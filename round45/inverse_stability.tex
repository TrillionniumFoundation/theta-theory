\section{Effective Jacobi inversion with a complete condition calculation}
\label{sec:quantitative-jacobi}
Throughout this section, vectors have the maximum norm unless an explicit
\(\ell^1\) norm is written; matrix infinity norms are induced row-sum norms.
Set
\[
 B_0=\max\{1,b_++2a_+\},\quad \Lambda=1+c_++B_0,\quad
 Q=8\max\{2,\Lambda,T,T^{-1},a_-^{-1}\}.
\]
For a depth \(J\ge0\), write \(j=J+1\) and define
\begin{equation}
 R_J=4J+8,\qquad L_J=Q^{64(J+1)^2}.
 \label{eq:effective-LJ}
\end{equation}
All constants below are arithmetic expressions in the declared coefficient
box and probe horizon. In particular, none is an infimum of an inverse map.

\subsection{Response derivatives, moments, and finite Gram systems}
Put \(\mu_k=\langle e_0,J_\beta^ke_0\rangle\) and
\(v_r=h_\beta^{(r+2)}(0)\). Then \(\mu_0=v_0=1\) and \(c=-v_1\).
The large-frequency resolvent expansion gives the convergent identity
\begin{equation}
 \sum_{r\ge0}v_rx^r
 =\langle e_0,(I+cx+x^2J_\beta)^{-1}e_0\rangle
 =\sum_{k\ge0}(-1)^k\mu_kx^{2k}(1+cx)^{-k-1}.
 \label{eq:response-moment-generating-identity}
\end{equation}
Indeed, factor out \(1+cx\), expand the remaining bounded-operator
geometric series near zero, and apply the boundary vector. Comparing the
coefficient of \(x^{2m}\) yields
\begin{equation}
 \mu_m=(-1)^m\left[v_{2m}-\sum_{k=0}^{m-1}(-1)^k
 \binom{2m-k}{2m-2k}c^{2m-2k}\mu_k\right].
 \label{eq:explicit-moment-recursion}
\end{equation}
This also proves exact response--moment triangularity; no analytic
continuation is used in the finite calculation.

\begin{lemma}[Moment-map condition bound]
\label{lem:response-moment-triangularity}
For \(M\ge1\), two admissible parameter values satisfy
\begin{equation}
 \max\left\{|c-\widetilde c|,\max_{k\le M}|\mu_k-\widetilde\mu_k|\right\}
 \le Q^{4(M+1)^2}\max_{0\le r\le2M}|v_r-\widetilde v_r|.
 \label{eq:quantified-moment-map}
\end{equation}
For \(M=0\), include \(v_1\) in the input maximum to recover damping.
\end{lemma}
\begin{proof}
Let \(S=Q/8\ge2\), and denote the input difference by \(e\).
Set \(P_0=0\) and, for \(m\ge1\),
\[
 P_m=1+\sum_{k<m}\binom{2m-k}{2m-2k}
 \{S^{2m-2k}P_k+(2m-2k)S^{2m-k-1}\}.
\]
The identity \(|x^r-y^r|\le rS^{r-1}|x-y|\), the bound
\(|\mu_k|\le B_0^k\le S^k\), and \eqref{eq:explicit-moment-recursion}
prove \(|\mu_m-\widetilde\mu_m|\le P_me\) by induction.
With \(U_m=\max(1,P_0,\ldots,P_m)\),
\[
 U_m\le[1+m(1+2m)4^mS^{2m}]U_{m-1}\le Q^{4m}U_{m-1}.
\]
For the last inequality, \(m(1+2m)\le16^m\) (induction, starting at
\(m=1\)) and \(1+64^mS^{2m}\le2Q^{2m}\le Q^{4m}\).
Thus \(U_M\le Q^{2M(M+1)}\le Q^{4(M+1)^2}\). Damping costs only the
factor one because \(c=-v_1\). This specifies every operation and norm.
\end{proof}

Let \(H_j=(\mu_{r+s})_{0\le r,s\le j}\), \(\Delta_j=\det H_j\),
and \(\Delta_{-1}=1\). For \(j\ge1\) solve
\begin{equation}
 H_{j-1}\alpha^{(j)}=-(\mu_j,\ldots,\mu_{2j-1})^T,
 \qquad p_j(x)=x^j+\sum_{r<j}\alpha_r^{(j)}x^r,
 \quad p_0=1.
 \label{eq:gram-linear-system}
\end{equation}
Write \(\rho_j=p_j^TH_jp_j\) and
\(t_j=\sum_{r,s\le j}p_{j,r}p_{j,s}\mu_{r+s+1}\). Orthogonality gives
\begin{equation}
 b_j=t_j/\rho_j,\quad a_j^2=\rho_{j+1}/\rho_j,\quad
 \rho_j=\prod_{r<j}a_r^2,\quad
 \Delta_j=\prod_{r<j}a_r^{2(j-r)}.
 \label{eq:gram-recover-b}
\end{equation}
Here \(p_j\) is the monic orthogonal polynomial of the boundary spectral
measure. The three-term recurrence proves
\(p_j(x)=\det(xI-J_{\beta,[0,j-1]})\); consequently its roots lie in
\([0,B_0]\), and its coefficient vector obeys
\begin{equation}
 \|p_j\|_1\le (1+B_0)^j=:S_j.
 \label{eq:monic-coefficient-bound}
\end{equation}
The recurrence also proves the norm and determinant products in
\eqref{eq:gram-recover-b}: starting with \(\rho_0=1\), orthogonality
implies \(\rho_{j+1}=a_j^2\rho_j\), and the unitriangular change from
monomials to monic polynomials gives \(\det H_j=\prod_{r\le j}\rho_r\).

\begin{lemma}[Finite-map condition ledger]
\label{prop:effective-gram-reconstruction}
Define, for \(j\ge1\),
\begin{align*}
 A_j&=j!B_0^{2j(j-1)}a_-^{-j(j-1)},&
 V_j&=jA_j(1+jS_j),\\
 R_j&=S_j^2+2B_0^{2j}S_jV_j,&
 T_j&=S_j^2+2B_0^{2j+1}S_jV_j,\qquad l_j=a_-^{2j}.
\end{align*}
Set \(S_0=l_0=T_0=1\), \(V_0=R_0=0\). For two admissible moment
vectors at maximum distance \(e\), the following are valid:
\begin{align}
 \|H_{j-1}^{-1}\|_\infty&\le A_j,&
 \|p_j-\widetilde p_j\|_1&\le V_je,\notag\\
 |\rho_j-\widetilde\rho_j|&\le R_je,&
 |t_j-\widetilde t_j|&\le T_je,\notag\\
 |b_j-\widetilde b_j|&\le (T_j+b_+R_j)e/l_j,&
 |a_j-\widetilde a_j|&\le
 (R_{j+1}+a_+^2R_j)e/(2a_-l_j).
 \label{eq:finite-map-ledger}
\end{align}
Only moments through \(2J+2\) are required through depth \(J\), and
\begin{equation}
 \max_{r\le J}(|a_r-\widetilde a_r|\vee|b_r-\widetilde b_r|)
 \le Q^{17(J+1)^2}\max_{k\le2J+2}|\mu_k-\widetilde\mu_k|.
 \label{eq:effective-gram-lipschitz}
\end{equation}
\end{lemma}
\begin{proof}
Each cofactor of the \(j\)-by-\(j\) matrix \(H_{j-1}\) is bounded by
\((j-1)!B_0^{2j(j-1)}\), using the determinant expansion and the entry
bound \(B_0^{2j}\). Its determinant is at least
\(a_-^{j(j-1)}\). Summing \(j\) cofactors in each inverse row proves
the first bound. Subtract the two linear systems, using the first inverse
and the second solution:
\[
 \|\alpha^{(j)}-\widetilde\alpha^{(j)}\|_\infty
 \le A_j(e+je\|\widetilde\alpha^{(j)}\|_\infty)
 \le A_j(1+jS_j)e.
\]
Multiplication by \(j\) gives the stated coefficient \(\ell^1\) bound.
For quadratic forms, the moment-matrix difference contributes at most
\(S_j^2e\); changing the two polynomial factors contributes at most
\(2B_0^{2j}S_jV_je\), or \(2B_0^{2j+1}S_jV_je\) for the shifted
matrix. These are exactly \(R_j,T_j\). Since \(\rho_j\ge l_j\) and
\(t_j=b_j\rho_j\), subtracting the ratios gives the bound for \(b_j\).
Subtract \(\rho_{j+1}/\rho_j\), use \(a_j^2\le a_+^2\), and divide by
\(a_j+\widetilde a_j\ge2a_-\) to obtain the last bound.

For clarity, the all-depth exponent propagation is displayed separately.
For every integer \(j\ge1\), \(j!\le2^{j^2}\), \(j\le Q^j\),
\(B_0,a_+,a_-^{-1}\le Q\), and \(1+B_0\le Q\). Therefore
\[
 \begin{array}{c|ccccc}
 \text{quantity}&S_j&A_j&V_j&R_j&T_j\\ \hline
 \text{upper bound}&Q^j&Q^{4j^2}&Q^{8j^2}&Q^{13j^2}&Q^{14j^2}
 \end{array}
\]
Indeed the exponents before simplification for \(V_j,R_j,T_j\) are,
respectively, \(4j^2+3j+1\), \(8j^2+3j+2\), and
\(8j^2+3j+3\); each is bounded by the corresponding table entry for
\(j\ge1\). The two ratio constants in \eqref{eq:finite-map-ledger}
are at most \(Q^{17j^2}\) and \(Q^{15(j+1)^2}\). At \(j=0\), the first
is one and the second is at most \(Q^{14}\), by the stated conventions.
Taking the maximum through \(J\) gives \eqref{eq:effective-gram-lipschitz}.
No constants are multiplied over unnecessary reconstruction levels.
\end{proof}

\begin{theorem}[All-depth effective response-jet inverse]
\label{lem:effective-response-jet}
For all \(J\ge0\) and \(\beta,\gamma\in\mathfrak B\),
\begin{equation}
 d_J(\beta,\gamma)\le L_J\max_{0\le r\le R_J}
 |h_\beta^{(r)}(0)-h_\gamma^{(r)}(0)|.
 \label{eq:effective-jet-inverse}
\end{equation}
\end{theorem}
\begin{proof}
Take \(M=2J+2\) in the moment lemma and compose with the Gram lemma.
Derivatives through \(4J+6\) suffice, hence those through \(R_J\) suffice.
For \(j=J+1\ge1\), the total exponent is bounded by
\[
 4(2j+1)^2+17j^2\le36j^2+17j^2=53j^2\le64j^2.
\]
This calculation includes every shallow depth, without an exceptional-case
adjustment or a change of the base \(Q\).
\end{proof}

\subsection{Finite Taylor grids and the sampling law}
For \(R\ge1\), set
\[
 V_R=(k^r/r!)_{0\le k,r\le R},\quad C_R=\|V_R^{-1}\|_\infty,
 \quad M_R=2e^{\Lambda T}\Lambda^R,\quad
 H_R=\frac{C_RM_RR^{R+1}}{(R+1)!}.
\]
The rational number \(C_R\) satisfies
\begin{equation}
 C_R\le2^{R+1}(R+1)!.
 \label{eq:vandermonde-inverse-bound}
\end{equation}
To see this, the coefficient sum of the Lagrange numerator
\(\prod_{l\ne k}(x-l)\) is at most \((R+1)!/(k+1)\).
The denominator has absolute value \(k!(R-k)!\); multiply coefficients by
\(r!\le R!\) and sum over \(k\), using
\(\sum_k1/(k!(R-k)!)=2^R/R!\).

\begin{lemma}[Finite-grid derivative certificate]
\label{lem:finite-grid-certificate}
For \(g=h_\beta-h_\gamma\) and \(0<\varepsilon\le\min(1,T/R)\),
\begin{equation}
 \max_{r\le R}|g^{(r)}(0)|\le
 C_R\varepsilon^{-R}\max_{0\le k\le R}|g(k\varepsilon)|
 +H_R\varepsilon.
 \label{eq:finite-grid-derivative}
\end{equation}
The value \(g(0)=0\) is deterministic; precisely \(R\) positive-time
values, not \(R+1\) measured cells, are required.
\end{lemma}
\begin{proof}
Since \(h^{(r)}(t)=\ell A^{r-1}e^{tA}B\) for \(r\ge1\),
\(\sup_{t\le T}|g^{(R+1)}(t)|\le M_R\).
Taylor expansion at the grid gives \(V_R(\varepsilon^rg^{(r)}(0))_r\)
plus a remainder of maximum norm at most
\(M_R(R\varepsilon)^{R+1}/(R+1)!\). Invert the matrix and use
\(\varepsilon^{-r}\le\varepsilon^{-R}\).
\end{proof}

Use labeled diagnostic cells, allowing distinct labels at the same time.
Assign mass \(2^{-m-1}\), \(m\ge1\), to a fixed dense enumeration of
rational points of \((0,T]\), and define additional cells
\begin{equation}
\begin{aligned}
 q&=(J,s,k),\quad J\ge0,\ s\ge1,\ 1\le k\le R_J,\\
 \varepsilon_{J,s}&=\zeta_{R_J}2^{-s},\quad \zeta_R=\min\{1,T/(2R)\},\\
 t_q&=k\varepsilon_{J,s},\qquad \omega_q=2^{-J-s-2}/R_J.
\end{aligned}
 \label{eq:multiscale-design}
\end{equation}
Each family has total mass one half. Duplicate times are summed in the
response metric; there is no zero-time sampling cell.
For \(0<\delta\le1\), let \(s_J(\delta)\) be the least positive integer
with \(H_{R_J}\varepsilon_{J,s}\le\delta/(2L_J)\). Define
\begin{align}
 \underline\kappa_J(\delta)&=
 \frac{2^{-J-s_J(\delta)-2}}{R_J}
 \left(\frac{\delta\varepsilon_{J,s_J(\delta)}^{R_J}}
 {2C_{R_J}L_J}\right)^2,\label{eq:effective-kappa}\\
 Z_J&=\max\{2/\zeta_{R_J},4L_JH_{R_J}\},\qquad
 K_J=\frac{2^{-J-4}}{R_J\zeta_{R_J}C_{R_J}^2L_J^2Z_J^{2R_J+1}}.
 \label{eq:explicit-kappa-prefactor}
\end{align}

\begin{theorem}[Effective depth-dependent inverse stability]
\label{thm:effective-jacobi-stability}
For \(d_J(\beta,\gamma)\ge\delta\),
\begin{equation}
 D(\beta,\gamma)\ge\underline\kappa_J(\delta)
 \ge K_J\delta^{2R_J+3}
 \ge e^{-A_0(J+1)^3}\delta^{20(J+1)},
 \label{eq:kappa-asymptotic-lower}
\end{equation}
where the explicit box constant is
\begin{equation}
 A_0=5000\{1+\Lambda T+|\log T|+\log Q\}.
 \label{eq:explicit-A0}
\end{equation}
\end{theorem}
\begin{proof}
The jet inverse and Taylor certificate force one positive grid value to
have magnitude at least
\(\delta\varepsilon_{J,s}^{R_J}/(2C_{R_J}L_J)\).
Its weighted square is the first bound. Minimality of \(s\) gives
\(\varepsilon_{J,s}\ge\delta/Z_J\): this is immediate when \(s=1\),
and otherwise \(2H_{R_J}\varepsilon_{J,s}>\delta/(2L_J)\).
Substitution of \(2^{-s}=\varepsilon_{J,s}/\zeta_{R_J}\) proves the
\(K_J\) bound.

Here is the full logarithmic audit. Put \(j=J+1\) and
\(F=1+\Lambda T+|\log T|+\log Q\). Since \(R_J\le8j\),
\(R_J+1\le9j\), and \(\log x\le x\),
\[
 \log C_{R_J}\le90j^2F,\quad
 \log\zeta_{R_J}^{-1}\le17jF,\quad
 \log H_{R_J}\le171j^2F,\quad
 \log Z_J\le240j^2F.
\]
The third inequality follows by dropping the negative factorial term and
adding \(90j^2F+F+8jF+72j^2F\); the fourth follows by adding
\(\log4+64j^2\log Q\) to the third and comparing with
\(\log2+17jF\). Finally \(\log\zeta_{R_J}\le0\), so
\[
 -\log K_J\le
 12jF+308j^2F+(16j+1)240j^2F\le4400j^3F<A_0j^3.
\]
Also \(2R_J+3=8J+19\le20j\). This proves the last bound for
\(0<\delta\le1\), with every depth and every constant accounted for.
\end{proof}
