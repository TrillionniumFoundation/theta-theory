\section{Five physical coefficients from one boundary coordinate}
\label{sec:lattice}
Let \(\mathcal H=\ell^2(\N_0;\R)\), \(\Hh=\mathcal H\oplus\mathcal H\),
\(Pq=\langle q,e_0\rangle e_0\), and \(Q_0=I-P\). The half-line
Laplacian is \((Lq)_0=q_0-q_1\) and
\((Lq)_j=2q_j-q_{j-1}-q_{j+1}\) for \(j\ge1\); it is nonnegative,
self-adjoint, and bounded by four. Let all endpoints of the fixed box
\[
 \vartheta=(c,k,\epsilon,c_b,k_b)\in\Theta
 =[c_-,c_+]\times[k_-,k_+]\times[\epsilon_-,\epsilon_+]
 \times[c_{b,-},c_{b,+}]\times[k_{b,-},k_{b,+}]
\]
be finite and strictly positive, with nonempty interior. Set
\[
 C_\vartheta=cP+c_bQ_0,\quad
 K_\vartheta=kP+k_bQ_0+\epsilon L,\quad
 A_\vartheta=\begin{pmatrix}0&I\\-K_\vartheta&-C_\vartheta\end{pmatrix},
 \quad B=\binom0{e_0},\quad \ell(q,v)=q_0.
\]
Thus \(\dot q=v\), \(\dot v=-K_\vartheta q-C_\vartheta v+e_0u(t)\).

\begin{lemma}[Uniform physical stability]
\label{lem:five-energy}
There are box constants \(M\ge1\), \(\lambda>0\) such that
\(\|e^{tA_\vartheta}\|\le Me^{-\lambda t}\) for all \(t\ge0\).
Every fixed-order parameter derivative is bounded by
\(M_r e^{-\lambda_rt}\), for \(M_r<\infty\), \(\lambda_r>0\).
Every fixed-order derivative of a uniformly bounded-input trajectory is
uniformly bounded, independently of its switching history.
\end{lemma}
\begin{proof}
Let \(c_*\) and \(k_*\) be the minimum damping and pinning endpoints,
and \(c^*\), \(k^*\) upper bounds on \(C_\vartheta,K_\vartheta\).
Take \(d=\min(c_*/2,\sqrt{k_*}/2)\) and use the energy
\[
 E_\vartheta(q,v)=\tfrac12\|v\|^2+\tfrac12\langle K_\vartheta q,q\rangle
 +d\langle q,v\rangle+\tfrac d2\langle C_\vartheta q,q\rangle.
\]
Young's inequality makes this uniformly equivalent to \(\|(q,v)\|^2\).
Differentiation along the unforced motion gives
\[
 E_\vartheta'=-\langle(C_\vartheta-dI)v,v\rangle
 -d\langle K_\vartheta q,q\rangle
 \le-\min(c_*-d,dk_*)\|(q,v)\|^2.
\]
This proves common exponential stability. The generator is affine in the
five parameters with bounded derivatives. Ordered Duhamel integrals for
order \(r\) are bounded by \(C_rt^re^{-\lambda t}\); decreasing the decay
rate absorbs the polynomial. Integrating these bounds against a bounded
input proves the final assertion.
\end{proof}

\subsection{The response jet and six-duration embedding}
Write
\(h_\vartheta(t)=\ell\int_0^te^{sA_\vartheta}Bds
=\sum_{j\ge0}r_jt^{j+1}/(j+1)!\), and put \(s=k+\epsilon\).
Successive application of the banded generator gives
\begin{equation}
\begin{aligned}
 r_0&=0,&r_1&=1,&r_2&=-c,\\
 r_3&=c^2-s,&r_4&=c(-c^2+2s),\\
 r_5&=c^4-3c^2s+s^2+\epsilon^2,\\
 r_6&=-c^5+4c^3s-2c\epsilon^2-3cs^2-c_b\epsilon^2,\\
 r_7&=c^6-5c^4s+3c^2\epsilon^2+6c^2s^2+2cc_b\epsilon^2\\ &\quad
       +c_b^2\epsilon^2-2\epsilon^3-2\epsilon^2s-s^3-\epsilon^2k_b.
\end{aligned}
\label{eq:eight-jets}
\end{equation}
The map \(\mathcal J(\vartheta)=(r_2,r_3,r_5,r_6,r_7)\) has the
following triangular inverse, for \(x=(x_2,x_3,x_5,x_6,x_7)\):
\begin{align}
 c&=-x_2,\quad s=c^2-x_3,\quad
 \epsilon=(x_5-c^4+3c^2s-s^2)^{1/2},\quad k=s-\epsilon,
 \label{eq:jet-inverse-one}\\
 c_b&=(-c^5+4c^3s-2c\epsilon^2-3cs^2-x_6)/\epsilon^2,
 \label{eq:jet-inverse-two}\\
 k_b&=\epsilon^{-2}[c^6-5c^4s+3c^2\epsilon^2+6c^2s^2
 +2cc_b\epsilon^2+c_b^2\epsilon^2\notag\\
 &\hspace{28mm}-2\epsilon^3-2\epsilon^2s-s^3-x_7].
 \label{eq:jet-inverse-three}
\end{align}
The radicand is at least \(\epsilon_-^2\) on the image. These formulas
are therefore smooth on a neighbourhood of the compact image. A smooth
cutoff supported in that neighbourhood extends the inverse to a globally
bounded smooth map \(G:\R^5\to\R^5\).

At \(t_j=j\tau\), \(1\le j\le6\), remove the known term \(t_j^2/2\)
and let \(y_j=h_\vartheta(t_j)-t_j^2/2\). Define
\[
 V_{j,\ell-1}=j^{\ell+1}/(\ell+1)!,\quad 2\le\ell\le7,
 \qquad D_\tau=\operatorname{diag}(\tau^3,\ldots,\tau^8),
 \qquad \widehat R_\tau=D_\tau^{-1}V^{-1}(y_j)_{j=1}^6.
\]
Dividing each row of \(V\) by \(j^3\) reduces its determinant to a
nonzero Vandermonde determinant, with nonzero column factors. The omitted
Taylor series starts at order nine, uniformly also for one parameter
derivative. Multiplication by \(D_\tau^{-1}\) amplifies at most by
\(\tau^{-8}\). Hence
\begin{equation}
 \|\widehat R_\tau-(r_2,\ldots,r_7)\|_{C^1(\Theta)}\le C\tau,
 \qquad 0<\tau\le1.
 \label{eq:recovered-jets}
\end{equation}

\begin{theorem}[Global six-duration mechanical embedding]
\label{thm:jet-embedding}
For sufficiently small fixed \(\tau>0\), the response map
\(H_\tau(\vartheta)=(h_\vartheta(j\tau))_{j=1}^6\) satisfies
\begin{equation}
 |H_\tau(\vartheta)-H_\tau(\vartheta')|\ge b_\tau|\vartheta-\vartheta'|,
 \qquad |DH_\tau(\vartheta)v|\ge b_\tau|v|,
 \qquad b_\tau>0.
 \label{eq:six-bi-lipschitz}
\end{equation}
\end{theorem}
\begin{proof}
Let \(P_5\) select jet coordinates \((r_2,r_3,r_5,r_6,r_7)\).
The smooth reconstructed map \(F_\tau=G(P_5\widehat R_\tau)\) obeys
\(\|F_\tau-\operatorname{Id}\|_{C^1}\le C\tau\).
Choose \(\tau\) so this norm is at most one half. Integration along a
segment in the convex parameter box gives
\(|F_\tau(\vartheta)-F_\tau(\vartheta')|\ge|
\vartheta-\vartheta'|/2\), and the same lower bound holds for tangent
vectors. The map taking six responses to \(F_\tau\) has a finite global
Lipschitz constant, because \(G\) does. Divide by twice that constant.
\end{proof}

\subsection{Finite-prefix calibration and exact chronological contrast}
\label{sec:balanced-design}
Fix a pulse amplitude \(a>0\), an exploitation-force bound \(U<\infty\),
exploitation durations in \([\tau_{\min},\tau_{\max}]\) with
\(0<\tau_{\min}\le\tau_{\max}<\infty\), and constants
\(\rho\in(0,1]\), \(C_{\rm cal}\ge0\). Calibration membership is
predictable from \(\mathcal F_{i-1}\), and
\begin{equation}
 N_{\rm cal}(m)\ge\rho m-C_{\rm cal},\qquad 1\le m\le n.
 \label{eq:prefix-calibration}
\end{equation}
These constants are common to all horizons and all admissible policies.
Calibration slots in their own chronological order are grouped into windows
of six. Before drawing the fresh sign, choose the duration predictably from
those not yet used in the current window; thus each complete window uses
\(\tau,2\tau,\ldots,6\tau\) once. Then apply \(aS_i\) and finally observe
with independent Gaussian noise. In symbols,
\[
 \mathcal F_{i-1}\longrightarrow s_i\longrightarrow S_i
 \longrightarrow\xi_i\longrightarrow Y_i.
\]
Exploitation may be arbitrary bounded history-dependent control whose full
current action is selected before the readout. The class is nonempty even
when \(C_{\rm cal}=0\): a deterministic schedule with
\(N_{\rm cal}(m)=\lceil\rho m\rceil\) satisfies the condition.

\begin{lemma}[Complete windows in every prefix]
\label{lem:prefix-window-count}
Pathwise,
\begin{equation}
 W_n=\lfloor N_{\rm cal}(n)/6\rfloor
 \ge\rho n/6-C_{\rm win},\qquad C_{\rm win}=C_{\rm cal}/6+1.
 \label{eq:complete-window-count}
\end{equation}
\end{lemma}
\begin{proof}
Apply \(\lfloor x\rfloor\ge x-1\) and \eqref{eq:prefix-calibration}.
Only the last incomplete window is lost.
\end{proof}
For zero-state means under the entire realized action history, at a
calibration slot
\begin{equation}
 m_i^\vartheta-m_i^{\vartheta_0}
 =D_{i,s_i}(\vartheta,\vartheta_0)
 +aS_i\{h_\vartheta(s_i)-h_{\vartheta_0}(s_i)\},
 \label{eq:sign-decomposition}
\end{equation}
where \(D_{i,s_i}\) is \(\mathcal F_{i-1}\)-measurable. Conditional
expectation of its square is
\begin{equation}
 D_{i,s_i}^2+a^2\{h_\vartheta(s_i)-h_{\vartheta_0}(s_i)\}^2.
 \label{eq:sign-square}
\end{equation}
Summing over a complete window gives at least
\(a^2b_\tau^2|\vartheta-\vartheta_0|^2\), regardless of the adaptive
order of its durations.

\begin{proposition}[Full-index empirical contrast]
\label{prop:balanced-contrast}
Let \(\bar c_\tau=\rho a^2b_\tau^2/6\). For every fixed
\(0<c_*<\bar c_\tau\), common constants \(C_1,C_2>0\) satisfy
\begin{equation}
 P_E\left\{\inf_{\vartheta\ne\vartheta_0}
 \frac{n^{-1}\sum_i(m_i^\vartheta-m_i^{\vartheta_0})^2}
 {|\vartheta-\vartheta_0|^2}<c_*\right\}\le C_1e^{-C_2n}.
 \label{eq:full-empirical-contrast}
\end{equation}
On the same good event,
\(n^{-1}\sum_i(v^TDm_i^{\vartheta_0})^2\ge c_*|v|^2\) for all \(v\).
\end{proposition}
\begin{proof}
Use the compact radial index set
\[
 \mathscr K=\{(\theta,r,v):\theta\in\Theta,\ v\in S^4,
 r\ge0,\ \theta+rv\in\Theta\},\qquad
 q_i(\theta,r,v)=\int_0^1v^TDm_i^{\theta+srv}ds.
\]
For \(r>0\), this is the divided mean difference; at zero it is the
tangent derivative. Lemma~\ref{lem:five-energy} bounds \(q_i\) and two
parameter derivatives uniformly, hence \(q_i^2\) is bounded and
Lipschitz on this fixed compact set, independently of the policy and
horizon. Let \(X_i=q_i^2-E[q_i^2\mid\mathcal F_{i-1}]\).
The window identity and count give, for every index,
\[
 \sum_iE[q_i^2\mid\mathcal F_{i-1}]
 \ge\bar c_\tau n-a^2b_\tau^2 C_{\rm win}.
\]
Each \(X_i\) is a bounded martingale difference with a common Lipschitz
constant \(L_X\). A deterministic finite \(\delta\)-net and Azuma's
inequality give, except on probability \(C'e^{-c'n}\),
\(\inf_{\mathscr K}n^{-1}\sum_iX_i\ge-\eta-L_X\delta\).
Choose \(\eta,\delta\) with
\(\eta+L_X\delta<(\bar c_\tau-c_*)/2\).
For all sufficiently large \(n\), the displayed predictable lower bound
then proves the assertion. Enlarge \(C_1\) for the remaining finite
number of horizons. Taking \(r=0\) gives the tangent assertion.
\end{proof}

\subsection{The mechanical posterior and evidence}
Define
\[
 x_i^\vartheta=\int_0^{t_i}e^{(t_i-s)A_\vartheta}Bu(s)ds,
 \quad m_i^\vartheta=\ell x_i^\vartheta,
 \quad b_i^\vartheta(z)=\ell e^{t_iA_\vartheta}z.
\]
Every block has length at least
\(\underline\tau=\min(\tau,\tau_{\min})>0\), so
\(t_i\ge i\underline\tau\). Every fixed derivative of \(b_i\) is
bounded by a common summable exponential envelope. Put
\[
 I_n=(n\sigma^2)^{-1}\sum_i Dm_i^{\vartheta_0}(Dm_i^{\vartheta_0})^T,
 \qquad \Delta_n=(\sigma^2\sqrt n)^{-1}\sum_i Dm_i^{\vartheta_0}\xi_i.
\]

\begin{theorem}[Five-parameter mechanical quasi-Bernstein--von Mises]
\label{thm:lattice-bvm}
Let the true parameter range over a fixed compact interior subset of
\(\Theta\), let \(\|z_0\|\le R\), and let \(\nu_n\) be arbitrary
working probabilities on the same Hilbert ball. Assume the parameter-prior
conditions of Assumption~\ref{ass:triangular}. Uniformly over these choices
and all prefix-balanced policies, including horizon-dependent ones,
\begin{equation}
 \TV\left(\mathcal L\{\sqrt n(\vartheta-\vartheta_0)\mid\mathcal F_n\},
 N(I_n^{-1}\Delta_n,I_n^{-1})\right)=o_{\mathfrak E}(1),
 \label{eq:five-bvm}
\end{equation}
with the singular-event convention of Theorem~\ref{thm:abstract-bvm}, and
\begin{equation}
 \frac{\mathcal Z_n^{\nu_n}}{L_n^{\nu_n}(\vartheta_0)}
 =\frac{(2\pi)^{5/2}\pi(\vartheta_0)}{n^{5/2}\sqrt{\det I_n}}
 e^{\Delta_n^TI_n^{-1}\Delta_n/2}[1+o_{\mathfrak E}(1)].
 \label{eq:five-evidence}
\end{equation}
The pseudo-true parameter has displacement \(O(n^{-1})\), including when
\(z_0\notin\supp\nu_n\). No limit of \(I_n\) is assumed.
\end{theorem}
\begin{proof}
The energy lemma proves uniform response smoothness and transient
summability of every fixed order. Proposition~\ref{prop:balanced-contrast}
proves global contrast with exponential failure probability. Apply
Theorem~\ref{thm:abstract-bvm} and Proposition~\ref{prop:pseudotrue}.
\end{proof}
