\section{Complete recovery of an unknown infinite Jacobi bath}
\label{sec:infinite-jacobi}
Fix \(0<c_-<c_+<\infty\), \(0<a_-<a_+<\infty\), and
\(2a_+<b_-<b_+<\infty\). The compact metrizable parameter space is
\begin{equation}
 \mathfrak B=[c_-,c_+]\times\prod_{j\ge0}[a_-,a_+]
 \times\prod_{j\ge0}[b_-,b_+],\qquad \beta=(c,(a_j),(b_j)).
 \label{eq:jacobi-parameter}
\end{equation}
On \(\ell^2(\N_0)\), let
\[
 (J_\beta q)_0=b_0q_0-a_0q_1,\qquad
 (J_\beta q)_j=b_jq_j-a_{j-1}q_{j-1}-a_jq_{j+1},\quad j\ge1.
\]
The operator satisfies \((b_--2a_+)I\le J_\beta\le(b_++2a_+)I\).
Put
\[
 A_\beta=\begin{pmatrix}0&I\\-J_\beta&-cI\end{pmatrix},\qquad
 B=\binom0{e_0},\quad\ell(q,v)=q_0,\quad
 h_\beta(t)=\ell\int_0^te^{sA_\beta}Bds.
\]

\begin{lemma}[Product continuity and common stability]
\label{lem:product-continuity}
The map \(\beta\mapsto h_\beta|_{[0,T]}\) is continuous into
\(C([0,T])\). There are common constants \(C_*\ge1\), \(\lambda_*>0\)
with \(\sup_\beta\|e^{tA_\beta}\|\le C_*e^{-\lambda_*t}\).
\end{lemma}
\begin{proof}
Apply the energy proof of Lemma~\ref{lem:five-energy} with \(C=cI\)
and \(K=J_\beta\). Coordinatewise convergence gives convergence of
\(J_\beta q\) on finitely supported vectors. Uniform operator bounds and
density extend this to strong convergence on \(\ell^2\). Powers and
uniformly convergent exponential series then give strong semigroup
convergence uniformly on compact time intervals. Apply to \(B\), observe
with \(\ell\), and integrate.
\end{proof}

\subsection{Weyl reconstruction}
For the tail starting at \(j\), define
\(m_j(w)=\langle e_j,(w+J_\beta^{(j)})^{-1}e_j\rangle\), \(w>0\).
Laplace transformation and a Schur complement give
\begin{equation}
 \widehat h_\beta(z)=z^{-1}m_0(z^2+cz),\qquad
 c=-h_\beta'''(0),\qquad
 m_j(w)^{-1}=w+b_j-a_j^2m_{j+1}(w).
 \label{eq:jacobi-transfer}
\end{equation}
Consequently
\begin{align}
 b_j&=\lim_{w\to\infty}\{m_j(w)^{-1}-w\},\notag\\
 a_j^2&=\lim_{w\to\infty}w\{w+b_j-m_j(w)^{-1}\},\qquad
 m_{j+1}(w)=\{w+b_j-m_j(w)^{-1}\}/a_j^2.
 \label{eq:recover-tail}
\end{align}

\begin{theorem}[Complete Jacobi reconstruction]
\label{thm:jacobi-reconstruction}
The boundary response determines all of \(\beta\). The infinite-chain
Weyl function is nonrational, so no fixed finite Jacobi chain has the same
response.
\end{theorem}
\begin{proof}
Recover damping from the third derivative. The map \(z\mapsto z^2+cz\)
is a bijection of \((0,\infty)\), so the Laplace transform determines
\(m_0(w)\) there. Since \(m_{j+1}(w)=w^{-1}+O(w^{-2})\),
\eqref{eq:recover-tail} recovers \(b_j,a_j,m_{j+1}\) successively; take
the positive square root for \(a_j\).
The vector \(e_0\) is cyclic: the recurrence and \(a_j>0\) inductively
put every basis vector in the span of \(\{J_\beta^ke_0\}\).
A rational Stieltjes transform has a finite-support spectral measure and
therefore a finite-dimensional cyclic subspace, a contradiction.
The inverse spectral recursion is classical \cite{Teschl2000}; the finite
condition estimates used statistically are proved separately below.
\end{proof}

For a fixed dense duration law \(\omega\) on \((0,T]\), define
\begin{equation}
 D(\beta,\gamma)=\sum_m\omega_m
 [h_\beta(t_m)-h_\gamma(t_m)]^2,\qquad
 d_J(\beta,\gamma)=\max\{|c-\widetilde c|,
 \max_{j\le J}|a_j-\widetilde a_j|,\max_{j\le J}|b_j-\widetilde b_j|\}.
 \label{eq:response-metric}
\end{equation}
The square root of \(D\) induces the product topology: the response map
is a continuous injection from a compact space into a Hausdorff space.
Injectivity on \([0,T]\) follows from bounded-generator analyticity and
the reconstruction theorem. Hence, whenever the constraint is nonempty,
\begin{equation}
 \kappa_J(\delta)=\inf_{d_J(\beta,\gamma)\ge\delta}D(\beta,\gamma)>0.
 \label{eq:jacobi-separation-modulus}
\end{equation}
The effective lower bound in Theorem~\ref{thm:effective-jacobi-stability}
will replace this qualitative compactness argument in every rate assertion.

\section{The exact posterior under logarithmic washout}
\label{sec:jacobi-statistics}
At stage \(i\), a parameter-independent bounded feedback segment of
length at most \(T_J\) and force at most \(U_J\) is followed by zero force
for
\begin{equation}
 w_i=\frac{1+\epsilon_w}{\lambda_*}\log(i+1),\qquad \epsilon_w>0.
 \label{eq:logarithmic-washout}
\end{equation}
Then draw \(M_i\sim\omega\), draw an independent Rademacher \(S_i\),
apply \(aS_i\) for duration \(t_{M_i}\), and finally draw the independent
\(N(0,\sigma^2)\) readout. All constants are common across policies and
horizons. The initial state lies in the fixed radius-\(R\) Hilbert ball.
Writing \(U_*=\max(U_J,a)\), common stability gives
\begin{equation}
 Y_i=aS_ih_{\beta_0}(t_{M_i})+\eta_i^{\beta_0}(z_0)+\xi_i,
 \qquad \sup_{\beta,\|z\|\le R}|\eta_i^\beta(z)|
 \le r_i:=C_*^2(R+U_*/\lambda_*)(i+1)^{-1-\epsilon_w}.
 \label{eq:washed-observation}
\end{equation}
Indeed the state preceding washout has norm at most
\(C_*(R+U_*/\lambda_*)\), and propagation through washout and the probe
multiplies this by at most \(C_*e^{-\lambda_*w_i}\).

\begin{lemma}[All-order washout differentiation at the original decay rate]
\label{lem:washout-derivatives}
For each fixed finite-cylinder multi-index \(\alpha\),
\begin{equation}
 \sup_{\beta,\|z\|\le R}|\partial^\alpha\eta_i^\beta(z)|
 \le C_\alpha(1+\log(i+1))^{|\alpha|}(i+1)^{-1-\epsilon_w}.
 \label{eq:polynomial-washout-envelope}
\end{equation}
For every \(0<\bar\epsilon<\epsilon_w\), this is bounded by
\(C_{\alpha,\bar\epsilon}(i+1)^{-1-\bar\epsilon}\), hence is summable.
\end{lemma}
\begin{proof}
The generator is affine in cylinder coordinates. An order-\(r\)
Duhamel derivative of \(e^{wA_\beta}\) is a sum of at most \(r!\)
ordered integrals. Each integrand is bounded by
\(C_*^{r+1}D^re^{-\lambda_*w}\), because the durations of its \(r+1\)
semigroup factors add to \(w\); \(D\) bounds the relevant generator
derivatives. The simplex volume is \(w^r/r!\). Thus the derivative is
bounded by \(C_r w^re^{-\lambda_*w}\), retaining the \emph{same}
\(\lambda_*\). Derivatives of the pre-washout state are uniformly
bounded by integration of \(C_r(1+t)^re^{-\lambda_*t}\) against the
bounded realized input; all actions are held fixed. Derivatives during
the bounded probe interval are bounded as well. Leibniz's rule therefore
gives \(C_\alpha(1+w_i)^{|\alpha|}e^{-\lambda_*w_i}\).
Substitute \eqref{eq:logarithmic-washout}. Finally every fixed power of
\(1+\log x\) is bounded by a constant times
\(x^{\epsilon_w-\bar\epsilon}\) for \(x\ge1\).
\end{proof}

Let \(L_n^{\nu_n}\) be the exact integrated likelihood of the full
realized action record, with any normalized working initial-state prior
\(\nu_n\) on the same Hilbert ball. Let \(\Pi\) be a fixed Borel prior
with full support on \(\mathfrak B\). It defines the exact working
posterior \(\Pi_n^{\nu_n}\) by normalization of \(L_n^{\nu_n}d\Pi\).

\begin{theorem}[Full good posterior LDP]
\label{thm:jacobi-rate}
For every admissible policy sequence and every chosen triangular coupling
as specified in Section~\ref{sec:finite-sample-geometry}, on one event of
probability one, for every Borel set \(A\subset\mathfrak B\),
\begin{equation}
 -\inf_{A^\circ}I_{\beta_0}\le
 \liminf_n n^{-1}\log\Pi_n^{\nu_n}(A)\le
 \limsup_n n^{-1}\log\Pi_n^{\nu_n}(A)
 \le-\inf_{\overline A}I_{\beta_0},\qquad
 I_{\beta_0}(\beta)=\frac{a^2}{2\sigma^2}D(\beta,\beta_0).
 \label{eq:jacobi-posterior-ldp}
\end{equation}
This includes any fixed infinite admissible policy. Finite-n probability
bounds are uniform over all policies with the declared common bounds.
\end{theorem}
\begin{proof}
Use \(f_i(\beta)=aS_i[h_\beta(t_{M_i})-h_{\beta_0}(t_{M_i})]\)
and \(d(\beta,\gamma)=a\|h_\beta-h_\gamma\|_\infty\) in
Proposition~\ref{prop:finite-n-field-concentration}. This is a compact
metric geometry, and \(|f_i|\le2aC_*/\lambda_*\).
Conditional squared means are exactly \(a^2D(\beta,\beta_0)\).
The summable residual in \eqref{eq:washed-observation} meets
Lemma~\ref{lem:finite-n-exact-correction}, including the candidate residual
from every past forcing segment. Hence
\[
 \sup_\beta|n^{-1}\log[L_n^{\nu_n}(\beta)/L_n^{\nu_n}(\beta_0)]
             +I_{\beta_0}(\beta)|\to0
\]
almost surely by the summable finite-n bounds. Only undifferentiated
residuals are needed here.
The posterior normalizer has exponent zero: its upper bound follows from
\(I\ge0\), and integration over the open set \(\{I<\eta\}\), of positive
prior mass, gives lower exponent at least \(-\eta\).
For an open set integrate a small neighbourhood of any of its points;
for a closed set use the uniform likelihood upper bound and compactness.
Subtract the normalizer exponent. This proves the bounds simultaneously
for all sets, as in Theorem~\ref{thm:adaptive-laplace-tracking}.
\end{proof}

\begin{corollary}[Strong consistency and effective cylinder exponent]
\label{thm:jacobi-consistency}
For every product neighbourhood \(U\) of \(\beta_0\),
\(\Pi_n^{\nu_n}(U^c)\to0\) almost surely. Under the multiscale design
of \eqref{eq:multiscale-design},
\[
 \limsup_n n^{-1}\log\Pi_n^{\nu_n}\{d_J(\beta,\beta_0)\ge\delta\}
 \le-\frac{a^2}{2\sigma^2}\underline\kappa_J(\delta).
\]
\end{corollary}
\begin{proof}
The continuous good rate is positive off its unique zero. Apply the closed
upper bound and, for cylinders, the effective inverse-stability theorem.
\end{proof}

The physical clock satisfies
\begin{equation}
 \mathsf T_n=\kappa_w n\log n+O(n),\quad
 \kappa_w=(1+\epsilon_w)/\lambda_*,\quad
 v(t)=\frac{t}{\kappa_w W(t/\kappa_w)},\quad
 v(\mathsf T_n)/n\to1,
 \label{eq:physical-time-complexity}
\end{equation}
where \(W\) is the principal Lambert function. This follows by summing
\(w_i\) and using \(\log((n+1)!)=n\log n+O(n)\); other stage durations
are uniformly bounded. Thus the same LDP holds at elapsed-time speed
\(v(\mathsf T_n)\).
