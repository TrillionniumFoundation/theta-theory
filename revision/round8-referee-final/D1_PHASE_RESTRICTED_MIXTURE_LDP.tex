\section{Round-eight reconstruction: phase-restricted microscopic laws and stratified mixture LDPs}
\label{sec:r8-d1}

Coexistence is handled by genuine phase-restricted finite-volume objects and an
exponential mixture theorem.  Lee--Yang zeros may pinch the coexistence set;
there is no common zero-free tube there.  The minimum of phase rates is proved
from the microscopic mixture and is not inferred from the conjugate of a
maximum pressure.

\subsection{Phase-restricted partition functions}

Let $j=1,\ldots,J$ label metastable boundary conditions or spectral phase
projectors.  Define finite positive phase laws $\mathbb P_{\varepsilon,j}$ and
restricted partition functions
\[
 Z_{\varepsilon,j}(\theta)
 =\mathbb E_{\varepsilon,j}e^{\mu_\varepsilon\theta\cdot X_\varepsilon}.
\]
On the interior $\mathcal U_j$ of phase $j$, assume the restricted transfer
operator has one isolated eigenvalue.  Its logarithm is analytic on local
complex tubes compactly contained in $\mathcal U_j$.

\begin{lemma}[Local phase atlas only]
\label{lem:r8-d1-atlas}
For every compact $K\Subset\mathcal U_j$, the functions
$\mu_\varepsilon^{-1}\log Z_{\varepsilon,j}$ converge analytically on a fixed
complex neighborhood of $K$.  No such neighborhood is asserted across a
coexistence point.  Zeros of the total partition function are allowed to
approach coexistence at order $\mu_\varepsilon^{-1}$.
\end{lemma}

\begin{proof}
The isolated phase eigenvalue and its Riesz projector persist on a compact
subdomain of one phase, giving a nonvanishing restricted leading term and
analytic perturbation.  At coexistence two restricted exponentials have equal
real part, so their sum can have nearby complex zeros; excluding that set from
the phase tube is both necessary and consistent.
\end{proof}

\subsection{Phase-wise finite-dimensional LDPs}

For a finite projection $\pi_m$, let $Q_{m,j}$ be the phase pressure and
$I_{m,j}=Q_{m,j}^*$ on the interior mean image.  Boundary points are recovered
inside the same microscopic phase by normal concentration followed by a
finite tangential tilt.

\begin{theorem}[Phase-wise face LDP]
\label{thm:r8-d1-phase}
Each restricted law $\mathbb P_{\varepsilon,j}\circ\pi_m^{-1}$ satisfies a full
good LDP with rate $I_{m,j}$ on its attainable compact support.  For an exposed
face $F$, the lower recovery is obtained by first choosing a large but finite
normal source, passing $\varepsilon\to0$, and then sending the normal source to
infinity while applying the phase-specific tangential coefficient theorem.
\end{theorem}

\begin{proof}
Interior points use the analytic phase saddle.  Compact support gives the
upper bound without an infinite source domain.  For a face, exponential
tightness in the normal coordinate makes the finite normal tilt concentrate in
a small neighborhood of $F$.  Conditional on that neighborhood, the
tangential restricted transfer operator remains inside phase $j$ and its
local-limit coefficient gives the lower bound.  A diagonal sequence in the
normal source completes the recovery.
\end{proof}

\subsection{The microscopic mixture theorem}

Assume the physical law has the decomposition
\[
 \mathbb P_\varepsilon
 =\sum_{j=1}^Jw_{\varepsilon,j}\mathbb P_{\varepsilon,j}
 +R_\varepsilon,
\]
where
\[
 \mu_\varepsilon^{-1}\log w_{\varepsilon,j}\to0,
 \qquad
 \|R_\varepsilon\|\le e^{-c\mu_\varepsilon}.
\]

\begin{theorem}[Coexistence mixture LDP]
\label{thm:r8-d1-mixture}
The physical finite-dimensional law satisfies a full LDP with rate
\[
 I_m(x)=\min_j I_{m,j}(x).
\]
Its pressure is $Q_m(\theta)=\max_jQ_{m,j}(\theta)$, but the rate identity is a
consequence of the microscopic mixture, not the generally false formula
$(\max_jQ_{m,j})^*=\min_jQ_{m,j}^*$.
\end{theorem}

\begin{proof}
For a closed set, sum the phase upper bounds; the largest exponential term is
governed by $\inf_x\min_jI_{m,j}(x)$.  For an open set and a point $x$, choose a
phase attaining the minimum and use its lower recovery from
Theorem~\ref{thm:r8-d1-phase}; the subexponential mixture weight does not change
the rate.  The signed remainder is exponentially negligible.  The pressure
maximum follows separately by the log-sum-exp identity.
\end{proof}

\subsection{Projective passage and topology upgrade}

Let $I_j=\sup_m I_{m,j}\circ\pi_m$ and $I=\min_jI_j$.

\begin{theorem}[Stratified projective LDP]
\label{thm:r8-d1-projective}
If every phase family is exponentially tight in the target weighted topology
and every finite-rate point has a phase-wise recovery sequence exponentially
concentrated in one compact rate sublevel, then the physical laws satisfy a
full target-topology LDP with rate $I$.
\end{theorem}

\begin{proof}
Apply the projective theorem phase by phase.  Compact rate sublevels make the
projective and target topologies agree on the recovery supports.  For a
target-open set, choose a projective neighborhood whose intersection with the
selected compact sublevel lies inside it.  The recovery law puts
superexponentially small mass outside that compact, so its projective lower
bound is also a target lower bound.  Combine phases with
Theorem~\ref{thm:r8-d1-mixture}.
\end{proof}

\subsection{Conditioning and exact local likelihoods}

\begin{theorem}[Phase-stratified thin-shell conditioning]
\label{thm:r8-d1-shell}
Suppose each active phase has the A2 or B1 exact finite-source shell coefficient
uniformly on its phase chart.  Conditioning the microscopic mixture on the
shell produces a new subexponential mixture of the conditioned phase laws.
The conditional LDP has the minimum of their constrained rates.
\end{theorem}

\begin{proof}
Apply the phase coefficient theorem to each numerator and denominator.  Phases
whose shell exponent is larger are exponentially suppressed; phases tying at
the minimum retain subexponential weights.  The conditioned law is therefore
again of the form required by Theorem~\ref{thm:r8-d1-mixture}, with each phase
rate restricted to the shell.
\end{proof}

\begin{theorem}[Exact finite-centered likelihood and contraction]
\label{thm:r8-d1-likelihood}
Within every active phase, the local likelihood ratio is centered at the exact
finite-volume mean and has the Gaussian quadratic form supplied by the phase
covariance.  Under any continuous contraction, phase rates contract first and
are then minimized over phases; this order agrees with direct contraction of
the microscopic mixture.
\end{theorem}

\begin{proof}
Taylor expand the restricted finite log partition function at its exact saddle;
the score has zero finite expectation and the Hessian is the exact covariance.
For contraction, push forward each phase law and apply the ordinary contraction
principle.  Pushforward commutes with the finite mixture, so
Theorem~\ref{thm:r8-d1-mixture} gives the minimum of the phase-wise contracted
rates.
\end{proof}
