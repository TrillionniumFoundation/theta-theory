\section{Round-eight reconstruction: regenerative compound smoothing and exact saddle coefficients}
\label{sec:r8-b1}

The continuous mark law of one particle is allowed to be singular.  Smoothing
is obtained from linearly many independent full-rank blocks in a typical
compound-Poisson sample; the complement of this regenerative event is
exponentially small.  The covariance is computed from the typical compound
law rather than from an exponentially rare fixed-particle sector.

\subsection{Admissible constraint marks}

Let
\[
 X(v)=\bigl(v,|v|^2/2,\chi_1(v),\ldots,\chi_d(v)\bigr)
 \in\mathbb R^{4+d}.
\]
We call the constraint family \emph{admissible} on a source chart $K$ if the
tilted second-moment matrices satisfy
\[
 \inf_{\lambda\in K}\inf_{|u|=1}
 \int (u\cdot X(v))^2e^{\lambda\cdot X(v)}\,d\nu(v)>0,
\]
and if some product patch $U\subset(\operatorname{supp}\nu)^{n_0}$ makes the
sum map
\[
 \mathsf S(v_1,\ldots,v_{n_0})=\sum_{j=1}^{n_0}X(v_j)
\]
a submersion.  These are explicit affine-span and transversality conditions;
they exclude redundant constraints such as $\chi_1\equiv0$.

\begin{lemma}[Finite submersion cover]
\label{lem:r8-b1-submersion}
For an admissible family there are finitely many patches $U_\alpha$ and smooth
cutoffs $\zeta_\alpha$ such that the restricted pushforwards
$\mathsf S_*(\zeta_\alpha\nu^{\otimes n_0})$ have compactly supported
$W^{k,1}$ densities, uniformly for every prescribed finite $k$ and every
source in $K$.
\end{lemma}

\begin{proof}
On a full-rank patch choose a nonvanishing $(4+d)$-minor and use the
corresponding coordinates as transverse variables.  Coarea gives a density;
differentiating under the fiber integral gives $W^{k,1}$ bounds after the
patch is shrunk away from the zero set of the minor.  Compactness of the source
chart and the support used by the saddle gives a finite cover.  General
velocity variations, not only radial variations in a coordinate plane, supply
the momentum directions.
\end{proof}

\subsection{Good-block regeneration}

Let $N_\mu$ be Poisson with mean $\mu a_\lambda$ and split the first
$n_0\lfloor N_\mu/n_0\rfloor$ marks into predetermined blocks.  A block is
\emph{good} if it lies in one of the submersion patches, with the cutoff label
chosen by an independent partition coin.  Its probability is at least
$p_*>0$, uniformly on $K$.

\begin{lemma}[Linearly many smooth blocks]
\label{lem:r8-b1-good-blocks}
There are $c,C>0$ such that
\[
 \mathbb P_\lambda\{M_\mu<c\mu\}\le Ce^{-c\mu},
\]
where $M_\mu$ is the number of good disjoint blocks.  Conditional on the
remaining marks and on $M_\mu=M$, the continuous sum law is a convolution of
$M$ uniformly $W^{k,1}$ densities with an arbitrary residual probability
measure.
\end{lemma}

\begin{proof}
Poisson concentration gives $N_\mu\ge c_0\mu$ outside an exponentially small
event.  Conditional on $N_\mu$, the indicators of the predetermined good
blocks are independent Bernoulli variables with success probability at least
$p_*$.  Chernoff's bound gives the first assertion.  Independence of disjoint
blocks gives the convolution decomposition; all singular marks are kept in the
residual factor rather than declared smooth.
\end{proof}

\begin{theorem}[Global characteristic-function estimate]
\label{thm:r8-b1-characteristic}
For every $k$ there are $C,c,R>0$ such that the compound characteristic
function satisfies
\[
 |\Phi_{\mu,\lambda}(\xi)|
 \le Ce^{-c\mu}+C_k(1+|\xi|)^{-k},
 \qquad |\xi|\ge R,
\]
uniformly for $\lambda\in K$.  Lattice coordinates are restricted to one
fundamental domain.  Empty and low-particle sectors are contained explicitly
in the term $Ce^{-c\mu}$.
\end{theorem}

\begin{proof}
On the event in Lemma~\ref{lem:r8-b1-good-blocks}, Fourier transformation gives
\[
 |\Phi(\xi\mid M)|\le \prod_{j=1}^{M}|\widehat g_j(\xi)|.
\]
A $W^{k,1}$ density has $|\widehat g_j(\xi)|\le C_k|\xi|^{-k}$ at large
frequency and, by nonlattice affine span, is uniformly less than one on every
compact annulus away from zero.  Since $M\ge c\mu$, the annular part is
exponentially small and the large-frequency part has arbitrary polynomial
decay.  The complementary event, including every fixed small-particle sector,
has probability $Ce^{-c\mu}$.
\end{proof}

\subsection{Typical-sector covariance}

The normalized compound pressure is
\[
 Q(\lambda)=\int\bigl(e^{\lambda\cdot X(v)}-1\bigr)d\nu(v).
\]

\begin{lemma}[Uniform covariance from compound activity]
\label{lem:r8-b1-covariance}
For an admissible mark family,
\[
 D^2Q(\lambda)=\int X(v)X(v)^T e^{\lambda\cdot X(v)}d\nu(v)
 \ge c_KI,
 \qquad\lambda\in K.
\]
Moreover, for the total mark $C_{\rm tot}$,
\[
 \mu^{-1}\operatorname{Cov}_\lambda(C_{\rm tot})=D^2Q(\lambda).
\]
\end{lemma}

\begin{proof}
The compound-Poisson logarithmic moment generating function is $\mu Q$.
Differentiating twice proves both identities.  The affine-span condition and
compactness of $K$ give the uniform positive lower bound.  No conditioning on
$N_\mu=n_0$ occurs.
\end{proof}

\subsection{Mixed lattice--continuous coefficient}

Let $L_\mu\in\mathbb Z^r$ be the lattice constraints and
$Y_\mu\in\mathbb R^s$ the continuous constraints.  Let
$(\lambda_\mu,\theta_\mu)$ be the unique exact finite saddle for the extensive
target $(\ell_\mu,y_\mu)$.

\begin{theorem}[Exact finite-saddle coefficient]
\label{thm:r8-b1-coefficient}
Suppose the continuous shell is
\[
 Y_\mu\in y_\mu+w_\mu B,
 \qquad w_\mu=o(\sqrt\mu),
\]
with $w_\mu$ measured in the extensive variable.  Then
\[
 \mathbb P\{L_\mu=\ell_\mu,
 Y_\mu\in y_\mu+w_\mu B\}
 =e^{-\mu I_\mu}
 \frac{1}{(2\pi\mu)^{r/2}}
 \int_{w_\mu B/\sqrt\mu}
 \varphi_{\Sigma_{c,\mu}}(z)\,dz
 \bigl(1+o(1)\bigr),
\]
uniformly on compact admissible target charts.  The formula includes the
empty/small-sector exponential remainder.
\end{theorem}

\begin{proof}
Tilt at the exact finite saddle, so the Fourier phase has zero linear term.
On the major arc use the third-order Taylor expansion and
Lemma~\ref{lem:r8-b1-covariance}.  The intermediate and high-frequency arcs are
controlled by Theorem~\ref{thm:r8-b1-characteristic}.  Fourier inversion in
$lattice\times continuous$ variables gives the coefficient.  Undoing the tilt
produces the exact exponent $I_\mu$ and no centering mismatch.
\end{proof}

\begin{theorem}[Microcanonical transfer and prepared LDP]
\label{thm:r8-b1-preparation}
The exact coefficient theorem transfers bounded marked pressures, covariances,
and finite-dimensional LDPs from the grand-canonical law to the extensive
microcanonical shell.  The transferred covariance is the Schur complement of
$D^2Q$ along the constrained directions.
\end{theorem}

\begin{proof}
Apply Theorem~\ref{thm:r8-b1-coefficient} to the numerator with a bounded mark
source and to the denominator without that source.  The leading shell factors
cancel uniformly.  Differentiating the resulting pressure ratio at zero gives
the conditional Schur complement.  Exponential tightness and the finite-source
Laplace principle give the prepared LDP.
\end{proof}
