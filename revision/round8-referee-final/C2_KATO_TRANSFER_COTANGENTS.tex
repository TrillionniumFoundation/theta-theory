\section{Round-eight reconstruction: closed linear null spaces and Kato transfer bundles}
\label{sec:r8-c2}

The invariant-integral null space is a closed linear span.  Invariant
separators are obtained only under a uniform Cesaro Lyapunov bound.  Source
response is transported in one common transfer-operator Banach bundle; no
Radon--Nikodym derivative between mutually singular infinite-volume Gibbs
measures is used.

\subsection{Two closed linear null spaces}

Let $\mathscr A_W$ be the weighted path-observable Banach space.  Define
\[
 \mathscr N_{\rm int}
 =\overline{\operatorname{span}}
 \{U-U\circ\Theta_t:U\in\mathscr A_W,\ t\ge0\},
\]
\[
 \mathscr N_{\rm pr}=\mathscr N_{\rm int}\oplus\mathbb R\mathbf1.
\]

\begin{theorem}[Invariant-integral quotient]
\label{thm:r8-c2-null}
Suppose every separating weighted functional satisfies the uniform Cesaro
Lyapunov estimate
\[
 \sup_{T\ge1}\frac1T\int_0^T
 \int W\circ\Theta_t\,d|\mu|\,dt<\infty.
\]
Then $F-G\in\mathscr N_{\rm int}$ exactly when all invariant probabilities have
the same integral on $F$ and $G$.  Normalized pressure equivalence uses
$\mathscr N_{\rm pr}$ and therefore also quotients constants.
\end{theorem}

\begin{proof}
Every coboundary integrates to zero under an invariant law, and linear closure
preserves this property.  Conversely, separate $F-G$ from
$\mathscr N_{\rm int}$ by a weighted Radon functional $\mu$.  Average it:
\[
 \mu_T=T^{-1}\int_0^T(\Theta_t)_\#\mu\,dt.
\]
The uniform Cesaro Lyapunov bound and compact sublevels of $W$ give tightness.
Any weak limit is invariant, annihilates every coboundary, and retains the
nonzero pairing with $F-G$, contradicting equality of invariant integrals.
Adding constants gives the normalized pressure quotient.
\end{proof}

\subsection{A common transfer-space bundle}

Let $\mathscr B$ be one fixed strong/weak transfer scale after moving-chart
identifications.  The source-dependent Ruelle operators
$\mathcal L_\eta:\mathscr B^+\to\mathscr B^-$ form a $C^m$ family with an
isolated spectral projector $P_\eta$.  Equilibrium measures are dual
eigenvectors in $\mathscr B^*$; they need not be mutually absolutely
continuous on infinite path space.

\begin{lemma}[Kato parallel transport]
\label{lem:r8-c2-kato}
The equation
\[
 U_\eta'=[P_\eta',P_\eta]U_\eta,
 \qquad U_0=I,
\]
has a bounded invertible solution on $\mathscr B$ and
\[
 P_\eta U_\eta=U_\eta P_0.
\]
It transports the resolved spectral subspace without density square roots.
\end{lemma}

\begin{proof}
The commutator is a bounded continuous operator on the common scale, so the
operator ODE has a unique product-integral solution.  Differentiate
$P_\eta U_\eta-U_\eta P_0$ and use $P_\eta^2=P_\eta$; the derivative is the
same homogeneous commutator equation with zero initial value, hence vanishes.
\end{proof}

Define the covariant derivative of a resolved section by
\[
 \nabla_\eta s=P_\eta U_\eta
 \frac d{d\eta}(U_\eta^{-1}s).
\]

\begin{theorem}[Covariant Doob response without mutual absolute continuity]
\label{thm:r8-c2-doob}
The derivatives of the transfer eigenfunction, dual eigenfunctional, pressure,
Doob operator, and resolved projector exist in the common Banach bundle and are
obtained by the reduced resolvent and Kato connection.  Finite-time likelihood
cocycles implement the same derivative on path cylinders; no infinite-path
Radon--Nikodym derivative is required.
\end{theorem}

\begin{proof}
Differentiate the isolated Riesz projector and use the reduced resolvent on
$(I-P_\eta)\mathscr B$.  Normalize the right and left eigenvectors by their
dual pairing.  Kato transport identifies the moving ranges.  On a finite path
cylinder all source laws have positive densities relative to the same reference
kernel, so ordinary likelihood differentiation is valid there.  The uniform
transfer estimates pass the finite-time formulas to the Banach-bundle limit.
Distinct infinite-volume ergodic laws may remain mutually singular.
\end{proof}

\subsection{Causal memory and moving resolved subspaces}

Let $C_\eta(z)$ be the compressed resolvent on
$P_\eta\mathscr B$ and $A_\eta=P_\eta L_\eta P_\eta$.  Define
\[
 \widehat K_\eta(z)=zP_\eta-A_\eta-C_\eta(z)^{-1}.
\]

\begin{theorem}[Covariant causal-memory derivative]
\label{thm:r8-c2-memory}
Under Kato transport,
\[
 \nabla_\eta\widehat K_\eta
 =z\nabla_\eta P_\eta-
 \nabla_\eta A_\eta+
 C_\eta^{-1}(\nabla_\eta C_\eta)C_\eta^{-1}.
\]
All connection terms from the moving projector and eigenvectors are included.
\end{theorem}

\begin{proof}
Differentiate the correctly signed identity
$\widehat K=zP-A-C^{-1}$ in the common bundle.  The inverse rule gives
$\nabla(C^{-1})=-C^{-1}(\nabla C)C^{-1}$, producing the plus sign in the last
term.  Lemma~\ref{lem:r8-c2-kato} supplies the derivative of the moving range,
and Theorem~\ref{thm:r8-c2-doob} supplies the eigenvector and pairing terms.
\end{proof}

\subsection{Optional projections}

For the A4 past filtration let
\[
 M_t^\varepsilon=
 \frac{\mathbb E[\exp\{\theta\Phi(X_T^\varepsilon)\}
 \mid\mathcal F_t^-]}{
 \mathbb E\exp\{\theta\Phi(X_T^\varepsilon)\}}.
\]

\begin{theorem}[Stable optional-projection convergence]
\label{thm:r8-c2-optional}
The exact likelihood martingales converge jointly with the A4 past kernels to
the diffusion optional projection, stably under bounded initial-past
multipliers.
\end{theorem}

\begin{proof}
Write the conditional expectation as the A4 past-kernel integral of the
terminal functional.  Uniform quenched kernel convergence on compact past sets
gives convergence there; exponential containment removes the complement.
Multiplication by a bounded initial-past variable is absorbed into the same
kernel integral, which proves stable convergence.
\end{proof}

\subsection{Nonlinear contractions without a fictitious adjoint}

For a continuous nonlinear contraction $\mathcal C:X\to Y$, define the test
pullback
\[
 \mathcal C^\sharp\phi=\phi\circ\mathcal C.
\]

\begin{theorem}[Typed pressure--rate contraction chain rule]
\label{thm:r8-c2-contraction}
If $Q_X$ is differentiable on a source chart, then
\[
 Q_Y(\phi)=Q_X(\mathcal C^\sharp\phi),
 \qquad
 DQ_Y(\phi)h=
 DQ_X(\phi\circ\mathcal C)(h\circ\mathcal C).
\]
The rate on $Y$ is
\[
 I_Y(y)=\inf_{\mathcal C(x)=y}I_X(x).
\]
For smooth parameterized contractions, the next derivative uses the ordinary
Frechet chain rule for $\phi\circ\mathcal C_\eta$.
\end{theorem}

\begin{proof}
The Laplace functional of the pushforward is the Laplace functional of the
pulled-back test.  Differentiate this identity on the declared source chart.
The rate formula is the contraction principle.  No linear adjoint of the
nonlinear map $\mathcal C$ is introduced.
\end{proof}
