\section{Round-twelve positive closure: deterministic-clock Markov-renewal path LDP}
\label{sec:r12-a3}

The process is treated directly as a countable Markov-renewal process stopped
at a deterministic collision or physical clock.  The terminal incomplete
excursion is part of the state from the beginning, and no finite cemetery
quotient is used.

\subsection{The compactified excursion kernel}

Let $E$ be the countable set of complete first-return branches and let
$P(a,b)$ be the normalized Gibbs transition kernel.  A mark
\[
 \Xi(a)=(r(a),\mathfrak t(a),K_a,\mathfrak s_a)
\]
contains collision length, roof, the complete graph-completed excursion
profile, and singularity exposure.  Normalize a long excursion by its
collision length and take the weighted weak closure; the resulting compact
mark space is $\overline E$.  The exponential return and roof estimates give
\[
 \sup_a\sum_bP(a,b)e^{\eta(r(b)+\mathfrak t(b))}<\infty.
\]

\begin{lemma}[Lyapunov--minorization for the excursion chain]
\label{lem:r12-a3-harris}
There are $V\ge1$, a finite small set $C$, $\rho<1$, and $b<\infty$ such that
$PV\le\rho V+b1_C$, and a fixed iterate of $P$ minorizes one probability on
$C$.  The same estimates hold for all bounded profile sources and for clock
sources in a common open interval.
\end{lemma}

\begin{proof}
Take $V(a)=e^{\eta_0(r(a)+\mathfrak t(a))}$ with $\eta_0<\eta$.  The transfer
operator exponential-moment estimate gives the drift after removing a finite
return core.  The finite connector family sends every core state to one common
regular branch in bounded time; bounded distortion gives the minorization.
Source multiplication changes the constants continuously on the declared
ball.
\end{proof}

\subsection{A controlled renewal representation}

For collision horizon $n$ define $S_k=\sum_{j<k}r(A_j)$ and
$N(n)=\max\{k:S_k\le n\}$.  The state consists of the empirical transition
flow of $A_0,\ldots,A_{N(n)-1}$, the clock-weighted complete-excursion
profiles, and the pointed prefix of $A_{N(n)}$ at age $n-S_{N(n)}$.

\begin{theorem}[Entropy representation at deterministic collision time]
\label{thm:r12-a3-control}
For every bounded continuous functional $F$ of the preceding state,
\[
 -{1\over n}\log\mathbb E e^{-nF}
 =\inf_{\mathbf Q}
 \mathbb E_{\mathbf Q}\left[
 F+{1\over n}\sum_{j<N(n)}
 D(Q_j(\cdot\mid\mathcal F_j)\Vert P(A_j,\cdot))
 +{1\over n}D(Q_{\rm term}\Vert P_{\rm term})
 \right],
\]
where the infimum is over predictable controlled transition and terminal
residual kernels.  The identity remains valid for unbounded clock/profile
coordinates with finite exponential moment.
\end{theorem}

\begin{proof}
Apply the Gibbs variational identity conditionally at each transition and at
the final residual choice.  Iteration gives the formula for bounded stopping
levels.  Stop first when $S_k\wedge k$ reaches $M$, then let $M\to\infty$;
Lemma~\ref{lem:r12-a3-harris} gives uniform integrability of the entropy and
clock terms.  Monotone truncation proves the unbounded-coordinate statement.
\end{proof}

\begin{theorem}[Full collision-clock LDP]
\label{thm:r12-a3-collision}
The deterministic-horizon states satisfy a full good LDP with speed $n$.  Its
rate is the lower-semicontinuous control cost
\[
 I^c(\lambda)=
 \inf_{\mathbf Q:\,\mathcal L(\mathbf Q)=\lambda}
 \liminf_{n\to\infty}{1\over n}
 \mathbb E_{\mathbf Q}
 \left[\sum_{j<N(n)}D(Q_j\Vert P(A_j,\cdot))
       +D(Q_{\rm term}\Vert P_{\rm term})\right].
\]
The state includes recurrent flows, arbitrary finite-cost one-big-excursion
profiles, and the stopped terminal prefix.  Every finite-rate point has a
recovery by actual admissible excursions.
\end{theorem}

\begin{proof}
The entropy representation gives the Laplace upper bound after tightness.
The drift condition makes clock-weighted empirical measures exponentially
tight on $\overline E$, while the A2 singularity shield gives exponentially
good continuous approximations of every finite path window.  For the lower
bound, approximate a finite-cost control by stationary finite-memory controls
on a finite Lyapunov core, retain each long-excursion control as an actual
kernel rather than collapsing it, and join core pieces with the finite
connector family.  The connector clock and entropy are $o(n)$.  The controlled
law of large numbers, including the terminal residual kernel, gives the
recovery.  The Laplace principle and exponential tightness are equivalent to
the stated full good LDP.
\end{proof}

\subsection{Recession and singularity exposure}

\begin{proposition}[Actual one-excursion epigraph]
\label{prop:r12-a3-epigraph}
Let $\mathcal R_L$ be the set of actual first-return excursions with
$r\ge L$, represented by normalized profile and cost
$-r^{-1}\log P(a\mid\text{entrance})$.  The closed epigraphs converge in the
Painlev\'e--Kuratowski sense to a nonempty closed epigraph on the boundary
mark space.  This boundary cost is exactly the restriction of $I^c$ to a
single macroscopic terminal excursion.
\end{proposition}

\begin{proof}
Exponential clock moments give epigraph compactness.  Every limit point is a
liminf by lower semicontinuity of relative entropy.  Conversely choose an
approximating actual excursion from the defining closed epigraph and use it as
the terminal control in Theorem~\ref{thm:r12-a3-control}; no concatenation is
needed.  The resulting cost and profile converge, proving recovery and the
identification with the full rate.
\end{proof}

\begin{lemma}[Uniform singularity shield on recurrent and recession sectors]
\label{lem:r12-a3-shield}
For every $M$ and $\epsilon$ there are a depth $q$, a neighborhood width
$\delta$, and a continuous shielded profile map such that, uniformly over
controls of cost at most $M$, the probability that the shielded and actual
clock-weighted profiles differ by more than $\epsilon$ is at most $e^{-Mn}$.
\end{lemma}

\begin{proof}
On one homogeneous excursion, branch complexity is exponential in depth while
its Gibbs mass and singularity-neighborhood mass have a strictly stronger
exponential decay after choosing $q$ and then $\delta$.  Entropy inequality
transfers this estimate uniformly to every bounded-cost control.  The same
calculation is divided by the actual excursion clock, so it remains valid for
a macroscopic recession excursion.  Summing recurrent excursions and the one
terminal excursion proves the claim.
\end{proof}

\subsection{Physical time}

For physical horizon $T$, stop at
$N_p(T)=\max\{k:\sum_{j<k}\mathfrak t(A_j)\le T\}$ and retain the age and
position inside the final free flight.

\begin{theorem}[Physical-clock LDP]
\label{thm:r12-a3-physical}
The physical-time empirical path states satisfy a full good LDP with speed
$T$.  Their rate is the same entropy-control formula as
Theorem~\ref{thm:r12-a3-collision}, divided by the controlled mean physical
clock and including the physical residual kernel.  Collision- and
physical-time rates are two applications of the deterministic-clock theorem,
not a change of speed by an unproved contraction.
\end{theorem}

\begin{proof}
Repeat the entropy representation with the roof stopping time.  The joint
exponential moment of $(r,\mathfrak t)$ gives tightness and makes the number of
transitions linear in $T$ on every finite-cost sublevel.  The lower-bound
control is stopped inside its final free flight, whose conditional residual
law is included in the state.  The same upper and recovery arguments therefore
produce the physical-clock Laplace principle.
\end{proof}

\begin{theorem}[Round-twelve A3 closure]
\label{thm:r12-a3-main}
The Sinai inducing process has full recurrent--recession empirical-path LDPs
at deterministic collision and physical clocks, including actual long
excursions, singularity exposure, and the pointed terminal prefix.
\end{theorem}

\begin{proof}
Combine Theorems~\ref{thm:r12-a3-collision} and
\ref{thm:r12-a3-physical}, Proposition~\ref{prop:r12-a3-epigraph}, and
Lemma~\ref{lem:r12-a3-shield}.
\end{proof}
