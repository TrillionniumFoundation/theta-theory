\section{Round-twelve positive closure: Harris past kernels, quenched rough limits, and renewal memory}
\label{sec:r12-a4}

The complete past is the left-infinite Markov-renewal history from A3.  Its
conditional future kernel is constructed from the normalized Gibbs kernel,
not inferred from the path LDP.  The Poisson algebra, nonlinear normalization,
and the causal memory sign are fixed throughout.

\subsection{A weighted spectral theorem for the actual past kernel}

Let $H_k=(\ldots,A_{k-1},A_k)$ and let $P$ append one excursion according to
the A3 kernel.  Put
\[
 W(H)=1+\sum_{j\ge0}e^{-\gamma j}
       e^{\eta(r(A_{-j})+\mathfrak t(A_{-j}))}.
\]
Let $\mathcal H_{W,\alpha}$ be the weighted dynamically H\"older functions on
pasts.

\begin{theorem}[Past-kernel Harris--Doeblin theorem]
\label{thm:r12-a4-harris}
There are $C<\infty$, $\rho<1$, a compact small set $K$, and a probability
$\nu_K$ such that
\[
 PW\le\rho W+C,
 \qquad
 P^m(H,\cdot)\ge\epsilon\nu_K(\cdot),\quad H\in K.
\]
Consequently $P$ is Feller on the weighted history topology and
\[
 \|P^nf-\pi(f)\|_{W,\alpha}
 \le C\rho^n\|f\|_{W,\alpha}.
\]
All constants are locally uniform under the bounded source twists used in A2
and A3.
\end{theorem}

\begin{proof}
The newest coordinate contributes the A3 exponential return moment; all older
coordinates are shifted and multiplied by $e^{-\gamma}$.  Choosing $\gamma$
and the moment exponent gives the drift.  On a finite return/homogeneity core,
the connector theorem supplies one common regular word of bounded length.
Bounded distortion gives a uniform lower density on its terminal cylinder,
which is the minorization.  Coupling at the minorization time and using the
drift on excursions from $K$ proves the weighted contraction.  Continuity of
the Gibbs branch weights and of the shift gives the Feller property.
\end{proof}

\subsection{Poisson martingales and the quenched suspension limit}

For centered $g\in\mathcal H_{W,\alpha}$ define
\[
 h=\sum_{j\ge0}P^jg,
 \qquad
 D_{k+1}=h(H_{k+1})-Ph(H_k).
\]
Then $D_{k+1}$ is a martingale difference and
\[
 g(H_k)=D_{k+1}+h(H_k)-h(H_{k+1}).
\]

\begin{lemma}[Uniform Poisson and bracket estimates]
\label{lem:r12-a4-poisson}
The series for $h$ converges in $\mathcal H_{W,\alpha}$.  For every compact
$W$-sublevel of initial pasts, predictable brackets of the martingale array
converge locally uniformly to the Green--Kubo covariance, the conditional
Lindeberg condition holds, and the coboundary divided by $\sqrt n$ vanishes
uniformly in probability.
\end{lemma}

\begin{proof}
Theorem~\ref{thm:r12-a4-harris} sums the Poisson series.  Apply the same
spectral estimate to the centered conditional second moment
$P(hh^T)-(Ph)(Ph)^T$ to obtain bracket convergence.  Exponential clock and
observable moments imply Lindeberg by truncation.  The drift inequality makes
$h(H_n)/\sqrt n$ uniformly negligible on compact initial $W$-sublevels.
\end{proof}

\begin{theorem}[Quenched enhanced suspension invariance principle]
\label{thm:r12-a4-rough}
For every finite vector of centered dynamically H\"older observables, the
collision sums and their canonical iterated integrals converge, uniformly on
compact past sets, to a Brownian rough path with covariance
\[
 D=\pi(DD^T)
\]
and deterministic antisymmetric area correction
\[
 \Gamma={1\over2}\sum_{k\ge1}
 \pi(g\otimes P^kg-P^kg\otimes g).
\]
The same statement holds in physical time after A3 roof stopping; the terminal
flight is jointly negligible at diffusive scale.
\end{theorem}

\begin{proof}
Lemma~\ref{lem:r12-a4-poisson} and the martingale functional central limit
theorem give stable first-level convergence.  Burkholder estimates and the
bracket theorem give tightness and convergence of the martingale iterated
integrals.  Expanding the coboundary cross terms yields the absolutely
convergent area series.  The A3 joint roof LDP and exponential moment imply a
uniform renewal law of large numbers and $o(\sqrt T)$ residual clock; the
random-time-change theorem then gives the physical suspension result.
\end{proof}

\subsection{The genuine nonlinear history semigroup}

For a bounded potential $V$, let $T_t^V$ be the Feynman--Kac semigroup.  Let
$e^{t\lambda_V}$ and $h_V>0$ be its leading eigen-data from
Theorem~\ref{thm:r12-a4-harris}.  Define
\[
 \widehat P_t^VF
 =e^{-t\lambda_V}h_V^{-1}T_t^V(h_VF),
 \qquad
 \mathcal E_t^VF=\log\widehat P_t^V(e^F).
\]

\begin{proposition}[Doob nonlinear tower]
\label{prop:r12-a4-doob}
$\widehat P_t^V$ is a Markov semigroup on the weighted history space and
$\mathcal E_{s+t}^V=\mathcal E_s^V\mathcal E_t^V$.  Its generator on the
Doob graph core is
$e^{-F}\widehat L_Ve^F$.
\end{proposition}

\begin{proof}
The eigenfunction relation cancels the Feynman--Kac growth and gives
$\widehat P_t^V1=1$.  The linear tower then survives conjugation by the same
eigenfunction, and exponentiation gives the nonlinear tower.  Differentiation
on the graph core gives the generator.
\end{proof}

\subsection{Continuous-time renewal resolvent and memory}

Let $L_V$ be the closed generator of the Doob suspension in
$L^2(\widehat\pi_V)$ and let $R$ be a finite-rank orthogonal resolved
projection.  Entrance, complete-excursion, and terminal-flight operators give
for $\Re z>0$
\[
 (z-L_V)^{-1}=E_V(z)(I-\mathcal L_{V,z})^{-1}F_V(z)+G_V(z).
\]

\begin{theorem}[Renewal resolvent strip]
\label{thm:r12-a4-resolvent}
There are $\gamma>0$ and $\delta>0$ such that the preceding identity extends
meromorphically to $\Re z> -\gamma$.  After subtracting the finitely many
large-$z$ polynomial terms and the finitely many poles/zeros in
$\{|\Im z|\le B\}$, the residual compressed transfer function and its first
two vertical derivatives are $O((1+|\Im z|)^{-1-\delta})$ on every smaller
strip.  The number of transmission zeros there is finite.
\end{theorem}

\begin{proof}
The renewal identity follows by splitting a suspension path into entrance,
complete returns, and the terminal flight.  The A2 small/compact/moderate/
very-high estimates apply to $(I-\mathcal L_{V,z})^{-1}$; the entrance and
flight operators gain one vertical power by integration over age.  Repeating
that integration twice gives the stated integrable derivatives.  High
frequency invertibility follows from the multi-block estimate.  Analyticity
then confines poles and determinant zeros to a compact rectangle, where they
are finite unless the determinant is identically zero; strict accretivity in
the right half-plane excludes that alternative.
\end{proof}

Define
\[
 C_V(z)=R(z-L_V)^{-1}R,
 \qquad
 \widehat K_V(z)=zR-RL_VR-C_V(z)^{-1}.
\]

\begin{theorem}[Unique normalized memory decomposition]
\label{thm:r12-a4-memory}
There is a unique decomposition, normalized by controllable--observable
minimal realization,
\[
 K_V=\sum_{j=0}^{J}A_j\delta_0^{(j)}
    +\sum_{\ell=1}^{N}e^{\zeta_\ell t}P_\ell
    +K_V^{\rm reg}(t),
\]
where $\Re\zeta_\ell<0$ and
$\|K_V^{\rm reg}(t)\|\le Ce^{-\gamma' t}$.  For every resolved observable the
projected Koopman evolution satisfies the exact causal Volterra equation with
this kernel.  Descriptor realizations are unique up to similarity after
minimality and the monic high-frequency normalization are imposed.
\end{theorem}

\begin{proof}
Polynomial large-$z$ terms invert to the displayed instantaneous
distributions.  Residues of poles and transmission zeros give a finite
minimal descriptor system.  The residual transform and two vertical
derivatives are integrable by Theorem~\ref{thm:r12-a4-resolvent}; shifting the
Bromwich contour gives the exponentially decaying function.  Laplace
transformation of the projected resolvent identity gives the Volterra
equation.  Standard controllability/observability reduction and monic
normalization give uniqueness up to similarity.
\end{proof}

\begin{theorem}[Round-twelve A4 closure]
\label{thm:r12-a4-main}
The Sinai past kernel has a proved weighted spectral gap, a uniform quenched
rough suspension limit, an exact Doob nonlinear history semigroup, and the
normalized distributional-plus-decaying memory decomposition of
Theorem~\ref{thm:r12-a4-memory}.
\end{theorem}

\begin{proof}
Combine Theorems~\ref{thm:r12-a4-harris},
\ref{thm:r12-a4-rough}, \ref{thm:r12-a4-resolvent}, and
\ref{thm:r12-a4-memory}, together with Proposition~\ref{prop:r12-a4-doob}.
\end{proof}
