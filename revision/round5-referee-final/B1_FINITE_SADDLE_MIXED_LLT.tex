\section{Round-five positive closure: the exact finite-volume saddle and the primitive mixed shell coefficient}
\label{sec:r5-b1}

The conditioning variables are particle number, total momentum and total
energy.  Every Fourier estimate is centered at the exact finite-volume saddle.
The lattice coordinate is handled by its span-one activity, while the
continuous Cram\'er condition is proved from a finite velocity block by
coarea.

\subsection{The exact source-dependent saddle}

Let
\[
 C(Z_N)=\left(N,\sum_{i=1}^Nv_i,
                   \sum_{i=1}^N|v_i|^2\right)
 \in\mathbb Z\times\mathbb R^3\times\mathbb R.
\]
For a bounded path/contact source \(H\) in the B2 grand-canonical chart and a
multiplier \(\lambda=(\lambda_N,\lambda_P,\lambda_E)\), put
\[
 Q_\varepsilon(H,\lambda)=
 \frac1{\mu_\varepsilon}
 \log\mathbb E_\varepsilon^{\rm gc}
 \exp\{\mu_\varepsilon H+\lambda\cdot C\}.
\]
Let \(a_\varepsilon\) be a compatible target with
\(a_\varepsilon\to a\) and with an integer particle-number coordinate.  The
finite-volume saddle is defined by
\[
 D_\lambda Q_\varepsilon(H,\lambda_{H,\varepsilon})
 =a_\varepsilon.
\]

\begin{lemma}[Uniform strict constraint convexity]
\label{lem:r5-b1-convex}
On compact regular source and target sets there are \(0<c<C<\infty\) such
that
\[
 cI_5\le D_{\lambda\lambda}^2Q_\varepsilon(H,\lambda)
 \le CI_5
\]
uniformly for small \(\varepsilon\).  Hence the exact saddle exists uniquely,
\[
 \lambda_{H,\varepsilon}\to\lambda_H,
\]
and its first two source derivatives converge locally uniformly.
\end{lemma}

\begin{proof}
The Hessian is
\(\mu_\varepsilon^{-1}\operatorname{Cov}(C,C)\) under the tilted law.
The one-particle connected activity contributes a strictly positive second
derivative in the number direction: varying the activity changes the random
number of singleton components, so this coordinate is not a centered
constant.  For the momentum--energy block, take \(m_0=5\) singleton velocity
marks.  The map
\[
 \mathcal S(v_1,\ldots,v_{m_0})=
 \left(\sum_iv_i,\sum_i|v_i|^2\right)
\]
has rank four whenever the velocities are not all equal.  Coarea and the
Gaussian lower bound on a compact velocity box show that its pushforward has
a \(C^3\) density bounded below on a compact subset of
\[
 e>|p|^2/m_0.
\]
The covariance of this block is therefore positive in every nonzero
momentum--energy direction.  A mixed number/continuous vector has positive
variance because its restriction to the one-particle and the \(m_0\)-particle
sectors cannot both be constant.  The connected sectors with at least two
collisions are \(O(T)\) in the B2 tree norm; choose the common short horizon so
that they cannot close the singleton/block spectral gap.  This proves the
bounds.  The analytic implicit-function theorem at the exact finite-volume
critical equation and local uniform convergence of the B2 pressure give the
remaining assertions.
\end{proof}

\subsection{The primitive continuous block}

Let \(g_{H,\lambda}^{(m_0)}(p,e)\) be the pushforward density of the selected
\(m_0\)-particle singleton block.  In center-of-mass variables
\(v_i=p/m_0+w_i\), \(\sum_iw_i=0\), coarea gives
\[
 g_{H,\lambda}^{(m_0)}(p,e)=
 1_{e>|p|^2/m_0}
 \left(e-|p|^2/m_0\right)^{(3m_0-5)/2}
 b_{H,\lambda}(p,e),
\]
where \(b_{H,\lambda}\) is \(C^3\), positive on compact interior sets and has
Gaussian tails, uniformly on compact source charts.

\begin{lemma}[Uniform mixed Cram\'er condition]
\label{lem:r5-b1-cramer}
Let \(t\in[-\pi,\pi]\) be the number frequency and
\(u\in\mathbb R^4\) the momentum--energy frequency.  For every compact set
not containing \((0,0)\), the centered characteristic function of the exact
tilt satisfies
\[
 |\Phi_{H,\varepsilon}(t,u)|
 \le e^{-c\mu_\varepsilon}
 +C_M(1+\sqrt{\mu_\varepsilon}|u|)^{-M}
\]
for every fixed \(M\), with a uniform \(c>0\) on compact frequency annuli.
There are no mixed lattice resonances other than \((0,0)\).
\end{lemma}

\begin{proof}
At \(u=0\), the one-particle activity contributes to the normalized connected
logarithm the real part
\[
 a_1(H,\lambda)(\cos t-1)
 =-2a_1(H,\lambda)\sin^2(t/2),
\]
with \(a_1\) uniformly positive.  Hence every \(t\ne0\) in the fundamental
interval has a strict gap.  When \(t\) is small and \(u\) belongs to a compact
annulus, the \(m_0\)-particle block contributes
\[
 e^{im_0t}\widehat g_{H,\lambda}^{(m_0)}(u)-\|g_{H,\lambda}^{(m_0)}\|_1.
\]
The absolutely continuous nonlattice density has strictly smaller Fourier
modulus at \(u\ne0\); compactness makes the gap uniform.  Potential resonances
with \(e^{im_0t}=1\) are still excluded by the one-particle term unless
\(t=0\).  For large \(u\), three integrations by parts in the explicit coarea
density give polynomial decay; increasing the block size gives arbitrary
fixed order \(M\).  The B2 connected remainder is \(O(T)\) on compact
frequencies and has the same integration-by-parts bound after source
insertions.  The short-time choice from
Lemma~\ref{lem:r5-b1-convex} preserves half of every block gap, proving the
claim.
\end{proof}

\subsection{Central expansion at the exact saddle}

Under the tilt \((H,\lambda_{H,\varepsilon})\), define
\[
 \widetilde C_\varepsilon=C-\mu_\varepsilon a_\varepsilon
\]
and
\[
 \Phi_{H,\varepsilon}(t,u)=
 \mathbb E_{H,\lambda_{H,\varepsilon},\varepsilon}
 e^{it\widetilde N+i u\cdot\widetilde Y}.
\]
The exact saddle makes the linear term vanish.

\begin{lemma}[Uniform central characteristic expansion]
\label{lem:r5-b1-central}
There are \(c,\delta>0\) such that
\[
 |\Phi_{H,\varepsilon}(t,u)|
 \le e^{-c\mu_\varepsilon(t^2+|u|^2)}
\]
for \(|t|+|u|\le\delta\), and uniformly for
\(|t|+|u|\le\mu_\varepsilon^{-2/5}\),
\[
 \log\Phi_{H,\varepsilon}(t,u)=
 -\frac{\mu_\varepsilon}{2}
 (t,u)^T\Sigma_{H,\varepsilon}(t,u)
 +O\bigl(\mu_\varepsilon(|t|+|u|)^3\bigr),
\]
where
\(cI\le\Sigma_{H,\varepsilon}\le CI\).
\end{lemma}

\begin{proof}
Taylor-expand the finite-volume pressure at the exact critical point.  The
first derivative is zero by its defining equation.  The Hessian bounds are
Lemma~\ref{lem:r5-b1-convex}; normal convergence on a larger complex source
chart gives a uniform third derivative.  Shrinking \(\delta\) absorbs the
cubic real part into the quadratic form.
\end{proof}

\subsection{The exact mixed shell coefficient}

Write the continuous target as
\(Y=(P,E)\).  Let \(N_\varepsilon\) be the compatible integer target and let
\(B_\varepsilon\subset\mathbb R^4\) be a centered rectangular or smooth shell
with diameter \(\delta_\varepsilon\) satisfying
\[
 \delta_\varepsilon\downarrow0,
 \qquad \sqrt{\mu_\varepsilon}\delta_\varepsilon\to\infty.
\]
The shell is centered at the exact tilted mean
\(\mu_\varepsilon a_{Y,\varepsilon}\).

\begin{theorem}[Source-uniform primitive shell coefficient]
\label{thm:r5-b1-coefficient}
Uniformly on compact regular source and target sets,
\[
 \begin{aligned}
 &\mathbb P_{H,\lambda_{H,\varepsilon},\varepsilon}
 \left\{
  N=N_\varepsilon,
  Y-\mu_\varepsilon a_{Y,\varepsilon}\in
  \mu_\varepsilon B_\varepsilon
 \right\}\\
 &\qquad=
 \mu_\varepsilon^{-1/2}
 \left[
  (2\pi\sigma_{N\mid Y,H}^2)^{-1/2}+o(1)
 \right],
 \end{aligned}
\]
where
\[
 \sigma_{N\mid Y,H}^2
 =\Sigma_{NN}-\Sigma_{NY}\Sigma_{YY}^{-1}\Sigma_{YN}
\]
is uniformly bounded above and below by positive constants.  In particular
the logarithm of the shell probability is
\(o(\mu_\varepsilon)\).
\end{theorem}

\begin{proof}
Use lattice Fourier inversion in \(t\in[-\pi,\pi]\) and smooth upper and lower
approximations of the continuous shell.  In the central region scale
\((t,u)\) by \(\mu_\varepsilon^{-1/2}\) and use
Lemma~\ref{lem:r5-b1-central}.  Conditional Gaussian integration gives the
lattice prefactor with variance \(\sigma_{N\mid Y,H}^2\).  Because
\(\sqrt{\mu_\varepsilon}\delta_\varepsilon\to\infty\), the continuous shell
contains an increasing number of conditional standard deviations and its
conditional Gaussian mass tends to one.

On compact minor arcs apply Lemma~\ref{lem:r5-b1-cramer}; on unbounded
continuous frequencies choose the smoothing order larger than four and use
its polynomial term.  Both contributions are
\(o(\mu_\varepsilon^{-1/2})\).  Uniform Schur-complement bounds follow from
Lemma~\ref{lem:r5-b1-convex}.  Centering at
\(\lambda_{H,\varepsilon}\), rather than at its limit, removes any unproved
moderate-deviation displacement.
\end{proof}

\subsection{Source-dependent microcanonical transfer}

Let \(\mathcal S_\varepsilon(a_\varepsilon)\) be the primitive shell.  Exact
exponential tilting gives
\[
 \begin{aligned}
 &\frac1{\mu_\varepsilon}
 \log\mathbb E_\varepsilon^{\rm gc}
 \left[e^{\mu_\varepsilon H}\mid
       \mathcal S_\varepsilon(a_\varepsilon)\right]\\
 &\quad=
 Q_\varepsilon(H,\lambda_{H,\varepsilon})
 -\lambda_{H,\varepsilon}\cdot a_\varepsilon
 -Q_\varepsilon(0,\lambda_{0,\varepsilon})
 +\lambda_{0,\varepsilon}\cdot a_\varepsilon+o(1).
\end{aligned}
\]

\begin{theorem}[Exact source-dependent microcanonical pressure]
\label{thm:r5-b1-transfer}
The conditional pressure converges locally uniformly to
\[
 Q^a(H)=
 \inf_\lambda\{Q(H,\lambda)-\lambda\cdot a\}
 -\inf_\lambda\{Q(0,\lambda)-\lambda\cdot a\}.
\]
The minimizer \(\lambda_H\) is unique and analytic.  Its first derivative is
the prepared mean and its Hessian is the Schur complement
\[
 D^2Q^a=
 Q_{HH}-Q_{H\lambda}Q_{\lambda\lambda}^{-1}Q_{\lambda H}.
\]
The prepared initial empirical measure satisfies the corresponding good
microcanonical LDP.
\end{theorem}

\begin{proof}
Apply Theorem~\ref{thm:r5-b1-coefficient} to the numerator and denominator
after exact finite-volume tilting.  The ratio of shell coefficients has
logarithm \(o(\mu_\varepsilon)\), uniformly on the source chart.  The saddle
convergence from Lemma~\ref{lem:r5-b1-convex} gives the limiting formula.
Differentiate the exact critical equation and the envelope to obtain the
Schur complement.  The conditional Laplace principle and exponential
tightness inherited from the grand-canonical Maxwellian law give the initial
LDP.
\end{proof}

\begin{theorem}[Round-five B1 closure]
\label{thm:r5-b1-main}
Primitive microcanonical preparation is centered at the unique exact
finite-volume source-dependent saddle.  The mixed lattice--continuous shell
coefficient follows from a proved span-one number gap and a coarea-derived
momentum--energy Cram\'er block, and conditioning yields the analytic prepared
pressure, Schur-complement covariance and initial LDP.
\end{theorem}

\begin{proof}
Combine Lemma~\ref{lem:r5-b1-convex},
Theorem~\ref{thm:r5-b1-coefficient}, and
Theorem~\ref{thm:r5-b1-transfer}.
\end{proof}
