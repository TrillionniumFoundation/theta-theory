\section{Round-nine reconstruction: quenched past kernels and distributionally separated memory}
\label{sec:r9-a4}

The first part constructs a Markov/Feller process from the genuine symbolic
past and proves a quenched enhanced invariance principle.  The second part is
an operator theorem for compressed resolvents.  Instantaneous Dirac terms are
separated before any exponential-decay statement, and the full unresolved
space is retained in the realization.

\subsection{The natural-extension past process}

Let $(\widehat\Omega,\widehat\sigma,\widehat\nu_\Psi)$ be one recurrent Gibbs
component from A3, with countable alphabet, summable-variation potential, and
roof $\tau$.  A history is
\[
 h=(\ldots,\omega_{-2},\omega_{-1},\omega_0,s),
 \qquad 0\le s<\tau(\omega_0).
\]
Equip histories with
\[
 d_\beta(h,h')=|s-s'|\wedge1+
 \sum_{j\ge0}e^{-\beta j}1_{\{\omega_{-j}\ne\omega'_{-j}\}}
\]
and weight $W(h)=1+\tau(\omega_0)+\sum_{j\ge0}e^{-\beta j}r(\omega_{-j})$.
Let $C_0^W$ be the weighted uniformly continuous functions vanishing at
$W$-infinity.

The normalized $g$-function of the natural extension gives the conditional
law of the next symbol given the past.  Flowing the age coordinate until the
roof is reached and then using this kernel defines $P_t(h,\cdot)$.

\begin{theorem}[Weighted past-kernel Feller semigroup]
\label{thm:r9-a4-feller}
The kernels $(P_t)_{t\ge0}$ satisfy Chapman--Kolmogorov and define a strongly
continuous positive contraction semigroup on $C_0^W$.  On every compact
history set,
\[
 |P_tF(h)-P_tF(h')|
 \le C_T\omega_F(C_Td_\beta(h,h'))+C_T\|F\|_W e^{-cL}
\]
when the two pasts agree through depth $L$.  The associated resolvent is
normal and separating, hence its Ray compactification carries a
right-continuous strong Markov realization.
\end{theorem}

\begin{proof}
Summable variations make the one-step $g$-kernel Lipschitz in the weighted past
metric.  Couple equal finite pasts with the same future symbols until the
coupling fails; the failure probability is bounded by the variation tail.
Exponential return tails control the number of renewals before time $T$ and
the $W$-tail uniformly on compact sets.  This gives the displayed modulus,
Feller continuity, and preservation of $C_0^W$.  Right continuity of the age
flow and the renewal tail give strong continuity.  Splitting at the first
renewal proves Chapman--Kolmogorov.  The Laplace resolvent is conservative,
normal by monotone convergence, and separates histories through cylinder
functions, so the Ray theorem applies without using a process constructed in
advance.
\end{proof}

\subsection{The oriented conditional Poisson equation}

Let $A$ be a centered vector observable in the A2/A3 weighted H\"older class.
Write $\mathsf P$ for the one-step past-state Markov operator.  The spectral
gap on the recurrent component gives
\[
 \chi=\sum_{n\ge1}\mathsf P^nA
\]
in the weighted H\"older norm.  Define
\[
 D_{n+1}=A(H_n)+\chi(H_{n+1})-\mathsf P\chi(H_n).
\]

\begin{lemma}[Conditional martingale--coboundary decomposition]
\label{lem:r9-a4-martingale}
For the past filtration $\mathcal F_n^-=\sigma(H_0,\ldots,H_n)$,
\[
 E(D_{n+1}\mid\mathcal F_n^-)=0,
\]
and
\[
 \sum_{j<n}A(H_j)=\sum_{j<n}D_{j+1}
 +\chi(H_0)-\chi(H_n)+R_n,
\]
where the roof-centering remainder $R_n$ has exponential moments uniformly for
initial histories in a compact set.
\end{lemma}

\begin{proof}
The definition of $\mathsf P$ gives
$E(\chi(H_{n+1})\mid\mathcal F_n^-)=\mathsf P\chi(H_n)$, proving the martingale
property.  The Poisson identity
$(I-\mathsf P)\chi=\mathsf PA$ gives the telescoping decomposition after one
index shift.  The suspension interpolation and unfinished final roof segment
form $R_n$; the A3 pointed-excursion tail supplies its uniform exponential
moment.
\end{proof}

\begin{theorem}[Uniform quenched enhanced invariance principle]
\label{thm:r9-a4-quenched}
For every compact history set, the conditional laws of
\[
 \epsilon\int_0^{t/\epsilon^2}A(H_s)\,ds
\]
and their canonical second iterated integrals converge uniformly to a Brownian
rough path.  The covariance is the conditional Green--Kubo form and the
antisymmetric second-level drift is the absolutely convergent area anomaly.
The convergence is stable with respect to bounded initial-history variables.
\end{theorem}

\begin{proof}
Lemma~\ref{lem:r9-a4-martingale} reduces the first level to a triangular
martingale array.  The conditional brackets are additive functions of the past
state; the Feller coupling and recurrent spectral gap make their time averages
converge uniformly on compact histories.  Uniform exponential moments give
conditional Lindeberg and tightness.  The martingale rough invariance theorem
then gives the first and second levels.  The coboundary contributes the
antisymmetric limit
\[
 \Gamma=\tfrac12\sum_{n\in\mathbb Z}
 \bigl(E[A_0\otimes A_n]-E[A_n\otimes A_0]\bigr),
\]
which is absolutely convergent.  Multiplying by an initial-history test does
not alter the conditional estimates, proving stable convergence.
\end{proof}

\subsection{Compressed resolvents and instantaneous distributions}

Let $\mathcal H=L^2(\widehat\nu_\Psi)$, let $L$ be the closed Doob generator,
and let $P$ be an arbitrary finite-rank orthogonal projection with
$P\mathcal H\subset D(L^{M+2})$.  Put $Q=I-P$ and
\[
 C(z)=P(z-L)^{-1}P,
 \qquad
 \widehat K(z)=zP-PLP-C(z)^{-1}.
\]
The exact full unresolved realization is the block operator
\[
 L=\begin{pmatrix}PLP&PLQ\\QLP&QLQ\end{pmatrix}
 \quad\text{on }P\mathcal H\oplus Q\mathcal H.
\]

\begin{lemma}[Instantaneous/residual decomposition]
\label{lem:r9-a4-instant}
The high-frequency expansion of $C(z)$ determines a polynomial
$\mathcal P_M(z)$ such that
\[
 \widehat K(z)=\mathcal P_M(z)+\widehat K_{\rm reg}(z),
 \qquad
 \|\widehat K_{\rm reg}(-\gamma+ib)\|
 \le C(1+|b|)^{-1-\delta}
\]
whenever the A2 resolvent has the corresponding strip and derivative bounds.
The inverse Laplace transform of $\mathcal P_M$ is a finite sum of
$\delta_0$ and its derivatives; only $K_{\rm reg}$ is called the decaying
memory function.
\end{lemma}

\begin{proof}
Expand $(z-L)^{-1}$ on $P\mathcal H$ through order $M+2$ and invert the finite
matrix series for $C(z)$.  Nonnegative powers of $z$ and constants form
$\mathcal P_M$.  The declared graph-domain resolvent estimates allow
$M+2$ integrations by parts on a vertical line, leaving the stated integrable
decay.  Polynomial Laplace transforms are exactly instantaneous distributions,
so they cannot be included in an ordinary exponential-decay claim.
\end{proof}

\begin{theorem}[Exact Volterra equation and exponential regular memory]
\label{thm:r9-a4-memory}
The resolved correlation $C(t)=Pe^{tL}P$ satisfies, in causal distributions,
\[
 C'(t)=PLP\,C(t)+(K*C)(t),
 \qquad C(0)=P,
\]
with
\[
 \widehat K(z)=zP-PLP-C(z)^{-1}.
\]
After the instantaneous part of
Lemma~\ref{lem:r9-a4-instant} is removed,
\[
 \|K_{\rm reg}(t)\|\le Ce^{-\gamma t}.
\]
\end{theorem}

\begin{proof}
Laplace transformation of the Volterra equation gives
$zC(z)-P=PLP\,C(z)+\widehat K(z)C(z)$, which is equivalent to the displayed
Schur formula.  For the regular part, shift the Bromwich contour to
$\Re z=-\gamma$; the integrable vertical bound from
Lemma~\ref{lem:r9-a4-instant} justifies the shift and yields the exponential
factor.
\end{proof}

\subsection{Finite meromorphic modes without passivity}

Suppose the finite matrix function $\widehat K(z)$ has finitely many poles in
the strip after the instantaneous polynomial is removed.  Its principal part
is rational.

\begin{theorem}[Descriptor realization of finite principal parts]
\label{thm:r9-a4-descriptor}
There are a finite-dimensional space $Z$ and a regular matrix pencil
$zE_Z-A_Z$ such that
\[
 \widehat K(z)=\mathcal P_M(z)
 +C_Z(zE_Z-A_Z)^{-1}B_Z
 +\widehat K_{\rm dec}(z),
\]
where $\widehat K_{\rm dec}$ is analytic with integrable vertical decay in the
strip.  The complete state realization is
\[
 P\mathcal H\oplus Q\mathcal H\oplus Z;
\]
the $Q$-space is never replaced by $Z$.  No positivity, passivity, or claim
that a transmission zero is an eigenmode of $QLQ$ is used.
\end{theorem}

\begin{proof}
Write each matrix principal part in partial fractions and realize every
Jordan term $(z-z_j)^{-k}M_{j,k}$ by a finite companion block in a descriptor
pencil.  Their direct sum gives $Z,E_Z,A_Z,B_Z,C_Z$.  Subtracting the exact
principal part leaves an analytic function with the vertical decay of
Theorem~\ref{thm:r9-a4-memory}.  Adjoining this finite input--output system to
the already exact block realization on $P\mathcal H\oplus Q\mathcal H$ gives
the stated descriptor realization.  It is an operator identity, not a unitary
microscopic decomposition.
\end{proof}

\subsection{Fixed-phase nonlinear history pressure}

For a bounded history source $V$, the Feynman--Kac semigroup on $C_0^W$ is
\[
 P_t^VF(h)=E_h\left[e^{\int_0^tV(H_s)ds}F(H_t)\right].
\]
Define $\mathcal E_t^VF=\log P_t^V(e^F)-\log P_t^V1$.

\begin{theorem}[History pressure and rough tangent]
\label{thm:r9-a4-pressure}
The family $(\mathcal E_t^V)$ is a strongly continuous nonlinear semigroup on
bounded subsets of $C_0^W$.  Its cylinder generator is the exponential
transform of the past-kernel generator.  Under diffusive scaling it converges,
conditionally on compact pasts, to the risk-sensitive diffusion semigroup
driven by Theorem~\ref{thm:r9-a4-quenched}, including the area anomaly.
The fixed-phase memory is the one in
Theorems~\ref{thm:r9-a4-memory}--\ref{thm:r9-a4-descriptor}; no generic Doob
conjugacy invariance is asserted.
\end{theorem}

\begin{proof}
The weighted Feller theorem gives the linear Feynman--Kac tower and strong
continuity; exponentiation gives the nonlinear tower.  Differentiating on the
cylinder core yields the exponential generator.  Uniform conditional rough
convergence and exponential moments pass the Feynman--Kac functionals to the
rough diffusion.  The memory statement is fixed in one Hilbert phase, so its
pairing and resolved projection do not change.
\end{proof}
