
\section{Round-fourteen positive closure: positive disintegrations, blown-up evidence, and growing belief coordinates}
\label{sec:r14-c1}

Posterior probabilities and geometric currents are different objects.
Observations are represented by a positive joint kernel and Rokhlin
disintegration; currents are used only to calculate regular coarea densities.
The exact information state is the full unnormalized belief.  At zero evidence
we use the real oriented blow-up of the positive cone and do not identify
different limiting directions.

\subsection{Positive observation disintegration}

Let \(E\) be the compact weighted microstate/trajectory set selected by the
B2 containment function, and let \(\mu\in\mathcal P(E)\).  A control \(u\)
chooses a positive observation kernel \(L_u(x,dy)\) on a Polish space \(Y\).
The joint prediction-observation measure is
\[
 J_{\mu,u}(dx,dy)=\mu(dx)L_u(x,dy).
\]
Let \(\nu_{\mu,u}\) be its \(Y\)-marginal and
\(\mu_{u,y}\) a Rokhlin conditional law:
\[
 J_{\mu,u}(dx,dy)=\mu_{u,y}(dx)\nu_{\mu,u}(dy).
\]

\begin{theorem}[Positive posterior kernel]
\label{thm:r14-c1-posterior}
The map \((\mu,u)\mapsto J_{\mu,u}\) is positive and affine in \(\mu\).
For every regular noiseless observation
\(O_u:E\to\mathbb R^d\) which is a submersion on a finite Whitney
stratification, \(\nu_{\mu,u}\) and \(\mu_{u,y}\) have the coarea
representation
\[
 \nu_{\mu,u}(dy)
 =
 \sum_S\left[
 \int_{S\cap O_u^{-1}(y)}
 {f_S(x)\over J_SO_u(x)}\,d\mathcal H^{\dim S-d}(x)
 \right]dy,
\]
and \(\mu_{u,y}\) is the normalized positive fiber measure.  Oriented normal
currents may encode the fiber geometry but are never paired with the scalar
\(1\) to define evidence.
\end{theorem}

\begin{proof}
Existence of the disintegration is Rokhlin's theorem on standard Borel
spaces.  Positivity and affinity are immediate from the joint kernel.
On a smooth stratum, apply the coarea formula to the positive density
\(f_S\,d\mathcal H^{\dim S}\).  Lower-rank strata are included as separate
positive components of the joint measure, with their own Hausdorff dimension.
The evidence is the mass of these positive coefficient measures, not the mass
or orientation of a current.
\end{proof}

\subsection{The blown-up zero-evidence cone}

Let \(\mathcal K_M\subset\mathcal P(E)\) be a compact moment sublevel.  Define
\[
 \operatorname{Bl}_0\mathcal M_+(\mathcal K_M)
 =
 \bigl((0,1]\times\mathcal K_M\bigr)
 \cup\bigl(\{0\}\times\mathcal K_M\bigr).
\]
The point \((r,\nu)\) represents the finite measure \(r\nu\).  At \(r=0\),
different directions \(\nu\) remain different.  Give this space the product
topology.

\begin{lemma}[Compact positive evidence blow-up]
\label{lem:r14-c1-blowup}
The blown-up cone is compact.  The collapse map
\((r,\nu)\mapsto r\nu\) is continuous but not injective at zero.  A flat
family with oscillating normalized direction has convergent subsequences in
the blow-up precisely when its directions do; no fictitious canonical
direction is assigned.
\end{lemma}

\begin{proof}
It is the compact product \([0,1]\times\mathcal K_M\).  The remaining
statements follow directly from the definition.  In particular the family
\(e^{-1/t^2}\nu_t\) is represented by
\((e^{-1/t^2},\nu_t)\), so its oscillation is visible rather than collapsed.
\end{proof}

Prediction and observation are positively homogeneous on finite measures.
For \(\Lambda=r\nu\), let \(\mathsf P_u\Lambda\) be the predicted finite
measure and let \(\mathsf O_{u,y}\Lambda\) be its positive fiber coefficient.

\begin{theorem}[Feller belief transition]
\label{thm:r14-c1-feller}
Assume the prediction kernel is Wasserstein--Lipschitz and the regular
observation densities satisfy, on \(\mathcal K_M\),
\[
 \|g_{\nu,u}-g_{\tilde\nu,u}\|_{L^1(Y)}
 +\int g_{\nu,u}(y)
   W_1(\nu_{u,y},\tilde\nu_{u,y})\,dy
 \le L W_1(\nu,\tilde\nu).
\]
Then the prediction--observation transition extends continuously to the
blown-up cone:
\[
 (r,\nu)\longmapsto
 \int \delta_{(r g_{\nu,u}(y),\,\nu_{u,y})}\,dy .
\]
At zero evidence the direction evolves as an explicit state coordinate; it
is not required to be determined by the zero measure alone.
\end{theorem}

\begin{proof}
Positive homogeneity gives the new mass \(rg_{\nu,u}(y)\).
The displayed estimate controls both evidence mass and normalized posterior
in the product Wasserstein metric.  Dominated convergence proves continuity
for \(r>0\).  If \(r_n\to0\) and \(\nu_n\to\nu\), the mass coordinate tends to
zero while the direction transition converges by the same estimate.  Thus the
extension is Feller on the product blow-up.
\end{proof}

For the hard-sphere observation family, the estimate follows from the B1
inserted coefficient theorem below; for the Sinai family it follows from the
A2 insertion LLT.

\subsection{Exact full-belief dynamic programming}

Let \(b=(r,\nu)\) be the blown-up unnormalized belief.  Prediction under a
control and subsequent observation use the kernel of
Theorem~\ref{thm:r14-c1-feller}.  Let running rewards be bounded and
continuous.

\begin{theorem}[Belief-state game]
\label{thm:r14-c1-game}
For finite or continuous time, the exact partially observed risk-sensitive
game is Markov on the complete state \((\text{physical state},b)\).  Its
Bellman operator is Feller, monotone, and cash additive.  One-time
microcanonical preparation, reward-only feedback, and adaptive canonical-law
control remain three distinct games.
\end{theorem}

\begin{proof}
Ionescu--Tulcea constructs the controlled joint law from the prediction and
observation kernels.  Conditional expectation of every future payoff depends
on the observation history only through the current unnormalized posterior,
including its zero-mass direction.  Composition of the Feller kernels gives
the Bellman recursion.  Preparation changes the initial belief once; a
reward control changes only the running payoff; a canonical-law control
changes the transition likelihood.  Their strategy spaces and timing are
therefore not interchangeable.
\end{proof}

\subsection{Inserted local coefficients}

Let \(Y^\varepsilon\) be a finite vector of additive particle-path and
actual-contact observations.  Insert a bounded central belief test \(G\) in
the B1 exact-number coefficient.

\begin{theorem}[Uniform inserted coefficient]
\label{thm:r14-c1-inserted}
On a compact regular source, control, and observation chart,
\[
 \mathbb E\left[
 G;\sqrt{\mu_\varepsilon}
 (Y^\varepsilon-y)\in W_\varepsilon
 \right]
 =
 c_\varepsilon(y)\,
 \mathbb E_{y}^{\rm tilt}G\,(1+o(1)),
\]
where \(c_\varepsilon(y)\) is the B1 shell coefficient and the error is
uniform for \(G\) in the unit ball of a fixed finite-history Lipschitz class.
The denominator has the same coefficient, so the posterior ratio converges
to the tilted conditional law.
\end{theorem}

\begin{proof}
Adjoin the observation source to the coefficient-level trajectory polymer
gas of B1.  Normal convergence holds uniformly over the compact control
chart.  The central spectral coefficient with insertion is obtained by
differentiating the same exact-\(N\) Cauchy integral; the left and right
coefficient eigenvectors pair with \(G\).  The compact annulus and
speed-dependent high-frequency estimates are unchanged because \(G\) affects
only finitely many history coordinates.  Fourier inversion gives the same
Gaussian coefficient times the tilted expectation of \(G\).
\end{proof}

This theorem proves the quantitative observation estimate assumed in
Theorem~\ref{thm:r14-c1-feller} by duality over bounded Lipschitz tests.

\subsection{Growing finite belief coordinates}

A fixed finite list of moments does not determine a probability law.  Let
\(\mathcal K_M\) be compact in \(W_1\).  For \(\epsilon>0\), choose a finite
\(\epsilon\)-net \(\{\phi_1,\ldots,\phi_{K(\epsilon)}\}\) in the unit ball of
bounded Lipschitz functions restricted to \(\mathcal K_M\), and define
\[
 C_\epsilon(\nu)
 =(\nu(\phi_1),\ldots,\nu(\phi_{K(\epsilon)})).
\]
Choose a measurable nearest-point reconstruction
\(R_\epsilon(C_\epsilon\nu)\in\mathcal K_M\).

\begin{lemma}[Quantitative growing-coordinate reconstruction]
\label{lem:r14-c1-coordinates}
Uniformly on \(\mathcal K_M\),
\[
 W_1\bigl(\nu,R_\epsilon C_\epsilon\nu\bigr)\le3\epsilon .
\]
If the belief kernel has Lipschitz constant \(L\), the \(N\)-stage value
computed with reconstructed beliefs differs from the full-belief value by at
most
\[
 3\epsilon\,L_V\sum_{j=0}^{N-1}L^j .
\]
Thus choosing \(\epsilon=\epsilon_\varepsilon\downarrow0\), equivalently
\(K_\varepsilon\to\infty\), gives a uniform \(o(1)\) reduction on every
fixed horizon.
\end{lemma}

\begin{proof}
By Kantorovich duality, \(W_1\) is the supremum over the bounded-Lipschitz unit
ball.  Approximate the maximizing test by one of the finite net functions and
use the nearest-coordinate reconstruction; the three approximation errors
give \(3\epsilon\).  Couple the exact and reconstructed belief chains at each
step.  The kernel Lipschitz estimate propagates the current error by \(L\);
the value Lipschitz constant \(L_V\) and telescoping give the displayed sum.
This constructs a reduced Markov kernel by pushforward through
\(C_\epsilon\) and \(R_\epsilon\), rather than treating moment proximity as
an exact state.
\end{proof}

\subsection{A fully specified local statistical limit}

Let \(\theta\in\Theta\subset\mathbb R^d\) parametrize a regular prepared
source.  Assume a prior density positive and \(C^1\) near the true
\(\theta_0\), uniform identifiability away from \(\theta_0\), and the exact
finite-centred LAN expansion
\[
 \log{dP_{\theta_0+h/\sqrt\mu}^\varepsilon
          \over dP_{\theta_0}^\varepsilon}
 =
 h\cdot Z_\varepsilon
 -\tfrac12h^TI_{\theta_0}h+o_{P_{\theta_0}}(1),
\]
uniformly for \(h\) in compact sets.  Here \(Z_\varepsilon\) is the B3 joint
density/contact score and \(I_{\theta_0}>0\).

\begin{theorem}[Controlled Bernstein--von Mises theorem]
\label{thm:r14-c1-bvm}
On every compact informative strategy class for which the inserted
coefficient and LAN expansion are uniform, the posterior law of
\(\sqrt{\mu_\varepsilon}(\theta-\theta_0)\) converges in total variation to
\[
 N\!\left(I_{\theta_0}^{-1}Z,\ I_{\theta_0}^{-1}\right).
\]
The convergence is uniform over the strategy class.
\end{theorem}

\begin{proof}
The inserted coefficient gives a common local likelihood normalization and
the B3 process theorem gives \(Z_\varepsilon\Rightarrow
N(0,I_{\theta_0})\).  Uniform identifiability makes the posterior outside a
fixed local neighborhood exponentially small.  On compact \(h\)-sets,
uniform LAN and the \(C^1\) prior reduce numerator and denominator to the
Gaussian integral.  Tail tightness follows from the positive information
matrix.  Scheff\'e's lemma gives total variation convergence uniformly in the
strategy.
\end{proof}

\begin{theorem}[Round-fourteen C1 closure]
\label{thm:r14-c1-main}
The exact information state is a positive disintegrated full belief on a
compact blown-up evidence cone; the belief game is Feller; local observations
have an inserted coefficient theorem; growing finite coordinates approximate
the full game quantitatively; and the regular controlled statistical model
satisfies the stated Bernstein--von Mises limit.
\end{theorem}

\begin{proof}
Combine Theorems~\ref{thm:r14-c1-posterior},
\ref{thm:r14-c1-feller}, \ref{thm:r14-c1-game},
\ref{thm:r14-c1-inserted}, and \ref{thm:r14-c1-bvm}, with
Lemma~\ref{lem:r14-c1-coordinates}.
\end{proof}
