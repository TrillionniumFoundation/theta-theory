\section{Round-eleven positive closure: centered Poisson theory and renewal memory}
\label{sec:r11-a4}

Let $P$ be the Markov kernel of the genuine past-history chain under an
exposed phase, and let $g$ be centered.  The previous noncentered increment is
replaced completely.

\begin{theorem}[Correct Poisson martingale decomposition]
\label{thm:r11-a4-poisson}
Assume the spectral gap of the past kernel on the weighted Holder space and
put
\[
 h=\sum_{j\ge0}P^jg,
 \qquad (I-P)h=g.
\]
Then
\[
 D_{k+1}=h(H_{k+1})-Ph(H_k)
\]
is a martingale difference and
\[
 \sum_{k=0}^{n-1}g(H_k)
 =\sum_{k=0}^{n-1}D_{k+1}+h(H_0)-h(H_n).
\]
The same identities hold vectorwise and in the weighted norm.
\end{theorem}

\begin{proof}
Conditional expectation gives
$\mathbb E[D_{k+1}\mid\mathcal F_k]=Ph(H_k)-Ph(H_k)=0$.
Since $g=h-Ph$,
\[
 g(H_k)=h(H_k)-h(H_{k+1})+D_{k+1},
\]
and summation telescopes.  The spectral gap makes the Poisson series converge
with uniform weighted bounds.
\end{proof}

\begin{theorem}[Weighted Feller and quenched enhanced limit]
\label{thm:r11-a4-quenched}
The past kernel satisfies a Lyapunov estimate $PW\le\rho W+C$ and is Feller
on $C_0^W$.  For every compact set of admissible pasts, the suspension
martingale, its bracket, roof interpolation, and second iterated integral
converge uniformly in the initial past to a Brownian rough path with the
explicit Green--Kubo covariance and area anomaly.
\end{theorem}

\begin{proof}
The A3 recurrent--recession rate gives the Lyapunov drift and excludes escape
on compact rate sublevels.  Couple two histories until their first common
return; the return tail and singularity shield give the Feller modulus.
Theorem~\ref{thm:r11-a4-poisson} reduces the additive functional to a
martingale plus a uniformly negligible coboundary.  Conditional bracket
convergence follows from the same coupling and the ergodic theorem applied to
the bracket observable; the estimates are uniform on compact history sets.
Martingale rough-path tightness and deterministic roof inversion finish the
proof.
\end{proof}

\subsection{A genuine nonlinear semigroup}
Let $T_t^V$ be the positive Feynman--Kac semigroup with leading eigenpair
$(e^{t\lambda_V},h_V)$.  Define the Doob Markov semigroup
\[
 P_t^V F=h_V^{-1}e^{-t\lambda_V}T_t^V(h_VF).
\]

\begin{theorem}[Doob log semigroup]
\label{thm:r11-a4-doob}
The nonlinear operators
\[
 \mathcal E_t^V F=\log P_t^V(e^F)
\]
are monotone, cash additive, strongly continuous on bounded weighted
cylinders, and satisfy
$\mathcal E_{s+t}^V=\mathcal E_s^V\mathcal E_t^V$.
Their generator is the exponential transform of the closed Doob generator.
\end{theorem}

\begin{proof}
The eigenfunction conjugacy makes $P_t^V1=1$ and gives the ordinary Markov
semigroup law.  Exponentiation and logarithm therefore preserve composition;
no statewise division by $T_t^V1$ occurs.  Strong continuity and the
generator formula follow from the weighted Feller theorem and the closed
Doob generator.
\end{proof}

\subsection{From the roof transfer operator to continuous-time memory}
For resolved observables let $\mathcal C(z)=P(z-L_V)^{-1}P$.

\begin{theorem}[Renewal--resolvent transference]
\label{thm:r11-a4-resolvent}
The A2 vector--roof operator identity gives a meromorphic continuation of each
resolved correlation entry to a common strip.  In that strip the resolvent
has explicit vertical bounds after subtracting finitely many polynomial and
resonant principal parts.  Transmission zeros are recorded as finite
descriptor modes.  On the remaining minimum-phase block, $\mathcal C(z)$ is
invertible and its inverse has the same vertical bounds.
\end{theorem}

\begin{proof}
Laplace-transform the suspension renewal equation.  Its denominator is the
A2 roof-twisted transfer operator, while the numerator consists of bounded
entrance and residual-flight operators.  The four Fourier ranges of A2 give
polynomial vertical estimates.  Riesz residues account for poles; a finite
Smith--McMillan factorization on the resolved matrix accounts for zeros.
After adjoining those finite modes, the residual determinant is bounded away
from zero on a slightly smaller strip.
\end{proof}

\begin{theorem}[Exact causal memory and decay]
\label{thm:r11-a4-memory}
With the descriptor modes included, the exact resolved transfer function has
memory transform
\[
 \widehat K_V(z)=zP-PL_VP-\mathcal C(z)^{-1}.
\]
It decomposes uniquely into instantaneous distributions, finitely many
resonant exponentials, and an ordinary kernel satisfying
\[
 \|K_V^{\rm reg}(t)\|\le Ce^{-\gamma t},
 \qquad
 \int_0^\infty(1+t)\|K_V^{\rm reg}(t)\|dt<\infty.
\]
The projected observable satisfies the corresponding causal Volterra equation.
\end{theorem}

\begin{proof}
The large-$z$ expansion separates every polynomial term, including possible
Dirac derivatives at zero.  Theorem~\ref{thm:r11-a4-resolvent} permits a
Bromwich contour shift for the regular remainder and gives the exponential
bound.  Laplace inversion of the Schur identity yields the exact Volterra
equation with the displayed sign.
\end{proof}

\begin{theorem}[Round-eleven A4 closure]
\label{thm:r11-a4-main}
Prepared Sinai histories possess a correctly centered martingale--coboundary
decomposition, a locally uniform quenched enhanced limit, a genuine Doob
nonlinear semigroup, and a billiard-specific renewal-resolvent memory theorem
with instantaneous terms and transmission zeros explicitly separated.
\end{theorem}

\begin{proof}
Combine Theorems~\ref{thm:r11-a4-poisson},
\ref{thm:r11-a4-quenched}, \ref{thm:r11-a4-doob},
\ref{thm:r11-a4-resolvent}, and \ref{thm:r11-a4-memory}.
\end{proof}
