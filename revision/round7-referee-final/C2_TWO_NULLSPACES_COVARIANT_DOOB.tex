\section{Round-seven reconstruction: two null spaces, coarse-history optional projections, and a covariant Doob connection}
\label{sec:r7-c2}

This section replaces the round-six controlling module.  Equality of invariant
integrals is separated from pressure equivalence modulo additive constants.
Invariant separators are obtained by an explicit Folner averaging theorem on a
weighted Radon dual.  Optional projections use the coarse A4 filtration.  When
the source changes the Doob data, differentiation is covariant and includes
the eigenfunction, invariant measure, inner product, and projection.

\subsection{Two equivalence relations}

Let $\mathscr A_W=C_{W,0}\cap C^\eta_{\rm dyn}$ be the weighted dynamically
Holder algebra.  Define
\[
 \mathscr N_{\rm int}=
 \overline{\{U-U\circ\Theta_t:U\in\mathscr A_W,\ t\ge0\}},
\]
\[
 \mathscr N_{\rm press}=
 \mathbb R\mathbf1+\mathscr N_{\rm int}.
\]
Constants are absent from $\mathscr N_{\rm int}$ and present in
$\mathscr N_{\rm press}$.

\begin{theorem}[Integral and pressure cotangents]
\label{thm:r7-c2-cotangent}
For $F,G\in\mathscr A_W$ the following are equivalent:
\begin{enumerate}[label=(\roman*)]
\item $\nu(F)=\nu(G)$ for every invariant probability with finite $W$ moment;
\item $F-G\in\mathscr N_{\rm int}$.
\end{enumerate}
Moreover $F$ and $G$ define the same normalized equilibrium phases and the
same pressure differences if and only if
$F-G\in\mathscr N_{\rm press}$.  If $G=F+c$, then
$P(G)=P(F)+c$ and invariant integrals differ by $c$.
\end{theorem}

\begin{proof}
Every coboundary has zero invariant integral, proving one inclusion.  For the
converse suppose $F-G$ is not in $\mathscr N_{\rm int}$.  Hahn--Banach gives a
continuous functional on $C_{W,0}$ annihilating the coboundary closure and not
$F-G$.  The weighted Riesz theorem represents it by a finite signed Radon
measure.  Lemma~\ref{lem:r7-c2-average} below makes it invariant without
changing the pairing with $F-G$.  Jordan decomposition and normalization then
produce invariant probabilities separating the two integrals.  Adding
constants gives the pressure statement and explicitly accounts for the
constant shift.
\end{proof}

\subsection{Invariant averaging in the weighted Radon dual}

Let $\mu$ be a finite signed measure with $\int Wd|\mu|<\infty$.  For the
continuous-time action define the Cesaro averages
\[
 \mu_T=\frac1T\int_0^T(\Theta_t)_\#\mu\,dt.
\]
Assume the platform rate--moment containment
\[
 \sup_{0\le t\le T}\int W\circ\Theta_t\,d|\mu|
 \le C_T\int(1+W)d|\mu|
\]
and the compact-sublevel property of $W$.

\begin{lemma}[Weighted Folner averaging]
\label{lem:r7-c2-average}
The family $\mu_T$ is weak-star relatively compact on every bounded
rate--moment set.  Every limit is invariant.  If a functional annihilates all
coboundaries, then its Cesaro averages have the same value on every
$F\in\mathscr A_W$.  Hence a separating functional has an invariant weighted
Radon separator.
\end{lemma}

\begin{proof}
The moment estimate and compact sublevels give weighted tightness.  For fixed
$s$,
\[
 (\Theta_s)_\#\mu_T-\mu_T
 =T^{-1}\left(\int_T^{T+s}-\int_0^s\right)(\Theta_t)_\#\mu\,dt,
\]
which tends to zero weak-star.  Thus every limit is invariant.  If the original
functional vanishes on $F-F\circ\Theta_t$, its pairing with every translate of
$F$ equals its pairing with $F$, so averaging does not alter that pairing.
\end{proof}

For Sinai platforms the rate--moment estimate follows from A3's canonical
compactification; for hard spheres it follows from B2's conserved Maxwellian
moment and contact entropy.  These are proved inputs rather than hypotheses
inserted only in C2.

\subsection{Exact finite likelihoods on the coarse filtration}

Let $\mathcal F_t^s$ be the coarse A4 history filtration and let $\Phi$ be a
bounded terminal functional.  Define
\[
 M_t^\varepsilon=
 \frac{\mathbb E[\exp\{\theta\Phi(X_T^\varepsilon)\}
      \mid\mathcal F_t^s]}
      {\mathbb E\exp\{\theta\Phi(X_T^\varepsilon)\}}.
\]

\begin{lemma}[Exact history likelihood]
\label{lem:r7-c2-likelihood}
The process $M_t^\varepsilon$ is a positive mean-one martingale.  Its kernel
representation uses the A4 coarse-history transition kernel.  If one instead
conditions on the full deterministic microscopic history, the conditional
expectation is the terminal functional evaluated on the unique future and no
nondegenerate diffusion-kernel statement is made.
\end{lemma}

\begin{proof}
This is the tower property of conditional expectation.  The distinction of
filtrations is part of the definition, not an after-the-fact reinterpretation.
\end{proof}

\begin{theorem}[Optional-projection convergence]
\label{thm:r7-c2-optional}
Under the A4 compact coarse-history conditional invariance principle,
$M^\varepsilon$ converges in path law and stably with respect to bounded
initial coarse-history variables to
\[
 M_t=
 \frac{\mathbb E[\exp\{\theta\Phi(X_T)\}
      \mid\mathcal F_t^X]}
      {\mathbb E\exp\{\theta\Phi(X_T)\}}.
\]
The convergence includes the filtrations through convergence of the conditional
kernels; ordinary weak path convergence alone is not used to pass conditional
expectations.
\end{theorem}

\begin{proof}
Write the finite optional projection as the integral of the terminal
functional against the A4 coarse-history kernel.  Uniform kernel convergence
on compact histories gives convergence there.  Exponential containment removes
the complement.  Multiplication by a bounded initial-history variable is
absorbed in the same kernel integral, proving stable convergence.  The limiting
kernel is the diffusion kernel and hence the displayed optional projection.
\end{proof}

\subsection{Fixed-Doob memory and its covariant source derivative}

For a source parameter $\eta$, let
$h_\eta$, $\lambda_\eta$, and $\nu_\eta^D$ be the Doob eigenfunction,
eigenvalue, and invariant probability.  Let
$\mathcal H_\eta=L^2(\nu_\eta^D)$ and let $P_\eta$ be the orthogonal projection
onto the resolved subspace transported by multiplication with $h_\eta$.
The family $\{\mathcal H_\eta\}$ is a Hilbert bundle.  Define its connection by
\[
 \nabla_\eta u
 =\partial_\eta u+rac12
  \partial_\eta\log\frac{d\nu_\eta^D}{d\nu_0^D}\,u
 -h_\eta^{-1}(\partial_\eta h_\eta)u.
\]
This connection differentiates both the inner product and the Doob
conjugation.  The projection derivative is the covariant derivative
$\nabla_\eta P_\eta$ determined by $P_\eta^2=P_\eta$ and orthogonality.

For fixed $\eta$ define
\[
 C_\eta(z)=P_\eta(z-L_\eta^D)^{-1}P_\eta,
 \qquad
 \widehat K_\eta(z)=C_\eta(z)^{-1}-zP_\eta-A_\eta.
\]
The A4 Riesz--Schur dilation supplies the finite zero modes and a zero-free
residual inverse on its domain.

\begin{theorem}[Covariant Doob-memory response]
\label{thm:r7-c2-memory}
For a bounded source insertion $V_\eta$, the covariant derivative of the Doob
semigroup is
\[
 \nabla_\eta e^{tL_\eta^D}
 =\int_0^t e^{(t-s)L_\eta^D}
  (\nabla_\eta L_\eta^D)e^{sL_\eta^D}ds,
\]
where $\nabla_\eta L_\eta^D$ contains the derivatives of the eigenvalue and
eigenfunction.  The derivative of the compressed resolvent and memory also
contains $\nabla_\eta P_\eta$ and the derivative of the Hilbert pairing.  With
these terms included, pressure differentiation, Doob response, and memory
compression commute as covariant maps of the Hilbert bundle.
\end{theorem}

\begin{proof}
Trivialize the Hilbert bundle by the unitary density square root and the Doob
multiplication.  In that fixed Hilbert space ordinary Duhamel differentiation
applies.  Transforming back produces the connection terms written above.
Differentiate $P_\eta^2=P_\eta$ and the orthogonality relation to obtain
$\nabla_\eta P_\eta$.  The resolvent identity then gives
\[
 \nabla C=P( z-L)^{-1}(\nabla L)(z-L)^{-1}P
 +(\nabla P)(z-L)^{-1}P+P(z-L)^{-1}(\nabla P).
\]
Differentiating the inverse and the Riesz--Schur block formula proves the
memory statement.
\end{proof}

\subsection{Girsanov and the limiting filtration}

Assume the limiting diffusion has bounded $C^3$ coefficients and uniformly
nondegenerate covariance on the resolved range.  Let $u$ solve the backward
Kolmogorov equation for the terminal exponential payoff.

\begin{theorem}[Microscopic likelihood to diffusion Girsanov]
\label{thm:r7-c2-girsanov}
The limit in Theorem~\ref{thm:r7-c2-optional} is
\[
 M_t=\exp\left\{
 \int_0^t\sigma^T\nabla\log u(s,X_s)dW_s
 -\frac12\int_0^t|\sigma^T\nabla\log u|^2ds\right\}.
\]
Under the changed law, the diffusion drift receives the usual
$\sigma\sigma^T\nabla\log u$ correction.  The associated logarithmic value
solves the risk-sensitive BSDE.  This conclusion is tied to the proved coarse
filtration convergence.
\end{theorem}

\begin{proof}
The classical solution is bounded away from zero by the bounded terminal
payoff.  Ito's formula gives the stochastic exponential.  Uniform integrability
comes from bounded gradient on the finite horizon.  Girsanov's theorem gives
the drift and the logarithmic transform gives the BSDE.
\end{proof}

\subsection{Linear and nonlinear contractions}

For a continuous linear contraction $C$, the pulled-back source is $C^*\eta$.
For a $C^2$ nonlinear contraction $\mathcal C$, use the first and second
Frechet derivatives:
\[
 D(Q\circ\mathcal C^*)=DQ\,D\mathcal C^*,
\]
\[
 D^2(Q\circ\mathcal C^*)=
 D\mathcal C\,D^2Q\,D\mathcal C^*
 +DQ\,D^2\mathcal C^*.
\]
The second term is retained.

\begin{theorem}[Typed contraction functor]
\label{thm:r7-c2-contraction}
Pressure, rate, and covariance commute with continuous linear contractions on
their proved source charts.  They commute with nonlinear $C^2$ contractions
through the full chain rule above whenever the pullback remains in the local
analytic domain and the map is exponentially good on rate sublevels.  No
adjoint is assigned to a nonlinear map.
\end{theorem}

\begin{proof}
The linear statement is the exact finite-volume identity followed by the
contraction principle and source derivative convergence.  The nonlinear
statement follows from exponential goodness, the ordinary Frechet chain rule,
and the displayed second derivative.  Domain preservation is an explicit
hypothesis verified for the admitted platform maps.
\end{proof}

\begin{theorem}[Round-seven C2 closure]
\label{thm:r7-c2-main}
Invariant-integral equality is quotiented only by coboundaries, while pressure
equivalence also quotients constants; invariant separators are weighted Radon
measures obtained by Folner averaging; finite likelihoods and optional
projections use the A4 coarse filtration; source-dependent memory response is
covariant in the Doob Hilbert bundle; and nonlinear contractions use the full
Frechet chain rule.
\end{theorem}

\begin{proof}
Combine Theorems~\ref{thm:r7-c2-cotangent},
\ref{thm:r7-c2-optional},
\ref{thm:r7-c2-memory},
\ref{thm:r7-c2-girsanov}, and
\ref{thm:r7-c2-contraction}, together with
Lemmas~\ref{lem:r7-c2-average} and
\ref{lem:r7-c2-likelihood}.
\end{proof}
