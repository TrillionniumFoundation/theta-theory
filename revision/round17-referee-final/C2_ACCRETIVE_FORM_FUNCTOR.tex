\section{Round-seventeen positive closure: strict duals, covariant form response, and optional-projection limits}
\label{sec:r17-c2}

The three topologies used in this paper are kept distinct.  The weighted
strict topology has countably additive signed Radon measures as its dual.  The
strong anisotropic/graph topology carries spectral response distributions.
Memory is defined on a fixed operator core or, more generally, through a
closed coercive form.  Parameter derivatives are taken after transport to a
fixed reference realization and include the metric connection.

\subsection{Weighted strict duality}

Let $E$ be a Polish path space, $W\ge1$ continuous with compact sublevels, and
\[
 C_W(E)=\{F\in C(E):F/W\text{ vanishes at weighted infinity}\}.
\]
Give $C_W(E)$ the weighted strict topology $\beta_W$: a net $F_i$ converges
when it is uniformly bounded in $\|F/W\|_\infty$ and converges uniformly on
every $W$-compact set after division by $W$.  Let
$\mathcal M_W(E)$ be the signed Radon measures with
$\int Wd|\mu|<\infty$.

\begin{theorem}[Exact strict dual and invariant annihilator]
\label{thm:r17-c2-dual}
The continuous dual of $(C_W(E),\beta_W)$ is $\mathcal M_W(E)$ under
integration.  If $\Theta_t$ is a continuous weighted flow, then the
annihilator of the closed linear span
\[
 \mathscr N=\overline{\operatorname{span}}^{\beta_W}
 \{F-F\circ\Theta_t:F\in C_W(E),t\in\mathbb Q_+\}
\]
is exactly the space of invariant signed Radon measures in
$\mathcal M_W(E)$.  Constants are not in $\mathscr N$.
\end{theorem}

\begin{proof}
A $\beta_W$-continuous functional is bounded on one set
$\{F:\|F/W\|_\infty\le1,\operatorname{supp}F\subset K\}$
for every compact $K$.  Riesz representation gives compatible signed Radon
measures on an increasing compact exhaustion.  Continuity on the bounded set
forces uniform summability of the weighted variations, producing one
$\mu\in\mathcal M_W(E)$.  Conversely, weighted tightness and dominated
convergence make integration continuous for $\beta_W$.  Invariance plainly
annihilates every coboundary.  Conversely, if a signed Radon functional
annihilates the rational-time coboundaries, continuity of the flow extends the
identity to all times, hence the measure is invariant.  A nonzero constant
has nonzero integral under a probability and therefore cannot lie in the
closed coboundary span.
\end{proof}

\subsection{Transported Hilbert and form bundles}

Fix one platform $\alpha$.  Let $H_\eta$ be the physical $L^2$ realization of
its prepared phase, with inner product
$(u,v)_\eta=(G_\eta u,v)_0$ on a fixed vector space $H_0$, where $G_\eta$ is a
positive boundedly invertible $C^2$ family.  Put
$T_\eta=G_\eta^{1/2}:H_\eta\to H_0$ and transport the generator and projection:
\[
 \widetilde L_\eta=T_\eta L_\eta T_\eta^{-1},
 \qquad
 \widetilde P_\eta=T_\eta P_\eta T_\eta^{-1}.
\]
The connection is
$\nabla_\eta u=\partial_\eta u+\frac12G_\eta^{-1}
(\partial_\eta G_\eta)u$ in the untransported coordinates.

Let $V\subset H_0$ be a common dense form domain and
$a_\eta:V\times V\to\mathbb C$ a closed sectorial form satisfying
\[
 \Re a_\eta(u,u)+c\|u\|_{H_0}^2\ge c_0\|u\|_V^2,
\]
with two bounded parameter derivatives as forms $V\to V^*$.

\begin{theorem}[Covariant type-(B) resolvent response]
\label{thm:r17-c2-connection}
The transported operators form a Kato type-(B) family.  For
$\Re z>z_0$,
\[
 D_\eta(z-\widetilde L_\eta)^{-1}
 =(z-\widetilde L_\eta)^{-1}
 (D_\eta\widetilde L_\eta)
 (z-\widetilde L_\eta)^{-1}
\]
as an identity $V^*\to V$, where
$D_\eta\widetilde L_\eta$ includes the derivative of
$T_\eta$, and hence the metric connection.  In untransported coordinates
this is the covariant derivative formula; no Hilbert-metric term is omitted.
\end{theorem}

\begin{proof}
The uniform coercive estimate gives an isomorphism
$z+A_\eta:V\to V^*$ by Lax--Milgram for $\Re z$ sufficiently large.
The form derivatives are bounded in $\mathcal L(V,V^*)$, so the inverse map is
$C^2$ and differentiation of
$(z+A_\eta)R_\eta(z)=I$ yields the formula.  Since
$T_\eta$ and $T_\eta^{-1}$ are $C^2$, differentiating the transported
generator explicitly produces the commutator with
$(\partial_\eta T_\eta)T_\eta^{-1}$, which is exactly the
metric-connection term.
\end{proof}

\subsection{Accretive compressed resolvents and form-level memory}

Let $\mathcal R\subset V$ be finite dimensional.  If
$\mathcal R\subset D(\widetilde L_\eta)$, let
$P_\eta$ be its orthogonal projection and define
\[
 C_\eta(z)=P_\eta(z-\widetilde L_\eta)^{-1}P_\eta.
\]
If only $\mathcal R\subset V$, define the instantaneous compression by the
matrix $a_\eta(r_i,r_j)$ and every Schur complement through the weak form.

\begin{theorem}[Dissipative compression and memory response]
\label{thm:r17-c2-memory}
Assume the prepared Markov generator is dissipative on $H_0$.  Then, for
$\Re z>0$ and $0\ne r\in\mathcal R$,
\[
 \Re(C_\eta(z)r,r)_{H_0}
 =\Re z\,\|(z-\widetilde L_\eta)^{-1}r\|^2
 -\Re(\widetilde L_\eta y,y)>0,
\]
where $y=(z-\widetilde L_\eta)^{-1}r$.  Thus
$C_\eta(z)$ is invertible in the right half-plane.  The memory transform
\[
 \widehat K_\eta(z)
 =zP_\eta-P_\eta\widetilde L_\eta P_\eta
 -C_\eta(z)^{-1}
\]
is well typed on a common operator core; on a mere form domain the same
formula holds with $P\widetilde LP$ replaced by the form matrix.  Its
covariant derivative consists of the derivatives of $C$, $P$, the form, and
the metric connection, all on the fixed reference realization.
\end{theorem}

\begin{proof}
The displayed identity follows from $r=(z-\widetilde L)y$ and
dissipativity.  Strict positivity gives injectivity and finite dimensionality
gives invertibility.  The common operator-core formula is then algebraic.  In
the form setting, block the weak equation
$(z+A)y=f$ with respect to $\mathcal R\oplus\mathcal R^\perp$ and
eliminate the second component; the resulting finite matrix is the form-level
Schur complement.  Differentiate after the fixed transport of
Theorem~\ref{thm:r17-c2-connection}; the inverse derivative is
$D(C^{-1})=-C^{-1}(DC)C^{-1}$ and every projection/metric term remains
visible.
\end{proof}

\subsection{Model-specific rigidity}

For a potential $F$ on one platform, let
\[
 \mathcal P_F(G)=Q(F+G)-Q(F)
\]
be the full perturbed pressure functional on a neighborhood of zero in the
model's strong source space.

\begin{theorem}[Pressure-functional rigidity]
\label{thm:r17-c2-rigidity}
If two regular potentials $F_1,F_2$ on the same platform satisfy
\[
 \mathcal P_{F_1}(G)=\mathcal P_{F_2}(G)
\]
for every $G$ in one strong neighborhood, then their equilibrium path laws
coincide.  For the Sinai platform their difference is a constant plus a closed
dynamical coboundary in the A3 potential space.  For the hard-sphere platform
their dual fields differ by the closed balance gauge
$(r,-\Delta r)$ and the finite collision invariants.
\end{theorem}

\begin{proof}
Differentiating in every strong source direction gives equality of all local
path-cylinder expectations.  The cylinder algebra is convergence determining
and dense in the strict topology on weighted sublevels, hence the equilibrium
Radon laws coincide.  On the Sinai platform, periodic orbit approximation for
the proved countable renewal code implies that all periodic sums of
$F_1-F_2$ equal one constant times the period.  The Livšic theorem on the
A3 summable-variation/weighted space and closedness of the coboundary range
give the representation.  On the hard-sphere platform, equality against all
balanced finite-action variations and the closed-range theorem of B3 identify
the annihilator as the full balance gauge plus invariants.  The argument uses
the complete perturbation functional, not one scalar pressure value.
\end{proof}

\subsection{Changing filtrations and optional projections}

Let $(X^\varepsilon,L^\varepsilon)$ be the microscopic path and its
canonical likelihood ratio, and let
$\mathcal G_t^\varepsilon$ be the coarse observation filtration.  Assume
the A4/C1 conditional kernels converge in the regular chart and the likelihood
ratios are uniformly bounded in $L^{1+\delta}$.

\begin{theorem}[Extended weak convergence of optional projections]
\label{thm:r17-c2-optional}
The triples
\[
 \left(X^\varepsilon,L^\varepsilon,
  \mathbf E[L_T^\varepsilon\mid\mathcal G_t^\varepsilon]
 \right)
\]
converge in the extended weak topology to the corresponding limiting path,
likelihood, and optional projection.  Consequently the projected likelihood
is the limiting stochastic exponential, and the Girsanov and bounded
risk-sensitive BSDE representations commute with the microscopic limit.
\end{theorem}

\begin{proof}
Convergence of the conditional kernels means that, for every bounded
continuous future cylinder $F$, the conditional expectation
$E[F(X^\varepsilon)\mid\mathcal G_t^\varepsilon]$ converges
jointly with the past to its limiting kernel evaluation, uniformly on compact
regular histories.  Approximate $L_T$ in $L^1$ by bounded continuous
cylinders using the uniform $L^{1+\delta}$ bound.  The approximation and
kernel convergence identify all finite-dimensional optional projections;
Doob's maximal inequality gives tightness.  The limiting Gaussian likelihood
from B3/A4 has the Cameron--Martin stochastic-exponential form, and standard
Girsanov and exponential-transform identities then yield the representations.
\end{proof}

\subsection{The typed contraction category}

Objects are tuples
$(E,\tau, I,\mathscr X, H, V, L)$ consisting of a
platform label, path topology, rate, strong source space, Hilbert realization,
form domain, and generator.  A morphism consists of a continuous or
exponentially approximable state map, its strict adjoint on Radon tests, a
bounded pullback on strong sources, and an intertwining map on the transported
form bundle.

\begin{theorem}[Naturality of rates, tangents, and memory]
\label{thm:r17-c2-functor}
For every typed morphism $T$, contraction of rates and pullback of pressure
commute,
\[
 I_T(y)=\inf_{Tx=y}I(x),
 \qquad Q_T(G)=Q(T^*G).
\]
At regular exposed points, Gaussian covariance and cotangents push forward as
$T\Sigma T^*$ and $T^*$, and compressed memory commutes whenever the
resolved subspace and form intertwiner form a commuting square.  Composition
of morphisms preserves these identities.
\end{theorem}

\begin{proof}
The pressure identity is exact at finite volume and the rate identity is the
extended contraction theorem.  Differentiating the exact pressure identity on
the transported strong spaces gives the covariance and cotangent formulas.
The form-level Schur complement is natural under a commuting block
intertwiner, giving memory compatibility.  All maps and topologies are part of
the object/morphism definition, so composition follows by ordinary
composition rather than by a bookkeeping declaration.
\end{proof}

\begin{theorem}[Round-seventeen C2 closure]
\label{thm:r17-c2-main}
Each platform has a correct signed-Radon strict dual, a transported strong
response realization with metric connection, a well-typed accretive
form/operator memory, model-specific pressure rigidity, convergent optional
projections, and a genuine typed contraction functor.
\end{theorem}

\begin{proof}
Combine Theorems~\ref{thm:r17-c2-dual},
\ref{thm:r17-c2-connection}, \ref{thm:r17-c2-memory},
\ref{thm:r17-c2-rigidity}, \ref{thm:r17-c2-optional}, and
\ref{thm:r17-c2-functor}.
\end{proof}
