\section{Round-seventeen positive closure: state-dependent controls, nonlinear resolvents, and kinetic semigroup convergence}
\label{sec:r17-b4}

The finite deterministic gas, its probability law, its factorial hierarchy,
and the limiting density value are kept as distinct typed objects.  The
kinetic transition semigroup is constructed from the dynamic B2 action; no
initial preparation cost is charged inside a transition.  Comparison is
obtained from the nonlinear resolvent and a state-dependent control-transfer
lemma, not by applying one collision measure to two different density paths.

\subsection{Exact finite objects and the law--hierarchy chart}

Let $\Phi_t^\varepsilon$ be the hard-sphere flow on the regular phase space,
$U_t^\varepsilon F=F\circ\Phi_t^\varepsilon$ its Koopman group, and
$(\Phi_t^\varepsilon)_\#P$ the push-forward of an initial law.  On laws with a
uniform exponential particle-number and energy moment, let
$\mathfrak K(P)=(g_j(P))_{j\ge0}$ be the factorial-correlation transform.
Equip the hierarchy with
\[
 \|g\|_{\alpha,\beta}=
 \sum_{j\ge0}{e^{\alpha j}\over j!}
 \int |g_j(Z_j)|e^{\beta|V_j|^2}\,dZ_j.
\]

\begin{theorem}[Exact law--hierarchy realization]
\label{thm:r17-b4-intertwiner}
On the exponential-moment law class, $\mathfrak K$ is injective, its image is
characterized by Lenard positivity, consistency, and the exponential bound,
and
\[
 \mathfrak K((\Phi_t^\varepsilon)_\#P)
 =\mathscr U_t^\varepsilon\mathfrak K(P),
\]
where $\mathscr U_t^\varepsilon$ is the BBGKY hierarchy evolution on the
realizable cone.  Polynomial law cylinders and their exponential transforms
are determined by absolutely convergent factorial series on this cone.
\end{theorem}

\begin{proof}
The exponential particle-number bound makes the Janossy generating functional
entire near the unit function.  Its derivatives are the correlation
functions, so equality of all $g_j$ implies equality of the generating
functional and hence of the law.  Lenard positivity and consistency reconstruct
the finite-volume Janossy measures; the exponential bound normalizes their
sum.  Integrating the Liouville equation against factorial observables gives
BBGKY and proves the intertwining.  The factor $1/j!$ together with the
exponential moment gives normal convergence of polynomial and exponential
cylinders.
\end{proof}

\subsection{The dynamic action and state-dependent control transfer}

For a positive density path $f$ and a contact intensity
$d\Gamma=q\,dA_f$, define the transition action
\[
 \mathcal A_{s,t}(f,\Gamma)
 =\int_s^t\int\ell(q)\,dA_f,
 \qquad \ell(q)=q\log q-q+1,
\]
subject to the exact weak balance and the initial condition $f(s)=f_s$.  The
initial preparation rate is absent.

\begin{lemma}[Multiplier transfer between nearby states]
\label{lem:r17-b4-transfer}
Let $(f,\Gamma=qA_f)$ be a regular controlled path with
$0<q_-\le q\le q_+$, and let $g_s$ be close to $f_s$ in the weighted
bounded-Lipschitz metric and in energy.  Then there is a balanced path
$(g,\widetilde\Gamma)$ beginning at $g_s$ with
\[
 \widetilde\Gamma=qA_g+R(f,g),
\]
where $R$ is an exactly conservative correction satisfying
\[
 \|R(f,g)\|_{L^1_w}
 +|\mathcal A(g,\widetilde\Gamma)-
   \mathcal A(f,\Gamma)|
 \le C\,d_*(f_s,g_s).
\]
The correction preserves strict positivity for sufficiently close states.
\end{lemma}

\begin{proof}
Use the same relative multiplier $q$, not the same collision measure.  The
biased Boltzmann equation with intensity $qA_g$ is locally well posed in the
regular weighted chart, and stability of its mild equation gives
$\sup_t d_*(f_t,g_t)\le C d_*(f_s,g_s)$.  The residual in the finite family of
mass, momentum, energy, and balance coordinates is therefore $O(d_*)$.  Apply
the bounded right inverse for the linearized balance from Paper B3 to a smooth
space--time approximation of that residual, and add a fixed positive
Maxwellian collision background of size $2C d_*$.  The resulting correction
has negative part dominated by the background and hence preserves positivity.
The perspective entropy is locally Lipschitz on
$q\in[q_-/2,2q_+]$; its weighted integral gives the action estimate.
\end{proof}

\begin{lemma}[Compact dynamic action sublevels]
\label{lem:r17-b4-compact}
Fix a compact initial family with uniformly bounded mass and energy.  For each
$M$, the set of balanced paths satisfying
$\mathcal A_{0,T}(f,\Gamma)\le M$ is compact in the topology of weighted
weak density paths and weak contact measures, strengthened by the time
modulus on a countable smooth test family.
\end{lemma}

\begin{proof}
The entropy inequality
$q r\le \ell(q)+e^r-1$ controls the contact mass against every bounded
weighted test.  Elastic balance conserves mass, momentum, and energy, while a
superlinear velocity test and the fixed initial exponential moment give
uniform velocity tightness.  The weak balance bounds increments of
$\langle f_t,\phi_k\rangle$ by transport moments plus contact mass, yielding a
uniform modulus.  Prokhorov, Arzelà--Ascoli on the determining coordinates,
and closedness of the balance relation give sequential compactness.  Lower
semicontinuity follows from the joint lower semicontinuity of the perspective
integrand $(f,\Gamma)\mapsto\int
d\Gamma\log(d\Gamma/dA_f)-d\Gamma+dA_f$.
\end{proof}

\subsection{Nisio semigroup and nonlinear resolvent}

For $h\in\mathrm{BUC}(\mathcal E)$ define
\[
 (S_t h)(f_0)=
 \sup_{(f,\Gamma):f(0)=f_0}
 \{h(f_t)-\mathcal A_{0,t}(f,\Gamma)\}.
\]

\begin{theorem}[Kinetic Nisio semigroup]
\label{thm:r17-b4-nisio}
The operators $(S_t)_{t\ge0}$ form a strongly continuous, monotone,
cash-additive contraction semigroup on weighted uniformly continuous
functions.  For every $\lambda>0$ the nonlinear resolvent
\[
 R_\lambda h=
 \sup_{(f,\Gamma):f(0)=f_0}
 \int_0^\infty e^{-\lambda t}
 \bigl[\lambda h(f_t)-
  \ell(d\Gamma/dA_f)A_f(dt)\bigr]
\]
is well defined, satisfies
\[
 R_\lambda-R_\mu
 =(\mu-\lambda)R_\lambda R_\mu
\]
in the nonlinear resolvent sense, and is a sup-norm contraction.
\end{theorem}

\begin{proof}
Concatenation of balanced paths preserves the state at the joining time and
adds the dynamic action, proving the semigroup law in both directions.
Lemma~\ref{lem:r17-b4-compact} gives attainment after relaxation.  Strong
continuity follows from the short-time balance estimate and the zero-cost
Boltzmann path.  For the resolvent, split the discounted integral at a stopping
time $s$, optimize the tail conditional on $f_s$, and use the semigroup law.
Integrating the resulting dynamic-programming identity against
$\lambda e^{-\lambda s}$ gives the nonlinear resolvent identity.  Taking the
same admissible path in two terminal payoffs gives contraction; no control is
transported between distinct initial states in this argument.
\end{proof}

\begin{theorem}[Full range, $m$-dissipativity, and comparison]
\label{thm:r17-b4-comparison}
Define the graph $\mathcal H$ by
\[
 (u,g)\in\mathcal H
 \quad\Longleftrightarrow\quad
 u=R_\lambda(u+\lambda^{-1}g)
\]
for one, hence every, $\lambda>0$.  Then $\mathcal H$ is closed,
$m$-dissipative, and independent of $\lambda$.  The range of
$I-\lambda^{-1}\mathcal H$ is all of
$\mathrm{BUC}(\mathcal E)$.  Consequently the resolvent equation and the
evolution equation have comparison, and $S_t$ is the unique nonlinear
semigroup generated by $\mathcal H$.
\end{theorem}

\begin{proof}
The resolvent identity in Theorem~\ref{thm:r17-b4-nisio} proves independence
of $\lambda$ and the range statement directly: for every $h$, $u=R_\lambda h$
solves $h\in u-\lambda^{-1}\mathcal H u$.  Contraction gives dissipativity
and closedness.  The nonlinear Hille--Yosida--Crandall--Liggett theorem then
produces a unique contraction semigroup.  Its exponential formula agrees with
$S_t$ because both satisfy the same dynamic-programming resolvent.  Comparison
is the contraction estimate for sub- and supersolutions of the resolvent,
followed by the exponential formula.  Lemma~\ref{lem:r17-b4-transfer} supplies
continuity in the state needed to pass from the regular core to the closure.
\end{proof}

\subsection{Observable graph core and a single perturbed-test sequence}

Let $\mathscr D_\varepsilon$ be the algebra of smooth symmetric observables
satisfying specular boundary compatibility, completed in the graph norm of the
Liouville generator.  Let $\mathscr D$ be the cylinder core on densities whose
collision increment is bounded.

\begin{theorem}[Graph-core hierarchy corrector]
\label{thm:r17-b4-corrector}
For every $F\in\mathscr D$ there are integers $K(\varepsilon)\uparrow\infty$
and exact observables $F_\varepsilon\in\mathscr D_\varepsilon$ such that
\[
 F_\varepsilon\to F,
 \qquad
 \mathscr H_\varepsilon F_\varepsilon
 \to\mathbb H\left(f,{\delta F\over\delta f}\right)
\]
locally uniformly on action sublevels.  The construction uses the full
triangular BBGKY Duhamel chain with $1/j!$ weights and yields one diagonal
sequence, not an iterated limit.
\end{theorem}

\begin{proof}
Galerkin-project the observable generator to the first $K$ hierarchy levels
and solve the finite terminal corrector equation by the forward
remaining-time propagator.  The $j$th connected defect is bounded by the B2
factorial majorant $j!C^jT^{j-1}$; multiplication by the hierarchy weight
$e^{\alpha j}/j!$ leaves a geometric tail $C\rho^K$.  The finite-$\varepsilon$
recollision defect is bounded by $C_K\varepsilon^\gamma$.  Choose
$K(\varepsilon)$ increasing slowly enough that
$\rho^{K(\varepsilon)}+C_{K(\varepsilon)}\varepsilon^\gamma\to0$.
Resolvent regularization in the Liouville graph norm puts every finite
corrector in $\mathscr D_\varepsilon$ while preserving the boundary jump.
The ensemble boundary identity then gives the exponential collision factor
$e^{\Delta p}-1$; it is not attributed to the deterministic Koopman chain
rule.
\end{proof}

\begin{theorem}[Nonlinear Trotter--Kato convergence]
\label{thm:r17-b4-limit}
The finite ensemble log-Laplace semigroups converge locally uniformly on
compact action sublevels to $S_t$.  The hypotheses of nonlinear
Trotter--Kato hold: compact containment, a separating graph core, upper and
lower generator recovery, and uniqueness of the limiting resolvent.
\end{theorem}

\begin{proof}
B2 gives exponential compact containment and the joint dynamic action.  The
corrector theorem supplies both generator inequalities on a separating core.
Lemma~\ref{lem:r17-b4-compact} gives equicoercivity.  Half-relaxed resolvent
limits are respectively sub- and supersolutions of the limiting graph; the
comparison theorem identifies them with $R_\lambda$.  Nonlinear
Trotter--Kato therefore gives resolvent and semigroup convergence.  Exact
law--hierarchy realizability from Theorem~\ref{thm:r17-b4-intertwiner} ensures
that the microscopic values are computed from genuine probability laws, not
from arbitrary formal hierarchies.
\end{proof}

\begin{theorem}[Round-seventeen B4 closure]
\label{thm:r17-b4-main}
The hard-sphere ensemble admits a typed exact law--hierarchy representation,
a compact dynamic action, a strongly continuous kinetic Nisio semigroup, an
$m$-dissipative nonlinear resolvent with comparison, a single graph-domain
corrector sequence, and microscopic nonlinear-semigroup convergence.
\end{theorem}

\begin{proof}
Combine Theorems~\ref{thm:r17-b4-intertwiner},
\ref{thm:r17-b4-nisio}, \ref{thm:r17-b4-comparison},
\ref{thm:r17-b4-corrector}, and~\ref{thm:r17-b4-limit}.
\end{proof}
