\section{Round-seventeen positive closure: latent phases, adapted posterior dynamics, and nonlinear aggregation}
\label{sec:r17-d1}

A phase is a genuine positive latent variable in the microscopic probability
space, not a smooth partition weight renamed as a discrete event.  Dynamic
consistency is obtained by carrying its posterior distribution as part of the
state.  The marginal log-sum value is then the projection of an augmented
semigroup; it is not asserted to be a semigroup on the physical state alone.

\subsection{Positive latent-phase experiments}

Let $J\in\{0,\ldots,m\}$ be sampled at time zero with positive weights
$w_{\varepsilon,j}$.  Conditional on $J=j$, the microscopic path has law
$P_{\varepsilon,j}$ on a common measurable path space.  In the Sinai
platform these conditional laws are obtained from positive recurrent
components of the inducing graph plus a boundary component.  In the
hard-sphere platform they are the positive source/preparation phases
constructed by B1--B4.  The physical law is exactly
\[
 P_\varepsilon=\sum_{j=0}^m w_{\varepsilon,j}
 P_{\varepsilon,j}.
\]
No spectral projection is used as a probability.

\begin{theorem}[Joint labelled large deviations]
\label{thm:r17-d1-joint}
Assume each positive conditional experiment satisfies a good path LDP at
speed $a_\varepsilon$ with rate $I_j$, and
\[
 -a_\varepsilon^{-1}\log w_{\varepsilon,j}\to\alpha_j.
\]
Then $(J,X_\varepsilon)$ satisfies a good LDP on the discrete--path
product with rate
\[
 \mathcal I(j,x)=\alpha_j+I_j(x)-\min_k\alpha_k.
\]
The physical marginal has rate
\[
 I(x)=\min_j\{\alpha_j+I_j(x)\}
 -\min_k\alpha_k.
\]
The component rates are those of genuine conditional laws, not envelopes
defined after the fact.
\end{theorem}

\begin{proof}
For a closed set $F$ in the finite product,
\[
 P_\varepsilon((J,X)\in F)
 =\sum_j w_{\varepsilon,j}
 P_{\varepsilon,j}(F_j).
\]
The largest exponential term gives the upper bound; the corresponding open
slice gives the lower bound.  Exponential tightness is inherited from the
finite family.  Contraction over $j$ gives the marginal rate.  The formula
uses actual positive conditional probabilities throughout.
\end{proof}

\subsection{Restricted pressure with one sign convention}

Define the normalized component pressure
\[
 \widetilde Q_{\varepsilon,j}(F)
 ={1\over a_\varepsilon}
 \log E_{\varepsilon,j}e^{a_\varepsilon F}
\]
and the unnormalized contribution
\[
 Q_{\varepsilon,j}^{\rm un}(F)
 ={1\over a_\varepsilon}
 \log E_\varepsilon[e^{a_\varepsilon F};J=j].
\]

\begin{lemma}[Exact phase-cost sign]
\label{lem:r17-d1-sign}
With
$\alpha_{\varepsilon,j}
 =-a_\varepsilon^{-1}\log w_{\varepsilon,j}$,
\[
 Q_{\varepsilon,j}^{\rm un}(F)
 =-\alpha_{\varepsilon,j}
  +\widetilde Q_{\varepsilon,j}(F).
\]
Consequently
\[
 Q_\varepsilon(F)
 ={1\over a_\varepsilon}\log
 \sum_j\exp\left\{
a_\varepsilon[-\alpha_{\varepsilon,j}
 +\widetilde Q_{\varepsilon,j}(F)]\right\}
\]
and the phase cost is counted exactly once.
\end{lemma}

\begin{proof}
The first identity is
$E[e^{aF};J=j]=w_jE_j[e^{aF}]$ followed by a logarithm.  Summing the exact
positive contributions gives the second formula.
\end{proof}

\subsection{Adapted posterior phase state}

Let $\mathcal F_t$ be the observed filtration.  The phase posterior is
\[
 \rho_t(j)=P(J=j\mid\mathcal F_t).
\]
If the component transition kernels have likelihood increments
$L_{s,t}^j$, Bayes' rule gives
\[
 \rho_t(j)=
 {\rho_s(j)L_{s,t}^j
  \over\sum_k\rho_s(k)L_{s,t}^k}.
\]
This process is adapted and automatically consistent under horizon
restriction.

\begin{theorem}[Augmented phase-posterior semigroup]
\label{thm:r17-d1-semigroup}
Let $S_t^j$ be the component risk-sensitive transition semigroup, including
its physical/filter state.  On the augmented state $(x,\rho)$ define
$\mathbf S_t$ by evolving each component and updating $\rho$ with its
likelihood increment.  Then $(\mathbf S_t)$ is a nonlinear semigroup.
For terminal payoff $h$ its marginal value is the exact nonlinear aggregation
\[
 \mathcal V_t(x,\rho;h)
 ={1\over\theta}
 \log\sum_j\rho(j)
 \exp\{\theta S_t^jh(x)\}
\]
when the component likelihood and risk coefficient are typed in the same
canonical experiment.  The map $x\mapsto\mathcal V_t$ alone need not
form a semigroup; the augmented pair does.
\end{theorem}

\begin{proof}
Condition on $(J,\mathcal F_s)$.  Each component obeys its own tower law,
and Bayes' update of $\rho$ at time $s$ followed by that at time $s+t$ is
identical to the one-step update because likelihood ratios multiply.  This
proves the augmented semigroup law.  Summing the positive conditional
exponential values gives the displayed log-sum formula.  Retaining the
posterior is exactly the state variable missing from a naive marginal
log-sum operator.
\end{proof}

\subsection{Phase-local analytic charts}

For a component $j$, let $K_j$ be a positive transfer/coefficient operator on
the recurrent interior of that component.  Assume its simple leading
eigenvalue is separated by a gap and that the leakage to other components is
exponentially small on a complex source neighborhood.

\begin{theorem}[Quantitative zero-free phase charts]
\label{thm:r17-d1-charts}
There are component neighborhoods $U_j$ on which the restricted partition
function is zero free and
$\widetilde Q_j$ is analytic.  More precisely, if
\[
 Z_{\varepsilon,j}(F)=
 c_{\varepsilon,j}(F)e^{a_\varepsilon q_j(F)}
 +R_{\varepsilon,j}(F),
\]
with $|c|\ge c_0$ and
$|R|\le\frac12c_0e^{a_\varepsilon\Re q_j}$ on
$U_j$, then $Z_{\varepsilon,j}$ has no zero there and all pressure
derivatives converge locally uniformly.  These estimates are supplied by the
A2/A3 spectral component or B1/B2 positive coefficient component; positivity
on the real axis alone is not used.
\end{theorem}

\begin{proof}
On the boundary of each compact subchart, the remainder is strictly smaller
than the leading analytic term.  Rouché's theorem gives zero freeness and the
same number of zeros as the nonvanishing leading term, namely none.  A
consistent analytic logarithm exists, and Cauchy's formula gives convergence
of all derivatives.  The spectral/coefficient gap is a quantitative complex
estimate, not a consequence of real positivity.
\end{proof}

\subsection{Second-order labelled limits at coexistence}

Assume
\[
 w_{\varepsilon,j}
 =e^{-a_\varepsilon\alpha_j}
 a_\varepsilon^{-\gamma_j/2}
 (c_j+o(1))
\]
and, under component $j$,
\[
 \sqrt{a_\varepsilon}
 (T X_\varepsilon-m_j)\Rightarrow N(0,\Sigma_j).
\]

\begin{theorem}[Labelled Gaussian mixtures and tie weights]
\label{thm:r17-d1-gaussian}
At a unique minimizing phase, the usual component Gaussian and
Cameron--Martin tangent holds.  At a tie set $J_*$, the joint labelled field
converges to the mixture
\[
 \sum_{j\in J_*}\omega_j
 \delta_j\otimes N(m_j,\Sigma_j),
\qquad
 \omega_j={c_j\mathbf 1_{\{\gamma_j=\gamma_*\}}
 \over\sum_{k\in J_*}c_k
 \mathbf 1_{\{\gamma_k=\gamma_*\}}}.
\]
Conditioning sequences with different subexponential asymptotics produce the
corresponding explicitly tracked posterior weights; no single covariance or
Cameron--Martin space is assigned to an unlabeled tie mixture.
\end{theorem}

\begin{proof}
Apply the component central limit theorem jointly with the discrete label.
Laplace's method in the exact finite sum selects the phases with minimum
exponential cost and then the minimum polynomial exponent; the coefficients
$c_j$ normalize the remaining weights.  Keeping the label gives joint weak
convergence.  Marginalization yields the Gaussian mixture, while conditional
Cameron--Martin shifts act componentwise.
\end{proof}

\subsection{Typed contractions and calibration}

\begin{theorem}[Phase-aware contraction theorem]
\label{thm:r17-d1-contraction}
For every typed observable map $T$, the labelled rate contracts to
\[
 I_T(y)=\min_j\inf_{Tx=y}
 \{\alpha_j+I_j(x)\}-\min_k\alpha_k.
\]
Conditioning, component pressure differentiation, Gaussian tangent formation,
and posterior aggregation commute on the augmented labelled state whenever
the upstream component interfaces hold.  A mechanical preparation multiplier
is identified with a risk-sensitive coefficient only under an explicit
canonical-cocycle compatibility axiom; phase decomposition alone does not
make that identification.
\end{theorem}

\begin{proof}
Apply the contraction theorem to the joint rate of
Theorem~\ref{thm:r17-d1-joint} and then minimize over the finite label.
Componentwise analytic differentiation and Gaussian convergence commute with
$T$ by the upstream typed contraction theorem of C2.  Bayes aggregation is an
exact finite conditional-expectation identity, so it commutes on the
augmented state.  Finally, equality of the mechanical and valuation exponents
is an extra equality of Radon--Nikodym cocycles; without it the two scalar
parameters are independent.
\end{proof}

\begin{theorem}[Round-seventeen D1 closure]
\label{thm:r17-d1-main}
Positive latent phases, their conditional path laws, adapted posterior
updates, component LDPs, zero-free response charts, nonlinear augmented
semigroups, and labelled Gaussian mixtures form one causal and
horizon-consistent phase theory.  Phase costs have one sign and are counted
once; the standalone theorem does not rely on signed spectral labels or a
non-semigroup marginal log-sum operator.
\end{theorem}

\begin{proof}
Combine Theorems~\ref{thm:r17-d1-joint},
\ref{thm:r17-d1-semigroup}, \ref{thm:r17-d1-charts},
\ref{thm:r17-d1-gaussian}, and~\ref{thm:r17-d1-contraction}, together with
Lemma~\ref{lem:r17-d1-sign}.
\end{proof}
