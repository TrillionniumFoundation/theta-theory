\subsection{Energy-compatible primal and Radon dual spaces}

Fix \(0<\delta<1\) and put
\[
 w_\delta(v)=1+|v|^{2-\delta},
 \qquad
 W_c(v,v_*)=1+|v|^2+|v_*|^2.
\]
Let
\[
 \mathcal X_T=C([0,T];\mathcal M_{w_\delta}
 (\mathbb T^3\times\mathbb R^3))
\]
with compact-uniform weighted weak topology, and let
\(\mathcal M_{W_c}^+(\mathfrak C_T)\) be the exchange/pre-post quotient
Radon collision measures with finite \(W_c\)-moment.  Energy bounds give
uniform integrability for \(w_\delta\), while the collision entropy gives
uniform integrability for \(W_c\) on every action sublevel.

For a nonnegative density path define
\[
 dA_f=\frac12ff_*((v-v_*)\cdot\omega)_+
 dt\,dx\,dv\,dv_*\,d\omega.
\]
A pair \((f,\Gamma)\) is balanced when
\[
 \langle p_T,f_T\rangle-\langle p_0,f_0\rangle
 -\int_0^T\langle(\partial_t+v\cdot\nabla_x)p,f\rangle dt
 =\int\Delta_\omega p\,d\Gamma
\]
for every test in the graph space below.

Let \(\mathcal Y_p\) be the completion of smooth tests for the graph norm
\[
 \|p\|_{\mathcal Y_p}
 =\sum_{e=0,T}\|p_e/w_\delta\|_\infty
 +\|((\partial_t+v\cdot\nabla_x)p)/w_\delta\|_\infty
 +\|\Delta p/W_c\|_\infty.
\]
Endpoint traces, the transport derivative, and
\(p\mapsto\Delta p\) extend continuously by definition.

Let
\[
 C_{W_c}(\mathfrak C_T)
 =\{z\in C(\mathfrak C_T):\|z/W_c\|_\infty<\infty\}
\]
with the weighted bounded-strict topology \(\beta_{W_c}\).  Its continuous
dual is the countably additive signed Radon space
\(\mathcal M_{W_c}(\mathfrak C_T)\).  The entropy conjugacy below uses the
bounded continuous subspace and then monotone truncation; no function is
identified modulo \(A_f\)-null sets.

\begin{lemma}[Closed balance]
\label{lem:r3-b3-balance-closed}
On sets with uniformly bounded energy and collision entropy, the balance
relation is closed in
\(\mathcal X_T\times\mathcal M_{W_c}\).
\end{lemma}

\begin{proof}
First truncate velocities and replace \(p\) by a compactly supported smooth
test.  Every term is then continuous.  The density tails are controlled by
energy and the strict inequality \(2-\delta<2\).  For the collision tails,
Young's inequality
\[
 ab\le e^a-1+\ell(b),\qquad \ell(b)=b\log b-b+1,
\]
with a small multiple of \(W_c\), gives uniform integrability on entropy
sublevels.  Let the truncation radius tend to infinity and then use density of
smooth tests in the graph norm.
\end{proof}

\subsection{Integral duality which detects singular collision current}

Put
\[
 \mathfrak J_f(\Gamma)=
 \begin{cases}
 \displaystyle\int\ell(d\Gamma/dA_f)dA_f,&\Gamma\ll A_f,\\
 +\infty,&\text{otherwise}.
 \end{cases}
\]
For \(p\in\mathcal Y_p\) and bounded continuous collision work \(\psi\), set
\[
 \begin{aligned}
 \mathfrak L_{f,\Gamma}(p,\psi)
 ={}&\langle p_T,f_T\rangle-\langle p_0,f_0\rangle
 -\int\langle(\partial_t+v\cdot\nabla_x)p,f\rangle dt\\
 &+\int\psi\,d\Gamma
 -\int(e^{\Delta p+\psi}-1)dA_f.
 \end{aligned}
\]

\begin{theorem}[Radon path-space Fenchel duality]
\label{thm:r3-b3-duality}
For every finite-energy pair,
\[
 \sup_{p\in\mathcal Y_p,\ \psi\in C_b(\mathfrak C_T)}
 \mathfrak L_{f,\Gamma}(p,\psi)
 =\begin{cases}
 \mathfrak J_f(\Gamma),&(f,\Gamma)\text{ balanced},\\
 +\infty,&\text{otherwise}.
 \end{cases}
\]
The same value is obtained from smooth compactly supported tests followed by
closure in the graph and bounded-strict topologies.
\end{theorem}

\begin{proof}
If balance fails, closedness and Hahn--Banach give a graph test \(p\) with
nonzero balance defect.  Approximate \(-\Delta p\) by bounded continuous
collision tests and scale; the exponential term remains bounded while the
balance defect diverges.

On a balanced pair, the expression depends only on
\(z=\Delta p+\psi\):
\[
 \int z\,d\Gamma-\int(e^z-1)dA_f.
\]
If \(\Gamma=qA_f\), bounded truncations of \(\log q\) and monotone convergence
give \(\int\ell(q)dA_f\), while scalar Fenchel inequality gives the reverse
bound.

If \(\Gamma\not\ll A_f\), choose a compact set \(K\) with
\(A_f(K)=0\), \(\Gamma(K)>0\).  By outer regularity choose open
\(O_n\supset K\) with \(A_f(O_n)\le e^{-2n}\), and by Urysohn choose
\(0\le\chi_n\le1\), \(\chi_n=1\) on \(K\), supported in \(O_n\).  Taking
\(z=n\chi_n\) gives
\[
 \int z\,d\Gamma-\int(e^z-1)dA_f
 \ge n\Gamma(K)-e^nA_f(O_n)\to\infty.
\]
These are genuine continuous functions, not zero equivalence classes.  Smooth
approximation follows from weighted tightness and the graph completion.
\end{proof}

\subsection{Exact balance-adjoint gauge complex}

Define
\[
 \mathfrak D(p,\psi)=\Delta p+\psi,
 \qquad
 \mathfrak G r=(r,-\Delta r).
\]
Use on the source product the graph topology on \(p\) and the inductive
bounded-strict topology over bounded continuous \(\psi\)-sets.

\begin{theorem}[Exact Radon gauge complex]
\label{thm:r3-b3-gauge}
The sequence
\[
 0\longrightarrow\mathcal Y_p
 \xrightarrow{\mathfrak G}
 \mathcal Y_p\times C_b(\mathfrak C_T)
 \xrightarrow{\mathfrak D}
 C_b(\mathfrak C_T)
 \longrightarrow0
\]
is topologically exact on every bounded source set.  The graph of
\(r\mapsto-\Delta r\) is closed, and the quotient is canonically identified
with the invariant collision work \(z=\Delta p+\psi\).  For
\(q=d\Gamma/dA_f>0\), the unique optimal quotient class is
\(z=\log q\), understood through bounded truncations if necessary.
\end{theorem}

\begin{proof}
Continuity of \(\Delta\) is part of the graph norm.  Surjectivity follows
from \(z=\mathfrak D(0,z)\), and
\(\ker\mathfrak D=\{(r,-\Delta r)\}\).  A convergent graph sequence
\((r_n,-\Delta r_n)\) has a graph-norm limit with the same relation, proving
closed range.  The quotient map and its inverse
\(z\mapsto[(0,z)]\) are continuous on bounded source sets.  Theorem
\ref{thm:r3-b3-duality} shows that balanced pairings depend only on \(z\), and
strict convexity of \(e^z-1\) gives uniqueness.
\end{proof}

\subsection{Density of balanced local variations}

\begin{lemma}[Balanced tangent density and annihilator]
\label{lem:r3-b3-local-density}
At a smooth strictly positive regular pair, compact free-flight variations and
compact non-grazing actual-collision variations span a dense subspace of the
linearized balanced tangent space subject to mass, momentum, and energy.  Its
annihilator in the Radon source pair is exactly
\(\operatorname{Ran}\mathfrak G\).
\end{lemma}

\begin{proof}
Local density variations are generated by perturbing the incoming activity on
short free-flight tubes.  Local collision-current variations are generated by
bounded collision sources on non-grazing boxes.  A partition of unity reduces
a smooth balanced tangent to finitely many such local pieces.  The residual
balance defect is orthogonal to the five collision invariants and is corrected
by the positive collision right inverse of B2; its graph norm tends to zero
under refinement.  Energy truncation and the strict Radon topology extend the
density statement.

If \((p,\psi)\) annihilates all local collision variations, then
\(\Delta p+\psi=0\) on every non-grazing box.  Continuity and exhaustion imply
this everywhere in the Radon source domain, hence
\((p,\psi)=\mathfrak Gp\).  The converse is the linearized balance identity.
\end{proof}

\subsection{Nuclear-space joint density--collision Gaussian tangent}

Let \(\mathscr S_\beta\) be the nuclear Fr\'echet space obtained by intersecting
all weighted Sobolev spaces whose velocity weights are strictly below the
Gaussian exponent, together with smooth time tests.  Its strong dual is
\(\mathscr S_\beta'\).  Under a regular finite-volume biased law define
finite-mean centered fields
\[
 \zeta^\varepsilon
 =\sqrt{\mu_\varepsilon}
 (\pi^\varepsilon-\mathbb E\pi^\varepsilon),
 \qquad
 \Xi^\varepsilon
 =\sqrt{\mu_\varepsilon}
 (\Gamma^\varepsilon-\mathbb E\Gamma^\varepsilon).
\]

\begin{lemma}[Mitoma and increment bounds]
\label{lem:r4-b3-tightness}
For every bounded set \(B\subset\mathscr S_\beta\),
\[
 \sup_{\phi\in B}
 \mathbb E|\langle\zeta_t^\varepsilon-\zeta_s^\varepsilon,\phi\rangle|^4
 \le C_B|t-s|^2,
\]
and the analogous estimate holds for collision tests and for mixed pairs.
Consequently tightness of every scalar projection implies tightness in
\(D([0,T];\mathscr S_\beta'\times\mathscr S_\beta')\).
\end{lemma}

\begin{proof}
Expand the fourth increment cumulant with the B2 marked connected series.
Every connected four-point term has pressure-scale size
\(O(\mu_\varepsilon^{-1})\) after central normalization, and the time support
of an increment supplies two ordered collision/free-flight integrations,
giving \(C|t-s|^2\).  Products of second cumulants obey the same estimate.
Uniformity on bounded nuclear test sets follows from the Gaussian velocity
weights.  The countable Hilbertian presentation of \(\mathscr S_\beta\) and
the scalar bounds give the stated Mitoma criterion.
\end{proof}

Let the limiting biased collision density be
\[
 q_f=e^{\Delta p[f]+\psi},
\]
where \(p[f]\) is the unique regular backward adjoint solution from B2.  For a
density tangent \(\zeta\), let \(\dot p[\zeta]\) solve its linearized backward
adjoint equation and define
\[
 Dq_f[\zeta]=q_f\Delta\dot p[\zeta].
\]
This specifies the derivative which was previously left untyped.

\begin{theorem}[Prepared joint fluctuating Boltzmann limit]
\label{thm:r3-b3-gaussian}
The pair \((\zeta^\varepsilon,\Xi^\varepsilon)\) converges in the preceding
nuclear dual path space to the unique centered Gaussian solution of the
linearized biased balance.  For density tests,
\[
 d\langle\zeta_t,\phi\rangle
 =\langle\zeta_t,
 v\cdot\nabla_x\phi+\mathcal L_{f,q}^*\phi\rangle dt
 +dM_t(\phi),
\]
with
\[
 d\langle M(\phi),M(\chi)\rangle_t
 =\int q_f(\Delta\phi)(\Delta\chi)dA_f\,dt.
\]
For collision tests,
\[
 \Xi(\psi_0)
 =M^{\rm coll}(\psi_0)
 +\int\psi_0\,D A_f[\zeta]q_f
 +\int\psi_0\,A_fDq_f[\zeta],
\]
where
\(d\langle M^{\rm coll}(\psi_0),M^{\rm coll}(\psi_1)\rangle
 =\int q_f\psi_0\psi_1dA_fdt\).
The initial covariance is the B1 Schur complement, and every finite test
covariance equals the limiting pressure Hessian.
\end{theorem}

\begin{proof}
For a finite family of tests, normal convergence on a larger complex source
neighborhood bounds the order-\(k\) centered cumulant by
\(C_k\mu_\varepsilon^{1-k/2}\).  Thus cumulants of order at least three
vanish and the second converges to the B2/B1 Hessian.  Lemma
\ref{lem:r4-b3-tightness} supplies process tightness, upgrading the
finite-dimensional Gaussian limit.

Differentiate the biased B2 hierarchy and its actual-contact intensity.  The
density derivative gives the linearized Boltzmann drift; differentiation of
\(A_fq_f\) gives the two predictable collision-response terms above, and the
second collision-source derivative gives the white contact bracket.  The
linear backward adjoint equation defines \(Dq_f\) and has a unique solution on
the regular short-time interval.  Uniqueness of this linear Gaussian martingale
problem identifies the full limit.
\end{proof}
