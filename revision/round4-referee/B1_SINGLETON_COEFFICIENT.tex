\begin{lemma}[Uniform strict constraint convexity]
\label{lem:r3-b1-convex}
There are a compact real source--multiplier set
\(\mathcal U\times\mathcal V\), a short time \(T_*>0\), and
\(0<c<C<\infty\) such that, for \(T\le T_*\),
\[
 cI_d\le D^2_{\lambda\lambda}Q_\varepsilon(H,\lambda)
 \le CI_d
\]
uniformly in small \(\varepsilon\).  The same inequalities hold for the
limit, and the first three derivatives converge locally uniformly.
\end{lemma}

\begin{proof}
Separate the connected logarithm into its one-label sector and its connected
sector with at least two labels:
\[
 Q_{\varepsilon,T}(H,\lambda)
 =Q_{\varepsilon,T}^{(1)}(H,\lambda)
  +Q_{\varepsilon,T}^{(\ge2)}(H,\lambda).
\]
The one-label trajectory has no collision and is a translated free flight.
On the compact source--multiplier set its initial density is bounded above and
below by positive multiples of a Maxwellian.  By the assumed linear
independence of the coordinates of \(C\), there are finitely many compact
phase-space patches on which their centered vectors span \(\mathbb R^d\).
Consequently the singleton covariance satisfies
\[
 c_1I_d\le D^2_{\lambda\lambda}Q^{(1)}\le C_1I_d.
\]

The B2 tree norm, with two constraint derivatives inserted at time zero,
gives
\[
 \sup_{\mathcal U\times\mathcal V}
 \|D^2_{\lambda\lambda}Q_{arepsilon,T}^{(\ge2)}\|
 \le C_2T+o_\varepsilon(1).
\]
The static hard-core correction is \(O(\varepsilon)\) at pressure scale and
is included in the final term.  Choose \(T_*\) so that
\(C_2T_*<c_1/4\), then take \(\varepsilon\) small.  This gives the lower
bound; the cluster majorant gives the upper bound.  Normal convergence on a
larger complex chart and Cauchy's formula give convergence of the first three
derivatives.
\end{proof}

% B1_COEFF
\subsection{Singleton-dominated mixed lattice--continuous coefficient}

Let \(N\) be particle number and let \(Y\in\mathbb R^{d-1}\) be the
continuous constraint vector.  Under the real tilted law at
\((H,\lambda_H)\), write
\[
 \Phi_\varepsilon(t,u)
 =\mathbb E_{H,\lambda_H,\varepsilon}
 \exp\{it(N-\mathbb EN)+i\mu_\varepsilon u\cdot(Y-\mathbb EY)\}.
\]
For the singleton phase density \(m_{H,\lambda}\), define
\[
 \chi_{H,\lambda}(t,u)
 =\int e^{it+i u\cdot(C'-\bar C')}
 m_{H,\lambda}(z)dz,
 \qquad
 a_{H,\lambda}=\int m_{H,\lambda}(z)dz.
\]
On compact source sets, \(a_{H,\lambda}\) is bounded above and below by
positive constants.

\begin{lemma}[Singleton pressure gap]
\label{lem:r4-b1-singleton-gap}
For every \(\delta\in(0,\pi)\) there are \(c_\delta>0\) and
\(T_\delta>0\) such that, for \(T\le T_\delta\),
\[
 \sup_{\delta\le|t|\le\pi,\ |u|\le\delta}
 |\Phi_\varepsilon(t,u)|
 \le e^{-c_\delta\mu_\varepsilon}.
\]
For every compact set not containing \((0,0)\), the same estimate holds, and
for \(|u|\ge1\)
\[
 |\Phi_\varepsilon(t,u)|
 \le e^{-c\mu_\varepsilon}
 +C_M(1+\mu_\varepsilon|u|)^{-M}
\]
for every fixed \(M\).
\end{lemma}

\begin{proof}
Insert the complex constraint source
\((it,i u)\) into the connected logarithm.  The one-label part contributes
\[
 \mu_\varepsilon
 \left(\chi_{H,\lambda}(t,u)-a_{H,\lambda}
 -i\,D\chi_{H,\lambda}(0,0)(t,u)\right).
\]
At \(u=0\), its real part is exactly
\[
 -2\mu_\varepsilon a_{H,\lambda}\sin^2(t/2).
\]
For \(|u|\le\delta\), smoothness and nondegeneracy of the singleton constraint
density preserve a fixed fraction of this gap when
\(\delta\le|t|\le\pi\).

A connected cluster with \(k\ge2\) labels acquires the factor
\(e^{ikt+i u\cdot\sum C'_j}-1\).  The B2 marked tree estimate with two
frequency derivatives gives, uniformly for \(|t|+|u|\le1\),
\[
 \left|\Re\Delta Q_T^{(\ge2)}(t,u)ight|
 \le CT(t^2+|u|^2),
\]
and, on the whole compact frequency domain,
\(|\Re\Delta Q_T^{(\ge2)}|\le CT\).  Taking \(T\) sufficiently small lets
the singleton negative real part dominate this remainder.  Exponentiating the
normalized connected logarithm proves the compact minor-arc estimate.

For continuous frequencies, the singleton measure has a smooth density with
Gaussian tails.  On each direction of the unit sphere one of the spanning
constraint patches has a nonvanishing directional derivative; a finite cover
and repeated integration by parts give
\(|\chi_{H,\lambda}(0,u)|\le C_M(1+|u|)^{-M}\).  Once this transform is below
\(a_{H,\lambda}/2\), the singleton logarithm has a fixed negative real part.
The connected remainder remains \(O(T)\), proving the exponential term.  The
polynomial term records the finite-\(\varepsilon\) smoothing error and is
more than sufficient for Fourier inversion.
\end{proof}

\begin{lemma}[Central complex expansion]
\label{lem:r3-b1-gap}
There are \(c,\delta>0\) such that
\[
 |\Phi_\varepsilon(t,u)|
 \le\exp\{-c\mu_\varepsilon(t^2+|u|^2)\},
 \qquad |t|+|u|\le\delta.
\]
Moreover, uniformly for \(|(t,u)|\le\mu_\varepsilon^{-2/5}\),
\[
 \log\Phi_\varepsilon(t,u)
 =-\frac{\mu_\varepsilon}{2}
 \langle(t,u),\Sigma_{H,\varepsilon}(t,u)\rangle
 +O(\mu_\varepsilon|(t,u)|^3),
\]
where \(cI\le\Sigma_{H,\varepsilon}\le CI\).
\end{lemma}

\begin{proof}
Taylor-expand the complex pressure at the real saddle.  The linear term
vanishes by centering.  Lemma~\ref{lem:r3-b1-convex} gives the uniform
quadratic bounds, and normal convergence on a larger complex polydisc gives a
uniform third derivative.  Taking the neighborhood smaller absorbs the cubic
remainder into the quadratic real part.
\end{proof}

Let the continuous shell widths satisfy
\[
 \delta_\varepsilon\downarrow0,
 \qquad
 \sqrt{\mu_\varepsilon}\delta_\varepsilon\to\infty,
\]
and let \(N_\varepsilon-\mu_\varepsilon a_0=O(1)\).

\begin{theorem}[Source-uniform shell coefficient]
\label{thm:r3-b1-coefficient}
Uniformly on compact source and target sets,
\[
 \mathbb P_{H,\lambda_H,\varepsilon}
 \left(N=N_\varepsilon,
 |Y-a'|\le\delta_\varepsilon\right)
 =\mu_\varepsilon^{-1/2}(c_H+o(1)),
\]
where \(0<c_-\le c_H\le c_+<\infty\).  In particular the logarithm of the
shell probability is \(o(\mu_\varepsilon)\).
\end{theorem}

\begin{proof}
Use lattice Fourier inversion in \(t\in[-\pi,\pi]\) and smooth upper and lower
approximations of the continuous shell.  In the central region
\(|(t,u)|\le\mu_\varepsilon^{-2/5}\), scale by
\(\mu_\varepsilon^{-1/2}\) and use
Lemma~\ref{lem:r3-b1-gap}.  The continuous shell contains a number of standard
deviations tending to infinity, so its Gaussian probability tends to one.
The lattice coordinate contributes
\[
 c_H=(2\pi\sigma_{N\mid Y,H}^2)^{-1/2},
\]
where \(\sigma_{N\mid Y,H}^2\) is the Schur complement of the continuous
constraint covariance.

On the remaining compact frequencies,
Lemma~\ref{lem:r4-b1-singleton-gap} gives an exponentially small integral.
For unbounded continuous frequencies, choose the smoothing order larger than
the dimension and use the polynomial estimate in the same lemma.  These
regions are \(o(\mu_\varepsilon^{-1/2})\).  Uniform covariance bounds give
the positive uniform bounds on \(c_H\).
\end{proof}
