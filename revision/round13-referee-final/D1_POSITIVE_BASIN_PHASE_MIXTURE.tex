\section{Round-thirteen positive closure: canonical positive basins and phase-mixture contractions}
\label{sec:r13-d1}

Phase labels are measurable events on the original microscopic sample space.
They are not Riesz projections.  A finite family of separated compact
components of a platform rate is converted into disjoint positive basins, with
one explicit boundary component that is retained at its actual exponential
cost.

\subsection{Canonical basin labels}

Let $X_n$ take values in a Polish platform path space $\mathcal X$ and satisfy
a good LDP with rate $I$.  Let
$M:\mathcal X\to\mathbb R^d$ be a continuous order parameter.  On a regular
finite-phase chart assume that the relevant compact sublevel contains finitely
many disjoint compact components
$K_1,\ldots,K_J$ separated by distance $4\delta>0$, and that every finite-rate
point in the chart lies either in one basin or in the declared boundary band.
These components are obtained from the A3 or B2 platform rate, not from an
operator spectral projection.

Define the hard measurable label
\[
 J_n(x)=
 \begin{cases}
 j,&d(M(x),K_j)<\delta
     \text{ and }j\text{ is the unique nearest component},\\
 \partial,&\text{otherwise}.
 \end{cases}
\]
Let $B_j=\{J_n=j\}$ and $B_\partial=\{J_n=\partial\}$; the sets do
not depend on the observable projection level.

\begin{theorem}[Exact positive microscopic disintegration]
\label{thm:r13-d1-disintegration}
For every $n$,
\[
 \mathbb P_n=
 \sum_{j=1}^Jw_{n,j}\mathbb P_{n,j}
 +w_{n,\partial}\mathbb P_{n,\partial},
\]
where $w_{n,j}=\mathbb P_n(B_j)$ and
$\mathbb P_{n,j}=\mathbb P_n(\cdot\mid B_j)$.  All weights are nonnegative,
all component laws are probabilities, and the same label is available in
every finite projection.  The boundary component is never declared
superexponential unless its rate actually is infinite.
\end{theorem}

\begin{proof}
The nearest-component rule is a Borel partition of the original sample space.
The displayed identity is the ordinary law of total probability.  Since
$J_n$ is defined from the full path order parameter, every projected label is
its contraction and is automatically projectively compatible.  No positivity
property of a nonleading spectral projection is used.
\end{proof}

\subsection{Restricted and conditional LDPs}

Put
\[
 \alpha_j=\inf_{x\in\overline B_j}I(x),
 \qquad
 \alpha_\partial=\inf_{x\in\overline B_\partial}I(x).
\]
Assume the basin boundaries are $I$-continuity sets and that every finite-rate
point in $B_j$ has a recovery sequence remaining in $B_j$.

\begin{theorem}[Positive component LDP]
\label{thm:r13-d1-component}
The unnormalized measures
$\mathbb P_n(\cdot;B_j)$ satisfy the LDP with rate $I$ restricted to
$\overline B_j$.  If $w_{n,j}>0$ and
$-n^{-1}\log w_{n,j}\to\alpha_j$, the conditional laws
$\mathbb P_{n,j}$ satisfy a good LDP with
\[
 I_j(x)=
 \begin{cases}
 I(x)-\alpha_j,&x\in\overline B_j,\\
 +\infty,&\text{otherwise}.
 \end{cases}
\]
The same theorem applies to the boundary label.  Contracting the exact label
gives
\[
 I(x)=\min_{j\in\{1,\ldots,J,\partial\}}
       \{\alpha_j+I_j(x)\},
\]
with no convexification.
\end{theorem}

\begin{proof}
For a closed set $F$, apply the platform LDP upper bound to
$F\cap\overline B_j$.  For an open set and a point in $B_j$, use the assumed
interior recovery sequence; boundary points follow by lower-semicontinuous
closure.  Taking the whole basin gives the weight exponent $\alpha_j$.
Dividing by the weight subtracts exactly that exponent and proves the
conditional rate.  The contraction principle for the finite positive label
then gives the pointwise minimum.
\end{proof}

\subsection{Normalized component pressures}

Define
\[
 \widetilde Q_{n,j}(\theta)
 ={1\over n}\log
 \mathbb E_n\left[e^{n\langle\theta,X_n\rangle}
                  \mid B_j\right],
\]
while the unnormalized restricted pressure is
\[
 Q_{n,j}^{\rm un}(\theta)
 ={1\over n}\log
 \mathbb E_n\left[e^{n\langle\theta,X_n\rangle};B_j\right].
\]

\begin{proposition}[Single-count phase pressure]
\label{prop:r13-d1-pressure}
Whenever the component Laplace principles hold,
\[
 Q_j^{\rm un}(\theta)=-\alpha_j+\widetilde Q_j(\theta),
 \qquad
 Q(\theta)=
 \max_{j\in\{1,\ldots,J,\partial\}}
 \{-\alpha_j+\widetilde Q_j(\theta)\}.
\]
The phase cost is counted once.
\end{proposition}

\begin{proof}
The first identity is the exact factorization of the restricted expectation
into its weight and conditional expectation.  Sum the positive restricted
expectations and take $n^{-1}\log$ to obtain the maximum.
\end{proof}

\subsection{Phase-local analytic charts}

Inside a basin suppose the platform tilted rate has one unique regular
minimizer separated from the basin boundary by a rate gap $g>0$.  Let
$\chi_j$ be a smooth positive cutoff equal to one in a neighborhood of that
minimizer and supported in $B_j$.

\begin{theorem}[Localized zero-free and response chart]
\label{thm:r13-d1-local}
On a sufficiently small complex source ball, the localized partition
function with cutoff $\chi_j$ equals the unlocalized simple phase contribution
plus an error $O(e^{-gn/2})$ relative to that contribution.  Hence it is
zero-free there for large $n$, has the same limiting analytic pressure, and
its first two derivatives give the phase mean and covariance.  The boundary
label has its own real LDP and is not assigned a fictitious analytic chart.
\end{theorem}

\begin{proof}
The platform source theorem gives a simple analytic phase contribution near
the unique interior minimizer.  On the support of $1-\chi_j$, the real part of
the tilted rate is larger by $g$; compact complex perturbations reduce the gap
by at most $g/2$.  The relative error is therefore $O(e^{-gn/2})$ uniformly on
the complex ball.  Rouché's theorem preserves zero-freeness and the limiting
pressure.  Cauchy estimates give derivative convergence.
\end{proof}

\subsection{Shell, Gaussian, and dynamic mixtures}

Assume the A2 or B1 local coefficient theorem holds uniformly on every active
regular basin.  Let $c_{n,j}(a)$ be the corresponding shell coefficient.

\begin{theorem}[Exact phase-conditioned shell mixture]
\label{thm:r13-d1-shell}
Conditioning the original law on a regular shell gives phase weights
proportional to
\[
 w_{n,j}c_{n,j}(a)
 e^{-nI_j(a)}
\]

after including the boundary label.  Every subsequential limit is the
corresponding mixture of the component conditioned laws.  If one exponent is
strictly minimal, that phase is selected; at a tie, subexponential weights and
coefficients determine the limiting mixture.
\end{theorem}

\begin{proof}
Intersect the exact positive disintegration of
Theorem~\ref{thm:r13-d1-disintegration} with the shell.  Apply the uniform
component coefficient theorem to every term and retain the boundary term at
its actual rate.  Divide by the sum.  This is an exact finite positive mixture,
so no signed cancellation or double-counted phase cost occurs.
\end{proof}

\begin{theorem}[Labelled commutation theorem]
\label{thm:r13-d1-commute}
On the joint labelled law, typed observable contraction, primitive-shell
conditioning, finite-dimensional Gaussian scaling, and the component dynamic
semigroups commute with the finite label disintegration.  After the limits are
proved componentwise, forgetting the label gives the positive mixture and the
minimum rate of Theorem~\ref{thm:r13-d1-component}.
\end{theorem}

\begin{proof}
All four operations are applied to the same positive measures
$w_{n,j}\mathbb P_{n,j}$.  Contraction is push-forward, conditioning is
restriction and normalization, Gaussian scaling is another push-forward, and
dynamic evolution acts linearly on the component laws before the nonlinear
logarithm is taken.  Finite summation commutes with these exact operations;
uniform component limits permit passage to the limit.  Forgetting the label
is an ordinary contraction.
\end{proof}

\begin{theorem}[Round-thirteen D1 closure]
\label{thm:r13-d1-main}
Every regular finite-phase chart of the Sinai or hard-sphere platform admits a
canonical positive basin label on the original microscopic sample space, good
component and boundary LDPs, normalized component pressures, phase-local
response charts, and exact shell/Gaussian/dynamic mixture formulas.  No
nonleading Riesz projection is interpreted as a probability.
\end{theorem}

\begin{proof}
Combine Theorems~\ref{thm:r13-d1-disintegration},
\ref{thm:r13-d1-component}, \ref{thm:r13-d1-local},
\ref{thm:r13-d1-shell}, and \ref{thm:r13-d1-commute}, with
Proposition~\ref{prop:r13-d1-pressure}.
\end{proof}
