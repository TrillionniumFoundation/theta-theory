\section{Round-thirteen positive closure: graph-completed stopped renewal LDPs}
\label{sec:r13-a3}

The singularity information is retained in the state and charged at its true
large-deviation cost.  It is never removed uniformly over an order-$n$
entropy-control class.  The terminal incomplete excursion is generated by the
same controlled transition as the completed excursions, so there is no
separate terminal likelihood and no double counting.

\subsection{The stopped Markov-renewal state}

Let $\mathcal A$ be the countable inducing alphabet.  A symbol $a$ carries its
source and target vertices, collision length $r(a)\ge1$, physical roof
$\tau(a)>0$, and a graph-completed excursion
$K_a:[0,r(a)]\to\overline\Omega_{\rm det}$.  The graph completion records the
one-sided continuation, signed distance to every singularity chart, and the
homogeneity label.  Let $P(a,db)$ be the normalized Gibbs transition kernel.

For a deterministic collision horizon $n$ put
\[
 S_m=\sum_{j=1}^m r(A_j),\qquad
 N_n=\max\{m:S_m\le n\},\qquad
 \alpha_n=n-S_{N_n}.
\]
The stopped state $Z_n^c$ contains
\begin{enumerate}[label=(\roman*)]
\item the transition occupation measure of $A_0,\ldots,A_{N_n+1}$;
\item the length-weighted empirical graph-completed profiles of all completed
      excursions;
\item the next symbol $A_{N_n+1}$, its age $\alpha_n$, and the pointed prefix
      $K_{A_{N_n+1}}|_{[0,\alpha_n]}$.
\end{enumerate}
All masses are divided by the deterministic clock $n$.

\begin{lemma}[Drift and compact containment for the actual renewal kernel]
\label{lem:r13-a3-drift}
There are $\eta>0$, $0<\rho<1$, a finite set $C\subset\mathcal A$, and
$b<\infty$ such that for
$V(a)=e^{\eta(r(a)+\tau(a))}$,
\[
 PV(a)\le\rho V(a)+b1_C(a).
\]
Moreover the graph-completed profile laws with
$\int V\,d\nu\le M$ are compact in the weighted local-Skorokhod topology.
\end{lemma}

\begin{proof}
The Young-tower return tail and bounded Gibbs distortion give an exponential
moment for every branch leaving the finite connector set.  Enlarging the
connector set makes the tail contraction smaller than one and yields the
drift inequality.  A bound on $V$ gives uniform clock tails.  On a fixed
clock level, finite horizon and homogeneity coordinates give equicontinuity on
each regular segment, while the graph completion records every one-sided jump
at a singularity.  Diagonal Arzelà--Ascoli over path windows and the
exponential tail yield compactness.  No supremum over all entrance symbols is
assumed; the drift controls the entrance dependence.
\end{proof}

\subsection{Exact entropy representation}

A predictable control is a sequence of kernels
$Q_j(\cdot\mid A_0,\\ldots,A_j)$ absolutely continuous with respect to
$P(A_j,\cdot)$.  It chooses $A_{j+1}$, including the symbol that contains the
terminal residual.

\begin{theorem}[Exact stopped-control variational formula]
\label{thm:r13-a3-control}
For every bounded continuous $F$ on the stopped state space,
\[
 -{1\over n}\log\mathbb E e^{-nF(Z_n^c)}
 =\inf_Q\mathbb E_Q\left[
 F(Z_n^{c,Q})+{1\over n}
 \sum_{j=0}^{N_n}
 D\bigl(Q_j(\cdot\mid\mathcal F_j)\Vert P(A_j,\cdot)\bigr)
 \right].
\]
There is no independent terminal entropy term: the last summand is the
likelihood of the branch whose prefix is observed at time $n$.
\end{theorem}

\begin{proof}
Apply the Gibbs variational formula successively to the conditional law of
$A_{j+1}$ until the stopping index is reached.  Truncate at
$N_n\wedge L$, use the ordinary finite-product chain rule for relative
entropy, and let $L\to\infty$.  Lemma~\ref{lem:r13-a3-drift} gives uniform
integrability of the stopping index and its entropy costs.  The terminal age
and prefix are measurable functions of the same last branch, so the chain
rule counts that branch exactly once.
\end{proof}

\subsection{The fixed recurrent and recession rates}

Let $\mathcal G$ be the set of invariant edge measures
$\eta(da,db)=\eta_1(da)q(a,db)$ with equal incoming and outgoing marginals and
finite mean clock
$\bar r(\eta)=\int r(b)\,d\eta$.  Its recurrent entropy per collision clock is
\[
 \mathcal R_c(\eta)
 ={\int D(q(a,\cdot)\Vert P(a,\cdot))\,\eta_1(da)
   \over \bar r(\eta)}.
\]

For a normalized graph-completed excursion profile $\xi$, define the actual
one-excursion recession cost by
\[
 J_c(\xi)=
 \sup_{U\ni\xi}\lim_{L\to\infty}
 \inf_{\ell\ge L}
 \left[-{1\over\ell}\log
 P\left(r(A_1)=\ell,
       \ell^{-1}K_{A_1}\in U\mid A_0\in C\right)
 \right]^{\rm lsc},
\]
where connector changes alter the numerator by $O(1)$ and hence do not change
the clock-normalized limit.

\begin{theorem}[Joint recurrent--recession rate]
\label{thm:r13-a3-rate}
The collision-clock rate is the fixed lower-semicontinuous functional
\[
 I^c(z)=\inf\left\{
 (1-\alpha)\mathcal R_c(\eta)+
 \alpha\int J_c(\xi)\,\zeta(d\xi):
 \mathcal C_c(\eta,\alpha,\zeta,\rho)=z
 \right\}.
\]
Here $0\le\alpha\le1$ is the fraction of collision clock carried by
macroscopic excursions, $\zeta$ is their clock-weighted profile law, $\rho$
is the pointed terminal residual law, and $\mathcal C_c$ is the explicit
clock-weighted reconstruction map.  The sublevels of $I^c$ are compact.
No singularity shield is discarded: singularity exposure is part of
$\xi$ and $\rho$ and is charged by $J_c$.
\end{theorem}

\begin{proof}
The entropy bound and Lemma~\ref{lem:r13-a3-drift} give tightness of controlled
edge occupations after separating a vanishing number of excursions whose
individual lengths are of order $n$.  The recurrent part converges to an
invariant edge measure and the entropy chain rule gives
$(1-\alpha)\mathcal R_c$.  Each macroscopic excursion is evaluated at its
actual length; the definition of $J_c$ gives the Gamma-liminf.  Conversely,
truncate $\eta$ to a finite connector core, approximate it by a periodic
Markov flow with the same vertex balance, and concatenate actual long
excursions realizing finitely supported approximations of $\zeta$.  Connector
cost and the pointed terminal prefix are $o(n)$ except when the prefix carries
a positive clock fraction, in which case it is included in $\rho$ and paid by
the same one-excursion cost.  Diagonal approximation gives recovery.  Clock
exponential moments and graph-profile compactness make the sublevels compact.
\end{proof}

\subsection{Full collision-clock LDP}

\begin{theorem}[Good stopped collision-clock LDP]
\label{thm:r13-a3-collision}
The states $Z_n^c$ satisfy a full weak LDP with deterministic speed $n$ and
good rate $I^c$.  The lower bound holds at every finite-rate state, not only
at exposed points.  Contracting the graph-completed profile to the physical
collision path gives the physical empirical-path LDP; finite-rate states give
zero mass to genuinely multivalued grazing continuations.
\end{theorem}

\begin{proof}
Theorem~\ref{thm:r13-a3-control} and compactness of bounded-cost controls give
the Laplace upper bound by the weak-convergence method.  The liminf is
Theorem~\ref{thm:r13-a3-rate}.  Its recovery construction, inserted in the
variational formula, gives the matching lower bound.  Laplace and LDP forms
are equivalent on the Polish compact-containment topology.  The graph-to-
physical map is continuous at every deterministic graph-completed path.  A
law charging a genuinely multivalued singular continuation has arbitrarily
large one-excursion exposure cost and therefore infinite rate.
\end{proof}

\subsection{Physical-time stopping}

Let
$\Theta_m=\sum_{j=1}^m\tau(A_j)$ and stop at deterministic physical time $T$.
The state $Z_T^p$ retains the current excursion, the physical age, and the
partial free flight.  Define invariant controls with mean roof
$\bar\tau(\eta)$ and recurrent cost
\[
 \mathcal R_p(\eta)=
 {\int D(q\Vert P)\,d\eta_1\over\bar\tau(\eta)}.
\]
The physical recession cost $J_p$ is defined by normalizing actual
one-excursion probabilities by physical duration.

\begin{theorem}[Physical-clock LDP]
\label{thm:r13-a3-physical}
As $T\to\infty$, $Z_T^p$ satisfies a full good LDP with speed $T$ and rate
\[
 I^p(z)=\inf\left\{(1-\alpha)\mathcal R_p(\eta)+
 \alpha\int J_p(\xi)\,d\zeta:
 \mathcal C_p(\eta,\alpha,\zeta,\rho)=z\right\}.
\]
This is obtained from a second stopped-control representation and not by
dividing a nonstationary collision-clock cost by one informal mean.
\end{theorem}

\begin{proof}
Repeat Theorem~\ref{thm:r13-a3-control} with the stopping index
$N(T)=\max\{m:\Theta_m\le T\}$.  Tight controlled occupation measures carry
their roof coordinate, so recurrent entropy is normalized by the limiting
roof occupation and macroscopic excursions by their actual physical
duration.  The same liminf and recovery arguments apply with the partial
terminal flight included in $\rho$.  This proves the rate and its compactness
directly at deterministic speed $T$.
\end{proof}

\begin{theorem}[Round-thirteen A3 closure]
\label{thm:r13-a3-main}
The Sinai inducing process has full good empirical-path LDPs at deterministic
collision and physical clocks, including recurrent flow, macroscopic
excursions, singularity exposure, and the pointed terminal residual in one
exact state.  Finite-rate singular events are retained rather than declared
uniformly negligible under entropy controls.
\end{theorem}

\begin{proof}
Combine Theorems~\ref{thm:r13-a3-control},
\ref{thm:r13-a3-rate}, \ref{thm:r13-a3-collision}, and
\ref{thm:r13-a3-physical}.
\end{proof}
