\section{Round-thirteen positive closure: number-first coefficient extraction and the primitive shell}
\label{sec:r13-b1}

The empty grand-canonical sector is never bounded by an integrable continuous
Fourier estimate.  The exact target particle-number coefficient is extracted
first.  Only then is continuous Fourier inversion performed.  Every remaining
monomial contains exactly $N_\varepsilon$ labels, which supplies the smoothing
and removes the empty-sector obstruction.

\subsection{Grand-canonical and canonical generating functions}

Let $H$ denote the bounded particle-path and actual-contact source supplied by
the grand-canonical part of B2.  Let $z$ be the activity and let
$Y^\varepsilon$ be the vector of continuous initial constraints (momentum,
energy, and the declared macro statistics).  Define
\[
 \mathcal Z_\varepsilon(z,H,u)
 =\mathbb E_\varepsilon^{\rm gc}
 \left[z^{N^\varepsilon}
 e^{\mu_\varepsilon\langle H,X^\varepsilon\rangle
   +i\mu_\varepsilon u\cdot Y^\varepsilon}
 \right]
\]

after dividing by the source-zero normalization.  The exact-number
partition function is
\[
 Z_{\varepsilon,N}(H,u)=[z^N]\mathcal Z_\varepsilon(z,H,u).
\]

\begin{theorem}[Number-coefficient trajectory-polymer expansion]
\label{thm:r13-b1-coefficient-expansion}
For $T\le T_*$ and $H$ in a fixed bounded real or complex source ball,
$Z_{\varepsilon,N}(H,u)$ has a normally convergent canonical polymer
expansion, uniform for
$N/\mu_\varepsilon$ in a compact density interval.  The logarithm divided by
$\mu_\varepsilon$ converges locally uniformly, with all source and multiplier
derivatives, to a canonical pressure $Q_N(H,u)$.  The coefficient is obtained
by a steepest contour through the unique activity saddle; no zero-particle or
wrong-number sector appears in the continuous characteristic function.
\end{theorem}

\begin{proof}
Expand the real-trajectory grand partition function into connected polymers
and apply Cauchy's formula in $z$.  Normal convergence on an annulus about the
positive saddle permits coefficient extraction term by term.  The saddle
equation fixes exactly $N$ labels, and the Gaussian activity direction gives
the standard $N^{-1/2}$ coefficient.  Tree-graph bounds and the B2 all-contact
majorant are uniform on the compact density/source chart.  Cauchy estimates
then give convergence of every derivative.  Since $[z^N]1=0$ for $N>0$, the
empty atom is removed before $u$ is integrated.
\end{proof}

\subsection{The exact constraint Hessian and finite saddle}

Let
\[
 C(x,v)=\left(1,v,{|v|^2\over2},\chi_1(x,v),\ldots,\chi_m(x,v)\right).
\]
Assume its affine span on the positive one-particle activity contains the
full declared constraint space.  For a multiplier $\lambda$ write the exact
finite pressure $Q_\varepsilon(H,\lambda)$.

\begin{lemma}[Uniform finite-volume mean map]
\label{lem:r13-b1-meanmap}
On a common compact multiplier domain,
\[
 cI\le D^2_{\lambda\lambda}Q_\varepsilon(H,\lambda)\le CI.
\]
The number coordinate and all cross terms are included.  For every regular
relative-interior target $a$ there is a unique finite saddle
$\lambda_{\varepsilon,H,a}$ satisfying
$D_\lambda Q_\varepsilon(H,\lambda)=a$, and these saddles converge locally
uniformly to $\lambda_{H,a}$.
\end{lemma}

\begin{proof}
The finite Hessian is the activity second moment
\[
 {1\over\mu_\varepsilon}
 \operatorname{Cov}_{H,\lambda}
 \left(\sum_i C(z_i),\sum_iC(z_i)\right),
\]
not a normalized one-mark covariance.  The canonical coefficient expansion
contains a linear fraction of labels in finitely many phase-space patches on
which the vectors $C$ span the constraint space.  The polymer correction is
small on the short-time/source chart, giving the lower bound; Gaussian moments
give the upper bound.  Strict convexity and properness on the relative-
interior chart yield the unique saddle by the degree-one mean map.  Local
uniform pressure convergence and the inverse-function theorem give saddle
convergence.
\end{proof}

\subsection{Full continuous Fourier control after number extraction}

Let $\phi_{\varepsilon,N,H}(u)$ be the continuous characteristic function
under the exact-number, source-tilted canonical law centered at its finite
saddle.

\begin{theorem}[Integrable canonical Fourier theorem]
\label{thm:r13-b1-fourier}
There are $\delta,R,c>0$ and, for every $s>d+3$, constants $C_s$ such that
uniformly on the regular source/target chart:
\begin{enumerate}[label=(\roman*)]
\item for $|u|\le\delta$,
\[
 |\phi_{\varepsilon,N,H}(u)|
 \le e^{-cN|u|^2};
\]
\item for $\delta\le|u|\le R$,
\[
 |\phi_{\varepsilon,N,H}(u)|\le e^{-cN};
\]
\item for $|u|>R$,
\[
 |\phi_{\varepsilon,N,H}(u)|
 \le C_s(1+|u|)^{-s-cN_0}
      +C_se^{-cN}(1+|u|)^{-s},
\]
where $N_0\ge1$ is fixed and the first exponent may be replaced by
$s+cN$ on the good-label sector.
\end{enumerate}
The right side in (iii) is integrable and its integral is $o(N^{-1/2})$.
The same theorem holds jointly with the remaining lattice angles.
\end{theorem}

\begin{proof}
The small-frequency estimate is the finite Hessian expansion in
Lemma~\ref{lem:r13-b1-meanmap}.  On a compact annulus, equality in the
triangle inequality would force a nontrivial affine character of $C$ to be
constant on every spanning patch; the affine-span assumption excludes this,
and the canonical polymer expansion preserves the gap.

For large $u$, examine each exact-$N$ polymer monomial.  A dependency-graph
matching lemma gives one of two alternatives.  Either a linear number of
labels are leaves whose continuous marks can be integrated independently
conditional on the remaining graph, producing a power
$(1+|u|)^{-s-cN}$; or the graph contains a linear number of extra polymer
bonds, whose absolute activity is $e^{-cN}$.  In the latter sector at least
one label remains, and $s$ integrations by parts in that label give the factor
$(1+|u|)^{-s}$.  Thus even the exponentially small bad sector has frequency
decay; no constant is integrated over an infinite Fourier line.  Summing the
normally convergent canonical polymers proves (iii).  The lattice angles are
compact and are handled by the same annular character argument.
\end{proof}

\subsection{The sharp primitive-shell coefficient}

Let the particle number be the exact integer $N_\varepsilon$, and under the
finite saddle write
\[
 \sqrt{\mu_\varepsilon}\bigl(Y^\varepsilon-a'\bigr)\in W_\varepsilon,
\]
where $W_\varepsilon$ is a centered Lipschitz domain whose Gaussian boundary
mass tends to zero and whose Gaussian mass
$\gamma_{H,a}(W_\varepsilon)$ satisfies
\[
 \gamma_{H,a}(W_\varepsilon)
 \gg \mu_\varepsilon^{-K}
\]

after comparison with the explicit Fourier remainder.  Expanding centered
boxes with
$\sqrt\mu\,\delta_\varepsilon\to\infty$ are included.

\begin{theorem}[Uniform mixed local coefficient]
\label{thm:r13-b1-local}
Uniformly on compact regular source and target sets,
\[
 \mathbb P_{H,\lambda_{\varepsilon,H,a}}
 \left(N=N_\varepsilon,
 \sqrt{\mu_\varepsilon}(Y-a')\in W_\varepsilon\right)
 =\mu_\varepsilon^{-1/2}
 c_{H,a}\,\gamma_{H,a}(W_\varepsilon)
 \bigl(1+o(1)\bigr),
\]
with $0<c\le c_{H,a}\le C$.  The error is relative to the displayed Gaussian
mass.
\end{theorem}

\begin{proof}
Cauchy coefficient extraction in the number variable is performed first.  On
the canonical coefficient, Fourier inversion in the continuous coordinates
has a Gaussian central region, an exponentially small compact annulus, and an
integrable high-frequency tail by
Theorem~\ref{thm:r13-b1-fourier}.  Smooth inner and outer approximations of
$W_\varepsilon$ have boundary error controlled by its Gaussian boundary mass.
The assumed comparison with the explicit remainder makes every noncentral
term $o$ of the main coefficient.  The activity saddle contributes the
one-dimensional lattice coefficient $c_{H,a}\mu^{-1/2}$.
\end{proof}

\subsection{Source-dependent microcanonical transfer}

\begin{theorem}[Primitive-shell pressure and covariance]
\label{thm:r13-b1-transfer}
Conditioning the grand-canonical hard-sphere law on the primitive shell gives
\[
 Q^{\rm mc,a}(H)
 =Q(H,\lambda_{H,a})-\lambda_{H,a}\cdot a
  -Q(0,\lambda_{0,a})+\lambda_{0,a}\cdot a.
\]
The initial empirical rate is the grand-canonical rate restricted to the
constraint surface and normalized by its minimum.  Its covariance is the
Schur complement
\[
 Q_{HH}-Q_{H\lambda}Q_{\lambda\lambda}^{-1}Q_{\lambda H}
\]

evaluated at $(0,\lambda_{0,a})$.
\end{theorem}

\begin{proof}
Apply the exact source-dependent change of measure separately to numerator
and denominator.  Theorem~\ref{thm:r13-b1-local} shows that both shell
coefficients are subexponential and gives their uniform ratio.  The exponential
parts are the two saddle values displayed above.  Differentiating the saddle
equation gives the Schur complement; Legendre duality gives the constrained
initial rate.
\end{proof}

\begin{theorem}[Round-thirteen B1 closure]
\label{thm:r13-b1-main}
The primitive hard-sphere shell has a source-dependent exact finite saddle, an
integrable number-first Fourier coefficient theorem, a sharp regular-shell
coefficient, and the constrained pressure/covariance of
Theorem~\ref{thm:r13-b1-transfer}.  The empty grand-canonical sector is
eliminated by exact particle-number extraction rather than contradicted by a
false pointwise bound.
\end{theorem}

\begin{proof}
Combine Theorems~\ref{thm:r13-b1-coefficient-expansion},
\ref{thm:r13-b1-fourier}, \ref{thm:r13-b1-local}, and
\ref{thm:r13-b1-transfer}.
\end{proof}
