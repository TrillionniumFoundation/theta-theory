\section{Round-five referee closure: the hierarchy convolution algebra and a finite Hamiltonian domain}
\label{sec:r5-b4}

The exact microscopic state is the complete factorial-correlation hierarchy.
Its observables form a convolution algebra, while the limiting density
Hamiltonian is restricted to a Maxwellian exponential-moment state and a
cotangent core on which the collision exponential is finite.

\subsection{The exact hierarchy algebra}

For \(\alpha,\beta>0\), let
\[
 {\mathfrak X}_{\alpha,\beta}
 =
 \left\{G=(g_n)_{n\ge0}:
 \sum_{n\ge0}\frac{e^{\alpha n}}{n!}
 \int |g_n(Z_n)|e^{\beta|V_n|^2}\,dZ_n<\infty\right\}.
\]
For symmetric hierarchy functionals \(F=(F_n)\) and \(G=(G_n)\), define the
label convolution
\[
 (F\star G)_n(Z_n)
 =
 \sum_{I\sqcup J=\{1,\ldots,n\}}
 F_{|I|}(Z_I)G_{|J|}(Z_J).
\]

\begin{lemma}[Correlation convolution algebra]
\label{lem:r5-b4-algebra}
With the exponential generating norm, the functional space
\({\mathfrak A}_{\alpha,\beta}\) is a commutative Banach algebra under
\(\star\), and
\[
 \|F\star G\|_{\alpha,\beta}
 \le\|F\|_{\alpha,\beta}\|G\|_{\alpha,\beta}.
\]
The exponential of every finite density/collision cylinder belongs to
\({\mathfrak A}_{\alpha,\beta}\).
\end{lemma}

\begin{proof}
For a fixed \(n\), group the ordered decompositions by \(|I|=k\).  The factor
\(\binom nk\) cancels the factorials in the exponential generating norm, and
Fubini gives the product estimate.  A finite cylinder has coefficients
bounded by \(C^n\) times finitely many weighted tests; its exponential series
is dominated by \(\exp(Ce^{-\alpha})\), hence belongs to the algebra.
\end{proof}

Let \({\mathscr U}_\varepsilon(t)\) be the BBGKY/Liouville semigroup on
hierarchies and let \({\mathscr U}_\varepsilon(t)^*\) act on the algebra.

\begin{theorem}[Exact hierarchy nonlinear tower]
\label{thm:r5-b4-exact}
For \(F\in{\mathfrak A}_{\alpha,\beta}\), define
\[
 {\mathscr S}_\varepsilon(t)F(G)
 =
 \frac1{\mu_\varepsilon}
 \log\langle
 {\mathscr U}_\varepsilon(t)G,\,
 \exp_\star(\mu_\varepsilon F)
 \rangle.
\]
On complete hierarchy states this value is exact and satisfies the nonlinear
tower.  No random hierarchy state is identified with a deterministic
one-particle density: the intermediate state is
\({\mathscr U}_\varepsilon(s)G\) in the complete hierarchy space.
\end{theorem}

\begin{proof}
Factorial correlations determine every symmetric expectation, and
\(\star\)-multiplication is the product of observables under the correlation
pairing.  The linear semigroup property of \({\mathscr U}_\varepsilon\)
therefore gives the tower after exponentiation and logarithmic normalization.
All intermediate objects remain deterministic correlation hierarchies
encoding the full law.
\end{proof}

\subsection{A Hamiltonian domain with finite collision exponentials}

For \(\theta<\beta/8\), put
\[
 {\mathcal E}_{M,\theta}
 =
 \left\{f\ge0:
 \int e^{\theta|v|^2}f(dx\,dv)\le M\right\}.
\]
Let \({\mathscr D}_\theta\) be the cylinder cotangent core for which
\[
 |\Delta_\omega p+\psi|
 \le c_0+c_1(|v|^2+|v_*|^2),
 \qquad 2c_1<\theta.
\]

\begin{lemma}[Finite and continuous Hamiltonian]
\label{lem:r5-b4-hamiltonian}
For \(f\in{\mathcal E}_{M,\theta}\) and
\((p,\psi)\in{\mathscr D}_\theta\),
\[
 {\mathbb H}(f,p,\psi)
 =
 \langle f,v\cdot\nabla_xp\rangle
 +\int(e^{\Delta p+\psi}-1)\,dA_f
\]
is finite.  On bounded subsets of
\({\mathcal E}_{M,\theta}\times{\mathscr D}_\theta\) it is continuous in the
weighted weak topology generated by tests dominated by
\(e^{\theta|v|^2/2}\).
\end{lemma}

\begin{proof}
The collision kernel grows linearly in relative velocity.  The source bound
uses less than half of the available exponential moment in each velocity, so
Cauchy--Schwarz and
\(|v-v_*|\le1+|v|+|v_*|\) give integrability.  Truncate velocities, use weak
continuity on the compact region, and control the tails by the unused
exponential moment.  This proves continuity.  Energy alone is not invoked.
\end{proof}

\subsection{Convergent hierarchy correctors}

Write the nonlinear BBGKY generator as
\[
 {\mathscr H}_\varepsilon F
 =\mu_\varepsilon^{-1}
 e_\star^{-\mu_\varepsilon F}\star
 {\mathscr A}_\varepsilon
 e_\star^{\mu_\varepsilon F}.
\]
Connected defects are graded by the number of labels in the minimal
contact tree.

\begin{theorem}[Convergent connected corrector series]
\label{thm:r5-b4-correctors}
For every bounded subset of \({\mathscr D}_\theta\) there is \(T_*>0\) such
that
\[
 F_\varepsilon^{\rm corr}
 =
 F+\sum_{n\ge2}\mu_\varepsilon^{1-n}{\mathfrak c}_n[F]
\]
converges in a weaker hierarchy algebra and
\[
 {\mathscr H}_\varepsilon F_\varepsilon^{\rm corr}
 \longrightarrow
 {\mathbb H}\left(f,\frac{\delta F}{\delta f},\psi\right)
\]
locally uniformly on hierarchy states whose first marginal lies in
\({\mathcal E}_{M,\theta}\).  The tail after order \(N\) is bounded by
\(C_F(CT_*)^N\), and recollision terms are
\(O(\varepsilon^\alpha)\).
\end{theorem}

\begin{proof}
The B2 all-contact estimate bounds the connected \(n\)-label defect by
\(n!C^nT^{n-1}\) after the exponential velocity weight is split among the
labels.  Define \({\mathfrak c}_n\) by the backward Duhamel integral of that
defect under the free hierarchy.  The homological equation cancels grade
\(n\) and moves the remainder to grade \(n+1\).  Dividing by \(n!\) in the
hierarchy norm makes the series geometric for \(CT_*<1\).  B2's unchanged-
trajectory pivotal estimate supplies the uniform recollision error.
Termwise application of the generator is justified in the weaker algebra,
and the surviving binary term is exactly the displayed Hamiltonian.
\end{proof}

\subsection{Action semigroup and comparison}

For a balanced density--collision pair put
\[
 {\mathcal A}_{s,t}(f,\Gamma)
 =
 \int_s^t
 \ell\left(\frac{d\Gamma}{dA_f}\right)dA_f.
\]
For bounded uniformly continuous \(\Phi\) on
\({\mathcal E}_{M,\theta}\), define
\[
 (S_{s,t}\Phi)(f_s)
 =
 \sup_{f(s)=f_s}
 \{\Phi(f_t)-{\mathcal A}_{s,t}(f,\Gamma)\}.
\]

\begin{theorem}[Maxwellian kinetic semigroup]
\label{thm:r5-b4-action}
The value \(S_{s,t}\) is well defined on every finite horizon by block
composition, preserves bounded uniform continuity, and satisfies
\(S_{r,t}=S_{r,s}S_{s,t}\).  On the cotangent core
\({\mathscr D}_\theta\) it is the viscosity solution of
\[
 \partial_sU+
 {\mathbb H}\left(f,\frac{\delta U}{\delta f},0\right)=0.
\]
\end{theorem}

\begin{proof}
B2 entropy coercivity and the exponential-moment estimate make action
sublevels compact in the chosen weighted topology.  Concatenation preserves
balance and adds the action, giving both inequalities in the dynamic
programming identity.  Short-time duality with
Lemma~\ref{lem:r5-b4-hamiltonian} gives the viscosity inequalities; finite
horizons are covered by the B2 block continuation.
\end{proof}

\begin{theorem}[Comparison on the exponential-moment domain]
\label{thm:r5-b4-comparison}
Bounded upper-semicontinuous subsolutions and bounded lower-semicontinuous
supersolutions on \({\mathcal E}_{M,\theta}\), tested by
\({\mathscr D}_\theta\) and the containment function
\[
 \Upsilon_\theta(f)=\int e^{\theta|v|^2}f,
\]
satisfy comparison.  Hence the action value in
Theorem~\ref{thm:r5-b4-action} is unique.
\end{theorem}

\begin{proof}
Double variables using finitely many weighted weak coordinates and add
\(\delta(\Upsilon_\theta(f)+\Upsilon_\theta(g))\).  Elastic collisions
preserve \(|v|^2+|v_*|^2\), so the collision part of
\({\mathbb H}(f,D\Upsilon_\theta,0)\) vanishes after the pre/post change of
variables; transport is controlled on the torus.  The penalty keeps
maximizers in a compact exponential-moment sublevel.  The Hamiltonian
continuity of Lemma~\ref{lem:r5-b4-hamiltonian} sends the doubled difference
to zero, yielding the standard contradiction.
\end{proof}

\subsection{Microscopic convergence and microcanonical lift}

\begin{theorem}[Hierarchy-to-density convergence]
\label{thm:r5-b4-limit}
For every compact family of prepared hierarchy states and bounded terminal
data, the exact values of
Theorem~\ref{thm:r5-b4-exact} converge locally uniformly to
\(S_{s,t}\Phi\).  Under a microcanonical preparation, the finite-volume
multiplier is retained as a static coordinate at time zero and is eliminated
only after the full kinetic value is evaluated.
\end{theorem}

\begin{proof}
The B2 joint LDP and Laplace principle give the action formula, uniformly by
exponential compact containment.  The corrector convergence in
Theorem~\ref{thm:r5-b4-correctors} identifies the same infinitesimal
Hamiltonian on a separating core, while
Theorem~\ref{thm:r5-b4-comparison} gives uniqueness.  B1 conditions the
initial hierarchy at its exact finite saddle.  Since that conditioning occurs
once, keeping the multiplier as an initial static coordinate preserves the
tower; the terminal contraction then eliminates it.
\end{proof}

\begin{theorem}[Round-five B4 closure]
\label{thm:r5-b4-main}
The finite hard-sphere gas has an exact nonlinear tower on a genuine hierarchy
convolution algebra, its connected correctors converge, and the limiting
Hamilton--Jacobi semigroup is finite and unique on a Maxwellian
exponential-moment domain.
\end{theorem}

\begin{proof}
Combine Theorems~\ref{thm:r5-b4-exact},
\ref{thm:r5-b4-correctors}, \ref{thm:r5-b4-action},
\ref{thm:r5-b4-comparison}, and \ref{thm:r5-b4-limit}.
\end{proof}
