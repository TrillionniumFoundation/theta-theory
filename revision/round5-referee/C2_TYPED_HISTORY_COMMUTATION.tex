\section{Round-five referee closure: countable platform sums, history likelihoods, and strict commutation}
\label{sec:r5-c2}

The global object is a countable locally convex coproduct of typed physical
platforms.  Finite-scale likelihoods live on complete-history filtrations and
are Doob martingales there; no slow coordinate is treated as Markov without a
sufficient-statistic theorem.

\subsection{The platform direct sum}

Let \({\mathfrak A}\) be the countable set of constructed platforms.  For
each \(\alpha\), let \(X_\alpha=C_{W_\alpha}(\Omega_\alpha)\) carry the
weighted strict topology and let
\(M_\alpha=M_{W_\alpha}(\Omega_\alpha)\) be its Radon dual.  Define
\[
 X=\bigoplus_{\alpha\in{\mathfrak A}}X_\alpha
\]
with the locally convex direct-sum topology and
\[
 M=\prod_{\alpha\in{\mathfrak A}}M_\alpha
\]
with the product weak-star topology.

\begin{theorem}[Typed direct-sum duality]
\label{thm:r5-c2-sum}
The continuous dual of \(X\) is \(M\), with pairing
\[
 \langle F,\nu\rangle
 =\sum_\alpha\int F_\alpha\,d\nu_\alpha,
\]
where the sum is finite for each \(F\in X\).  For componentwise good rates
\(I_\alpha\), the coproduct rate
\[
 I(\alpha,\nu)=I_\alpha(\nu)
\]
has conjugate
\[
 Q(F)=\sup_{\alpha,\nu}
 \{\nu(F_\alpha)-I_\alpha(\nu)\}.
\]
No minimizing sequence changes platform label unless an additional
cross-platform scaling map is proved.
\end{theorem}

\begin{proof}
A continuous functional on a locally convex direct sum is exactly a family
of continuous component functionals, with no summability condition because
each vector has finite support.  The strict dual theorem on each component
identifies those functionals with \(M_\alpha\), giving the product dual.
The conjugacy formula is the supremum over the disjoint components.
The discrete label makes each component open and closed, so convergence
cannot change labels.
\end{proof}

\subsection{Coercive strict phase compactness}

For each platform choose a continuous weight \(W_\alpha\ge1\) with compact
sublevels and assume the pressure domain contains a positive multiple of
\(W_\alpha\).  Let
\[
 Q_\alpha(F)=\sup_\nu\{\nu(F)-I_\alpha(\nu)\}.
\]

\begin{proposition}[Strict compact subdifferential]
\label{prop:r5-c2-strict}
On every bounded strict source set,
\(Q_\alpha\) is continuous and
\[
 \partial Q_\alpha(F)
 =
 \operatorname*{argmax}_\nu
 \{\nu(F)-I_\alpha(\nu)\}
\]
is a nonempty compact set of countably additive weighted Radon probabilities.
\end{proposition}

\begin{proof}
Tilting by a small positive multiple of \(W_\alpha\) gives
\[
 I_\alpha(\nu)\ge c\,\nu(W_\alpha)-C
\]
on the relevant variational sublevel.  Hence maximizing sequences are
weighted tight.  Goodness and lower semicontinuity give attainment and
compactness.  The strict dual is the Radon space, so the usual Fenchel
subgradient argument identifies maximizers and subgradients without
finitely additive functionals.
\end{proof}

\subsection{Typed contraction registry}

A component map
\(C_\alpha:\Omega_\alpha\to E_\alpha\) is admissible when it is continuous
on the weighted graph completion or has continuous truncations
\(C_{\alpha,L}\) satisfying
\[
 \lim_{L\to\infty}\limsup_\varepsilon
 \frac1{a_\varepsilon}
 \log\mathbb P_\varepsilon
 \bigl(d(C_\alpha,C_{\alpha,L})>\delta\bigr)=-\infty.
\]

\begin{theorem}[Strict typed contraction]
\label{thm:r5-c2-contraction}
Every admissible component map satisfies
\[
 I_{\alpha,C}(y)
 =\inf\{I_\alpha(\nu):C_{\alpha\#}\nu=y\},
 \qquad
 Q_{\alpha,C}(\eta)=Q_\alpha(C_\alpha^*\eta).
\]
The A1 recut work, A2--A3 clock maps, A4 history/memory coordinates, and
B2 actual-contact maps are admissible with the repaired source topologies.
\end{theorem}

\begin{proof}
The continuous case is the contraction theorem.  In the approximable case
exponential equivalence gives the extended contraction theorem and passes
the pressure identity from the truncations.  A1 work is a descended continuous
cocycle; A3 proves exponentially good clock truncations; A4 history
coordinates are continuous on the Ray/Skorokhod realization; B2 uses weighted
weak actual-contact measures and finite-source exponential containment.
\end{proof}

\subsection{History likelihood martingales}

Let \(P_{s,t}^{W}\) and \(g_s=P_{s,t}^{W}g_t\) be the A4 Feynman--Kac kernel
and its positive space-time harmonic function.  On the complete-history
filtration define
\[
 M_t
 =
 \frac{e^{W_{0,t}}g_t(H_t)}{g_0(H_0)}.
\]

\begin{theorem}[Finite history likelihood and optional projection]
\label{thm:r5-c2-likelihood}
The process \(M_t\) is a positive mean-one martingale on the complete-history
filtration, and its tilted transition kernel is the A4 Doob kernel
\[
 {\mathsf P}_{s,t}^{W,g}F
 =g_s^{-1}P_{s,t}^{W}(g_tF).
\]
For any smaller resolved filtration \({\mathcal G}_t\), the likelihood is
\(\mathbb E[M_t\mid{\mathcal G}_t]\).  If resolved kernels converge in the
A4 weighted history topology, these optional projections converge jointly
with the resolved process.
\end{theorem}

\begin{proof}
The additive work identity and
\(g_s=P_{s,t}^{W}g_t\) give
\(\mathbb E[M_t\mid{\mathcal F}^{\mathsf H}_s]=M_s\).
Bayes' formula gives the Doob transition kernel.  Optional projection of a
uniformly integrable martingale is a martingale in the smaller filtration.
A4 coupling gives uniform convergence of conditional kernels on bounded
history cylinders, and the finite-source exponential moment gives uniform
integrability; hence the optional projections converge.
\end{proof}

\subsection{Pressure derivatives and memory}

Let
\({\mathcal E}^{W,g}_{s,t}F
=\log{\mathsf P}^{W,g}_{s,t}e^F\).  Its derivative at zero is the same Doob
kernel, not \(P_{s,t}F/P_{s,t}1\) with an unrelated normalization.

\begin{theorem}[History-pressure/memory identity]
\label{thm:r5-c2-memory}
For resolved observables \(A,B\),
\[
 \int_0^\infty e^{-zt}
 D{\mathcal E}^{W,g}_{0,t}(0)[A](B)\,dt
 =
 \langle B,P(z-L^{W,g})^{-1}PA\rangle,
 \qquad\Re z>-\gamma.
\]
The compressed memory transform is
\[
 \widehat K^{W,g}(z)
 =zP-PL^{W,g}P-
 [P(z-L^{W,g})^{-1}P]^{-1}.
\]
Source differentiation on either side gives the same Duhamel insertion.
\end{theorem}

\begin{proof}
A4's nonlinear tower gives
\(D{\mathcal E}^{W,g}_{0,t}(0)A
={\mathsf P}^{W,g}_{0,t}A=e^{tL^{W,g}}A\).
Pairing with \(B\) and Laplace transforming gives the compressed resolvent.
The Schur-complement identity gives \(\widehat K\).  The A4 exponential
memory bound supplies a common integrable majorant, so source differentiation
commutes with the integral and equals the resolvent Duhamel formula.
\end{proof}

\subsection{Diffusion and BSDE representation}

Under A4 rough convergence, let
\[
 X_t=x+bt+\sigma W_t
\]
with covariance \(A=\sigma\sigma^T\) and area anomaly \(\Gamma\).  For
terminal \(\Phi\), define the Doob value
\[
 u(t,x)=\theta^{-1}
 \log\mathbb E_{t,x}e^{\theta\Phi(X_T)}.
\]

\begin{corollary}[Likelihood, anomaly-corrected diffusion, and BSDE]
\label{cor:r5-c2-bsde}
The finite history likelihoods of
Theorem~\ref{thm:r5-c2-likelihood} converge to
\[
 {\mathcal E}\left(
 \int_0^\cdot\theta\sigma^TDu(s,X_s)\,dW_s\right).
\]
The limiting controlled rough equation includes the deterministic bracket
drift induced by \(\Gamma\).  In the diffusion coordinates,
\[
 dX_t=(b+\theta A Du+\mathfrak b_\Gamma)dt+\sigma\,dW_t^\theta,
\]
and
\[
 Y_t=\Phi(X_T)+\int_t^T\frac\theta2|Z_s|^2ds
       -\int_t^TZ_s\,dW_s
\]
with \(Y_t=u(t,X_t)\), \(Z_t=\sigma^TDu(t,X_t)\).
\end{corollary}

\begin{proof}
The history likelihood theorem and A4 rough convergence give joint
convergence and uniform integrability.  It\^o's formula for the limiting
classical value gives the stochastic exponential, Girsanov drift, and BSDE.
The rough area correction contracts \(\Gamma\) with the Lie brackets of the
driving vector fields, producing \(\mathfrak b_\Gamma\).
\end{proof}

\begin{theorem}[Round-five C2 closure]
\label{thm:r5-c2-main}
The platform coproduct has a rigorous countable direct-sum dual, every
mechanical contraction is typed, finite likelihoods are complete-history
Doob martingales, and pressure, optional projection, memory, diffusion, and
BSDE representations commute on their proved domains.
\end{theorem}

\begin{proof}
Combine Theorems~\ref{thm:r5-c2-sum},
\ref{thm:r5-c2-contraction}, \ref{thm:r5-c2-likelihood}, and
\ref{thm:r5-c2-memory}, together with
Corollary~\ref{cor:r5-c2-bsde}.
\end{proof}
