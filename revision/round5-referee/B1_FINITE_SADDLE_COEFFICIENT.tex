\section{Round-five referee closure: exact finite-volume saddles and mixed shell coefficients}
\label{sec:r5-b1}

The coefficient extraction is performed at the finite-volume saddle.  The
limiting saddle is used only after the exact finite system has been centered.

\subsection{Normalization and the finite saddle}

Let \(\mu_\varepsilon\to\infty\) be the Boltzmann--Grad activity scale.  For a
bounded finite-mean trajectory source \(H\) and a constraint vector
\(C=(N/\mu_\varepsilon,Y)\in\mathbb R^d\), define
\[
 Q_\varepsilon(H,\lambda)
 =\frac1{\mu_\varepsilon}
  \log\mathbb E_\varepsilon
  \exp\{\mu_\varepsilon H+\mu_\varepsilon\lambda\cdot C\}.
\]
The one-particle activity measure is normalized so that its mass is
\(\mu_\varepsilon a(H,\lambda)\).  Thus the singleton sector contributes
\(a(H,\lambda)(e^{it}-1)\) to the number-frequency pressure, with no hidden
extra factor of \(\mu_\varepsilon\).

\begin{lemma}[Uniform finite-volume convexity]
\label{lem:r5-b1-convex}
For every finite source radius \(R\) there are \(T_R>0\), a compact
multiplier set \(K_R\), and \(0<c_R<C_R<\infty\) such that
\[
 c_RI\le D_{\lambda\lambda}^2Q_\varepsilon(H,\lambda)
       \le C_RI
\]
for \(\|H\|\le R\), \(T\le T_R\), \(\lambda\in K_R\), and all sufficiently
small \(\varepsilon\).  The first three derivatives converge locally
uniformly to those of \(Q\).
\end{lemma}

\begin{proof}
The singleton covariance is
\[
 \int(C-\bar C)\otimes(C-\bar C)\,
       e^{H_0+\lambda\cdot C}\,dm.
\]
Linear independence of the constraints supplies finitely many compact
phase-space patches on which the centered constraint vectors span
\(\mathbb R^d\); the tilted density is bounded above and below on those
patches, giving a positive lower eigenvalue.  The B2 finite-source connected
tree estimate, with two or three constraint insertions at time zero, bounds
the sector with at least two labels by \(C_Re^{C_RR}T\).  Choose \(T_R\) so
that this is less than one half of the singleton lower eigenvalue.  Normal
convergence on a slightly larger complex source polydisc gives uniform
convergence of the first three derivatives by Cauchy's formula.
\end{proof}

\begin{theorem}[Exact finite-volume saddle]
\label{thm:r5-b1-saddle}
Let \(a\) belong to a compact subset of the regular mean image.  For every
\(\|H\|\le R\) and \(T\le T_R\), there is a unique
\(\lambda_{\varepsilon,H,a}\in K_R\) satisfying
\[
 D_\lambda Q_\varepsilon(H,\lambda_{\varepsilon,H,a})=a.
\]
Moreover
\[
 \lambda_{\varepsilon,H,a}\longrightarrow\lambda_{H,a}
\]
locally uniformly, and
\[
 \|\lambda_{\varepsilon,H,a}-\lambda_{H,a}\|
 \le c_R^{-1}
 \sup_{\lambda\in K_R}
 \|D_\lambda Q_\varepsilon(H,\lambda)-D_\lambda Q(H,\lambda)\|.
\]
\end{theorem}

\begin{proof}
Lemma~\ref{lem:r5-b1-convex} makes the finite mean map strongly monotone and
proper on the compact regular chart.  The inverse-function theorem gives a
unique local solution; strong monotonicity excludes a second one.  Subtract
the finite and limiting saddle equations and pair with the multiplier
difference.  The lower Hessian bound gives the displayed quantitative
estimate and local uniform convergence.
\end{proof}

\subsection{Fourier bounds at the exact center}

Under the tilted law at \(\lambda_{\varepsilon,H,a}\), set
\[
 \widetilde N=N-\mathbb E_\varepsilon N,\qquad
 \widetilde Y=Y-\mathbb E_\varepsilon Y
\]
and
\[
 \Phi_\varepsilon(t,u)
 =\mathbb E_{\varepsilon,H,a}
   e^{it\widetilde N+i\mu_\varepsilon u\cdot\widetilde Y}.
\]

\begin{lemma}[Central expansion and lattice minor arcs]
\label{lem:r5-b1-fourier}
There are \(\delta,c>0\), uniform on compact regular charts, such that
\[
 |\Phi_\varepsilon(t,u)|
 \le e^{-c\mu_\varepsilon(t^2+|u|^2)}
 \quad (|t|+|u|\le\delta),
\]
and
\[
 \log\Phi_\varepsilon(t,u)
 =-\frac{\mu_\varepsilon}{2}
  \langle(t,u),\Sigma_{\varepsilon,H,a}(t,u)\rangle
 +O(\mu_\varepsilon|(t,u)|^3),
\]
where \(cI\le\Sigma_{\varepsilon,H,a}\le CI\).  For every
\(\delta_0>0\),
\[
 \sup_{\delta_0\le|t|\le\pi,\ |u|\le\delta_0}
 |\Phi_\varepsilon(t,u)|
 \le e^{-c_{\delta_0}\mu_\varepsilon}.
\]
\end{lemma}

\begin{proof}
The finite saddle removes the linear term exactly.  Taylor expansion of the
complex pressure and Lemma~\ref{lem:r5-b1-convex} give the central statements.
For the lattice minor arcs, the singleton logarithm has real part
\[
 -2\mu_\varepsilon a(H,\lambda_{\varepsilon,H,a})
   \sin^2(t/2)+O(\mu_\varepsilon|u|).
\]
The activity is uniformly positive.  The connected sector is
\(O(\mu_\varepsilon T_R)\) on the compact frequency set and can be made a
strictly smaller fraction of the singleton gap.  This proves the
exponential estimate.
\end{proof}

To control unbounded continuous frequency, split the one-particle phase
space into finitely many patches on each of which one constraint coordinate
has derivative bounded away from zero in the chosen direction.  A smooth
partition of unity depends continuously on the direction, and compactness of
the unit sphere gives a finite directional cover.

\begin{lemma}[Global continuous-frequency decay]
\label{lem:r5-b1-high}
For every \(M\) there are \(C_M,c>0\) such that
\[
 |\Phi_\varepsilon(t,u)|
 \le e^{-c\mu_\varepsilon}
     +C_M(1+\mu_\varepsilon|u|)^{-M}
 \qquad(|u|\ge1,\ t\in[-\pi,\pi]).
\]
\end{lemma}

\begin{proof}
On each directional patch use the constraint coordinate as one integration
variable.  Repeated integration by parts moves derivatives onto the smooth
tilted singleton density, whose derivatives have Gaussian-weighted
\(L^1\) bounds.  The finite partition gives the polynomial term uniformly in
the direction.  Where the singleton transform is already below half of its
zero-frequency mass, its logarithm has a fixed negative real part; the
connected tree remainder is again absorbed by the choice of \(T_R\), giving
the exponential term.
\end{proof}

\subsection{The mixed shell coefficient}

Let the target number satisfy
\[
 N_\varepsilon-\mu_\varepsilon a_N=O(1)
\]
and let the continuous shell width satisfy
\[
 \delta_\varepsilon\downarrow0,\qquad
 \sqrt{\mu_\varepsilon}\,\delta_\varepsilon\to\infty.
\]
The finite saddle is chosen for the exact target
\[
 a_\varepsilon=
 \left(N_\varepsilon/\mu_\varepsilon,a_Y\right).
\]

\begin{theorem}[Source-uniform finite-saddle coefficient]
\label{thm:r5-b1-coefficient}
Uniformly for \(\|H\|\le R\) and regular targets,
\[
 \mathbb P_{\varepsilon,H,a_\varepsilon}
 \left(
 N=N_\varepsilon,\,
 |Y-a_Y|\le\delta_\varepsilon
 \right)
 =
 \mu_\varepsilon^{-1/2}
 \left[
 (2\pi\sigma^2_{N\mid Y,H,a})^{-1/2}+o(1)
 \right],
\]
where \(\sigma^2_{N\mid Y,H,a}\) is the Schur complement of the continuous
constraint covariance and is bounded above and below uniformly.
\end{theorem}

\begin{proof}
Use lattice Fourier inversion in \(t\in[-\pi,\pi]\) and smooth upper and
lower approximations of the continuous shell.  In the central region scale
\((t,u)\) by \(\mu_\varepsilon^{-1/2}\) and apply
Lemma~\ref{lem:r5-b1-fourier}.  The finite saddle centers both coordinates
exactly.  The continuous shell contains a diverging number of standard
deviations and therefore contributes probability \(1+o(1)\); integrating
out the continuous Gaussian leaves the conditional number variance.
Lattice minor arcs are exponentially small by
Lemma~\ref{lem:r5-b1-fourier}; unbounded continuous frequencies are
\(o(\mu_\varepsilon^{-1/2})\) by
Lemma~\ref{lem:r5-b1-high} with \(M>d+2\).  Uniform covariance bounds finish
the proof.
\end{proof}

\subsection{Conditioning and exponential concentration}

Define the conditioned pressure by
\[
 Q_\varepsilon^a(H)
 =
 Q_\varepsilon(H,\lambda_{\varepsilon,H,a})
 -\lambda_{\varepsilon,H,a}\cdot a
 -Q_\varepsilon(0,\lambda_{\varepsilon,0,a})
 +\lambda_{\varepsilon,0,a}\cdot a.
\]

\begin{theorem}[Finite-source microcanonical transfer]
\label{thm:r5-b1-transfer}
For every finite \(R\) and \(T\le T_R\), exact conditioning on
\(N=N_\varepsilon\) and the shrinking continuous shell has log-Laplace
functional \(Q^a(H)=\lim Q_\varepsilon^a(H)\).  Its Hessian is the Schur
complement
\[
 D^2Q^a
 =Q_{HH}-Q_{H\lambda}
  Q_{\lambda\lambda}^{-1}Q_{\lambda H}.
\]
For every closed set separated from the conditioned mean there is \(c>0\)
such that its probability is at most \(e^{-c\mu_\varepsilon}\).
\end{theorem}

\begin{proof}
Exponential tilting at the exact finite saddle is an identity.  The shell
coefficient in Theorem~\ref{thm:r5-b1-coefficient} contributes
\(o(\mu_\varepsilon)\) to the logarithm, uniformly in \(H\).  Passing to the
limit gives the displayed pressure.  Differentiating the exact saddle
equation gives the Schur complement.  Strict convexity and Chernoff's
inequality on a compact source neighbourhood give the exponential
concentration statement with the declared speed.
\end{proof}

\begin{theorem}[Round-five B1 closure]
\label{thm:r5-b1-main}
Microcanonical preparation is centered at the unique exact finite-volume
saddle, has a source-uniform mixed lattice--continuous coefficient, and
produces the constrained pressure and Gaussian Schur complement without a
mean-mismatch penalty.
\end{theorem}

\begin{proof}
Combine Theorems~\ref{thm:r5-b1-saddle},
\ref{thm:r5-b1-coefficient}, and
\ref{thm:r5-b1-transfer}.
\end{proof}
