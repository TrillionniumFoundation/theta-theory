\section{Round-five referee closure: survivor pressure, entropy-preserving recovery, and direct two-clock LDP}
\label{sec:r5-a3}

The lower bound is reconstructed without periodic-orbit approximation of
positive-entropy measures.  Escape is retained as a state coordinate, and
finite-cut Markov approximations preserve both entropy and the potential
integral.

\subsection{The compactified induced state}

Let \(Y_R\) be the graph-completed collision space and let \(F_R:Y_R\to Y_R\)
be the induced map on the magnet.  Write \(r\in\mathbb N\) for return length,
\(\tau>0\) for physical roof, and \(j\in\mathbb N\) for the homogeneity index.
Choose
\[
 V=1+r+\tau+\log(1+j).
\]
The compactification \(Y_R^\dagger=Y_R\cup\{\partial\}\) records every
excursion removed by a cut at the cemetery point \(\partial\).  A probability
\(\nu\) on \(Y_R^\dagger\) has escape mass
\(e(\nu)=\nu(\{\partial\})\).

For \(L\ge1\), let \(K_L=\{V\le L\}\) away from the \(L^{-1}\)-neighbourhood
of the singular set.  The killed transfer operator is
\[
 {\mathcal L}_{R,L,\Phi}h
 ={\mathcal L}_{R,\Phi}(1_{K_L}h).
\]

\begin{lemma}[Exponential Lyapunov inequality]
\label{lem:r5-a3-lyapunov}
For every compact source set there are \(a,c,C>0\), uniform in \(R\), such
that
\[
 {\mathcal L}_{R,\Phi}e^{aV}
 \le e^{-c}e^{aV}+C
\]
in the weak moving-cut norm.  Consequently
\[
 \sup_R\frac1n\log\mathbb P\!\left(
 \frac1n\sum_{k<n}V\circ F_R^k>M\right)
 \longrightarrow-\infty
 \quad(M\to\infty),
\]
uniformly in compact source sets.
\end{lemma}

\begin{proof}
Finite horizon controls the one-flight roof.  A long return to the magnet
requires either a long sequence of regular cells or an approach to a
homogeneity boundary.  The growth lemma gives an exponential tail for the
regular sequence; the homogeneous Jacobian supplies a summable
\(j^{-2}\)-tail for boundary depth.  Multiplication by a source in a compact
weighted ball consumes only a fixed fraction of those exponents.  Summing
over inverse branches gives the drift inequality.  Iteration and exponential
Markov inequality give the displayed compact-containment bound.
\end{proof}

\subsection{Survivor pressure and the escape branch}

Let \(\lambda_{R,L}(\Phi)\) be the leading eigenvalue of the killed operator
on the finite moving-cut bundle and put
\[
 P_{R,L}^{\rm surv}(\Phi)=\log\lambda_{R,L}(\Phi).
\]
The one-big-excursion contribution is
\[
 P_R^{\rm esc}(\Phi)
 =\limsup_{m\to\infty}\frac1m
   \log\sup_{x:r(x)\ge m}
   \exp S_{r(x)}\Phi(x).
\]

\begin{theorem}[Renewal--survivor pressure identity]
\label{thm:r5-a3-pressure}
The full induced pressure exists and
\[
 P_R(\Phi)
 =\max\left\{
 \lim_{L\to\infty}P_{R,L}^{\rm surv}(\Phi),
 P_R^{\rm esc}(\Phi)
 \right\}.
\]
The convergence of the survivor branch is locally uniform on compact source
sets.  If the survivor branch is strictly larger, its equilibrium law is the
limit of the finite-cut equilibrium laws.  If the escape branch is larger,
the maximizing sequence has positive cemetery mass in the compactified
state.  Equality of the branches is represented by their compact convex
mixture; no branch is discarded.
\end{theorem}

\begin{proof}
Decompose every orbit segment at the first exit from \(K_L\).  Segments that
never exit are generated by the killed operator.  An exiting segment contains
one first excursion with \(V>L\), followed by a fresh return; the renewal
convolution therefore bounds the partition function above by the convolution
of the survivor partition function and the one-excursion partition function.
The exponential rate of a convolution is the maximum of the two rates.
Conversely, survivor words and one-excursion words are subsets of the full
language and give the reverse inequality.  Lemma~\ref{lem:r5-a3-lyapunov}
makes the convolution tails locally uniform.  Compactness on
\(Y_R^\dagger\) gives the phase statements, including the coexistence case.
\end{proof}

\subsection{Entropy-preserving finite-cut approximation}

Let \({\mathcal P}_L\) be the finite partition into depth-\(L\) regular
cylinders inside \(K_L\), with one boundary atom.  For an invariant finite-rate
law \(\nu\), let \(\nu^{[m,L]}\) be the stationary Markov chain whose
\(m\)-block transition probabilities are the conditional block probabilities
of \(\nu\), after replacing the boundary atom by the normalized conditional
law on its regular components.

\begin{lemma}[Markov recovery with entropy convergence]
\label{lem:r5-a3-entropy}
For every invariant \(\nu\) with \(\nu(V)<\infty\) and finite relative
entropy rate, there are \(L_q,m_q\to\infty\) such that
\[
 \nu^{[m_q,L_q]}\Longrightarrow\nu,\qquad
 \int\Phi\,d\nu^{[m_q,L_q]}\to\int\Phi\,d\nu,
 \qquad
 h(\nu^{[m_q,L_q]})\to h(\nu).
\]
The approximants are supported on finite regular cut systems.  In particular,
positive entropy is not replaced by periodic-orbit entropy zero.
\end{lemma}

\begin{proof}
First choose \(L\) so that
\(\nu(V1_{\{V>L\}})\) and the entropy contribution of the boundary atom are
small; this is possible by finite rate and
Lemma~\ref{lem:r5-a3-lyapunov}.  On the resulting finite alphabet, the
\(m\)-step Markov approximation has the same \((m+1)\)-block distribution as
\(\nu\).  Hence it converges on cylinders and
\[
 h(\nu^{[m,L]})
 =H_\nu(X_m\mid X_0,\ldots,X_{m-1})
 \downarrow h(\nu)
\]
as \(m\to\infty\).  Replacing the boundary atom by its regular conditional
components changes total variation, the weighted potential integral, and
conditional entropy by the chosen tail error.  Taking \(m\to\infty\) after
\(L\to\infty\) proves all three limits.
\end{proof}

\subsection{Direct induced empirical-path LDP}

Let
\[
 L_n=\frac1n\sum_{k=0}^{n-1}\delta_{\Theta^k\omega}
\]
be the empirical path law on the graph-completed path space.  Define
\[
 I_R^{\rm ind}(\nu)
 =
 \begin{cases}
 P_R(0)-h(\nu)-\int\phi_R\,d\nu,&
 \nu\text{ invariant},\ \nu(V)<\infty,\\
 +\infty,&\text{otherwise},
 \end{cases}
\]
with the cemetery/escape contribution interpreted through
Theorem~\ref{thm:r5-a3-pressure}.

\begin{theorem}[Full induced path LDP]
\label{thm:r5-a3-induced}
The laws of \(L_n\) satisfy a full good LDP in the weighted weak topology
generated by continuous \(F\) with \(|F|\le C(1+V)\), with rate
\(I_R^{\rm ind}\), uniformly on compact radius and source sets.
\end{theorem}

\begin{proof}
On each finite cut system the spectral theorem gives the level-three LDP and
the entropy variational formula.  Exponential compact containment follows
from Lemma~\ref{lem:r5-a3-lyapunov}; hence the finite-cut upper bounds pass to
the full space.  For the lower bound, take a finite-rate \(\nu\) and the
entropy-preserving approximants of
Lemma~\ref{lem:r5-a3-entropy}.  The finite-cut lower bound gives every open
neighbourhood of each approximant the correct rate.  Weighted convergence,
entropy convergence, and potential convergence pass that estimate to \(\nu\).
The escape branch is obtained by the one-excursion words used in
Theorem~\ref{thm:r5-a3-pressure}.  Compact sublevels follow from the
Lyapunov inequality and upper semicontinuity of entropy on each cut followed
by the same approximation.
\end{proof}

\subsection{Collision and physical clocks}

For an induced path \(\omega\), let
\[
 {\mathfrak C}(\omega)
 \quad\text{and}\quad
 {\mathfrak T}(\omega)
\]
be respectively the collision Palm expansion and the roof-time Palm
expansion.  Truncate both at \(V\le L\) and at distance \(L^{-1}\) from
singularities.

\begin{lemma}[Bounded terminal clock surgery]
\label{lem:r5-a3-surgery}
There is \(C<\infty\), uniform in \(R\), such that any regular truncated word
of collision length \(n\) can be extended at both ends by at most \(CL^2\)
collisions to begin and end on the magnet with prescribed regular entrance
and exit atoms.  The added physical time is at most \(CL^2\), and the
Jacobian cost is \(e^{o(n)}\) whenever \(L=o(\sqrt n)\).
\end{lemma}

\begin{proof}
The finite cut graph on \(K_L\) is irreducible after splitting its period.
Choose one connector between every ordered pair of atoms and let \(C_L\) be
the maximum connector length.  The number of atoms is polynomial in \(L\)
because finite horizon and the homogeneous depth cut give polynomial
complexity; a breadth-first path gives \(C_L\le CL^2\).  All connector
collisions stay a distance \(L^{-1}\) from singularities, so distortion is
\(O(C_L\log L)\).  For \(L=o(\sqrt n)\) this is \(o(n)\).  The roof bound
follows from finite horizon.
\end{proof}

\begin{theorem}[Two-clock empirical-path LDP]
\label{thm:r5-a3-two-clock}
The collision empirical path law and the physical-time empirical path law
satisfy full good LDPs obtained by the typed contractions
\({\mathfrak C}\) and \({\mathfrak T}\) from
Theorem~\ref{thm:r5-a3-induced}.  Their rates include the survivor, escape,
and coexistence branches.  The lower bounds hold at zero frequency and at
non-exposed finite-rate phases.
\end{theorem}

\begin{proof}
The truncated Palm maps are continuous.  Lemma~\ref{lem:r5-a3-lyapunov}
makes the discarded tails exponentially good, while
Lemma~\ref{lem:r5-a3-surgery} removes initial and terminal residuals at
subexponential cost.  The extended contraction theorem gives the upper
bounds.  For the lower bounds, apply the Markov recovery
Lemma~\ref{lem:r5-a3-entropy}, perform the bounded surgery on each finite-cut
approximant, and then send \(m\), \(L\), and \(n\) in that order.  This
argument does not require uniqueness or exposedness.
\end{proof}

\begin{theorem}[Round-five A3 closure]
\label{thm:r5-a3-main}
The induced system has a direct survivor/escape level-three LDP with
entropy-preserving recovery, and both collision and physical clocks inherit
full lower and upper bounds through explicit exponentially good maps.
\end{theorem}

\begin{proof}
Combine Theorems~\ref{thm:r5-a3-pressure},
\ref{thm:r5-a3-induced}, and
\ref{thm:r5-a3-two-clock}.
\end{proof}
