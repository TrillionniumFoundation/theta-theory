\section{Round-five referee closure: a typed weighted gauge complex and the balanced Gaussian process}
\label{sec:r5-b3}

The collision source space is chosen to contain quadratic collision increments.
The Orlicz heart has order-continuous norm, so its continuous dual contains no
singular finitely additive functionals.

\subsection{Weighted primal and source spaces}

Put
\[
 W(v,v_*)=1+|v|^2+|v_*|^2
\]
and fix \(0<\eta<\beta/8\), where \(\beta\) is the Maxwellian exponent of the
prepared phase.  Let \({\mathcal Y}_p^\eta\) be the completion of smooth
density tests for
\[
 \|p\|_{p,\eta}
 =
 \sup\frac{|p|+|(\partial_t+v\cdot\nabla_x)p|}{1+|v|^2}
 +\sup\frac{|\Delta_\omega p|}{W}.
\]
Let \({\mathcal H}_{\Phi_\eta}\) be the exponential Orlicz heart on collision
space associated with
\[
 \Phi_\eta(z)=e^{\eta|z|}-1.
\]
Define
\[
 {\mathcal Y}_\psi^\eta
 =\{\psi:\psi/W\in{\mathcal H}_{\Phi_\eta}\},
 \qquad
 {\mathcal Y}_z^\eta={\mathcal Y}_\psi^\eta.
\]

\begin{lemma}[Typed collision increment]
\label{lem:r5-b3-typed}
The map
\[
 \Delta:{\mathcal Y}_p^\eta\longrightarrow{\mathcal Y}_z^\eta
\]
is continuous.  If \(f\) has a Maxwellian moment
\(\int e^{\beta|v|^2}f<\infty\), then
\[
 \int e^{|\Delta p+\psi|}\,dA_f<\infty
\]
on every sufficiently small bounded set of
\({\mathcal Y}_p^\eta\times{\mathcal Y}_\psi^\eta\).
\end{lemma}

\begin{proof}
The first statement is built into the norm.  For the second, Young's
inequality gives
\[
 |\Delta p+\psi|
 \le \eta W+C_{\eta,\psi}
\]
in the Orlicz modular on a bounded source set.  Relative velocity grows only
linearly, while \(ff_*\) has Maxwellian exponent \(\beta\).  Since
\(2\eta<\beta/4\), the exponential is integrable against \(A_f\).
\end{proof}

\subsection{A quantitative balance right inverse}

For a signed density path \(g\) and signed collision current \(J\), define
the balance defect
\[
 {\mathfrak B}(g,J)(p)
 =
 \langle p_T,g_T\rangle-\langle p_0,g_0\rangle
 -\int\langle(\partial_t+v\cdot\nabla_x)p,g\rangle\,dt
 -\int\Delta p\,dJ.
\]
Let \({\mathcal Z}_0\) be the weighted defect space annihilating mass,
momentum, and energy.

\begin{theorem}[Bounded right inverse of the balance map]
\label{thm:r5-b3-right-inverse}
There is a continuous linear operator
\[
 {\mathcal R}:{\mathcal Z}_0
 \longrightarrow
 {\mathcal X}_g\times{\mathcal X}_J
\]
such that
\[
 {\mathfrak B}({\mathcal R}b)=b,\qquad
 \|{\mathcal R}b\|_{{\mathcal X}_g\times{\mathcal X}_J}
 \le C\|b\|_{{\mathcal Z}_0}.
\]
Its collision component is a finite sum of localized binary blocks and its
density component is obtained by transport along characteristics.
\end{theorem}

\begin{proof}
First solve the endpoint and transport part by
\[
 g(t,x,v)=\int_0^t b_{\rm tr}
 (s,x-(t-s)v,v)\,ds
\]
after representing the transport defect by the weighted Riesz pairing.
The residual defect is local in time and annihilates collision invariants.
Choose finitely many compact velocity boxes and collision normals such that
the vectors of collision increments of a fixed finite test basis span the
moment-free subspace.  The corresponding Gram matrix is positive definite;
a partition of unity in \((t,x)\) gives coefficients bounded by its inverse
smallest eigenvalue.  The resulting localized collision blocks correct the
residual exactly.  Refining the finite test basis gives a Cauchy sequence
because the Gram inverse and weighted block norms are uniform.  Its limit is
\({\mathcal R}b\), and the construction gives the displayed estimate.
\end{proof}

\subsection{The exact gauge complex}

Define
\[
 {\mathfrak D}(p,\psi)=\Delta p+\psi,
 \qquad
 {\mathfrak G}r=(r,-\Delta r).
\]

\begin{theorem}[Closed weighted gauge complex]
\label{thm:r5-b3-gauge}
The sequence
\[
 0\longrightarrow{\mathcal Y}_p^\eta
 \xrightarrow{{\mathfrak G}}
 {\mathcal Y}_p^\eta\times{\mathcal Y}_\psi^\eta
 \xrightarrow{{\mathfrak D}}
 {\mathcal Y}_z^\eta
 \longrightarrow0
\]
is exact and has closed first range.  Hence
\[
 ({\mathcal Y}_p^\eta\times{\mathcal Y}_\psi^\eta)/
 \operatorname{Ran}{\mathfrak G}
 \cong{\mathcal Y}_z^\eta
\]
as Hausdorff Banach spaces.  On balanced pairs the dual action depends only
on \(z=\Delta p+\psi\).
\end{theorem}

\begin{proof}
Lemma~\ref{lem:r5-b3-typed} makes both arrows continuous.
Surjectivity follows from \(z={\mathfrak D}(0,z)\).  The kernel identity is
algebraic:
\({\mathfrak D}(p,\psi)=0\) iff
\((p,\psi)={\mathfrak G}p\).  The range of \({\mathfrak G}\) is the graph of
the continuous map \(-\Delta\), hence closed.  On a balanced pair, the
endpoint and transport terms equal \(\int\Delta p\,d\Gamma\); combining with
the collision source leaves only \(z\).
\end{proof}

\subsection{Countably additive Orlicz duality}

Let \(M^{\Phi_\eta^*}\) be the weighted signed Radon measures with finite
complementary Orlicz modular.  The exponential Orlicz heart is the closure of
bounded compactly supported functions in modular norm.

\begin{proposition}[No singular dual]
\label{prop:r5-b3-orlicz}
The continuous dual of \({\mathcal Y}_z^\eta\) is precisely
\(M^{\Phi_\eta^*}\) under integration.  In particular every continuous
functional is countably additive; no \(ba\)- or singular Orlicz functional
occurs.
\end{proposition}

\begin{proof}
The heart has order-continuous norm.  If \(z_n\downarrow0\), dominated modular
convergence gives \(\|z_n\|\downarrow0\).  Therefore every continuous positive
functional is order continuous and the Daniell construction represents it by
a countably additive Radon measure.  Jordan decomposition treats signed
functionals.  H\"older's inequality for complementary Orlicz functions gives
the converse and identifies the norm.
\end{proof}

For a positive feasible pair put
\[
 dA_f=\frac12ff_*((v-v_*)\cdot\omega)_+\,dt\,dx\,dv\,dv_*\,d\omega.
\]

\begin{theorem}[Closed Radon Fenchel duality]
\label{thm:r5-b3-duality}
For every finite-moment pair \((f,\Gamma)\),
\[
 \sup_{p,\psi}
 \left\{
 \langle p_T,f_T\rangle-\langle p_0,f_0\rangle
 -\int\langle D_tp,f\rangle dt
 +\int\psi\,d\Gamma
 -\int(e^{\Delta p+\psi}-1)dA_f
 \right\}
\]
equals
\[
 \int\ell(d\Gamma/dA_f)\,dA_f
\]
when the pair is balanced and \(\Gamma\ll A_f\), and equals \(+\infty\)
otherwise.
\end{theorem}

\begin{proof}
If balance fails, Theorem~\ref{thm:r5-b3-right-inverse} and Hahn--Banach give
a test \(p\) with nonzero defect.  Taking \(\psi=-\Delta p\) and scaling
makes the supremum infinite.  On balanced pairs the expression reduces to
\[
 \int z\,d\Gamma-\int(e^z-1)dA_f.
\]
The integral conjugacy of \(e^z-1\) on the Orlicz heart gives the entropy for
the absolutely continuous part.  If a singular Radon part is present,
continuous compactly supported tests concentrating on its support and
Proposition~\ref{prop:r5-b3-orlicz} make the supremum infinite.
\end{proof}

\subsection{Covariance cohomology and the process limit}

Let \(Q_\varepsilon\) be the finite-volume joint pressure and define
\[
 \zeta^\varepsilon
 =\sqrt{\mu_\varepsilon}(\pi^\varepsilon-f),\qquad
 \Xi^\varepsilon
 =\sqrt{\mu_\varepsilon}(\Gamma^\varepsilon-\Gamma).
\]

\begin{lemma}[Covariance nullspace]
\label{lem:r5-b3-null}
A finite family of density/collision tests has zero limiting variance if and
only if its class in the quotient of
Theorem~\ref{thm:r5-b3-gauge} is zero, together with a finite linear
combination of imposed microcanonical constraints.
\end{lemma}

\begin{proof}
The ``if'' direction follows from the exact microscopic balance and fixed
constraints.  Conversely, zero variance makes the observable constant on
every open regular hard-sphere phase cell.  Varying one free flight identifies
its density part as a transport coboundary; varying one non-grazing binary
contact identifies the residual collision part as \(-\Delta r\).  The bounded
right inverse in Theorem~\ref{thm:r5-b3-right-inverse} closes the local
identities in the weighted norm.  The only remaining constants are the
imposed constraint coordinates.
\end{proof}

\begin{theorem}[Balanced joint Gaussian process]
\label{thm:r5-b3-gaussian}
The pair \((\zeta^\varepsilon,\Xi^\varepsilon)\) converges in the dual of a
countably Hilbert nuclear test space to the centered Gaussian solution of the
linearized balanced equation.  For density tests \(\phi,\chi\), its martingale
bracket is
\[
 \langle M(\phi),M(\chi)\rangle_t
 =
 \int_0^t\!\int
 q\,(\Delta_\omega\phi)(\Delta_\omega\chi)\,dA_{f_s}.
\]
There is no extra copy of \(\Delta\phi\); in particular the diagonal bracket
uses \((\Delta_\omega\phi)^2\), not \(4(\Delta_\omega\phi)^2\).
After microcanonical preparation the initial covariance is
\[
 \Pi_{\rm bal}
 \left[
 \Sigma-\Sigma C^*(C\Sigma C^*)^{-1}C\Sigma
 \right]\Pi_{\rm bal}^*.
\]
\end{theorem}

\begin{proof}
B2 finite-source normal convergence implies convergence of all finite
collections of cumulants.  The \(k\)-th centered cumulant is
\(O(\mu_\varepsilon^{1-k/2})\), so orders \(k\ge3\) vanish and the Hessian
gives the covariance.  Differentiating one actual-collision exponential
produces exactly one increment \(\Delta\phi\) in each test direction, yielding
the displayed cross bracket.  For tightness, choose Hilbertian seminorms whose
unit balls are compact in the preceding one.  Exponential moment bounds and
the exact balance give Aldous bounds for every test coordinate; Mitoma's
criterion lifts them to the nuclear dual.  The nullspace lemma gives
uniqueness on the balanced quotient, and B1 differentiation gives the Schur
complement before the balanced projection.
\end{proof}

\begin{theorem}[Round-five B3 closure]
\label{thm:r5-b3-main}
The Hamilton--Boltzmann cotangent complex is typed in a quadratic weighted
exponential Orlicz heart, its dual is countably additive Radon current, the
balance map has a bounded right inverse, and the joint Gaussian process has
the correctly normalized collision bracket.
\end{theorem}

\begin{proof}
Combine Theorems~\ref{thm:r5-b3-right-inverse},
\ref{thm:r5-b3-gauge}, \ref{thm:r5-b3-duality}, and
\ref{thm:r5-b3-gaussian}.
\end{proof}
