\section{Round-ten reconstruction: covariance first, short-time constrained curvature, and the complete gauge}
\label{sec:r10-b3}

The collision action is not assigned a Riemannian metric by formal
inspection.  Its raw second variation may be indefinite.  We first construct
the Gaussian covariance from exact finite-volume cumulants.  We then prove a
short-time energy estimate for the actual linearized balanced dynamics and
identify the second epi-derivative of the exposed rate with the inverse
covariance on its range.  No fictitious curvature is attributed to the linear
balance constraint.

\subsection{The weighted balance complex}

Fix $m>8$.  Let $\mathsf X_T$ be the closure of smooth density tangents for
\[
 \|u\|_{\mathsf X_T}^2
 =\sup_{t\le T}\int \frac{|u_t|^2}{f_t}(1+|v|^m)
 +\int_0^T\|u_t\|_{H^{-1}_{x,v}(w_m)}^2dt,
\]
and let $\mathsf Y_T=L^2(qA_f)$ be the contact tangent space.  Write a
contact tangent as
\[
 J=q\,DA_f[u]+qA_f\,\xi .
\]
The linearized weak balance is
\[
 \mathcal B_{f,q}(u,\xi)
 =\partial_tu+v\cdot\nabla_xu
  -\Delta^*(qDA_f[u])-\Delta^*(qA_f\xi)=0,
\]
with the prepared initial conservation constraints.  Its closed kernel is
$\mathsf T_{f,q}$.

The raw second variation of the exposed action is
\[
 \mathfrak h_{f,q}(u,\xi)
 =D^2I_{0,a}(f_0)[u_0,u_0]
  +\int_0^T\!\int qA_f|\xi|^2
  +\int_0^T\!\int(1-q)D^2A_f[u,u].
\]
The last integral has no fixed sign.

\begin{theorem}[Linearized biased Boltzmann well-posedness]
\label{thm:r10-b3-linear}
On every compact regular phase chart there is $T_*>0$ such that, for
$T\le T_*$, the equation $\mathcal B_{f,q}(u,\xi)=0$ with prepared initial
datum has a unique solution in $\mathsf X_T$ and
\[
 \|u\|_{\mathsf X_T}^2
 \le C\left(
  \|u_0\|_{L^2(f_0^{-1}w_m)}^2
  +\int_0^T\!\int qA_f|\xi|^2
 \right).
\]
The same estimate holds for the backward adjoint, including its endpoint and
B2 Green-trace terms.
\end{theorem}

\begin{proof}
Decompose $u/f$ into the five collision invariants and their orthogonal
complement in the weighted collision Hilbert space.  On the complement, the
symmetric part of the regular biased collision operator obeys the uniform
spectral-gap estimate
\[
 -\langle h,L_{f,q}^{\rm sym}h\rangle
 \ge c\|h\|_{\nu}^2.
\]
The invariant coefficients solve the finite transport system obtained by
integrating against $1,v,|v|^2$; their initial values are fixed by the
microcanonical tangent constraints.  Pair the equation with $u/f$ after a
velocity truncation.  Free transport is skew after the endpoint terms, B2's
closed Green graph cancels incoming and outgoing specular traces, and the
forcing $\Delta^*(qA_f\xi)$ is bounded by Cauchy--Schwarz in the collision
Dirichlet norm.  The high-velocity remainder tends to zero by the $m$th
moment.  Gronwall on a compact coefficient chart gives the estimate and
uniqueness.  Reversing time and using the adjoint reflection law gives the
backward estimate.
\end{proof}

\begin{corollary}[Positive short-time second epi-curvature]
\label{cor:r10-b3-curvature}
After reducing $T_*$ if necessary, the restriction of $\mathfrak h_{f,q}$ to
$\mathsf T_{f,q}$ is coercive:
\[
 \mathfrak h_{f,q}(u,\xi)
 \ge c\left(
  \|u_0\|_{L^2(f_0^{-1}w_m)}^2
  +\|u\|_{\mathsf X_T}^2
  +\int qA_f|\xi|^2
 \right).
\]
No curvature is added from the linear balance multiplier.
\end{corollary}

\begin{proof}
Bilinearity of $A_f$ and the moment bounds give
\[
 \left|\int_0^T\!\int(1-q)D^2A_f[u,u]\right|
 \le CT\|q-1\|_{\infty,w}\|u\|_{\mathsf X_T}^2.
\]
Use Theorem~\ref{thm:r10-b3-linear} and choose $T_*$ so that this term is
absorbed by the positive initial entropy and collision residual.  This is a
direct constrained energy argument, not a KKT-curvature assertion.
\end{proof}

\subsection{Closed range and the balance gauge}

Let $\mathsf P_T$ be the weighted backward transport Sobolev space with
endpoint traces, and let
\[
 \mathfrak D(p,\psi)=\Delta p+\psi,
 \qquad
 \mathfrak G r=(r,-\Delta r).
\]
Constants and collision invariants are fixed by endpoint normalization.

\begin{theorem}[Backward observability and exact gauge]
\label{thm:r10-b3-gauge}
For every normalized cotangent,
\[
 \|(p,\psi)\|_{\mathsf P_T\times L^2(qA_f)}
 \le C\left(
  \|\mathcal B_{f,q}^*(p,\psi)\|_{\mathsf T_{f,q}^*}
  +\|\Delta p+\psi\|_{L^2(qA_f)}
 \right).
\]
Consequently the balance operator has closed range and
\[
 \ker\mathfrak D=\operatorname{Ran}\mathfrak G.
\]
The cotangent space is the Hausdorff quotient
$(\mathsf P_T\times L^2(qA_f))/\operatorname{Ran}\mathfrak G$.
\end{theorem}

\begin{proof}
Solve the backward equation by the micro--macro decomposition used in
Theorem~\ref{thm:r10-b3-linear}.  The collision gap controls the microscopic
part, while the five macroscopic coefficients solve a finite hyperbolic
system controlled by the endpoint normalization and the adjoint balance
functional.  The B2 trace graph supplies the boundary Green identity and
excludes grazing defect functionals.  The resulting observability estimate
implies closed range by the Hilbert closed-range theorem.  If
$\Delta p+\psi=0$, the pair equals $(p,-\Delta p)=\mathfrak Gp$; conversely
such pairs vanish on every balanced tangent by the exact weak balance.
\end{proof}

\subsection{Finite-volume covariance first}

For a regular finite source $\Theta$ define the exactly centered field
\[
 Z_\varepsilon
 =\sqrt{\mu_\varepsilon}
 \left(X_\varepsilon-DQ_\varepsilon(\Theta)\right),
 \qquad X_\varepsilon=(\pi^\varepsilon,\Gamma^\varepsilon).
\]

\begin{theorem}[Projective Gaussian covariance]
\label{thm:r10-b3-finite}
For every finite family of weighted density/contact tests, the covariance
matrices
\[
 D^2Q_\varepsilon(\Theta)
 =\mu_\varepsilon\operatorname{Cov}_{\Theta,\varepsilon}(X_\varepsilon)
\]
converge to a compatible positive semidefinite family $\Sigma$.  All scaled
cumulants of order at least three vanish, so the finite projections of
$Z_\varepsilon$ converge to $N(0,\Sigma)$.  The null space is precisely the
prepared conservation space plus the closed balance gauge.
\end{theorem}

\begin{proof}
The B2 normal complex-source expansion gives local uniform convergence of all
finite derivatives.  A connected $k$th cumulant of the scaled field is
$O(\mu_\varepsilon^{1-k/2})$.  Hence only the Hessian survives.  Positivity
and projective compatibility are inherited from finite covariance matrices.
If a class has zero limiting variance, its pairing vanishes against every
regular local tilted perturbation.  Density of those perturbations in the
balanced tangent core and Theorem~\ref{thm:r10-b3-gauge} identify the class
with the gauge plus the prepared constraints.
\end{proof}

\begin{theorem}[Second epi-derivative and covariance inverse]
\label{thm:r10-b3-mosco}
Let $I$ be the exposed joint rate and $x=DQ(\Theta)$.  The second
epi-derivative
\[
 d_e^2I(x\mid\Theta)(z)
 =\Gamma\!\operatorname{-lim}_{t\downarrow0}
 \frac{2}{t^2}
 \bigl[I(x+tz)-I(x)-t\langle\Theta,z\rangle\bigr]
\]
is the closed quadratic form
\[
 d_e^2I(x\mid\Theta)(z)
 =\begin{cases}
 \langle z,\Sigma^\dagger z\rangle,&z\in\overline{\operatorname{Ran}\Sigma},\\
 +\infty,&\text{otherwise}.
 \end{cases}
\]
On the regular short-time balanced core it equals the closure of
$\mathfrak h_{f,q}$ from Corollary~\ref{cor:r10-b3-curvature}.
\end{theorem}

\begin{proof}
On each finite projection, analytic convex duality gives the classical
second-order conjugacy between the pressure Hessian and the second
epi-derivative of its Legendre transform.  Tilts
$\Theta+th$ provide recovery sequences on the covariance range; separation
makes the value infinite off its closure.  Increasing the projections gives
a monotone family of closed quadratic forms and hence a Mosco limit.  For
smooth balanced tangents, differentiating the actual action and using
Theorem~\ref{thm:r10-b3-linear} yields exactly $\mathfrak h_{f,q}$; no
additional balance curvature occurs.  Coercivity makes this core dense in the
form domain, proving equality of the closures.
\end{proof}

\subsection{Process tightness}

\begin{lemma}[Localized connected-cumulant estimate]
\label{lem:r10-b3-cumulants}
For deterministic intervals $I_1,\ldots,I_k$ and compactly supported weighted
tests, the connected cumulant has an $L^1$ density in one anchor time and
exponentially decaying relative-time tree distance.  In particular,
\[
 \mathbb E|Z_\varepsilon([s,t],F)|^4
 \le C|t-s|^2+C\mu_\varepsilon^{-1}|t-s|,
 \qquad
 \mathbb E|Z_\varepsilon([s,t],F)|^2\le C|t-s|.
\]
\end{lemma}

\begin{proof}
Localize the B2 connected trajectory expansion by inserting the deterministic
time indicators before summing graphs.  Root each connected graph at its
earliest contact.  Every additional marked time is joined to the root by a
real trajectory edge whose renewal density decays exponentially on the
short-time source chart.  Integrating the tree majorant gives an $L^1$ anchor
density and the displayed second/fourth moment bounds, including intervals
sharing an endpoint.  Boundary atoms are already included in the flux trace
measure and contribute the term of order $\mu_\varepsilon^{-1}|t-s|$.
\end{proof}

\begin{theorem}[Prepared density--contact Gaussian process]
\label{thm:r10-b3-main}
The exactly centered microscopic fields converge in the weighted nuclear dual
path space to the unique Gaussian balanced process with covariance $\Sigma$.
Its Cameron--Martin space is the quotient form domain of
Theorem~\ref{thm:r10-b3-mosco}; the initial covariance is the B1
microcanonical Schur complement.
\end{theorem}

\begin{proof}
Finite-dimensional convergence is Theorem~\ref{thm:r10-b3-finite}.
Lemma~\ref{lem:r10-b3-cumulants} gives Aldous estimates on every member of a
countable nuclear test core; Mitoma's criterion gives tightness.  The weak
balance passes to the limit by closedness of B2's Green graph.  The limiting
finite-dimensional laws and balance determine a unique Gaussian process.
Theorem~\ref{thm:r10-b3-mosco} identifies its Cameron--Martin form, and B1
identifies the prepared initial block.
\end{proof}
