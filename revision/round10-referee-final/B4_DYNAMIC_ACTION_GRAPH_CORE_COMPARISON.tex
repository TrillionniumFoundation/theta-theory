\section{Round-ten reconstruction: typed microscopic objects, a graph core, and the kinetic action semigroup}
\label{sec:r10-b4}

Five objects are kept distinct: the microscopic state, the deterministic
Koopman semigroup on microscopic observables, its dual push-forward on laws,
finite-horizon log-Laplace values under an initial ensemble, and the limiting
nonlinear semigroup on kinetic densities.  The limiting semigroup is obtained
from the proved B2 action and the Laplace principle, not by applying a state
resolvent to an observable.

\subsection{Microscopic state, observable, and law semigroups}

Let $E_\varepsilon$ be the graph-completed hard-sphere phase space augmented
by the cumulative actual-contact ledger, and let $\Phi_{s,t}^\varepsilon$ be
the deterministic flow.  Define
\[
 P_{s,t}^\varepsilon F(x)=F(\Phi_{s,t}^\varepsilon x),
 \qquad
 P_{s,t}^{\varepsilon *}\rho=(\Phi_{s,t}^\varepsilon)_\#\rho.
\]
The first acts on observables, the second on laws.  If $G$ is a terminal
macro-observable and $\rho$ an initial ensemble, put
\[
 \mathcal V_{s,t}^\varepsilon(G;\rho)
 =\mu_\varepsilon^{-1}
 \log\int_{E_\varepsilon}
 e^{\mu_\varepsilon G(X_t^\varepsilon(x))}\,d\rho(x).
\]

\begin{proposition}[Typed finite-horizon tower]
\label{prop:r10-b4-tower}
The Koopman and push-forward families satisfy their exact dual towers.  The
finite-horizon value obeys the Bayes tower
\[
 \mathcal V_{r,t}^\varepsilon(G;\rho)
 =\mathcal V_{r,s}^\varepsilon
 \left(
  x\mapsto
  \mathcal V_{s,t}^\varepsilon(G;\delta_x);
  \rho
 \right),
\]
with the inner function understood before the outer logarithm.  No law
functional is evaluated at a microscopic point, and no push-forward
resolvent is applied to an observable.
\end{proposition}

\begin{proof}
Both linear towers follow from
$\Phi_{r,t}=\Phi_{s,t}\circ\Phi_{r,s}$.  Disintegrate the initial integral at
time $s$ and apply the elementary identity
$e^{\mu\mathcal V_{s,t}(G;\delta_x)}=
 P_{s,t}^\varepsilon(e^{\mu G})(x)$.  This proves the displayed formula with
all domains typed.
\end{proof}

\subsection{A genuine observable graph core}

Let $A_\varepsilon$ be the closed generator of the Koopman semigroup on the
weighted observable space.  For $\lambda>0$ define the observable resolvent
\[
 R_{\lambda,\varepsilon}^{\rm obs}F
 =\int_0^\infty e^{-\lambda t}P_{0,t}^\varepsilon F\,dt.
\]
Let $\mathcal D_0$ be the smooth weighted configuration/contact cylinders
satisfying the specular matching condition, and set
\[
 \mathcal C_\varepsilon
 =\operatorname{span}\{R_{\lambda,\varepsilon}^{\rm obs}F:
  F\in\mathcal D_0,\ \lambda\in\mathbb Q_+\}.
\]

\begin{theorem}[Observable resolvent core]
\label{thm:r10-b4-core}
$\mathcal C_\varepsilon\subset D(A_\varepsilon)$ and
\[
 (\lambda-A_\varepsilon)
 R_{\lambda,\varepsilon}^{\rm obs}F=F.
\]
Its closure in the graph norm is $D(A_\varepsilon)$.  Smooth bounded
functional calculus on finite collections of core functions preserves the
specular graph domain.  No entire cutoff is used.
\end{theorem}

\begin{proof}
The B2 reflected transport semigroup is strongly continuous on observables,
so Hille--Yosida gives the resolvent identity.  The range of a fixed
$\lambda-A_\varepsilon$ on the dense cylinder class is dense in the
observable space; applying the bounded inverse proves graph-core density.
The ordinary chain rule holds on smooth compatible cylinders, while specular
matching is preserved because both one-sided traces have equal cylinder
values and equal tangential derivatives.  Approximation in the resolvent
range passes this property to the graph closure.
\end{proof}

\subsection{Finite-generator correctors and the cylinder Hamiltonian}

For a kinetic cylinder $F$ let $F\circ X^\varepsilon$ be its microscopic
lift.  Expand the B2 marked pressure on a deterministic time block and denote
by $D_{j,\varepsilon}$ the connected $j$-label defect after lower orders have
been removed.  The full finite hierarchy is triangular on the factorial
label scale; let $V_\varepsilon(t,s)$ be its backward observable evolution,
not a closed order-$j$ evolution.

\begin{lemma}[Full-hierarchy terminal corrector]
\label{lem:r10-b4-corrector}
The vector corrector
\[
 c_\varepsilon(t)
 =\int_t^T V_\varepsilon(t,s)D_\varepsilon(s)\,ds
\]
belongs to the product observable graph domain, has zero terminal value, and
solves
\[
 \partial_tc_\varepsilon+A_\varepsilon(t)c_\varepsilon
 =-D_\varepsilon(t).
\]
Its factorial connected expansion and the expansion of its generator image
are normally summable on a slightly smaller label weight.
\end{lemma}

\begin{proof}
B2's positive boundary-renewal hierarchy constructs the full product
evolution by Picard iteration.  The factorial estimate
$j!C^jT^{j-1}$ is summable after multiplication by
$\mu_\varepsilon^{1-j}$ and one loss of label weight.  Differentiating the
Duhamel integral gives the equation in the closed graph domain.  The equation
itself expresses the generator image as a normally summable defect plus a
time derivative, so the infinite corrector may be interchanged with the
finite generator.
\end{proof}

\begin{theorem}[Cylinder Hamiltonian convergence]
\label{thm:r10-b4-hamiltonian}
For every bounded-increment kinetic cylinder there are corrected microscopic
observables $F_\varepsilon\in D(A_\varepsilon)$ such that the finite-volume
source derivative of the logarithmic ensemble value converges locally
uniformly to
\[
 \mathbb H(f,p,\psi)
 =\langle f,v\cdot\nabla_xp\rangle
 +\int\left(e^{\Delta p+\psi}-1\right)dA_f.
\]
The statement concerns ensemble log-Laplace derivatives and does not identify
the deterministic push-forward on laws with a nonlinear Markov generator.
\end{theorem}

\begin{proof}
Insert the full-hierarchy corrector into the exact B2 connected source
expansion.  It cancels every correlation defect in the deterministic block.
Creation trees converge by collision coarea to the displayed exponential,
while the integrated first-surplus estimate removes cycles.  Normal complex
convergence permits the source derivative and corrector sum to pass to the
limit.
\end{proof}

\subsection{The dynamic action semigroup}

For a balanced pair on $[s,t]$ define only the dynamic cost
\[
 \mathcal A_{s,t}(f,\Gamma)
 =\int_s^t\!\int
 \ell\left(\frac{d\Gamma}{dA_f}\right)dA_f.
\]
The prepared initial rate is charged once at time zero.

\begin{theorem}[Additive kinetic Lax--Oleinik semigroup]
\label{thm:r10-b4-action}
For bounded uniformly continuous terminal data,
\[
 (S_{s,t}\Phi)(f_s)
 =\sup_{(f,\Gamma):f(s)=f_s}
 \{\Phi(f_t)-\mathcal A_{s,t}(f,\Gamma)\}
\]
is finite on moment sublevels, has compact relaxed maximizer sets, and
satisfies $S_{r,t}=S_{r,s}S_{s,t}$.  The prepared value is
\[
 \sup_{f_0}\{S_{0,T}\Phi(f_0)-I_{0,a}(f_0)\}.
\]
\end{theorem}

\begin{proof}
B2 gives compact action sublevels and closed balance.  Restriction and
concatenation of a balanced pair preserve the intermediate trace and add the
two entropy integrals, proving both dynamic-programming inequalities.  The
initial rate is absent from every transition interval and appears only in the
last displayed boundary optimization.
\end{proof}

\subsection{Two-scale comparison}

For $M>0$, restrict $q=d\Gamma/dA_f$ to
$e^{-M}\le q\le e^M$ and denote the Hamiltonian by $\mathbb H_M$.  Let
$d_\lambda$ be the Tataru distance generated by the resolvent core truncated
at slope $\lambda$.

\begin{theorem}[Coercive diagonal comparison]
\label{thm:r10-b4-comparison}
For fixed $M$, bounded upper and lower viscosity solutions of
\[
 \partial_tU+\mathbb H_M(f,DU)=0
\]
satisfy comparison.  One may use the penalty
\[
 \frac{d_\lambda(f,g)^2}{2\epsilon}
 +\delta\bigl(\Upsilon(f)+\Upsilon(g)\bigr),
\]
first sending $\epsilon\downarrow0$ at fixed $\lambda$, and only then
$\lambda\uparrow\infty$.  The first limit forces the diagonal; the second
recovers the full source topology without ever evaluating the Hamiltonian on
an uncontrolled test jet.
\end{theorem}

\begin{proof}
At a doubled maximum the coercive quadratic penalty implies
$d_\lambda(f,g)=O(\sqrt\epsilon)$.  Its derivative belongs to the finite
resolvent core and has slope bounded by $\lambda$.  For fixed $(M,\lambda)$,
Hamiltonian continuity on a common moment sublevel is uniform, so subtraction
of the viscosity inequalities and $\epsilon\downarrow0$ gives the
contradiction.  Monotonicity of $d_\lambda$ and density of the resolvent core
then permit $\lambda\uparrow\infty$; the Lyapunov term is removed last.  No
post hoc saturation changes the maximizing penalty.
\end{proof}

\begin{theorem}[Untruncated semigroup and microscopic Laplace limit]
\label{thm:r10-b4-main}
The truncated semigroups $S^M$ converge locally uniformly to $S$ as
$M\to\infty$, comparison passes to the entropy-coercive full Hamiltonian, and
for every prepared regular initial family
\[
 \mathcal V_{0,T}^\varepsilon(\Phi\circ X^\varepsilon;\rho_\varepsilon)
 \longrightarrow
 \sup_{f_0}\{S_{0,T}\Phi(f_0)-I_{0,a}(f_0)\}.
\]
\end{theorem}

\begin{proof}
Entropy coercivity makes the action carried by
$q\notin[e^{-M},e^M]$ vanish uniformly on compact action sublevels after
truncation and exact B2 balance repair.  Hence $S^M\to S$ locally uniformly;
stability of viscosity solutions and Theorem~\ref{thm:r10-b4-comparison}
give full comparison.  The final convergence is the Laplace principle for
the B1--B2 prepared good LDP.  It uses the correctly typed finite-horizon
ensemble value from Proposition~\ref{prop:r10-b4-tower}, not an ill-typed law
semigroup.
\end{proof}
