
\section{Five physical coefficients from one boundary coordinate}
\label{sec:lattice}

Let \(\mathcal H=\ell^2(\N_0;\R)\) and \(\Hh=\mathcal H\oplus\mathcal H\),
with state \(z=(q,v)\).  Put \(Pq=\ip{q}{e_0}e_0\) and \(Q=I-P\).  The
half-line graph Laplacian is
\[
 (Lq)_0=q_0-q_1,
 \qquad
 (Lq)_j=2q_j-q_{j-1}-q_{j+1}\quad(j\ge1).
\]
It is bounded, nonnegative, and self-adjoint, with \(\norm L\le4\).

The unknown parameter is
\[
 \vartheta=(c,k,\epsilon,c_b,k_b)\in
 \Theta=[c_-,c_+]\times[k_-,k_+]\times
 [\epsilon_-,\epsilon_+]\times[c_{b,-},c_{b,+}]
 \times[k_{b,-},k_{b,+}],
 \qquad \epsilon_->0.
\]
Define
\[
 C_\vartheta=cP+c_bQ,\qquad
 K_\vartheta=kP+k_bQ+\epsilon L,\qquad
 A_\vartheta=\begin{pmatrix}0&I\\-K_\vartheta&-C_\vartheta\end{pmatrix},
 \qquad B=\binom0{e_0},\qquad \ell(q,v)=q_0.
\]
The forced equations are
\begin{equation}
 \dot q=v,\qquad
 \dot v=-C_\vartheta v-K_\vartheta q+e_0u(t).
 \label{eq:five-mechanics}
\end{equation}
Thus all five parameters occur in the physical generator.  The bath damping
and pinning are no longer known constants.

\subsection{Uniform energy and response smoothness}

Set
\[
 c_* =\min(c_-,c_{b,-}),\qquad
 c^*=\max(c_+,c_{b,+}),\qquad
 k_* =\min(k_-,k_{b,-}),\qquad
 k^*=\max(k_+,k_{b,+})+4\epsilon_+.
\]

\begin{lemma}[Uniform physical stability]
\label{lem:five-energy}
There are \(M\ge1\) and \(\lambda>0\), depending only on the parameter box,
such that
\[
 \norm{e^{tA_\vartheta}}\le Me^{-\lambda t}
 \quad(t\ge0,\ \vartheta\in\Theta).
\]
For every fixed \(r\), there are \(M_r,\lambda_r>0\) for which
\[
 \sup_{\vartheta\in\Theta}
 \norm{\partial_\vartheta^\alpha e^{tA_\vartheta}}
 \le M_re^{-\lambda_rt},
 \qquad \abs\alpha\le r.
\]
All fixed-order parameter derivatives of a bounded-input trajectory are
uniformly bounded.
\end{lemma}

\begin{proof}
Choose
\(\delta=\min(c_*/2,\sqrt{k_*}/2)\) and define
\[
 \mathcal E_\vartheta(q,v)
 =\tfrac12\norm v^2+\tfrac12\ip{K_\vartheta q}{q}
 +\delta\ip qv+\tfrac\delta2\ip{C_\vartheta q}{q}.
\]
Uniform coercivity and boundedness follow from
\(k_*I\le K_\vartheta\le k^*I\) and
\(c_*I\le C_\vartheta\le c^*I\).  Along the homogeneous motion,
\[
 \mathcal E_\vartheta'
 =-\ip{(C_\vartheta-\delta I)v}{v}
  -\delta\ip{K_\vartheta q}{q}
 \le-\eta\norm{(q,v)}^2,
 \quad \eta=\min(c_*-\delta,\delta k_*)>0.
\]
Equivalence of \(\mathcal E_\vartheta\) with the squared Hilbert norm gives
the first estimate.  Every first parameter derivative of \(A_\vartheta\) is
a bounded operator of norm bounded uniformly, and every higher derivative is
zero.  Repeated Duhamel differentiation produces finite sums of ordered
integrals bounded by \(C_rt^re^{-\lambda t}\); a smaller exponential rate
absorbs the polynomial.  Convolution with a bounded input proves the final
claim.
\end{proof}

Define the boundary step response
\[
 h_\vartheta(t)=\ell\int_0^te^{sA_\vartheta}B\,ds
 =\sum_{j\ge0}\frac{r_j(\vartheta)t^{j+1}}{(j+1)!},
 \qquad r_j(\vartheta)=\ell A_\vartheta^jB.
\]

\subsection{A triangular eight-jet coordinate system}

Put \(s=k+\epsilon\).  Direct application of the banded generator gives
\begin{equation}
\begin{aligned}
 r_0&=0,&r_1&=1,&r_2&=-c,\\
 r_3&=c^2-s,&
 r_4&=c(-c^2+2s),\\
 r_5&=c^4-3c^2s+s^2+\epsilon^2,\\
 r_6&=-c^5+4c^3s-2c\epsilon^2-3cs^2-c_b\epsilon^2,\\
 r_7&=c^6-5c^4s+3c^2\epsilon^2+6c^2s^2+2cc_b\epsilon^2
      +c_b^2\epsilon^2-2\epsilon^3-2\epsilon^2s-s^3-\epsilon^2k_b.
\end{aligned}
\label{eq:eight-jets}
\end{equation}
The first bath coefficient \(c_b\) appears at order \(t^7\), and the bath
pinning \(k_b\) at order \(t^8\).  This is why two short durations cannot
identify the five-parameter model.

Let
\[
 J(\vartheta)=(r_2,r_3,r_5,r_6,r_7)\in\R^5.
\]
For \(x=(x_2,x_3,x_5,x_6,x_7)\), define successively
\begin{align}
 c(x)&=-x_2, & s(x)&=c(x)^2-x_3,\notag\\
 \epsilon(x)&=\sqrt{x_5-c(x)^4+3c(x)^2s(x)-s(x)^2},
 \label{eq:jet-inverse-one}\\
 k(x)&=s(x)-\epsilon(x),\notag\\
 c_b(x)&=\frac{-c(x)^5+4c(x)^3s(x)-2c(x)\epsilon(x)^2
                -3c(x)s(x)^2-x_6}{\epsilon(x)^2},
 \label{eq:jet-inverse-two}\\
 k_b(x)&=\epsilon(x)^{-2}\bigl[c(x)^6-5c(x)^4s(x)
 +3c(x)^2\epsilon(x)^2+6c(x)^2s(x)^2
 +2c(x)c_b(x)\epsilon(x)^2+c_b(x)^2\epsilon(x)^2\notag\\
 &\hspace{37mm}-2\epsilon(x)^3-2\epsilon(x)^2s(x)-s(x)^3-x_7\bigr].
 \label{eq:jet-inverse-three}
\end{align}

\begin{lemma}[Global jet inversion]
\label{lem:jet-inversion}
The map \(J:\Theta\to\R^5\) is one-to-one and has a \(C^\infty\) inverse on
an open neighbourhood of \(J(\Theta)\).  There is a globally bounded
\(C^2\) map \(G:\R^5\to\R^5\) which agrees with \(J^{-1}\) on a
neighbourhood of \(J(\Theta)\).
\end{lemma}

\begin{proof}
Equations \eqref{eq:jet-inverse-one}--\eqref{eq:jet-inverse-three} recover
\((c,k,\epsilon,c_b,k_b)\) from \(J(\vartheta)\).  The radicand equals
\(\epsilon^2\ge\epsilon_-^2\), so the positive square root is smooth in a
neighbourhood of the compact image.  Division by \(\epsilon^2\) is uniformly
safe.  The displayed formulas therefore define a smooth inverse on some open
set \(U\supset J(\Theta)\).  A compactly supported smooth cutoff equal to one
near \(J(\Theta)\), with support in \(U\), extends the inverse to a globally
bounded \(C^2\) map.
\end{proof}

\subsection{Six durations recover the jets}

Fix \(\tau>0\) and let \(t_j=j\tau\), \(j=1,\ldots,6\).  Remove the
parameter-free leading term:
\[
 y_j(\vartheta)=h_\vartheta(t_j)-\tfrac12t_j^2.
\]
Let \(V\) be the \(6\times6\) matrix
\[
 V_{j,\ell-1}=\frac{j^{\ell+1}}{(\ell+1)!},
 \qquad j=1,\ldots,6,\quad \ell=2,\ldots,7,
\]
and \(D_\tau=\diag(\tau^3,\ldots,\tau^8)\).  Dividing row \(j\) by
\(j^3\) reduces the determinant of \(V\), up to nonzero factors, to the
ordinary Vandermonde determinant on \(1,\ldots,6\); hence \(V\) is
invertible.  Define the recovered jet vector
\begin{equation}
 \widehat R_\tau(\vartheta)=D_\tau^{-1}V^{-1}
 (y_1(\vartheta),\ldots,y_6(\vartheta))^T.
 \label{eq:recovered-jets}
\end{equation}
Its coordinates correspond to \((r_2,\ldots,r_7)\).

\begin{lemma}[Uniform jet interpolation]
\label{lem:jet-interpolation}
There is \(C<\infty\) such that, for \(0<\tau\le1\),
\[
 \norm{\widehat R_\tau-(r_2,\ldots,r_7)}_{C^1(\Theta)}\le C\tau.
\]
\end{lemma}

\begin{proof}
The omitted series begins with \(r_8t^9/9!\).  Lemma~\ref{lem:five-energy},
or directly the boundedness of \(A_\vartheta\), gives a uniform \(C^1\)
bound \(C t^9\) for the response remainder on \(0\le t\le6\).  The fixed
matrix \(V^{-1}\) preserves that bound.  The largest amplification in
\(D_\tau^{-1}\) is \(\tau^{-8}\), leaving \(C\tau\).
\end{proof}

Let \(P:\R^6\to\R^5\) select coordinates \((r_2,r_3,r_5,r_6,r_7)\), and
put
\begin{equation}
 F_\tau(\vartheta)=G(P\widehat R_\tau(\vartheta)),\qquad
 H_\tau(\vartheta)=(h_\vartheta(\tau),\ldots,h_\vartheta(6\tau)).
 \label{eq:six-response-map}
\end{equation}

\begin{theorem}[Global six-duration mechanical embedding]
\label{thm:jet-embedding}
There exist \(\tau_0>0\) and \(b_\tau>0\) such that for every
\(0<\tau\le\tau_0\) and \(\vartheta,\vartheta'\in\Theta\),
\begin{equation}
 \abs{H_\tau(\vartheta)-H_\tau(\vartheta')}
 \ge b_\tau\abs{\vartheta-\vartheta'},
 \qquad
 \abs{DH_\tau(\vartheta)v}\ge b_\tau\abs v.
 \label{eq:six-bi-lipschitz}
\end{equation}
One may take
\[
 b_\tau=\frac1{2\Lip(GPV_\tau)},
 \quad
 V_\tau=D_\tau^{-1}V^{-1},
\]
where the Lipschitz norm is evaluated on the response range.  In particular
\(b_\tau>0\) for every fixed admissible \(\tau\).
\end{theorem}

\begin{proof}
By Lemma~\ref{lem:jet-interpolation} and bounded derivatives of \(G\),
\[
 \norm{F_\tau-\Id}_{C^1(\Theta)}\le C\tau.
\]
Choose \(\tau_0\) so that this norm is at most \(1/2\).  Integration along a
line segment in the convex box gives
\(\abs{F_\tau(\vartheta)-F_\tau(\vartheta')}\ge
\abs{\vartheta-\vartheta'}/2\) and
\(\abs{DF_\tau(\vartheta)v}\ge\abs v/2\).  Since \(F_\tau\) is the
composition of \(H_\tau\) with the fixed smooth reconstruction map, its
Lipschitz constant with respect to the six response values yields
\eqref{eq:six-bi-lipschitz}.
\end{proof}

\begin{remark}[Why the bath is visible]
The triangular inverse uses \(r_6\) and \(r_7\), whose coefficients contain
\(-c_b\epsilon^2\) and \(-k_b\epsilon^2\).  Thus the proof does not identify
the boundary pair while treating the bath as known.  The same bath parameters
control the memory kernel in Section~\ref{sec:memory}.
\end{remark}

\subsection{Balanced calibration under arbitrary feedback exploitation}
\label{sec:balanced-design}

Fix a calibration amplitude \(a>0\), an exploitation-force bound
\(U<\infty\), exploitation-duration bounds
\(0<\tau_{\min}\le\tau_{\max}<\infty\), a calibration density
\(\rho\in(0,1]\), and a prefix defect \(C_{\rm cal}<\infty\).  These
constants are common to the entire triangular class, including policies that
depend on the terminal size \(n\).  Exploitation blocks may use arbitrary
feedback forces bounded in absolute value by \(U\), and their durations lie
in \([\tau_{\min},\tau_{\max}]\).  All policy randomizers and the complete
action of a block are selected before its readout noise.

Let \(N_{\rm cal}(m)\) be the number of calibration slots among the first
\(m\) blocks.  The admissible policy class is defined by the finite-prefix
condition
\begin{equation}
 N_{\rm cal}(m)\ge \rho m-C_{\rm cal}
 \qquad(1\le m\le n),
 \label{eq:prefix-calibration}
\end{equation}
not merely by a terminal or asymptotic density.  Calibration slots, in their
own chronological order, are partitioned into consecutive windows of six.
At a calibration slot \(i\), the duration \(s_i\) is
\(\cF_{i-1}\)-measurable and is selected from the durations not yet used in
the current window; the six realized durations are therefore a permutation of
\(\{\tau,2\tau,\ldots,6\tau\}\).  Only after \(s_i\) is fixed is an
independent Rademacher sign \(S_i\) drawn.  Force \(aS_i\) is then held for
\(s_i\), and finally an independent \(N(0,\sigma^2)\) readout noise is drawn.
Thus the calibration chronology is
\[
 \cF_{i-1}\longrightarrow s_i\longrightarrow S_i
 \longrightarrow\xi_i\longrightarrow Y_i.
\]
The class is nonempty: a deterministic Beatty-type calibration schedule has
\(N_{\rm cal}(m)=\lfloor\rho m\rfloor\), and the six durations may be cycled
in any predictable order.

\begin{lemma}[Complete windows in every finite prefix]
\label{lem:prefix-window-count}
Let \(W_n\) be the number of complete six-slot calibration windows contained
in the first \(n\) blocks.  Then, pathwise and uniformly over the policy
class,
\begin{equation}
 W_n=\left\lfloor\frac{N_{\rm cal}(n)}6\right\rfloor
 \ge \frac{\rho n}{6}-C_{\rm win},
 \qquad C_{\rm win}=\frac{C_{\rm cal}}6+1.
 \label{eq:complete-window-count}
\end{equation}
The sole loss from the unfinished terminal window is the additive constant in
\eqref{eq:complete-window-count}.
\end{lemma}

\begin{proof}
Use \eqref{eq:prefix-calibration} at \(m=n\) and
\(\lfloor x\rfloor\ge x-1\).  The window partition is chronological, so
exactly \(\lfloor N_{\rm cal}(n)/6\rfloor\) windows are complete.
\end{proof}

The state is never reset.  Let \(m_i^\vartheta\) be the zero-initial-state
boundary mean generated by the entire realized action record.  At a
calibration slot with duration \(s_i\), for any two parameters,
\begin{equation}
 m_i^\vartheta-m_i^{\vartheta_0}
 =D_{i,s_i}(\vartheta,\vartheta_0)
  +aS_i\{h_\vartheta(s_i)-h_{\vartheta_0}(s_i)\},
 \label{eq:sign-decomposition}
\end{equation}
where \(D_{i,s_i}\) is \(\cF_{i-1}\)-measurable because the duration
was fixed before the fresh sign.  Therefore
\begin{equation}
 \E\left[(m_i^\vartheta-m_i^{\vartheta_0})^2
 \mid\cF_{i-1}\right]
 =D_{i,s_i}^2+a^2\{h_\vartheta(s_i)-h_{\vartheta_0}(s_i)\}^2.
 \label{eq:sign-square}
\end{equation}
Summing a complete window and using Theorem~\ref{thm:jet-embedding} gives
\begin{equation}
 \sum_{i\in W}\E[(m_i^\vartheta-m_i^{\vartheta_0})^2
 \mid\cF_{i-1},s_i]
 \ge a^2b_\tau^2\abs{\vartheta-\vartheta_0}^2.
 \label{eq:window-contrast}
\end{equation}
The same inequality holds for the tangent sensitivity in every direction.

\begin{proposition}[Full-index empirical contrast]
\label{prop:balanced-contrast}
Put
\[
 \bar c_\tau=\frac{\rho a^2b_\tau^2}{6}.
\]
For every fixed \(c_*\in(0,\bar c_\tau)\) there are
\(C_1,C_2>0\), independent of the policy, true parameter, true initial state,
and sample size, such that
\begin{equation}
 P_E\left\{
 \inf_{\vartheta\ne\vartheta_0}
 \frac{n^{-1}\sum_{i=1}^n
 (m_i^\vartheta-m_i^{\vartheta_0})^2}
      {\abs{\vartheta-\vartheta_0}^2}<c_*
 \right\}\le C_1e^{-C_2n}.
 \label{eq:full-empirical-contrast}
\end{equation}
On the same event,
\[
 \frac1n\sum_{i=1}^n
 \{v^TDm_i^{\vartheta_0}\}^2\ge c_*\abs v^2
 \quad\hbox{for all }v\in\R^5.
\]
The predictable contrast before martingale fluctuation is
\(\bar c_\tau n-O(1)\), with the explicit incomplete-window loss
\(a^2b_\tau^2C_{\rm win}\).
\end{proposition}

\begin{proof}
The compact radial index set is
\[
 \mathscr K=\{(\theta_0,r,v):\theta_0\in\Theta,\ v\in S^4,\ r\ge0,
 \ \theta_0+rv\in\Theta\}.
\]
At \(r=0\), define
\[
 q_i(\theta_0,0,v)=v^TDm_i^{\theta_0};
\]
for \(r>0\), set
\[
 q_i(\theta_0,r,v)=
 \frac{m_i^{\theta_0+rv}-m_i^{\theta_0}}r
 =\int_0^1v^TDm_i^{\theta_0+srv}\,ds.
\]
Let \(\bar U=\max(a,U)\).  Uniform exponential stability and Duhamel
iteration show that every fixed-order parameter derivative of the forced
trajectory is bounded by a deterministic constant depending only on
\(\bar U,\tau_{\max},\tau\) and the parameter box, not on the number or
ordering of blocks.  Hence \(q_i^2\) is uniformly bounded and uniformly
Lipschitz on \(\mathscr K\), simultaneously in \(i,n\) and the policy.
The continuous definition at \(r=0\) makes the true parameter and tangent
boundary part of the same compact net.

For each \(\kappa\in\mathscr K\), put
\[
 X_i(\kappa)=q_i(\kappa)^2-
 \E\{q_i(\kappa)^2\mid\cF_{i-1}\}.
\]
Because the current action is chosen before the readout and \(q_i\) is a
function of the action record, \((X_i(\kappa),\cF_i)\) is a bounded
martingale-difference sequence.  At every complete calibration window,
\eqref{eq:sign-square} and Theorem~\ref{thm:jet-embedding}, divided by
\(r^2\) and continuously extended to \(r=0\), give
\[
 \sum_{i\in W}\E\{q_i(\kappa)^2\mid\cF_{i-1}\}
 \ge a^2b_\tau^2.
\]
Lemma~\ref{lem:prefix-window-count} therefore yields, uniformly in
\(\kappa\),
\begin{equation}
 \sum_{i=1}^n\E\{q_i(\kappa)^2\mid\cF_{i-1}\}
 \ge \bar c_\tau n-a^2b_\tau^2C_{\rm win}.
 \label{eq:predictable-prefix-contrast}
\end{equation}
Noncalibration blocks contribute nonnegative conditional squares.

Let \(Q\) bound \(\abs{q_i}\), and let \(L_X\) be a common Lipschitz
constant for \(X_i\); conditional expectation is a contraction in supremum
norm, so \(L_X\) is deterministic.  Fix \(c_*<\bar c_\tau\), choose
\(\eta>0\) and a mesh \(\delta>0\) so that
\(\eta+L_X\delta<(\bar c_\tau-c_*)/2\), and take a finite
\(\delta\)-net of \(\mathscr K\).  At each net point, Azuma--Hoeffding gives
\[
 P_E\left\{\sum_{i=1}^nX_i<-\eta n\right\}
 \le \exp\{-\eta^2n/(8Q^4)\}.
\]
A union bound over the fixed net and Lipschitz interpolation give, outside an
event \(C'_1e^{-C'_2n}\),
\[
 \inf_{\kappa\in\mathscr K}\frac1n\sum_{i=1}^nX_i(\kappa)
 \ge-\eta-L_X\delta.
\]
Combining this with \eqref{eq:predictable-prefix-contrast} proves the desired
lower bound for all sufficiently large \(n\).  Enlarging \(C_1\) absorbs the
finitely many smaller sizes.  Substitution of
\(r=\abs{\vartheta-\vartheta_0}\) proves
\eqref{eq:full-empirical-contrast}; \(r=0\) gives the simultaneous tangent
information inequality.
\end{proof}

\subsection{The five-parameter quasi-posterior theorem}

At block \(i\), the complete trajectory under the realized actions is
\[
 x_i^\vartheta=\int_0^{t_i}e^{(t_i-s)A_\vartheta}Bu(s)ds,
 \qquad m_i^\vartheta=\ell x_i^\vartheta,
 \qquad b_i^\vartheta(z)=\ell e^{t_iA_\vartheta}z.
\]
Every block duration is bounded below by
\(\underline\tau=\min(\tau,\tau_{\min})>0\), hence the physical endpoint
time satisfies \(t_i\ge i\underline\tau\).  Lemma~\ref{lem:five-energy}
therefore gives
\[
 \sup_{\vartheta,\norm z\le R}
 \abs{\partial_\vartheta^\alpha b_i^\vartheta(z)}
 \le C_re^{-\lambda_ri\underline\tau}.
\]
The nuisance is therefore summable to every fixed derivative order.  Define
\[
 I_n=\frac1{n\sigma^2}\sum_{i=1}^n
 Dm_i^{\vartheta_0}(Dm_i^{\vartheta_0})^T,
 \qquad
 \Delta_n=\frac1{\sigma^2\sqrt n}
 \sum_{i=1}^nDm_i^{\vartheta_0}\xi_i.
\]

\begin{theorem}[Five-parameter mechanical quasi-Bernstein--von Mises]
\label{thm:lattice-bvm}
Let \(\vartheta_0\) range over a compact subset of the interior of \(\Theta\),
let \(\norm{z_0}\le R\), let \(\nu_n\) be any Borel probability supported in
the same Hilbert ball, and let the policy range over the balanced class above,
including sample-size-dependent policies.  Then
\begin{equation}
 \TV\left(
 \mathcal L\{\sqrt n(\vartheta-\vartheta_0)\mid\cF_n\},
 N(I_n^{-1}\Delta_n,I_n^{-1})
 \right)\longrightarrow0
 \label{eq:five-bvm}
\end{equation}
in probability uniformly over this class.  The evidence obeys
\begin{align}
 \frac{\mathcal Z_n^{\nu_n}}{L_n^{\nu_n}(\vartheta_0)}
 ={}&\frac{(2\pi)^{5/2}\pi(\vartheta_0)}{n^{5/2}\sqrt{\det I_n}}
 \exp\{\tfrac12\Delta_n^TI_n^{-1}\Delta_n\}
 [1+o_{\frE}(1)].
 \label{eq:five-evidence}
\end{align}
The conclusions do not require convergence of \(I_n\).  If the working
initial-state prior excludes \(z_0\), they are frequentist asymptotics for the
stated quasi-posterior, whose pseudo-true parameter differs from
\(\vartheta_0\) by \(O(n^{-1})\).
\end{theorem}

\begin{proof}
Uniform response derivatives follow from Lemma~\ref{lem:five-energy};
transient summability was just established; and
Proposition~\ref{prop:balanced-contrast} proves the global contrast with
exponentially small failure probability.  All hypotheses of
Theorem~\ref{thm:abstract-bvm} hold with \(p=5\).  Equations
\eqref{eq:five-bvm}--\eqref{eq:five-evidence} and the pseudo-true statement
are exactly Theorem~\ref{thm:abstract-bvm} and
Proposition~\ref{prop:pseudotrue} in this model.
\end{proof}

\begin{remark}[What the design does and does not impose]
The finite-prefix lower bound \eqref{eq:prefix-calibration}, rather than a
bare terminal liminf, is the exact uniformity condition needed when policies
may depend on \(n\).  It still avoids the stronger requirement that every
duration--sign atom have positive conditional mass at every block.  The order
inside each complete window may be selected adaptively; the fresh sign is the
randomizer that prevents cancellation by the existing state.
\end{remark}
