\begin{abstract}
\noindent\textbf{Platform: }\texttt{OB3-PE-v1}.\par
We derive a physical stochastic limit from the real specular path ensembles of
Paper I.  Fix a geometry parameter $\rho$ and a prepared impact frequency
$r$.  The conjugate field $\theta=I_\rho'(r)$ selects an equilibrium state on
the same deterministic billiard.  The centered mechanical collision current
$g=J-r$ has collision variance
$\Sigma_{\rho,r}=\Lambda_\rho''(\theta)>0$, exactly the pressure Hessian.

The fast orbit is deterministic after its initial point is chosen.  A
vector-valued almost sure invariance principle for the current and the actual
free-flight roof yields collision-clock and physical-time Brownian limits.  If
$\bar\tau_{\rho,r}$ is the mean roof, the physical covariance is
$A_{\rho,r}=\Sigma_{\rho,r}/\bar\tau_{\rho,r}$.  The same coefficient is the
second derivative of the zero-pressure physical cumulant root.

A joint spectral moment bound gives sub-Gaussian control of the inverse clock.
We prove an exponential random-window theorem strong enough to pass
Doob-normalized entropic values from a random physical horizon to a
deterministic collision horizon.  No microscopic noise or substituted roof is
used.
\end{abstract}

\section{Prepared deterministic billiard dynamics}\label{sec:prepared}
Fix $\rho\in I_\rho$ and $r\in(0,1/2)$.  Put
\[
 \theta=\theta_\rho(r),\qquad
 \nu=\nu_{\rho,\theta},\qquad
 \tau=\widehat\tau_\rho,
\]
where the objects are those of Paper I.  Let
\[
 g=J-r.
\]
Then $g$ is a locally constant mechanical impact fluctuation and
\[
 \int g\,d\nu=0.
\]
Given $\omega\in\Sigma$, the sequence
$g(\sigma^k\omega),\tau(\sigma^k\omega)$ is deterministic and is the symbolic
image of one actual specular billiard orbit.

\begin{lemma}[Transfer from the one-sided roof to the physical roof]\label{lem:roof-cohomology}
Let $\tau^{\rm phys}_\rho$ be the two-sided actual free-flight roof.  There is
a bounded Holder function $U_\rho$ such that
\[
 \widehat\tau_\rho
 =\tau^{\rm phys}_\rho+U_\rho-U_\rho\circ\sigma.
\]
Hence for every $n$,
\[
 \abs{S_n\widehat\tau_\rho-S_n\tau^{\rm phys}_\rho}
 \le2\norm{U_\rho}_\infty.
\]
The two inverse collision clocks differ by a uniformly bounded number of
collisions.  Therefore all diffusive, rough, and exponential random-window
statements proved with $\widehat\tau_\rho$ transfer to the actual physical
roof, with at most $O(\varepsilon)$ additional slow-path error.
\end{lemma}
\begin{proof}
The standard Sinai cohomology construction replaces a two-sided Holder
observable by a one-sided representative.  The sum identity telescopes.  Since
both roofs are bounded below by $\tau_*>0$, a bounded time discrepancy changes
the inverse count by at most a uniform constant.  Each slow collision jump is
$O(\varepsilon)$ and each drift/work contribution is $O(\varepsilon^2)$.
\end{proof}

\section{Collision pressure and Green--Kubo variance}\label{sec:variance}
Define the collision cumulant pressure
\begin{equation}\label{eq:K}
 K_{\rho,r}(q)
 =P_\sigma(\varphi_\rho+(\theta+q)J)
 -P_\sigma(\varphi_\rho+\theta J)-qr.
\end{equation}

\begin{theorem}[Pressure--variance identity]\label{thm:variance}
The function $K_{\rho,r}$ is analytic near zero and
\[
 K_{\rho,r}'(0)=0,
 \qquad
 \Sigma_{\rho,r}:=K_{\rho,r}''(0)
 =\Lambda_\rho''(\theta)>0.
\]
Moreover
\[
 \Sigma_{\rho,r}
 =\int g^2\,d\nu
 +2\sum_{n\ge1}\int g\,g\circ\sigma^n\,d\nu
 =\lim_{n\to\infty}\frac1n
 \int(S_ng)^2\,d\nu.
\]
\end{theorem}
\begin{proof}
Equation \eqref{eq:K} is
$\Lambda_\rho(\theta+q)-\Lambda_\rho(\theta)-qr$.
Differentiate and use $r=\Lambda_\rho'(\theta)$.  The Ruelle resolvent
expansion of the second derivative gives the absolutely convergent
Green--Kubo series and the variance limit.  Positivity was proved in Paper I
from the physical period-two orbits.
\end{proof}

\section{Joint invariance principle with the actual roof}\label{sec:joint}
Put
\[
 \bar\tau_{\rho,r}=\int\tau\,d\nu,
 \qquad \tau^\circ=\tau-\bar\tau_{\rho,r}.
\]

\begin{theorem}[Joint deterministic ASIP]\label{thm:ASIP}
There is a two-dimensional Brownian motion $(B^g,B^\tau)$ and, on an enlarged
probability space, a coupling such that for some $\delta>0$,
\[
 \sup_{k\le n}\abs{
 \sum_{j=0}^{k-1}(g,\tau^\circ)\circ\sigma^j
 -(B^g_k,B^\tau_k)}
 =O(n^{1/2-\delta})
\]
almost surely.  In particular, the polygonal lift of
$n^{-1/2}S_{\lfloor nt\rfloor}g$ converges in every scalar rough-path
$p$-variation topology, $p>2$, to Brownian motion with variance
$\Sigma_{\rho,r}$.
\end{theorem}
\begin{proof}
The shift is a mixing subshift of finite type, $\nu$ is a Holder equilibrium
state, and $(g,\tau^\circ)$ is Holder.  The vector-valued ASIP follows from the
spectral/martingale approximation for hyperbolic systems
\cite{MelbourneNicol2009}.  The scalar enhanced statement follows from the
canonical polygonal lift.
\end{proof}

\section{Physical time and the pressure root}\label{sec:physical}
Let
\[
 S_n^\tau=\sum_{k=0}^{n-1}\tau\circ\sigma^k,
 \qquad
 N(t)=\max\{n:S_n^\tau\le t\}.
\]
Define the joint pressure
\[
 \Psi_{\rho,r}(q,s)
 =P_\sigma(\varphi_\rho+\theta J+qg-s\tau)
 -P_\sigma(\varphi_\rho+\theta J).
\]
Since
\[
 \partial_s\Psi_{\rho,r}(0,0)=-\bar\tau_{\rho,r}<0,
\]
there is an analytic root $s=\kappa_{\rho,r}(q)$ satisfying
$\Psi(q,\kappa(q))=0$.

\begin{theorem}[Physical covariance identity]\label{thm:physical-covariance}
One has
\[
 \kappa_{\rho,r}'(0)=0,
 \qquad
 A_{\rho,r}:=\kappa_{\rho,r}''(0)
 =\frac{\Sigma_{\rho,r}}{\bar\tau_{\rho,r}}>0.
\]
Moreover
\[
 n^{-1/2}S_{N(nt)}g
 \Longrightarrow
 \sqrt{A_{\rho,r}}\,W_t.
\]
\end{theorem}
\begin{proof}
Differentiate the zero-pressure equation.  The first derivative vanishes
because $g$ is centered.  At second order all terms involving
$\kappa'(0)$ vanish, leaving
$0=\Sigma_{\rho,r}-\bar\tau_{\rho,r}\kappa''(0)$.
The process convergence also follows from \cref{thm:ASIP} and continuity of
monotone inverse-clock time change.
\end{proof}

\section{A deterministic slow system}\label{sec:slow}
Let $b:\R\to\R$ be bounded and globally Lipschitz.  For a deterministic
billiard code $\omega$ define
\begin{align*}
 X_{k+1}^\varepsilon
 &=X_k^\varepsilon
 +\varepsilon g(\sigma^k\omega)
 +\varepsilon^2\tau(\sigma^k\omega)b(X_k^\varepsilon),\\
 t_{k+1}^\varepsilon
 &=t_k^\varepsilon+\varepsilon^2\tau(\sigma^k\omega).
\end{align*}

\begin{theorem}[Physical diffusion from specular collisions]\label{thm:SDE}
If the initial billiard point is prepared according to $\mu_{\rho,\theta}$,
the physical-time interpolation converges weakly to
\[
 dX_t=b(X_t)\,dt+\sqrt{A_{\rho,r}}\,dW_t.
\]
The convergence is locally uniform over initial slow states and holds jointly
with the collision and roof additive functionals.
\end{theorem}
\begin{proof}
Apply the joint ASIP to the fast observables.  The drift sum is a Riemann sum in
actual physical time, while \cref{thm:physical-covariance} gives the martingale
part.  Global Lipschitz continuity gives uniqueness and continuity of the
solution map, as in deterministic homogenization theory
\cite{KellyMelbourne2016}.
\end{proof}

\section{Uniform exponential moment control}\label{sec:mgf}
\begin{lemma}[Spectral Bernstein bound]\label{lem:bernstein}
For compact $K\Subset I_\rho\times(0,1/2)$ there are $u_0,C<\infty$ such that,
uniformly for $(\rho,r)\in K$, $m\ge1$, $\abs u\le u_0$, and every initial one-sided symbolic state $z$,
\[
 \E_z e^{uS_mg}\le C e^{C u^2m},
 \qquad
 \E_z e^{uS_m\tau^\circ}\le C e^{C u^2m}.
\]
The stationary bounds follow by integration.  The same estimates hold for
forward and backward maximal partial sums, with a possibly larger constant.
\end{lemma}
\begin{proof}
The leading eigenvalue of each Fourier--Laplace Ruelle family is analytic with
zero first derivative for the centered observable and bounded second
derivative on compact parameter sets.  The remaining spectrum is uniformly
contracting.  This gives the moment bound.  A dyadic decomposition and the
Markov property of the normalized Ruelle operator give the maximal version.
\end{proof}

\begin{corollary}[Inverse-clock concentration]\label{cor:clock}
For every compact $K$ as above there are $c,C>0$ such that
\[
 \Pp\left(
 \abs{N(t)-t/\bar\tau_{\rho,r}}>u\sqrt t
 \right)\le Ce^{-cu^2}
\]
for $t$ in bounded multiples of $\varepsilon^{-2}$ and admissible $u$, uniformly in the initial symbolic state.
\end{corollary}
\begin{proof}
Use \cref{lem:bernstein} for $S_m\tau^\circ$ and monotone inversion, noting
$\tau\ge\tau_*>0$.
\end{proof}

\section{Exponential random-window theorem}\label{sec:window}
Put
\[
 n_\varepsilon(T)=\left\lfloor
 \frac{T}{\varepsilon^2\bar\tau_{\rho,r}}\right\rfloor,
 \qquad
 N_\varepsilon(T)=N(T/\varepsilon^2).
\]

\begin{theorem}[Spectral random-window Orlicz bound]\label{thm:window}
For every compact $K\Subset I_\rho\times(0,1/2)$, $T<\infty$, and
$\lambda>0$, there is $C_{K,T,\lambda}$ such that
\[
 \sup_z\log\E_z\exp\left\{
 \lambda\varepsilon
 \abs{\sum_{k=n_\varepsilon(T)}^{N_\varepsilon(T)-1}
 g\circ\sigma^k}
 \right\}
 \le C_{K,T,\lambda}\sqrt\varepsilon,
\]
and
\[
 \sup_z\log\E_z\exp\left\{
 \lambda\varepsilon^2
 \abs{N_\varepsilon(T)-n_\varepsilon(T)}
 \right\}
 \le C_{K,T,\lambda}\varepsilon.
\]
The reversed-sum convention is used when $N_\varepsilon<n_\varepsilon$.
\end{theorem}
\begin{proof}
By \cref{cor:clock}, the random window has sub-Gaussian length with typical
size $O(\varepsilon^{-1})$.  On a deterministic window of length $m$,
\cref{lem:bernstein} and its reverse-time version give sub-Gaussian exponential
moments for the centered current.  Split the window length into dyadic shells
$2^j\varepsilon^{-1}\le m<2^{j+1}\varepsilon^{-1}$.  Holder's inequality
combines the current maximal moment with the exponentially decaying shell
probability.  The resulting series is bounded by
$\exp\{C\sqrt\varepsilon\}$.  The second estimate follows directly from the
inverse-clock tail.
\end{proof}

\section{Geometry and preparation response}\label{sec:response}
\begin{corollary}[Coefficient regularity]\label{cor:regularity}
On compact subsets of $I_\rho\times(0,1/2)$, the maps
\[
 (\rho,r)\mapsto
 \theta_\rho(r),\quad
 \Sigma_{\rho,r},\quad
 \bar\tau_{\rho,r},\quad
 A_{\rho,r}
\]
are $C^N$ for every prescribed finite $N$ allowed by the obstacle regularity.
In particular,
\[
 \partial_r\theta_\rho(r)=\Sigma_{\rho,r}^{-1}.
\]
\end{corollary}
\begin{proof}
Combine Paper I's joint perturbation theorem with
$\Sigma=\Lambda''$, the equilibrium-state response of the roof, and the
quotient formula $A=\Sigma/\bar\tau$.
\end{proof}

\section{Main theorem and scope}\label{sec:main}
\begin{theorem}[Specular deterministic-to-stochastic closure]\label{thm:main}
For every geometry $\rho$ and prepared mechanical impact frequency $r$ in the
declared compact scope, the same deterministic specular billiard supplies:
\begin{enumerate}[label=(\roman*)]
\item the canonical equilibrium path preparation selected in Paper I;
\item a centered collision current with positive pressure-Hessian variance;
\item a deterministic collision and physical-time invariance principle;
\item the actual nonconstant free-flight clock;
\item the physical diffusion coefficient
$A_{\rho,r}=\Lambda_\rho''(\theta_\rho(r))/\bar\tau_{\rho,r}$;
\item exponential random-horizon control for nonlinear path evaluations.
\end{enumerate}
No Brownian forcing or stochastic clock is present microscopically.
\end{theorem}
\begin{proof}
Combine \cref{thm:variance,thm:ASIP,thm:physical-covariance,thm:SDE,thm:window}.
\end{proof}

\section*{Scope and review status}
The result is for the geometry-selected equilibrium preparations on the
non-wandering set of a no-eclipse open billiard.  A compact periodic Sinai
Liouville theorem is not claimed.  Independent review remains pending.
