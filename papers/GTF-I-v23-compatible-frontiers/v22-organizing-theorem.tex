\section{The constructive resource theorem}
\label{sec:v23-preserved-main}
The constructive statement and its proof are retained here as a single reference point for the supporting development.

\begin{theorem}[Continuation profiles and statistical resources]
\label{thm:v22-main}
The following statements hold in the declared clocked finite-state
model.

\smallskip\noindent
\textup{(i)} Every strict $K$-response cover of envelope $v$ has, for
integers $m,n,r\ge1$ and dyadic calibration error $\rho$, a rational
whole-schedule realization with
\[
 N=n(K-1),\qquad W\le3dK(2m+1),
\]
and worst-case score at least
\[
 v-2^{1-r}-2\rho-4(K-1)\theta
       -2(K-1)\bigl(e^{-4\theta m}+e^{-2n\theta^2}\bigr).
\]
Here $K\ge2$; one response needs no training. The previously available
constant-width realization gives, in the same model,
$\kappa_c\le W_c\le W_c^{\rm rat}\le12d\kappa_c$.

\smallskip\noindent
\textup{(ii)} Suppose a parameter in $\{1,\ldots,M\}$ has known ordered
prior $\pi$, and report $t$ independently reveals that parameter with
probability $q_t$ and otherwise reports $*$. Among all stochastic
machines having at most $K_t$ labels after each report, the exact
Bayesian success probability is
\[
 a\pi_1+\sum_{t=1}^n w_t\sum_{j=1}^{k_t}\pi_j,
 \quad
 k_t=\min(M,K_t,\ldots,K_n),
\]
where $a=\prod_t(1-q_t)$ and
$w_t=q_t\prod_{s>t}(1-q_s)$. One compatible deterministic machine
attains all the cut bounds simultaneously. The formula yields exact
sample--peak tradeoffs and exact losses under checkpoint insertion.

\smallskip\noindent
\textup{(iii)} If $P_t,Q_t$ are the induced retained-state transition
kernels of a fixed execution with common initial law, its terminal-law
change is bounded by
\[
 \min\left(1,\sum_t\E_P
           \|P_t(X_{t-1},\cdot)-Q_t(X_{t-1},\cdot)\|_{\TV}\right).
\]
For absorbed reset comparisons this prices active updates even when
training laws differ. It does not remove ignored preparations from
the schedule or from the full raw-data law.

\smallskip\noindent
\textup{(iv)} In the marked positive-noise collision experiment, a
$320000$-training-trial audit with peak width at most $1604$ has score
greater than $2/5$. The $399$-trial, $40602$-state audit and the
$12$-state long-clock audit are also retained. The exact row-indexed
decision width remains three. Separate finite-state amplifiers yield
specified type-I and type-II errors, with a polynomial-horizon growing
counter or a constant register and a longer clocked schedule.
\end{theorem}
\begin{proof}
Part (i) is Theorem~\ref{thm:v22-tournament} together with
Theorem~\ref{thm:v21-endtoend}. Part (ii) is
Theorem~\ref{thm:v22-revelation} and its corollaries. Part (iii) is
Theorem~\ref{thm:v22-active} and
Corollary~\ref{cor:v22-active-training}. Part (iv) follows from
Theorem~\ref{thm:v22-physical},
Corollary~\ref{cor:v22-physical-regimes}, and
Theorem~\ref{thm:v22-confidence}.
\end{proof}
