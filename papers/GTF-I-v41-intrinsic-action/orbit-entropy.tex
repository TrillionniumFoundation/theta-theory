\section{An orbital entropy occupation theorem}\label{sec:orbit}
Lebesgue measure is used on $V=\R^D$ and normalized Haar measure on $G$.
For a function on $V$, set
\begin{equation}\label{eq:operators}
 \mathcal U_a f(z)=f(U_a^{-1}z),\quad
 P_t=\sum_a p_t(a)\mathcal U_a,\quad
 \Pi f(z)=\int_G f(U_g^{-1}z)\,dm_G(g).
\end{equation}
Thus $\mathcal U_a^*=\mathcal U_{a^{-1}}$, $\Pi$ is an orthogonal
projection, and $P_t\Pi=\Pi P_t=\Pi$. The probability law $p_t$ need not
be symmetric. Its required squared norm gap is
\begin{equation}\label{eq:gap}
 \norm{P_t f}_2^2\le\norm{\Pi f}_2^2+(1-g_t)\norm{f-\Pi f}_2^2,
 \qquad 0\le g_t\le1.
\end{equation}
Invariance in this formula means $G$-invariance, not necessarily radial
symmetry. In a reducible representation invariant functions can carry
many parameters.

\begin{lemma}[Transfer of a group norm gap]\label{lem:transfer}
If right convolution $R_{p_t}F(g)=\sum_a p_t(a)F(ga)$ on
$L^2(G)$ has mean-zero norm at most $\sqrt{1-g_t}$, then
\eqref{eq:gap} holds in every orthogonal representation on $V$.
\end{lemma}
\begin{proof}
For each $z$ apply the group inequality to
$F_z(g)=f(U_g^{-1}z)$. Its mean is $\Pi f(z)$ and
$R_{p_t}F_z(g)=P_tf(U_g^{-1}z)$. Integrate the resulting inequality
in $z$. Orthogonal changes of variables and Fubini identify the three
integrals with those in \eqref{eq:gap}. First take bounded compactly
supported $f$ and then use density. This proof does not replace the full
group gap by a gap in the defining representation.
\end{proof}

\subsection{The conditional-mean cost}
Let $G_D$ be a standard independent real Gaussian vector and let
$\gamma_\tau$ be the density of $\tau G_D$.
\begin{lemma}[A one-sided Gaussian coupling inequality]\label{lem:centroid}
For bounded $X$ and a finite-valued variable $J$, put $M=\E[X\mid J]$.
For every $\tau>0$,
\begin{equation}\label{eq:entropy-cost}
 \Ent(X+\tau G_D)-\Ent(M+\tau G_D)
       \le\frac{\E\norm{X-M}^2}{2\tau^2}.
\end{equation}
\end{lemma}
\begin{proof}
Write $f=\Law(M)*\gamma_\tau$. Differentiation of the Gaussian mixture
and the conditional covariance identity give
\begin{equation}\label{eq:hessian}
 \nabla^2\log f(y)=-\tau^{-2}I+
 \tau^{-4}\operatorname{Cov}(M\mid M+\tau G_D=y)\succeq-\tau^{-2}I.
\end{equation}
Hence $\phi(x)=-\E\log f(x+\tau G_D)$ has Hessian at most
$\tau^{-2}I$. Taylor's inequality at $M$ yields
$\phi(X)\le\phi(M)+\langle\nabla\phi(M),X-M\rangle+
\norm{X-M}^2/(2\tau^2)$. The conditional expectation of the linear
term given $J$ is zero. If $g=\Law(X)*\gamma_\tau$, nonnegativity of
$D(g\Vert f)$ gives $\Ent(g)\le-\int g\log f=\E\phi(X)$,
whereas $\E\phi(M)=\Ent(f)$. Boundedness of $X,M$ provides Gaussian
bounds with finite second moments and justifies differentiation and
integration. These identities prove \eqref{eq:entropy-cost}.
\end{proof}
This elementary consequence of Gaussian posterior identities is used
as an estimate, not as a claim of priority over the Gaussian estimation
literature \cite{AKG,PW}. The estimate is one-sided and can have a
negative left side. Here $J$ can be a finite register with no Gaussian
observation model; the smoothing noise is added after conditioning.

\begin{lemma}[Entropy production away from invariant functions]
\label{lem:production}
For any bounded-support Gaussian mixture density $f$,
\begin{equation}\label{eq:production}
 \Ent(P_t f)-\Ent(f)\ge g_t\norm{\sqrt f-\Pi\sqrt f}_2^2\ge0.
\end{equation}
\end{lemma}
\begin{proof}
Put $f_a=\mathcal U_a f$ and $h=P_t f$. Orthogonality preserves entropy,
so $\Ent(h)-\Ent(f)=\sum_a p_t(a)D(f_a\Vert h)$.
For probability densities $u,v$,
$D(u\Vert v)\ge\norm{\sqrt u-\sqrt v}_2^2$; it follows, for example,
from $-\log x\ge2(1-\sqrt x)$ and integration. The mean is the
least-squares minimizer, and thus
\[
 \sum_a p_t(a)\norm{\sqrt{f_a}-\sqrt h}_2^2
 \ge\sum_a p_t(a)\norm{\sqrt{f_a}-P_t\sqrt f}_2^2
 =1-\norm{P_t\sqrt f}_2^2.
\]
Apply \eqref{eq:gap} and $\norm{\sqrt f}_2=1$.
\end{proof}
No logarithmic Sobolev inequality or point-mass mixing assertion is
used in this step.

\subsection{Sparsity away from invariant densities}
Suppose that the representation obeys the uniform orbital bound
\begin{equation}\label{eq:orbital}
 m_G\{g:\norm{U_gu-v}\le r\}\le A r^s
 \quad(u,v\in\mathbb S(V),\ 0<r\le1),
\end{equation}
where $A\ge1$ and $s>0$. In particular the representation has no
nonzero fixed vector. It is essential that $u$ range over all possible
centroid directions, not merely the initial seed orbit.

\begin{lemma}[Finite centroids cannot be almost invariant]
\label{lem:sparse}
Suppose $Z$ has at most $k$ values in the unit ball and
$\E\norm Z\ge\kappa$, where $0<\kappa\le1$. Set
\begin{equation}\label{eq:scale}
 r=\left(\frac{\kappa}{16Ak}\right)^{1/s},\qquad
 \tau=\frac{\kappa r}{8\sqrt{2D}},\qquad f=\Law(Z)*\gamma_\tau.
\end{equation}
Then $\norm{\sqrt f-\Pi\sqrt f}_2^2\ge\kappa/16$.
\end{lemma}
\begin{proof}
Atoms of norm at least $\kappa/2$ have total mass at least $\kappa/2$.
Let $\Omega$ be the union of radius-$r$ caps around their directions,
and let $C_\Omega$ be its cone. By Markov's inequality,
$\Pp(\norm{G_D}\le\sqrt{2D})\ge1/2$. On this event, perturbing a
high-norm atom by $\tau G_D$ changes its direction by at most
$4\tau\sqrt{2D}/\kappa=r/2$. Therefore $\int_{C_\Omega}f\ge\kappa/4$.

For any $G$-invariant $u\in L^2(V)$, Haar averaging and
\eqref{eq:orbital} give
\[
 \int_{C_\Omega}|u(z)|^2\,dz
 =\int |u(z)|^2\int_G\one_{C_\Omega}(U_gz)\,dm_G(g)\,dz
 \le kAr^s\norm u_2^2.
\]
The point $z=0$ is Lebesgue-null. Taking $u=\Pi\sqrt f$, whose norm
is at most one, bounds this integral by $\kappa/16$. The reverse
triangle inequality in $L^2(C_\Omega)$ now gives a norm difference at
least $\sqrt{\kappa/4}-\sqrt{\kappa/16}=\sqrt\kappa/4$.
Repeated directions and zero centroids cause no change.
\end{proof}

\begin{theorem}[Orbital occupation budget]\label{thm:occupation}
Under \eqref{eq:gap} and \eqref{eq:orbital}, let a finite hidden register
track a unit-vector command process as in \eqref{eq:centroids}.
If $q_N\ge\kappa>0$, then, for every $k\ge1$,
\begin{equation}\label{eq:occupation}
 \sum_{\substack{0\le t<N\\K_t\le k}}g_t
 \le B_{D,A,s}\kappa^{-3-2/s}k^{2/s},\qquad
 B_{D,A,s}=2048D(16A)^{2/s}.
\end{equation}
The same inequality holds with $K_t$ replaced by the number of
positive-probability labels under the chosen test law. No bound is
imposed on the other epochs.
\end{theorem}
\begin{proof}
Fix $k$ and the single scale \eqref{eq:scale} for the entire run.
Write $f_t=\Law(Z_t)*\gamma_\tau$. Isotropy of the Gaussian makes the
density of $U_aZ_t+\tau G_D$ equal to $P_tf_t$. By
Lemmas~\ref{lem:centroid} and \ref{lem:production},
\[
 \Ent(f_{t+1})-\Ent(f_t)
 \ge g_t\norm{\sqrt{f_t}-\Pi\sqrt{f_t}}_2^2
       -\frac{\Delta_t}{2\tau^2}.
\]
At an epoch with at most $k$ labels, Lemma~\ref{lem:sparse} bounds the
norm term below by $\kappa/16$. Elsewhere it is nonnegative.
The initial density is a translate of $\gamma_\tau$; the terminal
second moment is at most $1+D\tau^2$. The Gaussian maximum-entropy
bound and \eqref{eq:conditional} yield
\begin{align*}
 \frac\kappa{16}\sum_{K_t\le k}g_t
 &\le\frac D2\log\left(1+\frac1{D\tau^2}\right)
       +\frac1{2\tau^2}
 \le\frac1{\tau^2}.
\end{align*}
Substitute \eqref{eq:scale}. The same Gaussian scale is retained through
all epochs; no repeated mixing-time charge is incurred.
\end{proof}

\begin{corollary}[Actual terminal error]\label{cor:actual-error}
For a constant gap $g>0$, let $\mathcal E_N(\mathbf K)$ be the minimum
actual wordwise binary-TV error over a prescribed available profile,
and put $K=\max_tK_t$. Then
\begin{equation}\label{eq:actual-error}
 \mathcal E_N(\mathbf K)\ge\frac12
 \left[\frac\rho{\sqrt D}
 -\left(\frac{B_{D,A,s}K^{2/s}}{gN}\right)^{1/(3+2/s)}\right]_+.
\end{equation}
In particular, $K=o(N^{s/2})$ implies
$\liminf\mathcal E_N\ge\rho/(2\sqrt D)$.
\end{corollary}
\begin{proof}
If the actual error is below the calibration threshold, combine
\eqref{eq:kappa} with \eqref{eq:occupation} and solve for it. Otherwise
the displayed lower bound already holds. Attainment or a minimizing
sequence gives the same conclusion for the optimum.
\end{proof}
Local simulation errors may cancel. Nothing in this corollary sums such
errors or rules out cancellation; its objective is the actual final law.
