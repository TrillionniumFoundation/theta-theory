\section{Physical source, signed measures, and retained estimates}
\label{sec:v94-inputs}
Fix $m,R,n,k$ until a uniform norm is explicitly taken. Write
$\bar\nu=\one_\Pi\nu_R^*$ for the original exact-event source.
The retained Borel parametrization is a bijection between its regular
part and maximal segments $(\sigma,u)$, $a_\sigma<u<b_\sigma$.
Its transverse measure $d\sigma$ is the disjoint angular measure and
its density is
\[
 A_\sigma(u)=\frac{\omega(\Psi_\sigma(u))}{4\pi c_*\chi(\Psi_\sigma(u))},
 \qquad d\bar\nu=A_\sigma(u)\,du\,d\sigma.
\]
Here $F=u$, $z=|\nabla_qF|$, $q=(\alpha_0,\alpha_m)$,
$\omega=|\partial_{p_0}\alpha_m|^{-1}$, and
$\chi=|\det D_q(F,\arg\nabla_qF)|$. All fibers and signed measures
below use these same kernels. Let $z_\sigma^-=z(\Psi_\sigma(a_\sigma+))$
and $\ell_\sigma=b_\sigma-a_\sigma$.

\begin{inputstatement}[Complete segment geometry and signed balance]
\label{input:v94-geometry}
Put $\lambda_\circ=9\sqrt{19}/200$ and $\Lambda_\circ=47/6$.
On every regular segment,
\begin{align}
 2\lambda_\circ(v-u)&\le z(v)^2-z(u)^2
                    \le2\Lambda_\circ(v-u),\label{eq:v94-radial}\\
 |A_\sigma'|&\le C A_\sigma I_m/z,\qquad
 |(W^\varepsilon)'|\le C\varepsilon^{-1}/z,
 \quad I_m=1+\sum_{i=1}^{m-1}q_0^{d_i}/c_i.\label{eq:v94-derivative}
\end{align}
The complete density $f_\sigma^\varepsilon=A_\sigma W^\varepsilon$
has one-sided traces and is absolutely continuous on the closed
segment. The sum of its endpoint traces and absolute derivative
integrals is finite at each fixed count. A grazing endpoint has zero
trace; a critical backward endpoint need not. The geometric domain
is a finite definable family at each fixed count. Its component count
is used for existence of measures only, not for normalized estimates.

For every Borel prefix endpoint $c_\sigma\in[a_\sigma,b_\sigma]$ put
\[
 R_c^\varepsilon(t)=\int\one_{\{a_\sigma<t<c_\sigma\}}
                              f_\sigma^\varepsilon(t)\,d\sigma.
\]
The original union $z<\zeta$ or $\ell<\delta$ is exactly such a
prefix, and $c_\sigma=b_\sigma$ gives $b^\varepsilon$. Define
\begin{align*}
 \mathsf I_c^\varepsilon&=\int_{a_\sigma<c_\sigma}
                  f_\sigma^\varepsilon(a_\sigma+)\delta_{a_\sigma}\,d\sigma,\\
 \mathsf D_c^\varepsilon&=\int_{a_\sigma<c_\sigma}
                  f_\sigma^\varepsilon(c_\sigma-)\delta_{c_\sigma}\,d\sigma,\\
 \mathsf G_c^{\varepsilon,\pm}(H)&=\int\int_{(a_\sigma,c_\sigma)\cap H}
                      [\pm(f_\sigma^\varepsilon)'(u)]_+\,du\,d\sigma.
\end{align*}
Their complete finite-count signed identity is
\begin{equation}\label{eq:v94-signed-input}
 D R_c^\varepsilon=\mathsf I_c^\varepsilon+
    \mathsf G_c^{\varepsilon,+}-\mathsf D_c^\varepsilon-
    \mathsf G_c^{\varepsilon,-}.
\end{equation}
Every artificial prefix cut is negative. All true entrance types and
one-sided incident physical components are retained.
\end{inputstatement}
This is the endpoint-field, optical, Borel-segment and signed-prefix
chain, culminating in retained modules 220--222. In particular,
$A_\sigma$ includes both angular factors. No reversal of the billiard
record or change of the section is used at a backward end.

\begin{inputstatement}[Backward-end and critical volume estimates]
\label{input:v94-volume}
Set $\alpha=129/128$ and $\gamma=1/32$. Let
\[
 e_s^-(t)=\int\one_{\{t-a_\sigma\le s\}}\,d\mu_t,\qquad
 u_q(t)\,dt=F_\#(I_m\one_{\{z<q\}}\bar\nu),\qquad
 p_q(t)\,dt=F_\#(\one_{\{z<q\}}\bar\nu).
\]
For every fixed auxiliary band $B_*$ there are $C_{B_*}<\infty$ and
$\theta_{B_*}<1$ such that, at finite count and small $s,q>0$,
\begin{align}
 m^2\int_J e_s^-&\le C s^\gamma(|J|+B_*^{-1})
 +C_{B_*}(1+B_*|J|)m^3\min\{\theta_{B_*}^m,\sqrt s\},
                                            \label{eq:v94-end-finite}\\
 m^2\int_J u_q&\le C q^\alpha(|J|+B_*^{-1})
 +C_{B_*}(1+B_*|J|)m^2q^{1/2}\theta_{B_*}^m.
                                            \label{eq:v94-critical-finite}
\end{align}
The constants are uniform in radius, exact labels, marks where present,
and interval location. Since $I_m\ge1$, the latter also bounds $p_q$.
Consequently, at every fixed $s,L>0$,
\[
 \limsup_m\sup_{R,n,k}\sup_{|J|=L}m^2\int_J e_s^-\le CLs^\gamma.
\]
For the actual positive source
$v^\rho(t)\,dt=F_\#(I_m z^{-1}\one_{\{z\ge\rho\}}\bar\nu)$,
\begin{equation}\label{eq:v94-cutoff}
 \limsup_m\sup_{R,n,k}\sup_{|J|=L}m^2\int_J v^\rho\le C L
                  \quad(\rho,L>0\text{ fixed}).
\end{equation}
The leading constant is independent of the fixed positive cutoff;
its finite-count remainder is not.
\end{inputstatement}
Equation~\eqref{eq:v94-end-finite} is the same-event backward-end
estimate of module 211, including critical entrances. For
\eqref{eq:v94-critical-finite}, multiply the actual fractional-source
bound of module 187, at exponent $1/4$, by $q^{1/4}$: pointwise,
$I_m\one_{z<q}\le q^{1/4}I_mz^{-1/4}\one_{z<q}$.
The leading power is $q^\alpha$ and the remainder power is $q^{1/2}$.
The audited cutoff estimate is retained in modules 196 and 222.
These are volume-source theorems, not assumptions on entrance flux.

\begin{inputstatement}[Original scalar reconstruction]
\label{input:v94-scalar}
For every fixed $\varepsilon,r>0$,
\begin{equation}\label{eq:v94-average-input}
 \limsup_m\sup_{R,n,k}m^2\|A_rb^\varepsilon\|_\infty
       \le C\varepsilon^{1/16},\qquad
 A_rf(t)=\frac1{2r}\int_{t-r}^{t+r}f(u)\,du.
\end{equation}
For large fixed $B$ and $\varepsilon(B)=A_0B^{-1/12}$,
\begin{equation}\label{eq:v94-remainder}
 m^2p-\mathcal L=m^2b^{\varepsilon(B)}+e_B,
 \qquad \limsup_m\sup_{R,n,k}\esssup_{\mathcal C(M)}|e_B|
                                      \le C_MB^{-1/192}.
\end{equation}
Every auxiliary band is fixed before the collision limit and enlarged
only afterwards. The reconstruction band $B$ is outermost.
\end{inputstatement}
We use $\mathcal H_M(h)=\limsup_m\sup_{R,n,k}m^2
\esssup_{t\in\mathcal C_{m,R,n,k}(M)}h(t)$ for a nonnegative source
family. Unqualified $\|\cdot\|_\infty$ denotes essential supremum
on the full roof line. Identities hold for common kernels at fixed
$m,R,n,k$; the uniform order is $\sup_R\esssup_t$. Countably many
chosen radii share versions. No simultaneous Lebesgue-point assertion
for every uncountable selector family is required.
