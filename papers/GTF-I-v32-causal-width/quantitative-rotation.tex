\section{An explicit multi-cut inequality}
\label{sec:quantitative}
We now give a quantitative expansion estimate for the rotation in
Theorem~\ref{thm:headline}. Throughout this section,
$\alpha=(\sqrt5-1)/2$, $R$ is rotation by $2\pi\alpha$, and
$a=1/10$. The Euclidean unit disk is denoted by $\B^2$.

\begin{lemma}[Finite orbit covering]\label{lem:orbit-net}
For every integer $M\ge3$, there is an integer $q$ with
$32M\le q<64M$ such that $\{R^ku:0\le k<q\}$ is a
$3\pi/q$-net of the unit circle in angular distance, for every unit $u$.
\end{lemma}
\begin{proof}
Let $F_0=0,F_1=1,F_{m+1}=F_m+F_{m-1}$ and take the first Fibonacci
number $q=F_m$ at least $32M$. Consecutive positive Fibonacci numbers
have ratio less than two, so $q<64M$. Put $p=F_{m-1}$. The elementary
identities
\[
 \gcd(p,q)=1,\qquad F_m\alpha-F_{m-1}=(-1)^{m-1}\alpha^m,
 \qquad F_m\alpha^m<1
\]
follow by induction (or by the formula for $F_m$ obtained from the two
roots of $z^2=z+1$). Thus $|\alpha-p/q|<1/q^2$. The first $q$
multiples of $p/q$ form the uniform $q$-grid modulo one. Each differs
from the corresponding multiple of $\alpha$ by less than $1/q$ in
circular distance. Every point of the circle is within $1/(2q)$ of a
grid point and hence within $3/(2q)$ turns of an orbit point. Multiplying
by $2\pi$ proves the assertion, also after a common rotation by $u$.
\end{proof}

\begin{lemma}[A polygon cannot be too nearly invariant]
\label{lem:hausdorff}
If $a\B^2\subset P\subset\sqrt2\B^2$ is a polygon with at most
$M\ge3$ vertices, then
\begin{equation}\label{eq:hausdorff-lower}
                         d_H(P,RP)\ge\frac{a}{64M^3}.
\end{equation}
\end{lemma}
\begin{proof}
Write $\varepsilon=d_H(P,RP)$ and $r=\max_{z\in P}\norm z_2$.
Choose a vertex $v$ of norm $r$. Rotational invariance of Hausdorff
distance and the triangle inequality give
$d_H(P,R^kP)\le k\varepsilon$.
By Lemma~\ref{lem:orbit-net}, for every unit direction $u$ some $k<q$
puts $R^kv$ within angular distance $3\pi/q$ of $u$. For the support
function $h_P$, it follows that
\begin{equation}\label{eq:orbit-support}
 h_P(u)\ge r\cos(3\pi/q)-q\varepsilon,
 \qquad r-h_P(u)\le\frac{9\pi^2\sqrt2}{2q^2}+q\varepsilon.
\end{equation}

The directions of the at most $M$ vertices have an empty circular arc
of length at least $2\pi/M$. At its midpoint $u_0$, every vertex has
scalar product at most $r\cos(\pi/M)$. Consequently
\[
 r-h_P(u_0)\ge r(1-\cos(\pi/M))\ge\frac{2a}{M^2}.
\]
Here $1-\cos s\ge2s^2/\pi^2$ for $0\le s\le\pi$, and $r\ge a$.
Using $q\ge32M$, $\pi^2<10$ and $\sqrt2<3/2$ shows that the first
term on the right of \eqref{eq:orbit-support} is less than $a/M^2$.
It follows that $q\varepsilon\ge a/M^2$.
The bound $q<64M$ proves \eqref{eq:hausdorff-lower}.
\end{proof}

\begin{lemma}[Area pays for Hausdorff displacement]\label{lem:area}
If $a\B^2\subset P\subset R_*\B^2$ and $U=\conv(P\cup RP)$, for
an arbitrary planar rotation $R$, then
\begin{equation}\label{eq:area-hausdorff}
           \area(U)-\area(P)\ge\frac{a}{R_*}d_H(P,RP)^2.
\end{equation}
\end{lemma}
\begin{proof}
Let $d_+=\max_{u\in\mathbb S^1}(h_{RP}(u)-h_P(u))_+$.
When $d_+>0$, choose a maximizing direction $u$ and $x\in RP$ with
$u\cdot x=h_P(u)+d_+$. The triangle with vertices $x$ and the two
points $\pm av$, where $v\perp u$ is a unit vector, lies in $U$.
Its part in $\{z:u\cdot z>h_P(u)\}$ lies outside $P$. Similar
triangles give its area as
\[
            \frac{a d_+^2}{u\cdot x}\ge\frac{a d_+^2}{R_*}.
\]
The zero case needs no argument. Repeating this with $P$ and $RP$
interchanged gives the same bound with $d_-$. Rotation preserves area,
so the two excess areas coincide. For compact convex sets,
$d_H(P,RP)=\max(d_+,d_-)$, proving the claim.
\end{proof}

\begin{theorem}[Quantitative temporal obstruction]\label{thm:quantitative}
For the golden-angle rotation and $S,D$ of Section~\ref{sec:rotation},
\begin{equation}\label{eq:delta-power}
                   \delta_M\ge\frac{1}{6000000M^6}\quad(M\ge3).
\end{equation}
Every exact realization of $C_{N,R}$ therefore satisfies
\begin{equation}\label{eq:profile-power}
                    \sum_{t=0}^{N-1}2^{-6K_t}\le24000000.
\end{equation}
In particular $W_N(R)\ge\frac16\log_2(N/24000000)$.
\end{theorem}
\begin{proof}
A polygon feasible for \eqref{eq:expansion} contains $a\B^2$ and lies
in $\sqrt2\B^2$. The last two lemmas give
\[
 \area(\conv(P\cup RP)) -\area(P)
 \ge\frac{a^3}{4096\sqrt2\,M^6}
 >\frac{1}{6000000M^6},
\]
using $a=1/10$ and $\sqrt2<10/7$. Minimize over $P$ to obtain
\eqref{eq:delta-power}. Apply Theorem~\ref{thm:volume} and bound
$\area(D)-\area(S)$ by $4$. If $K_t\le W$ at all times, the left
side of \eqref{eq:profile-power} is at least $N2^{-6W}$.
\end{proof}

Combining Proposition~\ref{prop:static-three},
Theorems~\ref{thm:rotation-unbounded}, \ref{thm:polygon-upper} and
\ref{thm:quantitative} proves Theorem~\ref{thm:headline}.
The profile inequality gives more than a bound on one minimum peak.
For example, at most $24000000\,2^{6k}$ of the command-input cuts can
have width at most $k$, independently of the widths at the other cuts.
The same statement follows directly by restricting the nonnegative sum.

\begin{proposition}[Robustness at a fixed excluded profile]
\label{prop:robust-profile}
Fix a finite horizon and a profile excluded for $C_{N,R}$ by either
volume bound. There is a strictly positive uniform row-total-variation
distance from $C_{N,R}$ to all arrays realized with that profile. For the
rational family, a positive rational lower distance can be computed from
the fully listed array and profile by real quantifier elimination.
\end{proposition}
\begin{proof}
Initialization, update and decoder rows range over a finite product of
compact simplices. The realized array is a polynomial, and hence
continuous, function of those rows. Its image is compact. An excluded
array has strictly positive distance to this image in the maximum of
the finitely many row-TV distances. For rational target entries the
minimum distance is a semialgebraic optimization, after expressing
absolute values by auxiliary variables; real quantifier elimination
and root isolation give the stated rational lower bound \cite{BPR}.
No uniform lower distance as the horizon or profile varies follows from
this argument.
\end{proof}

The exact golden-angle lower bound uses a specified computable real
rotation; it is not presented as an algebraic-input computation. The
rational family supplies algebraic excluded-horizon and fixed-profile
robustness certificates instead. Neither the numerical values of its
$\delta_M$ nor optimal rates are asserted by the finite regression tests.
