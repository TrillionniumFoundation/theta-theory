\section{Pressure, distortion and survival}\label{sec:v5-pressure}
We use the Gibbs and variational principles on a finite full shift in
their usual H\"older form \cite[Chapters~1--2]{Bowen2008}. For a
H\"older function $u$, these principles give a shift-invariant
probability $\nu_u$ and $C_u\ge1$ with
\begin{equation}\label{eq:v5-classicalgibbs}
 C_u^{-1}e^{S_nu(\omega)-nP(u)}\le\nu_u[w]
 \le C_ue^{S_nu(\omega)-nP(u)},
 \quad P(u)=\sup_{\nu\in\mathcal M_S}
           \left(h_\nu(S)+\int u\,d\nu\right).
\end{equation}
Here $\mathcal M_S$ is the set of shift-invariant probabilities.
Only these finite-alphabet statements, not a countable-state
spectral or suspension theorem, are used below.

\begin{lemma}[The mixed partition sum]\label{lem:v5-pressure-sum}
Fix $0<s<1$, and set
\[
 A(s)=P(s\varphi+2s\psi),\qquad
 B_q(s)=P(s\varphi)+s\log q.
\]
There are constants $c_s,C_s'>0$, independent of $n$, such that
\begin{equation}\label{eq:v5-pressurebounds}
 c_s\max\{e^{nA(s)},e^{nB_q(s)}\}
 \le \sum_{|w|=n}a_q(w)^s
 \le C_s'\sum_{k=0}^{n}e^{kB_q(s)+(n-k)A(s)}.
\end{equation}
\end{lemma}

\begin{proof}
For a word $w$ of length $n$ and $\omega\in[w]$, the chain rule
and bounded distortion give
\[
 r_{w_{k:}}\asymp e^{S_{n-k}\psi(S^k\omega)},\qquad
 \mu[w]\asymp e^{S_n\varphi(\omega)},
\]
with constants independent of $n,k,w$. The term $k=0$ and the
terminal term in \eqref{eq:v5-cost}, together with
\eqref{eq:v5-classicalgibbs}, give the two lower bounds in
\eqref{eq:v5-pressurebounds}.

For the upper bound use $(\sum x_j)^s\le\sum x_j^s$ for
nonnegative $x_j$. At a fixed splitting time $k<n$, the resulting
sum, up to a constant and the factor $(1-q)^s$, is
\[
 \sum_{|w|=n}\exp\left\{
 sS_k\varphi(\omega)+ks\log q
 +S_{n-k}(s\varphi+2s\psi)(S^k\omega)\right\}.
\]
The variations of $S_ku$ inside any $k$-cylinder are uniformly
bounded for a H\"older $u$. Since every prefix can be concatenated
with every suffix on the full shift, this sum is comparable to
the product of the length-$k$ partition sum for $s\varphi+s\log q$
and the length-$(n-k)$ partition sum for $s\varphi+2s\psi$.
Each is comparable to its pressure exponential by
\eqref{eq:v5-classicalgibbs}. The terminal term is the same
calculation with $k=n$. Summing over $k$ proves the assertion.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:v5-pressure}]
Both pressures at $s=0$ equal $\log b$. At $s=1$ they are negative:
$A(1)\le2\log r_+<0$ and $B_q(1)=\log q<0$.
There is a uniform positive lower bound on every one-symbol
conditional cylinder mass. Since there are at least two symbols,
there is consequently $\eta<1$ such that $\mu[w]\le\eta^{|w|}$.
Equation~\eqref{eq:v5-gibbs} implies
$S_n\varphi\le n\log\eta+\log C_G$. Integration against any
invariant probability gives $\int\varphi\,d\nu\le\log\eta<0$.
The variational principle now shows that both $A$ and $B_q$ are
strictly decreasing, with their slopes bounded above by a
negative constant. Their unique zeros therefore lie in $(0,1)$.
Equation~\eqref{eq:v5-pressurebounds} proves
\eqref{eq:v5-mixedpressure}: the sum on its right has at most
$n+1$ terms, each at most the exponential of
$n\max\{A(s),B_q(s)\}$.

We prove the exponent without assuming any quantization dimension.
Let $s>s_*(q)$, $s<1$. Both pressures are negative, so
\eqref{eq:v5-pressurebounds} is summable in $n$. Hence
\[
 H_s:=\sum_{w\text{ finite}}a_q(w)^s<\infty.
\]
Balance in the greedy tree yields
$L_N(a_0\theta_N)^s\le H_s$, and thus
\begin{equation}\label{eq:v5-upperexponent}
 G_N\le C_s''N^{-(1-s)/s}.
\end{equation}

For the reverse estimate choose a Gibbs measure $\nu_*$ for a
zero-pressure potential as follows. When $s_*=s_g$, use
$s_g\varphi+2s_g\psi$; otherwise use
$s_t(q)\varphi+s_t(q)\log q$. The spatial summand or the terminal
summand of $V_q$, respectively, shows that
\[
 \nu_*[w]\le C_*a_q(w)^{s_*}
 \quad\hbox{for every word }w.
\]
Since $\mathcal P_N$ is a cylinder partition,
$1\le C_*L_N\theta_N^{s_*}$. Therefore
\begin{equation}\label{eq:v5-lowerexponent}
 G_N\ge c_*N^{-(1-s_*)/s_*}.
\end{equation}
Let $s$ decrease to $s_*$ in \eqref{eq:v5-upperexponent} and use
\eqref{eq:v5-lowerexponent}. This proves the exponent limit for
$G_N$, and Theorem~\ref{thm:v5-intrinsic} proves the other limits.
Finally $P(s_g\varphi)>0$ because $s_g<1$ and $P(\varphi)=0$.
Thus the number $q_c$ in \eqref{eq:v5-qcritical} lies in $(0,1)$.
Strict monotonicity shows that $s_t(q)\le s_g$ precisely when
$B_q(s_g)\le0$, which is equivalent to $q\le q_c$.
\end{proof}

\begin{corollary}[Entropy--expansion ratios]\label{cor:v5-variational}
Put $I_\nu=-\int\varphi\,d\nu$ and
$\chi_\nu=-\int\psi\,d\nu$. Then
\begin{equation}\label{eq:v5-variational}
 s_g=\sup_{\nu\in\mathcal M_S}
       \frac{h_\nu(S)}{I_\nu+2\chi_\nu},\qquad
 s_t(q)=\sup_{\nu\in\mathcal M_S}
       \frac{h_\nu(S)}{I_\nu+\log(1/q)}.
\end{equation}
All denominators are strictly positive. The quantities
$\chi_\nu$ are the mean expansion rates of $T$ in the coding
coordinates.
\end{corollary}

\begin{proof}
The preceding proof gives $I_\nu\ge-\log\eta>0$, and
$\chi_\nu\ge-\log r_+>0$. By the variational principle,
$A(s)=\sup_\nu\{h_\nu-s(I_\nu+2\chi_\nu)\}$.
Its zero is the supremum of the displayed ratios. For example,
at that supremum all the affine expressions are nonpositive,
while ratios approaching the supremum make their values approach
zero, since the denominators are uniformly bounded above as well
as below. The second formula follows in the same way from $B_q$.
The chain rule identifies $-S_n\psi$ with the logarithmic
expansion of $T^n$ on its cylinder.
\end{proof}

\begin{proposition}[Finite pressure enclosures]\label{prop:v5-enclosures}
For a H\"older potential $u$, define
\[
 Z_n^-(u)=\sum_{|w|=n}\exp\{\inf_{[w]}S_nu\},\qquad
 Z_n^+(u)=\sum_{|w|=n}\exp\{\sup_{[w]}S_nu\}.
\]
If $\operatorname{var}_j(u)=\sup\{|u(x)-u(y)|:x_{0:j}=y_{0:j}\}$,
then
\begin{equation}\label{eq:v5-enclosures}
 \frac1n\log Z_n^-(u)\le P(u)\le\frac1n\log Z_n^+(u),\qquad
 \frac1n\log\frac{Z_n^+(u)}{Z_n^-(u)}
 \le\frac1n\sum_{j\ge1}\operatorname{var}_j(u).
\end{equation}
Here $x_{0:j}=y_{0:j}$ means equality of the first $j$ symbols.
Thus certified cylinder bounds on the potentials give convergent
finite enclosures for the two pressure roots and for $q_c$.
\end{proposition}

\begin{proof}
Concatenation gives $Z_{n+k}^-\ge Z_n^-Z_k^-$ and
$Z_{n+k}^+\le Z_n^+Z_k^+$. Their normalized logarithms have the
same limit $P(u)$, since the oscillation of $S_nu$ in a cylinder
is at most $\sum_{j=1}^n\operatorname{var}_j(u)$. Superadditivity
and subadditivity prove the enclosure, and the same oscillation
bound gives its width. Strict monotonicity of the pressures
then converts pressure enclosures into root enclosures.
\end{proof}

These are finite enclosures in the intrinsic potentials. As usual
with arbitrary real-valued smooth input data, an effective numerical
algorithm requires certified access to those data. Neither the
resource model nor this proposition asserts a Turing complexity
bound for an unspecified real function.

\subsection{Markov branch weights and unequal multipliers}
\begin{corollary}[A Perron--Frobenius formula]\label{cor:v5-markov}
Suppose $h_i(x)=c_i+r_ix$ and the symbols have a stationary Markov
law with a strictly positive stochastic matrix $P=(p_{ij})$.
Then $s_g$ and $s_t(q)$ are the unique roots
\begin{equation}\label{eq:v5-matrixroots}
 \rho\bigl((p_{ij}^{s_g}r_i^{2s_g})_{ij}\bigr)=1,
 \qquad q^{s_t(q)}\rho\bigl((p_{ij}^{s_t(q)})_{ij}\bigr)=1.
\end{equation}
In particular,
\[
 q_c=\rho\bigl((p_{ij}^{s_g})_{ij}\bigr)^{-1/s_g}.
\]
For independent symbols with probabilities $p_i>0$, these equations
reduce to $\sum_i(p_ir_i^2)^{s_g}=1$ and
$q^{s_t}\sum_i p_i^{s_t}=1$.
\end{corollary}

\begin{proof}
Take $\varphi(\omega)=\log p_{\omega_0\omega_1}$ and
$\psi(\omega)=\log r_{\omega_0}$. The Markov cylinder probability
has a stationary initial factor and one fewer transition than
$e^{S_n\varphi}$, so the Gibbs ratio is bounded above and below.
The pressure of an edge potential $u(i,j)$ is
$\log\rho((e^{u(i,j)})_{ij})$, by summing products along words
and applying Perron--Frobenius to the positive matrix. This yields
\eqref{eq:v5-matrixroots}. When all rows equal $(p_i)_i$, the
matrix has rank one and its nonzero eigenvalue is the displayed
sum.
\end{proof}

The powers of $p_{ij}$ in \eqref{eq:v5-matrixroots} are
\emph{entrywise powers}. Replacing them by a matrix power of $P$
would erase precisely the information pressure at issue.

\begin{example}[A nonlinear variable-expansion family]\label{ex:v5-variable}
On $[0,1]$ take
\[
 h_1(x)=\frac{x}{5}+\varepsilon x(1-x),\qquad
 h_2(x)=\frac34+\frac{x}{6}+\varepsilon x(1-x),
 \qquad 0\le\varepsilon\le\frac1{100}.
\]
Their image intervals are $[0,1/5]$ and $[3/4,11/12]$.
The derivatives lie, respectively, in $[19/100,21/100]$ and
$[1/6-1/100,1/6+1/100]$. Thus all hypotheses, including a common
positive separation and distortion bound, hold uniformly. Equip
the symbols, for example, with the stationary Markov law
\[
 P=\begin{pmatrix}4/5&1/5\\7/20&13/20\end{pmatrix}.
\]
At $\varepsilon=0$, Corollary~\ref{cor:v5-markov} gives two
explicit positive-matrix equations. At $\varepsilon>0$,
Theorem~\ref{thm:v5-pressure} and
Proposition~\ref{prop:v5-enclosures} apply to the genuinely
nonconstant derivative potential.

This family is not obtained by a smooth change of coordinate
from a map with a common multiplier at every fixed point.
The two inverse branches have fixed points with derivatives in
the two disjoint intervals just displayed. A $C^1$ conjugacy
with nonzero derivative at those points preserves the fixed-point
multipliers, which are their reciprocals. Their inequality is
therefore an obstruction to such a conjugacy. There remains a
topological full-shift coding, as there must for this repeller
class; the assertion concerns the metric expansion, not the
absence of symbolic coordinates.
\end{example}
