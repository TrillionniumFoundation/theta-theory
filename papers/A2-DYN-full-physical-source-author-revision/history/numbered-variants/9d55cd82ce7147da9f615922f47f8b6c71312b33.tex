\section{Recovering positive trace missed by the relative clock}
\label{sec:v103-recovery}
For each of the two flows define, on its comparison domain,
\begin{equation}\label{eq:v103-gap-clock}
 k_{r,\Delta}^\pm(y)=\frac1r\int_{-r}^r
 \one_{\{\Theta_h^\pm y\text{ exists}\}}
                            \min\{1,J_h^\pm(y)\}\,dh,
\end{equation}
and set it to zero elsewhere. It is Borel and lies in $[0,2]$.
The roof kernels are those of the same original positive paired
measure as in Section~\ref{sec:v102-pair-coarea}.

\begin{theorem}[Pointwise height from the two endpoint-distance clocks]
\label{thm:v103-gap-height}
Let $0<\delta<\Delta$, $r>0$ and $0<q\le1$. For an original
Borel selector $H$ supported on $\ell\le\delta$, let
\[
 G=\{\max(k_{r,\Delta}^-,k_{r,\Delta}^+)\ge q\}.
\]
At every finite collision count and almost every original target roof,
\begin{equation}\label{eq:v103-gap-finite}
 T_{H\cap G}^\varepsilon(t)
 \le\frac{2}{qr}\int_{t-r}^{t+r}
                         \mathsf T_\Delta^\varepsilon(u)\,du.
\end{equation}
Consequently
\begin{equation}\label{eq:v103-gap-height}
 \mathcal H_M(T_{H\cap G}^\varepsilon)
 \le\frac Cq\{\varepsilon^{1/16}+
                                    (1+\varepsilon^{-1})\Delta\}.
\end{equation}
The estimate acts on the positive endpoint minimum itself. There is
no lower residence, initial-gradient or internal-incidence cutoff.
The numerical factor two counts comparison directions, not words.
\end{theorem}
\begin{proof}
Disjointize $G$ into $G_- =\{k^-\ge q\}$ and
$G_+=\{k^-<q,\ k^+\ge q\}$. On either set, integrate $q\le k^\pm$
against $\one_H\,d\pi_t^\varepsilon$. For each time $h$, use
$\min(1,J_h^\pm)\le J_h^\pm$ and
\eqref{eq:v103-gap-covariance}. The transported restriction is
bounded above by the complete paired target with $\ell<\Delta$,
after all physical sheets have been summed. The target need not
satisfy the original selector $H$, which is held at the starting
roof; it has the same exact event and labels. This gives
\[
 qT_{H\cap G_\pm}^\varepsilon(t)
 \le\frac1r\int_{-r}^r\mathsf T_\Delta^\varepsilon(t+h)\,dh.
\]
Adding the two estimates proves \eqref{eq:v103-gap-finite}.
Apply \eqref{eq:v102-paired-average} to obtain
\eqref{eq:v103-gap-height}. All widths and thresholds are fixed
before the collision limit. No separate estimate of sheet count,
no differentiation of the selector, and no signed cancellation enters.
\end{proof}

Keep the original schedule and original selector $\mathsf S$ in
\eqref{eq:v102-new-schedule} and \eqref{eq:v102-transverse-residual}.
In addition to its unchanged relative clock with collar $2\delta$,
use the two new clocks with
\begin{equation}\label{eq:v103-schedule}
 r_g=\delta,\qquad\Delta_g=4\delta,\qquad
 q_g=\varepsilon(B)^{1/32}.
\end{equation}
Let $F_B$ be the original relative-clock failure set, so that
$\mathsf U_B=T_{\mathsf S\cap F_B}^{\varepsilon(B)}$.
With $G_B$ the gap-admitted set from Theorem~\ref{thm:v103-gap-height},
define the positive sources
\begin{equation}\label{eq:v103-residual-split}
 \mathsf V_B=T_{\mathsf S\cap F_B\cap G_B}^{\varepsilon(B)},
 \qquad
 \mathsf U_B^{\rm rem}
       =T_{\mathsf S\cap F_B\cap G_B^c}^{\varepsilon(B)}.
\end{equation}
They satisfy $\mathsf U_B=\mathsf V_B+\mathsf U_B^{\rm rem}$
on the original kernels. In particular
$0\le\mathsf U_B^{\rm rem}\le\mathsf U_B$ at finite count.

\begin{theorem}[A smaller positive failure source with paid recovery]
\label{thm:v103-recovery}
For all sufficiently large outer $B$,
\begin{equation}\label{eq:v103-recovered}
 \mathcal H_M(\mathsf V_B)\le C_MB^{-1/384}.
\end{equation}
Moreover the original scalar error satisfies
\begin{equation}\label{eq:v103-scalar}
 \mathcal E_M\le\mathcal H_M(\mathsf U_B^{\rm rem})
                                      +C_MB^{-1/384}.
\end{equation}
Along this same schedule the original pointwise theorem is equivalent
to $\mathcal H_M(\mathsf U_B^{\rm rem})\to0$ for every fixed $M$.
The first assertion estimates the newly recovered part; the last
limit is not asserted here.
\end{theorem}
\begin{proof}
Apply \eqref{eq:v103-gap-height} with $H=\mathsf S\cap F_B$.
Since $\delta=O(\varepsilon^4)$, the upper bound is
\[
 C\varepsilon^{-1/32}
        \{\varepsilon^{1/16}+(1+\varepsilon^{-1})4\delta\}
 \le C(\varepsilon^{1/32}+\varepsilon^{95/32}).
\]
This is $O(B^{-1/384})$. Substitute the exact positive split
\eqref{eq:v103-residual-split} in Theorem~\ref{thm:v102-scalar}.
This proves \eqref{eq:v103-scalar} and sufficiency. Conversely,
scalar closure implies $\mathcal H_M(\mathsf U_B)\to0$ by the
retained necessity argument; domination then implies the stated
limit for $\mathsf U_B^{\rm rem}$. No limit is exchanged with a
collision-dependent collar or a changing Fourier band.
\end{proof}

The next section proves a geometric admission theorem which is
uniform in the relative coordinate, including its approach to a
backward critical entrance and to a relative-coordinate fold. Thus
\eqref{eq:v103-residual-split} is accompanied by a new pointwise
estimate and a new comparison direction, not merely by new names
for the old positive remainder. Failure of all three clocks still
retains its full positive source, including short angular reach and
uncontrolled variation. It is not inferred small from its definition.
