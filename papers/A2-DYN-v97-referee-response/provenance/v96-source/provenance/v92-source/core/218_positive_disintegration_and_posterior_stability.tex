\section{Positive physical disintegration and posterior stability}
\label{sec:v92-paths}
The lower denominator bound permits a limited, but unconditional,
normalization theorem at positive-reference roofs. Its topology must
be distinguished from uniform pointwise convergence of the conditional
laws. The latter still requires exclusion of the upper spikes.

Let $\mathcal X$ and $\mathsf W_R$ have exactly the centered collision
path convention of Input~\ref{input:v92-path}. Write
\[
 \mathbf P=P\mathsf Q_t,\qquad
 \mathfrak d_m(t)=\|\mathbf P-G\mathsf W_R\|_{\mathrm{BL}^*}.
\]
The constant test belongs to the bounded-Lipschitz class. A countable
norming class and the inherited common kernels fix measurability of
all displayed norms before any supremum is taken.

\begin{theorem}[Pointwise identification of the numerator excess]
\label{thm:v92-numerator}
For every fixed $M$,
\begin{equation}\label{eq:v92-numerator-gap}
 0\le\mathfrak d_m-|P-G|,\qquad
 \sup_{R,n,k}\esssup_{\mathcal C(M)}
       \bigl(\mathfrak d_m-|P-G|\bigr)\longrightarrow0.
\end{equation}
In particular $\mathfrak d_m$ tends to zero in the unweakened critical
weak norm, locally and in the complete diffusive window. This theorem
identifies the two discrepancies up to a uniform additive error; it
does not assert that either discrepancy is uniformly small.
\end{theorem}
\begin{proof}
At every previously fixed $B$, the positive physical path remainder
satisfies
\[
 \mathbf P-G\mathsf W_R=\mathbf b_B+\mathbf E_B,
 \qquad \|\mathbf b_B\|_{\mathrm{BL}^*}
       =\mathbf b_B(1)=b_B.
\]
If $\eta_B(t)=\|\mathbf E_B(t)\|_{\mathrm{BL}^*}$, the constant
test and positivity give
\[
 |P-G|\le\mathfrak d_m\le b_B+\eta_B
                                      \le |P-G|+2\eta_B.
\]
The ordered essential height of $\eta_B$ is at most $C_MB^{-1/192}$.
Take the collision limsup first, and then $B\to\infty$.
This proves \eqref{eq:v92-numerator-gap} without subtracting infinite
limsup heights. The discrepancies are finite almost everywhere at each
finite count. The critical-norm conclusion follows from the uniform
additive error and Input~\ref{input:v92-critical}, using the fixed
measure of the local or rescaled window.
\end{proof}

\begin{proposition}[The exact physical mixture]
\label{prop:v92-mixture}
On $D_m(d)$, fix $B$ sufficiently large and then $m$ sufficiently
large that $\eta_B\le d/2$. Put
\[
 A_B=P-b_B,\qquad \beta_B=b_B/P.
\]
The measure $\mathbf A_B=\mathbf P-\mathbf b_B$ is nonnegative,
$A_B=G+\mathbf E_B(1)\ge d/2$, and
\begin{equation}\label{eq:v92-mixture}
 \mathsf Q_t=(1-\beta_B)\mathsf Q_{B,t}^{\rm good}
                       +\beta_B\mathsf Q_{B,t}^{\rm bad},\qquad
 \mathsf Q_{B,t}^{\rm good}=\mathbf A_B/A_B.
\end{equation}
The bad kernel can be assigned arbitrarily where $b_B=0$. Moreover,
\begin{align}
 d_{\mathrm{BL}}(\mathsf Q_{B,t}^{\rm good},\mathsf W_R)
                    &\le4\eta_B/d,\label{eq:v92-good-kernel}\\
 d_{\mathrm{BL}}(\mathsf Q_t,\mathsf W_R)
                    &\le2\beta_B+4\eta_B/d,\label{eq:v92-mixture-bound}\\
 \left|\beta_B-(1-G/P)\right|&\le2\eta_B/d.
                                                   \label{eq:v92-posterior-ratio}
\end{align}
The weights in \eqref{eq:v92-mixture} are the posterior masses of the
original guard and its complement, not a new source normalization.
\end{proposition}
\begin{proof}
The positive path remainder is the pushforward of the original guard
restriction. Since $0\le W^{\varepsilon(B)}\le1$, its complement
is a positive submeasure of $\mathbf P$. Subtracting that remainder
gives $\mathbf A_B=G\mathsf W_R+\mathbf E_B$ and the mass identity.
After normalization,
\[
 \|\mathsf Q_{B,t}^{\rm good}-\mathsf W_R\|_{\mathrm{BL}^*}
 \le\frac{\eta_B+|\mathbf E_B(1)|}{A_B}\le4\eta_B/d.
\]
Disintegration of the two positive restrictions proves the exact
mixture, and the distance between two probability measures in this
bounded-Lipschitz convention is at most two. Finally
$\beta_B=1-G/P-\mathbf E_B(1)/P$ and $P\ge d/2$.
All identities use the original common coarea version.
\end{proof}

\begin{theorem}[Finite-norm stability of same-roof collision kernels]
\label{thm:v92-kernels}
Under Inputs~\ref{input:v92-scalar}--\ref{input:v92-path}, for every
fixed $d,M,L>0$ and every finite $p\ge1$,
\begin{equation}\label{eq:v92-kernel-local}
 \sup_{R,n,k}\sup_{|J|=L}
 \left\|\one_{D_m(d)}
 d_{\mathrm{BL}}(\mathsf Q_t,\mathsf W_R)\right\|_{L^p(J)}
                                                    \longrightarrow0.
\end{equation}
The same assertion holds on the complete central window with measure
$dt/\sqrt m$. At each roof the distance already takes the supremum
over all bounded-Lipschitz path tests; the exceptional roof is not
chosen separately for each test. This is not path-space total variation,
not essential-supremum convergence in the roof, and not a new
collision-to-return clock theorem.
\end{theorem}
\begin{proof}
Let $\rho_m$ be the supremum of the nonnegative gap in
\eqref{eq:v92-numerator-gap}; then $\rho_m\to0$. For $P>0$,
\[
 P(\mathsf Q_t-\mathsf W_R)
   =(\mathbf P-G\mathsf W_R)+(G-P)\mathsf W_R.
\]
On $D_m(d)$ this gives
\begin{equation}\label{eq:v92-kernel-pointwise}
 d_{\mathrm{BL}}(\mathsf Q_t,\mathsf W_R)
       \le\min\{2,\,2|1-G/P|+2\rho_m/d\}.
\end{equation}
Apply Theorem~\ref{thm:v92-reciprocal}; the additive error is uniform.
The diffusive assertion is the same inequality in rescaled coordinates.
The countable norming class fixes the discrepancy as one measurable
function of the roof and therefore includes all path tests before
integration. No independent numerator convergence premise is inserted
at this step: it was deduced from the stated positive path remainder.
\end{proof}

\subsection{Posterior mass of the original exceptional union}
For arbitrary $\varepsilon>0$ and positive cutoffs $a,\delta$, let
\[
 H_{a,\delta}^\varepsilon(t)
   =m^2\int W^\varepsilon(x)
          \one_{\{z(x)<a\ \mathrm{or}\ \ell(x)<\delta\}}\,d\mu_t(x).
\]
On $D_m(d)$ its posterior mass is
$\pi_{a,\delta}^\varepsilon=H_{a,\delta}^\varepsilon/P$.
Extend this posterior by zero outside $D_m(d)$. It belongs to $[0,1]$.
Its two alternatives form one original union;
the intersection is not counted twice.

\begin{corollary}[All finite posterior moments of the same source]
\label{cor:v92-posterior}
Put $a=h^{1/8}$ and $\delta=h^{1/4}$. For every finite $p\ge1$,
\begin{equation}\label{eq:v92-posterior-power}
 \limsup_m\sup_{R,n,k}\sup_{|J|=L}\sup_{\varepsilon>0}
 \|\pi_{a,\delta}^\varepsilon\one_{D_m(d)}\|_{L^p(J)}
 \le C_{M,L,d,p}
 \begin{cases}
 h^{31/4096},&p<Q,\\
 h^{31/4096}(1+|\log h|)^{1/Q},&p=Q,\\
 h^{1/(128p)},&p>Q.
 \end{cases}
\end{equation}
The same bound dominates every smaller pair of cutoffs, without
assuming monotonicity of $W^\varepsilon$ in $\varepsilon$.
No conclusion at $p=\infty$ is asserted.
\end{corollary}
\begin{proof}
The original weighted union is bounded by the unweighted union $U$ in
Input~\ref{input:v92-critical}, for every $\varepsilon$. Thus its
posterior is at most $\min\{1,2m^2U/d\}$ on $D_m(d)$.
The critical norm of this common upper function has ordered bound
$C_{M,L,d}h^{31/4096}$. Apply Lemma~\ref{lem:v92-bounded-critical}
to the upper function at every finite count, and then take the limsup.
For $p>Q$ use $Q(31/4096)=1/128$. At $p=Q$ the logarithm is kept.
Positive domination also covers all smaller cutoffs simultaneously.
\end{proof}
