\subsection{A phase-dependent inverse for actual functions}
\label{sec:v76-envelope-principle}

The scalar tail estimate above can be organized by the phases visited by
a half-line.  This gives an inverse criterion that retains heterogeneous
contraction and separates it from the conditioning of finite jets.
The criterion applies to actual functions and to an integrated nonlinear
envelope; it does not require an analytic power series.

\begin{proposition}[Matrix domination of a contracted envelope]
\label{prop:v76-envelope}
Let $h_b,g_b$ be continuous functions on $[-R,R]$, $0\le b<r$,
vanishing at zero and with finite
weighted order-$m$ norms.  Suppose
\begin{equation}\label{eq:v76-abstract-envelope}
 g_b(u)=\alpha_b(u)h_b(u)
   +\int_0^1\sum_{j\ge1}v_{b,j}^t(u)
                       h_{c(b,j)}(\psi_{b,j}^t(u))\,dt,
\end{equation}
where $\alpha_b\ge\alpha_*>0$,
$|v_{b,j}^t|\le V_{b,j}$ and
$|\psi_{b,j}^t(u)|\le\gamma_{b,j}|u|$ with
$0\le\gamma_{b,j}\le1$.  Define the nonnegative matrix
\begin{equation}\label{eq:v76-domination-matrix}
 K_m(b,c)=\alpha_*^{-1}
     \sum_{j:\,c(b,j)=c}V_{b,j}\gamma_{b,j}^m.
\end{equation}
Assume its entries are finite and its spectral radius is less than one.
Writing
\[
 H_b=\sup_{0<|u|\le R}\frac{|h_b(u)|}{|u|^m},\qquad
 G_b=\sup_{0<|u|\le R}\frac{|g_b(u)|}{|u|^m},
\]
one has the componentwise estimate
\begin{equation}\label{eq:v76-vector-inverse}
 H\le\alpha_*^{-1}(I-K_m)^{-1}G.
\end{equation}
In particular, if $w_b>0$ and $K_mw\le\theta w$ for some
$\theta<1$, then
\begin{equation}\label{eq:v76-weight-inverse}
 \|h\|_{m,R,w}\le\frac{\alpha_*^{-1}}{1-\theta}
                      \|g\|_{m,R,w},\qquad
 \|h\|_{m,R,w}:=\max_b\sup_{0<|u|\le R}
                     \frac{|h_b(u)|}{w_b|u|^m}.
\end{equation}
Such a weight exists whenever the stated spectral-radius condition holds.
\end{proposition}
\begin{proof}
The summability in \eqref{eq:v76-domination-matrix} is an absolute
majorant for the envelope on the weighted space.  Dividing
\eqref{eq:v76-abstract-envelope} by $\alpha_b(u)|u|^m$ and taking
suprema yields $H\le\alpha_*^{-1}G+K_mH$.  Iterate this inequality
$n$ times.  The finite matrix $K_m^n$ tends to zero and
$\sum_{j\ge0}K_m^j=(I-K_m)^{-1}$, whose entries are nonnegative.
Taking the limit proves \eqref{eq:v76-vector-inverse}.
For a weight satisfying $K_mw\le\theta w$, divide the componentwise
inequality by $w_b$ and take a maximum to prove
\eqref{eq:v76-weight-inverse}.  Conversely choose any
$\theta\in(\rho(K_m),1)$ and set
$w=(I-K_m/\theta)^{-1}\mathbf1$.
Then $w\ge\mathbf1$ and $K_mw=\theta(w-\mathbf1)\le\theta w$.
All statements concern a fixed integrated envelope.  Its coefficients
may depend on both candidates; no observation-only iteration is asserted.
\end{proof}

\begin{corollary}[Phase-weighted smooth separation]
\label{cor:v76-phase-separation}
For the graph interpolation in
Proposition~\ref{prop:v68-weighted-separation}, suppose the one-step
maps satisfy, on a common collar and for all interpolation parameters,
\[
 |(X_i^t)'(u)|\le\widehat a_i<1,\qquad 0<\widehat a_i,
 \qquad \overline a=\left(\prod_{i=0}^{r-1}\widehat a_i\right)^{1/r}.
\]
For any integer $m\ge3$ such that $\overline a^m<1/7$, two graph
tuples with equal jets through degree $m-1$ satisfy
\begin{equation}\label{eq:v76-phase-inverse}
 \|G-F\|_{m,R,w}\le
 \frac{2}{1-\vartheta_m}\|\mathcal S(G)-\mathcal S(F)\|_{m,R,w},
 \qquad \vartheta_m=\frac{6\overline a^m}{1-\overline a^m},
\end{equation}
where $w_0=1$ and
\begin{equation}\label{eq:v76-phase-weights}
 w_b=\left(\prod_{i=0}^{b-1}\frac{\overline a}{\widehat a_i}\right)^m
 \quad(1\le b<r).
\end{equation}
In the unweighted norm the right-side constant is multiplied by
$\kappa(w):=\max_bw_b/\min_bw_b$.
\end{corollary}
\begin{proof}
The mean value theorem and the constant-one composition identity in
Lemma~\ref{lem:v68-real-visits} give
\[
 |x_{b,j}^t(u)|\le
       \left(\prod_{\ell=0}^{j-1}\widehat a_{b+\ell}\right)|u|.
\]
Use $\alpha_*=1/2$, $V_{b,j}=3$ and $c(b,j)=b+j\pmod r$ in
Proposition~\ref{prop:v76-envelope}.  If
$(A_mz)_b=\widehat a_b^m z_{b+1}$, its domination matrix is
\[
 K_m=6\sum_{j\ge1}A_m^j=6A_m(I-A_m)^{-1}.
\]
The positive weight \eqref{eq:v76-phase-weights} obeys
$A_mw=\overline a^m w$, including the closing phase, because
$\overline a^r=\prod_i\widehat a_i$.
It follows that $K_mw=\vartheta_m w$.  In fact conjugation by
$\diag(w)$ changes $A_m$ into $\overline a^m$ times the cyclic
permutation matrix, so its nonnegative geometric sum has spectral
radius $\vartheta_m$.  The hypothesis is exactly
$\vartheta_m<1$.  The preceding proposition proves
\eqref{eq:v76-phase-inverse}.  The norm inequalities
\[
 \|h\|_{m,R}\le(\max_bw_b)\|h\|_{m,R,w},\qquad
 \|g\|_{m,R,w}\le(\min_bw_b)^{-1}\|g\|_{m,R}
\]
prove the unweighted estimate.  The actual envelope, its terminal-limit
justification and the finite-jet factorization are those already proved;
only the estimate of its tail has changed.
\end{proof}

\begin{corollary}[Finite-smoothness inverse with phase bounds]
\label{cor:v76-phase-stability}
Fix a bounded positive marked class and an enlarged class for the
polynomial alignments of Lemma~\ref{lem:v68-finite-comparison}.
Suppose the latter has common one-step bounds $\widehat a_i$ as above.
Choose $m\ge3$ with $\overline a^m<1/7$ and retain the
$C^{m+3}$ graph prior and the positive density, gate and offset floors.
Then the conclusions \eqref{eq:v68-real-profile-stability} and
\eqref{eq:v68-law-profile-stability} hold at this order $m$ on a
smaller common collar, for sufficiently small data discrepancy.
\end{corollary}
\begin{proof}
The finite-jet comparison lemma holds at every stipulated finite order:
its proof uses the global quadratic inverse, the signed finite blocks
and the forward polynomial variation estimate, not the scalar tail
threshold.  Set $\eta=\|\mathcal S(F)-\mathcal S(G)\|_{C^m}$ and
form its polynomial correction $P$ and $G^\sharp=G+P$ exactly as in
Theorem~\ref{thm:v68-smooth-stability}.  Then
$\|P\|_{C^{m+3}}\le C\eta$.  For small enough $\eta$, both required
interpolations remain in the fixed enlarged positive class.  The actions
of $F$ and $G^\sharp$ have equal jets through order $m-1$, and
Taylor's formula and the forward variation bound give
\[
 \|\mathcal S(F)-\mathcal S(G^\sharp)\|_{m,R,w}
 \le \frac{C\eta}{\min_bw_b}.
\]
Apply \eqref{eq:v76-phase-inverse}, multiply by
$R^m\max_bw_b$, and add $\|P\|_{C^0}$.  This proves the action
estimate with a finite class-dependent constant.  The two-offset
quotient on the same positive gate gives the density estimate.
No curvature bound is inferred from the resulting $C^0$ graph error;
curvatures were controlled in the preceding jet alignment.
\end{proof}

The choice $\widehat a_i=a$ recovers the previous sufficient condition
and constant.  On a heterogeneous class the geometric-mean criterion
may allow a smaller order than $6(\max_i\widehat a_i)^m/
[1-(\max_i\widehat a_i)^m]<1$.  The price is the explicit weight
condition number, as well as the unchanged finite-jet constants.
Thus this refinement improves a sufficient inverse criterion without
asserting that the selected order is optimal or that the unweighted
inverse is uniformly better.  It also illustrates the reusable content
of the smooth argument: invert the finite jets, then dominate the
remaining actual-function envelope by a nonnegative visit matrix.
