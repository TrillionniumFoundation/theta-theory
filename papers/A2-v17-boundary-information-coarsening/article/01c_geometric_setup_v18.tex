\section{Geometric setting and the principal relative theorem}
\label{sec:v18-formal-setup}

Fix a full-rank lattice $\Lambda\subset\mathbb R^2$ and finitely many
bounded $C^\infty$ strictly convex obstacles $D_1,\ldots,D_s$ with
positive curvature. Distinct lattice translates have disjoint closures.
Let $A>0$ be the free area per lattice cell and prepare the unit-speed
billiard with normalized phase volume $dq\,d\vartheta/(2\pi A)$ on the
torus complement. We observe the collision count and, when specified, the
first and last collision positions. Finite horizon, congruence and symmetry
are not assumed for the general forward or boundary-law results.

Consider a fixed configuration and a smooth finite-dimensional family
$D_{a,\xi}$ through it, with $\xi$ in a compact set $K_0$ contained in a
sufficiently small parameter neighborhood. At every fixed derivative order,
constants may depend on the corresponding higher norms of this family, on
positive separation and curvature bounds, and on the lattice, but not on
the collision order.

Let $\mathcal E$ be the finite set, modulo common lattice translation, of
oriented closest pairs attaining the minimum gap at the reference
configuration. An index $e=(a,b,\lambda)$ represents a segment from
$D_{a,\xi}$ to $D_{b,\xi}+\lambda$; reversing it gives
$\bar e=(b,a,-\lambda)$. Write
\begin{equation}\label{eq:g-general-data}
 \begin{split}
 g_*&=\min_{e\in\mathcal E}g_e,\qquad
 c_{e,b}=1+g_e\kappa_{e,b}\quad(b=0,1),\\
 c_e&=\sqrt{c_{e,0}c_{e,1}},\qquad
 \gamma_e=\operatorname{arcosh}c_e,
 \qquad d_{j,e}=t-jg_e.
 \end{split}
\end{equation}
Here $\kappa_{e,0}$ and $\kappa_{e,1}$ are the positive curvatures at the
two contact points and need not be equal. A channel event $E_{j,e}(d)$
consists of the $j+1$ impacts alternating between the specified facing
patches in the time window $t=jg_e+d$. These patches are fixed around the
reference contacts before observation and define a measurable part of the
physical record. The localization theorem below proves that the relevant
pairs continue smoothly and that a common onset collar is available.

Parameter derivatives are multi-index derivatives in a fixed Euclidean
parameter chart; offset derivatives hold $d$ fixed as an independent
coordinate. Matrix norms are Euclidean operator norms unless a trace norm is
specified.

\subsection{Local multiplier convention}
\label{sec:v15-notation}
Suppress the channel index for the local hyperbolic calculations and set
\begin{equation}\label{eq:v15-multiplier-convention}
 \begin{aligned}
 \varrho&=e^{-\gamma},&\qquad \lambda&=\varrho^2=e^{-2\gamma},\\
 \mathfrak r_b&=\sqrt{c_b/c_{1-b}},&
 r_b&=\mathfrak r_b\varrho\quad(b=0,1).
 \end{aligned}
\end{equation}
The number $\mathfrak r_b$ is a curvature ratio, whereas $r_b$ is the
derivative at zero of the physical first-hit map $\varphi_b$. Thus
$\mathfrak r_0\mathfrak r_1=1$, $r_0r_1=\lambda$, and
$R_b=\varphi_{1-b}\circ\varphi_b$ has multiplier $\lambda$. The integer
$j$ counts flights; $n$ iterations of $R_b$ mean $j=2n$. The symbol
$\rho$ in exponential estimates denotes a freely chosen strict decay
margin, not the exact factor $\varrho$.

\subsection{The nonlinear relative law}
Fix an oriented channel $e$ and abbreviate $g=g_e$, $\gamma=\gamma_e$.
Use transverse endpoint coordinates $u,v$ at its two contacts. Let
$W_j(u,v)$ be the stationary length of its $j$ alternating flights,
$E_j=W_j-jg$, $d_j^0=-W_{j,uv}(0,0)$, and $b_j=-W_{j,uv}/d_j^0$.
For $p=j\bmod2$ put
\[
 a_b=\frac{\sqrt{c_0c_1}\sinh\gamma}{g c_{1-b}},\qquad
 q_p=\sqrt{a_0a_p},\qquad
 d_j^0=\frac{q_p}{\sinh(j\gamma)}.
\]

\begin{theorem}[Relative boundary law]\label{thm:v8-main-relative}
There are positive smooth functions $B_b$ and strictly convex smooth
functions $S_b$, $b=0,1$, with
\[
 S_b(0)=S_b'(0)=0,\qquad S_b''(0)=a_b,\qquad B_b(0)=1,
\]
a common endpoint box, a number $d_*>0$, and $0<\tau<1$ such that, for
every fixed derivative order $k$ and every $j\ge1$,
\begin{equation}\label{eq:v8-relative}
 \|E_j-S_0\oplus S_p\|_{C^k}
 +\|b_j-B_0\otimes B_p\|_{C^k}\le C_k\tau^j,
 \qquad p=j\bmod2.
\end{equation}
The action difference vanishes to second order at the origin. Define
\begin{equation}\label{eq:v8-boundary-integral}
 \mathcal F_p(d)=\frac{q_p}{\pi d^2}
 \int(d-S_0(u)-S_p(v))_+B_0(u)B_p(v)\,du\,dv.
\end{equation}
Then $\mathcal F_p$ extends smoothly to $[0,d_*]$, is positive there,
and has value one at zero. The exact selected-channel probabilities obey
\begin{equation}\label{eq:v8-relative-integral}
 \left\|\frac{2A\sinh(j\gamma)}{d^2}\Pr(E_{j,e}(d))
 -\mathcal F_p(d)\right\|_{C^k([0,d_*])}\le C_k\tau^j.
\end{equation}
The norm includes every fixed mixed geometric and offset derivative. The
limiting conditional endpoint--residual law at $d>0$ has density
proportional to
\[
 B_0(u)B_p(v)\mathbf 1_{\{r>0,\ S_0(u)+S_p(v)+r<d\}}.
\]
All assertions hold uniformly on the compact local families specified
above. For analytic families the endpoint constructions are analytic.
\end{theorem}

\begin{proof}[Proof and dependencies]
Lemma~\ref{lem:g-relative} constructs the finite bridge and its relative
determinant. Lemma~\ref{lem:v4-halfline} constructs $S_b,B_b$ with the
normalizations above. Theorem~\ref{thm:v4-factorization} proves
\eqref{eq:v8-relative} by comparing two separated perturbation blocks in
trace norm, including mixed derivatives. Theorem~\ref{thm:v4-law} proves
\eqref{eq:v8-relative-integral} using the common Morse domain of
Lemma~\ref{lem:g-radial} and identifies the conditional density. The full
proofs occur in Part I.
\end{proof}
