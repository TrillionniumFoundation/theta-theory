\section{The affine quotient and its observable residue condition}
\label{sec:residue}
Throughout this section $d=1$ and $\ell=3$. The parameters are as in
\eqref{eq:affine}. Write $P_j=P(T_j)$ and $\gamma=\|B\|_F$.

\begin{lemma}[The third clock and the pole-shift fibre]\label{lem:pole}
If $\gamma>0$, put $X=P_1-P_2$, $Y=P_2-P_3$, and
$a_0=(T_2-T_1)/(T_3-T_2)$. Then
\begin{equation}\label{eq:pole}
 r:=\frac{\langle X,Y\rangle_F}{\|Y\|_F^2}
       =a_0\frac{T_3-h}{T_1-h},\qquad
 h=\frac{rT_1-a_0T_3}{r-a_0},\qquad \|Y\|_F\asymp\gamma.
\end{equation}
The map from $h\in[L,D]$ to $r$ is uniformly bi-Lipschitz. If two
model observations are at distance $\delta\le c\gamma$, then
$|h-h'|\le C\delta/\gamma$. Also $\tau(P)\asymp\gamma$.

Let $H_0=(T_1-h)(T_2-h)$. For sufficiently small real $u$, define
\begin{align}
 a_x(u)&=1+u(h-x)/H_0,\nonumber\\
 x_u&=\frac{x+u\{1+(h-x)(T_1+T_2-h)/H_0\}}{a_x(u)},\nonumber\\
 \alpha_{b,u}&=\alpha_b a_{\xi_b}(u),\qquad h_u=h+u.
                                                        \label{eq:gauge}
\end{align}
With the same channels, these parameters give exactly the same two
observations at $T_1,T_2$, whenever they remain admissible. At interior
parameters they give a nonconstant exact fibre, since
\begin{equation}\label{eq:gauge-derivative}
 \left.\frac{d x_u}{du}\right|_{u=0}
                  =\frac{(T_1-x)(T_2-x)}{H_0}>0.
\end{equation}
\end{lemma}
\begin{proof}
Subtract $P(T)=A+B/(T-h)$ at consecutive clocks. This gives
\eqref{eq:pole}; the derivative of $r(h)$ is
$a_0(T_3-T_1)/(T_1-h)^2$, bounded above and below by positive
constants. At error $\delta\le c\gamma$, both difference norms are
bounded above by $C\gamma$ and the perturbed denominator norm is
bounded below by $c\gamma$. Expanding the ratio gives error
$C\delta/\gamma$, hence the pole estimate.

For $d=1$, $\cN_P$ has one column. Since $\Omega$ annihilates
constant and affine clock vectors,
\[
 \tau(P)=\|B\|_F\left\|\Omega\big((T_j-h)^{-1}\big)_{j=1}^3\right\|_2.
\]
The last norm is continuous and strictly positive for $h\in[L,D]$:
if a degree-one polynomial agreed with $(T-h)^{-1}$ at three distinct
points, multiplying by $T-h$ would give an impossible polynomial
identity. Compactness proves the comparison.

The affine interpolant of $p_b(T)$ at $T_1,T_2$ is
\[
 Q_b(z)=\alpha_b\left[1+(h-\xi_b)
                              \frac{T_1+T_2-h-z}{H_0}\right].
\]
The polynomial $\alpha_b(z-\xi_b)-uQ_b(z)$ has slope
$\alpha_{b,u}$ and root $(\xi_b)_u$. Summing these polynomials gives
$z-h-u$, so the weights sum to one and their weighted actions to
$h+u$. At either clock the polynomial equals
$(T_j-h-u)p_b(T_j)$. This proves the fibre assertion and differentiation
gives \eqref{eq:gauge-derivative}. In the perturbation arguments below
this gauge polynomial is used algebraically even if it leaves the
parameter interval; its slopes stay positive and its roots remain in
a fixed slightly larger interval.
\end{proof}

Let $\mathcal R$ be the subset of the affine model image on which
$\rank P_1=k$ and $P_2\ne P_3$. On $\mathcal R$, the preceding lemma
recovers $h$, then the two coefficients $A,C$ of $K(z)=zA-C$.
Choose orthonormal bases $Q_1,Q_2$ of the column and row spaces of $A$,
and set
\begin{equation}\label{eq:reduced}
 A_0=Q_1^{\mathsf T}AQ_2,\quad C_0=Q_1^{\mathsf T}CQ_2,
 \quad M(z)=zA_0-C_0.
\end{equation}
For each distinct generalized eigenvalue $x$, define
\begin{equation}\label{eq:eta}
 R_x=\Res_{z=x}M(z)^{-1},\qquad
 \beta(P)=\sum_x\|R_x\|_\op,\qquad
 \eta(P)=(\beta(P)+\gamma(P)^{-1})^{-1}.
\end{equation}
Set $\eta=0$ off $\mathcal R$. The residues are residues of a matrix
function, not rank-one factors chosen inside repeated eigenspaces.

\begin{theorem}[An observable affine singular invariant]\label{thm:residue}
The quantities $\beta$ and $\eta$ do not depend on the chosen
orthonormal bases or on a latent representation. On the affine model
image,
\begin{equation}\label{eq:zero-set}
 \eta(P)>0\quad\Longleftrightarrow\quad
 \sig_*(P)>0\quad\Longleftrightarrow\quad
 \rank P_1=k\ \hbox{ and }\ P_2\ne P_3.
\end{equation}
For every representation of $P\in\mathcal R$,
\begin{equation}\label{eq:beta-bounds}
 \sigma_k(A)^{-1}\le\beta(P)\le\frac{k}{\alpha_*\kappa},
 \qquad \eta(P)\ge c\sig_*(P).
\end{equation}
Every competitor in the full closed affine model satisfies
\begin{equation}\label{eq:eta-inverse}
 d_\infty(\xi,\xi')+|h-h'|
     \le C\min\{1,\delta/\eta(P)\}.
\end{equation}
All constants have precisely the fixed-parameter dependencies stated
in Section~1; no individual action gap enters them.

On $\mathcal R$, $\beta$ is lower semicontinuous and is continuous on
each fixed action-multiplicity stratum. The extension of $\eta$ by
zero is upper semicontinuous on the whole model image, continuous on
those strata, and tends to zero at its boundary zero set.
The envelope $\sig_*$ is upper semicontinuous on the whole image.
\end{theorem}
\begin{proof}
For any representation, put $S=Q_1^{\mathsf T}U$ and
$T=Q_2^{\mathsf T}V$. These are invertible, and
\begin{equation}\label{eq:residue-formula}
 M(z)^{-1}=T^{-\mathsf T}
       \diag\left(\frac{1}{\alpha_b(z-\xi_b)}\right)S^{-1}
       =\sum_x\frac{R_x}{z-x},\qquad
 R_x=T^{-\mathsf T}\diag(\alpha_b^{-1}\mathbf1_{\xi_b=x})S^{-1}.
\end{equation}
Thus every pole is simple, even when the corresponding action has
higher multiplicity. An orthogonal change of bases multiplies each
residue on its left and right by orthogonal matrices, preserving its
operator norm. The construction otherwise uses only $h,A,C$.
Taking the coefficient at infinity gives
$\sum_xR_x=A_0^{-1}$. Each residue norm is at most
$1/(\alpha_*\kappa)$, proving \eqref{eq:beta-bounds} before its last
assertion. Every representation of a rank-$k$ observation has full
rank, since all diagonal probabilities are positive. Further,
$P_2=P_3$ is equivalent to $B=0$. This proves \eqref{eq:zero-set}.
The bounds on $\beta$ imply $\eta\ge c\sig$ for every
representation, hence $\eta\ge c\sig_*$.

To prove the inverse, it suffices to consider
$\delta(\beta+\gamma^{-1})$ small. The pole lemma gives
$u=h'-h=O(\delta/\gamma)$. Let $Q(z)$ be the affine interpolant of
$P_1,P_2$, and form the true gauge polynomial
$\overline K_u(z)=K(z)-uQ(z)$. Its reduced inverse is
\begin{equation}\label{eq:gauged-resolvent}
 \overline M_u(z)^{-1}
       =\sum_x\frac{R_x/a_x(u)}{z-x_u}.
\end{equation}
For sufficiently small $|u|$, uniformly in the actions,
$1/2\le a_x(u)\le3/2$ and $|x_u-x|\le C|u|$.
The fractional-linear map $x\mapsto x_u$ has positive derivative
close to one on $[L,D]$, so it preserves distinct groups and their
multiplicities. Consequently
\[
 \|\overline M_u(z)^{-1}\|
                 \le\frac{2\beta(P)}{\dist(z,\{x_u\})}.
\]
On the other hand, $M'(z)=Q_1^{\mathsf T}K'(z)Q_2$ satisfies
$M'=\overline M_u+E$, where $E$ is the affine interpolant of
\[
 (T_j-h')Q_1^{\mathsf T}(P'_j-P_j)Q_2,\qquad j=1,2.
\]
Its coefficient norm is at most $C\delta$, with no channel inverse.
The leading inverse of $\overline M_u$ has norm at most $2\beta$,
so the leading coefficient remains invertible throughout
$\overline M_u+sE$, $0\le s\le1$. The same resolvent bound first
excludes roots outside a fixed disk and then excludes them outside
disks of radius $C\beta\delta$ about the $x_u$. The contour and
multiplicity argument in Lemma~\ref{lem:roots}, now with degree $k$,
gives a matching within $C\beta\delta$. The invertible leading
coefficient of $M'$ forces both reduced competitor channel factors
to be invertible. Its roots are therefore precisely $\xi'$.
Combining this matching with $|x_u-x|\le C\delta/\gamma$ proves
\eqref{eq:eta-inverse}. Compactness covers larger errors.

For regularity, let $P^{(n)}\to P$ in $\mathcal R$. Locally one can
choose continuous orthonormal frames for the rank-$k$ column and row
spaces, for example by projecting fixed independent columns and
orthonormalizing. The recovered $h,A,C$, hence the reduced pencils,
then converge. Choose disjoint circles enclosing the distinct actions
of the limit pencil and no others. The inverse pencils converge
uniformly on these circles. Integrating gives
\[
 \sum_{y\ \mathrm{in\ the\ cluster\ of}\ x}R_y^{(n)}\longrightarrow R_x.
\]
The triangle inequality and summation imply
$\beta(P)\le\liminf_n\beta(P^{(n)})$. If the multiplicity pattern
is fixed locally, each circle contains one distinct pole, so each
residue converges and $\beta$ is continuous. Since $\gamma>0$ is
continuous here, this gives the assertions for $\eta$ on $\mathcal R$.

At a constant-curve limit, the difference formula in the pole lemma,
uniformly for $h\in[L,D]$, gives $\gamma\to0$, hence $\eta\to0$.
At a nonconstant rank-loss limit, $h,A$ still converge and
$\sigma_k(A)\to0$. Since $\beta\ge\sigma_k(A)^{-1}$, again
$\eta\to0$. This proves the extension statement. Finally, choose
maximizing representatives in \eqref{eq:envelope} along a sequence
realizing a limsup. A compact subsequence converges to a representation
of the limit datum. Continuity of $\sig$ proves upper semicontinuity
of $\sig_*$.
\end{proof}

\begin{remark}[Why semicontinuity is the appropriate statement]
Take $k=m_1=m_2=3$, $U=V=I$, $\alpha_b=1/3$, and actions
$(a-t,a+t,b)$ with $a<b$ and small $t\ge0$. For $t>0$ the three
residue norms equal $3$, so $\beta=9$. At $t=0$ the residue at $a$
is $3\diag(1,1,0)$, while the residue at $b$ has norm $3$; hence
$\beta=6$. The contrast stays positive. Thus $\eta$ has an upward
jump at this partial collision and is not globally continuous on
$\mathcal R$. This example does not assert discontinuity of
$\sig_*$. The zero-set statement is a characterization of the
certificate, not a claim that every rank-deficient datum loses every
possible spectral feature, nor a distance-to-discriminant formula.
\end{remark}

\begin{proposition}[Exact factors and the original affine modulus]
\label{prop:factors}
For a full-rank nonconstant affine datum, every exact representation
has the same $h$ and spectrum and has full-rank channels. If the
spectrum is simple, the channels and weights are unique up to a common
permutation. At a repeated action, its total matrix coefficient is
unique, but its individual positive rank-one factors need not be.
Moreover, if $\xi_k-\xi_1\ge\Delta>0$, then
$\sig\ge c_\Delta\kappa$. The original one-sided estimate with
$\sig_*$ and all its gap-free consequences follow from
\eqref{eq:eta-inverse}.
\end{proposition}
\begin{proof}
The pole is determined by \eqref{eq:pole}. Hence $A,C$ are determined,
and $\rank A=k$. In an alternative exact representation
$A=U'\diag(\alpha')V'^{\mathsf T}$; its rank forces both alternative
channels to have full column rank. This justifies applying spectral
reconstruction to every point of the exact fibre, not just to a
selected full-rank representative.

Choose row and column index sets $I,J$ with $A_{IJ}$ invertible.
The spectral projector $\Pi_x$ of
$K(T_1)_{IJ}A_{IJ}^{-1}$ at $T_1-x$ gives
\[
 E_x=A_{:J}A_{IJ}^{-1}\Pi_x A_{I:}
                =\sum_{\xi_b=x}\alpha_b U_{:b}V_{:b}^{\mathsf T}.
\]
For a simple action, entry sum, row sums and column sums of $E_x$
recover $\alpha_b,U_{:b},V_{:b}$. At a repeated action only their
displayed sum is determined. Finally,
\[
 \gamma\ge\kappa\left(\sum_b\alpha_b^2(h-\xi_b)^2\right)^{1/2}
                    \ge\tfrac12\alpha_*\Delta\kappa.
\]
Together with $\sig=\kappa\gamma/(\kappa+\gamma)$ this proves the
last claim. In particular, on every simple-spectrum exact fibre,
$\sig_* =\sig$.
\end{proof}
