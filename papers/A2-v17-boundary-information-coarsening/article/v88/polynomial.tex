\section{Normalization and the finite-clock fibre}
\label{sec:polynomial}
The normalizing polynomial is constrained by the requirement that
$q(T_j)P(T_j)$ interpolate a matrix polynomial of degree at most $d$.
The operator in \eqref{eq:normalization} records precisely this
constraint. We first determine its nullspace.

\begin{lemma}[The normalization defect]\label{lem:defect}
Under the rank assumption of Theorem~\ref{thm:main}, the greatest common
divisor of all entries of $K$ is $g=\gcd(f_1,\ldots,f_k)$. If
$\ell\ge2d+1$, then \eqref{eq:kernel} holds.
\end{lemma}
\begin{proof}
Every entry of $K$ is a constant linear combination of the $f_b$, so
$g$ divides those entries. Conversely, multiplication by the constant
left inverse $U^\dagger$ and right inverse $(V^\dagger)^{\mathsf T}$
recovers $\diag(\alpha_b f_b)$. Every common divisor of the entries
therefore divides every $f_b$.

Let $r$ have degree less than $d$ and belong to $\ker\cN_P$.
There is a matrix polynomial $H$ of degree at most $d$ such that
$H(T_j)=r(T_j)P(T_j)$. Each entry of $qH-rK$ has degree at most $2d$
and vanishes at $2d+1$ distinct clocks. Hence $qH=rK$ identically.
Write $K=g\widetilde K$ and $q=g\widetilde q$. The entries of
$\widetilde K$ have greatest common divisor one. Their Bezout identity
implies that $\widetilde q$ divides $r$. Thus $r=\widetilde q s$,
where $\deg s<\deg g$. Conversely, for every such $s$ the polynomial
$H=\widetilde K s$ has degree at most $d-1$ and satisfies $qH=rK$.
This proves both assertions, including the case $\deg g=d$.
\end{proof}

The defect is an actual normalization ambiguity. Cancelling $g$ changes
the visible degree from $d$ to $d-\deg g$; the zeros of the cancelled
factor cannot be read from the normalized curve. At a shorter design,
aliasing can produce a normalization defect even when $g=1$.

\begin{lemma}[Positive local fibres]\label{lem:local-fibre}
Suppose $\ell\ge d+1$, the true channels have full column rank,
$\alpha_b>\alpha_*$, and all roots of each $f_b$ are simple and lie
in $(L,D)$. If $0\ne r\in\ker\cN_P$, there is a smooth curve of
parameters with the same $U,V$, unchanged observations at every
sampled clock, and a nonconstant root multiset.
\end{lemma}
\begin{proof}
Let $H$ be the degree-at-most-$d$ interpolant of $r(T_j)P(T_j)$.
All its values lie in the space
$\{U\diag(x_b)V^{\mathsf T}:x\in\R^k\}$.
Interpolation at the first
$d+1$ nodes shows that
$H(z)=U\diag(h_b(z))V^{\mathsf T}$ with $\deg h_b\le d$.
Since the sum of the entries of $P(T_j)$ is one, the polynomial
$\sum_bh_b-r$ vanishes at $d+1$ nodes and is identically zero.

For small real $u$, put
\[
 q_u=q+ur,\qquad K_u=K+uH,
 \qquad w_{b,u}=\alpha_b f_b+u h_b.
\]
Let $\alpha_{b,u}$ be the leading coefficient of $w_{b,u}$ and set
$f_{b,u}=w_{b,u}/\alpha_{b,u}$. Then
$\sum_b\alpha_{b,u}=1$ and $q_u=\sum_b\alpha_{b,u}f_{b,u}$.
The strict weight inequalities persist. Real simple roots in an open
interval persist under sufficiently small real coefficient
perturbations: choose disjoint brackets about the roots with opposite
signs at their endpoints. Each bracket retains a root, accounting for
all $d$ roots. These roots are smooth by the implicit function theorem.
Moreover $K_u(T_j)=q_u(T_j)P(T_j)$, as required.

If the root multiset were unchanged for sufficiently small nonzero
$u$, each component root, being close to its original value, would have
to equal that value. Thus $f_{b,u}=f_b$ for every $b$.
Applying the true channel inverses to the equality of observations
then gives $\alpha_{b,u}/\alpha_b=q_u(T_j)/q(T_j)$ for every $b,j$.
Summing the weights makes this common ratio one. Consequently $r$
vanishes at $d+1$ nodes, which contradicts $r\ne0$ and $\deg r<d$.
\end{proof}

\begin{proposition}[Sharp worst-case clock count]\label{prop:clocks}
For every admissible fixed $k,m_1,m_2$, there are positive full-rank
channels and interior, relatively prime component polynomials for
which any specified $2d$ distinct clocks admit the fibre of
Lemma~\ref{lem:local-fibre}.
\end{proposition}
\begin{proof}
First take $k=2$ and write
$P(T)=E_2+(E_1-E_2)w(T)$, where
$E_b=U_{:b}V_{:b}^{\mathsf T}$ and
$w=\alpha_1f_1/q$. Since $\Omega(r(T_j))_j=0$ for $\deg r<d$,
\[
 \cN_Pr=\big[\Omega(r(T_j)w(T_j))_j\big]
                    \otimes\operatorname{vec}(E_1-E_2).
\]
At $2d$ nodes the residual scalar space has dimension $d-1$.
Thus the map from the $d$ coefficients of $r$ has nonzero kernel,
including when $f_1,f_2$ are relatively prime.

For arbitrary $k$, choose two close, relatively prime monic
polynomials $f,g$ with simple roots in $(L,D)$. They can be chosen so
that every $(1-s)f+sg$, $0\le s\le1$, has simple roots in that
interval, by the bracket argument in the preceding lemma. Choose
$s_b\in[0,1]$, including both endpoints, and set
$f_b=(1-s_b)f+s_bg$. The collection is relatively prime. Its matrix
numerator belongs to the span of two fixed matrices, and normalization
restricts it to an affine line, since every observed matrix has entry
sum one. The same residual-dimension argument applies. Take
$\alpha_b=1/k$ and any strictly positive full-rank stochastic channels.
Lemma~\ref{lem:local-fibre} finishes the proof.
\end{proof}

\subsection{Root matching without cross-component gaps}
For a polynomial matrix $F(z)=\sum_{r=0}^dF_rz^r$, write
$\|F\|_{\mathrm{coef}}=\sum_r\|F_r\|_\op$.
The next lemma states explicitly the root-counting step used below.

\begin{lemma}[Diagonal polynomial perturbations]\label{lem:roots}
Let $D_f(z)=\diag(\alpha_b f_b(z))$ be as in \eqref{eq:model}.
If $\|E\|_{\mathrm{coef}}\le\varepsilon$ is sufficiently small,
then $\det(D_f+E)$ has degree $kd$ and its root multiset $Z_E$
satisfies
\[
 d_\infty(\Lambda,Z_E)\le C\varepsilon^{1/d}.
\]
The roots in $Z_E$ may be complex. If the roots within each $f_b$
are simple and $\nu$-separated, the bound is $C_\nu\varepsilon$.
No separation between roots of different components is required.
\end{lemma}
\begin{proof}
The leading coefficient of $D_f+sE$ is invertible for $0\le s\le1$,
provided $\varepsilon<\alpha_*/2$. For $|z|$ larger than a fixed
constant, $|f_b(z)|\ge c|z|^d$, whereas
$\|E(z)\|\le C\varepsilon|z|^d$. A Neumann series therefore excludes
zeros outside a fixed disk, uniformly in $s$.
Inside that disk, if $\rho=\dist(z,\Lambda)$, then
\[
 \|D_f(z)^{-1}\|\le\alpha_*^{-1}\rho^{-d},
 \qquad \|E(z)\|\le C\varepsilon.
\]
Thus every zero along the homotopy lies in the union of disks of radius
$r=(C_1\varepsilon)^{1/d}$ about the true roots, with $C_1$ fixed
large enough to make the exclusion strict outside these disks.

For completeness, fix any $r'>r$ which is not one of the finitely many
radii at which two of the disks first touch. Distinct components of
the union of the closed $r'$-disks have positive distance from one
another. Each can be surrounded by finitely many piecewise smooth
contours contained in the complement of the closed $r$-disks, enclosing
that component and no other. Slight outer perturbations of the
boundary of its disk union give such contours; holes may be filled,
since the zeros already lie in the $r$-disk union. The determinant is
nonzero on the contours for the whole homotopy. The argument principle
therefore keeps the number of enclosed roots, counted with algebraic
multiplicity, equal to its value at $s=0$.

If a component contains $n$ centres with multiplicity, a chain of its
disks joins any two distinct centres in at most $n-1$ steps. Their
distance is at most $2(n-1)r'$. Every root enclosed in that component
is within $r$ of one of its centres. Since there are exactly $n$
roots, any bijection inside the component has distance at most
$(2n-1)r'$. Let $r'\downarrow r$ through nonexceptional radii.
There are only finitely many permutations, so one can take a
subsequence with a fixed matching. This proves the asserted estimate,
including touching components and repeated centres. For real spectra,
uncrossing inverted pairs proves the equivalent ordered matching.

Under the within-component separation assumption, near a root of
$f_b$ all its other factors have magnitude bounded below in terms of
$\nu$. Away from those neighbourhoods $|f_b|$ is uniformly positive.
On the fixed disk this gives
$|f_b(z)|\ge c_\nu\min\{1,\dist(z,Z(f_b))\}$.
The same argument then uses radius $C_\nu\varepsilon$.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:main}]
The kernel and fibre assertions have been proved. For a competitor,
$\cN_{P'}a'=b_{P'}$. Projecting its cleared observations gives
\begin{equation}\label{eq:normalization-difference}
 \cN_P(a'-a)=(\Omega\otimes I)
       \big(q'(T_j)\operatorname{vec}(P(T_j)-P'(T_j))\big)_j.
\end{equation}
Uniform coefficient and clock-value bounds for $q'$ imply
$\|a'-a\|\le C\delta/\tau$. Compactness supplies the cap in
\eqref{eq:q-bound}.

Let $L_U=U^\dagger$, $R_V=(V^\dagger)^{\mathsf T}$, so that
$\|L_U\|\|R_V\|=\kappa^{-1}$. Set $M(z)=L_UK'(z)R_V$.
At the first $d+1$ clocks,
\begin{align}
 M(T_j)-\diag(\alpha_b f_b(T_j))
 &=q'(T_j)L_U(P'(T_j)-P(T_j))R_V\nonumber\\
 &\quad +(q'(T_j)-q(T_j))\diag(p_b(T_j)).
                                                        \label{eq:diagonal-error}
\end{align}
The second term is diagonal. Interpolation at this fixed design gives
\[
 \|M-D_f\|_{\mathrm{coef}}
              \le C\delta(\kappa^{-1}+\tau^{-1})=:C\epsilon.
\]
In particular the normalization error has not acquired a factor
$\kappa^{-1}$. For small $\epsilon$, the leading coefficient of $M$
is invertible. It is also
$(L_UU')\diag(\alpha')(V'^{\mathsf T}R_V)$, so both square outer
factors must be invertible. Therefore
\[
 \det M(z)=\det(L_UU')\det(V'^{\mathsf T}R_V)
                                  \prod_b\alpha'_b f'_b(z).
\]
Its roots are exactly the competitor multiset, even though no initial
rank assumption was made on that competitor. Lemma~\ref{lem:roots}
proves both small-error bounds. For larger $\epsilon$, all target
roots belong to $[L,D]$, which proves the capped estimates.
\end{proof}

\begin{remark}
The exponent $1/d$ is a uniform upper bound, not a claim of an optimal
exponent at every real-rooted singularity. The linear result requires
separation within a component polynomial, not separation of the
aggregate spectrum. In degree one this additional assumption is empty.
For $d>1$, the kernel identity identifies common-factor cancellation,
whereas the root modulus also records multiple roots within a scalar
component. These are different mechanisms.
\end{remark}
