\section{Vector boundary information and a self-contained LAN theorem}
\label{sec:v20-vector-information}

The scalar theorem of Section~\ref{sec:v17-boundary-information} identifies the
borderline logarithmic information carried by a density which vanishes linearly
on a moving smooth boundary.  We record here the vector form in a self-contained
way.  This section is an abstract statistical statement on a fixed observation
space.  The billiard realization on parameter-independent physical windows is
given in Section~\ref{sec:v20-raw-physical}.

Let $\Theta\subset\mathbb R^r$ be a neighborhood of the origin.  On a fixed open
set $U\subset\mathbb R^m$ suppose
\begin{equation}\label{eq:v20-vector-density}
 f_\theta(x)=a_\theta(x)(w_\theta(x))_+,
 \qquad \int_U f_\theta(x)\,dx=1.
\end{equation}
Assume, uniformly for $\theta$ near zero, that $a_\theta,w_\theta$ are $C^3$ in
$(\theta,x)$ on a common compact set, $a_\theta$ is positive there, and
\[
 \Sigma:=\{w_0=0\}\cap\operatorname{supp}a_0
\]
is a compact regular hypersurface with $|\nabla w_0|\ge c_*>0$.  Shrinking the
parameter neighborhood if necessary, all moving boundaries are contained in a
fixed tubular collar.  Put
\begin{equation}\label{eq:v20-boundary-velocity-matrix}
 V(x)=D_\theta w_\theta(x)|_{\theta=0}\in\mathbb R^r,
 \qquad
 \mathcal J_\Sigma=
 \int_\Sigma\frac{a_0(x)V(x)V(x)^t}{|\nabla w_0(x)|}\,d\sigma(x).
\end{equation}
The matrix is positive semidefinite.  Multiplying $w_\theta$ by a positive
smooth factor and dividing $a_\theta$ by the same factor leaves
$\mathcal J_\Sigma$ unchanged.

\begin{lemma}[Uniform vector collar estimates]
\label{lem:v20-vector-collar}
Let
\[
 \mathsf S(x)=D_\theta\log a_\theta(x)|_0+\frac{V(x)}{w_0(x)},
 \qquad D=\{w_0>0\}.
\]
There are $q_0,C>0$ such that, for $0<q<q_0$ and unit vectors
$u,v\in\mathbb R^r$,
\begin{align}
 \mathbb E_0[(u^t\mathsf S)(v^t\mathsf S)
       \mathbf 1_{\{w_0\ge q\}}]
 &=u^t\mathcal J_\Sigma v\log(1/q)+O(1),
                                      \label{eq:v20-vector-second}\\
 \mathbb E_0[|u^t\mathsf S|^3\mathbf 1_{\{w_0\ge q\}}]
 &\le Cq^{-1},                       \label{eq:v20-vector-third}\\
 \mathbb E_0[|u^t\mathsf S|^4\mathbf 1_{\{w_0\ge q\}}]
 &\le Cq^{-2}.                       \label{eq:v20-vector-fourth}
\end{align}
Moreover
\begin{equation}\label{eq:v20-vector-centering}
 \left\|\mathbb E_0[\mathsf S\mathbf 1_{\{w_0\ge q\}}]\right\|
 \le Cq,
 \qquad
 \mathbb P_0(0<w_0<q)\le Cq^2.
\end{equation}
All constants are uniform for the indicated unit directions.
\end{lemma}

\begin{proof}
Use normal coordinates $(y,s)$ in a fixed tubular neighborhood of
$\Sigma$, with $s=w_0$ and Jacobian
$|\nabla w_0|^{-1}+O(s)$.  Since $f_0=a_0s$ in the interior collar, the
singular part of
$(u^t\mathsf S)(v^t\mathsf S)f_0$ is
\[
 \frac{a_0(u^tV)(v^tV)}s.
\]
Integrating $s^{-1}$ from $q$ to a fixed collar width gives
\eqref{eq:v20-vector-second}; all mixed terms containing at most one factor
$s^{-1}$ are integrable and contribute $O(1)$.  The same coordinates give
$s^{1-k}$ for the $k$th singular score moment, proving
\eqref{eq:v20-vector-third}--\eqref{eq:v20-vector-fourth}.  They also give
quadratic collar mass.  Differentiating the normalization in
\eqref{eq:v20-vector-density} yields $\mathbb E_0\mathsf S=0$ in the
improper integrable sense; the omitted collar contributes $O(q)$ to its
first moment, which proves \eqref{eq:v20-vector-centering}.
\end{proof}

\begin{lemma}[Uniform truncated likelihood expansion]
\label{lem:v20-vector-likelihood}
Let $\delta_n\downarrow0$, $0<p_n\le1$, and
\begin{equation}\label{eq:v20-vector-rate}
 n p_n\delta_n^2\log(1/\delta_n)\longrightarrow1.
\end{equation}
Put
\[
 L_n=\log(1/\delta_n)^{1/4},\qquad q_n=\delta_nL_n.
\]
For $h$ in a fixed compact subset of $\mathbb R^r$, define
\[
 P_{n,h}=(1-p_n)\delta_\dagger+p_nf_{\delta_nh}(x)\,dx.
\]
Under $P_{n,0}^{\otimes n}$ the probability that a successful observation
falls in $0<w_0<q_n$, or in the support-exclusive part of a bounded local
alternative, tends to zero uniformly in $h$.  On the complementary event,
\begin{equation}\label{eq:v20-vector-uniform-expansion}
 \log\frac{dP_{n,h}^{\otimes n}}{dP_{n,0}^{\otimes n}}
 =h^t\Delta_n-\frac12h^tQ_nh+o_P(1)
\end{equation}
uniformly on compact $h$-sets, where, writing $Y_i$ for the success
indicator and $X_i\sim f_0$ on success,
\begin{align}
 \Delta_n&=\delta_n\sum_{i=1}^nY_i
 \mathsf S(X_i)\mathbf 1_{\{w_0(X_i)\ge q_n\}},
                                                  \label{eq:v20-delta}\\
 Q_n&=\delta_n^2\sum_{i=1}^nY_i
 \mathsf S(X_i)\mathsf S(X_i)^t
 \mathbf 1_{\{w_0(X_i)\ge q_n\}}. \label{eq:v20-Q}
\end{align}
Furthermore
\begin{equation}\label{eq:v20-central-limits}
 \Delta_n\Rightarrow N_r(0,\mathcal J_\Sigma),
 \qquad Q_n\longrightarrow\mathcal J_\Sigma
 \quad\text{in probability}.
\end{equation}
\end{lemma}

\begin{proof}
Lemma~\ref{lem:v20-vector-collar} gives
\[
 np_nq_n^2
 =O\!\left(\frac{L_n^2}{\log(1/\delta_n)}\right)=o(1),
\]
and the moving-support strip has the same order uniformly for bounded
$h$.  On the common support a third-order Taylor expansion in the parameter
and in the logarithm gives
\[
 \log\frac{f_{\delta_nh}}{f_0}
 =\delta_n h^t\mathsf S
 -\frac{\delta_n^2}{2}(h^t\mathsf S)^2
 +\delta_n^2R_{n,h}.
\]
The parameter-Taylor remainder is bounded by the same
$1+w_0^{-1}$ envelope as in the scalar proof, while the cubic logarithmic
remainder is controlled by \eqref{eq:v20-vector-third}.  After summation
both are $o_P(1)$ uniformly on compact $h$-sets.  This proves
\eqref{eq:v20-vector-uniform-expansion}.

The centering error of the truncated score has norm at most
$Cnp_n\delta_nq_n$.  By \eqref{eq:v20-vector-rate} this is
$O(L_n/\log(1/\delta_n))=o(1)$.  The covariance of $\Delta_n$ converges
to $\mathcal J_\Sigma$ by \eqref{eq:v20-vector-second}.  On the truncated
set the largest normalized scalar score contribution is $O(L_n^{-1})$;
therefore Lindeberg holds uniformly on the unit sphere.  Cram\'er--Wold
gives the first limit in \eqref{eq:v20-central-limits}.  Finally
\eqref{eq:v20-vector-fourth} and the rate relation imply that every entry
of $Q_n$ has vanishing variance after centering, while its mean tends to
the corresponding entry of $\mathcal J_\Sigma$.  This proves the second
limit.
\end{proof}

\begin{theorem}[Vector boundary LAN and local minimax bound]
\label{thm:v20-vector-boundary-lan}
Under the hypotheses above,
\begin{equation}\label{eq:v20-lan}
 \log\frac{dP_{n,h}^{\otimes n}}{dP_{n,0}^{\otimes n}}
 =h^t\Delta_n-\frac12h^t\mathcal J_\Sigma h+o_P(1)
\end{equation}
uniformly for $h$ in compact subsets of $\mathbb R^r$.  Hence every fixed
finite local subexperiment converges to the Gaussian shift with information
matrix $\mathcal J_\Sigma$.  In particular, for fixed $h$,
\begin{equation}\label{eq:v20-testing-profile}
 \|P_{n,h}^{\otimes n}-P_{n,0}^{\otimes n}\|_{\rm TV}
 \longrightarrow
 2\Phi\!\left(\frac{\sqrt{h^t\mathcal J_\Sigma h}}2\right)-1.
\end{equation}
If $\mathcal J_\Sigma$ is positive definite, the local central-sequence
estimator
\begin{equation}\label{eq:v20-local-estimator}
 \widehat h_n=\mathcal J_\Sigma^{-1}\Delta_n
\end{equation}
satisfies, under $P_{n,h}^{\otimes n}$,
\[
 \widehat h_n\Rightarrow N_r(h,\mathcal J_\Sigma^{-1})
\]
uniformly on compact local parameter sets.

For a positive semidefinite matrix $W$ and truncated quadratic losses
$\ell_M(z)=\min\{z^tWz,M\}$, the local asymptotic minimax theorem gives
\begin{equation}\label{eq:v20-minimax-truncated}
 \lim_{R\to\infty}\liminf_{n\to\infty}
 \inf_{T_n}\sup_{\|h\|\le R}
 \mathbb E_{n,h}\ell_M(T_n-h)
 \ge
 \mathbb E\ell_M(Z),
 \qquad Z\sim N_r(0,\mathcal J_\Sigma^{-1}).
\end{equation}
Letting $M\to\infty$ yields the quadratic lower bound
$\operatorname{tr}(W\mathcal J_\Sigma^{-1})$.  For
\eqref{eq:v20-local-estimator}, the fourth-moment estimates above imply
uniform integrability on each fixed compact local parameter set, so its
quadratic risk converges there to the same value.
\end{theorem}

\begin{proof}
The LAN expansion follows immediately from
Lemmas~\ref{lem:v20-vector-likelihood} and
\ref{lem:v20-vector-collar}.  Le Cam's third lemma gives the translated
Gaussian limit of $\Delta_n$ under fixed local alternatives and hence
\eqref{eq:v20-local-estimator}.  Formula~\eqref{eq:v20-testing-profile}
is the exact total variation distance between equal-covariance Gaussian
shifts.  The minimax assertion is the finite-dimensional local asymptotic
minimax theorem applied first to the bounded continuous losses
$\ell_M$; see \cite{IbragimovHasminskii,HiranoPorter}.  Monotone
convergence then gives the quadratic lower bound.  The same fourth-moment
bounds used for $Q_n$ give uniform integrability of the displayed local
estimator on fixed compact sets.
\end{proof}

\begin{remark}[What the theorem does and does not say]
The estimator \eqref{eq:v20-local-estimator} is a local central-sequence
estimator centered at a specified reference parameter.  No claim is made
here that it is, by itself, a globally implementable estimator when the
reference geometry, coordinate registration or acquisition window is
unknown.  Section~\ref{sec:v20-raw-physical} supplies the physical local
experiment and states exactly which registration variables are fixed and
which geometric coordinates are unknown.
\end{remark}
