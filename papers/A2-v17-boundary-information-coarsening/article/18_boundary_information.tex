\section{Boundary information after residual-time integration}
\label{sec:v17-boundary-information}
Residual-time integration has two different effects. It preserves the
smooth relative law of Theorem~\ref{thm:v8-main-relative}, but it changes
the statistical singularity of a moving support. Before integration the
conditional density is positive up to the moving boundary. Afterwards it
vanishes there to first order. We give a boundary integral for the
resulting logarithmic information, and then identify it for physical
collision records. The limit below is not obtained by contracting the
full-record total variation estimate: that estimate is no longer sharp
after this observation is removed.

\subsection{A linearly vanishing density}
Let $U$ be a bounded open subset of $\R^m$. For $|t|<t_0$, let $a_t$
and $w_t$ be real functions which are $C^3$ jointly on a neighborhood of
$[-t_0,t_0]\times\overline U$. Assume that $a_t$ is bounded above and
bounded away from zero, that $w_t$ is uniformly negative near $\partial U$,
and that
\[
 D=\{x\in U:w_0(x)>0\}\ne\varnothing,\qquad
 \Sigma=\{x\in U:w_0(x)=0\},\qquad
                  |\nabla w_0|>0\quad\hbox{on }\Sigma.
\]
Thus $\overline D\subset U$ and $\Sigma=\partial D$ is a compact
regular hypersurface. Decrease $t_0$ when necessary. Suppose
\begin{equation}\label{eq:v17-linear-density}
       f_t(x)=a_t(x)(w_t(x))_+,
       \qquad \int_U f_t(x)\,\dd x=1.
\end{equation}
Normalizing an initially unnormalized density is allowed: its positive
normalizing integral is $C^3$ by a common tubular chart for the moving
boundary. Put
\begin{equation}\label{eq:v17-boundary-form}
 v(x)=\left.\partial_t w_t(x)\right|_{t=0},\qquad
 \mathcal I_{\Sigma}=
       \int_\Sigma\frac{a_0(x)v(x)^2}{|\nabla w_0(x)|}\,
                                      \mathrm d\sigma(x).
\end{equation}
Here $\mathrm d\sigma$ is hypersurface measure. We use the convention
\[
 H^2(P,Q)=\int(\sqrt{\mathrm dP/\mathrm d\nu}
                   -\sqrt{\mathrm dQ/\mathrm d\nu})^2\,\mathrm d\nu,
 \qquad \|P-Q\|_{\rm TV}=\sup_B|P(B)-Q(B)|,
\]
where $\nu$ dominates both measures. In particular $0\le H^2\le2$.
The distribution function of a standard normal variable is denoted by
$\Phi$.

\begin{theorem}[Logarithmic boundary information]
\label{thm:v17-boundary-information}
Under \eqref{eq:v17-linear-density},
\begin{equation}\label{eq:v17-hellinger}
 H^2(f_t\,\dd x,f_0\,\dd x)
       =\frac{\mathcal I_\Sigma}{4}\,t^2\log\frac1{|t|}
                                                    +O(t^2)
                      \qquad(t\longrightarrow0).
\end{equation}
Assume $\mathcal I_\Sigma>0$. Let $t_n\to0$, $t_n\ne0$, let
$0<p_n\le1$, and consider $n$ independent observations from the pair
\begin{equation}\label{eq:v17-thinning}
 P_{n,0}=(1-p_n)\delta_\dagger+p_n f_0\,\dd x,
 \qquad
 P_{n,1}=(1-p_n)\delta_\dagger+p_n f_{t_n}\,\dd x.
\end{equation}
The failure symbol $\dagger$ is distinct from $U$ and is retained.
If
\begin{equation}\label{eq:v17-information-scale}
        n p_n t_n^2\log(1/|t_n|)\longrightarrow b\in[0,\infty],
\end{equation}
then
\begin{equation}\label{eq:v17-general-testing}
 \|P_{n,1}^{\otimes n}-P_{n,0}^{\otimes n}\|_{\rm TV}
        \longrightarrow
            2\Phi\bigl(\sqrt{\mathcal I_\Sigma b}/2\bigr)-1.
\end{equation}
At $b=\infty$ the right side is one. At $0<b<\infty$, the logarithm
of the likelihood of the absolutely continuous part of
$P_{n,1}^{\otimes n}$ relative to $P_{n,0}^{\otimes n}$ converges under
the latter law to
\begin{equation}\label{eq:v17-general-likelihood}
             \mathcal N(-\mathcal I_\Sigma b/2,\mathcal I_\Sigma b).
\end{equation}
The probability of a support-exclusive observation tends to zero under
either hypothesis at this scale. The minimum equal-prior testing error
therefore tends to $\Phi(-\sqrt{\mathcal I_\Sigma b}/2)$.
\end{theorem}

The limiting normal law is compatible with a moving support because the
exclusive boundary layer is smaller than the layer generating the
logarithmic information. Classical nonregular likelihood theory already
contains the change of rate when a density vanishes linearly at a moving
endpoint; see Smith~\cite{Smith1985}. The theorem records the relevant
hypersurface coefficient and the independent rare-event charge in the
form needed here. We prove it directly, rather than invoking a
finite-Fisher-information theorem whose hypotheses fail in this setting.

\subsection{Coarea estimates}
For $x\in D$ define the ordinary pointwise score
\[
 S(x)=\frac{\dot a_0(x)}{a_0(x)}+\frac{v(x)}{w_0(x)}.
\]
It is integrable under $f_0\,\dd x$, but has infinite second moment
when $\mathcal I_\Sigma>0$. For $h>0$ put
$D_h=\{w_0\ge h\}$ and $S_h=S\one_{D_h}$.

\begin{lemma}[Truncated score moments]\label{lem:v17-coarea}
As $h\downarrow0$,
\begin{align}
 \Pr_0\{0<w_0<h\}&=O(h^2),\qquad
                         \mathbb E_0 S_h=O(h),
                                           \label{eq:v17-strip}\\
 \mathbb E_0 S_h^2&=\mathcal I_\Sigma\log(1/h)+O(1),
                                           \label{eq:v17-score-two}\\
 \mathbb E_0|S_h|^3&=O(h^{-1}),\qquad
 \mathbb E_0|S_h|^4=O(h^{-2}).              \label{eq:v17-score-higher}
\end{align}
For $|t|$ sufficiently small, the mass of the support of $f_t$ outside
$D$, and the mass under $f_0$ where $f_t=0$, are both $O(t^2)$.
All estimates are uniform on compact families with the stated strict
positivity and regularity bounds.
\end{lemma}
\begin{proof}
Choose $h_0>0$ so that $|\nabla w_0|$ is bounded away from zero on
$|w_0|\le2h_0$. The flow of
$\nabla w_0/|\nabla w_0|^2$ identifies the level sets $w_0=z$ for
$|z|\le h_0$ with $\Sigma$, with uniformly bounded $C^2$ Jacobians.
The coarea formula consequently writes integration in this collar as
integration first over $w_0=z$ and then over $z$. In particular,
\[
 J(z)=\int_{\{w_0=z\}}
              \frac{a_0v^2}{|\nabla w_0|}\,\mathrm d\sigma
           =\mathcal I_\Sigma+O(|z|).
\]
The collar contribution of $v^2/w_0^2$ to the second score moment is
$\int_h^{h_0}J(z)\,\mathrm dz/z$. The remaining terms in $S^2$
are integrable, and the part outside the collar is bounded. This proves
\eqref{eq:v17-score-two}. Similarly $f_0=O(w_0)$ in the collar gives
\[
 \int_{\{0<w_0<h\}} f_0\,\dd x=O(h^2),\qquad
 \int_{D_h}|S|^r f_0\,\dd x
       \le C\left(1+\int_h^{h_0}z^{1-r}\,\mathrm dz\right)
                                                     \quad(r=3,4).
\]

Differentiating $\int f_t=1$ once at zero gives
$\int_D(\dot a_0w_0+a_0v)\,\dd x=0$. This differentiation is
justified by dominated convergence: the positive-part map is Lipschitz,
the coefficients have bounded derivatives, and the zero set of $w_0$
has Lebesgue measure zero. Thus $\mathbb E_0 S=0$. The omitted collar
integral of $Sf_0=\dot a_0w_0+a_0v$ is $O(h)$, proving the second
estimate in \eqref{eq:v17-strip}.

Uniform differentiability gives $|w_t-w_0|\le C|t|$. The two
support-exclusive regions therefore lie in $|w_0|\le C|t|$.
There, both relevant densities are bounded by $C|t|$, and the
collar has volume $O(|t|)$. Their masses are $O(t^2)$. The same
bounds in finitely many uniformly controlled level-set charts prove
the uniform assertion.
\end{proof}

\subsection{Proof of the information theorem}
\begin{proof}[Proof of Theorem~\ref{thm:v17-boundary-information}]
Write $t\ne0$ and $\ell=\log(1/|t|)$. On $D_h$, whenever
$h\ge C_0|t|$ for a sufficiently large fixed $C_0$, the two densities
are positive and Taylor's formula gives
\begin{equation}\label{eq:v17-density-ratio}
 \frac{f_t}{f_0}=1+tS+t^2R_t,
             \qquad |R_t(x)|\le C(1+w_0(x)^{-1}).
\end{equation}
The bound follows by expanding $a_t w_t$ before dividing by $a_0w_0$;
it does not differentiate the support indicator.

First set $h=C_0|t|$. The integral of $f_0+f_t$ outside their common
part $D_h$ is $O(t^2)$, by the same collar argument as in the lemma.
On $D_h$ set $z=tS+t^2R_t$. Taking $C_0$ sufficiently large and then
$t$ small gives $|z|\le1/2$. The identity
\[
             (\sqrt{1+z}-1)^2=z^2/4+O(|z|^3)
\]
and the coarea bounds imply
\[
 \int_{D_h}z^2 f_0\,\dd x
       =t^2\mathbb E_0S_h^2+O(|t|^3\log(1/|t|)),\qquad
 \int_{D_h}|z|^3f_0\,\dd x=O(t^2).
\]
For example, the last bound follows from
$|t|^3\mathbb E_0|S_h|^3=O(|t|^3/h)$ and the corresponding, smaller,
remainder terms in \eqref{eq:v17-density-ratio}. Substitution of
\eqref{eq:v17-score-two} proves \eqref{eq:v17-hellinger}.

We next prove the likelihood assertion when $0<b<\infty$. In the
$n$th row put
\[
 t=t_n,\quad p=p_n,\quad \ell=\log(1/|t|),\quad
               L=\ell^{1/4},\quad h=|t|L.
\]
Represent each null observation by a Bernoulli success indicator $Y_i$
with mean $p$ and, on success, an independent $X_i$ of density $f_0$.
With null probability tending to one, all successful $X_i$ lie in $D_h$:
indeed the exceptional probability is at most
\[
                   Cnp h^2=C(np t^2\ell)L^2/\ell=o(1).
\]
The same statement holds under the alternative, because its density
has mass $O(h^2)$ in the corresponding enlarged collar. In particular
support-exclusive observations have probability $O(np t^2)=o(1)$
under both hypotheses.

On this high-probability event, expand
$\log(f_t/f_0)$ using \eqref{eq:v17-density-ratio}. The log likelihood
is
\begin{equation}\label{eq:v17-log-expansion}
 t\sum_{i=1}^nY_iS_h(X_i)
       -\frac{t^2}{2}\sum_{i=1}^nY_iS_h(X_i)^2+o_{P_{n,0}^{\otimes n}}(1).
\end{equation}
For completeness, the sum of the absolute expectations of the
$t^2R_t$ terms is $O(np t^2)=o(1)$, since
$\int_D(1+w_0^{-1})f_0<\infty$. The cubic logarithmic remainder is
bounded in expectation by
$Cnp|t|^3/h=O((np t^2\ell)/(L\ell))=o(1)$.
Replacing $z^2$ by $t^2S_h^2$ creates an error
$O(np|t|^3\log(1/h)+np t^4\log(1/h))=o(1)$.
These estimates justify the remainder in \eqref{eq:v17-log-expansion}
without an untruncated third score moment.

The mean of the first sum in \eqref{eq:v17-log-expansion} is
$O(np|t|h)=O((np t^2\ell)L/\ell)=o(1)$. Its variance converges to
$\mathcal I_\Sigma b$ by \eqref{eq:v17-score-two}, since
$\log(1/h)/\ell\to1$. The contribution of the squared mean is at
most $Cnp^2t^2h^2=o(1)$. Each summand has absolute value at most
$C/L$, apart from its vanishing centering term. The Lindeberg condition
is therefore automatic, and the triangular-array central limit theorem
gives convergence of this sum to $\mathcal N(0,\mathcal I_\Sigma b)$.
The mean of the second, unsigned sum converges to
$\mathcal I_\Sigma b$, and its variance is at most
\[
       Cnp t^4\mathbb E_0|S_h|^4
             \le Cnp t^4h^{-2}=C(np t^2\ell)/(L^2\ell)=o(1).
\]
This proves \eqref{eq:v17-general-likelihood}.

Let $Z_n$ be this log likelihood, with $Z_n=-\infty$ on the
null-exclusive set. Even when the alternative has a singular part,
\[
 1-\|P_{n,1}^{\otimes n}-P_{n,0}^{\otimes n}\|_{\rm TV}
       =\mathbb E_{P_{n,0}^{\otimes n}}\min\{1,e^{Z_n}\}.
\]
The function on the right is bounded and continuous on the extended
real line. The limit just proved gives, for
$v=\mathcal I_\Sigma b$,
\[
 \mathbb E\min\{1,e^{\sqrt v G-v/2}\}=2\Phi(-\sqrt v/2),
                        \qquad G\sim\mathcal N(0,1).
\]
The equality follows by splitting at $G=\sqrt v/2$ and completing
the square in the other Gaussian integral. This proves
\eqref{eq:v17-general-testing} for finite positive $b$; no uniform
integrability of an unbounded likelihood is being assumed.

Finally the one-observation affinity of the thinned pair is exactly
$1-pH^2(f_t,f_0)/2$. Affinities multiply under independent products,
and for any probabilities $P,Q$ their affinity $\mathcal A$ satisfies
\[
             1-\mathcal A\le\|P-Q\|_{\rm TV}
                              \le\sqrt{1-\mathcal A^2}.
\]
The first inequality follows from $\min(f,g)\le\sqrt{fg}$; the
second is Cauchy--Schwarz applied to
$|f-g|=|\sqrt f-\sqrt g|(\sqrt f+\sqrt g)$.
Equation~\eqref{eq:v17-hellinger} now proves the cases $b=0$ and
$b=\infty$. The equal-prior risk is $(1-\|P-Q\|_{\rm TV})/2$.
\end{proof}

\begin{remark}[The boundary coefficient is intrinsic]
\label{rem:v17-invariance}
Replacing the defining function by $\widetilde w_t=c_t w_t$, where
$c_t>0$ is smooth, and replacing $a_t$ by $a_t/c_t$ leaves the density
unchanged. On $\Sigma$ one has
$\dot{\widetilde w}_0=c_0v$ and
$|\nabla\widetilde w_0|=c_0|\nabla w_0|$, so
\eqref{eq:v17-boundary-form} is unchanged as well. Equivalently, if
$V_\nu=-v/|\nabla w_0|$ is the normal velocity of the support boundary,
then $\mathcal I_\Sigma=\int_\Sigma a_0|\nabla w_0|V_\nu^2\,
\mathrm d\sigma$. If this integral vanishes, the theorem supplies only
$H^2=O(t^2)$; a nontrivial logarithmic critical scale is not asserted.
The usual smooth interior Fisher information is a lower-order term
when $\mathcal I_\Sigma>0$.
\end{remark}
