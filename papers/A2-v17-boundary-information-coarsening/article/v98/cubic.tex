\section{A rank-deficient cubic stratum}\label{sec:cubic}
Let $\mathfrak F$ denote the degree-three centres with
\[
 f=z(z-r)^2,\qquad g=(z-s)^3,\qquad 0<s<r<D,
\]
strict weights $\pi<\alpha<1-\pi$, strictly positive $U,V$, $\det V\ne0$,
and at least seven clocks. The first channel may have either rank. Put
$\gamma=1-\alpha$ and
\begin{equation}\label{eq:ABE}
 A=(r+3s)(2r-3s)^2,\quad B=(3D-r)s-2rD,\quad E=\gamma A-4\pi r^3.
\end{equation}
Here $A$ is a scalar, not the target polynomial.

\begin{theorem}[The entire cubic fibre]\label{thm:cubic-fibre}
The exact spectral fibre is a singleton precisely on
\begin{equation}\label{eq:I}
 \mathfrak I=\{\det U\ne0\}\cup\{B<0\}\cup\{E<0\}.
\end{equation}
On its complement, besides $(f,g)$ all component-polynomial pairs are,
up to permutation,
\begin{equation}\label{eq:remote-interval}
 (f,h_c),\quad h_c=(1-c)f+cg,\quad
 c_*\le c\le\min(c_D,\gamma/\pi),
\end{equation}
where
\[
 c_*=\frac{4r^3}{A},\quad b=\frac{rs}{3s-2r},\quad
 R(z)=\frac{f(z)}{g(z)},\quad c_D=\frac{R(D)}{R(D)-1}.
\]
The interval is nonempty even at equality in either constraint, and
every one of its pairs has a positive stochastic realization. Conditional
on the component pair, that realization is unique.
\end{theorem}
\begin{proof}
Lemma~\ref{lem:normalizer} fixes $q,K$ for every competitor. If $U$ is
invertible, Lemma~\ref{lem:fullrank} applies; in particular
$\det K/\det K_3=fg$ gives spectral identification.

If $\det U=0$, stochasticity gives $u_1=u_2=u$. The second marginal
$V\diag(\alpha,\gamma)(f,g)^{\mathsf T}$ has coefficient rank two.
Every competitor therefore has invertible second channel and its two
monic polynomials lie in the affine pencil $h_c$. Expansion gives
\begin{equation}\label{eq:disc}
 \disc(h_c)=c(c-1)^2(r-s)^3(Ac-4r^3),\qquad h_c(0)=-cs^3,
\end{equation}
and
\[
 \frac{R'(z)}{R(z)}=\frac{(2r-3s)z+rs}{z(z-r)(z-s)},\qquad
 A=4r^3-27s^2(r-s)<4r^3.
\]
Nonnegative roots exclude $c<0$, and the discriminant excludes $0<c<1$.
For $c>1$ there is a root in $(s,r)$ and no root in $[0,s]$. The other
two admissible roots must be above $r$, solving $R(z)=c/(c-1)>1$.
They exist only for $s>2r/3$. In that case $R$ increases to its maximum
at $b$ and then decreases to one. The double-root value is $c_*>1$;
the larger root increases from $b$ to $D$ as $c$ increases from $c_*$
to $c_D$. Thus the whole real-rooted pencil in $[0,D]$ is $\{0,1\}$,
with $[c_*,c_D]$ added exactly when $b\le D$.

For a competing pair with pencil coordinates $c_1,c_2$, the weights obey
$\alpha'_1c_1+\alpha'_2c_2=\gamma\in(0,1)$. One coordinate must be
zero and the other is one or a value $c\ge c_*$. The latter weight is
$\gamma/c$, so the floor imposes $c\le\gamma/\pi$; the other weight
is automatically at least $\alpha>\pi$. Conversely put
\[
 \beta_f=\alpha v_1+\gamma(1-1/c)v_2,\qquad
 \beta_h=\gamma v_2/c.
\]
Both vectors are positive and
$\beta_ff+\beta_hh_c=\alpha v_1f+\gamma v_2g$.
Their entry sums are the stated weights; normalizing them and keeping
first channel $(u,u)$ gives the realization. Coefficient functionals
dual to the independent polynomials recover the two rank-one coefficient
matrices uniquely, hence their weights and normalized marginals.
Finally $b\le D$ is equivalent to $B\ge0$ and, on this set,
$c_*\le\gamma/\pi$ is equivalent to $E\ge0$.
\end{proof}

\subsection{Quantitative inversion across the rank locus}
All coefficient norms below are Euclidean, including the leading
coefficient when a monic polynomial is represented as a vector. Constants
are specified from a fixed compact $K_0\subset\mathfrak F$. Norm
comparisons between these finite-dimensional spaces are included in
these constants. Put
\[
 \delta=\max_j\|P'_j-P_{\theta,j}\|_F.
\]
The coefficient matrix of the second marginal is denoted by
$\mathsf B_\theta$; its rows are the coefficients of that two-vector
polynomial. Set
\[
 \beta_0=\min_{K_0}\sigma_2(\mathsf B_\theta)>0,\quad
 d_{fg}=\min_{K_0}\|g-f\|_{\coef}>0.
\]
Normalizer recovery supplies $C_B$ with
$\|\mathsf B'-\mathsf B_\theta\|\le C_B\delta$.
Let $K_d$ bound $\|\diag(\alpha')F'_{\coef}\|$ on all closed-model
competitors for the designs in $K_0$, and set
\begin{equation}\label{eq:affine-constants}
 W_0=\frac{2K_d}{\pi\beta_0},\qquad
 C_f=\max_{K_0}\|f\|_{\coef},\qquad
 K_{\rm aff}=W_0C_B(1+C_f).
\end{equation}

\begin{proposition}[From the second marginal to the pencil]\label{prop:affine}
If $\delta\le\beta_0/(2C_B)$, every closed competitor has invertible
$V'$ and its component polynomials satisfy
\begin{equation}\label{eq:affine-approx}
 f'_b=h_{c_b}+e_b,\qquad \deg e_b<3,\qquad
 \|e_b\|_{\coef}\le K_{\rm aff}\delta.
\end{equation}
Moreover $\|\diag(\alpha')^{-1}V'^{-1}\|\le W_0$. If the ordered
component pair is within $\rho$ of $(f,g)$, then
\begin{equation}\label{eq:pencil-radius}
 |c_1|,\ |c_2-1|\le(\rho+K_{\rm aff}\delta)/d_{fg}.
\end{equation}
No lower bound on $|\det U|$ is used.
\end{proposition}
\begin{proof}
Since $\mathsf B'=V'\diag(\alpha')F'_{\coef}$,
\[
 \beta_0/2\le\sigma_2(\mathsf B')
        \le\sigma_{\min}(V')K_d.
\]
This proves the inverse bound, using $\alpha'_a\ge\pi$.
Write $C_\theta$ for the matrix with coefficient rows $f,g$ and put
\[
 T=\diag(\alpha')^{-1}V'^{-1}V\diag(\alpha,\gamma),\qquad
 E_0=\diag(\alpha')^{-1}V'^{-1}(\mathsf B'-\mathsf B_\theta).
\]
Then $F'_{\coef}=TC_\theta+E_0$ and $\|E_0\|\le W_0C_B\delta$.
If a row of $T$ is $(l,c)$, monicity says
$l+c+(E_0)_{\rm lead}=1$. Subtracting $(E_0)_{\rm lead}f$ from
that error row gives the exact decomposition
$f'_b=(1-c)f+cg+e_b$, with the bound in
\eqref{eq:affine-constants} and zero leading coefficient in $e_b$.
Finally $h_c-f=c(g-f)$ and $h_c-g=(c-1)(g-f)$ imply
\eqref{eq:pencil-radius} by the triangle inequality.
\end{proof}

\begin{lemma}[Two robust isolation estimates]\label{lem:pencil-isolation}
There are explicit positive $r_f,r_g,\varepsilon_0,C_0,C_1$, depending
only on $K_0$, with the following property. If $h_c+e$ is monic with
all roots in $[0,D]$, $\deg e<3$ and $\|e\|_{\coef}\le\varepsilon$,
then
\[
 |c|\le r_f,\ \varepsilon\le\varepsilon_0
       \ \Longrightarrow\ |c|\le C_0\varepsilon,
 \qquad
 |1-c|\le r_g,\ \varepsilon\le\varepsilon_0
       \ \Longrightarrow\ |1-c|\le C_1\varepsilon.
\]
The constants require only root-separation and coefficient bounds on
$K_0$, not a sign or a nonzero margin for $B,E$.
\end{lemma}
\begin{proof}
Let $s_0=\min_{K_0}s>0$ and
$d_0=\min_{K_0}4r^3(r-s)^3>0$. Choose $L_D$ as a Lipschitz bound for
the discriminant on a coefficient neighbourhood of
$\{h_c:\theta\in K_0,|c|\le1\}$, and $R_D$ such that
\[
 |\disc(h_c)+4r^3(r-s)^3c|\le R_Dc^2\quad(|c|\le1).
\]
These constants are finite because the discriminant is polynomial.
Take $r_f\le\min(1,d_0/(2R_D))$, omitting the second bound if $R_D=0$.
The endpoint condition gives $-cs^3+e(0)\le0$, hence
$c\ge-\varepsilon/s_0^3$. For $0\le c\le r_f$, nonnegative
discriminant gives
$0\le-d_0c+R_Dc^2+L_D\varepsilon$, hence
$c\le2L_D\varepsilon/d_0$. One may take
$C_0=\max(s_0^{-3},2L_D/d_0)$.

Near $c=1$, put $v=1-c$ and translate $z=s+x$. There are exact identities
\[
 h_c(s+x)=x^3+vA_2x^2+vA_1x+vA_0,\quad
 A_2=3s-2r,\ A_1=f'(s),\ A_0=f(s)=s(r-s)^2.
\]
Let $L_s$ bound coefficient translation of $e$ on $K_0$, set
$E_s=L_s\varepsilon$, and write the perturbed coefficients as
$u_i=vA_i+e_i$, $|e_i|\le E_s$. Choose
$M\ge\max_{K_0}(|A_1|,|A_2|)$ and $a_*=\min_{K_0}A_0>0$.
Depressing the perturbed cubic by its own quadratic coefficient gives
\[
 p=u_1-u_2^2/3,\qquad q=u_0-u_1u_2/3+2u_2^3/27.
\]
For $u=|v|+E_s\le1$,
\[
 |p|\le C_pu,\quad |q-A_0v|\le E_s+C_qu^2,
 \quad C_p=M+1+(M+1)^2/3,\quad
 C_q=(M+1)^2/3+2(M+1)^3/27.
\]
Real-rootedness is exactly $p\le0$ and
$|q|\le c_{\rm cub}(-p)^{3/2}$ with $c_{\rm cub}=2/(3\sqrt3)$. Thus
\begin{equation}\label{eq:depressed-absorption}
 a_*|v|\le E_s+C_qu^2+c_{\rm cub}C_p^{3/2}u^{3/2}.
\end{equation}
Choose
\[
 \eta=\min\{1,\ a_*/(4C_q),\ (a_*/(4c_{\rm cub}C_p^{3/2}))^2\},
 \quad r_g=\eta/2,\quad \varepsilon_0\le\eta/(2L_s).
\]
Then $u\le\eta$, the last two terms in
\eqref{eq:depressed-absorption} are at most $a_*u/2$, and
$|v|\le(2/a_*+1)E_s$. Take $C_1=(2/a_*+1)L_s$.
Decrease $\varepsilon_0$ also to stay in the discriminant neighbourhood
used in the first argument. Every threshold is determined by the
stated positive lower margins and finite polynomial upper bounds.
\end{proof}

Let $d_{\rm par}$ be Euclidean distance in monic coefficients and
stochastic entries, minimized over the component permutation.

\begin{lemma}[A local inverse through the walls]\label{lem:local-cubic}
For every compact $K_0\subset\mathfrak F$, there exist $\rho,M,t_0>0$
such that a closed-model competitor whose component pair is within
$\rho$ of $(f,g)$, up to permutation, satisfies
\begin{equation}\label{eq:local-inverse}
 d_{\rm par}(\xi,\theta)\le M\max_j\|F_T(\xi)_j-P_{\theta,j}\|_F
\end{equation}
whenever the observation error is below $t_0$. These constants require
no lower bound on $|\det U|$, $-B$, or $-E$.
For a compact $H\subset\mathfrak I$ and small fixed $\rho$, the remote
separation
\begin{equation}\label{eq:separation}
 \eta_H(\rho)=\min_{\substack{\theta\in H,\ \xi\text{ in the closed model}\\
                         d_{\rm par}(\xi,\theta)\ge\rho}}
                   h_w(F_T(\xi),P_\theta)
\end{equation}
is positive if the minimizing set is nonempty. The whole-model inverse
therefore holds uniformly on $H$.
\end{lemma}
\begin{proof}
Let $r_* =\min(r_f,r_g)$ from Lemma~\ref{lem:pencil-isolation} and choose
\[
 \rho\le d_{fg}r_*/4,\qquad
 t_0\le\min\{\beta_0/(2C_B),\rho/K_{\rm aff},
                       \varepsilon_0/K_{\rm aff}\}.
\]
Proposition~\ref{prop:affine} gives an error at most
$\varepsilon=K_{\rm aff}\delta$ in the affine pencil and pencil
coordinates in the required neighbourhoods. The two isolation estimates
then give $|c_1|+|c_2-1|\le(C_0+C_1)K_{\rm aff}\delta$.
Together with \eqref{eq:affine-approx}, this controls both polynomial
coefficient vectors by a common multiple of $\delta$.

To recover the stochastic parameters without dividing by $\det U$,
choose coefficient functionals $\ell_1,\ell_2$ dual to $f,g$. Their
norms are bounded by the inverse Gram matrix of this pair, uniformly
on $K_0$. The matrix $T'_{ij}=\ell_i(f'_j)$ is $I+O_{K_0}(\delta)$.
Decrease $t_0$ so that $\|T'-I\|\le1/2$; then $\|(T')^{-1}\|\le2$.
Apply the two functionals entrywise to $K'$. Inverting $T'$ recovers
$M'_a=\alpha'_au'_av_a'{}^{\mathsf T}$ with $O_{K_0}(\delta)$ error
from $M_a$. Entry sums give the weights, and row and column sums divided
by those weights give the two channels. These divisions have lower
bound $\pi$. This proves \eqref{eq:local-inverse} with constants
constructed from normalizer, second-marginal, Gram, root and weight
margins only. In particular no full-rank inverse has been continued
through its singular determinant.

For the last assertion the feasible set in \eqref{eq:separation} is
compact, allowing the centre's design to vary in $H$. A zero minimum
would give an exact competitor separated from that centre. The entire
fibre and conditional uniqueness in Theorem~\ref{thm:cubic-fibre}
exclude this on $H$. Small whole-model observation balls thus lie in
the local neighbourhood. It is this exclusion radius, rather than a
local coefficient constant, that may collapse at a remote wall.
\end{proof}

\subsection{The complete local Fisher fibre}
Use $a,X,Y\ge0$, free $b,c$, and five free stochastic coordinates $\eta$.
The component jets are
\begin{equation}\label{eq:cubic-jets}
 p_f=-a(z-r)^2-bz(z-r)-Xz,\qquad
 p_g=-c(z-s)^2-Y(z-s).
\end{equation}
Let $L$ be the seven free score columns and $D_0$ the three columns
indexed by $0,x,y$ for $(a,X,Y)$. Define $Q$ by \eqref{eq:mult-Q} and
\begin{equation}\label{eq:cubic-kappa}
 J(X,Y)=\min_{a\ge0}(a,X,Y)Q(a,X,Y)^{\mathsf T},\quad
 \kappa_x=\min_{Y\ge0}J(1,Y),\quad
 \kappa_y=\min_{X\ge0}J(X,1).
\end{equation}

\begin{proposition}[Coercivity and finite supports]\label{prop:cubic-Q}
Throughout $\mathfrak F$, $L$ has rank seven, $Q_{00}>0$, and
$\kappa_x,\kappa_y$ are positive and locally Lipschitz. For every compact
$K_0\subset\mathfrak F$,
\[
 \|S_\theta(a,b,c,\eta,X,Y)\|_{I,\theta}
       \ge c_{K_0}\|(a,b,c,\eta,X,Y)\|\quad(a,X,Y\ge0).
\]
Each $\kappa_i$ is given by four finite support tests. For
$A\subseteq\{0,x,y\}\setminus\{i\}$ with $Q_{AA}$ positive definite, put
\[
 v_A=-Q_{AA}^{-1}Q_{Ai},\qquad k_{i,A}=Q_{ii}-Q_{iA}Q_{AA}^{-1}Q_{Ai}.
\]
The test is $v_A\ge0$ and $Q_{ji}+Q_{jA}v_A\ge0$ for inactive $j$.
The empty support has value $Q_{ii}$. At least one test succeeds and
all successful tests give $\kappa_i$.
\end{proposition}
\begin{proof}
Every feasible jet is realized by roots
\[
 ta,\qquad r+tb/2\pm\sqrt{tX},\qquad
 s+tc/3+\sqrt t\,y_i,\quad \sum_i y_i=0,\quad\tfrac12\sum_i y_i^2=Y,
\]
and order-$t$ stochastic increments. Their observation increment is
$tS+O(t^{3/2})$. A zero score, combined with
Lemma~\ref{lem:local-cubic}, forces the component jets and $\eta$ to
vanish. Coefficient comparison in \eqref{eq:cubic-jets} then forces
every variable to vanish. Compactness of the feasible unit sphere gives
coercivity, uniformly on $K_0$. This also proves independence of the free
columns, $Q_{00}>0$, and existence, positivity and boundedness of the
constrained minimizers.

Choose a minimizer with minimal positive support in the projected score
columns. Those supported columns are independent: a nonzero dependence
would allow a signed change of positive coefficients, preserving the
represented vector, until one coefficient vanished. On an independent
support, stationarity and the inactive inequalities are exactly the
stated tests. Conversely their KKT conditions prove optimality by
convexity. This handles a singular full $Q$ without inverting it.
Uniformly bounded minimizers and smooth score dependence give local
Lipschitz dependence by evaluating each problem at a minimizer of the
other problem.
\end{proof}

\begin{theorem}[Local normal form and whole-model restriction]\label{thm:cubic-local}
At every centre of $\mathfrak F$ the local leading root set is
\begin{equation}\label{eq:cubic-leading}
 \bigl(\{0\},\{-\sqrt X,\sqrt X\},Z(z^3-Yz-Z)\bigr),\qquad
 X,Y\ge0,\quad J(X,Y)\le4,\quad27Z^2\le4Y^3.
\end{equation}
On compact subsets of $\mathfrak F$, scaled local observation balls
converge to this set with Hausdorff error $O(\sqrt t)$. Hence
\begin{equation}\label{eq:cubic-C}
 \omega_{\theta,\loc}(t)=C_{\loc}(\theta)\sqrt t+O(t),\qquad
 C_{\loc}=\max\{\sqrt2\,\kappa_x^{-1/4},\ 2\sqrt{2/3}\,\kappa_y^{-1/4}\}.
\end{equation}
The error is compact-uniform, and $C_{\loc}$ is positive and locally
Lipschitz on all of $\mathfrak F$. On compact subsets of $\mathfrak I$
these statements hold for the whole-model modulus. The support tests,
\eqref{eq:I}, and $9\kappa_y-16\kappa_x$ give a finite semialgebraic
partition. On its constant choices,
\[
 C_{\loc}^4=\max\{4/\kappa_x,64/(9\kappa_y)\}
\]
is one rational function, and $C_{\loc}$ is Nash after refinement.
\end{theorem}
\begin{proof}
The local inverse bounds coefficients and stochastic increments by
$O(t)$ for every local competitor. Its first component has exact roots
$ta,r+tb/2\pm\sqrt{tX}$ with bounded $a,b,X$ and $a,X\ge0$.
Its second component has exact roots $s+tc/3+\sqrt t\,y_i$ with
$\sum_i y_i=0$ and bounded squared sum. Put
$Y=\tfrac12\sum_i y_i^2$ and $Z=\prod_i y_i$. The real-cubic condition
is exactly $Y\ge0,27Z^2\le4Y^3$. Expansion gives
$F=P+tS+O(t^{3/2})$ and $h_w=t\|S\|_I/2+O(t^{3/2})$ uniformly on
these bounded coordinates. Radial contraction of all first-order
variables, together with $Z\mapsto\rho^{3/2}Z$, proves the outer
Hausdorff inclusion. Conversely choose bounded minimizing free and
endpoint variables and realize the displayed root arcs with
$\rho=1-M\sqrt t$. They lie in the observation ball and give the inner
inclusion. The inequality boundary and all joint cubic coefficient
relations are therefore realized.

Homogeneity gives $X_{\max}=2/\sqrt{\kappa_x}$ and
$Y_{\max}=2/\sqrt{\kappa_y}$. The double-cluster diameter is
$\sqrt{X_{\max}}$. Lemma~\ref{lem:ordered} with $m=3$ gives the triple
diameter $2\sqrt{Y_{\max}/3}$, attained also by
$(-2h,h,h)$ and $(-h,-h,2h)$ with $h=\sqrt{Y_{\max}/3}$.
The maximum of marginal diameters gives \eqref{eq:cubic-C}.
Proposition~\ref{prop:cubic-Q}, cluster separation and the positive
remote exclusion in Lemma~\ref{lem:local-cubic} give the remaining
claims. The local formula remains valid at $B=0$ or $E=0$, but its
identification with the whole-model modulus does not.
\end{proof}

\begin{example}[The earlier rational phase values]
For $D=1$, $\pi=1/10$, $r=3/5$, $s=1/5$, $\alpha=2/5$,
$u_1=u_2=(1/3,2/3)^{\mathsf T}$,
$v_1=(1/4,3/4)^{\mathsf T}$, $v_2=(3/4,1/4)^{\mathsf T}$,
clocks $(6/5,7/5,8/5,9/5,2,5/2,3)$ and equal weights,
$B=-18/25$. The empty supports are admissible, the double cluster
dominates, and exact rational evaluation gives
\[
 C^4=\frac{78550423766145732273576408862601811503774391446}
            {385279214348552940073664077156875},\qquad
 3778.707<C<3778.708.
\]
Changing only $u_2$ to $(2/5,3/5)^{\mathsf T}$ gives the triple-dominant
phase, $59.970<C<59.971$. The exact arithmetic records for both values
are retained in the archival companion. These large values are not
evidence for an unspecified wall blow-up.
\end{example}

\begin{corollary}[Centre-known local risk]\label{cor:risk}
Let $H\subset\mathfrak I$ be compact, fix $0<c\le1/(8\sqrt2)$, and
observe independent categorical samples at the design clocks with
$n_j/N\to w_j$ uniformly on $H$ (integer rounding suffices).
For squared spectral loss on the specified neighbourhood
$h_w(F_T(\xi),P_\theta)\le cN^{-1/2}$, with the centre known, the
minimax risk $\mathcal R_N(\theta,c)$ satisfies uniformly on $H$
\[
 \frac{3c}{32}C_{\loc}(\theta)^2-O_H(N^{-1/4})
 \le\sqrt N\,\mathcal R_N(\theta,c)
 \le cC_{\loc}(\theta)^2+O_H(N^{-1/4}).
\]
\end{corollary}
\begin{proof}
A constant decision in the specified spectral ball has loss at most
its squared diameter. A diameter pair in that ball has weighted
Hellinger distance at most $2cN^{-1/2}$. Tensorization bounds its product
Hellinger distance by $2c\sqrt{\max_j n_j/(Nw_j)}$. Uniform convergence
and the positive minimum of the design weights on $H$ make the maximum
ratio at most $2$ for all sufficiently large $N$, uniformly on $H$.
Thus the product distance is at most $2c\sqrt2\le1/4$. Total variation
is bounded by this Hellinger distance in the present convention. The
two-point testing bound for squared metric loss is $d^2(1-\TV)/8$,
hence at least $3d^2/32$. Substitute \eqref{eq:cubic-C}. The centre and
its neighbourhood are known; unknown-centre adaptation and honest
confidence sets across a wall are not asserted.
\end{proof}
