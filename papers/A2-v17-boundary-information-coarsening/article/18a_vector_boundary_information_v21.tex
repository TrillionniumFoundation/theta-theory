\section{Vector boundary information in a non-dominated local family}
\label{sec:v21-vector-information}

The scalar theorem of Section~\ref{sec:v17-boundary-information} already
separates the likelihood of the absolutely continuous part from the small
support-exclusive mass.  We now give the corresponding vector theorem in a
form which keeps that distinction at every finite sample size.  In
particular, we never write a Radon--Nikodym derivative of a moving-support
alternative with respect to the null when such a derivative does not exist.
The original non-dominated experiment will instead be shown asymptotically
equivalent to an explicitly dominated common-collar experiment.  The latter
is uniformly LAN, and therefore the original experiment has the same
Gaussian limit experiment, testing profile and local minimax bound.

Let $\Theta\subset\mathbb R^r$ be a neighborhood of the origin.  On a fixed
bounded open set $U\subset\mathbb R^m$ suppose
\begin{equation}\label{eq:v21-vector-density}
 f_\theta(x)=a_\theta(x)(w_\theta(x))_+,
 \qquad \int_U f_\theta(x)\,dx=1.
\end{equation}
Assume that $a_\theta,w_\theta$ are $C^3$ jointly in $(\theta,x)$ on a
common compact neighborhood of all relevant supports, uniformly for
$\theta$ near zero, that $a_\theta$ is bounded above and bounded away from
zero there, and that
\[
 \Sigma=\{w_0=0\}\cap\operatorname{supp}a_0
\]
is a compact regular hypersurface with $|\nabla w_0|\ge c_*>0$.  Shrinking
$\Theta$ if necessary, all moving boundaries lie in one fixed tubular
collar.  Put
\begin{equation}\label{eq:v21-boundary-velocity-matrix}
 V(x)=D_\theta w_\theta(x)|_{\theta=0}\in\mathbb R^r,
 \qquad
 \mathcal J_\Sigma=
 \int_\Sigma\frac{a_0(x)V(x)V(x)^t}{|\nabla w_0(x)|}\,d\sigma(x).
\end{equation}
Multiplying $w_\theta$ by a positive smooth factor and dividing
$a_\theta$ by the same factor leaves $\mathcal J_\Sigma$ unchanged.

For the local triangular family let $\delta_n\downarrow0$, $0<p_n\le1$,
and
\begin{equation}\label{eq:v21-vector-rate}
 n p_n\delta_n^2\log(1/\delta_n)\longrightarrow1.
\end{equation}
For $h\in\mathbb R^r$ define on
$\Omega=U\sqcup\{\dagger\}$
\begin{equation}\label{eq:v21-thinned-family}
 P_{n,h}=(1-p_n)\delta_\dagger+p_nf_{\delta_nh}(x)\,dx.
\end{equation}
No domination relation between $P_{n,h}$ and $P_{n,0}$ is assumed.

\begin{lemma}[Uniform vector collar estimates]
\label{lem:v21-vector-collar}
Let
\[
 \mathsf S(x)=D_\theta\log a_\theta(x)|_0+\frac{V(x)}{w_0(x)},
 \qquad D=\{w_0>0\}.
\]
There are $q_0,C>0$ such that, for $0<q<q_0$ and unit vectors
$u,v\in\mathbb R^r$,
\begin{align}
 \mathbb E_0[(u^t\mathsf S)(v^t\mathsf S)
       \mathbf 1_{\{w_0\ge q\}}]
 &=u^t\mathcal J_\Sigma v\log(1/q)+O(1),
                                      \label{eq:v21-vector-second}\\
 \mathbb E_0[|u^t\mathsf S|^3\mathbf 1_{\{w_0\ge q\}}]
 &\le Cq^{-1},                       \label{eq:v21-vector-third}\\
 \mathbb E_0[|u^t\mathsf S|^4\mathbf 1_{\{w_0\ge q\}}]
 &\le Cq^{-2},                       \label{eq:v21-vector-fourth}\\
 \left\|\mathbb E_0[\mathsf S\mathbf 1_{\{w_0\ge q\}}]\right\|
 &\le Cq,
 \qquad
 \mathbb P_0(0<w_0<q)\le Cq^2.      \label{eq:v21-vector-centering}
\end{align}
The constants are uniform for the indicated unit directions and on compact
parameter sets satisfying the displayed regularity bounds.
\end{lemma}

\begin{proof}
Use normal coordinates $(y,s)$ with $s=w_0$ in a fixed tubular
neighborhood of $\Sigma$.  The Jacobian is
$|\nabla w_0|^{-1}+O(s)$ and $f_0=a_0s$ on the interior collar.  The only
nonintegrable contribution to the second score moment is therefore
\[
 \frac{a_0(u^tV)(v^tV)}s,
\]
whose integral from $q$ to a fixed collar width is
$u^t\mathcal J_\Sigma v\log(1/q)+O(1)$.  The $k$th singular score moment
has normal integrand $O(s^{1-k})$, which gives
\eqref{eq:v21-vector-third}--\eqref{eq:v21-vector-fourth}.  The same
coordinates give the quadratic collar mass.  Differentiating
$\int f_\theta=1$ at zero gives $\mathbb E_0\mathsf S=0$ in the improper
integrable sense; the omitted collar has first score moment $O(q)$.
\end{proof}

\begin{lemma}[Lebesgue decomposition and support-exclusive mass]
\label{lem:v21-support-decomposition}
For every compact $K\subset\mathbb R^r$ there is $C_K$ such that, uniformly
for $h\in K$ and all large $n$, the Lebesgue decomposition of $P_{n,h}$
relative to $P_{n,0}$ satisfies
\begin{align}
 P_{n,h}^{\perp 0}(\Omega)&\le C_Kp_n\delta_n^2,
                                      \label{eq:v21-alt-singular-mass}\\
 P_{n,0}\{f_{\delta_nh}=0<f_0\}&\le C_Kp_n\delta_n^2.
                                      \label{eq:v21-null-exclusive-mass}
\end{align}
Consequently the probability, under either product experiment, of seeing at
least one support-exclusive observation is at most
$C_Knp_n\delta_n^2=o(1)$.
\end{lemma}

\begin{proof}
Joint $C^1$ regularity gives
$|w_{\delta_nh}-w_0|\le C_K\delta_n$ on the common collar.  Every point on
which exactly one of $f_{\delta_nh},f_0$ is positive therefore lies in
$|w_0|\le C_K\delta_n$.  On that strip both relevant densities are
$O(\delta_n)$, while its Lebesgue volume is $O(\delta_n)$ by the tubular
coordinates.  This proves the one-observation bounds after multiplication by
$p_n$.  The product bounds follow by a union bound, and
\eqref{eq:v21-vector-rate} gives
$np_n\delta_n^2=O(1/\log(1/\delta_n))=o(1)$.
\end{proof}

The preceding lemma identifies the singular part but does not turn the
finite-$n$ family into a dominated one.  We next construct a common-collar
representative to which ordinary likelihood theory applies.

Put
\begin{equation}\label{eq:v21-truncation-scale}
 L_n=(\log(1/\delta_n))^{1/4},\qquad
 q_n=\delta_nL_n,\qquad C_n=\{w_0\ge q_n\}.
\end{equation}
Introduce a second cemetery symbol $\partial$ and let
$T_n:\Omega\to C_n\sqcup\{\dagger,\partial\}$ keep $\dagger$ and every
$x\in C_n$, while sending every successful $x\notin C_n$ to $\partial$.
Write
\begin{equation}\label{eq:v21-censored-law}
 \overline P_{n,h}=T_nP_{n,h}
 =(1-p_n)\delta_\dagger
 +p_nf_{\delta_nh}\mathbf1_{C_n}\,dx
 +p_nr_{n,h}\delta_\partial,
 \qquad
 r_{n,h}=\int_{C_n^c}f_{\delta_nh}\,dx.
\end{equation}

\begin{lemma}[Common-collar asymptotic equivalence]
\label{lem:v21-common-collar-equivalence}
For every compact $K\subset\mathbb R^r$,
\begin{equation}\label{eq:v21-collar-mass}
 \sup_{h\in K}r_{n,h}\le C_Kq_n^2,
 \qquad
 \sup_{h\in K}np_nr_{n,h}=o(1).
\end{equation}
Moreover, for all large $n$ the family
$\{\overline P_{n,h}:h\in K\}$ is dominated by
$\overline P_{n,0}$, and the product experiments
$\{P_{n,h}^{\otimes n}:h\in K\}$ and
$\{\overline P_{n,h}^{\otimes n}:h\in K\}$ have Le Cam distance at most
\begin{equation}\label{eq:v21-lecam-censoring}
 C_Knp_nq_n^2
 =O((\log(1/\delta_n))^{-1/2})=o(1).
\end{equation}
\end{lemma}

\begin{proof}
For $h\in K$, the moving support is contained in
$\{w_0>-C_K\delta_n\}$.  Integrating the bound
$f_{\delta_nh}=O(w_0+C_K\delta_n)$ over
$-C_K\delta_n<w_0<q_n$ gives $r_{n,h}=O(q_n^2)$ uniformly.  The second
part of \eqref{eq:v21-collar-mass} follows from
\eqref{eq:v21-vector-rate} because
$np_nq_n^2=O(L_n^2/\log(1/\delta_n))$.

On $C_n$ one has
$w_{\delta_nh}\ge q_n-C_K\delta_n\ge q_n/2$ for all large $n$, uniformly
in $h\in K$.  Hence all successful densities are mutually positive on
$C_n$; the atom at $\partial$ has positive mass at the null as well, and
the common failure atom causes no difficulty.  Thus the censored family is
dominated.

The map $T_n$ is a parameter-independent Markov kernel from the original
experiment to the censored one.  Conversely, use the identity kernel on
$C_n\sqcup\{\dagger\}$ and reconstruct an arbitrary fixed successful point
when $\partial$ occurs.  The reconstruction can disagree with the original
sample only on the event that at least one observation was censored.  Its
probability is at most $np_nr_{n,h}$, uniformly in $h\in K$.  This gives
\eqref{eq:v21-lecam-censoring}.
\end{proof}

\begin{lemma}[Uniform LAN for the dominated representative]
\label{lem:v21-dominated-lan}
For $h$ in a fixed compact set define, under
$\overline P_{n,0}^{\otimes n}$,
\begin{align}
 \Delta_n&=\delta_n\sum_{i=1}^nY_i
 \mathsf S(X_i)\mathbf1_{C_n}(X_i),
                                      \label{eq:v21-delta}\\
 Q_n&=\delta_n^2\sum_{i=1}^nY_i
 \mathsf S(X_i)\mathsf S(X_i)^t\mathbf1_{C_n}(X_i),
                                      \label{eq:v21-Q}
\end{align}
where $Y_i$ is the success indicator and, conditionally on success under the
null, $X_i$ has density $f_0$.  Then
\begin{equation}\label{eq:v21-dominated-expansion}
 \log\frac{d\overline P_{n,h}^{\otimes n}}
          {d\overline P_{n,0}^{\otimes n}}
 =h^t\Delta_n-\frac12h^tQ_nh+o_{\overline P_{n,0}}(1)
\end{equation}
uniformly for $h$ in compact subsets of $\mathbb R^r$, and
\begin{equation}\label{eq:v21-central-limits}
 \Delta_n\Rightarrow N_r(0,\mathcal J_\Sigma),
 \qquad Q_n\longrightarrow\mathcal J_\Sigma
 \quad\text{in probability}.
\end{equation}
The probability that the atom $\partial$ occurs is $o(1)$ uniformly on every
compact local parameter set.
\end{lemma}

\begin{proof}
The last assertion is Lemma~\ref{lem:v21-common-collar-equivalence}.  On its
complement the likelihood ratio involves only the common support $C_n$ and
the unchanged failure atom.  Taylor expansion before division by $w_0$ gives,
uniformly for bounded $h$ and $x\in C_n$,
\[
 \log\frac{f_{\delta_nh}(x)}{f_0(x)}
 =\delta_nh^t\mathsf S(x)
 -\frac{\delta_n^2}{2}(h^t\mathsf S(x))^2
 +R_{n,h}(x),
\]
where the regular parameter remainder has expectation
$O(\delta_n^2)$ per successful observation and the cubic logarithmic
remainder is bounded in expectation by
$C_K\delta_n^3\mathbb E_0(1+\|\mathsf S\|^3)\mathbf1_{C_n}$.
Lemma~\ref{lem:v21-vector-collar} therefore gives
\[
 np_n\delta_n^2=O(1/\log(1/\delta_n))=o(1),
 \qquad
 np_n\delta_n^3q_n^{-1}
 =O((L_n\log(1/\delta_n))^{-1})=o(1).
\]
This proves \eqref{eq:v21-dominated-expansion} uniformly on compact sets.

The truncated-score mean has norm at most $Cq_n$, so
$\|\mathbb E\Delta_n\|\le Cnp_n\delta_nq_n
=O(L_n/\log(1/\delta_n))=o(1)$.  Its covariance tends to
$\mathcal J_\Sigma$ because
$\log(1/q_n)/\log(1/\delta_n)\to1$.  On $C_n$ every scalar score
contribution multiplied by $\delta_n$ is $O(L_n^{-1})$, so Lindeberg holds
uniformly on the unit sphere and Cram\'er--Wold gives the first limit in
\eqref{eq:v21-central-limits}.  Finally
\[
 np_n\delta_n^4
 \mathbb E_0[\|\mathsf S\|^4\mathbf1_{C_n}]
 \le Cnp_n\delta_n^4q_n^{-2}=o(1),
\]
while the mean of $Q_n$ tends to $\mathcal J_\Sigma$.  Hence
$Q_n\to\mathcal J_\Sigma$ in probability.
\end{proof}

\begin{theorem}[Non-dominated vector boundary Gaussian limit]
\label{thm:v21-vector-boundary-gaussian}
Under \eqref{eq:v21-vector-density}--\eqref{eq:v21-vector-rate}, the following
statements hold.

\begin{enumerate}
\item For every compact local parameter set, the dominated common-collar
experiment $\{\overline P_{n,h}^{\otimes n}\}$ is uniformly LAN with
information matrix $\mathcal J_\Sigma$ in the sense of
\eqref{eq:v21-dominated-expansion}--\eqref{eq:v21-central-limits}.

\item The original non-dominated experiment
$\{P_{n,h}^{\otimes n}\}$ is uniformly asymptotically equivalent, on every
compact local parameter set, to that LAN representative.  In particular the
original and null local families are mutually asymptotically contiguous on
compact sets, despite the fact that finite-$n$ absolute continuity may fail.
Every fixed finite local subexperiment converges in Le Cam distance to the
Gaussian shift experiment
\[
                         Z=h+\mathcal J_\Sigma^{-1/2}G,
 \qquad G\sim N_r(0,I),
\]
with the usual interpretation on the range of $\mathcal J_\Sigma$ when the
matrix is singular.

\item For every fixed $h$,
\begin{equation}\label{eq:v21-testing-profile}
 \|P_{n,h}^{\otimes n}-P_{n,0}^{\otimes n}\|_{\rm TV}
 \longrightarrow
 2\Phi\!\left(\frac{\sqrt{h^t\mathcal J_\Sigma h}}2\right)-1.
\end{equation}
If $\mathcal J_\Sigma$ is positive definite, the statistic
\begin{equation}\label{eq:v21-local-estimator}
 \widehat h_n=\mathcal J_\Sigma^{-1}\Delta_n,
\end{equation}
with observations outside $C_n$ contributing zero, satisfies
\[
 \widehat h_n\Rightarrow N_r(h,\mathcal J_\Sigma^{-1})
\]
uniformly for $h$ in compact local sets.

\item For every positive semidefinite $W$ and
$\ell_M(z)=\min\{z^tWz,M\}$,
\begin{equation}\label{eq:v21-minimax-truncated}
 \lim_{R\to\infty}\liminf_{n\to\infty}
 \inf_{T_n}\sup_{\|h\|\le R}
 \mathbb E_{n,h}\ell_M(T_n-h)
 \ge \mathbb E\ell_M(Z_0),
 \qquad Z_0\sim N_r(0,\mathcal J_\Sigma^{-1}).
\end{equation}
Letting $M\to\infty$ gives the quadratic lower bound
$\operatorname{tr}(W\mathcal J_\Sigma^{-1})$.  On each fixed compact local
set the fourth moments of \eqref{eq:v21-local-estimator} are uniformly
bounded, and hence its quadratic risk converges uniformly there to the same
Gaussian value.
\end{enumerate}
\end{theorem}

\begin{proof}
Item 1 is Lemma~\ref{lem:v21-dominated-lan}.  In particular the likelihood
ratios of the dominated representative satisfy Le Cam's first lemma, and the
compact-uniform remainder plus the same argument applied to bounded local
sequences gives compact-uniform contiguity.  Lemma
\ref{lem:v21-common-collar-equivalence} transfers this statement and the
Gaussian limit experiment to the original non-dominated family, proving
item 2.  Notice that this conclusion uses asymptotic equivalence and does not
retroactively create a finite-$n$ Radon--Nikodym derivative for the original
family.

Le Cam's third lemma applied to the dominated representative translates the
central sequence by $\mathcal J_\Sigma h$.  The asymptotic equivalence then
transfers the conclusion to the original family.  The total variation
formula in item 3 is the distance between equal-covariance Gaussian shifts.

For the moment statement, on $C_n$ and for $h$ in a fixed compact set the
local density ratio is bounded above and below by constants tending to one,
because $\delta_n/w_0\le L_n^{-1}$.  Thus the null third- and fourth-moment
bounds remain uniform under local alternatives.  The standard fourth-moment
formula for sums, together with
\eqref{eq:v21-vector-second}--\eqref{eq:v21-vector-fourth}, yields
\[
 \sup_{h\in K}\mathbb E_{n,h}\|\Delta_n\|^4\le C_K.
\]
The rare censored event has probability $o(1)$ uniformly and contributes
zero to the displayed statistic.  Hence $\widehat h_n$ is uniformly
integrable for quadratic loss on compact sets.

Finally, finite-dimensional convergence of experiments gives the local
asymptotic minimax bound first for bounded continuous losses
$\ell_M$; see \cite{IbragimovHasminskii,HiranoPorter}.  Monotone convergence
in $M$ gives the quadratic lower bound, while the uniform fourth-moment bound
gives the matching compact-local risk limit for
\eqref{eq:v21-local-estimator}.
\end{proof}

\begin{lemma}[Quantitative stability of a linearly vanishing support]
\label{lem:v21-hypersurface-stability}
Let
$f=A(w)_+$ and $\widetilde f=\widetilde A(\widetilde w)_+$ be normalized
densities on the same bounded open set.  Suppose $A,\widetilde A$ are
uniformly positive and $C^1$, $w,\widetilde w$ have uniformly regular zero
sets in one fixed tubular collar, and
\[
 \|w-\widetilde w\|_{C^1}\le\varepsilon<1/2,
 \qquad
 \|A-\widetilde A\|_{C^0}\le\eta.
\]
Then
\begin{equation}\label{eq:v21-hypersurface-stability}
 H^2(f\,dx,\widetilde f\,dx)
 \le C\left\{\varepsilon^2\log\frac e\varepsilon+\eta^2\right\}.
\end{equation}
The constant is uniform on compact families with the stated bounds.
\end{lemma}

\begin{proof}
Use the tubular coordinate $s=w$.  On $|s|\le C\varepsilon$ the total mass
of both densities is $O(\varepsilon^2)$.  On
$s>C\varepsilon$ the supports agree and
\[
 \bigl|\sqrt{A w}-\sqrt{\widetilde A\widetilde w}\bigr|^2
 \le C\left(\eta^2s+\frac{\varepsilon^2}{s}\right).
\]
The first term is integrable with size $O(\eta^2)$ and the second integrates
to $O(\varepsilon^2\log(e/\varepsilon))$.  The same estimates cover the
part of the collar on which $\widetilde w$ is used as normal coordinate.
Away from the collar all denominators are uniformly bounded.  This proves
\eqref{eq:v21-hypersurface-stability}.
\end{proof}

\begin{remark}[Terminology]
The original family \eqref{eq:v21-thinned-family} is genuinely
non-dominated in outward-moving directions.  Accordingly, we reserve the
literal phrase ``LAN expansion'' for the dominated common-collar
representative and state the conclusion for the original family as
asymptotic equivalence to a uniformly LAN experiment, or equivalently as
convergence to the same Gaussian limit experiment.  This is a distinction at
finite sample size, not a loss of the local Gaussian information or minimax
conclusion.
\end{remark}
