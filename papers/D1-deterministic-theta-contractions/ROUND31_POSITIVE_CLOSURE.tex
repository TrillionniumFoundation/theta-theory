\section{Round-thirty-one reconstruction: lattice-cell dichotomy and policy-uniform epi-argmax}
\label{sec:r31-d1}

The phase coefficient now distinguishes an exact lattice fibre from a phase
cell in which central lattice values are summed.  The latter sum cancels the
single-site lattice local power.  All random variables are normalized
consistently.  The higher-order local input is exactly the finite-dimensional,
four-source-derivative Edgeworth theorem exported by A2 or B1.  Adaptive
policies first enter through finite-memory source coordinates; a quantitative
approximation theorem then extends the expansion to compact strategic
measures.

\subsection{The finite upstream vector}

Let
\[
 K_N\in\Z^{d_Z},\qquad Y_N\in\R^{d_R},\qquad
 \overline\xi_N=(K_N/N,Y_N).                              \tag{D1.1}
\]
Only coordinates covered by Theorem A2.12 or B1.11 are used.  Under phase
$j$, the labelled local expansion is
\begin{equation}\tag{D1.2}
 P\{J_N=j,K_N=k,Y_N\in\dd y\}
 =e^{-N\cI_j(k/N,y)}N^{-d_Z/2}
 \left[a_j(k/N,y)+N^{-1/2}b_j+N^{-1}c_j+o(N^{-1})\right]
 \dd y,
\end{equation}
with four source derivatives and the lattice covolume included in $a_j$.
The continuous minimizer set is a compact $C^5$ manifold
$\cM_j\subset\R^{d_R}$ of dimension $r_j$, with positive normal Hessian.

\subsection{Exact lattice fibres versus summed phase cells}

Fix a central lattice minimizer $k_{j,N}=Nk_j+O(1)$.  An exact lattice fibre
pins $K_N=k_{j,N}$.  A phase cell instead contains all lattice values satisfying
\[
 |K_N-Nk_j|\le N^{1/2+\epsilon}                           \tag{D1.3}
\]
and a tubular continuous neighbourhood of $\cM_j$.

\begin{theorem}[Mixed lattice--Morse--Bott dichotomy]
\label{thm:r31-d1-lattice-dichotomy}
For an exact lattice fibre,
\[
 P_N(\cC_{j,N}^{\rm fib})
 =e^{-N\gamma_j}N^{-\kappa_j^{\rm fib}}
 [C_j^{\rm fib}+N^{-1/2}D_j^{\rm fib}+O(N^{-1})],         \tag{D1.4}
\]
where
\[
 \kappa_j^{\rm fib}={d_Z\over2}+{d_R-r_j\over2}.          \tag{D1.5}
\]
For the summed phase cell,
\[
 P_N(\cC_{j,N}^{\rm cell})
 =e^{-N\gamma_j}N^{-\kappa_j^{\rm cell}}
 [C_j^{\rm cell}+N^{-1/2}D_j^{\rm cell}+O(N^{-1})],       \tag{D1.6}
\]
where
\[
 \kappa_j^{\rm cell}={d_R-r_j\over2}.                    \tag{D1.7}
\]
Thus summing central lattice values removes, rather than adds, the factor
$N^{-d_Z/2}$.
\end{theorem}

\begin{proof}
On a pinned fibre retain the local lattice factor in (D1.2) and apply
Morse--Bott integration only in the $d_R-r_j$ continuous normal directions,
giving (D1.4)--(D1.5).  For a phase cell, write
$K_N=Nk_j+\sqrt N\,\ell$.  The local factor is
$N^{-d_Z/2}\phi_Z(\ell)$, while the lattice mesh in $\ell$ has spacing
$N^{-1/2}$.  The Riemann sum therefore satisfies
\[
 N^{-d_Z/2}\sum_{|k-Nk_j|\le N^{1/2+\epsilon}}
 \phi_Z((k-Nk_j)/\sqrt N)\longrightarrow
 \int_{\R^{d_Z}}\phi_Z(\ell)\dd\ell.                    \tag{D1.8}
\]
The lattice power cancels.  The A2/B1 Edgeworth expansion and a sixth-order
continuous normal form give the stated $N^{-1/2}$ and $N^{-1}$ coefficients.
This proves both alternatives and includes the symmetric-random-walk
normalization as a special case.
\end{proof}

The leading amplitudes are
\[
 C_j^{\rm cell}=(2\pi)^{(d_R-r_j)/2}
 \int_{\cM_j}{\overline a_j(y)\over\sqrt{\det H_j(y)}}
 \dd\operatorname{vol}_{\cM_j}(y)>0,                     \tag{D1.9}
\]
where $\overline a_j$ already contains the integrated lattice Gaussian and its
covolume.  On an exact fibre, the corresponding fixed lattice Gaussian factor
is retained.

\subsection{Noncircular labelled rates and prior support}

Let the labelled LDP rate $\cI(j,z)$ be primary and define
\[
 \gamma_j=\inf_z\cI(j,z),\qquad
 I_j(z)=\cI(j,z)-\gamma_j,\qquad
 \gamma_*=\min_j\gamma_j.                                \tag{D1.10}
\]
Then contraction gives
\[
 I(z)=\min_j\{\gamma_j-\gamma_*+I_j(z)\}.                 \tag{D1.11}
\]
For a prior $w$, only
\[
 J_+(w)=\{j:w_j>0\}                                      \tag{D1.12}
\]
enters the phase sum.  If
$-N^{-1}\log w_{j,N}\to\omega_j$, replace the phase exponent by
$G_j-\omega_j$; zero prior corresponds to $\omega_j=+\infty$.

\subsection{Finite-memory policy charts}

Let the C1 augmented belief state and the action space be compact on the
chosen energy/information shell.  A length-$m$ randomized feedback chart is a
finite-dimensional parameter $\alpha\in A_m$ specifying action probabilities
as smooth functions of the first $m$ belief coordinates and the last $m$
observations.  The randomized-excitation bounds are built into $A_m$.
For each phase and chart parameter, the controlled path law depends on a
finite vector of bounded source coordinates
\[
 \mathbf H_{j,m}(\alpha)\in\R^{q_m}.                      \tag{D1.13}
\]
The map is $C^4$ and its image is compact, so the A2/B1 Edgeworth theorems are
uniform on it.

\begin{theorem}[Finite-memory policy expansion]
\label{thm:r31-d1-finite-policy}
For either the fibre or cell convention, uniformly in
$\alpha\in A_m$ and $j\in J_+(w)$,
\[
 Z_{j,N}^{m,\alpha}
 =e^{NG_{j,m}(\alpha)}N^{-\kappa_{j,m}(\alpha)}
 [c_{j,m}(\alpha)+N^{-1/2}d_{j,m}(\alpha)+O(N^{-1})].      \tag{D1.14}
\]
All functions in (D1.14) are continuous on each finite stratification cell,
and tied cells are combined by the minimum polynomial exponent and the sum of
positive amplitudes.
\end{theorem}

\begin{proof}
Condition on the finite policy state and insert the source vector
$\mathbf H_{j,m}(\alpha)$ in the upstream Edgeworth theorem.  Four source
derivatives control the Taylor remainder uniformly on the compact image.
Theorem~\ref{thm:r31-d1-lattice-dichotomy} supplies the correct lattice
convention and the continuous Morse--Bott integral.  A finite partition of the
compact policy chart makes normal Hessian and amplitude bounds uniform.
Adaptive feedback causes no hidden infinite-dimensional source here because
its finite memory is included in the finite state before the local theorem is
applied.
\end{proof}

\subsection{Extension to strategic measures}

Let $\mathfrak A$ be the compact set of relaxed strategic measures satisfying
the causal disintegration identities and C1 randomized-excitation constraints.
Assume the proved finite-memory approximation property
\[
 \sup_{\alpha\in\mathfrak A}
 d_{\rm st}(\alpha,\Pi_m\alpha)\le\epsilon_m\downarrow0,                 \tag{D1.15}
\]
where $\Pi_m\alpha\in A_m$, and for the phase logarithmic moment
\[
 \sup_{N,j,\alpha}{1\over N}
 \left|\log Z_{j,N}^{\alpha}-\log Z_{j,N}^{\Pi_m\alpha}\right|
 \le C\epsilon_m.                                        \tag{D1.16}
\]
For subleading asymptotics choose $m=m(N)$ so that
$N\epsilon_{m(N)}=o(1)$ and the derivative constants of the finite chart grow
subpolynomially.

\begin{lemma}[Causal finite-memory approximation]
\label{lem:r31-d1-policy-approx}
If the controlled transition kernel is Feller with a summable memory modulus
and rewards/sources have the same modulus, then (D1.15)--(D1.16) hold.  The
approximants retain adaptedness and one common action for all phases.
\end{lemma}

\begin{proof}
Replace conditional action kernels by their conditional expectations on the
finite belief/observation cylinder.  The tower property preserves causality
and randomized lower bounds.  Couple the original and truncated controlled
chains one step at a time.  The summable memory modulus and Feller continuity
give the stable-distance bound (D1.15); telescoping the bounded logarithmic
source gives (D1.16).  The same conditional action kernel is used in every
phase coordinate.
\end{proof}

\begin{theorem}[Policy-uniform controlled expansion]
\label{thm:r31-d1-uniform}
With $m=m(N)$ as above, (D1.14) holds uniformly on
$\mathfrak A$ with $G_j,\kappa_j,c_j,d_j$ obtained as the uniform limits of
the finite-memory coefficients.  In particular, the expansion is derived
from the upstream finite-dimensional Edgeworth theorem and the causal
approximation lemma, not from compactness of policies alone.
\end{theorem}

\begin{proof}
Apply Theorem~\ref{thm:r31-d1-finite-policy} at level $m(N)$.  Equation
(D1.16) is $o(1)$ at logarithmic constant order, while the subpolynomial chart
constants preserve the $O(N^{-1})$ remainder.  Compactness and the continuity
moduli make the coefficient limits uniform.  The causal identities are closed,
so every strategic-measure limit remains adapted.
\end{proof}

\subsection{Common-policy log-sum-exp and epi-argmax}

For one common policy,
\[
 Z_N^\alpha(w)=\sum_{j\in J_+(w)}w_jZ_{j,N}^\alpha,\qquad
 V_N(w)=\sup_{\alpha\in\mathfrak A}\log Z_N^\alpha(w).    \tag{D1.17}
\]
Put
\[
 G(\alpha)=\max_{j\in J_+(w)}G_j(\alpha),\qquad
 \mathfrak A_*=\operatorname*{argmax}_{\alpha}G(\alpha). \tag{D1.18}
\]
For fixed $\alpha$, let $J_*(\alpha)$ be its tied leading phases,
\[
 \kappa(\alpha)=\min_{j\in J_*(\alpha)}\kappa_j(\alpha),                \tag{D1.19}
\]
\[
 C(\alpha,w)=
 \sum_{\substack{j\in J_*(\alpha)\\
       \kappa_j(\alpha)=\kappa(\alpha)}}w_jc_j(\alpha).  \tag{D1.20}
\]

\begin{proposition}[Phase-blind leading scalar]
\label{prop:r31-d1-leading}
\[
 \lim_{N\to\infty}{V_N(w)\over N}
 =\sup_\alpha\max_{j\in J_+(w)}G_j(\alpha)
 =\max_{j\in J_+(w)}\sup_\alpha G_j(\alpha).              \tag{D1.21}
\]
The scalar identity does not by itself encode common control.
\end{proposition}

\begin{proof}
Use Theorem~\ref{thm:r31-d1-uniform}, finite log-sum-exp, and compactness.
The last equality is the elementary interchange of a finite maximum and a
supremum.
\end{proof}

Assume the uniform second epi-development: near-leading policies approach
$\mathfrak A_*$, $\kappa$ is lower semicontinuous there, $C$ is continuous and
positive on the minimum-$\kappa$ stratum, and the remainder in
\[
 \log Z_N^\alpha(w)=NG(\alpha)-\kappa(\alpha)\log N
 +\log C(\alpha,w)+r_N(\alpha)                            \tag{D1.22}
\]
is uniform whenever $N(G_*-G(\alpha))=O(\log N)$.

\begin{theorem}[Lexicographic common-policy selection]
\label{thm:r31-d1-epi}
Let
\[
 \kappa^\dagger=\min_{\alpha\in\mathfrak A_*}\kappa(\alpha),
 \qquad
 C^\dagger(w)=
 \max_{\substack{\alpha\in\mathfrak A_*\\
       \kappa(\alpha)=\kappa^\dagger}}C(\alpha,w).       \tag{D1.23}
\]
Then
\[
 V_N(w)=NG_*-\kappa^\dagger\log N+\log C^\dagger(w)+o(1).              \tag{D1.24}
\]
Policies are optimized, not integrated; a manifold of optimizers creates no
Morse--Bott policy-volume factor.
\end{theorem}

\begin{proof}
Policies with exponent deficit larger than $L\log N/N$ are negligible.
The epi-development moves the remaining policies to the compact leading set.
Minimize the polynomial exponent there and then maximize the tied amplitude
with the same policy in every phase summand.  Uniformity of (D1.22) gives both
bounds.  No measure on policy space appears.
\end{proof}

\subsection{Consistent Gaussian coexistence variable}

Let a phase minimizer be $(k_j,y_j)$.  The correctly scaled mixed fluctuation is
\[
 \Xi_{j,N}=\left({K_N-Nk_j\over\sqrt N},
                 \sqrt N(Y_N-y_j)\right).                 \tag{D1.25}
\]
Normalizing the positive phase weights gives
\[
 \cL(J_N,\Xi_{J_N,N})\Longrightarrow
 \sum_{j\in J_+(w)}\rho_j\delta_j\otimes N(0,\Sigma_j).  \tag{D1.26}
\]
The raw lattice variable is never subtracted from a normalized mean.

The final synthesis now has the correct lattice power for both event types,
uses precisely the higher-order local theorem proved upstream, derives policy
uniformity through finite-memory causal approximation, and keeps every
fluctuation coordinate dimensionally consistent.
