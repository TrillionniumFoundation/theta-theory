
\section{Round-twenty-seven reconstruction: full-saddle phase weights and subleading common control}
\label{sec:r27-d1}

The scalar max-plus limit cannot distinguish a policy shared across hidden
phases from phasewise pre-optimization.  The shared constraint is therefore
kept exactly at finite scale and in the first nontrivial subleading
asymptotics.  The leading scalar identity is recorded correctly rather than
assigned information it does not contain.

\subsection{A joint local theorem in every integrated coordinate}

Let $(M_N,Z_N)$ be the phase order parameter and all remaining
lattice/continuous path coordinates.  For phase $j$, write
$\xi=(m,z)$ and assume the source-uniform A2 or B1 theorem gives
\[
 P_N\{\xi_N\in d\xi,\ J_N=j\}
 =e^{-NJ_j(\xi)}N^{-\lambda_j}
 \left[a_j(\xi)+N^{-1/2}b_j(\xi)+O(N^{-1})\right]\,d\varkappa_N(\xi),   \tag{D1.1}
\]
with two uniform derivatives.  Every coordinate that is integrated in the
phase probability appears in $\xi$; none is frozen without conditioning.

Let the minimizer set $\mathcal M_j\subset\mathbb R^{d_j}$ be a compact
$C^3$ manifold of dimension $r_j$.  In a tubular chart
$\xi=(y,n)$,
\[
 J_j(y,n)=\gamma_j+\tfrac12
 \langle H_j(y)n,n\rangle+O(|n|^3),\qquad H_j(y)\ge cI.    \tag{D1.2}
\]
All exact-number, lattice, and coarea powers are included in $\lambda_j$.

\begin{theorem}[Full Morse--Bott phase coefficient]
\label{thm:r27-d1-weight}
For a rate-separated tubular phase cell $\mathcal C_{j,N}$,
\[
 P_N(\mathcal C_{j,N})
 =e^{-N\gamma_j}N^{-\kappa_j}
 \left(c_j+N^{-1/2}d_j+O(N^{-1})\right),                  \tag{D1.3}
\]
where
\[
 \kappa_j=\lambda_j+\tfrac12(d_j-r_j),\qquad
 c_j=(2\pi)^{(d_j-r_j)/2}
 \int_{\mathcal M_j}{a_j(y,0)\over\sqrt{\det H_j(y)}}\,
 d\operatorname{vol}_{\mathcal M_j}(y)>0.                 \tag{D1.4}
\]
The coefficient $d_j$ contains the cubic normal form, Jacobian, and
$b_j$ contribution in every integrated $m$ and $z$ direction.
\end{theorem}

\begin{proof}
Goodness of the joint rate gives a compact set containing all minimizers and
a positive rate gap on its complement.  Use tubular coordinates in the full
$\xi$ space, set $n=N^{-1/2}\eta$, and Taylor-expand the exponent, amplitude,
and tubular Jacobian.  Gaussian domination gives (D1.3)--(D1.4).  The
rate-separated complement is exponentially smaller.  Since $z$ is included
in $\xi$, its saddle dimensions and determinants are present rather than
silently omitted.
\end{proof}

\subsection{Labelled component rates}

Define the joint labelled rate directly from the upstream labelled LDP:
\[
 \mathcal I(j,z)=\gamma_j+I_j(z)-\gamma_*,
 \qquad \gamma_*=\min_k\gamma_k.                          \tag{D1.5}
\]
The conditional component rate is
\[
 I_j(z)=\mathcal I(j,z)-\inf_y\mathcal I(j,y).             \tag{D1.6}
\]
Then contraction gives
\[
 I(z)=\min_j\{\gamma_j-\gamma_*+I_j(z)\}.                  \tag{D1.7}
\]
Thus the notation is defined before use, and the polynomial weights
(D1.3) are kept separate from exponential rates.

\subsection{Exact common-policy state}

Let phase $j$ carry a C1 belief $\pi_j$ and let $w_j$ be an arbitrary prior
phase weight, $\sum_jw_j=1$.  The augmented belief is
\[
 \mathbf b=(w_1,\pi_1;\ldots;w_J,\pi_J).
\]
For one common action $a$,
\[
 r(\mathbf b,a,y)=\sum_jw_jr_j(\pi_j,a,y),\qquad
 w_j'={w_jr_j(\pi_j,a,y)\over r(\mathbf b,a,y)},           \tag{D1.8}
\]
and $\pi_j'=B_j(\pi_j,a,y)$.  The prior $w$ is not replaced by
$e^{-N\gamma_j}$ unless it is specifically the microscopic phase prior from
(D1.3).

For a common admissible policy $\alpha$ define
\[
 V_N^\alpha(\mathbf b)
 =\log\sum_jw_j
 E_j^\alpha e^{NF_j(Z_N)},\qquad
 V_N(\mathbf b)=\sup_{\alpha\ {\rm common}}V_N^\alpha(\mathbf b).       \tag{D1.9}
\]
Optimization remains outside the phase sum.

\subsection{Uniform control compactness}

Let $\mathfrak A$ be relaxed feedback policies on the compact augmented belief
space, with the stable topology of occupation kernels.  Assume rewards and
transition kernels are equicontinuous and the action costs have compact
sublevels.  C1's integrated Feller theorem implies that $\mathfrak A$ is
sequentially compact after relaxation.

For each phase define
\[
 G_j(\alpha)=\sup_z\{F_j^\alpha(z)-I_j^\alpha(z)\}.         \tag{D1.10}
\]

\begin{lemma}[Uniform controlled Laplace principle]
\label{lem:r27-d1-uniform}
Uniformly in $\alpha\in\mathfrak A$,
\[
 {1\over N}\log E_j^\alpha e^{NF_j(Z_N)}
 =G_j(\alpha)+o(1),                                      \tag{D1.11}
\]
and the subleading local expansion is uniform on every
$\varepsilon_N$-near-optimal policy set with
$\varepsilon_N\downarrow0$ and $N\varepsilon_N\to\infty$.
\end{lemma}

\begin{proof}
Exponential tightness is uniform because the controlled action sublevels are
compact and the policy occupation kernels are tight.  The upper bound follows
from a finite cover of each common compact sublevel.  For the lower bound,
regular exposed controls and local source insertions are dense uniformly in
the stable policy topology.  Equicontinuity transfers a finite family of
recovery controls to nearby policies.  The A2/B1 local expansion and its two
source derivatives are uniform on that compact family.  These estimates give
(D1.11) and the stated subleading uniformity.
\end{proof}

\begin{theorem}[Leading common-control limit]
\label{thm:r27-d1-leading}
\[
 \lim_{N\to\infty}{1\over N}V_N(\mathbf b)
 =\sup_{\alpha\in\mathfrak A}\max_jG_j(\alpha)
 =\max_j\sup_{\alpha\in\mathfrak A}G_j(\alpha).            \tag{D1.12}
\]
The second equality is an elementary scalar identity; consequently the
leading value alone does not encode the sharing constraint.
\end{theorem}

\begin{proof}
Lemma \ref{lem:r27-d1-uniform} permits the finite-$N$ supremum to pass through
the uniform limit.  Finite log-sum-exp gives the maximum over phases.  For a
finite maximum,
$\sup_\alpha\max_jG_j(\alpha)=\max_j\sup_\alpha G_j(\alpha)$.
\end{proof}

\subsection{The subleading shared-policy theorem}

Let
\[
 G_*=\sup_\alpha\max_jG_j(\alpha),\qquad
 \mathfrak A_*=\{\alpha:\max_jG_j(\alpha)=G_*\},
\]
and
\[
 J_*(\alpha)=\{j:G_j(\alpha)=G_*\}.
\]
For each active pair $(j,\alpha)$, the local theorem and
Theorem \ref{thm:r27-d1-weight} give
\[
 E_j^\alpha e^{NF_j(Z_N)}
 =e^{NG_j(\alpha)}N^{-\kappa_j(\alpha)}
 \left[c_j(\alpha)+N^{-1/2}d_j(\alpha)+o(N^{-1/2})\right].              \tag{D1.13}
\]
Put
\[
 \kappa_*(\alpha)=\min_{j\in J_*(\alpha)}\kappa_j(\alpha),
 \qquad
 C(\alpha,\mathbf b)
 =\sum_{\substack{j\in J_*(\alpha)\\
                   \kappa_j(\alpha)=\kappa_*(\alpha)}}
 w_jc_j(\alpha).                                          \tag{D1.14}
\]

\begin{theorem}[Subleading common-policy selection]
\label{thm:r27-d1-subleading}
Assume $G_j$ has a uniform quadratic drop away from the compact
near-optimal set.  Then
\[
 V_N(\mathbf b)
 =NG_*-\kappa^\dagger\log N+
 \log\sup_{\substack{\alpha\in\mathfrak A_*\\
                      \kappa_*(\alpha)=\kappa^\dagger}}
 C(\alpha,\mathbf b)+o(1),                                \tag{D1.15}
\]
where
\[
 \kappa^\dagger=\min_{\alpha\in\mathfrak A_*}\kappa_*(\alpha).
\]
If the displayed supremum is attained on a smooth policy manifold, the
$N^{-1/2}$ correction is obtained by replacing $c_j$ with
$c_j+N^{-1/2}d_j$ and applying Morse--Bott integration on that common-policy
manifold.  Formula (D1.15) generally differs from optimizing each phase
before the sum.
\end{theorem}

\begin{proof}
Policies with $G_*-G(\alpha)>\log^2N/N$ are smaller by
$e^{-\log^2N}$ and can be removed uniformly using Lemma
\ref{lem:r27-d1-uniform}.  On the remaining common near-optimal set, insert
(D1.13) before any phasewise optimization.  The smallest polynomial exponent
wins, and tied phases add with the same policy-dependent weights, giving
(D1.15).  Because one $\alpha$ appears in every summand of (D1.14), replacing
it by separate $\alpha_j$ changes the coefficient in general.  The final
statement is the ordinary uniform Morse--Bott expansion on the policy
manifold.
\end{proof}

\subsection{Complex phase charts}

The source-uniform A2/B1 theorem supplies
\[
 Z_{j,N}^\alpha(z)
 =e^{N\Lambda_j^\alpha(z)}
 N^{-\kappa_j(\alpha)}
 \left[a_j^\alpha(z)+O(N^{-1/2})\right]                   \tag{D1.16}
\]
on one common complex chart, uniformly in near-optimal policies.  Since
$a_j^\alpha(0)>0$, compactness permits a zero-free smaller chart.  If one
phase has a strict real-part gap, Rouche's theorem makes the full sum zero-free
there; on coexistence surfaces the finite analytic sum is retained and its
zeros are not suppressed.

\subsection{Labelled Gaussian coexistence}

Under phase $j$,
\[
 \sqrt N(X_N-m_j)\Rightarrow N(0,\Sigma_j).
\]
Normalizing the exact weights in (D1.3) yields, along every convergent
subsequence,
\[
 \mathcal L\bigl(J_N,\sqrt N(X_N-m_{J_N})\bigr)
 \Rightarrow\sum_j\rho_j\delta_j\otimes N(0,\Sigma_j).    \tag{D1.17}
\]
These same microscopic weights may initialize (D1.8), but arbitrary prior
weights remain allowed.

The full joint saddle, labelled rate, uniform policy compactness, finite-scale
log-sum-exp, phase-blind leading identity, and subleading common-policy
selection now form one consistent synthesis theorem.
