\section{Round-ten reconstruction: microscopic phase disintegration and commuting limits}
\label{sec:r10-d1}

This paper retains a standalone theorem, but phase labels are now genuine
microscopic objects.  They are not external tags appended to copies of the
same law.

\subsection{Phase-restricted finite-volume laws}

For every finite projection \(m\), let
\(B_{n,1},\ldots,B_{n,J_m}\) be disjoint measurable phase basins determined by
an order parameter in that projection.  Assume their uncovered probability is
superexponentially small and their basin boundaries have subexponential
probability under all compact real source tilts.  Define
\[
 w_{n,j}=\mathbb P_n(B_{n,j}),
 \qquad
 \mathbb P_{n,j}=\mathbb P_n(\cdot\mid B_{n,j}),
 \qquad
 Z_{n,j}(\theta)=
 \mathbb E_n[e^{n\theta\cdot X_n};B_{n,j}].
\]
The platform papers construct these basins from their exposed phase order
parameters and prove the boundary estimates.

\begin{theorem}[Microscopic phase mixture LDP]
\label{thm:r10-d1-mixture}
Suppose \(n^{-1}\log w_{n,j}\to-\alpha_j\) and every restricted law
\(\mathbb P_{n,j}\) satisfies a good LDP with rate \(I_j\).  Then the original
unlabelled law satisfies a good LDP with
\[
 I(x)=\min_j\{\alpha_j+I_j(x)\}.
\]
The joint labelled law has rate \(\alpha_j+I_j(x)\).  This conclusion follows
from the microscopic decomposition, not from the false identity
\((\max_jQ_j)^*=\min_jQ_j^*\).
\end{theorem}

\begin{proof}
For a closed set, sum the finitely many phase probabilities and use the
largest exponential term; the superexponential remainder is irrelevant.  For
an open set, choose one phase attaining the minimum and use its lower bound
together with the phase weight.  Exponential tightness of the restricted laws
and finiteness of the phase set give goodness.  Keeping the label makes the
same proof immediate on the disjoint union.
\end{proof}

\subsection{Phase-local complex charts}

\begin{theorem}[Phase-local response and Lee--Yang-compatible coexistence]
\label{thm:r10-d1-charts}
For every phase \(j\) and compact subset of its real regularity domain,
\[
 Q_{n,j}(\theta)=n^{-1}\log Z_{n,j}(\theta)
\]
has a uniform zero-free complex neighborhood and converges there with all
derivatives to \(Q_j\).  No common zero-free tube is asserted across a
coexistence surface.  The physical pressure is
\[
 Q(\theta)=\max_j\{Q_j(\theta)-\alpha_j\},
\]
and Lee--Yang zeros may approach its nondifferentiable phase boundary.
\end{theorem}

\begin{proof}
Inside a phase basin the platform transfer/cluster operator has a simple
isolated eigenvalue and the basin-boundary remainder is subexponential.  A
uniform Riesz contour gives the restricted zero-free logarithm and analytic
convergence.  At coexistence, two restricted exponentials have comparable
real parts and their sum can have zeros at distance \(O(n^{-1})\); the theorem
makes no contrary assertion.  Laplace's principle over the finite phase
mixture gives the maximum formula.
\end{proof}

\begin{corollary}[Exact finite-centred local likelihood]
\label{cor:r10-d1-lan}
Within a regular phase, for a finite source direction \(h\),
\[
 L_{n,j}(h)=
 \exp\left\{
 \sqrt n\,h\cdot(X_n-DQ_{n,j}(0))
 -n[Q_{n,j}(h/\sqrt n)-Q_{n,j}(0)
     -n^{-1/2}DQ_{n,j}(0)h]
 \right\}
\]
is an exact mean-one density and converges to the Cameron--Martin likelihood
with covariance \(D^2Q_j(0)\).
\end{corollary}

\begin{proof}
The normalization is the finite restricted log moment generating function, so
expectation is exactly one.  Taylor expansion on the phase-local complex chart
and the finite-dimensional Gaussian limit give convergence and uniform
integrability.
\end{proof}

\subsection{Faces and projective passage}

\begin{theorem}[Labelled projective LDP and topology upgrade]
\label{thm:r10-d1-projective}
If every finite observable projection has the phase-labelled LDP above and the
platform laws are exponentially tight in the weighted path topology, then the
joint labelled path law satisfies a full good LDP with the supremum of the
consistent finite rates.  Every finite-rate boundary-face point has a recovery
sequence obtained inside one restricted phase basin or by the microscopic
phase mixture.
\end{theorem}

\begin{proof}
Exact contraction of each restricted law gives projective consistency.
Dawson--G\"artner yields the projective LDP and exponential tightness upgrades
it to the weighted topology.  For a target-open set, choose a compact rate
sublevel and a finite projection whose inverse image separates the target
point inside that compact set.  Use the phase-specific recovery law; its
exponential tightness makes probability outside the compact negligible under
the same tilt.  Boundary faces are obtained by the platform's restricted
normal-source limit before the projective limit.  No Straszewicz statement is
substituted for microscopic recovery.
\end{proof}

\subsection{Thin shells through phase mixtures}

\begin{theorem}[Exact labelled and unlabelled shell conditioning]
\label{thm:r10-d1-shell}
Suppose the A2 or B1 local coefficient holds uniformly on every active phase
chart.  Conditioning a restricted phase law on a regular thin shell gives its
source-dependent information projection.  Conditioning the original mixture
gives the finite mixture of active projected phases with weights equal to the
phase weights times their shell coefficients.  The logarithmic conditioning
cost is the minimum of the phase costs.
\end{theorem}

\begin{proof}
Apply the phase-local coefficient to numerator and denominator within each
basin.  Sum over the finitely many phases.  Exponentially nonminimal terms
disappear; tied phases retain their explicit subexponential coefficients.
The basin-boundary remainder is superexponential.  This is an actual
microscopic conditional decomposition, not a formal label appended after the
limit.
\end{proof}

\subsection{The commutation theorem}

\begin{theorem}[Phase-labelled commutation principle]
\label{thm:r10-d1-main}
On the platform domains proved in A2--A4 and B1--B4, the following operations
commute within each regular restricted phase and then with the finite
microscopic mixture:
\begin{enumerate}[label=(\roman*)]
\item finite-source differentiation and the microscopic limit;
\item source-dependent thin-shell conditioning;
\item continuous typed contraction;
\item formation of the exact-centered Gaussian tangent and local likelihood;
\item dynamic action concatenation and the nonlinear semigroup.
\end{enumerate}
At coexistence, derivatives are phase-wise and the physical tangent is the
compact convex set or mixture selected by the microscopic phase weights.
\end{theorem}

\begin{proof}
Within one phase, uniform complex convergence gives derivative commutation;
A2/B1 give the shell coefficient; ordinary and extended contraction give the
observable rates; A4/B3 give Gaussian process convergence; and B4 gives the
additive dynamic semigroup.  The finite phase decomposition of
Theorem~\ref{thm:r10-d1-mixture} allows each identity to be summed before or
after the limit.  At coexistence no common analytic derivative is asserted;
the labelled statements and explicit mixture weights give the correct
set-valued or mixed result.
\end{proof}
