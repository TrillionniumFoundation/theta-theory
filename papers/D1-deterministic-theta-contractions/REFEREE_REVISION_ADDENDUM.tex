\subsection{Finite-volume normalization}

Let
\[
 Q_\epsilon(\Theta)
 =
 \mu_\epsilon^{-1}\log
 \mathbb E e^{\mu_\epsilon\Theta}.
\]

\begin{proposition}[Exact derivative identities]
\label{prop:rr2-d1-scale}
For bounded directions \(F,G\),
\[
 DQ_\epsilon(\Theta)[F]
 =
 \mathbb E_{\Theta,\epsilon}F,
 \qquad
 D^2Q_\epsilon(\Theta)[F,G]
 =
 \mu_\epsilon
 \operatorname{Cov}_{\Theta,\epsilon}(F,G).
\]
If
\(\widehat F_\epsilon=\sqrt{\mu_\epsilon}
(F-\mathbb E_{\Theta,\epsilon}F)\), then
\[
 D^2Q_\epsilon(\Theta)[F,G]
 =
 \operatorname{Cov}_{\Theta,\epsilon}
 (\widehat F_\epsilon,\widehat G_\epsilon).
\]
\end{proposition}

\begin{proof}
Differentiate the normalized logarithm of the partition function twice.
The second derivative produces two factors of \(\mu_\epsilon\) from the
exponent and one inverse factor from the normalization.
\end{proof}

\begin{theorem}[Deterministic pressure--action--fluctuation triangle]
\label{thm:rr2-d1-triangle}
On every regular exposed source branch established in B1--B4:
\[
 \begin{array}{c}
 DQ(\Theta)=\text{biased kinetic law},\\
 D^2Q(\Theta)=\text{covariance of the limiting }
 \sqrt{\mu_\epsilon}\text{-fluctuation field},\\
 Q^*=\text{joint density--collision action of B2}.
 \end{array}
\]
All three statements hold on one kinetic state space and one
Boltzmann--Grad speed.
\end{theorem}

\begin{proof}
Normal convergence of the marked cumulants gives convergence of the first
two finite-volume derivatives.  Proposition~\ref{prop:rr2-d1-scale} and B3
identify the Hessian with the fluctuation covariance.  The full LDP of B2
and Varadhan duality identify the Legendre transform with the action.
\end{proof}

\label{thm:rr2-d1-compiler}

\item homological current by continuous projection of a periodic-Sinai path;
\item physical diffusion by centering \(\kappa-v\tau\) and inverse-clock

\item scalar theta expectation by restriction to a one-dimensional cotangent

\end{enumerate}
The resulting rate or semigroup is never called a contraction of a different

\end{theorem}

\begin{proof}

use the marginal maps on the many-particle path space; exponential tightness
and the corresponding martingale problems identify their scaling limits.
The typed coproduct theorem of C2 records the platform label and prevents
cross-platform substitution.
\end{proof}

\subsection{Joint likelihood-ratio and Gaussian tangent}

Let \(Z^\epsilon\) be the centered fluctuation field and let
\(M^\epsilon\) be the canonical likelihood-ratio martingale for a smooth
source direction.

\begin{theorem}[Joint mod-Gaussian convergence]
\label{thm:rr2-d1-modgaussian}
There is a complex neighborhood of zero on which
\[
 \log\mathbb E
 \exp\left\{
 \langle z,Z^\epsilon\rangle+
 \log M_T^\epsilon
 \right\}
\]
converges locally uniformly to the cumulant of a Gaussian field under its
Cameron--Martin tilt.  Consequently
\[
 (Z^\epsilon,M^\epsilon)
 \Longrightarrow
 \left(Z,\mathcal E\!\left(\int h\,dW\right)\right)
\]
in the product path topology, and the likelihood ratios are uniformly
integrable.
\end{theorem}

\begin{proof}
The source-uniform analytic expansions in B2 and B3 give local uniform
convergence of the joint cumulant and a uniform exponential moment on a
larger complex ball.  Cauchy bounds make cumulants of order at least three
vanish under the central-limit scaling.  Tightness follows from B2/B3.
The limiting quadratic cumulant is the Cameron--Martin formula.  The
larger-ball exponential bound gives de la Vall\'ee--Poussin uniform
integrability.
\end{proof}

\begin{corollary}[Girsanov and quadratic BSDE representation]
\label{cor:rr2-d1-bsde}
For the diffusive contraction, the canonical tangent law is the Girsanov
law with drift \(D h\).  For bounded terminal work and bounded running cost,
the entropic value is the unique bounded solution of the corresponding
quadratic BSDE.
\end{corollary}

\begin{proof}
Theorem~\ref{thm:rr2-d1-modgaussian} identifies the stochastic exponential
and permits passage of expectations under the tilted laws.  Girsanov's
theorem identifies the drift.  Martingale representation and the exponential
transform give the quadratic BSDE; bounded terminal data give uniqueness.
\end{proof}

\begin{proof}
