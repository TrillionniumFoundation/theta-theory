\begin{abstract}
We organize the hard-sphere theory as derivatives and contractions of one
path pressure.  The first derivative is the biased Boltzmann law, the Hessian
is the fluctuating Boltzmann covariance, and the full convex dual is the
dynamic action.  Scalar calibrated rays satisfy entropic rigidity.  Tagged
Brownian motion, heat-bath energy rays and single-particle billiard theories
arise only through typed contractions.  At Gaussian contractions, likelihood
ratios yield Girsanov densities and quadratic BSDEs; no such representation is
substituted for the full kinetic Hamiltonian.
\end{abstract}

\section{Derivative hierarchy of one pressure functional}
Let $Q_T(\Theta)$ be limiting path pressure on the local source domain.  At an
exposed phase, define the tangent law $\mathbb P^\Theta$ by canonical tilt.

\begin{theorem}[Pressure tangent hierarchy]\label{thm:tangent}
For bounded directions $F,G$,
\[
 DQ_T(\Theta)[F]=\E_{\mathbb P^\Theta}F,
 \qquad
 D^2Q_T(\Theta)[F,G]=\Cov_{\mathbb P^\Theta}(F,G).
\]
After Boltzmann--Grad scaling, the first derivative gives the biased Boltzmann
path, the Hessian gives the fluctuating Boltzmann covariance, and the convex
dual gives the dynamic action.
\end{theorem}

\begin{proof}
Differentiate the normalized finite-volume exponential tilt and pass to the
limit using uniform source analyticity.  Paper II identifies the derivatives
with the kinetic objects.
\end{proof}

At phase coexistence the derivative is replaced by the compact convex set of
tangent phase laws.

\section{Scalar entropic rigidity and mechanical calibration}
Let a scalar mechanical constraint select $\Theta=\theta G$.  Assume a
certainty equivalent is monotone, law invariant, cash additive, relevant,
strongly time consistent and sufficiently regular for conditional composition.
The corresponding rigidity theorem is proved under precise hypotheses in
\cite{KupperSchachermayer2009}.

\begin{theorem}[Calibrated entropic rigidity]\label{thm:rigidity}
On a nontrivial scalar ray, the smooth certainty equivalents with those
properties are linear expectation and
\[
 \mathfrak C^\theta(X)=\frac1\theta\log\E e^{\theta X}.
\]
The mechanical preparation fixes $\theta$ as a Lagrange multiplier or rate
derivative; the axioms do not choose its numerical value.
\end{theorem}

\begin{proof}
Cash additivity for $u^{-1}(\E u(X))$ forces
$u(x+c)=A(c)u(x)+B(c)$.  Smooth solutions are affine or exponential.  Relevance
and nonlinearity exclude the affine branch; the path I-projection identifies
the multiplier.
\end{proof}

\section{Universal contractions}
\begin{enumerate}[label=(\alph*)]
\item Zero cotangent gives the Boltzmann equation.
\item The pressure Hessian gives the fluctuating Boltzmann equation.
\item A near-equilibrium tagged-particle contraction followed by fast
relaxation gives Brownian motion \cite{BGSRBrownian2016}.
\item An energy-transfer constraint gives the heat-bath ray, whose multiplier
is inverse temperature after physical calibration.
\item Freezing all but one particle against a periodic background recovers the
single-particle path theories as typed limiting models, not finite-system
identities.
\end{enumerate}

\begin{proposition}[Conservation gauge]
The collision part is invariant under adding
$a+b\cdot v+c|v|^2$ to the density cotangent.  Under microcanonical preparation,
these directions only change fixed boundary constants.
\end{proposition}

\section{Gaussian likelihood ratios, Girsanov and BSDE tangents}
At a finite-dimensional Gaussian contraction
\[
 dX_t=b_tdt+\sigma_tdW_t,
\]
a bounded drift cotangent $u_t$ gives
\[
 Z_T=\exp\left\{\int_0^Tu_t\cdot dW_t
 -\frac12\int_0^T|u_t|^2dt\right\}.
\]

\begin{theorem}[Diffusive tangent representations]\label{thm:girsanov}
The Gaussian tangent of the kinetic canonical likelihood ratio is $Z_T$.  The
scalar entropic value admits the quadratic BSDE
\[
 Y_t=\Phi(X_T)+\int_t^T
 \left(c_s+\frac\theta2|Z_s^{\rm BSDE}|^2\right)ds
 -\int_t^TZ_s^{\rm BSDE}\,dW_s.
\]
These are Gaussian-tangent representations; the full kinetic semigroup uses
the nonquadratic Hamilton--Boltzmann functional.
\end{theorem}

\begin{theorem}[v11 hierarchy closure]\label{thm:main}
The deterministic microcanonical path law, collision-marked LDP, cotangent
geometry, nonlinear semigroup, controlled typing and tangent contractions form
one closed short-time hard-sphere theory.
\end{theorem}

\begin{remark}[Stronger theorem classes]
The long-time Boltzmann law of large numbers in \cite{DengHaniMa2024} does not
supply a source-decorated long-time LDP.  Such an LDP, adaptive microscopic
collision control and a deterministic hydrodynamic Navier--Stokes fluctuation
theory are not claimed or used.
\end{remark}
