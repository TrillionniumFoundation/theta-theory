\begin{abstract}
We derive theta-expectation from a deterministic path ensemble rather than
postulating exponential utility.  A macro current $a$ in the area-preserving
collision map of Paper I selects the unique conjugate field
$\theta=I'(a)$.  The normalized canonical path partition function is exactly
the entropic expectation under the deterministic driven map.  This gives an
exact microscopic nonlinear dynamic programming operator.

Using the collision impulses and actual branch roof of Paper II, we prove
monotonicity, stability, a corrector-free local expansion, collision-clock
convergence, and an exponential random-horizon replacement.  The physical
limit is
\[
 -u_t=c+b\cdot Du+\frac1{2\bar\tau}C:D^2u
 +\frac{\theta}{2\bar\tau}Du^TC Du.
\]
Here both $C/\bar\tau$ and $\theta$ are derived from the same deterministic
macro preparation.  We separate the fixed-law entropic semigroup from the
controlled risk-sensitive game; this removes the type ambiguity that formerly
propagated into tangent-law theory.
\end{abstract}

\section{The canonical path partition function}
Fix $a\in(-1,1)$ and let
\[
 \theta=I'(a)=\frac12\log\frac{3(1+a)}{7(1-a)}.
\]
Let $m_a$ be the driven Lebesgue branch law from Paper I.  If $\theta=0$, all
formulas below are interpreted by continuous extension.

For a bounded terminal functional $F$ define
\[
 \cE_{s,t}^{\theta,a}[F]
 =\frac1\theta\log
 \E_{m_a}\left[e^{\theta F}\mid\mathcal F_s\right].
\]
Paper I proves that this is the excess pressure of the base deterministic path
ensemble after adding the terminal work $F$.

\begin{proposition}[Structural properties]\label{prop:structure}
The family $\cE^{\theta,a}$ is monotone, cash additive, constant preserving,
strictly relevant, and strongly time consistent.
\end{proposition}
\begin{proof}
All properties follow from positivity and the conditional-expectation tower
after exponentiation.  Strict relevance follows from strict positivity of all
branch weights.
\end{proof}

\section{The microscopic free-energy recursion}
Let $d_i,\tau_i,C(a),\bar\tau(a)$ be the data of Paper II.  Let $b,c$ be
bounded globally Lipschitz functions and $\Phi$ bounded Lipschitz.  Define
\[
 X_{k+1}=X_k+\varepsilon d_{I_k}
 +\varepsilon^2\tau_{I_k}b(X_k).
\]
For a deterministic collision horizon $N$, define backward
\[
 V_N^\varepsilon(x)=\Phi(x),
\]
\begin{equation}\label{eq:micro-dpp}
 V_k^\varepsilon(x)
 =\frac1\theta\log\sum_{i=1}^4w_i(a)
 \exp\left\{\theta\left[
 \varepsilon^2\tau_i c(x)
 +V_{k+1}^\varepsilon
 (x+\varepsilon d_i+
 \varepsilon^2\tau_i b(x))
 \right]\right\}.
\end{equation}

\begin{theorem}[Exact first-principles DPP]\label{thm:dpp}
Equation \eqref{eq:micro-dpp} is the conditional excess path pressure of the
deterministic collision ensemble.  It is not an independent stochastic or
utility postulate.
\end{theorem}
\begin{proof}
Condition on the next branch under the driven Lebesgue law $m_a$, add the
branch work $\varepsilon^2\tau_i c$, and apply the exact conditional
excess-pressure identity of Paper I.  Backward induction gives the recursion.
\end{proof}

\section{Consistency and the endogenous theta Hamiltonian}
Let $p=D\phi(x)$ and $X=D^2\phi(x)$.  Taylor expansion gives
\[
 \phi(x+\varepsilon d_i+
 \varepsilon^2\tau_ib)
 =\phi(x)+\varepsilon p\cdot d_i
 +\varepsilon^2\left(
 \tau_i b\cdot p+\frac12d_i^TXd_i
 \right)+O(\varepsilon^3).
\]

\begin{theorem}[One-step free-energy expansion]\label{thm:expansion}
Locally uniformly for smooth $\phi$,
\[
 \frac{\mathcal T_\varepsilon^{\theta,a}\phi-\phi}{\varepsilon^2}
 \longrightarrow
 \bar\tau(a)(c+b\cdot p)
 +\frac12C(a):X
 +\frac\theta2p^TC(a)p.
\]
\end{theorem}
\begin{proof}
The $O(\varepsilon)$ term has mean zero because
$\sum_iw_i(a)d_i=0$.  Expanding the logarithm of the exponential moment gives
its mean second-order term plus one half $\theta$ times the variance of
$p\cdot d_i$.  Paper II identifies that variance as $p^TC(a)p$.
\end{proof}

\section{Collision-clock convergence}
\begin{theorem}[Collision free-energy PDE]\label{thm:collision-pde}
With collision time $s=k\varepsilon^2$, the values converge locally uniformly
to the unique bounded viscosity solution
\[
 -u_s=\bar\tau(a)(c+b\cdot Du)
 +\frac12C(a):D^2u
 +\frac\theta2Du^TC(a)Du,
 \qquad u(T)=\Phi.
\]
\end{theorem}
\begin{proof}
The scheme is monotone because log-sum-exp is increasing in every argument,
and it is cash additive.  Bounded data give stability.  Consistency is
\cref{thm:expansion}.  Uniform positivity of $C(a)$ on compact current ranges gives uniform
parabolicity.  For $\theta\ne0$, the Cole--Hopf transform
$v=e^{\theta u}$ converts the equation into a linear uniformly parabolic
terminal problem with potential $\theta c$; for $\theta=0$ the equation is
already linear.  This gives global bounded comparison.  Apply the monotone
half-relaxed-limit theorem \cite{BarlesSouganidis1991,CrandallIshiiLions1992}.
\end{proof}

\section{Entropic physical-horizon replacement}
Let $V_{\rm phys}^{\varepsilon,a}(T,x)$ stop at the last collision before
physical time $T$, and let $V_{\rm coll}^{\varepsilon,a}$ use
$n_\varepsilon(T)=\lfloor T/(\varepsilon^2\bar\tau(a))\rfloor$ collisions.

\begin{lemma}[Tilted entropic Orlicz comparison]\label{lem:orlicz}
For bounded $X,Y$ and $\theta\ne0$, let
\[
 \frac{dQ_X}{dP}=\frac{e^{\theta X}}{\E e^{\theta X}},
 \qquad
 \frac{dQ_Y}{dP}=\frac{e^{\theta Y}}{\E e^{\theta Y}}.
\]
Then
\[
 \abs{
 \frac1\theta\log\E e^{\theta X}
 -\frac1\theta\log\E e^{\theta Y}}
 \le\frac1{\abs\theta}
 \max_{Q\in\{Q_X,Q_Y\}}
 \log\E_Qe^{\abs\theta\abs{X-Y}}.
\]
If $\abs X,\abs Y\le M$ and
$\E(e^{\abs\theta\abs{X-Y}}-1)\le\delta$, the right side is at most
\[
 \frac1{\abs\theta}\log(1+e^{2\abs\theta M}\delta).
\]
\end{lemma}
\begin{proof}
The exact ratio identity is
\[
 \frac{\E e^{\theta X}}{\E e^{\theta Y}}
 =\E_{Q_Y}e^{\theta(X-Y)}.
\]
Use it in both directions and dominate by the absolute exponential.  The
bounded-density estimate
$dQ_X/dP,dQ_Y/dP\le e^{2\abs\theta M}$ gives the second assertion.
\end{proof}

\begin{theorem}[Physical-time entropic replacement]\label{thm:physical-replacement}
For every compact $K\Subset(-1,1)$ and fixed horizon $T$,
\[
 \sup_{a\in K,x}
 \abs{V_{\rm phys}^{\varepsilon,a}(T,x)
 -V_{\rm coll}^{\varepsilon,a}
 (n_\varepsilon(T),x)}
 \le C_{K,T}\sqrt\varepsilon.
\]
\end{theorem}
\begin{proof}
Couple both recursions on the same deterministic branch path.  The terminal
payoff difference is bounded by its Lipschitz constant times the random-window
slow displacement; running work and drift differ by
$O(\varepsilon^2\abs{N_\varepsilon-n_\varepsilon})$.  The values are uniformly bounded over the fixed horizon.  Apply
\cref{lem:orlicz}; Paper II gives
$\E(e^{\abs\theta\abs{X-Y}}-1)=O(\sqrt\varepsilon)$ for the coupled payoff
difference.  This proves the stated uniform rate.
\end{proof}

\section{The physical theta-HJB equation}
\begin{theorem}[Endogenous theta-semigroup]\label{thm:theta-hjb}
The physical values converge to
\[
 -u_t=c+b\cdot Du+\frac1{2\bar\tau(a)}C(a):D^2u
 +\frac{\theta(a)}{2\bar\tau(a)}Du^TC(a)Du,
 \qquad u(T)=\Phi.
\]
Explicitly,
\[
 C(a)=\operatorname{diag}(1-a,6(1+a)),
 \qquad \bar\tau(a)=\frac{3-a}{2},
\]
so the nonlinear gradient term is
\[
 \frac{\theta(a)}{3-a}
 \left[(1-a)u_{x_1}^2+6(1+a)u_{x_2}^2\right].
\]
The solution operators form a physical theta-semigroup.
\end{theorem}
\begin{proof}
Combine \cref{thm:collision-pde,thm:physical-replacement} and rescale collision
time by $\bar\tau(a)$.  Time consistency follows already at the microscopic
level and passes to the unique limit.
\end{proof}

\section{Fixed-law, fixed-policy, and controlled types}
\begin{definition}[Three noninterchangeable exports]
The \emph{uncontrolled entropic semigroup} is \cref{thm:theta-hjb}.  A
\emph{fixed-policy entropic semigroup} freezes a prescribed feedback and has
the same log-moment structure.  A \emph{controlled risk-sensitive value}
takes a supremum or saddle over policies outside the log moment.  Only the
first two have pure covariance second variation without an optimizer-response
term.
\end{definition}

Let $U,V$ be finite or compact action sets and suppose $b^{uv},c^{uv}$ are
chosen before the branch.  The controlled one-step value is
\[
 \max_u\min_v\frac1\theta\log\sum_iw_i(a)e^{\theta[
 \varepsilon^2\tau_i c^{uv}+F(x+\varepsilon d_i+
 \varepsilon^2\tau_i b^{uv})]}.
\]
Its viscosity limit has the same diffusion and entropic term, with the local
max--min applied to $c^{uv}+b^{uv}\cdot p$.  Tangent derivatives of this
controlled value are handled by Paper IV, not by silently reusing the fixed-law
covariance formula.

\section{Main theorem}
\begin{theorem}[Deterministic first-principles to theta-expectation]\label{thm:main}
Within the declared full-cover collision platform,
\[
\begin{aligned}
 \text{deterministic mechanics}
 &\to \text{natural path law}
 \to \text{current LDP}
 \to \theta=I'(a),\\
 &\to \text{excess-pressure DPP}
 \to \text{physical theta-HJB}.
\end{aligned}
\]
is a closed theorem chain.  Neither Brownian forcing nor exponential utility
nor a free risk parameter is inserted at the microscopic level.
\end{theorem}
\begin{proof}
Combine Papers I--II with
\cref{thm:dpp,thm:expansion,thm:collision-pde,thm:physical-replacement,thm:theta-hjb}.
\end{proof}

\section*{Scope and review status}
The theorem is a complete internal proof in the declared deterministic
full-cover collision scope.  It does not claim that an arbitrary periodic
Sinai billiard has the same exact symbolic reset or that independent external
review has occurred.
