\section{Arithmetic local variation of the unmodified raw density}
\label{sec:v49-raw-local-TV}
The local variation estimate now removes every source guard from a
raw density theorem. It also allows a roof test to vary arbitrarily
with the return and collision indices. The arithmetic modulation is
not removed.

\begin{theorem}[The full-source local variation law]
\label{thm:v49-raw-local-TV}
For every fixed $h>0$,
\begin{equation}\label{eq:v49-raw-local-TV}
 \lim_{m\to\infty}\sup_{R,\,1\le n\le m,\,k,t}
 \frac1h\int_t^{t+h}
 \left|m^2p_{n,R}(k,m,u)
       -\mathcal L_{m,R}(k_1,k_2,u,n)\right|\,du=0.
\end{equation}
This is the density of the complete original return law, with respect
to counting measure in $(k,m)$ and Lebesgue measure in $u$.
No return index is averaged. The transition kernel is the one in
\eqref{eq:v41-transition-kernel}.

At a fixed radius, on every central compact set, it may be replaced
by $c\mathfrak a_R(k,n,m)g_{\Omega_R}(Z(u))$, where
\[
 Z(u)=\frac{(k_1,k_2,u,n)-m(0,0,\bar\tau_R,c)}{\sqrt m}.
\]
The equivalent return normalization has the same arithmetic factor
and the matrix $D_R$, as in
Theorem~\ref{thm:v43-arithmetic-raw-criterion}.
\end{theorem}
\begin{proof}
At every fixed reconstruction band $B$, the band-limited test theorem
already gives
\[
 \sup_{R,n,k,u}\left|m^2(K_B*p_{n,R}(k,m,\cdot))(u)
                 -\mathcal L_{m,R}(k_1,k_2,u,n)\right|\longrightarrow0.
\]
This is the last displayed estimate in the proof of
Theorem~\ref{thm:v43-arithmetic-raw-criterion}, on the full source.
Integrating its absolute error on any interval of length $h$, and
using Theorem~\ref{thm:v49-full-local-correction} with $w=1$, shows
that the limsup in \eqref{eq:v49-raw-local-TV} is at most
$CB^{-1/192}$. Let $B\to\infty$ after the collision limsup. This
proves the first assertion, without selecting pointwise representatives
of the densities. The fixed-radius specialization of
Theorem~\ref{thm:v41-uniform-transition} is uniform on a slightly
larger central compact set containing the intervals under discussion.
Its uniform error can be integrated, giving the second assertion.
\end{proof}

\begin{corollary}[Arbitrary bounded roof selectors]
\label{cor:v49-roof-selector-TV}
The supremum in \eqref{eq:v49-raw-local-TV} may equivalently be
written
\begin{equation}\label{eq:v49-selector-dual}
 \sup_{R,n,k,t}\ \sup_{\substack{\psi:\ |\psi|\le1\\
                                \operatorname{supp}\psi\subset[t,t+h]}}
 \frac1h\left|\int\psi(u)
       [m^2p_{n,R}(k,m,u)-\mathcal L_{m,R}(k_1,k_2,u,n)]\,du\right|.
\end{equation}
Here $\psi$ is measurable and may depend on every displayed parameter,
including $m$. In particular the approximation holds uniformly for
all Borel subsets of a translated fixed roof interval.
\end{corollary}
\begin{proof}
The total variation norm of an absolutely continuous signed measure
is its $L^1$ norm. The upper inequality is immediate; equality follows
by the measurable sign of its density. This is a selector in the
observed roof variable, not an arbitrary trajectory selector with a
new Gaussian amplitude.
\end{proof}

\subsection{A consequence for the original mixed measure}
The preceding theorem can also be summed on a bounded central region
at a fixed return index. For $M<\infty$ put
\[
 V_{n,R}(k,m,u)=
       n^{-1/2}(k_1,k_2,m-n/c,u-n\bar\tau_R/c),
 \qquad \mathcal C_{n,R}(M)=\{|V_{n,R}|\le M,\ m\ge n\}.
\]
The positive part below is needed because a finite transition kernel
need not be pointwise positive at finite count.

\begin{corollary}[Central mixed-measure total variation]
\label{cor:v49-mixed-TV}
For every fixed $M<\infty$,
\begin{align}
 \lim_{n\to\infty}\sup_R
 \sum_{m\ge n}\sum_{k\in\Z^2}
 \int_{\{u:(k,m,u)\in\mathcal C_{n,R}(M)\}}
 \left|p_{n,R}(k,m,u)
       -m^{-2}(\mathcal L_{m,R}(k_1,k_2,u,n))_+\right|\,du=0.
                                      \label{eq:v49-mixed-TV}
\end{align}
Thus this restriction of the original mixed measure is approximated
in total variation, not only weakly. No assertion outside the stated
central region or at a prescribed individual roof value is included.
\end{corollary}
\begin{proof}
On the central region, $m=n/c+O_M(\sqrt n)$ and hence $m\asymp n$.
There are $O_M(\sqrt n)$ collision labels and $O_M(n)$ displacement
labels. For each pair the roof set is contained in an interval of
length $O_M(\sqrt n)$, covered by $O_M(\sqrt n)$ unit intervals.
Theorem~\ref{thm:v49-raw-local-TV}, with $h=1$, bounds each integral
by $m^{-2}e_m$, where $e_m\to0$ uniformly in all targets. The total
number of unit intervals is $O_M(n^2)$, so their sum is bounded by
$C_M\sup_{m\ge a_M n}e_m\to0$. Finally, for $p\ge0$,
$|m^2p-\mathcal L_+|\le|m^2p-\mathcal L|$. This justifies the
positive part and proves the claim.
\end{proof}

\subsection{Conditional roof laws and the remaining pointwise issue}
For $I_t=[t,t+h]$, define
\[
 G_{m,R}^{n,k}(I_t)=\int_{I_t}
                    (\mathcal L_{m,R}(k_1,k_2,u,n))_+\,du.
\]
Consider any family of these targets on which
$G_{m,R}^{n,k}(I_t)\ge d h$ for a fixed $d>0$.

\begin{corollary}[Total variation of the exact conditional roof law]
\label{cor:v49-conditional-roof-TV}
Uniformly on such a target family, the law of the original roof
$T_{n,R}$ conditioned on
\[
 K_{n,R}=k,\qquad N_{n,R}=m,\qquad T_{n,R}\in I_t
\]
is asymptotic in total variation to the probability with density
\begin{equation}\label{eq:v49-conditional-roof-reference}
 \frac{\one_{I_t}(u)
      (\mathcal L_{m,R}(k_1,k_2,u,n))_+}
             {G_{m,R}^{n,k}(I_t)}.
\end{equation}
The convention is
$d_{\TV}(P,Q)=\sup_A|P(A)-Q(A)|=\tfrac12\int|dP-dQ|$.
The conditioning event is unchanged. This is not a bridge conditioned
at the single equation $T_{n,R}=t$.
\end{corollary}
\begin{proof}
Let $f=m^2p\one_{I_t}$, $g=\mathcal L_+\one_{I_t}$ and
$E=\|f-g\|_1$. By Theorem~\ref{thm:v49-raw-local-TV},
$E/h\to0$ uniformly. Write $F=\int f$ and $G=\int g$.
Then $|F-G|\le E$, $F\ge dh/2$ eventually, and
\[
 \frac12\left\|\frac fF-\frac gG\right\|_1
                        \le \frac E F\le\frac{2E}{dh}\longrightarrow0.
\]
The source normalization is the original $\nu_R^*$ on both sides of
the conditional probability, so no change of event or section occurs.
\end{proof}

\begin{corollary}[Size of the exceptional roof set]
\label{cor:v49-exceptional-roofs}
For every fixed $h,\lambda>0$,
\[
 \sup_{R,n,k,t}\frac1h\operatorname{Leb}
 \{u\in[t,t+h]:|m^2p_{n,R}(k,m,u)
           -\mathcal L_{m,R}(k_1,k_2,u,n)|>\lambda\}\longrightarrow0.
\]
For the full signed correction at fixed large $B$, its corresponding
limsup is at most $C\lambda^{-1}B^{-1/192}$.
\end{corollary}
\begin{proof}
Apply the elementary inequality
$\operatorname{Leb}\{|F|>\lambda\}\le\lambda^{-1}\int|F|$
to Theorem~\ref{thm:v49-raw-local-TV} and to
\eqref{eq:v49-full-local-correction}.
\end{proof}

\paragraph{Relation to the original raw endpoint.}
The preceding results include the entire incidence and clearance
source in a density-norm local limit and in exact-window conditional
roof laws. They strengthen a fixed-test local theorem: the measurable
roof selector in \eqref{eq:v49-selector-dual} is chosen after $m$,
not fixed before the limit. They do not upgrade this norm convergence
to uniform convergence of density values. Narrow physical-boundary
spikes are still compatible with all the estimates above. Excluding
those spikes, or estimating their signed correction at the same roof
value, remains the extra assertion in
\eqref{eq:v48-physical-residual-criterion}. Its main term continues
to carry $\mathfrak a_R$ at fixed radius and $\mathcal L_{m,R}$
uniformly. Neither residue triviality nor a single-roof conditioned
path bridge follows from local total variation.
