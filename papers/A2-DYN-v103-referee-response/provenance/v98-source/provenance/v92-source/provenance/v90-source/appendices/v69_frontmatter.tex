\section{The preceding opening statements}
\label{app:v70-retained-front}
The preceding abstract and introduction are retained verbatim below.
Their scoped theorems and the full pointwise target keep their original
mathematical content.
\subsection*{Preceding abstract}

We study the joint displacement, collision count and flight time of
actual returns in a triangular periodic Lorentz gas, retaining the
complete arithmetic record and the original pointwise inversion
problem. We construct a positive comparison trace on the unchanged
physical itineraries and exact labels. Absolute control of the
clearance guard then yields an ordered caustic-height theorem on
angular-persistence strata, including zero center guards and all
finite angular birth orders. The zero-center contribution has an
additional square-root gain in the collar width; infinitely flat
guards require no analytic classification. A second comparison uses
finite-scale angular occupation and applies beyond the certified
germ disk, including labels born only at positive radius. We give
its physically weighted tail transfer and an exact disjoint
source decomposition. The complete incidence height, the radial
tail and the exterior clearance height remain explicit in the
positive identity for the unrestricted pointwise arithmetic theorem.

\subsection*{Preceding introduction}


\paragraph{The physical record.}
For $R\in[9/20,47/100]$ let
$J_{n,R}=(K_{n,R},N_{n,R},T_{n,R})$ be the actual return record
to $Y_R^*$ under $\nu_R^*=c_*^{-1}\nu|_{Y_R^*}$, with
$c_*=91/(10000\pi)$. The density $p_{n,R}(k,m,t)$ uses counting
measure in $(k,m)$ and Lebesgue measure in $t$. Occupation counts
collisions $0,\ldots,m-1$; terminal section membership is imposed
at collision $m$. The original pointwise problem is
\begin{equation}\label{eq:intro-v69-pointwise-target}
 \sup_{R,n,k}\operatorname*{ess\,sup}_{t:\,|Z_{m,R}|\le M}
 \left|m^2p_{n,R}(k,m,t)-\mathcal L_{m,R}(k_1,k_2,t,n)\right|
                                      \longrightarrow0,
\end{equation}
where $Z_{m,R}=(k_1,k_2,t-m\bar\tau_R,n-mc_*)/\sqrt m$.
The arithmetic transition kernel in $\mathcal L_{m,R}$, including
its zero classes, is retained. Our argument concerns the positive
physical sources which enter this pointwise inversion, not a
replacement process or an averaged return label.

\paragraph{The comparison source.}
At a selected normal critical word write $F_z=t_z+|y|^2/2$,
$h_\chi=r_\chi^2/2$, and $K_\chi=C\chi^{-2}$. Let
$B_{z,\lambda}$ be the exact physical indicator with label
$\lambda=(n,k)$, and let $D_z^\varepsilon$ be the unchanged
first-clearance multiplier. Its center value $g_z$ may vanish;
its Lipschitz constant is at most $L_g=C_g\varepsilon^{-1}$.
The source with this multiplier omitted, but every physical and
label test retained, has density $\widehat b_{z,\lambda}$.
Pointwise domination gives
\[
 b_{z,\lambda}^{\varepsilon,\mathrm{clr}}(t_z+u)
 \le\min\{1,g_z+L_g\sqrt{2u}\}\,
                       \widehat b_{z,\lambda}(t_z+u).
\]
Its positive collar trace satisfies the same ordered two-endpoint
bound as the guarded trace, by a direct comparison with the
original exact-return event.

\begin{leadtheorem}[Zero-center and all-order collar height]
\label{thm:intro-v69-zero-guard}
Let $b^{\rm a,K,H;\le\delta}$ be the original clearance source
on the angular-persistence stratum
$\mathfrak p^{\rm a}_{z,\lambda}\le K$, with $g_z\le\delta$,
restricted to $0<F_z-t_z<H$. For fixed
$\chi,\varepsilon,K,H,\delta$ with $2H<h_\chi$ and
$2K\sqrt H<1$,
\[
 \limsup_m\sup_{R,\lambda}m^2
       \|b^{\rm a,K,H;\le\delta}_{m;\lambda,R}\|_\infty
 \le6CH F_{\chi,K}(H)\min\{1,\delta+L_g\sqrt{2H}\},
\]
where
\[
 F_{\chi,K}(H)=e^{(2+\sqrt2)K_\chi\sqrt H}
                 \frac{1+\sqrt2K\sqrt H}{1-2K\sqrt H}.
\]
All finite angular birth orders are included with the same constant.
There is no positive-center-guard assumption. At $\delta=0$ the
bound is $6\sqrt2 C C_g\varepsilon^{-1}H^{3/2}F_{\chi,K}(H)$.
\end{leadtheorem}
\begin{proof}
The angular certificate alone compares every inner level of the
unguarded physical density to one positive outer-annulus average.
Absolute guard domination is applied on the inner source, rather
than used in the denominator of the comparison. The contributing
critical centers lie in $[t-H,t]$. The positive comparison trace at
width $2H$ has ordered bound $3CH$. The complete proof is
Theorem~\ref{thm:v69-absolute-guard-height}.
\end{proof}

\paragraph{Beyond an angular germ.}
For fixed $0<S<h_\chi$ define
\[
 V_z(S)=\frac{\operatorname*{ess\,sup}_{0<u<S}
                      \ell_{z,\lambda}(\sqrt{2u})}
                  {S^{-1}\int_0^S\ell_{z,\lambda}(\sqrt{2u})\,du},
\]
with value zero when the denominator vanishes. This is a finite-scale
geometric ratio, not either of the earlier angular condition numbers.
It is finite at each fixed chart and label. Its tail must nevertheless
be controlled after the parameter and label suprema and the collision
limsup; fixed-count finiteness does not establish that control.

\begin{leadtheorem}[Complete fixed-radius comparison]
\label{thm:intro-v69-full-radius}
The original clearance source on $0<F_z-t_z<S$ and $V_z(S)\le T$
satisfies
\[
 \mathcal H(b^{(0,S),S,V\le T})
             \le2CTS e^{2K_\chi\sqrt{2S}}.
\]
This includes portions beyond any angular-germ certificate, zero
center guards, and labels absent near the center but present at
positive radius. For $\alpha>0$, the remaining $V_z(S)>T$ source
satisfies
\[
 \mathcal H(b^{(0,S),S,V>T})
 \le e^{2K_\chi\sqrt{2S}}T^{-\alpha}\mathcal M_\alpha(S),
\]
where $\mathcal M_\alpha(S)$ is the exact physical collar-trace
moment in \eqref{eq:v69-radial-moment}. This implication requires a
finite moment bound; such a bound is not supplied by finiteness of
individual ratios.
\end{leadtheorem}
\begin{proof}
The physical density distortion compares the radial supremum to its
positive collar average with cost $e^{2K_\chi\sqrt{2S}}V_z(S)$.
At roof $t$, centers lie in $[t-S,t]$. The ordered comparison trace
bounds this interval by $2CS$. The tail follows from
$V\one_{\{V>T\}}\le T^{-\alpha}V^{1+\alpha}$.
See Theorem~\ref{thm:v69-finite-scale-height} and
Proposition~\ref{prop:v69-radial-tail-transfer}.
\end{proof}
