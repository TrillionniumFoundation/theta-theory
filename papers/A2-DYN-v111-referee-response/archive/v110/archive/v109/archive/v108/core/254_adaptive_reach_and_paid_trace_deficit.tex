\section{Adaptive positive comparison at the actual reached scale}
\label{sec:v105-adaptive}
The preceding stationary-arm argument requires physical reach to the
outgoing levels $\delta/2,2\delta,3\delta$. On the same original
regular backward-critical arm, retain the incoming roof $a_*$ and
positive incoming trace
\[
f_-^\varepsilon(x)=N_*/Q_*(x),\qquad
\lambda_\circ\le Q_*(x)\le\Lambda_\circ ,
\]
and let $b(x)=a_*+d+\psi(x)$ with $0\le d\le\delta/2$.
Choose an \emph{outer, collision-independent} scale
$0<\rho\le\delta$. Assume $\psi$ is continuous and strictly
increasing and \emph{actually reaches}, before a true physical
boundary, the levels
\begin{equation}\label{eq:v105-levels}
\psi(x_1)=\rho/8,\qquad
\psi(x_2)=\rho/2,\qquad
\psi(x_3)=3\rho/4,\quad x_1<x_2<x_3.
\end{equation}
Set $I=(0,x_1)$ and $J=(x_2,x_3)$ on this very physical sheet.
\begin{theorem}[Smaller-reach critical-arm source with paid deficit]
\label{thm:v105-reach}
Suppose the comparison intervals of a family of such actual arms are
disjoint on the original incident-component pairing-sheet space.
Assume their geometric ratios satisfy
\[
 {\Lambda_\circ\over\lambda_\circ}
 {|I|\over|J|}\le\kappa
\]
with a common outer cost $\kappa>0$, independent of the collision
count and exact labels. For every original Borel selector $H$,
at finite collision count and almost every roof $t$,
\begin{align}
T_{H,\mathrm{reach}}^\varepsilon(t)
&\le {\kappa\over \rho/4}
       \int_t^{t+\rho/4}\mathsf T_{4\delta}^\varepsilon(u)\,du
\notag\\
&\quad+C\kappa(1+\varepsilon^{-1})
       \int_{t-\delta}^{t+4\delta}v^0(u)\,du .
\label{eq:v105-reach}
\end{align}
Thus
\begin{equation}\label{eq:v105-reach-height}
\mathcal H_M(T_{H,\mathrm{reach}}^\varepsilon)
\le C\kappa\{\varepsilon^{1/16}
                   +(1+\varepsilon^{-1})\delta\}.
\end{equation}
This is the complete source on the selected reached arms,
not only a trace-balanced fraction; no endpoint-speed lower bound
or outgoing trace-fraction bound is imposed.
\end{theorem}
\begin{proof}
For $x\in I$, $\ell(x)=d+\psi(x)\le5\delta/8<\delta$;
for $y\in J$, $\ell(y)<d+3\rho/4\le5\delta/4<4\delta$.
Moreover $a(y)=a(x)=a_*$ and
$b(y)-b(x)>3\rho/8>\rho/4$. Therefore the actual one-sided
inclusion \eqref{eq:v104-inclusion} holds at $r=\rho/4$.
For each arm put
\[
C_j=\int_Ic^\varepsilon,\quad P_j=\int_Jf_-^\varepsilon,\quad
D_j=\int_Jc^\varepsilon,\quad
E_j=\int_J[f_-^\varepsilon-f_+^\varepsilon]_+.
\]
The incoming critical trace and the geometric ratio give
$C_j\le\kappa P_j$, while the positive minimum gives
$D_j=P_j-E_j$. Hence
$(C_j-\kappa D_j)_+\le\kappa E_j$, even when $D_j=0$.
The positive comparison theorem \ref{thm:v104-block} pays the
admitted fraction on $(t,t+\rho/4)$. For any original source
segment containing $t$, one has $a_*<t<a_*+\delta$, while
every comparison segment lies in $(a_*,a_*+4\delta)$.
The inherited weighted one-sided trace-variation estimate
\eqref{eq:v104-weighted-variation-input}, integrated over the
actual comparison segments, pays the full $E_j$ in
$(t-\delta,t+4\delta)$. Disjointness of the physical arms gives
multiplicity one in the uncut $v^0$ source sum.
This proves \eqref{eq:v105-reach} before any limit.
With $\rho,\delta,\kappa,\varepsilon$ fixed before collision
limsup, apply \eqref{eq:v104-paired-average} and
\eqref{eq:v104-uncut-input} to the two fixed windows.
\end{proof}
\begin{corollary}[Elastic width criterion]\label{cor:v105-elastic}
If the actual $\psi$ has
$x\psi'(x)/\psi(x)\le K_B$ almost everywhere on $(x_2,x_3)$,
then $\kappa\le\Lambda_\circ K_B/
(\lambda_\circ\log(3/2))$ is admissible.
For $K_B\le C\varepsilon(B)^{-1/32}$,
the complete reached source has height $O(B^{-1/384})$.
The conclusion permits $\rho_B\ll\delta(B)$ and physical arms
which never reach the old level $3\delta(B)$.
\end{corollary}
\begin{proof}
Integrate $\psi'/\psi$ across the two actually reached levels:
$\log(3/2)\le K_B\log(x_3/x_2)$, so
$x_3-x_2\ge x_1\log(3/2)/K_B$.
Substitute in \eqref{eq:v105-reach-height}.
\end{proof}
No theorem here asserts that an arm reaching less than
$3\rho/4$ can be continued past its true physical boundary.
Its original positive source remains in the complementary class.
