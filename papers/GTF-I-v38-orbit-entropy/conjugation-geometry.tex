\section{Uniform geometry of Hermitian conjugation}\label{sec:conjugation}
Fix $q\ge2$ and let $V=\Herm(q)$ be the real space of traceless
Hermitian $q$ by $q$ matrices, with inner product $\tr(AB)$. Its dimension
is $D=q^2-1$. Unitary conjugation is orthogonal on this space.
The pure-state orbit has dimension $2(q-1)$, but normalized hidden
centroids can have arbitrary spectra. We therefore prove a bound on
\emph{every} unit sphere orbit.

\subsection{A Grassmann small-ball estimate}
Let $P_0$ be a rank-$k$ orthogonal projector in $\C^q$, $1\le k<q$,
and let $P$ be a Haar-distributed rank-$k$ projector. Put
$m=k(q-k)$ and
\begin{equation}\label{eq:grassmann-constant}
 b_{q,k}=\frac1{m!}\prod_{j=1}^k\frac{\Gamma(q-k+j)}{\Gamma(j)}.
\end{equation}
\begin{lemma}[Projector balls]\label{lem:grassmann}
For $0<R\le1$,
\begin{equation}\label{eq:grassmann-ball}
 \Pp(\norm{P-P_0}_{\HS}\le R)\le b_{q,k}R^{2k(q-k)}.
\end{equation}
\end{lemma}
\begin{proof}
By invariance take $P_0$ onto the first $k$ coordinates. Apart from a
null set, a Haar subspace is the graph of
$Z\in\C^{(q-k)\times k}$. Its density with respect to complex Euclidean
Lebesgue measure is
\begin{equation}\label{eq:chart-density}
 \pi^{-m}\prod_{j=1}^k\frac{\Gamma(q-k+j)}{\Gamma(j)}
             \det(I+Z^*Z)^{-q}.
\end{equation}
For completeness, take a $q$ by $k$ matrix of independent standard
complex Gaussian entries and split it into a square top block $A$ and
bottom block $B$. Its column space is Haar and $Z=BA^{-1}$ almost surely.
The change $B=ZA$ has real Jacobian $|\det A|^{2(q-k)}$.
Integrating $A$ with weight
$\exp[-\tr A^*(I+Z^*Z)A]$ gives the determinant factor $\det(I+Z^*Z)^{-q}$.
Complex Gaussian Gram--Schmidt gives
$\E|\det A|^{2(q-k)}=\prod_{j=1}^k\Gamma(q-k+j)/\Gamma(j)$,
since its squared orthogonal lengths have gamma shapes $1,\ldots,k$.
The powers of $\pi$ give \eqref{eq:chart-density}.

If $\theta_i$ are the principal angles, then
$\norm{P-P_0}_{\HS}^2=2\sum_i\sin^2\theta_i$ and
$\norm Z_{\HS}^2=\sum_i\tan^2\theta_i$. On the event in
\eqref{eq:grassmann-ball} each $\sin^2\theta_i\le R^2/2$, whence
\[
 \norm Z_{\HS}^2\le\frac{R^2}{2-R^2}\le R^2.
\]
The density \eqref{eq:chart-density} is bounded by its value at zero.
Multiply this value by the volume $\pi^m R^{2m}/m!$ of a real
$2m$-dimensional ball. This proves the assertion.
\end{proof}

Define the explicit finite constant
\begin{equation}\label{eq:Aq}
 A_q=\max_{1\le k<q}
 \left\{\max(1,b_{q,k})\,[9q(q-1)^2]^{k(q-k)}\right\}.
\end{equation}
It is rational and at least one; no optimality is claimed.
\begin{theorem}[All-spectra orbital anti-concentration]
\label{thm:all-spectra}
For all $A,B\in\Herm(q)$ of Hilbert--Schmidt norm one and all
$0<r\le1$,
\begin{equation}\label{eq:hermitian-ball}
 m_{\SU(q)}\{U:\norm{UAU^*-B}_{\HS}\le r\}
                        \le A_q r^{2(q-1)}.
\end{equation}
In particular, the estimate is uniform at repeated eigenvalues and over
mixed-state centroid directions. No lower bound on each eigenvalue gap
is assumed.
\end{theorem}
\begin{proof}
Let $\lambda_1\ge\cdots\ge\lambda_q$ be the eigenvalues of $A$.
Trace zero and $\sum_i\lambda_i^2=1$ imply
$\lambda_1-\lambda_q\ge q^{-1/2}$. Hence some $k$ has
\begin{equation}\label{eq:eigengap}
 \lambda_k-\lambda_{k+1}\ge\delta_q:=\frac1{\sqrt q(q-1)}.
\end{equation}
For two conjugates $C,D$ of $A$, let $P_C,P_D$ be their top-$k$
spectral projectors. Eigenvectors $u_i$ of $C$ and $v_j$ of $D$ obey
\[
 (\lambda_i-\lambda_j)\langle u_i,v_j\rangle
        =\langle u_i,(C-D)v_j\rangle.
\]
Summing squares for $i\le k<j$ and using equal projector ranks gives
\begin{equation}\label{eq:projector-stability}
 \norm{P_C-P_D}_{\HS}^2
 =2\sum_{i\le k<j}|\langle u_i,v_j\rangle|^2
 \le 2\delta_q^{-2}\norm{C-D}_{\HS}^2.
\end{equation}
If the event in \eqref{eq:hermitian-ball} is empty there is nothing to
prove. Otherwise choose one conjugate $D$ in that event. Every other
$C$ in the event has $\norm{C-D}_{\HS}\le2r$. The image $P_C$ is a
Haar rank-$k$ projector and lies within $2\sqrt2\delta_q^{-1}r$ of
$P_D$. Set $c_q=3\sqrt q(q-1)$, which is larger than this coefficient.
If $c_qr\le1$, Lemma~\ref{lem:grassmann} gives a bound
$b_{q,k}(c_qr)^{2k(q-k)}$. Since $k(q-k)\ge q-1$ and $r\le1$,
this is at most the right side of \eqref{eq:hermitian-ball}.
If $c_qr>1$, then $A_qr^{2(q-1)}\ge c_q^{2(q-1)}r^{2(q-1)}>1$,
so the trivial probability bound suffices. Haar conjugation under
$\SU(q)$ and under $U(q)$ has the same law, completing the proof.
\end{proof}
The proof isolates one uniformly separated spectral cut rather than
requiring a simple spectrum. This is why pure-state and highly mixed
centroids are handled by the same constant. The smallest possible
nontrivial Grassmann exponent is $2(q-1)$.

\begin{corollary}[Conjugation occupation]\label{cor:conjugation}
For conjugation commands satisfying a group gap $g_t$, any hidden
register with $q_N\ge\kappa>0$ obeys
\begin{equation}\label{eq:conjugation-budget}
 \sum_{K_t\le k}g_t
 \le C_q\kappa^{-3-1/(q-1)}k^{1/(q-1)},\qquad
 C_q=2048(q^2-1)(16A_q)^{1/(q-1)}.
\end{equation}
For a constant gap $g>0$ and wordwise error $\varepsilon$ with
$\kappa=\rho/\sqrt{q^2-1}-2\varepsilon>0$,
\begin{equation}\label{eq:q-lower}
 W_{N,\varepsilon}\ge
 \left(\frac{g\kappa^{3+1/(q-1)}}{C_q}\,N\right)^{q-1}.
\end{equation}
\end{corollary}
\begin{proof}
Apply Lemma~\ref{lem:transfer}, Theorem~\ref{thm:all-spectra} and
Theorem~\ref{thm:occupation}, then use Lemma~\ref{lem:terminal}.
\end{proof}
A group gap is stronger than a gap only on functions on the pure-state
homogeneous quotient. The proof genuinely needs uniform control of the
mixed-centroid directions, and does not silently replace this hypothesis
by the weaker quotient gap.
