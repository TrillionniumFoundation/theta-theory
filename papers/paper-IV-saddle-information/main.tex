\begin{abstract}
We extend the fixed-law endogenous theta-semigroup of Paper III to partially
observed and controlled models while preserving its deterministic origin.  A
noncompact refresh--autoregressive hidden coordinate is realized by a
deterministic product shift under a specified preparation measure.  Soft
observations give uniform filter contraction and an invariant posterior moment
ball, whereas exact pre-action observation obeys a sharp no-go theorem.

For smooth controlled theta-values, we prove a risk-sensitive saddle-envelope
theorem.  Under curvature compensation the local saddle is unique and smooth.
The first terminal derivative is expectation under the optimal tilted law.
The second derivative is not merely covariance: it is
\[
 \theta\operatorname{Cov}_{Q^*}-
 G_{\phi z}G_{zz}^{-1}G_{z\phi}.
\]
This optimizer-response correction closes the former type gap between a
controlled theta-game and the fixed-law tangent theory of Paper V.
\end{abstract}

\section{The deterministic product extension}
Let $B_a$ be the driven collision map of Papers I--III.  Independently, let a
bilateral shift carry a refresh bit and a Gaussian innovation coordinate.  Its
left shift is deterministic once the full initial sequence is fixed.  Under
the product preparation measure, the observed hidden state has transition
kernel
\[
 P_Y(y,\cdot)=\delta N(0,1)+(1-\delta)N(ry,1-r^2),
 \qquad 0<\delta<1,\quad\abs r<1.
\]
Thus stochastic filtering is a coarse conditional description of a
deterministic product system.

\section{Soft filtering}
Let $W(y)=1+y^2$ and let observations $o\in\{-1,1\}$ have likelihood
\[
 g_o(y)=\frac{1+o\epsilon\tanh y}{2},\qquad0<\epsilon<1.
\]
Write $g_\pm=(1\pm\epsilon)/2$.

\begin{theorem}[Noncompact filter packet]\label{thm:filter}
Let $C_B=2g_+/g_-$.  Choose $m_*$ so that
\[
 q_f=C_B(1-\delta)^{m_*}<1,
 \qquad
 \kappa_f=\frac{g_+}{g_-}\bigl((1-\delta)r^2\bigr)^{m_*}<1.
\]
Then one prediction-observation block contracts total variation by $q_f$ and
preserves every moment ball
\[
 \{\pi:\pi(W)\le R\},\qquad
 R\ge\frac{2g_+/g_-}{1-\kappa_f}.
\]
The bounds are uniform along common strategy and observation trees.
\end{theorem}
\begin{proof}
The common refresh component cancels in differences, giving prediction
contraction $1-\delta$.  The AR component gives
$P_YW\le\alpha W+2$ with $\alpha=(1-\delta)r^2$.  The Bayes map is
$C_B$-Lipschitz in total variation and multiplies the $W$ moment by at most
$g_+/g_-$.  Iterate.
\end{proof}

\begin{theorem}[Exact-observation obstruction]\label{thm:no-go}
If a centered nonzero fast payoff $h(z)$ is observed before choosing
$u\in\{-1,1\}$, then
\[
 \int\sup_u uh(z)\,d\mu=\int\abs h\,d\mu>0,
 \qquad
 \sup_u\int uh\,d\mu=0.
\]
Hence exact pre-action information cannot be replaced by blind averaging.
\end{theorem}

\section{Curvature compensation}
Let $z=(u,v)$ and let $G(\phi,z)$ be a one-step controlled free-energy
functional, concave in $u$ and convex in $v$.  Add actuator energies so that
for some $\lambda>0$,
\[
 -G_{uu}\ge\lambda I,
 \qquad G_{vv}\ge\lambda I.
\]

\begin{theorem}[Unique smooth saddle]\label{thm:saddle}
Assume $G$ is $C^3$, the saddle is interior, and the saddle Jacobian
$G_{zz}$ is invertible with the displayed uniform concave--convex bounds.
Then there is a unique saddle $z_*(\phi)$ and it is $C^2$ in terminal data.
\end{theorem}
\begin{proof}
The first-order saddle system $G_z(\phi,z)=0$ has a uniformly invertible
Jacobian.  Strong concavity--convexity gives global uniqueness and the implicit
function theorem gives smooth dependence.
\end{proof}

\section{The risk-sensitive saddle-envelope theorem}
Put
\[
 \mathcal V(\phi)=G(\phi,z_*(\phi)).
\]
Let $Q_\phi^*$ denote the exponential branch law at the optimal saddle.

\begin{theorem}[Saddle envelope and optimizer correction]\label{thm:envelope}
For terminal directions $\eta,\zeta$,
\[
 D\mathcal V(\phi)\eta
 =\E_{Q_\phi^*}\eta,
\]
and
\[
 D^2\mathcal V(\phi)(\eta,\zeta)
 =\theta\Cov_{Q_\phi^*}(\eta,\zeta)
 -G_{\phi z}[\eta]\,G_{zz}^{-1}\,G_{z\phi}[\zeta].
\]
The second term is the exact policy/saddle response.  It vanishes for a fixed
policy but not in general.
\end{theorem}
\begin{proof}
At the saddle $G_z=0$, so the first derivative is the partial terminal
derivative; differentiation of log-sum-exp gives the optimal tilted
probability.  Differentiate $G_z(\phi,z_*(\phi))=0$ to obtain
\[
 Dz_*[\zeta]=-G_{zz}^{-1}G_{z\phi}[\zeta].
\]
Differentiate the first envelope identity.  The pure terminal Hessian of
log-sum-exp is $\theta$ times covariance, and substitution gives the formula.
\end{proof}

\begin{corollary}[Correct tangent typing]\label{cor:typing}
The controlled risk-sensitive value has a probability-valued first tangent,
but its second tangent has pure covariance curvature only when the optimizer
response vanishes.  Therefore fixed-law tangent characterization and
controlled-game representation are distinct theorem branches.
\end{corollary}

\section{Controlled microscopic recursion}
With collision branches $i$ and controls selected before the branch, define
\[
 \mathcal T_\varepsilon F
 =\sup_u\inf_v\frac1\theta\log\sum_iw_i(a)
 e^{\theta[\varepsilon^2\tau_i c^{uv}
 +F(x+\varepsilon d_i+\varepsilon^2\tau_i b^{uv})]}.
\]

\begin{theorem}[Controlled theta-HJB]\label{thm:controlled-hjb}
Under the uniform saddle conditions and bounded Lipschitz coefficients, the
microscopic values converge to
\[
\begin{aligned}
 -u_t=\frac1{\bar\tau(a)}\sup_u\inf_v\Bigl\{
 &\bar\tau(a)(c^{uv}+b^{uv}\cdot Du)\\
 &+\frac12C(a):D^2u
 +\frac\theta2Du^TC(a)Du\Bigr\}.
\end{aligned}
\]
If the diffusion and entropic terms are control independent, the saddle acts
only on $c^{uv}+b^{uv}\cdot Du$.  In every uniform classical parabolic window, the first and second terminal
variations are typed by \cref{thm:envelope}; no differentiability claim is
made merely from viscosity convergence.
\end{theorem}
\begin{proof}
Monotonicity and stability are inherited from the one-step log partition.
Uniform smooth saddles give continuous selectors.  The expansion of Paper III
is uniform in controls; viscosity comparison yields convergence.  Apply
\cref{thm:envelope} at the discrete level and pass through the stable smooth
window.
\end{proof}

\section{Filtered extension}
The exact belief-state recursion uses state $(x,i,\pi)$, the collision branch,
the hidden prediction, and Bayes update.  Under \cref{thm:filter}, an initial
layer with a diverging number of filter blocks and vanishing physical duration
removes the initial belief.  Combining this with \cref{thm:saddle} yields the
partially observed physical theta-Isaacs equation on
\texttt{FB4-PE-v1}$\times$\texttt{AR-FILTER-v1}.

\section{Main theorem}
\begin{theorem}[Controlled and informational closure]\label{thm:main}
The partially observed extension has an exact deterministic product
realization, a positive soft-filter theorem, an exact-information no-go, a
unique pure saddle, and a complete first/second tangent calculus including the
optimizer-response correction.  No fixed-law covariance theorem is silently
reused for the controlled branch.
\end{theorem}
\begin{proof}
Combine \cref{thm:filter,thm:no-go,thm:saddle,thm:envelope,thm:controlled-hjb}.
\end{proof}

\section*{Scope and review status}
The theorem is a complete internal proof in the declared deterministic
full-cover collision scope.  It does not claim that an arbitrary periodic
Sinai billiard has the same exact symbolic reset or that independent external
review has occurred.
