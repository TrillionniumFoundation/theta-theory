\section{Referee-round positive closure: compensated blocks and three distinct games}
\label{sec:rr2-c1}

\subsection{Coboundary-compensated block preparation}

Let \(m_\epsilon\to\infty\) and partition the kinetic horizon into blocks.
The raw slow-variation error has a boundary part that must be telescoped.

\begin{lemma}[Compensated block error]
\label{lem:rr2-c1-block}
There exists a uniformly bounded corrector \(\chi^{u,v}\) such that the error
on block \(j\) has the decomposition
\[
 R_j^\epsilon
 =
 \epsilon m_\epsilon
 \big(\chi^{u_j,v_j}(X_{j+1})-\chi^{u_j,v_j}(X_j)\big)
 +O(\epsilon^2m_\epsilon^3)
 +O(\rho_*^{m_\epsilon}).
\]
If controls are held fixed on adjacent preparation blocks or pay a switching
cost controlling the corrector jump, then over a unit slow-time horizon
\[
 \sum_jR_j^\epsilon
 =
 O(\epsilon m_\epsilon)
 +O(\epsilon m_\epsilon^2)
 +O\!\left(\frac{\rho_*^{m_\epsilon}}
 {\epsilon m_\epsilon}\right).
\]
Hence the total error vanishes under
\[
 \epsilon m_\epsilon^2\to0,\qquad
 \rho_*^{m_\epsilon}/(\epsilon m_\epsilon)\to0.
\]
\end{lemma}

\begin{proof}
Solve the centered Poisson equation for the frozen prepared kernel on the
uniform spectral atlas.  Taylor expansion of the slow state over one block
gives the boundary coboundary plus a second-order remainder
\(O(\epsilon^2m_\epsilon^3)\).  The spectral remainder is
\(O(\rho_*^{m_\epsilon})\).  Summing the coboundary telescopes, leaving only
the two horizon endpoints and controlled switching jumps.  There are
\(O((\epsilon m_\epsilon)^{-1})\) blocks, giving the displayed total.
\end{proof}

\subsection{One-time preparation game}

Players choose \(u,v\) once at time zero, selecting a prepared phase
\(\nu^{u,v}\), and then evolution is uncontrolled.

\begin{theorem}[Outer preparation saddle]
\label{thm:rr2-c1-outer}
The value is
\[
 V^{\rm prep}_{s,t}(F)
 =
 \sup_u\inf_v\,
 \mathcal E_{s,t}^{\nu^{u,v}}(F).
\]
It satisfies time consistency only after the preparation label \((u,v)\) is
retained in the state.  No pointwise Isaacs maximization appears in its
generator.
\end{theorem}

\begin{proof}
The choice is made once, so conditioning at an intermediate time does not
permit reselection.  Apply the fixed-phase tower for each \((u,v)\), then
take the outer saddle.  Retaining the label makes the family Markov; removing
it mixes different kernels and destroys the tower.
\end{proof}

\subsection{Reward-only control on a fixed phase}

Here controls change only running and terminal work, not the transition law.

\begin{theorem}[Fixed-law risk-sensitive Isaacs equation]
\label{thm:rr2-c1-reward}
For compact actions, continuous rewards, and the fixed prepared generator
\(\mathcal L^\nu\), the value is the unique viscosity solution of
\[
 \partial_tU+\mathcal L^\nu U+
 \sup_u\inf_v
 \left\{c^{u,v}
 +\frac{\theta}{2}
 \langle DU,D^\nu DU\rangle\right\}=0.
\]
\end{theorem}

\begin{proof}
The exact log-Laplace backward recursion holds with one fixed transition
kernel.  Measurable selection follows from compactness and continuity.
The diffusion tangent of A4/B4 gives consistency; monotonicity, stability,
and comparison yield the viscosity limit.
\end{proof}

\subsection{Genuinely adaptive law control}

Now actions change the kinetic generator itself and may be reselected
nonanticipatively.  Let \(H^{u,v}\) be the controlled Hamiltonian, and include
a switching cost that controls the corrector jumps in
Lemma~\ref{lem:rr2-c1-block}.

\begin{theorem}[Adaptive law-control Isaacs equation]
\label{thm:rr2-c1-adaptive}
Under uniform spectral/cluster bounds over the compact action family and
nonanticipative Elliott--Kalton strategies, the block values converge to the
unique viscosity solution
\[
 \partial_tU+
 \sup_u\inf_v
 H^{u,v}\!\left(f,\frac{\delta U}{\delta f}\right)=0.
\]
This theorem concerns a different game from
Theorem~\ref{thm:rr2-c1-outer}.
\end{theorem}

\begin{proof}
Freeze controls over blocks, apply the controlled preparation and generator
estimates, and use Lemma~\ref{lem:rr2-c1-block} to sum errors.  The switching
cost makes the corrector boundary terms tight.  Dynamic programming for
nonanticipative strategies gives monotone discrete schemes.  Uniform
compact containment and comparison for the controlled Hamiltonian yield the
limit.
\end{proof}

\subsection{Specified observation model and filter contraction}

Let observations have density \(g(y\mid h)\) relative to a fixed measure.
At independent refresh times of intensity \(\lambda\), the observation
contains a component \(\varepsilon_0\nu_0(dy)\) independent of the hidden
history.  Assume a Lyapunov drift for a history moment \(V\).

\begin{theorem}[Posterior-history stability]
\label{thm:rr2-c1-filter}
The posterior history process is Feller on
\(\mathcal P_V(\mathsf H)\), leaves a uniform \(V\)-moment ball invariant,
and satisfies
\[
 \|\Pi_t^\mu-\Pi_t^{\tilde\mu}\|_{\rm TV}
 \le Ce^{-\gamma t}\|\mu-\tilde\mu\|_V .
\]
The posterior is a sufficient state for all bounded future work.
\end{theorem}

\begin{proof}
The refresh component gives a common minorizing observation event after
Bayes normalization.  Coupling at the first successful refresh coalesces the
filters; its tail is exponential.  The Lyapunov drift controls excursions
between refreshes and proves invariance of the moment ball.  Feller
continuity follows from dominated continuity of the prediction and Bayes
operators.  Sufficiency is the defining disintegration identity.
\end{proof}

\subsection{Saddle curvature}

\begin{theorem}[Typed second variation]
\label{thm:rr2-c1-hessian}
For a smooth interior continuous-action saddle,
\[
 D^2W
 =
 G_{\phi\phi}-G_{\phi z}G_{zz}^{-1}G_{z\phi}.
\]
For finite actions the corresponding statement is a directional
subdifferential formula over active saddles, not a Hessian.
\end{theorem}

\begin{proof}
For continuous actions, differentiate the first-order saddle equations and
substitute the implicit derivative of the optimizer.  For finite actions,
the value is a finite upper/lower envelope of smooth branches; standard
convex directional calculus gives the active-branch subdifferential.
\end{proof}
