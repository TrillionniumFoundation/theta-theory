\begin{abstract}
We type the decision and information extensions of the kinetic cotangent
theory.  Preparation decisions occur before the deterministic hard-sphere
evolution; path-work decisions evaluate a fixed prepared phase.  Optimized
second variations contain an explicit optimizer-response Schur complement and
therefore are not pure covariances.  The exact information state is a posterior
law on future empirical histories.  Complete microscopic observation removes
all stochastic uncertainty.  Adaptive controls changing the collision law are
excluded from the theorem.
\end{abstract}

\section{Typed decision interfaces}
We distinguish:
\begin{enumerate}[label=(\roman*)]
\item a fixed prepared law, selecting one cotangent phase;
\item a preparation decision, chosen before hard-sphere evolution;
\item a bounded path-work decision on a fixed prepared ensemble.
\end{enumerate}
No feedback in this paper changes sphere diameter, elastic scattering,
particle-number scaling or microscopic geometry during the Boltzmann--Grad
limit.

For preparation parameter $z$ and terminal work $\phi$, let
\[
 G(\phi,z)=\frac1\vartheta
 \log\E_{\mathbb P_z}e^{\vartheta\phi}+k(z),
 \qquad
 V(\phi)=\sup_u\inf_vG(\phi,z(u,v)).
\]
The fixed-law scalar branch is compatible with entropic rigidity
\cite{KupperSchachermayer2009}; optimization is typed outside the log-Laplace
functional.

\section{Risk-sensitive saddle envelope}
\begin{theorem}[Saddle envelope and optimizer response]\label{thm:saddle}
Suppose $G$ is twice Fr\'echet differentiable, the saddle $z^*(\phi)$ is
unique, and $G_{zz}$ is invertible with the appropriate saddle coercivity.
Then
\[
 DV(\phi)[h]=\E_{Q_{\phi,z^*}}h,
\]
and
\[
 \boxed{
 D^2V(\phi)[h,k]
 =\vartheta\Cov_{Q_{\phi,z^*}}(h,k)
 -G_{\phi z}[h]G_{zz}^{-1}G_{z\phi}[k].}
\]
\end{theorem}

\begin{proof}
Differentiate $G_z(\phi,z^*(\phi))=0$ to obtain
$Dz^*=-G_{zz}^{-1}G_{z\phi}$.  The first derivative is the envelope theorem;
a second derivative gives the Schur complement.  At fixed $z$, the
log-Laplace Hessian is $\vartheta$ times covariance.
\end{proof}

Thus optimized values are not silently identified with the fixed-law
semigroup of Paper III.

\section{Information states and deterministic no-go}
Let $\mathcal Y_t$ be generated by coarse empirical moments, collision counts
or noisy measurements.  The exact sufficient statistic is
\[
 \Pi_t^{\rm post}
 =\operatorname{Law}\left(
 (\pi^\varepsilon,\Gamma^\varepsilon)_{[t,T]}
 \mid\mathcal Y_t\right).
\]

\begin{proposition}[Posterior-history sufficiency]\label{prop:posterior}
Every bounded future path-work value is a measurable functional of
$\Pi_t^{\rm post}$, and the posterior kernel satisfies the Bayesian tower.  A
finite-dimensional filter requires a separate approximation theorem.
\end{proposition}

\begin{theorem}[Complete-observation no-go]\label{thm:nogo}
If the complete hard-sphere microstate is observed before a decision, then the
future microscopic trajectory is deterministic almost surely.  Conditional
variance or filtering noise at that information level is zero.
\end{theorem}

\begin{proof}
The hard-sphere flow is deterministic on a full Liouville-measure set, so the
future is a measurable function of the observed state.
\end{proof}

\section{Controlled kinetic Hamiltonians}
For a fixed finite set of preparation choices define
\[
 \mathbb H^{u,v}(f,p)
 =\mathbb H_{\Theta(z(u,v))}(f,p)+c^{u,v}(f).
\]
On a unique smooth saddle branch,
\[
 \partial_tU+\sup_u\inf_v
 \mathbb H^{u,v}\left(f,\frac{\delta U}{\delta f}\right)=0.
\]

\begin{theorem}[Typed controlled closure]\label{thm:main}
The equation is valid for initial preparation games and bounded path-work games
on already constructed kinetic phases.  Its first and second variations are
given by \cref{thm:saddle}.  Adaptive control of the microscopic collision law
is outside the theorem and supplies no load-bearing input.
\end{theorem}
