
\section{Round-twenty-seven reconstruction: deterministic mechanics with an explicit noisy observation channel}
\label{sec:r27-c1}

A complete mechanical state has deterministic future and deterministic
noiseless observables.  The dominated observation density in this paper comes
from an explicit independent measurement channel, not from relabelling an
aggregate local limit theorem as a pointwise conditional density.  This
choice makes the hidden/observation model internally consistent and provides
the positivity needed for recursive likelihood differentiation.

\subsection{The consistent partially observed model}

Let $X$ be the complete prepared mechanical/history state, $A$ a compact
action space, and $\Theta\subset\mathbb R^d$ compact.  The hidden transition is
\[
 M_\vartheta^a(x,dx')
 =\delta_{\Phi_\vartheta^a(x)}(dx').                      \tag{C1.1}
\]
For each observation stratum $s$ let
$\mathcal O_{\vartheta,s}^a:X\to E_s$ be a continuous deterministic signal
map.  The measurement noise $\xi_s$ is independent of the hidden state and
has a $C^3$ density $\rho_s$ with respect to a probability measure $\nu_s$.
The observation is
\[
 Y=\mathcal O_{\vartheta,s}^a(x)+\xi_s
\]
on Euclidean strata, with the analogous convolution or confusion matrix on
surface and countable strata.  Put
\[
 \nu=\sum_s2^{-|s|}\nu_s,\qquad
 g_\vartheta^a(x,y)
 =\sum_s2^{|s|}1_{E_s}(y)
   \rho_s\bigl(y-\mathcal O_{\vartheta,s}^a(x)\bigr).      \tag{C1.2}
\]
Thus $\nu$ is a finite probability envelope and
$\int g_\vartheta^a(x,y)\nu(dy)=1$.

On diagnostic strata assume
\[
 0<c_K\le\rho_s(y)\le C_K,\qquad
 |\partial^\alpha\log\rho_s(y)|\le C_K,\quad|\alpha|\le3,               \tag{C1.3}
\]
on every compact signal/noise region reached by the experiment.  Global
filter continuity only uses $L^1$ continuity and does not require (C1.3).

\begin{proposition}[Model-derived channel regularity]
\label{prop:r27-c1-channel}
On every prepared compact shell,
\[
 (\vartheta,a,x)\longmapsto g_\vartheta^a(x,\cdot)
\]
is continuous in $L^1(\nu)$ and is quadratically mean differentiable through
order two.  On diagnostic strata it has fixed support and three integrable
log-derivative envelopes.  Both deterministic hidden transition and smooth
observation density therefore hold without contradiction.
\end{proposition}

\begin{proof}
Translation is continuous in $L^1$ for every probability density.  Chain
rule through the $C^3$ signal map gives the derivatives.  The noise envelopes
and compact signal range provide domination.  Strict positivity in (C1.3)
fixes support and permits logarithmic differentiation.  The randomness is
measurement noise independent of the complete deterministic state.
\end{proof}

\subsection{Integrated Feller filtering}

For a belief $\pi$ set
\[
 \pi^-=(\Phi_\vartheta^a)_\#\pi,\qquad
 r_\vartheta(\pi,a,y)=\int g_\vartheta^a(x,y)\pi^-(dx),                  \tag{C1.4}
\]
and
\[
 N_\vartheta^\varphi(\pi,a,y)
 =\int\varphi(x)g_\vartheta^a(x,y)\pi^-(dx).
\]
Where $r>0$, define
$B_\vartheta(\pi,a,y)(\varphi)=N^\varphi/r$; its value at $r=0$ is arbitrary.
The belief transition is
\[
 Q_\vartheta^aF(\pi)
 =\int F(B_\vartheta(\pi,a,y))
       r_\vartheta(\pi,a,y)\nu(dy).                       \tag{C1.5}
\]

\begin{theorem}[Evidence-stable Feller filter]
\label{thm:r27-c1-feller}
For bounded continuous $F$, the map
$(\vartheta,a,\pi)\mapsto Q_\vartheta^aF(\pi)$ is continuous on every compact
reachable belief shell.
\end{theorem}

\begin{proof}
Weak convergence of predictions follows from continuity of $\Phi$.
Proposition \ref{prop:r27-c1-channel} gives $L^1$ convergence of evidences
and unnormalised numerators.  For bounded $\varphi$,
\[
 r_n\left|{N_n^\varphi\over r_n}-{N^\varphi\over r}\right|
 \le|N_n^\varphi-N^\varphi|+\|\varphi\|_\infty|r_n-r|.    \tag{C1.6}
\]
Sum this inequality over a countable bounded-Lipschitz determining family.
It gives convergence of posteriors integrated against their evidence, even
where the limiting evidence is small or zero.  Uniform continuity of $F$ on
the compact reachable shell proves (C1.5).
\end{proof}

\subsection{A forward-invariant finite-coordinate belief model}

Let $\mathcal K$ be a compact reachable belief set.  Its closed convex hull
$\overline{\operatorname{co}}\mathcal K$ is used as the regularity shell; the
deterministic prediction and noisy Bayes kernel extend continuously to it by
linearity and Theorem \ref{thm:r27-c1-feller}.  Choose a finite
bounded-Lipschitz family $\varphi_1,\ldots,\varphi_m$ and an $\epsilon$-net
$\pi_1,\ldots,\pi_J$ in the convex shell.  Define
\[
 T_m\pi=(\pi\varphi_1,\ldots,\pi\varphi_m),\qquad
 R_mz=\sum_i\psi_i(z)\pi_i,                               \tag{C1.7}
\]
where $\psi_i$ is a partition of unity subordinate to coordinate balls.

\begin{proposition}[Invariant finite-coordinate approximation]
\label{prop:r27-c1-finite}
The net and functions can be chosen so that
\[
 \sup_{\pi\in\overline{\operatorname{co}}\mathcal K}
 d_{\rm BL}(R_mT_m\pi,\pi)\le3\epsilon.                   \tag{C1.8}
\]
The approximate filter that applies $R_m$ after each update remains in the
same convex regularity shell, and finite-horizon values converge uniformly.
\end{proposition}

\begin{proof}
A finite separating family distinguishes the finite net with a fixed margin.
Every active partition weight corresponds to a net point within
$3\epsilon$; convexity of $d_{\rm BL}$ gives (C1.8).  Forward invariance follows
because $R_mz$ is a convex combination of shell points.  Iterate the Feller
modulus for a fixed horizon.
\end{proof}

\subsection{Risk-sensitive dynamic programming}

For bounded continuous running reward $c$ and terminal reward $\Psi$, define
\[
 (\mathcal T_\vartheta V)(\pi)
 =\inf_{a\in A}\int e^{c(\pi,a,y)}
 V(B_\vartheta(\pi,a,y))r_\vartheta(\pi,a,y)\nu(dy).       \tag{C1.9}
\]
Compactness of $A$ and Theorem \ref{thm:r27-c1-feller} give continuous values
and Borel selectors.  No hidden-state dominating measure is used.

\subsection{Derivative filter and noncircular excitation}

On the diagnostic experiment, (C1.3) gives $r_\vartheta(\pi,a,y)>0$.  Write
the unnormalised update as the positive linear operator
\[
 \mathsf G_{\vartheta,a,y}\mu(dx')
 =g_\vartheta^a(x',y)(\Phi_\vartheta^a)_\#\mu(dx').        \tag{C1.10}
\]
Its first three derivatives are bounded between weighted signed-measure
spaces.  Normalize only after differentiating:
\[
 D B={D\mathsf G\,\pi\over r}
      -B\,{D r\over r}.                                   \tag{C1.11}
\]

\begin{lemma}[Recursive QMD]
\label{lem:r27-c1-qmd}
The filter, predictive evidence, and their first two parameter derivatives
are stable in $L^2$ uniformly over the diagnostic action class.  The
one-step square-root likelihood expansion composes to a uniform local
quadratic-mean expansion for the observation experiment.
\end{lemma}

\begin{proof}
Differentiate (C1.10) in the signed-measure graph norm and use the positive
lower bound in (C1.3) only on the diagnostic strata.  The integrated Feller
contraction bounds propagation of old derivative errors; the derivative
envelopes bound new ones.  Induction gives the first two derivatives and a
summable third-order remainder.  Applying the same recursion to square roots
proves QMD.
\end{proof}

The admissible policy class is specified before Fisher information is
computed.  Let $A_1,\ldots,A_J$ be diagnostic actions and require that every
policy use action $A_j$ at least $\rho_jn+o(n)$ times, with
$\rho_j>0$.  Define the deterministic observability Gramian
\[
 \mathcal G(\pi,\vartheta)
 =\sum_j\rho_j
 E_{\vartheta,\pi}^{A_j}[s_js_j^\mathsf T].               \tag{C1.12}
\]
Assume the finite, checkable signal/noise condition
$\mathcal G(\pi,\vartheta)\ge cI_d$ on the convex shell.

\begin{theorem}[Derived persistent information]
\label{thm:r27-c1-info}
Every policy satisfying the action-frequency condition obeys
\[
 {1\over n}\sum_{k\le n}
 E[s_ks_k^\mathsf T\mid\mathcal F_{k-1}]
 \ge\tfrac12cI_d+o_P(1),                                  \tag{C1.13}
\]
and the upper bound and conditional Lindeberg condition are uniform.
\end{theorem}

\begin{proof}
The Feller filter and compact shell make the information contribution
uniformly continuous in the belief.  Group times by diagnostic action and use
their required frequencies.  The ergodic/martingale law of large numbers and
(C1.12) give (C1.13).  The score envelope from (C1.3) gives the upper and
Lindeberg bounds.  Thus the policy class is not defined by assuming the
desired information conclusion.
\end{proof}

\subsection{Adaptive LAN and full-annulus contraction}

Let
\[
 \Delta_n=n^{-1/2}\sum_{k\le n}s_k,\qquad
 I_n=n^{-1}\sum_{k\le n}E[s_ks_k^\mathsf T\mid\mathcal F_{k-1}].
\]

\begin{theorem}[Uniform adaptive LAN and Bernstein--von Mises]
\label{thm:r27-c1-bvm}
Uniformly over the declared policy class,
\[
 \log{dP_{\vartheta_0+h/\sqrt n}\over dP_{\vartheta_0}}
 =h^\mathsf T\Delta_n-\tfrac12h^\mathsf TI_nh+o_P(1)       \tag{C1.14}
\]
on every bounded $h$-set, and $\Delta_n\Rightarrow N(0,I)$ along every
subsequence with $I_n\to I$.  For a prior with positive continuous density at
$\vartheta_0$, the posterior of $\sqrt n(\vartheta-\vartheta_0)$ converges in
total variation to
\[
 N(I^{-1}\Delta_n,I^{-1}).                                \tag{C1.15}
\]
\end{theorem}

\begin{proof}
Lemma \ref{lem:r27-c1-qmd}, Theorem \ref{thm:r27-c1-info}, and the martingale
CLT give (C1.14).  For posterior contraction, split alternatives into:
fixed distance, annuli $M/\sqrt n\le|\vartheta-\vartheta_0|\le\delta$, and
the local ball.  Hellinger tests from the positive noisy channel are
exponentially consistent on the first set.  On the growing annuli, cover by
$O((\sqrt n/M)^d)$ local balls and use the quadratic information lower bound
to obtain error $e^{-cM^2}$; then let $M\to\infty$.  LAN controls the local
ball and Gaussian tails complete the total-variation argument.
\end{proof}

The observation randomness, finite envelope, fixed support, recursive
derivatives, invariant coordinate approximation, derived excitation, and
growing-annulus tests are now explicit parts of one consistent experiment.
