\section{Round-thirteen positive closure: disintegrated belief kernels and growing-moment reduction}
\label{sec:r13-c1}

The exact information state is a probability law, not an oriented current.
Currents retain collision and observation geometry, while positive Rokhlin
slices supply evidence and posterior probabilities.  No fixed finite list of
moments is declared sufficient: the reduced coordinate order grows with the
Boltzmann--Grad scale and has an explicit approximation error.

\subsection{Positive stratified disintegration}

Let $X$ be the graph-completed microscopic/history state and let
$O_u:X\to Y_u$ be one of a finite family of $C^2$ observation maps.  On every
Whitney stratum $S$, $O_u|_S$ has constant rank.  For a positive finite measure
$\nu$ whose stratum densities belong to the declared weighted Sobolev class,
coarea/Rokhlin disintegration gives
\[
 \nu(dx)=\int_{Y_u}\nu_{u,y}(dx)Z_\nu^u(y)\,dy,
 \qquad
 \nu_{u,y}(X)=1
\]

after omitting the zero-density fibers.  The oriented slice current and the
positive coefficient measure $Z_\nu^u(y)\nu_{u,y}$ are stored separately.

Define the observation kernel as the probability measure on
$Y_u\times\mathcal P_W(X)$
\[
 \mathbf O_u(\nu)(dy,d\pi)
 =Z_\nu^u(y)\,dy\,\delta_{\nu_{u,y}}(d\pi).
\]

\begin{theorem}[Feller disintegrated observation kernel]
\label{thm:r13-c1-feller}
On a compact regular stratum chart with uniform submersion, density, and
weighted-tail bounds, the map
$\nu\mapsto\mathbf O_u(\nu)$ is weakly continuous.  Hence, for every bounded
continuous posterior payoff $\Phi$,
\[
 \nu\longmapsto
 \int_{Y_u}\Phi(\nu_{u,y})Z_\nu^u(y)\,dy
\]

after including the observed reward is continuous.  Critical-rank strata are
included as separate coarea components.
\end{theorem}

\begin{proof}
Use a finite family of constant-rank charts and a partition of unity.  In each
chart the coarea formula writes the joint observation/posterior integral as an
ordinary integral over graph coordinates with a uniformly bounded Jacobian.
Weak convergence plus the uniform Sobolev/tail bounds gives convergence of
these integrals for a determining algebra on
$Y_u\times\mathcal P_W(X)$.  Tightness upgrades it to weak convergence of the
joint kernel.  Summing the strata proves the theorem.  No pointwise continuity
of an arbitrarily chosen version of $y\mapsto\nu_{u,y}$ is asserted.
\end{proof}

\subsection{The evidence cone}

Let $\mathcal K_M\subset\mathcal P_W(X)$ be a compact weighted belief set.
The compactified unnormalized state is
\[
 \widehat{\mathfrak B}_M
 =\{(r,\pi):0\le r\le1,\ \pi\in\mathcal K_M\}/\sim,
\]
where $(r,\pi)$ represents the positive measure $r\pi$, and at $r=0$ the
projective direction $\pi$ is retained.  On each regular blow-up chart the
direction is the normalized leading coarea coefficient.

\begin{proposition}[Continuous zero-evidence lift]
\label{prop:r13-c1-zero}
Prediction followed by the joint kernel of
Theorem~\ref{thm:r13-c1-feller} extends continuously to
$\widehat{\mathfrak B}_M$.  A zero-evidence observation has zero probability;
its retained direction is used only to make subsequent unnormalized
transitions continuous and does not alter any expected value.
\end{proposition}

\begin{proof}
Write the unnormalized sliced measure as
$r^kG(
u,u,y)+o(r^k)$ in a fixed constant-rank blow-up chart and normalize the
first nonzero positive coefficient.  The regularity assumptions make this
coefficient unique and continuous.  If all coefficients vanish, use the
predicted direction, which is likewise continuous.  Since the observation
kernel weights the state by its evidence $r^k$, the chosen zero-mass direction
does not change the integrated transition.  The joint-kernel continuity then
proves the extension.
\end{proof}

\subsection{Exact full-belief dynamic programming}

Let $\mathfrak B$ be the space of complete posterior laws.  A control first
chooses a canonical actual-contact likelihood as in C1's microscopic game,
then the physical law is propagated, and finally the new observation is
disintegrated by $\mathbf O_u$.

\begin{theorem}[Exact belief-state DPP]
\label{thm:r13-c1-dpp}
The prediction--observation kernels and the risk-sensitive reward define a
Feller controlled Markov kernel on $\widehat{\mathfrak B}_M$.  Nonanticipative
relaxed strategies satisfy the exact dynamic programming principle.  The
one-time microcanonical preparation variable is chosen once and retained as a
static coordinate; it is not reoptimized after every observation.
\end{theorem}

\begin{proof}
Ionescu--Tulcea constructs the controlled path law from the prediction kernel
and the joint disintegration kernel.  Theorem~\ref{thm:r13-c1-feller} and
Proposition~\ref{prop:r13-c1-zero} give Feller continuity and compact
containment.  Conditioning at an observation time and concatenating relaxed
strategies gives the DPP.  The initial shell multiplier belongs to the initial
law and therefore remains fixed under conditional updates.
\end{proof}

\subsection{A quantitative growing-moment reduction}

For a belief $\pi$, let $M_K(\pi)$ be all mixed moments of a fixed separating
countable observable family of total degree at most $K$.  Suppose
$\int e^{\alpha W}d\pi\le M$ on the reachable class.  Let
$K_\varepsilon\to\infty$ and $R_\varepsilon\to\infty$ satisfy
\[
 {R_\varepsilon\over K_\varepsilon}	o0,
 \qquad e^{-\alpha R_\varepsilon}=o(1).
\]
For example, take
$R_\varepsilon=\log\mu_\varepsilon$ and
$K_\varepsilon=(\log\mu_\varepsilon)^2$.

\begin{lemma}[Quantitative moment approximation]
\label{lem:r13-c1-moments}
If two beliefs in the reachable exponential-moment class have moment vectors
$M_{K_\varepsilon}$ differing by $o(K_\varepsilon^{-d})$ in the normalized
coordinates, then
\[
 W_{1,W}(\pi,\widetilde\pi)
 \le C{R_\varepsilon\over K_\varepsilon}
    +Ce^{-\alpha R_\varepsilon}+o(1).
\]
Thus a growing finite coordinate, not a fixed finite cumulant list, is
asymptotically determining.
\end{lemma}

\begin{proof}
Truncate to the compact set $\{W\le R_\varepsilon\}$.  Multivariate Bernstein
polynomials approximate every unit weighted-Lipschitz test there with error
$CR_\varepsilon/K_\varepsilon$.  Its polynomial integral is determined by the
moments through degree $K_\varepsilon$, and the assumed moment mismatch gives
the $o(1)$ term.  Exponential moments bound both tails by
$Ce^{-\alpha R_\varepsilon}$.  Kantorovich duality yields the result.  Since
moments and cumulants are triangular polynomial transforms at every finite
order, either coordinate system may be used.
\end{proof}

\begin{theorem}[Reduced DPP with a controlled error]
\label{thm:r13-c1-reduction}
Assume the B2 source chart gives the uniform exponential-moment bound on every
belief reachable by the declared controls, and choose
$K_\varepsilon$ so that the one-step error in
Lemma~\ref{lem:r13-c1-moments} is $o(J_\varepsilon^{-1})$, where
$J_\varepsilon$ is the number of decision stages.  Then the full-belief value
and the value on
$(r,M_{K_\varepsilon}(\pi))$ differ by $o(1)$ uniformly over admissible
strategies.
\end{theorem}

\begin{proof}
Couple one prediction--observation step through the common microscopic law and
apply Lemma~\ref{lem:r13-c1-moments} conditionally.  Bounded Lipschitz rewards
and the Feller kernel make the Bellman operator Lipschitz in weighted
Wasserstein distance.  Induction over the stages sums at most
$J_\varepsilon$ one-step errors.  The chosen growth of
$K_\varepsilon$ makes the total $o(1)$.  No claim is made that a fixed finite
moment vector determines the law.
\end{proof}

\subsection{Inserted coefficients and regular statistical limits}

Augment the B1 continuous constraint vector by a finite regular observation
statistic.  Exact number extraction is performed first.

\begin{theorem}[Inserted observation coefficient]
\label{thm:r13-c1-coefficient}
On a compact regular observation/source chart, the numerator with one bounded
posterior cylinder insertion and the denominator satisfy the same exact-number
mixed local theorem.  Their ratio converges to the tilted conditional
expectation, uniformly on regular observation fibers.  The large-frequency
bound is the integrable canonical coefficient theorem of B1, not a
full-grand-canonical atom-free assertion.
\end{theorem}

\begin{proof}
The insertion changes only the analytic amplitude surrounding the canonical
saddle.  B1's coefficient-first Fourier theorem supplies the common Gaussian
phase and an integrable relative remainder.  The leading Riesz/amplitude
coefficient in the numerator is the conditional tilted expectation; division
by the positive denominator proves the ratio.
\end{proof}

Let $\vartheta\in\mathbb R^d$ be a finite-dimensional identifiable phase
parameter.  Restrict strategies to the uniformly informative class
\[
 {1\over J}\sum_{j<J}
 \mathcal I_j(\vartheta_0\mid\mathcal F_{j-1})
 \ge cI_d
\]

after stopping on a compact regular chart.

\begin{theorem}[Controlled LAN and Bernstein--von Mises]
\label{thm:r13-c1-bvm}
For a prior with a positive continuous density near $\vartheta_0$, the exact
finite-centered log likelihood is uniformly LAN over the informative strategy
class.  Away from phase/coarea boundaries, the posterior of
$\sqrt{\mu_\varepsilon}(\vartheta-\vartheta_0)$ converges in total variation
to the Gaussian with the random efficient mean and inverse predictable
information.  No claim is made for uninformative strategies or coexistence
boundaries.
\end{theorem}

\begin{proof}
The inserted coefficient theorem gives a uniform third-order expansion of the
conditional log likelihood and exact normalization.  The predictable score is
a martingale; B3 supplies its joint Gaussian limit, and the uniform information
lower bound gives quadratic coercivity.  Exponential tests outside a shrinking
neighborhood give posterior tightness.  Taylor expansion of the prior and
Le Cam's completion-of-squares argument yield total-variation convergence on
the regular chart.
\end{proof}

\begin{theorem}[Round-thirteen C1 closure]
\label{thm:r13-c1-main}
The hard-sphere information problem has an exact Feller full-belief DPP, a
positive stratified disintegration with a continuous zero-evidence cone, a
quantitative growing-moment reduction, a number-first inserted coefficient,
and a controlled LAN/Bernstein--von Mises theorem on the uniformly informative
regular strategy class.
\end{theorem}

\begin{proof}
Combine Theorems~\ref{thm:r13-c1-feller},
\ref{thm:r13-c1-dpp}, \ref{thm:r13-c1-reduction},
\ref{thm:r13-c1-coefficient}, and \ref{thm:r13-c1-bvm}.
\end{proof}
