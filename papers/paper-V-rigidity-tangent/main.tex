\begin{abstract}
We prove that the theta-semigroup constructed from deterministic path
conditioning in Papers I--III is also the unique dynamically consistent
certainty equivalent in its law-invariant cash-additive class.  A direct
finite-state rigidity argument shows that a continuous strictly increasing
certainty-equivalent generator compatible with cash translations is either
affine or exponential.  Relevance and nonlinearity select the exponential
case.  Since the path large deviation principle already fixes
$\theta=I'(a)$, the usual free parameter of the axiomatic theorem is removed.

For the fixed-law and fixed-policy branches, first terminal derivatives are
expectations under exponential tangent laws, second derivatives are theta
covariances, and tangent kernels form a cocycle.  Microscopic collision tangent
laws converge to the physical diffusion tangent law.  The continuum density
is a genuine stochastic exponential and gives Girsanov and quadratic BSDE
representations.  The controlled branch imports Paper IV and carries its
optimizer-response correction rather than a false pure-covariance identity.
\end{abstract}

\section{A direct certainty-equivalent rigidity theorem}
Let $u:\R\to\R$ be continuous and strictly increasing.  On every finite
probability space define
\[
 \mathfrak C_u(X)=u^{-1}(\E u(X)).
\]

\begin{theorem}[Cash-additive rigidity]\label{thm:rigidity}
Assume
\[
 \mathfrak C_u(X+c)=\mathfrak C_u(X)+c
\]
for every bounded $X$ and $c\in\R$.  Then, up to an affine change of generator,
either
\[
 u(x)=x
\]
or
\[
 u(x)=e^{\theta x}
\]
for one nonzero constant $\theta$.
\end{theorem}
\begin{proof}
For fixed $c$ define on the interval $u(\R)$
\[
 F_c(y)=u(u^{-1}(y)+c).
\]
Apply cash additivity to every two-point random variable.  For all
$y_1,y_2$ and $p\in[0,1]$,
\[
 F_c(py_1+(1-p)y_2)=pF_c(y_1)+(1-p)F_c(y_2).
\]
Continuity makes $F_c$ affine:
$F_c(y)=A(c)y+B(c)$ with $A(c)>0$.  Therefore
\[
 u(x+c)=A(c)u(x)+B(c).
\]
The translation group gives
$A(c+d)=A(c)A(d)$, so $A(c)=e^{\theta c}$.  If $\theta=0$, continuity makes
$B(c)$ additive and $u$ affine.  If $\theta\ne0$, the cocycle identity gives
$B(c)=\beta(e^{\theta c}-1)$; an affine renormalization of $u$ yields the
exponential generator.
\end{proof}

\begin{corollary}[Dynamic uniqueness]\label{cor:dynamic-uniqueness}
Let
\[
 \mathfrak C_{s,t}(X)=u^{-1}(\E[u(X)\mid\mathcal F_s]).
\]
If the family is law invariant, cash additive, strictly relevant, strongly
time consistent, and nonlinear, then
\[
 \mathfrak C_{s,t}(X)
 =\frac1\theta\log\E[e^{\theta X}\mid\mathcal F_s].
\]
The general law-invariant rigidity theorem is consistent with
\cite{KupperSchachermayer2009}; \cref{thm:rigidity} gives the finite-state
argument needed here.
\end{corollary}
\begin{proof}
Strong time consistency follows from the conditional tower for every
certainty equivalent.  Apply \cref{thm:rigidity}; strict relevance and
nonlinearity exclude the affine case.
\end{proof}

\section{The parameter is no longer free}
For the collision path ensemble of Paper I,
\[
 \theta=I'(a)=\frac12\log\frac{3(1+a)}{7(1-a)}.
\]

\begin{theorem}[Existence plus uniqueness synthesis]\label{thm:synthesis}
The deterministic current preparation at value $a$ constructs an entropic
dynamic evaluation with parameter $I'(a)$.  Within the class of
\cref{cor:dynamic-uniqueness}, no other nonlinear evaluation is compatible
with law invariance, cash additivity, relevance, and strong time consistency.
Thus mechanics fixes existence and the axioms fix functional uniqueness.
\end{theorem}

\section{Fixed-law tangent probabilities}
Let $P_{s,t}^a$ be the physical diffusion transition family obtained in Paper
II, with fixed macro current $a$.  Define
\[
 \cE_{s,t}^{\theta,a}[\phi](x)
 =\frac1\theta\log P_{s,t}^a(e^{\theta\phi})(x).
\]

\begin{theorem}[Analytic terminal map and tangent law]\label{thm:tangent}
For bounded terminal data the map $\phi\mapsto\cE^{\theta,a}[\phi](x)$ is real
analytic.  Its first two derivatives are
\[
 D\cE[\phi]\eta=\E_{Q_\phi}\eta(X_t),
\]
\[
 D^2\cE[\phi](\eta,\zeta)
 =\theta\Cov_{Q_\phi}(\eta(X_t),\zeta(X_t)),
\]
where
\[
 \frac{dQ_\phi}{dP}
 =\frac{e^{\theta\phi(X_t)}}{\E e^{\theta\phi(X_t)}}.
\]
The tangent kernels satisfy the nonlinear chain-rule cocycle.
\end{theorem}
\begin{proof}
The exponential map is analytic on bounded sets and the denominator is bounded
away from zero.  Differentiate the logarithm.  The Markov property makes the
Radon--Nikodym factors multiply and proves the cocycle.
\end{proof}

\begin{theorem}[Covariance-curvature characterization]\label{thm:characterization}
Let a monotone cash-additive $C^2$ functional have probability-valued first
derivatives and
\[
 D^2\mathfrak E(\phi)(f,g)
 =\theta\Cov_{Q_\phi}(f,g)
\]
for one fixed $\theta$.  Then
\[
 \mathfrak E(\phi)=\frac1\theta\log Q_0(e^{\theta\phi}).
\]
\end{theorem}
\begin{proof}
Along $t\phi$, the tangent probabilities solve
\[
 \frac d{dt}Q_t(f)=\theta
 \{Q_t(f\phi)-Q_t(f)Q_t(\phi)\}.
\]
The unique solution is the exponential tilt of $Q_0$.  Integrate
$d\mathfrak E(t\phi)/dt=Q_t(\phi)$.
\end{proof}

\section{Microscopic tangent convergence}
Let $P_h^a$ be the path law of the deterministic collision recursion under the
driven preparation, and let $X_T^h$ be its terminal slow state.  Set
\[
 \frac{dQ_{h,\phi}^a}{dP_h^a}
 =\frac{e^{\theta\phi(X_T^h)}}
 {\E_{P_h^a}e^{\theta\phi(X_T^h)}}.
\]

\begin{theorem}[Collision tangent convergence]\label{thm:micro}
For bounded continuous $\phi,\eta,\zeta$, the weak convergence of Paper II
implies
\[
 Q_{h,\phi}^a\Longrightarrow Q_\phi^a,
\]
and convergence of the first two terminal derivative operators.
\end{theorem}
\begin{proof}
The exponential densities are uniformly bounded above and below.  Weak
convergence therefore passes to every numerator and denominator, including
products $\eta\zeta$.
\end{proof}

\section{Girsanov and BSDE representation}
Let
\[
 dX_t=b(X_t)dt+\sigma(a)dW_t,
 \qquad \sigma(a)\sigma(a)^T=A_{\rm phys}(a),
\]
and
\[
 u(t,x)=\frac1\theta\log\E_{t,x}e^{\theta\phi(X_T)}.
\]

\begin{theorem}[Genuine tangent density]\label{thm:girsanov}
For smooth bounded data,
\[
 M_s=\frac{\E[e^{\theta\phi(X_T)}\mid\mathcal F_s]}
 {\E e^{\theta\phi(X_T)}}
 =\mathcal E\left(
 \int_t^s\theta\sigma(a)^TDu(r,X_r)dW_r
 \right)
\]
is a uniformly integrable martingale.  Under $Q_\phi$, the process
\[
 W_s^Q=W_s-\int_t^s\theta\sigma(a)^TDu(r,X_r)dr
\]
is Brownian and
\[
 dX_s=(b+\theta A_{\rm phys}(a)Du)(X_s)ds+\sigma(a)dW_s^Q.
\]
\end{theorem}
\begin{proof}
Conditional expectation gives uniform integrability.  Ito's formula for
$e^{\theta u(s,X_s)}$ and the theta-HJB equation cancel the drift and identify
the stochastic exponential.  Bounded terminal data place its logarithm in the
standard BMO window, yielding Girsanov.
\end{proof}

\begin{corollary}[Quadratic and tangent BSDEs]\label{cor:bsde}
With $Y_s=u(s,X_s)$ and $Z_s=\sigma(a)^TDu(s,X_s)$,
\[
 Y_s=\phi(X_T)+\int_s^T\frac\theta2\abs{Z_r}^2dr
 -\int_s^TZ_rdW_r.
\]
A terminal direction solves a linear martingale BSDE under $Q_\phi$.
\end{corollary}

\section{The controlled branch}
For a fixed policy, \cref{thm:tangent} applies verbatim.  For an optimized
saddle value, Paper IV proves
\[
 D^2V=\theta\Cov_{Q^*}-G_{\phi z}G_{zz}^{-1}G_{z\phi}.
\]
Thus the first tangent is still a probability under the optimal tilted law,
but the covariance-curvature characterization applies only if the optimizer
correction vanishes.  This is a typed representation theorem rather than an
unqualified extension of the fixed-law result.

\section{Main theorem}
\begin{theorem}[Endogenous theta rigidity and representation]\label{thm:main}
For the fixed-law and fixed-policy physical semigroups derived in Papers
I--III:
\begin{enumerate}[label=(\roman*)]
\item the parameter is fixed by $\theta=I'(a)$;
\item entropic form is axiomatically unique in the declared class;
\item terminal tangents are probability laws with covariance curvature;
\item microscopic tangent laws converge;
\item Girsanov and quadratic BSDE representations are genuine.
\end{enumerate}
The controlled branch is represented with the optimizer-response correction of
Paper IV.
\end{theorem}
\begin{proof}
Combine \cref{thm:rigidity,cor:dynamic-uniqueness,thm:synthesis,thm:tangent,thm:characterization,thm:micro,thm:girsanov,cor:bsde} and Paper IV.
\end{proof}

\section*{Scope and review status}
The theorem is a complete internal proof in the declared deterministic
full-cover collision scope.  It does not claim that an arbitrary periodic
Sinai billiard has the same exact symbolic reset or that independent external
review has occurred.
