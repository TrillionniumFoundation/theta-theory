 F(z,u,v)=F_0(z,u,v)-\frac\mu2\abs u^2+\frac\nu2\abs v^2,
 \qquad \mu>L_u,\quad\nu>L_v.
\]
Define
\[
 \mathcal A_z(u,v)=(-D_uF(z,u,v),D_vF(z,u,v)).
\]

\begin{theorem}[Noncompact pure-saddle theorem]\label{thm:pure-saddle}
Assume $\mathcal A_z+N_{U\times V}$ is coercive.  Then it is strongly maximal
monotone with constant
\[
 \lambda=\min\{\mu-L_u,\nu-L_v\}>0.
\]
There is a unique global pure saddle $(u_*(z),v_*(z))$, and it depends
Lipschitz continuously on $z$ whenever $\mathcal A_z$ does.
\end{theorem}
\begin{proof}
For an increment $(\dot u,\dot v)$, mixed Hessian terms cancel in the symmetric
part of $D\mathcal A_z$ and
\[
 \pair{D\mathcal A_z(\dot u,\dot v),(\dot u,\dot v)}
 \ge\lambda(\abs{\dot u}^2+\abs{\dot v}^2).
\]
Strong monotonicity, maximality of the normal cone, and coercivity give a
unique variational-inequality solution.  Strong concavity in $u$ and strong
convexity in $v$ give the global saddle inequalities.  Comparing two
inclusions gives the Lipschitz bound.
\end{proof}

\section{The risk-sensitive saddle envelope}\label{sec:envelope}
Let $\phi$ denote terminal data and let $z=(u,v)$.  Consider a smooth local
risk-sensitive value
\[
 V(\phi)=\sup_u\inf_v\,G(\phi,u,v),
\]
where the unique saddle $z_\phi$ is supplied by \cref{thm:pure-saddle}.  Write
$G_{zz}$ for the signed saddle Hessian on the tangent space.

\begin{theorem}[First and second saddle-envelope derivatives]\label{thm:envelope}
If $G$ is $C^2$, the saddle is interior or the active face is locally fixed,
and $G_{zz}$ is invertible, then
\[
 DV(\phi)\eta=G_\phi(\phi,z_\phi)\eta,
\]
and
\[
 D^2V(\phi)(\eta,\zeta)
 =G_{\phi\phi}(\eta,\zeta)
 -G_{\phi z}[\eta]\,G_{zz}^{-1}\,G_{z\phi}[\zeta].
\]
For a fixed-policy entropic value,
$G_{\phi\phi}=q\,\Cov_{Q_\phi}(\eta,\zeta)$.  The second term is the genuine
optimizer-response correction and generally does not vanish.
\end{theorem}
\begin{proof}
Differentiate the saddle stationarity equation
$G_z(\phi,z_\phi)=0$ to obtain
\[
 Dz_\phi[\eta]=-G_{zz}^{-1}G_{z\phi}[\eta].
\]
The first derivative is the ordinary saddle envelope theorem.  Differentiate
again and substitute the response of $z_\phi$.
\end{proof}

\section{Controlled physical theta equation}\label{sec:hjb}
\begin{theorem}[Pure-strategy controlled limit]\label{thm:controlled-limit}
Assume the local payoff satisfies the uniform saddle conditions above.  Then
the lower and upper microscopic schemes converge to the same bounded viscosity
solution
\[
 -u_t
 =\sup_u\inf_v\{c^{uv}(x)+b^{uv}(x)u_x\}
 +\frac{D_{R,q}}2u_{xx}
 +\frac{qD_{R,q}}2(u_x)^2.
\]
The saddle feedback is the unique Lipschitz selector from
\cref{thm:pure-saddle}.
\end{theorem}
\begin{proof}
The saddle theorem identifies the local lower and upper operators.  The
billiard correctors, physical clock, and random-horizon bounds are those of
Papers II--III and are uniform under the controlled slow coefficients.
Monotonicity, stability, consistency, and comparison give convergence
\cite{FlemingSoner2006,CrandallIshiiLions1992}.
\end{proof}

\section{A deterministic soft-filter extension}\label{sec:filter}
Independently of the billiard, let a hidden coordinate be generated by a
measure-preserving two-sided product shift whose one-step kernel on the
observed factor is
\[
 P_Y(y,\cdot)=\delta\gamma+(1-\delta)N(ry,1-r^2),
 \qquad0<\delta<1,\quad\abs r<1.
\]
This is a deterministic shift under a specified product initial measure.  Let
bounded observation likelihoods satisfy $0<g_-\le g_o(y)\le g_+$.

\begin{theorem}[Soft-filter contraction]\label{thm:filter}
Choose $m$ such that
\[
 \frac{2g_+}{g_-}(1-\delta)^m<1
\]
and
\[
 \frac{g_+}{g_-}\bigl((1-\delta)r^2\bigr)^m<1.
\]
Then $m$ predictions followed by one Bayes update contract total variation and
preserve an explicit quadratic moment ball.  The contraction is uniform along
common observation and strategy trees.
\end{theorem}
\begin{proof}
The common refresh component contracts prediction by $1-\delta$.  Bayes
normalization expands total variation by at most $2g_+/g_-$.  The Gaussian
autoregressive part gives an affine drift for $1+y^2$; the second displayed
inequality makes the post-Bayes moment recursion strictly contractive.
\end{proof}
