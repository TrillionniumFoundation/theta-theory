\section{Pointwise removal of every section-decision layer}
\label{sec:v48-all-decisions}
We retain the physical incidence and clearance protection, but no longer
restrict the free section decisions to a fixed endpoint depth. The
uniform marked-strip estimate permits summation over all depths before
taking the collision limit. This removes the interior-decision part
which cannot be controlled by the fixed-history envelopes alone.

Write $d_j=\min(j,m-j)$, $e_j=q_0^{d_j/4}$ and retain $e'_j$ for
flights. For fixed $\eta,\varepsilon\in(0,\varepsilon_0)$ put
\begin{align}
 H_{m,R}^{\eta}
   &=\prod_{j=0}^m\Theta(c_j/(\eta e_j))
             \prod_{j=0}^{m-1}\Theta(\Delta_j/(\eta e'_j)),\notag\\
 D_{m,R}^{\varepsilon}
   &=\prod_{j=0}^m\Theta(s_j/(\varepsilon e_j)).
                                      \label{eq:v48-physical-decision-guards}
\end{align}
The first product contains only physical margins. The original full
protection is $\Xi_{m,R}^{\varepsilon}=H_{m,R}^{\varepsilon}
D_{m,R}^{\varepsilon}$. In all auxiliary continuations the measure
is $\nu/c$, with $c=91/(10000\pi)$; on the original initial section
this is exactly $\nu_R^*$.

\subsection{A positive telescoping by decision depth}
Set $D_m=\lfloor m/2\rfloor$. For $-1\le a\le D_m$ define
\begin{align}
 G_a&=H_{m,R}^{\eta}\prod_{d_j>a}\Theta(s_j/(\varepsilon e_j)),
       &G_{-1}&=H_{m,R}^{\eta}D_{m,R}^{\varepsilon},\notag\\
 Z_a&=G_a-G_{a-1}\ge0 &&(0\le a\le D_m).
                                      \label{eq:v48-depth-telescope}
\end{align}
Empty products equal one. Therefore
\begin{equation}\label{eq:v48-depth-exact-sum}
 \sum_{a=0}^{D_m}Z_a
       =H_{m,R}^{\eta}(1-D_{m,R}^{\varepsilon}).
\end{equation}
If $Z_a>0$, every physical margin is $\eta$-protected, every
section margin at depth greater than $a$ is $\varepsilon$-protected,
and at least one of the at most two contacts of depth $a$ satisfies
$s_j<2\varepsilon e_j$. Contacts of depth at most $a$ may cross the
section. At most $r_a=2a+2$ occupation summands can then change.
The terminal membership at $m$ is not one of the summands of $A_m$.

For $|w|\le M$ let $z^{a,w}_{n,k,m,R}(t)\,dt$ be the pushforward
of $\one_{\Pi_{n,k,m,R}}wZ_a\nu_R^*$ by $T_{n,R}$. These are
well-defined finite complex densities by domination by the complete
fixed-count source. Their labels have not been enlarged. We use an
enlargement only in a positive comparison below.

\begin{lemma}[One geometric continuation for all decision depths]
\label{lem:v48-depth-continuation}
Put $\zeta=\min(\eta,\varepsilon)$. There is a common endpoint
continuation of radius $r_*\zeta^3$ for every source point with
$Z_a>0$, with the decisions of depth at most $a$ freed. The radius,
Hessian and relative-density bounds are independent of $a$ and $m$.
For collision index $j$ its section-distance differential obeys
\begin{equation}\label{eq:v48-depth-response}
 |\nabla s_j|\le C\zeta^{-1}q_0^{3d_j/4}
                      \quad\hbox{almost everywhere}.
\end{equation}
The continuation preserves the physical word, displacement, collision
count and every retained decision. Its occupation lies in
$\{l:|l-n|\le r_a\}$ when it starts at the original coefficient $n$.
\end{lemma}
\begin{proof}
Every retained margin is bounded below by $\zeta$ times its geometric
weight. Apply the endpoint continuation and response proof of
Lemma~\ref{lem:v47-free-decision-continuation}, using $\zeta$ for
this common lower bound and omitting precisely the decisions of depth
at most $a$. That proof checks margins separately: the relative
change at depth $d$ is bounded by
$C\zeta^{-2}q_0^{d/2}(|du|+|dv|)$.
The number of omitted checks does not occur in the estimate.
Consequently it applies also when $a$ depends on $m$, up to $D_m$.
For overlapping endpoint layers omit duplicate contacts.
Equation~\eqref{eq:v48-depth-response} is the unscaled endpoint
response before division by $e_j$. It remains an almost-everywhere
bound at rectangle corners. The occupation assertion counts only the
free terms at times $0,\ldots,m-1$. The source density on the
continuation is the same $|F_{uv}|/(4\pi Rc)$ as in the earlier
physical continuation.
\end{proof}

\subsection{A summable high-gradient comparison}
Fix once, for these $\eta,\varepsilon$, a sufficiently small
$\delta>0$ satisfying
\begin{equation}\label{eq:v48-fixed-gradient-choice}
 \delta\le c_0\min\{\zeta^3,\varepsilon\zeta\}.
\end{equation}
The constant $c_0$ below is independent of the margin parameters.
This is not a count-dependent gradient scale. Use the roof vector
field $V=\nabla F/|\nabla F|^2$ on $|\nabla F|\ge\delta$ and the
common roof width
\begin{equation}\label{eq:v48-common-flow-width}
 h=c_1\min\{\zeta^3\delta^2,\varepsilon\zeta\delta\}>0.
\end{equation}
All these quantities are independent of the layer $a$.

\begin{proposition}[Uniform layer estimate away from critical points]
\label{prop:v48-high-depth}
Let $z^{a,+}$ denote the unweighted layer density restricted by
$|\nabla F|\ge\delta$. For every fixed sufficiently large auxiliary
band $B_*$,
\begin{align}
 m^2\|z^{a,+}_{n,k,m,R}\|_\infty
 &\le C(a+1)\varepsilon^{\alpha_0}q_0^{\alpha_0a/4}
                      (1+(B_*h)^{-1})\notag\\
 &\quad+C_{B_*,h}(a+1)m^2\rho_{B_*}^{m/2}.
                                      \label{eq:v48-high-depth}
\end{align}
The constants in the first line are independent of $a,m,\eta,
\varepsilon,\delta,B_*$. The constants in the second line need not
be uniform in $B_*$ or $h$.
\end{proposition}
\begin{proof}
The continuation, lower gradient and relative Jacobian estimates used
in Proposition~\ref{prop:v47-high-gradient-boundary} hold with
$\zeta$ in place of $\eta$. Along the flow,
\[
 e_j^{-1}|ds_j/dt|
       \le C\zeta^{-1}\delta^{-1}q_0^{d_j/2}.
\]
At a contact of depth $a$ which is initially bad, the choice
\eqref{eq:v48-common-flow-width} thus keeps its distance below
$C\varepsilon q_0^{a/4}$. It suffices to test this one contact, not
the entire history of length $a$. Its index is $a$ or $m-a$.
The flow crosses free decision cuts only; no incidence, clearance,
physical-word or retained section boundary is crossed.

For almost every level $F=t$, roof time and ODE uniqueness give the
same injective level-to-window map on each glued physical word as in
\eqref{eq:v47-flow-to-relaxed-window}. Thus the density is at most
$C/(hc)$ times the sum, over $|l-n|\le2a+2$ and the two possible
marks $j$, of
\[
 \nu\{K_m^{\rm c}=k,A_m=l,|S_m-t|<h,
                b_{R,C\varepsilon q_0^{a/4}}(T_R^jx)=1\}.
\]
There are at most $4a+5$ occupation coefficients; this allowance
appears only on the upper side. Apply
Theorem~\ref{thm:v48-marked-strip-local} at the common fixed interval
of length $2h$. Its strip width and mark are allowed to depend on
$m$. Division by $h$ proves \eqref{eq:v48-high-depth}.
This is the point where the new uniform error, absent from the old
finite-history envelope, is indispensable.
\end{proof}

\subsection{Critical collars retain the bad decision}
\begin{proposition}[Uniform layer estimate near critical points]
\label{prop:v48-low-depth}
Let $z^{a,-}$ be the unweighted layer density restricted by
$|\nabla F|<\delta$. There is a fixed roof width $r=C_0\delta^2>0$,
independent of $a,m$, such that
\begin{align}
 m^2\|z^{a,-}_{n,k,m,R}\|_\infty
 &\le C(a+1)\varepsilon^{\alpha_0}q_0^{\alpha_0a/4}
                       (1+(B_*r)^{-1})\notag\\
 &\quad+C_{B_*,r}(a+1)m^2\rho_{B_*}^{m/2}.
                                      \label{eq:v48-low-depth}
\end{align}
Here $B_*$ is fixed, as above. The continued collar source is $\nu/c$,
including when its critical center is outside the return section.
\end{proposition}
\begin{proof}
Complete a low-gradient source point to the unique physical
normal-to-normal center as in Lemma~\ref{lem:v47-free-critical-cluster}.
All retained margins are at least the common $\zeta$ threshold, so
the center lies within $C\delta$ and its critical roof within
$C\delta^2$. Choose $c_0$ in \eqref{eq:v48-fixed-gradient-choice}
so that the required collars remain inside the half-margin
continuations. Their radius in endpoint coordinates is $O(\delta)$.
Equation~\eqref{eq:v48-depth-response}, on the completed square with
half-margins, shows that an initially bad contact of depth $a$ stays
within
\[
 2\varepsilon q_0^{a/4}
       +C\delta\zeta^{-1}q_0^{3a/4}
           \le C'\varepsilon q_0^{a/4}
\]
of its section boundary throughout the collar. This retains the small
strip in the critical comparison. It is not enough to retain only the
two normal endpoint strips and then sum a bound independent of $a$.

Use the intrinsic physical coefficient $J_z$ of each completed center
and the mass collar of roof width $r=C_0\delta^2$, increasing $C_0$
to cover all completed source points and reducing $c_0$ to keep
$r<h_{\zeta/2}$. Each collar has mass at least $a_*rJ_z$ in
$\nu/c$ and density at most $A_*J_z$. For centers whose collars can
contribute at roof $t$, their critical roofs lie in an interval of
length $Cr$, and the collar union lies in a roof interval of length
$C'r$. Take each physical collar once. Endpoint injectivity makes
collars from different physical words disjoint. Splitting a word by
free decisions does not produce extra collars.

Every collar retains its original middle occupation, and at most
$2a+2$ free summands can differ from the original $n$. Hence the
whole collar union is included in the same occupation allowance,
not a second enlargement of it. It also has one of the two marked
strip conditions at $a$ or $m-a$ just proved. Apply
Theorem~\ref{thm:v48-marked-strip-local} to this positive union.
Dividing by $a_*r$ bounds the sum of the intrinsic coefficients.
The collar-density upper estimate then gives
\eqref{eq:v48-low-depth}. No extra normal-strip smallness is needed
here: the retained bad decision supplies the summable depth factor.
\end{proof}

\begin{theorem}[All-depth pointwise decision deconcentration]
\label{thm:v48-all-decision-smallness}
For fixed $\eta,\varepsilon\in(0,\varepsilon_0)$ and $|w|\le M$, let
\begin{equation}\label{eq:v48-all-decision-density}
 d^{\eta,\varepsilon,w}_{n,k,m,R}(t)\,dt
  =(T_{n,R})_*\bigl(\one_{\Pi_{n,k,m,R}}w
       H_{m,R}^{\eta}(1-D_{m,R}^{\varepsilon})\nu_R^*\bigr).
\end{equation}
There is an absolute $C$ such that
\begin{equation}\label{eq:v48-all-decision-limsup}
 \limsup_{m\to\infty}\sup_{R,n,k,\,|w|\le M}
        m^2\|d^{\eta,\varepsilon,w}_{n,k,m,R}\|_\infty
                                  \le CM\varepsilon^{1/16}.
\end{equation}
More precisely, for the tail $d^{\eta,\varepsilon,w}_{>J}$ obtained
by summing the layers $a>J$, $J\ge-1$, there are
$C_{\eta,\varepsilon}<\infty$ and $\rho_{\eta,\varepsilon}<1$ such
that, for every $m\ge2$ and every such $J$,
\begin{align}
 m^2\|d^{\eta,\varepsilon,w}_{>J,n,k,m,R}\|_\infty
 &\le CM\varepsilon^{1/16}(J+2)q_0^{(J+1)/64}
                  +C_{\eta,\varepsilon}M m^4
                                      \rho_{\eta,\varepsilon}^{m/2}.
                                      \label{eq:v48-decision-depth-tail}
\end{align}
A tail beyond $D_m$ is zero. In particular the normalized tail tends
to zero for any $J=J(m)\to\infty$, at fixed $\eta,\varepsilon$.
The estimate holds for arbitrary bounded measurable source insertions.
\end{theorem}
\begin{proof}
First take $w=1$. Choose $\delta,h,r$ as above, once for the fixed
margin parameters, and then choose one auxiliary band
$B_*\ge\max\{B_0,h^{-1},r^{-1}\}$. This band is fixed before the
collision limit and before summing the layers. The two layer
estimates now have leading bound
$C(a+1)\varepsilon^{1/16}q_0^{a/64}$ and an error
$C_{\eta,\varepsilon}(a+1)m^2\rho_{\eta,\varepsilon}^{m/2}$.
Summing the finite positive telescope first, use
\[
 \sum_{a>J}(a+1)x^a
 =x^{J+1}\left(\frac{J+2}{1-x}+\frac{x}{(1-x)^2}\right),
       \qquad x=q_0^{1/64}<1,
\]
and $\sum_{a=0}^{D_m}(a+1)\le C m^2$.
These give \eqref{eq:v48-decision-depth-tail}. In particular the
summed error vanishes; no interchange of a limsup and an uncontrolled
infinite depth sum is used. Setting $J=-1$ gives
\eqref{eq:v48-all-decision-limsup}. Radon--Nikodym domination by
$M$ times the positive source proves the assertion for all $w$.
Neither $w$ nor a long pullback of a section indicator is differentiated.
\end{proof}

\begin{corollary}[All roof bands and finite trace jumps]
\label{cor:v48-decision-all-band}
With the same sources,
\begin{equation}\label{eq:v48-decision-all-band}
 \limsup_{m\to\infty}\sup_{B>0,R,n,k,\,|w|\le M}
 m^2\|d^{\eta,\varepsilon,w}-K_B*d^{\eta,\varepsilon,w}\|_\infty
                                  \le CM\varepsilon^{1/16}.
\end{equation}
The analogous bound with the first term of
\eqref{eq:v48-decision-depth-tail} holds for a depth tail.
Where finite one-sided traces exist, each jump is bounded by twice
its essential-height estimate. No sum of absolute jump heights is
asserted.
\end{corollary}
\begin{proof}
Use $\|K_B\|_1=\|K_1\|_1$ and
$\|f-K_B*f\|_\infty\le(1+\|K_1\|_1)\|f\|_\infty$.
The supremum over the displayed reconstruction band $B$ is therefore
a convolution consequence, not a spectral estimate at a growing
band. The auxiliary band $B_*$ used in the preceding proof was fixed.
The trace assertion follows from the essential bound.
\end{proof}
