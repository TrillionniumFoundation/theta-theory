\section{The complete protected correction and collapsing margins}
\label{sec:v46-protected-reduction}
We combine the regular density estimate with the positive critical
collars. The resulting estimate concerns the entire smoothly protected
source, including its noncritical inverse. The remaining source is
identified positively and has uniformly small total mass as protection
is removed. Small mass will not be substituted for a pointwise density
estimate; the final arithmetic criterion retains that remaining signed
correction explicitly.

\subsection{Near-normal sources belong to actual critical collars}
\begin{lemma}[Completion to a protected normal critical word]
\label{lem:v46-near-critical-completion}
There are $d_*,C_*>0$, independent of $m,R,\varepsilon$, such that if
$x\in\mathsf P_{m,R}(\varepsilon)$ and
$|\nabla F(x)|\le2\delta$, with $0<\delta<d_*\varepsilon^3$, then
$x$ belongs to a collar $U_z(C_*\delta^2)$ from
Proposition~\ref{prop:v45-mass-collar}. Its critical center $z$ is an
$\varepsilon/2$-protected normal-to-normal word with exactly the same
$(n,k,m)$ labels. Moreover $C_*\delta^2<h_{\varepsilon/2}$.
\end{lemma}
\begin{proof}
The two endpoint tangential velocities at $x$ have modulus at most
$2\delta$. On the endpoint square in
Lemma~\ref{lem:v46-regular-endpoints}, reduce its fixed radius
constant so that they stay below $1/2$. Its reduced Hessian consequently
has lower and upper bounds $\lambda I_2\le\nabla^2F\le\Lambda I_2$
with absolute positive constants. On a ball of radius proportional to
$\varepsilon^3$ inside that square,
\[
 F(x+v)-F(x)\ge-2\delta|v|+\lambda|v|^2/2.
\]
For $d_*$ sufficiently small the minimum on this closed ball is
interior. Strict convexity makes it unique; call it $z$. The first
variation gives $p_0(z)=p_m(z)=0$, and strong monotonicity of the
gradient gives $|z-x|\le2\delta/\lambda$. Continuation preserved every
half-margin and every exact label, so $z$ is
$\varepsilon/2$-protected.

After a further fixed reduction of $d_*$, the original point $x$ lies
in the endpoint square of radius $r_*(\varepsilon/2)^3$ about $z$.
The continuations coincide by endpoint injectivity. Taylor's formula
there gives $0\le F(x)-F(z)\le C_*\delta^2$. Take $C_*$ slightly
larger if needed to make the upper inequality strict on the indicated
support. Finally choose $d_*$ so that
$C_*d_*^2<h_*2^{-6}$. This proves the last assertion and the inclusion
in the actual collar, without replacing the return event by a nearby
index.
\end{proof}

Let
\begin{equation}\label{eq:v46-cluster-budget}
 \mathcal C_m^{\eta}(r)=
      \sup_{R,n,k,t}m^2
          \mathcal J^{\eta}_{n,k,m,R}([t-r,t+r]).
\end{equation}
For fixed $\eta>0$ and $0<r<h_\eta$, the proved cluster bound is
\begin{equation}\label{eq:v46-cluster-budget-limit}
 \limsup_{m\to\infty}\mathcal C_m^{\eta}(r)\le Cr.
\end{equation}
Keep the kernel $K_B$ fixed as in the preceding section and put
$\kappa_0=1+\|K_1\|_1$. For an admissible insertion define the complete
protected density and its signed correction by
\begin{align}
 f^{\varepsilon,a}_{n,k,m,R}(t)\,dt
    &=(T_{n,R})_*(\one_{\Pi_{n,k,m,R}}
                         a\Xi_{m,R}^{\varepsilon}\,\nu_R^*),\notag\\
 \mathscr D^{\varepsilon,a}_{B,n,k,m,R}
    &=f^{\varepsilon,a}_{n,k,m,R}
                         -K_B*f^{\varepsilon,a}_{n,k,m,R}.
                                      \label{eq:v46-protected-correction}
\end{align}
The cutoff is the explicit physical guard
\eqref{eq:v46-source-guard}; no intrinsic edge coefficient is altered.

\begin{theorem}[The full smoothly protected signed correction]
\label{thm:v46-protected-correction}
For $0<\delta<d_*\varepsilon^3$ and every $B>0$, uniformly over all
exact labels and all admissible insertions with bounds $(M,L)$,
\begin{align}
 \sup_{R,n,k}m^2
       \|\mathscr D^{\varepsilon,a}_{B,n,k,m,R}\|_\infty
 &\le C\left[
 \frac{\mathcal T_{\varepsilon,\delta}(M,L)}{B}
                        \mathcal Q_m(h_{\varepsilon,\delta})
          +M\mathcal C_m^{\varepsilon/2}(C_*\delta^2)\right].
                                      \label{eq:v46-finite-time-bound}
\end{align}
In particular there are absolute constants $C,C_2$ such that, for
$B\ge C_2\varepsilon^{-12}$,
\begin{equation}\label{eq:v46-bandwidth-modulus}
 \limsup_{m\to\infty}\sup_{R,n,k,a}m^2
       \|\mathscr D^{\varepsilon,a}_{B,n,k,m,R}\|_\infty
                    \le C(M+L)B^{-1/2}.
\end{equation}
The bandwidth and protection are fixed in this limsup. Also, for every
fixed $\varepsilon>0$ and every deterministic $B_m\to\infty$,
\begin{equation}\label{eq:v46-growing-band-protected}
 \sup_{R,n,k,a}m^2
       \|\mathscr D^{\varepsilon,a}_{B_m,n,k,m,R}\|_\infty
                                  \longrightarrow0.
\end{equation}
No rate in $m$ is asserted in either conclusion.
\end{theorem}
\begin{proof}
Split the guard source using $H_\delta$ from
\eqref{eq:v46-regular-source}. Its high-gradient part is
$g^{\varepsilon,\delta,a}$, controlled by
Corollary~\ref{cor:v46-noncritical-unsmoothing}. Its remaining part
has modulus dominated by $M$ times the positive source with
$\Xi^{\varepsilon}>0$ and $|\nabla F|\le2\delta$.
Lemma~\ref{lem:v46-near-critical-completion} places this latter source
in the union of the $\varepsilon/2$-protected critical collars of roof
width $C_*\delta^2$. Equation~\eqref{eq:v45-critical-density-cluster}
therefore bounds its density in modulus by
\[
 MA_*\mathcal J^{\varepsilon/2}_{n,k,m,R}
                                      ([t-C_*\delta^2,t]).
\]
The operator $f\mapsto f-K_B*f$ has $L^\infty$ norm at most
$\kappa_0$, for every $B$. Combining the two bounds proves
\eqref{eq:v46-finite-time-bound}. It controls the actual regular
inverse and the actual near-critical source, not an absolute sum of
formal word jets.

For fixed $\varepsilon,\delta,B$, take the collision limsup using
\eqref{eq:v46-window-budget-limit} and
\eqref{eq:v46-cluster-budget-limit}. The result is
\begin{equation}\label{eq:v46-two-parameter-modulus}
 C\left[\frac{L\delta^{-1}
           +M(\varepsilon^{-3}\delta^{-1}+\delta^{-2})}{B}
                         +M\delta^2\right].
\end{equation}
For $\delta=B^{-1/4}$ the geometric restriction holds if
$B\ge C_2\varepsilon^{-12}$, with $C_2$ chosen strictly larger than
$d_*^{-4}$. The two terms $B^{-1}\delta^{-2}$ and $\delta^2$ equal
$B^{-1/2}$. Also
$\varepsilon^{-3}B^{-1}\delta^{-1}
   =\varepsilon^{-3}B^{-3/4}\le C B^{-1/2}$
under the same restriction, and $LB^{-3/4}\le LB^{-1/2}$.
This proves \eqref{eq:v46-bandwidth-modulus}.

For a prescribed diverging sequence $B_m$, keep
$\varepsilon,\delta$ fixed in \eqref{eq:v46-finite-time-bound}.
The first term tends to zero because its local-window budget is
eventually bounded. The second has limsup at most $CM\delta^2$.
Only then let $\delta\downarrow0$. This proves
\eqref{eq:v46-growing-band-protected}. It does not use
\eqref{eq:v46-bandwidth-modulus} with $B=B_m$ inside a collision
limsup. In particular it gives no polynomial rate along that sequence.
\end{proof}

\subsection{The positive source outside the smooth protection}
We next quantify the mass of the exact remainder, uniformly in the
length of the history. Its density is not bounded by its mass.

\begin{lemma}[One-flight boundary neighborhoods]
\label{lem:v46-margin-neighborhoods}
There is $C<\infty$, uniform in $R$, such that for $0<s<1$,
\begin{equation}\label{eq:v46-one-step-boundary}
 \nu\{c_0<s\}\le s^2,\qquad
 \nu\{\operatorname{dist}(x,\partial Y_R^*)<s\}\le Cs,
 \qquad \nu\{\Delta_0<s\}\le Cs.
\end{equation}
The last event refers to a regular physical flight and excludes its
two incident disks.
\end{lemma}
\begin{proof}
In the flat coordinates $(\alpha,p)$ the collision probability is
$(4\pi)^{-1}d\alpha\,dp$. Since $c_0=(1-p^2)^{1/2}$, the first mass
is $1-\sqrt{1-s^2}\le s^2$. The section is a fixed finite union of
moving rectangles with uniformly bounded perimeter, which proves the
second estimate, including their corners.

For the third, finite horizon restricts the relevant nonincident disks
to a fixed finite set. Take first $s<s_0$ with $s_0>0$ sufficiently
smaller than the disk separation $3/50$. A segment at clearance less
than $s$ from a nonincident disk attains its closest point in its
interior: both segment endpoints lie on other disks and are at least
$3/50$ from that disk. Write $r=|d-Rn_\alpha|\ge1-R\ge53/100$ for
the distance from the initial point to the candidate center. A regular
segment misses its open disk, so its line has perpendicular distance
in $[R,R+s]$. On either of the two tangent directions the derivative
of this distance with respect to the direction angle has modulus
\[
 \sqrt{r^2-(R+s)^2}\ge
     \sqrt{(53/100)^2-(47/100+s_0)^2}>0.
\]
Thus the allowed direction angles have total length at most $Cs$,
uniformly in the initial angle and in the candidate. Restricting to
outgoing directions or to the actual first-hit segment only decreases
the set. The density in the direction angle is bounded, since
$dp=\cos\varphi\,d\varphi$. Integrate in $\alpha$ and sum the fixed
candidate set. For $s\ge s_0$ increase $C$ and use the mass bound one.
\end{proof}

\begin{proposition}[Uniform exhaustion in original source mass]
\label{prop:v46-protection-mass}
The guards increase to one almost everywhere as
$\varepsilon\downarrow0$ for every fixed $m$, and
\begin{equation}\label{eq:v46-unprotected-mass}
 \sup_{m\ge1,\ R\in I}
       \int(1-\Xi_{m,R}^{\varepsilon})\,d\nu_R^*
                                  \le C\varepsilon.
\end{equation}
Define the nonnegative exact-label remainder density by
\begin{equation}\label{eq:v46-boundary-density}
 b^{\varepsilon}_{n,k,m,R}(t)\,dt
  =(T_{n,R})_*(\one_{\Pi_{n,k,m,R}}
                       (1-\Xi_{m,R}^{\varepsilon})\,\nu_R^*).
\end{equation}
It is supported by the source failing the $2\varepsilon$ protection
conditions, and
\begin{equation}\label{eq:v46-label-mass-sum}
 \sup_{m,R}\sum_{n=1}^m\sum_k\|b^{\varepsilon}_{n,k,m,R}\|_1
                                  \le C\varepsilon.
\end{equation}
For a bounded measurable insertion the same assertion holds in total
variation, multiplied by its supremum bound. The total-variation norm
here is the mass of the variation measure, without a factor $1/2$.
\end{proposition}
\begin{proof}
For numbers $0\le a_i\le1$, $1-\prod_i a_i\le\sum_i(1-a_i)$.
Apply this to \eqref{eq:v46-source-guard}; a summand can be nonzero
only when its margin is less than twice its threshold. Invariance of
$\nu$ and domination $\nu_R^*\le c^{-1}\nu$ permit
Lemma~\ref{lem:v46-margin-neighborhoods} to be applied at every
collision or flight. Consequently the expectation is at most
\[
 C\sum_{j=0}^{m}\bigl((2\varepsilon e_j)^2+2\varepsilon e_j\bigr)
     +C\sum_{j=0}^{m-1}2\varepsilon e'_j
                           \le C\varepsilon.
\]
At most two indices have any given distance from the ends, so all the
geometric sums are bounded independently of $m$. No independence or
mixing assertion is used here.

Every finite regular history away from section boundaries has finitely
many positive margins. Monotonicity of $\Theta$ therefore gives the
almost-everywhere exhaustion. The exact events $\Pi_{n,k,m,R}$ are
disjoint as $(n,k)$ varies at fixed $m$: collision $m$ has at most one
actual return index and displacement. Sum their restricted masses to
obtain \eqref{eq:v46-label-mass-sum}. Bounded insertions are dominated
in variation by the same positive source. Absolute continuity of its
roof pushforward follows also by domination by the complete density
in Theorem~\ref{thm:v44-complete-density-bound}.
\end{proof}

\subsection{Ordered protection scales and the arithmetic target}
Choose a fixed sufficiently large $A>0$ and, for all sufficiently large
$B$, put
\begin{equation}\label{eq:v46-ordered-protection}
 \varepsilon(B)=A B^{-1/12}<\varepsilon_0.
\end{equation}
Take $A^{12}\ge C_2$. Then the condition of
\eqref{eq:v46-bandwidth-modulus} holds for each such fixed $B$.
It follows that the complete protected correction is at most
$CB^{-1/2}$ at the normalized local scale after the collision limsup,
while the complementary source has total mass at most $CB^{-1/12}$.
These two estimates have different norms.

\begin{theorem}[Localization of the full arithmetic pointwise obstruction]
\label{thm:v46-boundary-reduction}
For every $m,R,n,k$ and $B>0$ the unweighted full source has the exact
almost-everywhere decomposition
\begin{align}
 p_{n,R}(k,m,\cdot)&=f^{\varepsilon,1}_{n,k,m,R}
                                      +b^{\varepsilon}_{n,k,m,R},\notag\\
 \mathscr D_{B,n,R}(k,m,\cdot)
   &=\mathscr D^{\varepsilon,1}_{B,n,k,m,R}
          +b^{\varepsilon}_{n,k,m,R}-K_B*b^{\varepsilon}_{n,k,m,R}.
                                      \label{eq:v46-full-source-split}
\end{align}
With $\varepsilon=\varepsilon(B)$, the radius-uniform raw arithmetic
local law with main term $\mathcal L_{m,R}$ holds if and only if,
for every central compact set $|Z|\le M_0$,
\begin{equation}\label{eq:v46-boundary-criterion}
 \lim_{B\to\infty}\limsup_{m\to\infty}
   \sup_{R,n,k}m^2\operatorname*{ess\,sup}_{t:\ |Z|\le M_0}
       |b^{\varepsilon(B)}_{n,k,m,R}(t)
                   -(K_B*b^{\varepsilon(B)}_{n,k,m,R})(t)|=0.
\end{equation}
Here $Z=((k_1,k_2,t,n)-m(0,0,\bar\tau_R,c))/\sqrt m$, exactly as
in Theorem~\ref{thm:v43-arithmetic-raw-criterion}. Its fixed-radius
specialization retains $\mathfrak a_R$; no arithmetic factor is set
to one.
\end{theorem}
\begin{proof}
The first identity is a partition of the original positive source.
The second follows by linearity and
\eqref{eq:v43-correction-identity}; it does not depend on any choice
of germ localization. For each fixed $B$, the full-source fixed-band
inverse has main term $\mathcal L_{m,R}$ with error $o(1)$ after
multiplication by $m^2$. Theorem~\ref{thm:v46-protected-correction}
bounds the protected term's limsup by $CB^{-1/2}$ under
\eqref{eq:v46-ordered-protection}. Thus
\eqref{eq:v46-boundary-criterion}, the triangle inequality, and the
order $m\to\infty$ followed by $B\to\infty$ imply the full raw local
law. Conversely, that law and the fixed-band inverse make the full
correction tend to zero at each fixed $B$. The same split bounds the
limsup of the boundary correction by $CB^{-1/2}$, proving the stated
criterion. The fixed-radius assertion is the unchanged specialization
of the transition kernel.
\end{proof}

\begin{corollary}[A count-dependent protection with no prescribed rate]
\label{cor:v46-slow-protection}
For any given $B_m\to\infty$ there is a deterministic
$\varepsilon_m\downarrow0$ for which
\[
 \sup_{R,n,k}m^2
       \|\mathscr D^{\varepsilon_m,1}_{B_m,n,k,m,R}\|_\infty\to0,
 \qquad
 \sup_R\sum_{n,k}\|b^{\varepsilon_m}_{n,k,m,R}\|_1\to0.
\]
No rate for $\varepsilon_m$ as a function of $m$ is supplied.
\end{corollary}
\begin{proof}
Take any decreasing $\varepsilon_j\to0$. For each $j$,
\eqref{eq:v46-growing-band-protected} supplies an integer $M_j$ beyond
which its normalized error is at most $2^{-j}$. Make these integers
strictly increasing and take $\varepsilon_m=\varepsilon_j$ on
$[M_j,M_{j+1})$. The mass assertion follows from
\eqref{eq:v46-label-mass-sum}. This diagonal argument does not replace
the $L^1$ conclusion by a density bound, nor does it evaluate the full
transition inverse at $B_m$.
\end{proof}

\paragraph{The remaining estimate.}
The protected noncritical inverse has now been controlled at the
local scale, and the critical collars complete the same statement for
the entire smooth protection. The remaining positive source in
\eqref{eq:v46-boundary-density} includes near-grazing, competing-hit and
section-decision histories with collapsing margins, together with their
finite jumps and noncritical residuals. Its total mass tends to zero
under the explicit ordered protection scale, but
\eqref{eq:v46-boundary-criterion} is still a pointwise signed estimate.
Neither small source mass nor the exponential fixed-count height bound
implies it. The criterion retains precisely that estimate for the
original four-coordinate record. The corresponding complete weighted
boundary estimate and pointwise roof-conditioned bridge are not
consequences of the regular insertion theorem above.
