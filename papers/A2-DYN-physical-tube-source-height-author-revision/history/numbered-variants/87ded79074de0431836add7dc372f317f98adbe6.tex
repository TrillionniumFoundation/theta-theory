\section{Actual entrances at every positive residence scale}
\label{sec:v94-entrances}
An incoming trace is attached to the incident maximal segment, not to
a boundary point without its physical component. We first separate its
types. If an endpoint or internal incidence vanishes, assign the end
to the grazing class. Among the remaining ends assign those with
$z_\sigma^-=0$ to the critical class. Every other end is a nongrazing
physical or section-decision entrance with $z_\sigma^->0$.
Simultaneous section and clearance failures can be assigned to the
first failing test in a fixed finite ordering. These assignments are
Borel: the limiting contact data and lifetimes are Borel, and all tests
are closed conditions on those data. They form a disjoint partition.
If all limiting physical margins and $z_\sigma^-$ were strictly
positive, regular continuation would contradict maximality of the
same exact-event segment. This exhausts the entrance types without
counting chart seams. The grazing trace vanishes; the critical
directional trace is retained.

Put $k_{r,t}(u)=(u-t+r)/r$ for $t-r<u\le t$. For fixed positive
$\rho,\delta$, define the noncritical, positive-residence entrance
assignment
\begin{equation}\label{eq:v94-entrance-assignment}
 B_{c,r}^{\varepsilon;\rho,\delta}(t)
 =\int_{a_\sigma<t<c_\sigma}
 \one_{\{t-r<a_\sigma,\ z_\sigma^-\ge\rho,\ \ell_\sigma\ge\delta\}}
          k_{r,t}(a_\sigma)f_\sigma^\varepsilon(a_\sigma+)\,d\sigma.
\end{equation}
Thus only entrances whose same selected segment reaches the target
are counted. A superscript $\sharp$ removes the ramp. In particular
no later re-entry is included.

\begin{theorem}[Complete noncritical entrance height]
\label{thm:v94-entrance-height}
For all sufficiently small fixed $r>0$ and every fixed
$\rho,\delta>0$,
\begin{equation}\label{eq:v94-entrance-height}
 \limsup_m\sup_{R,n,k,c}m^2
 \|B_{c,r}^{\varepsilon;\rho,\delta,\sharp}\|_\infty
                           \le C(r^{1/32}+r).
\end{equation}
The leading constant is independent of $\varepsilon$, $\rho$, $\delta$
and the Borel prefix. All noncritical nongrazing entrance types are
included. Each cutoff is positive and fixed before the collision
limit; the theorem does not assert the zero-cutoff conclusion.
\end{theorem}
\begin{proof}
It is enough to bound the larger source with entrance trace
$A_\sigma(a_\sigma+)$ and without the prefix restriction, since
$0\le W^\varepsilon\le1$. On a selected segment, radial monotonicity
keeps $z\ge\rho$. If $\ell_\sigma>2r$ and $a_\sigma\in(t-r,t)$,
absolute continuity gives
\[
 A_\sigma(a_\sigma+)\le
 \frac1r\int_{a_\sigma}^{a_\sigma+r}A_\sigma(u)\,du
 +C\int_{a_\sigma}^{a_\sigma+r}A_\sigma(u)I_m(u)/z(u)\,du.
\]
All these anchors lie in $(t-r,t+r)$ and have backward lifetime at
most $r$. The Borel segment map counts their interiors once. Summing
therefore bounds the long part by
\begin{equation}\label{eq:v94-long-inlet}
 \frac1r\int_{t-r}^{t+r}e_r^-(u)\,du
                         +C\int_{t-r}^{t+r}v^\rho(u)\,du.
\end{equation}
The ordered bound is $C(r^{1/32}+r)$ by the retained volume inputs.
This is a trace comparison with the original density, not an
identification of entrance flux and unweighted starting mass.

For a residence band $s<\ell_\sigma\le2s$, integrate the same
fundamental theorem over the complete segment:
\[
 A_\sigma(a_\sigma+)\le
 \frac1s\int_{a_\sigma}^{b_\sigma}A_\sigma(u)\,du
                  +C\int_{a_\sigma}^{b_\sigma}A_\sigma(u)I_m(u)/z(u)\,du.
\]
If the segment contains $t$, its interior lies in $(t-2s,t+2s)$ and
has backward lifetime at most $2s$. Complete summation gives
\begin{equation}\label{eq:v94-band-inlet}
 \int_{a_\sigma<t<b_\sigma}\one_{\{s<\ell_\sigma\le2s,\ z_\sigma^-\ge\rho\}}
          A_\sigma(a_\sigma+)\,d\sigma
 \le\frac1s\int_{t-2s}^{t+2s}e_{2s}^-+C\int_{t-2s}^{t+2s}v^\rho.
\end{equation}
Its ordered height is $C(s^{1/32}+s)$. For a previously fixed
$\delta>0$, only finitely many bands $s_j=2^{-j}r$ are needed to
cover $\delta\le\ell\le2r$. Sum them before the collision limsup.
Their leading bounds form convergent geometric series independent
of the number of bands. Combine with \eqref{eq:v94-long-inlet}.
No physical-word count or uniform estimate at $\rho=0$ was used.
\end{proof}

\subsection{The zero-residence inlet is an original source, not a new loss}
Keep $\rho>0$ fixed. For the full original source set
\begin{align}
 Z_\delta^{\varepsilon,\rho}(t)
 &=\int_{a_\sigma<t<b_\sigma}
       \one_{\{\ell_\sigma\le\delta,\ z_\sigma^-\ge\rho\}}
                         f_\sigma^\varepsilon(a_\sigma+)\,d\sigma,\notag\\
 Q_\delta^{\varepsilon,\rho}(t)
 &=\int_{a_\sigma<t<b_\sigma}
       \one_{\{\ell_\sigma\le\delta,\ z_\sigma^-\ge\rho\}}
                         f_\sigma^\varepsilon(t)\,d\sigma.
                                             \label{eq:v94-inlet-volume}
\end{align}
The second density is a restriction of the original weighted collision
source to the same actual segments, with no transported normalization.

\begin{proposition}[Exact identification of the noncritical inlet charge]
\label{prop:v94-inlet-charge}
For every fixed $\rho,\delta>0$,
\begin{equation}\label{eq:v94-inlet-volume-difference}
 |Z_\delta^{\varepsilon,\rho}(t)-Q_\delta^{\varepsilon,\rho}(t)|
 \le C(1+\varepsilon^{-1})\int_{t-\delta}^{t}v^\rho(u)\,du.
\end{equation}
Consequently their monotone zero-residence ordered heights agree:
\[
 \mathfrak E_{\varepsilon,M}^{\rho}
 :=\lim_{\delta\downarrow0}\mathcal H_M(Z_\delta^{\varepsilon,\rho})
  =\lim_{\delta\downarrow0}\mathcal H_M(Q_\delta^{\varepsilon,\rho}).
\]
If $B_r^{\varepsilon,\rho}$ is the full ramped entrance assignment
with $z_\sigma^-\ge\rho$ and no residence cutoff, then
\begin{equation}\label{eq:v94-inlet-sandwich}
 \mathfrak E_{\varepsilon,M}^{\rho}
 \le\mathcal H_M(B_r^{\varepsilon,\rho})
 \le\mathfrak E_{\varepsilon,M}^{\rho}+C(r^{1/32}+r).
\end{equation}
All assertions hold for extended nonnegative heights; their zero value
is not inferred from cutoff-independent leading constants.
\end{proposition}
\begin{proof}
On the selected segment $a_\sigma>t-\delta$ and $z\ge\rho$.
Integrate
$|(f_\sigma^\varepsilon)'|\le
 C(1+\varepsilon^{-1})A_\sigma I_m/z$
from $a_\sigma$ to $t$ and sum by the common Borel bijection.
This proves \eqref{eq:v94-inlet-volume-difference}. Its ordered norm
is at most $C(1+\varepsilon^{-1})\delta$, proving equality of the
two monotone limits by inequalities, without subtracting infinities.
For $0<\delta<r$ the ramp on every selected short segment lies between
$1-\delta/r$ and one. Hence
\[
 (1-\delta/r)Z_\delta^{\varepsilon,\rho}
 \le B_r^{\varepsilon,\rho}
 \le Z_\delta^{\varepsilon,\rho}
                   +B_r^{\varepsilon;\rho,\delta,\sharp}.
\]
Apply Theorem~\ref{thm:v94-entrance-height}, take the ordered height,
and only then decrease $\delta$. The first comparison remains valid
when the limiting height is infinite.
\end{proof}
