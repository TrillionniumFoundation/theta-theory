\section{Two-ended comparison on moving physical roofs}
\label{sec:v107-envelope}
Throughout this section the collision count, radius, exact labels and
$\varepsilon>0$ are fixed. All angular integrals are taken on the
original disjoint space $\Theta_{\rm phys}$ of incident pairing sheets. Put
\begin{equation}\label{eq:v107-envelope}
 \begin{aligned}
 f_-(\theta)&=f^\varepsilon(a+,\theta),&
 f_+(\theta)&=f^\varepsilon(b-,\theta),\\
 p_\vee(\theta)&=\max\{f_-(\theta),f_+(\theta)\},&
 e_\partial(\theta)&=|f_-(\theta)-f_+(\theta)|.
 \end{aligned}
\end{equation}
Thus $p_\vee=c^\varepsilon+e_\partial$.
The traces used below are the finite one-sided traces of the
inherited physical source. No critical-point formula is imposed.

\begin{lemma}[Endpoint passage]\label{lem:v107-endpoint}
On a regular segment where $f^\varepsilon$ is locally absolutely
continuous and has the stated one-sided traces,
\begin{equation}\label{eq:v107-endpoint-charge}
 e_\partial(\theta)\le C(1+\varepsilon^{-1})
       \int_{a(\theta)}^{b(\theta)} A(u,\theta)I_m(u,\theta)/z(u,\theta)\,du.
\end{equation}
\end{lemma}
\begin{proof}
On each compact subinterval the physical density $f^\varepsilon$
is absolutely continuous. Integrate
\eqref{eq:v104-weighted-variation-input} between two interior
points, then let the points tend monotonically to the two ends.
The left side converges by the one-sided trace convention and the
right side by monotone convergence. If that integral is infinite,
the asserted nonnegative inequality is still valid. Neither the
minimum, the maximum, nor a target selector has been differentiated.
\end{proof}

\begin{definition}[Physical comparison family]\label{def:v107-family}
Fix $0<r<\Delta$, $0<N<\infty$ and $0<\kappa<\infty$.
Let $I_j$ be countably many disjoint Borel angular source sets,
let $0\le s_j\le1$ be supported on $I_j$, and let $0\le w_j\le1$
be Borel comparison weights. For each $j$ choose $\tau_j\in\{-1,1\}$.
The following conditions refer to actual maximal physical segments:
\begin{align}
 &0<b(\eta)-a(\eta)<\Delta
                   &&\text{on }\{w_j>0\},\label{eq:v107-length}\\
 &\sum_j w_j(\eta)\le N
                   &&\text{for almost every }\eta\in\Theta_{\rm phys}.
                                      \label{eq:v107-overlap}
\end{align}
For product-almost every $(\xi,\eta)\in I_j\times\{w_j>0\}$ require
\begin{equation}\label{eq:v107-inclusion}
 (a(\xi),b(\xi))+\tau_j[0,r]\subset(a(\eta),b(\eta)).
\end{equation}
Here product-almost everywhere uses angular measure in the first
variable and $w_j(\eta)\,d\eta$ in the second. Define the finite
nonnegative capacities
\begin{equation}\label{eq:v107-capacities}
 \begin{aligned}
 C_j&=\int_{I_j}s_jc^\varepsilon\,d\theta,&
 P_j&=\int w_jp_\vee\,d\theta,\\
 D_j&=\int w_jc^\varepsilon\,d\theta,&
 E_j&=\int w_je_\partial\,d\theta.
 \end{aligned}
\end{equation}
In particular $P_j=D_j+E_j$. The same fixed exact labels are used
on the source and comparison sides.
\end{definition}

For a Borel selector $0\le H\le1$, held at its original target roof,
write
\[
 T_j(t)=\int_{I_j}s_j(\theta)c^\varepsilon(\theta)
       \one_{\{a(\theta)<t<b(\theta)\}}H(\Psi(\theta,t))\,d\theta.
\]
Weights $s_j,w_j$ do not depend on the target roof. In contrast,
$H$ may be any of the original roof-dependent failure selectors.

\begin{lemma}[Common window without a common entrance]
\label{lem:v107-window}
If $T_j(t)>0$, then for $w_j\,d\eta$-almost every comparison sheet
\begin{equation}\label{eq:v107-window}
 t+\tau_j[0,r]\subset(a(\eta),b(\eta))
                    \subset(t-\Delta,t+\Delta).
\end{equation}
\end{lemma}
\begin{proof}
By Fubini, outside a fixed angular null set of source points,
\eqref{eq:v107-inclusion} holds for almost every comparison point.
Positivity of $T_j(t)$ implies that the source points whose segment
contains $t$ and whose integrand is positive form a set of positive
angular measure. It therefore contains a point outside that null
set. Apply the interval inclusion at this point. Since the comparison
segment contains $t$ and has length less than $\Delta$, both its
ends lie within distance $\Delta$ of $t$.
For a countable family the angular exceptional sets can be combined
before choosing $t$. The inherited coarea identities are then used
on their common roof versions at fixed $m,R,n,k$. This does not
assert a null set uniform over all radii.
\end{proof}

\begin{theorem}[Two-ended positive-envelope estimate]
\label{thm:v107-envelope}
Let Definition~\ref{def:v107-family} hold and suppose
\begin{equation}\label{eq:v107-balance}
 C_j\le\kappa P_j\qquad\text{for every }j.
\end{equation}
Then, on the original roof kernels at finite count,
\begin{align}
 \sum_j T_j(t)
 &\le {N\kappa\over r}\int_{t-r}^{t+r}
                  \mathsf T_\Delta^\varepsilon(u)\,du\notag\\
 &\quad+C N\kappa(1+\varepsilon^{-1})
                      \int_{t-\Delta}^{t+\Delta}v^0(u)\,du.
                                      \label{eq:v107-finite}
\end{align}
If the hypotheses hold uniformly with previously fixed parameters,
\begin{equation}\label{eq:v107-height}
 \mathcal H_M\Bigl(\sum_jT_j\Bigr)
 \le C N\kappa\{\varepsilon^{1/16}
                         +(1+\varepsilon^{-1})\Delta\}.
\end{equation}
There is no lower bound on $D_j/P_j$ and neither endpoint roof
is required to be stationary.
\end{theorem}
\begin{proof}
Blocks with $C_j=0$ contribute nothing. On every other block set
$\alpha_j=\min\{1,\kappa D_j/C_j\}$. The balance and $P_j=D_j+E_j$
give the exact algebraic bound
\begin{equation}\label{eq:v107-shortage}
 (1-\alpha_j)C_j=(C_j-\kappa D_j)_+\le\kappa E_j.
\end{equation}
At an active roof $t$, Lemma~\ref{lem:v107-window} gives
\[
 rD_j\le\int_{t+\tau_j(0,r)}
           \int w_j(\eta)c^\varepsilon(\eta)
              \one_{\{a(\eta)<u<b(\eta)\}}\,d\eta\,du.
\]
Together with $T_j(t)\le C_j$ this bounds $\alpha_jT_j(t)$.
Sum over active blocks, extend both oriented windows to $(t-r,t+r)$,
and use \eqref{eq:v107-length}--\eqref{eq:v107-overlap}.
The result is the first term in \eqref{eq:v107-finite}.

For the complementary part use \eqref{eq:v107-shortage}, then
Lemma~\ref{lem:v107-endpoint}. By Lemma~\ref{lem:v107-window}
the entire variation integral for every active comparison sheet lies
in $(t-\Delta,t+\Delta)$. Consequently Tonelli's theorem and
physical overlap give
\[
 \sum_{j:T_j(t)>0}E_j
 \le C N(1+\varepsilon^{-1})
   \int_{\Theta_{\rm phys}}\int_{t-\Delta}^{t+\Delta}
       \one_{\{a(\eta)<u<b(\eta)\}}
                A(u,\eta)I_m(u,\eta)/z(u,\eta)\,du\,d\eta.
\]
The last physical integral is the original uncut volume integral
in \eqref{eq:v104-uncut-input}. This proves the second term.
For a countable family first sum over finitely many blocks and
increase that family; all terms are nonnegative and the same global
bound applies. No limit superior is interchanged with that sum.

Finally the first term of \eqref{eq:v107-finite} is
$2N\kappa A_r\mathsf T_\Delta^\varepsilon(t)$ and the second is
$2C N\kappa(1+\varepsilon^{-1})\Delta A_\Delta v^0(t)$.
Apply \eqref{eq:v104-paired-average} and
\eqref{eq:v104-uncut-input} at those previously fixed widths.
\end{proof}

\begin{corollary}[Nested moving roofs]\label{cor:v107-nested}
On an actual angular arm suppose $a(x)$ is nonincreasing and $b(x)$
is nondecreasing. Let $I=(0,x_1)$ and $J=(x_2,x_3)$, where
$x_1<x_2<x_3$ are attained before any true physical boundary.
Suppose $b(x_2)\ge b(x_1)+r$, comparison lengths are below $\Delta$,
and $\int_Ic^\varepsilon\le\kappa\int_Jp_\vee$.
Then this arm satisfies the positive-orientation hypotheses of
Theorem~\ref{thm:v107-envelope}. A family of such arms has the
stated estimate when its full comparison arms satisfy the physical
overlap bound. Disjoint source intervals alone do not imply that bound.
\end{corollary}
\begin{proof}
For $\xi\in I$ and $\eta\in J$, monotonicity gives
$a(\eta)\le a(\xi)$ and $b(\eta)\ge b(\xi)+r$.
Every point of the translated open source segment is therefore
strictly between the comparison ends. The capacity hypothesis is
\eqref{eq:v107-balance}, with the indicator of $J$ as comparison weight.
\end{proof}

\begin{proposition}[An actually reached level-width criterion]
\label{prop:v107-drifting}
Suppose an actual arm has
\[
 a(x)=a_*-\eta\psi(x),\qquad b(x)=a_*+d+\psi(x),\qquad
 0\le\eta\le1,\quad0\le d\le\delta/2,
\]
where $\psi$ is strictly increasing and reaches $\delta/2,2\delta,3\delta$
at $x_1,x_2,x_3$ before a true boundary. Assume $\psi$ is
continuously differentiable on $[x_2,x_3]$ and
$x\psi'(x)/\psi(x)\le K$ there, with $K>0$.
Suppose, for some $Q,L,l>0$, that $c^\varepsilon\le QL$ on $I=(0,x_1)$
and $p_\vee\ge Ql$ on $J=(x_2,x_3)$. Then the preceding corollary applies
with
\begin{equation}\label{eq:v107-drifting-cost}
 r=\delta,\qquad\Delta=8\delta,\qquad
 \kappa={LK\over l\log(3/2)}.
\end{equation}
\end{proposition}
\begin{proof}
The outgoing increment from $x_1$ to $x_2$ is $3\delta/2$.
On $J$ the residence is $d+(1+\eta)\psi(x)<13\delta/2<8\delta$.
Moreover
\[
 \log(3/2)=\int_{x_2}^{x_3}{\psi'(x)\over\psi(x)}\,dx
 \le K\log(x_3/x_2)
 \le K(x_3-x_2)/x_1.
\]
Hence $\int_Ic^\varepsilon\le QLx_1\le\kappa Ql(x_3-x_2)
\le\kappa\int_Jp_\vee$, as required.
\end{proof}

\begin{remark}[Scope and rates]\label{rem:v107-scope}
For $\eta>0$ both roofs in Proposition~\ref{prop:v107-drifting}
move. Taking $\psi(x)=x^k$ verifies its geometry with $K=k$,
including zero endpoint velocities at $x=0$ when $k>1$.
This is a diagnostic model of the comparison, not an asserted new
Lorentz realization or a uniform bound on the orders of all arms.
The theorem on actual arms uses exactly the stated reach, trace and
overlap hypotheses. It does not extend a segment past its physical
boundary.

Under \eqref{eq:v104-schedule}, $\Delta\le8\delta$ gives
$\varepsilon^{1/16}+(1+\varepsilon^{-1})\Delta=O(\varepsilon^{1/16})$.
Thus bounded $N\kappa$ gives $O(B^{-1/192})$; the larger budget
$N\kappa=O(\varepsilon^{-1/32})$ gives $O(B^{-1/384})$.
The width $r$ may be arbitrarily small at a fixed outer stage but
must be fixed before the collision limit and uniform over its family.
\end{remark}
