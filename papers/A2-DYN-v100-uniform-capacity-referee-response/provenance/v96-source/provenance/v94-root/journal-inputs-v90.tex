\section{The physical and analytic inputs used in the proof}
\label{sec:v90-inputs}
All new theorems use the following source-pinned assertions. They are
stated in their original physical normalization, with finite-count
remainders retained. The underlying anisotropic billiard literature
\cite{DPZ} is background; the four-record and new source assertions
are the repository's own explicitly inherited inputs.

\begin{inputstatement}[Same-label physical segments]
\label{input:v90-geometry}
Fix $m,R,n,k$. The regular exact event $\Omega$ is partitioned by a
Borel bijection, with Borel inverse, into maximal roof intervals
$a_\sigma<u<b_\sigma$ and disjoint angular charts. In endpoint angles
$q=(\alpha_0,\alpha_m)$, put
\[
 F=L,\quad g=\nabla_qF=R(-p_0,p_m),\quad z=|g|,\quad
 K=D_q^2F,\quad\vartheta=\arg g,
\]
\[
 \lambda_\circ=9\sqrt{19}/200,\quad\Lambda_\circ=47/6,\quad
 \omega=|\partial_{p_0}\alpha_m|^{-1},\quad
 \chi=|\det D_q(F,\vartheta)|.
\]
The bounds $\lambda_\circ I\le K\le\Lambda_\circ I$ and
$\lambda_\circ\le\chi\le\Lambda_\circ$ hold on all regular pieces.
The field $X_q=K^{-1}g/(g^{\mathsf T}K^{-1}g)$ satisfies
$XF=1$, $X\vartheta=0$, and
\begin{equation}\label{eq:v83-radial-input}
 2\lambda_\circ h\le z(\Phi_hx)^2-z(x)^2\le2\Lambda_\circ h.
\end{equation}
A same-segment motion of roof length $h$ has endpoint-curve length
at most $C\sqrt h$. Contact and flight-margin first derivatives in
endpoint angles are bounded by $Cq_0^{d_j}$ and $Cq_0^{d'_j}$.
Thus every true forward end within $h$ is in a layer
\begin{equation}\label{eq:v83-layer-input}
 c_j\le C_Eq_0^{d_j}\sqrt h,\quad
 \Delta_j\le C_Eq_0^{d'_j}\sqrt h,\quad
 \operatorname{dist}(T_R^jx,\partial Y_R^*)\le C_Eq_0^{d_j}\sqrt h.
\end{equation}
Forward critical termination cannot occur. The physical angular density
and its derivative are
\[
 A_\sigma(u)=\frac{\omega(\Psi_\sigma(u))}{4\pi c_*\chi(\Psi_\sigma(u))},
 \qquad |A_\sigma'(u)|\le C A_\sigma(u)I_m(u)/z(u),\qquad
 I_m=1+\sum_{i=1}^{m-1}q_0^{d_i}/c_i.
\]
The Borel disintegration identifies $A_\sigma(u)du\,d\sigma$ with
$d\bar\nu=\nu_R^*|_\Omega$, and its roof fibers with $d\mu_u$.
It counts each interior physical point once, retaining its incident
component. Artificial angular-chart seams are not ends.
\end{inputstatement}
These are the retained endpoint-field, all-contact, cubic-action and
Borel-segment theorems. In particular the Borel map is Supplement
Theorem~\ref{supp-v85archive:thm:v85-segment-map}. Its countable
kernel convention is used throughout: for fixed $m,R,n,k$ one fixes
joint measurable fiber and time kernels, not a separate density version
for each restriction. The parameter order remains $\sup_R\esssup_t$;
no common null set for every radius is asserted.

\begin{inputstatement}[Finite-count positive local bounds]
\label{input:v90-spectral}
At each fixed auxiliary band $B_*$ there are constants $C_{B_*}$
and $\rho_{B_*}<1$ such that, uniformly in the interval $J$,
finite collision count and exact labels, the following hold. A marked
section strip $S_{R,s}$ of width $Cs$ obeys
\begin{equation}\label{eq:v83-strip-input}
 m^2\nu\{K_m^{\rm c}=k,A_m=n,L_m\in J,T_R^jx\in S_{R,s}\}
 \le Cs^{1/16}(|J|+B_*^{-1})
       +C_{B_*}(1+B_*|J|)m^2\rho_{B_*}^{m/2}.
\end{equation}
The original bad source satisfies
\begin{equation}\label{eq:v90-bad-input}
 m^2\int_J b^\delta\le C\delta^{1/16}(|J|+B_*^{-1})
       +C_{B_*}(1+B_*|J|)m^3\rho_{B_*}^{m/2}.
\end{equation}
The extension to a fixed remaining compact range of $\delta$ is
included. At parameter one, $b^1=p$. The endpoint-only estimate has
leading factor $\zeta^{129/128}$ and remainder
$C_{B_*}(1+B_*|J|)m^2\zeta\rho_{B_*}^{m/2}$.
The actual fractional source
\[
 v(t)dt=F_\#(I_mz^{-1/4}\bar\nu)
\]
is finite at every count; after enlarging constants its complete
finite-count local bound has the form \eqref{eq:v90-volume-finite}.
No untruncated first inverse-gradient bound is part of this assertion.
\end{inputstatement}
The strip and bad-source results are Supplement
Theorems~\ref{supp-thm:v48-marked-strip-local} and
\ref{supp-thm:v49-physical-remainder-local}. The endpoint-only bound
is Supplement equation~\eqref{supp-v86archive:eq:v86-endpoint-input}.
For the complete finite-count fractional bound one may use Supplement
equation~\eqref{supp-v86archive:eq:v86-full-moment} with
$p=33/32$ and $\eta=1/4$, since $I_m\le I_m^p$ and
$1/4<1-p/2$. Its exponential remainder is enlarged to the common
one in \eqref{eq:v90-volume-finite}, not discarded. The actual
negative-residence source and physical clearance cap are retained in
module 205, compiled in the full preceding article.

\begin{inputstatement}[Original source and scalar endpoint]
\label{input:v90-scalar}
The bad-source density $b^\varepsilon$ is nonnegative and compactly
supported at finite count. For $\varepsilon(B)=A_0B^{-1/12}$ and
large fixed reconstruction band $B$,
\begin{equation}\label{eq:v76-remainder}
 m^2p-\mathcal L=m^2b^{\varepsilon(B)}+e_B,\qquad
 \limsup_m\sup_{R,n,k}\esssup_{\mathcal C(M)}|e_B|
                                  \le C_MB^{-1/192}.
\end{equation}
For the original exceptional union in \eqref{eq:v90-original-union},
let $\mathcal H_M(f)=\limsup_m\sup_{R,n,k}m^2\esssup_{\mathcal C(M)}f$.
Its cofinal charge satisfies
\[
 \mathfrak R_{\varepsilon,M}
   =\inf_{a,\delta>0}\mathcal H_M(R_{a,\delta}^\varepsilon),\qquad
 \mathcal E_M=\lim_{B\to\infty}\mathfrak R_{\varepsilon(B),M}.
\]
Neither charge is asserted zero by this input.
\end{inputstatement}
The scalar remainder is Supplement Theorem~\ref{supp-thm:v51-positive-error},
and the exact original-source equivalence is Supplement
Theorem~\ref{supp-v87archive:thm:v87-endpoint}. The critical weak
boundedness, logarithmically weakened critical convergence and full
central-window theorem of the preceding article are preserved. The last
uses the diffusive measure $dt/\sqrt m$, not unnormalized $dt$.
