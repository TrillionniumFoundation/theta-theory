\section{Mixed critical volume insertions}
\label{sec:v81-mixed}
The joint estimate has a critical-width gain greater than one.
It supplies a uniform estimate for the first inverse gradient with
a fixed positive truncation, and actual untruncated fractional
inverse-gradient insertions. The distinction is visible in the
finite-count remainder and will be retained in the roof-height argument.

Keep the original additive insertion $I_m$ of
\eqref{eq:v80-depth-insertion}. Write $z=|g|$ on the regular exact
event and, for a small fixed $b>0$, define
\begin{equation}\label{eq:v81-critical-volume}
 u^b(t)\,dt=L_\#(I_m\one_{\{z<b\}}\one_\Pi\nu_R^*).
\end{equation}
The threshold $b$ now measures $|g|$, not the momentum coordinate.
Since $\{z<b\}\subset D_{b/R_-}$, the difference changes only
uniform constants. There is a fixed $b_1>0$ for which the following
statements hold for $b<b_1$.

\begin{theorem}[Critical localization of the complete optical source]
\label{thm:v81-critical-volume}
For every fixed band $B_*$,
\begin{align}
 m^2\int_Ju^b(t)\,dt
 \le{}&Cb^\alpha(|J|+B_*^{-1})\notag\\
 &+C_{B_*}(1+B_*|J|)m^2 b^{1/2}\rho_{B_*}^{m/4}.
                                      \label{eq:v81-critical-finite}
\end{align}
Consequently, for every previously fixed roof width $r>0$,
\begin{equation}\label{eq:v81-critical-average}
       \limsup_m\sup_{R,n,k}m^2\|A_ru^b\|_\infty\le Cb^\alpha.
\end{equation}
Every inverse-incidence scale and every actual word is included
before the collision limit.
\end{theorem}
\begin{proof}
Insert the positive layer cake for
$c_j^{-1}\one_{\{c_j<s_0\}}$ into
\eqref{eq:v81-joint-local}. Its leading integral is finite because
$\beta>1$. In the capped remainder put $D=b\rho_{B_*}^{m/2}$.
Equation~\eqref{eq:v80-cap-integral}, with exponent one, bounds the
entire remainder by $C\sqrt D$; the cap is integrated before the
collision limit. This gives the stated $b^{1/2}$ error.

For the bounded complementary part of $c_j^{-1}$ and the constant
one, the endpoint-only pairing is $\ell(P_bQ^mP_b\one)$.
Its leading amplitude is at most
$Cb^{7/8+63/64}$, and its remainder is at most
$Cb\rho^m$. These are bounded by the respective terms in
\eqref{eq:v81-critical-finite}. Sum the internal marks with
weights $q_0^{d_j}$, whose sum is uniformly bounded. Dropping only
the section tests in each positive comparison retains $D_{b/R_-}$,
the mark and the same exact $(k,n,m)$ coefficient, with the sole
normalization factor $c_*^{-1}$. It does not alter the defined source.
Finally divide by $2r$, take the collision limit with $B_*$ fixed,
and enlarge that band afterwards. This proves
\eqref{eq:v81-critical-average}.
\end{proof}

\begin{theorem}[Actual fractional and truncated first inverse gradients]
\label{thm:v81-gradient-insertions}
For $0<\eta<1/2$, the actual measure
\begin{equation}\label{eq:v81-fractional-source}
 v^{\eta,b}(t)\,dt
   =L_\#(I_m z^{-\eta}\one_{\{0<z<b\}}\one_\Pi\nu_R^*)
\end{equation}
is finite at each count and satisfies
\begin{align}
 m^2\int_Jv^{\eta,b}(t)\,dt
 \le{}&C_\eta b^{\alpha-\eta}(|J|+B_*^{-1})\notag\\
 &+C_{B_*,\eta}(1+B_*|J|)m^2
                 b^{1/2-\eta}\rho_{B_*}^{m/4}.
                                      \label{eq:v81-fractional-finite}
\end{align}
In particular its fixed-width collision-normalized height is at
most $C_\eta b^{\alpha-\eta}$.

For $0<a<b<b_1$, put
\begin{equation}\label{eq:v81-truncated-source}
 v^{a,b}(t)\,dt
   =L_\#(I_m z^{-1}\one_{\{a\le z<b\}}\one_\Pi\nu_R^*).
\end{equation}
This source obeys
\begin{align}
 m^2\int_Jv^{a,b}(t)\,dt
 \le{}&Cb^\delta(|J|+B_*^{-1})\notag\\
 &+C_{B_*}(1+B_*|J|)m^2 a^{-1/2}\rho_{B_*}^{m/4}.
                                      \label{eq:v81-truncated-finite}
\end{align}
Thus, for each fixed $a,b,r>0$,
\begin{equation}\label{eq:v81-truncated-average}
 \limsup_m\sup_{R,n,k}m^2\|A_rv^{a,b}\|_\infty\le Cb^\delta,
 \qquad \delta=1/128,
\end{equation}
with a leading constant independent of the lower truncation $a$.
\end{theorem}
\begin{proof}
For the fractional assertion use
\[
 z^{-\eta}\one_{\{z<b\}}
   =b^{-\eta}\one_{\{z<b\}}
       +\eta\int_0^b q^{-\eta-1}\one_{\{z<q\}}\,dq
\]
on $z>0$. Apply \eqref{eq:v81-critical-finite} first with a positive
lower integration limit, and then use monotone convergence. Both
integrals are finite: their exponents are $\alpha-\eta>0$ and
$1/2-\eta>0$. This proves physical finiteness as well as
\eqref{eq:v81-fractional-finite}, not just convergence of a
formal leading term. The removed set $g=0$ has zero source area;
no value assigned there affects the measure.

For the first inverse power a convenient positive majorant is
\[
 z^{-1}\one_{\{a\le z<b\}}
 \le b^{-1}\one_{\{z<b\}}+
                    \int_a^b q^{-2}\one_{\{z<q\}}\,dq.
\]
The leading term in \eqref{eq:v81-critical-finite} gives
$C\{b^{\alpha-1}+\int_a^b q^{\alpha-2}\,dq\}\le Cb^\delta$.
The error gives
$C\{b^{-1/2}+\int_a^b q^{-3/2}\,dq\}\le Ca^{-1/2}$.
This proves \eqref{eq:v81-truncated-finite}. The ordered fixed-width
limits prove the remaining assertions.
\end{proof}

Let $v^a$ be the same exact source with insertion
$I_m z^{-1}\one_{\{z\ge a\}}$. Split it at $b_1/2$, and use
\eqref{eq:v80-depth-source} on the part above that threshold.
For every fixed $a,r>0$ this gives
\begin{equation}\label{eq:v81-full-truncation}
             \limsup_m\sup_{R,n,k}m^2\|A_rv^a\|_\infty\le C,
\end{equation}
where $C$ is independent of $a$. This statement does not assert the
same conclusion for the untruncated first inverse gradient. The
factor $a^{-1/2}$ in \eqref{eq:v81-truncated-finite} prevents taking
$a\downarrow0$ before the collision limit by this proof. By contrast,
\eqref{eq:v81-fractional-finite} is an actual untruncated insertion
throughout the stated range $\eta<1/2$.
