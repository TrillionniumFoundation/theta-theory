\begin{abstract}
We give an exact deterministic first-principles construction of a path
canonical ensemble.  The microscopic system is an invertible area-preserving
four-branch baker collision map with its natural Lebesgue probability.  A
branch current is an actual signed winding observable.  Its empirical mean
satisfies an explicit large-deviation principle.  Conditioning on an atypical
mean current produces, with a quantitative finite-window bound, a unique
Bernoulli-Gibbs path law.  The conjugate parameter is not selected externally:
\[
 \theta(a)=I'(a)=\frac12\log\frac{3(1+a)}{7(1-a)}.
\]
The canonical tilted law is exactly the Lebesgue branch law of another member
of the same moving-seam collision family.  Thus microcanonical conditioning,
Doob driving, moving-boundary response, and the later theta-semigroup all use
one typed deterministic platform.

For any bounded terminal work measured in the same units as the current, the
nonlinear expectation is the excess canonical path pressure, not a postulated
utility.  We prove exact dynamic composition of this pressure increment.  The
similarity Lorentz and open-billiard platforms remain comparison and physical
application branches but are not imported into the first-principles theorem.
\end{abstract}

\section{The deterministic collision family}
For $a\in(-1,1)$ put
\[
 w_1(a)=\frac{1-a}{6},\quad
 w_2(a)=\frac{1-a}{3},\quad
 w_3(a)=\frac{3(1+a)}{14},\quad
 w_4(a)=\frac{2(1+a)}7.
\]
Let $s_0=0$ and $s_i=\sum_{j\le i}w_j$.  On
$I_i(a)=[s_{i-1}(a),s_i(a))$ define
\[
 B_a(x,y)=\left(
 \frac{x-s_{i-1}(a)}{w_i(a)},
 s_{i-1}(a)+w_i(a)y
 \right).
\]
The phase space is the unit square modulo boundary identifications.

\begin{proposition}[Natural Liouville law]\label{prop:natural}
The map $B_a$ is invertible away from its seams, preserves Lebesgue area, and
is Bernoulli with branch weights $w(a)$.  For every cylinder
$[i_0\ldots i_{n-1}]$,
\[
 m_a[i_0\ldots i_{n-1}]=\prod_{k=0}^{n-1}w_{i_k}(a).
\]
Lebesgue measure is ergodic and its basin has full Lebesgue measure.
\end{proposition}
\begin{proof}
On branch $i$ the derivative is
$\operatorname{diag}(w_i^{-1},w_i)$ and has determinant one.  The four images
are the horizontal strips $[0,1]\times I_i(a)$ and partition the square.
Repeated inverse images of a branch cylinder have width equal to the product
of the declared branch widths.  Hence future branch symbols are independent
with law $w(a)$.  The usual generating partition argument gives the Bernoulli
property and ergodicity.
\end{proof}

The base preparation is $a_0=2/5$, for which
\[
 w(a_0)=(1/10,1/5,3/10,2/5).
\]
Define the mechanical signed winding current
\[
 c_1=c_2=-1,\qquad c_3=c_4=1,
\]
and
\[
 A_n=\sum_{k=0}^{n-1}c_{I_k}.
\]
The typical current under $m_{a_0}$ is $a_0$.

\section{The exact current large deviation principle}
Let $q=(1+a)/2$ and $q_0=7/10$.  The scaled cumulant generating function is
\[
 \Lambda(\theta)
 =\log\left(\frac3{10}e^{-\theta}+\frac7{10}e^\theta\right).
\]

\begin{theorem}[Explicit current LDP]\label{thm:ldp}
The variables $A_n/n$ satisfy an LDP on $[-1,1]$ with good strictly convex rate
\[
 I(a)=\frac{1+a}{2}\log\frac{1+a}{7/5}
      +\frac{1-a}{2}\log\frac{1-a}{3/5}.
\]
For $a\in(-1,1)$,
\[
 I'(a)=\theta(a)=\frac12\log\frac{3(1+a)}{7(1-a)},
 \qquad I''(a)=\frac1{1-a^2}>0,
\]
and $\Lambda'(\theta(a))=a$.
\end{theorem}
\begin{proof}
The signs $c_{I_k}$ are iid, with positive sign probability $q_0=7/10$.
Cramer's theorem gives the binary relative-entropy rate.  Direct
differentiation gives the displayed formulas and strict convexity
\cite{DemboZeitouni1998,Kifer1990}.
\end{proof}

\section{Quantitative microcanonical conditioning}
Let $K_n=(n+A_n)/2$ be the number of positive-current branches.  For an
admissible integer $k$ let $\mu_{n,k}$ be the branch-path law conditioned on
$K_n=k$.

\begin{theorem}[Finite-window ensemble equivalence]\label{thm:microcanonical}
Put $q_n=k/n$ and $a_n=2q_n-1$.  For every $\ell<n$, the total-variation
distance between the first $\ell$ branches under $\mu_{n,k}$ and iid branches
with weights $w(a_n)$ satisfies
\[
 \left\|
 \mu_{n,k}^{(\ell)}-m_{a_n}^{(\ell)}
 \right\|_{\TV}
 \le
 \sum_{r=0}^{\ell-1}\frac{r}{n-r}
 \le \frac{\ell(\ell-1)}{2(n-\ell+1)}.
\]
Consequently, if $k_n/n\to q$, every fixed path window converges to the
Lebesgue branch law of $B_{2q-1}$.
\end{theorem}
\begin{proof}
Conditioned on $K_n=k$, the sign sequence is sampling without replacement
from $k$ positive and $n-k$ negative symbols.  At stage $r$, after $u$
positive signs, the conditional positive probability is $(k-u)/(n-r)$; its
difference from $q_n$ is at most $r/(n-r)$.  Sequential maximal coupling gives
the sum bound.  Conditional on each sign, the negative subtypes have fixed
probabilities $(1/3,2/3)$ and the positive subtypes have fixed probabilities
$(3/7,4/7)$, both before and after conditioning.  Hence the same bound holds
for the four branches.
\end{proof}

\section{Canonical tilt and deterministic driven realization}
For $\theta\in\R$ define
\[
 \frac{d\mu_{\theta,n}}{dm_{a_0}^{(n)}}
 =\exp\{\theta A_n-n\Lambda(\theta)\}.
\]

\begin{theorem}[Exact driven-map theorem]\label{thm:driven}
Let $a=\Lambda'(\theta)$.  For every finite $n$, $\mu_{\theta,n}$ is the iid
branch law with weights $w(a)$.  Equivalently, it is the $n$-step branch law of
the deterministic map $B_a$ under its natural Lebesgue measure.  Thus the
canonical path ensemble has an exact deterministic driven realization.
\end{theorem}
\begin{proof}
Multiplication by $e^{\theta c_i}$ changes the total negative and positive
weights to $(3/10)e^{-\theta}$ and $(7/10)e^\theta$.  Normalization gives
positive probability
\[
 q(\theta)=\frac{(7/10)e^\theta}
 {(3/10)e^{-\theta}+(7/10)e^\theta}
 =\frac{1+a}{2}.
\]
The subtype ratios within each sign remain $(1:2)$ and $(3:4)$, hence the four
weights are exactly $w(a)$.  Independence is preserved by product tilting.
\end{proof}

\section{Canonical excess pressure and theta-expectation}
Let $F$ be a bounded terminal mechanical work, in the same current units.
For $\theta\ne0$ define
\[
 \cE_{0,n}^{\theta}[F]
 =\frac1\theta\log
 \frac{
 \E_{m_{a_0}}e^{\theta(A_n+F)}
 }{
 \E_{m_{a_0}}e^{\theta A_n}
 }.
\]

\begin{theorem}[Excess-pressure identity]\label{thm:excess}
If $a=\Lambda'(\theta)$, then
\[
 \cE_{0,n}^{\theta}[F]
 =\frac1\theta\log\E_{m_a}e^{\theta F}.
\]
For the branch filtration, the conditional operators
\[
 \cE_{r,t}^{\theta}[F]
 =\frac1\theta\log
 \E_{m_a}\left[e^{\theta F}\mid\mathcal F_r\right]
\]
are monotone, cash additive, constant preserving, and satisfy the exact tower
law
\[
 \cE_{r,t}^{\theta}
 =\cE_{r,s}^{\theta}\circ\cE_{s,t}^{\theta}.
\]
At $\theta=0$ the continuous limit is conditional expectation.
\end{theorem}
\begin{proof}
The Radon--Nikodym derivative in \cref{thm:driven} gives the first identity.
The conditional tower property follows by exponentiating, applying the linear
conditional-expectation tower, and taking the logarithm.
\end{proof}

\begin{remark}[Why theta is endogenous]
The macro-preparation specifies the atypical current $a$.  The conjugate field
is then uniquely fixed by $\theta=I'(a)$.  No risk parameter is inserted after
the path ensemble is chosen.  A different unit for terminal work introduces
only the corresponding physical conversion factor.
\end{remark}

\section{Moving seams and all-order response}
The inverse branches are
\[
 \psi_i^a(y)=s_{i-1}(a)+w_i(a)y,
 \qquad
 (L_ah)(y)=\sum_iw_i(a)h(\psi_i^a(y)).
\]
All $w_i$ and $\psi_i^a$ are affine in $a$.

For $r\ge1$ define the centered space
\[
 X_r=\left\{h\in W^{r,1}([0,1]):\int_0^1h=0\right\},
 \qquad
 \norm h_{X_r}=\sum_{j=1}^r\norm{h^{(j)}}_{L^1}.
\]
By the one-dimensional Poincare inequality this is equivalent to the usual
$W^{r,1}$ norm on $X_r$.

\begin{theorem}[All-order moving-seam response]\label{thm:response}
For every $r\ge1$, $k\ge0$, and compact $K\Subset(-1,1)$,
\[
 \partial_a^kL_a:W^{r+k,1}\longrightarrow W^{r,1}
\]
is uniformly bounded for $a\in K$.  Moreover,
\[
 \norm{L_a^mh}_{X_r}
 \le \rho_K^m\norm h_{X_r},
 \qquad
 \rho_K=\sup_{a\in K}\max_iw_i(a)<1,
\]
for every $h\in X_r$.  Consequently every finite reduced-resolvent word,
bilateral response array, and finite parameter difference has a summable
product tail on the declared graded ladder.
\end{theorem}
\begin{proof}
Repeated differentiation of $w_i(a)h(\psi_i^a)$ loses one Sobolev derivative
per parameter derivative and produces finitely many chain-rule terms.  For
$j\ge1$,
\[
 (L_ah)^{(j)}(y)=\sum_iw_i(a)^{j+1}h^{(j)}(\psi_i^a(y)).
\]
After the change of variables $x=\psi_i^a(y)$,
\[
 \norm{(L_ah)^{(j)}}_{L^1}
 \le \max_iw_i(a)^j\norm{h^{(j)}}_{L^1}
 \le \rho_K\norm{h^{(j)}}_{L^1}.
\]
Mass preservation leaves $X_r$ invariant, proving the contraction.  Apply it
before and after each response letter.  The reduced resolvent is the geometric
series on $X_r$; finite words are therefore well typed and product tails are
summable.  Taylor's formula and the common tail give complete
finite-difference convergence.
\end{proof}

\section{Main first-principles theorem}
\begin{theorem}[Deterministic path-ensemble closure]\label{thm:main}
On the single family $B_a$:
\begin{enumerate}[label=(\roman*)]
\item microscopic dynamics is deterministic and area preserving;
\item the reference probability is the natural full-volume Lebesgue law;
\item the current LDP and its rate are explicit;
\item a macro current $a$ determines the unique $\theta=I'(a)$;
\item microcanonical conditioning converges quantitatively to the driven law;
\item that law is realized by the deterministic moving-seam map $B_a$;
\item theta-expectation is the exact excess path pressure;
\item the response calculus remains on the same family.
\end{enumerate}
There is no arbitrary transition matrix, microscopic noise, or independent
risk parameter in this theorem.
\end{theorem}
\begin{proof}
Combine \cref{prop:natural,thm:ldp,thm:microcanonical,thm:driven,thm:excess,thm:response}.
\end{proof}

\section*{Scope and review status}
The theorem is a complete internal proof in the declared deterministic
full-cover collision scope.  It does not claim that an arbitrary periodic
Sinai billiard has the same exact symbolic reset or that independent external
review has occurred.
