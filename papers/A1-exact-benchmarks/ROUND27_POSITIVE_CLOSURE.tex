\section{Round-twenty-seven reconstruction: coherent probability fibres and certified current limits}
\label{sec:r27-a1}

The purpose of this paper is to place parameter response and Hilbert-valued
functional limits on one coherent domain.  Infinite Bernoulli products with
different symbol weights are usually mutually singular, so no Radon--Nikodym
derivative between the full measures is used.  Differentiation is performed
on a projective family of finite-coordinate fibres and is then closed by an
exponentially summable tail norm.  The same construction keeps track of the
loss of two current derivatives at every material differentiation.

\subsection{The fixed geometric current scale}

Let $\widehat S_a$ be the labelled corner resolution of the four-branch
symplectic baker section and let $F_a$ be its exact branch map.  The labelled
mapping torus
\[
 Q_a=(\widehat S_a\times[0,1])/(x,1)\sim(F_ax,0)
\]
has the exact symplectic form
\[
 \Omega_a=d(\lambda_a-E\,dt),
 \qquad \iota_{\partial_t}\Omega_a=dE.
\]
For every depth $n$, let $\mathscr F_n(a)$ be the finite set of bulk, face,
corner, and flag strata generated by the first $n$ seam preimages.  Affine
parent-fold transport identifies all $\mathscr F_n(a)$ with a fixed labelled
family $\mathscr F_n$.

Choose $m$ above every trace threshold and choose $0<\beta<1$ with
$\sum_n\beta^n\#\mathscr F_n<\infty$.  Define the positive test Hilbert scale
\[
 \Phi^{m+2r}
 =\left\{\phi=(\phi_F):
 \sum_{n\ge0}\beta^{-n}\sum_{F\in\mathscr F_n}
 \|\phi_F\|_{H^{m+2r+\operatorname{codim}F+2}(F)}^2<\infty\right\}.
\]
Every incidence or subdivision relation is one row of a bounded operator
\[
 \mathcal D_r:\Phi^{m+2r}\longrightarrow
 \bigoplus_{e\in\mathscr E}\beta^{-\operatorname{depth}(e)/2}
 H^{m+2r-\frac12}(e).
\]
The compatible test space is $\widehat\Phi^{m+2r}=\ker\mathcal D_r$ and the
current fibre is $\mathbb J^{m+2r}=(\widehat\Phi^{m+2r})'$.

\begin{lemma}[Closed subdivision relations]
\label{lem:r27-a1-closed}
For every $r\ge0$, $\widehat\Phi^{m+2r}$ is closed, parent-fold transport is a
uniformly bounded isomorphism on the scale, and the material connection
satisfies
\[
 \nabla_a:\mathbb J^{m+2r}\longrightarrow\mathbb J^{m+2r-2}.
\]
Consequently $j$ parameter derivatives are always interpreted in
$\mathbb J^{m+2r-2j}$; no derivative is silently returned to the original
current space.
\end{lemma}

\begin{proof}
Each trace row of $\mathcal D_r$ is bounded by the Sobolev trace theorem.  The
weighted direct sum is bounded because the number of rows at depth $n$ is
absorbed by $\beta^{-n}$, hence $\ker\mathcal D_r$ is closed.  Parent-fold
charts and their inverses have uniformly bounded derivatives on the compact
parameter interval, so interpolation gives uniform isomorphisms at every
level of the scale.  Differentiating a transported test consumes two
derivatives; transposition therefore gives the displayed closed operator on
currents.  Iteration gives precisely the stated scale loss.
\end{proof}

\subsection{A projective derivative without comparing singular products}

Let $\Sigma=\{1,2,3,4\}^{\mathbb Z}$ and let
$\mu_a=\bigotimes_{k\in\mathbb Z}w(a)$, where all $w_i(a)$ are $C^{r+1}$ and
bounded away from zero.  Write $\mathcal F_N=\sigma(\omega_k:|k|\le N)$ and
$\Pi_N^a=\mathbb E_{\mu_a}[\cdot\mid\mathcal F_N]$.

A coherent current jet of order $r$ is a family
\[
 \mathbf J^{(j)}_{a,N}\in
 L^2(\mu_a|_{\mathcal F_N};\mathbb J^{m+2r-2j}),
 \qquad 0\le j\le r,\quad N\ge0,
\]
such that restriction from $\mathcal F_{N+1}$ to $\mathcal F_N$ is conditional
expectation and, for every finite $N$,
\[
 \partial_a\mathbf J^{(j)}_{a,N}
 =\mathbf J^{(j+1)}_{a,N}
 +\mathbf B^{(j)}_{a,N}.                                  \tag{A1.1}
\]
Here $\mathbf B^{(j)}_{a,N}$ is the geometric boundary derivative created
when a transported seam crosses one of the finitely many cylinder
boundaries.  It is part of the data, not omitted as a null-set derivative.

For $M>N$, put
\[
 S_{a;N,M}=\sum_{N<|k|\le M}\partial_a\log w_{\omega_k}(a).
\]
The derivative of the parameter-dependent tail conditional expectation is
encoded by
\[
 \mathfrak T^{(j)}_{a,N}
 =\lim_{M\to\infty}
 \Pi_N^a\!\left[
 \bigl(\mathbf J^{(j)}_{a,M}-\Pi_N^a\mathbf J^{(j)}_{a,M}\bigr)
 S_{a;N,M}\right].                                      \tag{A1.2}
\]
This is the term missing from a calculation that differentiates only the
central finite product.

\begin{definition}[Certified projective response domain]
\label{def:r27-a1-domain}
For $\eta>0$, let $\mathfrak C_{\eta}^{r}$ be the coherent jets for which
\[
 \sum_{j=0}^{r}\sup_a\left[
 \sup_N\|\mathbf J^{(j)}_{a,N}\|_2+
 \sup_N e^{\eta N}
 \|\mathbf J^{(j)}_{a,N+1}-\mathbf J^{(j)}_{a,N}\|_2+
 \sup_N e^{\eta N}\|\mathbf B^{(j)}_{a,N}\|_2\right]<\infty.              \tag{A1.3}
\]
The projective limit in $L^2(\mu_a)$ is denoted $\mathbf J_a^{(j)}$.
\end{definition}

\begin{lemma}[Tail derivative identity]
\label{lem:r27-a1-tail}
For a jet in $\mathfrak C_\eta^r$, the limit in \emph{(A1.2)} exists
absolutely in $\mathbb J^{m+2r-2j}$ and
\[
 \partial_a\Pi_N^a\mathbf J_a^{(j)}
 =\Pi_N^a\mathbf J_a^{(j+1)}
  +\Pi_N^a\mathbf B_a^{(j)}
  +\mathfrak T^{(j)}_{a,N}
  +\Pi_N^a\!\left[
     (\mathbf J_a^{(j)}-\Pi_N^a\mathbf J_a^{(j)})
     S_{a;-N,N}\right],                                  \tag{A1.4}
\]
where $S_{a;-N,N}=\sum_{|k|\le N}\partial_a\log w_{\omega_k}(a)$.
Every series obtained by differentiating \emph{(A1.4)} at most $r-j$ times is
normally convergent.
\end{lemma}

\begin{proof}
Apply finite-product differentiation to $\mathbf J^{(j)}_{a,M}$ and split the
score into central and tail coordinates.  Centering the tail score permits
the factor in (A1.2) to be replaced by the martingale difference outside
$\mathcal F_N$.  The $L^2$ norm of a score on a shell of width $q$ is
$O(q^{1/2})$, whereas (A1.3) makes the corresponding current difference
$O(e^{-\eta q})$.  Cauchy--Schwarz therefore gives a summable bound
$C(1+q)^{1/2}e^{-\eta q}$.  Higher derivatives produce Bell polynomials of
degree at most $r$, hence only polynomial factors in $q$; the same exponential
tail gives normal convergence.  Passing $M\to\infty$ proves (A1.4).
\end{proof}

\begin{theorem}[Closed response formula]
\label{thm:r27-a1-response}
If $\mathbf J\in\mathfrak C_\eta^r$, then
\[
 a\longmapsto \mathcal M_a(\mathbf J)
 :=\lim_{N\to\infty}\int\mathbf J^{(0)}_{a,N}\,d\mu_a
 \in\mathbb J^{m}
\]
is $C^r$.  Its first derivative is
\[
 {d\over da}\mathcal M_a(\mathbf J)
 =\mathcal M_a(\mathbf J^{(1)}+\mathbf B^{(0)})
 +\sum_{k\in\mathbb Z}
 \mathbb E_{\mu_a}\!\left[
  \bigl(\mathbf J_a^{(0)}
       -\mathbb E[\mathbf J_a^{(0)}\mid\mathcal F_{|k|-1}]\bigr)
  \partial_a\log w_{\omega_k}(a)\right],                 \tag{A1.5}
\]
and the series is absolutely convergent in $\mathbb J^m$.  Higher derivatives
are the normally convergent Bell expansions generated by (A1.4).
\end{theorem}

\begin{proof}
The finite-cylinder means are $C^r$ by ordinary finite-dimensional
differentiation.  Lemma \ref{lem:r27-a1-tail} gives a common summable
majorant for the means and all derivatives through order $r$.  The Banach
valued Weierstrass theorem permits termwise differentiation.  Pairing the
tail shells by their outermost coordinate gives (A1.5).  At no point is a
density between two infinite product measures introduced.
\end{proof}

\subsection{The geometric seam current belongs to the domain}

Let $\mathbf J_a^{\rm seam}(\omega)$ be the signed current obtained by
intersecting the suspension orbit coded by $\omega$ with every labelled seam
and flag, using positive-side traces.  A depth-$n$ flag has weight $\beta^n$.

\begin{proposition}[Geometric-domain certificate]
\label{prop:r27-a1-geometric}
Suppose the branch widths, seam charts, and their first $r$ material
derivatives satisfy the uniform parent-fold distortion estimate
\[
 \sum_{F\in\mathscr F_n}
 \|\nabla_a^j[\mathbf J_a^{\rm seam}]_F\|_{\mathbb J^{m+2r-2j}}^2
 \le C_j\vartheta^{2n}\beta^{n},
 \qquad \vartheta<1.                                    \tag{A1.6}
\]
Then $\mathbf J^{\rm seam}\in\mathfrak C_\eta^r$ for every
$0<\eta<-\log\vartheta$.  Estimate (A1.6) holds for the labelled baker
benchmark because every depth-$n$ seam is an affine image with Jacobian equal
to a product of branch widths.
\end{proposition}

\begin{proof}
Changing coordinates outside $[-N,N]$ changes only flags of depth larger than
$N$.  Orthogonality of the weighted direct sum and (A1.6) bound the cylinder
tail by $C\vartheta^N$.  The same estimate after material differentiation
controls the boundary jet.  In the affine baker model, the product formula
for each branch Jacobian gives (A1.6) directly and the finite incidence
operator from Lemma \ref{lem:r27-a1-closed} preserves the estimate.
\end{proof}

\subsection{A maximal Hilbert martingale compiler}

Let $Y_k=\mathbf J_a^{(0)}\circ\sigma^k-\mathcal M_a(\mathbf J)$ and let
$P_jZ=\mathbb E[Z\mid\mathcal F_j^-]-\mathbb E[Z\mid\mathcal F_{j-1}^-]$ for
the past filtration.  The domain estimate gives
\[
 \sum_{q\in\mathbb Z}(1+|q|)^{1/2}
 \|P_0Y_q\|_{L^2(\mu_a;\mathbb J^m)}<\infty.              \tag{A1.7}
\]
Define $D_0=\sum_qP_0Y_q$ and $D_k=D_0\circ\sigma^k$.

\begin{lemma}[Maximal coboundary estimate]
\label{lem:r27-a1-maximal}
Under (A1.7),
\[
 {1\over\sqrt n}
 \max_{1\le j\le n}
 \left\|\sum_{k=0}^{j-1}(Y_k-D_k)\right\|_{\mathbb J^m}
 \longrightarrow0
 \quad\hbox{in }L^2.                                     \tag{A1.8}
\]
\end{lemma}

\begin{proof}
Truncate the projective series at range $2^\ell$.  The truncated remainder
is an exact difference $G_\ell-G_\ell\circ\sigma$, whose maximum is bounded by
$2\max_{j\le n}\|G_\ell\circ\sigma^j\|$.  The tail is split into dyadic
shells.  Doob's inequality on each shell and Cauchy--Schwarz with weights
$2^{\ell/2}$ give
\[
 \left\|\max_{j\le n}\Big\|\sum_{k<j}(Y_k-D_k)\Big\|\right\|_2
 \le o(\sqrt n)+C\sqrt n
 \sum_{\ell\ge L}2^{\ell/2}
 \sum_{2^\ell\le|q|<2^{\ell+1}}\|P_0Y_q\|_2.
\]
First let $n\to\infty$ for fixed $L$, then $L\to\infty$ using (A1.7).
\end{proof}

\begin{theorem}[Trace-class current FCLT]
\label{thm:r27-a1-fclt}
The polygonal process
\[
 W_n(t)=n^{-1/2}\sum_{k<nt}Y_k
\]
converges in $C([0,1];\mathbb J^m)$ to a Brownian motion with covariance
\[
 Qh=\mathbb E[\langle D_0,h\rangle D_0].
\]
The operator is positive and trace class, with
\[
 \operatorname{tr}Q=\mathbb E\|D_0\|_{\mathbb J^m}^2.
\]
The conclusion and covariance derivatives are locally uniform in $a$ through
order $r-1$.
\end{theorem}

\begin{proof}
The Hilbert martingale FCLT applies to $(D_k)$: finite-dimensional conditional
brackets converge by the ergodic theorem, $L^2$ truncation gives Lindeberg,
and Doob's inequality controls the orthogonal Hilbert tail.  Lemma
\ref{lem:r27-a1-maximal} transfers the functional limit to $(Y_k)$.  Parseval
and Tonelli give the trace identity.  Normal convergence of the response jets
permits differentiation of the covariance series.
\end{proof}

The construction simultaneously resolves the varying-measure fibre, the
parameter derivative of conditional expectation, the Hilbert-scale loss, the
closed infinite subdivision relation, the maximal remainder, and membership
of the motivating geometric current.
