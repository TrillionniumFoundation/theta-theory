\section{Coarse observations and visible--hidden information}\label{sec:information}
\subsection{Exact projection of a selected law}
For $R:\Omega\to Y$ and bounded $V$, let $P_y$ be a regular conditional law and set
\[
 V_R(y)=\log\int e^V\dd P_y.
\]
On zero-evidence points choose a version in $[\inf V,\sup V]$.

\begin{theorem}[Projection--tilting identities]\label{thm:projection}
Writing $P_Y=R_\#P$,
\[
 R_\#Q^V=\frac{e^{V_R}}{Pe^V}P_Y,
 \qquad Q^V(\dd\omega\mid y)=e^{V(\omega)-V_R(y)}P_y(\dd\omega).
\]
For a further projection $S=s(R)$,
$V_S=\log E_P[e^{V_R}\mid S]$.
\end{theorem}
\begin{proof}
Test the first equality against bounded $f(y)$ and condition $f(R)e^V$ on $R$. For the second, test against products $f(R)g(\omega)$ and use the displayed fiber normalization. The final identity is the tower property applied to $e^V$, followed by a logarithm. The source after projection is therefore not generally the conditional mean of the original source.
\end{proof}

\begin{proposition}[Conditional fluctuation gap]\label{prop:jensen}
If the conditional essential oscillation of $V$ is at most $b$, then
\[
 0\le V_R-E(V\mid R)\le b^2/8.
\]
For a bounded linear source $V=\xi\cdot C$, conditional derivatives are the tilted conditional mean and covariance of $C$.
\end{proposition}
\begin{proof}
On one fiber put $X=V-E(V\mid R)$ and $g(t)=\log E(e^{tX}\mid R)$. Then $g(0)=g'(0)=0$ and $g''(t)$ is a variance of a variable of oscillation at most $b$. If $X\in[l,l+b]$, the inequality $(X-l)(l+b-X)\ge0$ implies $\Var X\le b^2/4$. Consequently $g(1)=\int_0^1(1-t)g''(t)dt\le b^2/8$. Jensen gives the lower bound. Bounded domination permits conditional differentiation of the exponential and yields the mean and covariance formulas.
\end{proof}

\subsection{Entropy and its quadratic loss}
\begin{theorem}[Visible--hidden entropy decomposition]\label{thm:entropy-chain}
For $Q\ll P$ on standard Borel spaces,
\begin{equation}
 D(Q\Vert P)=D(Q_Y\Vert P_Y)
 +\int D(Q_y\Vert P_y)Q_Y(\dd y).
 \label{eq:entropy-chain}
\end{equation}
This holds in extended nonnegative values. At finite entropy, the second term vanishes exactly when the two conditional laws agree $Q_Y$-almost everywhere.
\end{theorem}
\begin{proof}
Let $q=dQ/dP$ and $b(y)=E_P(q\mid R=y)$. On $b>0$, the conditional density is $q/b$ and $Q_Y=bP_Y$. Decompose $q\log q=q\log b+q\log(q/b)$. The negative parts are integrable: the first total entropy integrand has negative part at most $1/e$; conditionally $E[q(\log b)^-\mid R]=b(\log b)^-\le1/e$; and $[q\log(q/b)]^-\le b/e$. Thus extended integration involves no undefined subtraction. Integrating the second and third expressions gives the two terms on the right. Conditional entropy is nonnegative, with equality only at the unchanged conditional law.
\end{proof}

For the selected law, hidden information equals $E_{Q^V}[V-V_R(R)]$. This is a loss under observation, not the same variational deficit as coarse preparation, although both concern unobserved fluctuations.

\begin{proposition}[Quadratic hidden-information tangent]\label{prop:hidden-tangent}
For bounded $F$ and $Q_t=e^{tF}P/Pe^{tF}$,
\[
 D(Q_t\Vert P)-D(R_\#Q_t\Vert R_\#P)
 =\tfrac12t^2E_P\Var_P(F\mid R)+O_{\norm F_\infty}(|t|^3).
\]
\end{proposition}
\begin{proof}
The full entropy is $t\Psi'(t)-\Psi(t)=t^2\Var(F)/2+O(t^3)$. The visible density is
$E(e^{tF}\mid R)/Ee^{tF}=1+t(E(F\mid R)-EF)+O(t^2)$ uniformly. Expand $(1+x)\log(1+x)=x+x^2/2+O(x^3)$ and use that its density perturbation integrates to zero. The visible entropy is $t^2\Var(E(F\mid R))/2+O(t^3)$. Total variance gives the claim. All remainders are uniform because $F$ is bounded.
\end{proof}

\subsection{Observation and parameter information}
Suppose a dominated differentiable experiment has score $\ell_a$ in $L^2(P_a)$. For a parameter-independent observation kernel or statistic, differentiation under a common envelope gives observed score
\[
 \ell_a^{\rm obs}=E_a[\ell_a\mid Y].
\]
Consequently
\begin{equation}
 E_a\ell_a^2-E_a(\ell_a^{\rm obs})^2
 =E_a\Var_a(\ell_a\mid Y)\ge0.
 \label{eq:fisher-loss}
\end{equation}
This is a calculation, not a claim that every deterministic path perturbation has a score. The event experiment in Section~\ref{sec:observation} has one; a moving Dirac path generally does not.

The same decomposition explains why a statistic sufficient for prediction at one parameter need not preserve derivatives across parameters. Response-aware sufficiency is formulated in Section~\ref{sec:predictors}.
