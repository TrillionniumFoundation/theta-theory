\section{Referee-round positive closure: actual-collision marked expansion and joint LDP}
\label{sec:rr2-b2}

\subsection{Orientation and exact microscopic balance}

For an unordered colliding pair \(i<j\), set
\[
 \omega=\frac{x_j-x_i}{\epsilon}.
\]
At an incoming contact,
\((v_i-v_j)\cdot\omega>0\).  The post-collisional velocities are
\[
 v_i'=v_i-\big((v_i-v_j)\cdot\omega\big)\omega,\qquad
 v_j'=v_j+\big((v_i-v_j)\cdot\omega\big)\omega.
\]
This convention is fixed throughout.

Define the empirical incoming collision measure by counting every actual
contact once:
\[
 \Gamma^\epsilon
 =
 \mu_\epsilon^{-1}
 \sum_{i<j}\sum_{t\in\mathcal C_{ij}}
 \delta_{(t,x_i(t),v_i(t-),v_j(t-),\omega(t))}.
\]

\begin{proposition}[Exact balance]
\label{prop:rr2-b2-balance}
For every smooth test \(\phi\),
\[
 \begin{aligned}
 \langle\phi_T,f_T^\epsilon\rangle-\langle\phi_0,f_0^\epsilon\rangle
 -\int_0^T\langle\partial_t\phi+v\cdot\nabla_x\phi,
 f_t^\epsilon\rangle\,dt
 =
 \int\Delta_\omega\phi\,d\Gamma^\epsilon .
 \end{aligned}
\]
The measure is invariant under simultaneous exchange
\((v,v_*,\omega)\mapsto(v_*,v,-\omega)\) after passing to the unordered
quotient.
\end{proposition}

\begin{proof}
Integrate \(\phi(t,x_i(t),v_i(t))\) along every free-flight segment.  Endpoint
terms telescope.  At a collision the jump of the two-particle sum is exactly
\(\Delta_\omega\phi\).  Summing over \(i<j\) counts each collision once.
The exchange statement follows from the definition of the quotient.
\end{proof}

\subsection{Marked trajectory polymers}

A trajectory polymer is a connected graph of labels equipped with ordered
collision times and the associated hard-sphere geometric constraints.  Give
each actual collision edge \(e\) the factor
\(\exp\{\psi(t_e,x_e,v_e,v_{*,e},\omega_e)\}\), including recollision edges.

\begin{lemma}[Factorial marked-polymer majorant]
\label{lem:rr2-b2-factorial}
For \(\|\psi\|_\infty\le\delta\), the total absolute weight of connected
\(k\)-label polymers with \(n\) collision events is bounded by
\[
 \frac{(C T e^{\delta})^n}{n!}\,(C\epsilon^2)^{k-1}
\]
times the product of the prescribed velocity weights.  The estimate is
uniform in \(\epsilon\) in the Boltzmann--Grad regime.
\end{lemma}

\begin{proof}
Order the collision times.  Integration over
\(0<t_1<\cdots<t_n<T\) gives \(T^n/n!\).  Each new label is introduced
through a collision cylinder of cross-section \(C\epsilon^2\).
A bounded mark contributes at most \(e^\delta\) per event.  Recollisions do
not introduce new labels; their geometric constraint removes one free
relative coordinate and is bounded by the same collision-cylinder estimate.
The tree-graph inequality sums graph choices into the constant \(C^n\).
\end{proof}

\begin{theorem}[Source-uniform marked cluster expansion]
\label{thm:rr2-b2-cluster}
There exist \(T_*,\delta>0\) such that for \(T<T_*\) and complex
\(\|h\|+\|\psi\|<\delta\), the scaled cumulant
\[
 Q_\epsilon(h,\psi)
 =
 \mu_\epsilon^{-1}
 \log\mathbb E_{\rm prep}
 \exp\left\{\mu_\epsilon
 \big(\langle h,f^\epsilon\rangle+
      \langle\psi,\Gamma^\epsilon\rangle\big)\right\}
\]
has an absolutely convergent connected-polymer expansion, uniformly in
\(\epsilon\), and converges with all source derivatives to
\(Q(h,\psi)\).
\end{theorem}

\begin{proof}
Lemma~\ref{lem:rr2-b2-factorial} gives a Koteck\'y--Preiss majorant with
activity \(CTe^\delta\).  Choose \(T_*,\delta\) so this activity is below the
convergence threshold.  Every actual collision is an event of exactly one
polymer, by connected-component decomposition of the labelled trajectory
graph.  Source-uniform recollision bounds follow from the same majorant
because the exponential mark has already been included in the polymer
activity.  Dominated convergence of each finite graph and summability of the
tail give convergence of the cumulant and all derivatives.
The prepared initial law is transferred with the source-dependent pressure
of the source-dependent constrained-pressure theorem of Paper B1.
\end{proof}

\subsection{State space and exponential tightness}

Use
\[
 \mathcal X_T=
 C([0,T],\mathcal P_{2+\eta})
 \times\mathcal M_+\big([0,T]\times\mathbb T^3
 \times\mathbb R^3\times\mathbb R^3\times\mathbb S^2\big),
\]
with weighted weak topology and the exchange quotient.

\begin{lemma}[Exponential compact containment]
\label{lem:rr2-b2-tight}
For every \(M\) there is a compact \(K_M\subset\mathcal X_T\) such that
\[
 \limsup_{\epsilon\to0}\mu_\epsilon^{-1}
 \log\mathbb P\big((f^\epsilon,\Gamma^\epsilon)\notin K_M\big)
 \le-M.
\]
\end{lemma}

\begin{proof}
The microcanonical energy bound controls the velocity \(2+\eta\) moments
after an exponential truncation using the prepared velocity tails.
The marked cluster theorem with a positive constant collision source gives
an exponential moment for \(\Gamma^\epsilon(1)\).  The exact balance controls
time increments of \(f^\epsilon\) against a countable dense test family.
Weighted Prokhorov compactness and Arzel\`a--Ascoli on the test coordinates
produce \(K_M\).
\end{proof}

\subsection{Identification of the joint rate}

Let
\[
 dA_f=\frac12 ff_*B((v-v_*)\cdot\omega)_+
 \,dt\,dx\,dv\,dv_*\,d\omega .
\]
A feasible pair satisfies the exact weak balance, exchange symmetry,
pre/post involution compatibility, and finite energy.

\begin{theorem}[Joint density--actual-collision LDP]
\label{thm:rr2-b2-ldp}
On \(\mathcal X_T\), \((f^\epsilon,\Gamma^\epsilon)\) satisfies a good LDP
with speed \(\mu_\epsilon\) and rate
\[
 I_T(f,\Gamma)
 =
 I_{0,a}(f_0)+
 \begin{cases}
 \displaystyle\int
 \ell\!\left(\frac{d\Gamma}{dA_f}\right)dA_f,
 & (f,\Gamma)\text{ feasible and }\Gamma\ll A_f,\\[2ex]
 +\infty,&\text{otherwise}.
 \end{cases}
\]
\end{theorem}

\begin{proof}
The convergent marked cumulants determine the upper Hamiltonian on a
separating cylinder algebra:
\[
 \mathbb H(f,p,\psi)=
 \langle f,v\cdot\nabla p\rangle+
 \int(e^{\Delta p+\psi}-1)\,dA_f .
\]
Lemma~\ref{lem:rr2-b2-tight} upgrades the finite-dimensional upper bounds to
the full upper bound.  Pointwise convex duality of \(e^z-1\), together with
the closed weak balance constraint, gives the displayed candidate rate.

For the lower bound, first take smooth strictly positive feasible pairs with
density \(q=d\Gamma/dA_f\) bounded above and below.  Tilt every actual
collision by \(\log q\) and tilt the initial law by the optimizer in
the prepared initial-LDP theorem of Paper B1.  The marked cluster expansion remains
uniform in this source ball and its first derivative gives the prescribed
biased Boltzmann equation.  Strict convexity gives concentration around
\((f,\Gamma)\), while the log Radon--Nikodym cost is
\(\int\ell(q)dA_f+I_{0,a}(f_0)\).
Approximate a general finite-rate pair by velocity truncation, convolution,
and addition of a small reference activity; entropy lower semicontinuity and
the balance equation pass to the limit.  This proves the lower bound and
goodness.
\end{proof}
