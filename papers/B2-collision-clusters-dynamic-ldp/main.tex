\begin{abstract}
We develop a collision-source-decorated real-trajectory cluster expansion for
deterministic hard spheres and derive a local joint large-deviation principle
for empirical one-particle paths and the actual incoming collision flow.  The
collision flow is counted once, satisfies the exact weak balance, and retains
recollision remainders.  The result is conditional only on the independently
stated microcanonical preparation theorem of Paper B1.
\end{abstract}

\section{Empirical paths and actual collision flow}
The real-trajectory cluster expansion of \cite{BGSSCluster2022} motivates the
following jointly marked objects.  Define
\[
 \Pi_T^\varepsilon
 =\frac1{\mu_\varepsilon}
 \sum_{i=1}^{N_\varepsilon}\delta_{z_i^\varepsilon(\cdot)}
 \quad\text{on }D([0,T],\T^3\times\R^3).
\]
For every unordered binary collision at time $t_c$, use the contact point
$x_c$, incoming velocities $(v_c,v_{*,c})$, and incoming normal $\omega_c$
with $(v_c-v_{*,c})\cdot\omega_c>0$.  Set
\[
 \Gamma_T^\varepsilon
 =\frac1{\mu_\varepsilon}
 \sum_c\delta_{(t_c,x_c,v_c,v_{*,c},\omega_c)}.
\]
For a test function $\phi$ write
\[
 \Delta_\omega\phi
 =\phi(x,v')+\phi(x,v_*')-\phi(x,v)-\phi(x,v_*).
\]
Then the exact microscopic balance is
\[
 \begin{aligned}
 \langle\pi_T^\varepsilon,\phi_T\rangle
 -\langle\pi_0^\varepsilon,\phi_0\rangle
 -\int_0^T\langle\pi_t^\varepsilon,
 \partial_t\phi+v\cdot\nabla_x\phi\rangle dt
 =\langle\Gamma_T^\varepsilon,\Delta_\omega\phi\rangle.
 \end{aligned}
\]
For bounded path and collision sources $(h,\psi)$ define
\[
 Q_{\varepsilon,T}(h,\psi)
 =\frac1{\mu_\varepsilon}
 \log\E\exp\left\{\mu_\varepsilon
 \big(\langle h,\Pi_T^\varepsilon\rangle
 +\langle\psi,\Gamma_T^\varepsilon\rangle\big)\right\}.
\]

\begin{theorem}[Collision-marked trajectory cluster expansion]\label{thm:marked-cluster}
For every source radius $r$ there exists $T_*(r)>0$ such that
$Q_{\varepsilon,T}(h,\psi)$ has an absolutely convergent dynamical cluster
expansion, uniformly in $\varepsilon$, for $T\le T_*(r)$ and
$\|h\|+\|\psi\|\le r$.  Its Boltzmann--Grad limit $Q_T(h,\psi)$ exists and is
analytic in the source ball.
\end{theorem}

\begin{proof}
Use the connected real-trajectory clusters of \cite{BGSSCluster2022}.  In the
boundary-flux representation, every created binary edge receives one factor
$e^{\psi}$ and every particle path receives $e^h$.  Recollision and overlap
graphs remain in the same negligible geometric class as in the undecorated
expansion.  The marked tree norm is bounded by the original norm with the time
constant replaced by $T e^{C(\|h\|+\|\psi\|)}$.  Shrinking $T_*(r)$ gives a
uniform polymer bound, while dominated convergence identifies the limiting
source-decorated Boltzmann trees.
\end{proof}

\section{Joint Hamiltonian and local density--collision large deviations}
For density $f=f(x,v)$ and cotangent fields $p=p(x,v)$ and
$\psi=\psi(x,v,v_*,\omega)$ define
\[
 \begin{aligned}
 \mathbb H(f,p,\psi)
 ={}&\int f(x,v)v\cdot\nabla_xp(x,v)\,dx\,dv\\
 &+\frac12\int f f_*\big((v-v_*)\cdot\omega\big)_+
 \left(e^{\Delta_\omega p+\psi}-1\right)
 \,dx\,dv\,dv_*\,d\omega.
 \end{aligned}
\]
The reference collision intensity is
\[
 dA_f=\frac12 f f_*\big((v-v_*)\cdot\omega\big)_+
 \,dt\,dx\,dv\,dv_*\,d\omega,
\]
and $\ell(q)=q\log q-q+1$.

\begin{theorem}[Local joint LDP]\label{thm:joint-ldp}
On the regular biased-solution class generated by bounded Lipschitz cotangents
in the source ball, $(\pi^\varepsilon,\Gamma^\varepsilon)$ satisfies a local
large-deviation principle at speed $\mu_\varepsilon$, with
\[
 \boxed{
 \cI_T(f,\gamma)=\cI_0(f_0)
 +\int\ell\left(\frac{d\gamma}{dA_f}\right)dA_f.}
\]
The formula applies when $\gamma\ll A_f$ and the weak kinetic balance holds;
otherwise the rate is $+\infty$.  Under the prepared law,
$\cI_0(f_0)=0$ at $f_a^0$ and is infinite off the prepared initial
constraint.
\end{theorem}

\begin{proof}
The density-only hard-sphere cumulant Hamiltonian and short-time LDP are
established in \cite{BGSS2023}.  The collision source above is the deterministic
hard-sphere realization of the kinetic flow dual appearing in
\cite{Rezakhanlou1998}.  The cluster limit gives the Hamilton--Jacobi equation
with Hamiltonian $\mathbb H$.  Pointwise Legendre duality uses
\[
 \sup_{\lambda\in\R}\{q\lambda-a(e^\lambda-1)\}
 =a\,\ell(q/a),
\]
and dualization in $p$ imposes the weak balance identity.  The preparation
transfer identifies the initial cost.  The local lower and upper bounds then
follow on the regular biased class by the same well-posedness and exponential
tightness mechanism as the density-only theorem.
\end{proof}

\begin{corollary}[Boltzmann law of large numbers]
The unique zero-cost path has $\gamma=A_f$ and satisfies the Boltzmann equation
with initial density $f_a^0$.
\end{corollary}

\section{Closure relative to the preparation theorem}
\begin{theorem}[Collision-cluster and local dynamic-LDP closure]\label{thm:b2-main}
Assume the microcanonical preparation-transfer theorem of Paper B1 in the
same short-time bounded-source window.  Then the actual incoming collision
flow and empirical particle paths admit the collision-marked trajectory
cluster expansion and the local joint density/collision-flow large-deviation
principle proved in this paper.  No stochastic collision rule is introduced.
\end{theorem}
\begin{proof}
The source-dependent preparation transfer supplies the initial exponential
functional.  The marked-cluster theorem controls connected real-trajectory
molecules and recollision remainders.  The joint-LDP theorem applies the local
Laplace principle and convex duality to the exact weak balance.  Their scopes
coincide by construction, proving the assertion.
\end{proof}
