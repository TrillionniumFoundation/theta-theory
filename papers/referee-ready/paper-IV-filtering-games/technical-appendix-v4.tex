\section{Bayes normalization in total variation}

For a bounded strictly positive likelihood $g$, define
\[
 \mathsf B_g(\mu)(f)={\mu(gf)\over\mu(g)}.
\]
Use
\[
 \norm{\mu-\nu}_{\TV}=\sup_{\abs f\le1}
 \abs{\mu(f)-\nu(f)}.
\]

\begin{lemma}[Sharp two-term Bayes bound]\label{lem:Bayes-bound}
If $0<g_-\le g\le g_+$, then
\[
 \norm{\mathsf B_g(\mu)-\mathsf B_g(\nu)}_{\TV}
 \le {2g_+\over g_-}\norm{\mu-\nu}_{\TV}.
\]
If the total-variation convention includes a factor one-half, the displayed
constant is divided by two.
\end{lemma}

\begin{proof}
For $\abs f\le1$,
\begin{align*}
 \mathsf B_g(\mu)(f)-\mathsf B_g(\nu)(f)
 =&{(\mu-\nu)(gf)\over\mu(g)}\\
 &+{\nu(gf)(\nu-\mu)(g)\over\mu(g)\nu(g)}.
\end{align*}
The first term is bounded by
$(g_+/g_-)\norm{\mu-\nu}_{\TV}$.  Since
$\abs{\nu(gf)}\le\nu(g)$, the second has the same bound.
\end{proof}

\section{Refresh--autoregression}

Let
\[
 P(y,\cdot)=\delta\gamma+(1-\delta)N(ry,1-r^2),
 \qquad0<\delta<1,\quad\abs r<1,
\]
where $\gamma=N(0,1)$, and put $W(y)=1+y^2$.

\begin{lemma}[Exact prediction identities]\label{lem:prediction}
For probability measures $\mu,\nu$,
\[
 \norm{\mu P-\nu P}_{\TV}
 \le(1-\delta)\norm{\mu-\nu}_{\TV}.
\]
With $\alpha=(1-\delta)r^2$,
\[
 PW(y)=\alpha W(y)+2(1-\alpha).
\]
Consequently
\[
 P^nW\le\alpha^nW+2.
\]
\end{lemma}

\begin{proof}
The common refresh part cancels in the difference; the remaining Gaussian
kernel is Markov and hence a TV contraction.  Direct calculation gives
\[
 \E[Y'^2\mid Y=y]
 =\delta+(1-\delta)(r^2y^2+1-r^2)
 =1-\alpha+\alpha y^2.
\]
Adding one gives the drift identity, and iteration gives the final bound.
\end{proof}

Let $g_o(y)=(1+o\epsilon\tanh y)/2$.  Then
$g_-=(1-\epsilon)/2$ and $g_+=(1+\epsilon)/2$.

\begin{theorem}[Posterior contraction and moment ball]\label{thm:posterior}
Choose $r_*$ so that
\[
 q={2g_+\over g_-}(1-\delta)^{r_*}<1,
 \qquad
 \kappa={g_+\over g_-}\alpha^{r_*}<1.
\]
Then one prediction--observation cycle contracts TV by $q$ and maps a prior
moment $M=\pi(W)$ to at most
\[
 \kappa M+{2g_+\over g_-}.
\]
Hence every
\[
 R\ge {2g_+/g_-\over1-\kappa}
\]
defines an invariant posterior moment ball.
\end{theorem}

\begin{proof}
Apply \cref{lem:prediction} for $r_*$ steps and then
\cref{lem:Bayes-bound}.  For the moment,
\[
 \mathsf B_g(\pi)(W)
 ={\pi(gW)\over\pi(g)}
 \le{g_+\over g_-}\pi(W).
\]
Combine this with $P^{r_*}W\le\alpha^{r_*}W+2$.
\end{proof}

\section{Strategy-tree stability and the slow initial layer}

Suppose the slow/control parameter changes between cycles.  Let
$\Phi_\theta$ be the complete filter update and assume
\[
 \norm{\Phi_\theta(\pi)-\Phi_\theta(\pi')}_{\TV}
 \le q\norm{\pi-\pi'}_{\TV},
\]
\[
 \norm{\Phi_\theta(\pi)-\Phi_{\theta'}(\pi)}_{\TV}
 \le L_R\abs{\theta-\theta'}
\]
on the invariant moment ball.

\begin{lemma}[Geometric slow convolution]\label{lem:convolution}
Along any common observation and control path,
\[
 \norm{\pi_n-\pi_n'}_{\TV}
 \le q^n\norm{\pi_0-\pi_0'}_{\TV}
 +L_R\sum_{j=0}^{n-1}q^{n-1-j}
 \abs{\theta_j-\theta_j'}.
\]
The constants are independent of the strategy tree.
\end{lemma}

\begin{proof}
Insert and subtract the update with one common parameter at each step.  The
resulting affine recursion is
$d_{n+1}\le qd_n+L_R\abs{\theta_n-\theta_n'}$.  Iterate it.
\end{proof}

If $\ell_\varepsilon$ cycles occur before the first order-one reward and their
slow duration is $\delta_\varepsilon^{\rm init}$, the value difference is
bounded by
\[
 C_Tq^{\ell_\varepsilon}\norm{\pi-\pi'}_{\TV}
 +C\delta_\varepsilon^{\rm init}
 +C_TL_R\sum_jq^j\omega_\varepsilon(j).
\]
This proves the value-level initial-belief collapse when all three terms vanish.

\section{Strong monotonicity of the saddle operator}

Let $U\subset\R^m$, $V\subset\R^n$ be closed convex.  Suppose
$F_0(z,u,v)$ is $C^2$ and
\[
 D_{uu}^2F_0\le L_uI,
 \qquad D_{vv}^2F_0\ge-L_vI.
\]
Set
\[
 F=F_0-{\mu\over2}\abs u^2+{\nu\over2}\abs v^2,
 \qquad \mu>L_u,\quad\nu>L_v,
\]
\[
 G_z(u,v)=(-D_uF(z,u,v),D_vF(z,u,v)).
\]

\begin{lemma}[Exact cancellation of mixed Hessians]\label{lem:mixed}
For $\dot w=(\dot u,\dot v)$,
\[
 \left\langle {DG_z+DG_z^T\over2}\dot w,\dot w\right\rangle
 =-\dot u^TD_{uu}^2F\dot u+
 \dot v^TD_{vv}^2F\dot v.
\]
In particular,
\[
 \langle G_z(w)-G_z(w'),w-w'\rangle
 \ge\lambda\abs{w-w'}^2,
 \quad
 \lambda=\min\{\mu-L_u,\nu-L_v\}.
\]
\end{lemma}

\begin{proof}
The upper-right block of $DG_z$ is $-D_{uv}^2F$, while the lower-left block is
$D_{vu}^2F=(D_{uv}^2F)^T$.  They cancel in the symmetric part.  The diagonal
bounds give the pointwise quadratic estimate; integrate it along the segment
from $w'$ to $w$.
\end{proof}

\section{The constrained variational inequality}

Let $K=U\times V$ and let $N_K$ be its normal cone.

\begin{theorem}[Unique constrained pure saddle]\label{thm:VI}
Assume $G_z$ is continuous, strongly monotone with constant $\lambda>0$, and
coercive on $K$.  Then
\[
 0\in G_z(w)+N_K(w)
\]
has a unique solution $w_*(z)=(u_*,v_*)$.  It satisfies
\[
 F(z,u,v_*)\le F(z,u_*,v_*)\le F(z,u_*,v)
\]
for all $(u,v)\in U\times V$.  If
\[
 \abs{G_z(w)-G_{z'}(w)}\le L_z\abs{z-z'},
\]
then
\[
 \abs{w_*(z)-w_*(z')}\le {L_z\over\lambda}\abs{z-z'}.
\]
\end{theorem}

\begin{proof}
The normal cone is maximal monotone and $G_z$ is continuous, strongly
monotone, and everywhere defined.  Their sum is strongly maximal monotone.
Coercivity gives surjectivity at zero, hence existence; strong monotonicity
gives uniqueness.  The inclusion is equivalent to
\begin{align*}
 \langle-D_uF(u_*,v_*),u-u_*\rangle&\ge0,\\
 \langle D_vF(u_*,v_*),v-v_*\rangle&\ge0.
\end{align*}
Strong concavity in $u$ and strong convexity in $v$ turn these first-order
inequalities into the global saddle inequalities.  Pair the two inclusions at
$z,z'$ with their difference; monotonicity of the normal cone and strong
monotonicity of $G_z$ give the Lipschitz estimate.
\end{proof}

\section{Coercive localization of the discrete game}

Consider the one-step expression
\begin{align*}
 \mathcal G_h(x,u,v;\phi)
 =&\sum_iw_i(a(x,u,v))
 \phi(x+hb(x,u,v)+\sqrt h\,m_i(x,u,v))\\
 &+hF_0(x,u,v)-h{\mu\over2}\abs u^2+h{\nu\over2}\abs v^2.
\end{align*}
Assume $b,m_i,a$ and their control derivatives grow at most linearly, while
$\phi$ is tested on a bounded gradient and Hessian window.

\begin{lemma}[Uniform optimizer localization]\label{lem:localization}
For every bounded jet window there is $R<\infty$, independent of small $h$,
such that the maximizing and minimizing optimizers of the consistency game lie
in $U\cap B_R$ and $V\cap B_R$.
\end{lemma}

\begin{proof}
Taylor expansion shows that the control-dependent terms arising from the test
function grow at most linearly in $\abs u+\abs v$ plus a bounded quadratic term
whose unfavourable curvature is already included in $L_u,L_v$.  The strict
energy margins $\mu-L_u$ and $\nu-L_v$ make the local payoff tend to
$-\infty$ as $\abs u\to\infty$ for the maximizer and to $+\infty$ as
$\abs v\to\infty$ for the minimizer, uniformly on the jet window.  Choose $R$
where the quadratic margin dominates all lower-order terms.
\end{proof}

\section{Common lower and upper consistency}

Let
\[
 \mathcal S_h^-\phi=\sup_u\inf_v\mathcal G_h(u,v;\phi),
 \qquad
 \mathcal S_h^+\phi=\inf_v\sup_u\mathcal G_h(u,v;\phi).
\]

\begin{proposition}[Pure consistency]\label{prop:pure-consistency}
For every smooth test $\phi$ and compact slow set,
\begin{align*}
 {\mathcal S_h^-\phi-\phi\over h}&\to H^-(x,D\phi,D^2\phi),\\
 {\mathcal S_h^+\phi-\phi\over h}&\to H^+(x,D\phi,D^2\phi),
\end{align*}
locally uniformly.  Under the curvature-compensation hypotheses,
$H^-=H^+=H$ and the common value is attained at the unique pure selector.
\end{proposition}

\begin{proof}
By \cref{lem:localization}, all optimizers lie in one compact control set.
The branch Taylor expansion is therefore uniform and gives the local
second-order game.  Its saddle operator has the same strict curvature margins,
so \cref{thm:VI} gives a unique saddle and equality of max--min and min--max.
Uniform convergence of the local payoffs permits passage of both optimization
orders to the limit.
\end{proof}

\section{Convergence of lower and upper schemes}

\begin{theorem}[Common Isaacs limit]\label{thm:common-limit}
Assume global continuity, uniform ellipticity, and the uniform curvature
margins above.  The lower and upper terminal recursions converge locally
uniformly to the same bounded viscosity solution of
\[
 -V_t-H(x,DV,D^2V)=0.
\]
\end{theorem}

\begin{proof}
Both one-step maps are monotone and stable.  Their consistency operators agree
by \cref{prop:pure-consistency}.  The unique pure selector is Lipschitz in the
jet variables on bounded sets.  The resulting Isaacs operator satisfies the
standard global structure estimate in the doubled-variable argument; the
quadratic control energies prevent optimizer escape.  Uniform ellipticity and
global coefficient continuity give comparison.  Half-relaxed limits of both
schemes are therefore the same solution.
\end{proof}

\section{Feedback verification beyond the smooth window}

For smooth terminal data and smooth coefficients, uniform parabolicity gives a
classical solution.  Let $w_*(t,x)$ be the pure selector evaluated at its jet.
It\^o's formula gives the lower and upper verification inequalities.

\begin{proposition}[Stable smooth approximation]\label{prop:smooth-approx}
Let bounded uniformly continuous terminal data be approximated uniformly by
smooth data, and mollify the coefficients while preserving ellipticity and
strict curvature margins.  The classical values converge locally uniformly to
the viscosity value, and the pure selectors converge locally in measure along
the controlled state laws.  The verification inequalities pass to the limit.
\end{proposition}

\begin{proof}
Comparison gives uniform stability of the values.  Strong monotonicity gives
\[
 \abs{w_*^n-w_*}
 \le\lambda^{-1}\abs{G^n(w_*)-G(w_*)},
\]
so the selectors converge wherever the approximating jets converge.  Uniform
parabolic Sobolev estimates give subsequential jet convergence locally in
measure.  Coercive quadratic energies provide uniform integrability of the
controls.  Pass to the limit in the It\^o verification inequalities by
localization and dominated convergence; comparison removes the subsequence.
\end{proof}

\section{Conclusion}

The noncompact filter uses separate mechanisms for metric contraction and
moment stability.  The pure game uses a constrained strongly monotone
variational inequality, not mixed minimax.  Coercive localization makes the
lower and upper discrete games uniformly consistent with the same pure
Hamiltonian, and stable parabolic approximation supplies the feedback value
beyond the classical window.
