\begin{abstract}
We prove a bilateral response theorem for transfer operators with moving
physical branch cuts.  Crossed source- and target-side estimates are
interpolated into an absolutely summable two-time product tail.  Difference
quotients converge in the complete $\ell^1(\N^2)$ response topology, and a
joint-Cauchy theorem totalizes the third source on a graded Banach scale.

The theory is verified on a genuinely nonconjugate four-branch deterministic
pinball billiard with three moving collision seams.  The centered transfer
operator contracts uniformly on periodic Sobolev spaces, and the $k$th
parameter derivative has an exact formula losing exactly $k$ derivatives.
For every prescribed finite order this gives product response, finite-DQ,
and all well-typed Kato words.  A second theorem proves exact predictable
innovations: even for history-dependent parameters, the current fast
coordinate is conditionally Lebesgue, the branch symbol has the declared
parameter-dependent law, and the image coordinate is again independent of the
enlarged past.  Hence the martingale arrays used downstream are the exact
symbolic process of the same deterministic moving-cut system.  We also prove
an all-order invariant-projector channel for a nonconjugate radial family of
finite-horizon specular Sinai billiards.
\end{abstract}

\section{Introduction}
For a fixed smooth operator family, isolated eigenvalues are governed by
classical perturbation theory \cite{Kato1995}.  Moving discontinuity or
collision sets create oriented currents and two independent waiting times.
An estimate $\rho^m+\rho^n$ cannot be summed over $(m,n)$; a genuine product
tail is required.

We separate three mechanisms and then verify them together on one model:
occurrencewise bilateral decay, physical totalization of higher sources, and a
graded regularity ladder.  The explicit model also supplies an exact
nonautonomous Bernoulli structure, removing the usual disconnect between a
response model and a homogenization model.  The framework is compatible with
strong--weak perturbation theory \cite{KellerLiverani1999}, higher response for
piecewise expanding systems \cite{BaladiSmania2008}, and anisotropic billiard
spaces \cite{DemersLiverani2008,BaladiDemersLiverani2018}.

\section{Bilateral two-time response}\label{sec:bilateral}
Let $L_a$ be a Perron operator, $\Pi_a$ its simple invariant projection, and
$Q_a=L_a-\Pi_a$.  A parameter source is represented as
\[
 \partial_aL_a=K_a^{\rm reg}+\sum_{\omega\in\Omega_a}J_{a,\omega},
\]
where an occurrence label records the physical face, side, owner, oriented
word, chamber, and refinement chart.  For centered $g$ and an admissible test
$f$, put
\[
 A_{m,n}^{\omega}(a;g,f)=
 \pair{Q_a^nJ_{a,\omega}Q_a^m g}{f}.
\]

\begin{assumption}[Crossed envelopes]\label{ass:crossed}
Uniformly on the declared unit balls,
\begin{align}
 \abs{A_{m,n}^{\omega}}
 &\le C_\omega^-\alpha^m\Gamma_+^n,
 &0<\alpha<1,\quad\Gamma_+\ge1,\label{eq:reverse}\\
 \abs{A_{m,n}^{\omega}}
 &\le C_\omega^+\Gamma_-^m\beta^n,
 &0<\beta<1,\quad\Gamma_-\ge1.\label{eq:forward}
\end{align}
\end{assumption}
Set
\[
 A_0=\log(1/\alpha),\quad B_0=\log(1/\beta),\quad
 C_0=\log\Gamma_+,\quad D_0=\log\Gamma_-.
\]

\begin{theorem}[Sharp bilateral product tail]\label{thm:bilateral}
If $A_0B_0>C_0D_0$, define
\[
 \lambda_*={B_0+D_0\over A_0+B_0+C_0+D_0},
 \qquad
 \log\rho_*={C_0D_0-A_0B_0\over A_0+B_0+C_0+D_0}.
\]
Then $0<\rho_*<1$ and
\[
 \abs{A_{m,n}^{\omega}}
 \le(C_\omega^-)^{\lambda_*}(C_\omega^+)^{1-\lambda_*}
 \rho_*^{m+n}.
\]
If the interpolated occurrence constants are summable, then
\[
 \sum_\omega\sum_{m,n\ge0}\abs{A_{m,n}^{\omega}}<\infty.
\]
The strict inequality is equivalent to the existence of an interpolation
parameter making both time factors strictly smaller than one.
\end{theorem}

\begin{proof}
Geometrically interpolate \eqref{eq:reverse} and \eqref{eq:forward}:
\[
 \abs{A_{m,n}^{\omega}}
 \le(C_\omega^-)^\lambda(C_\omega^+)^{1-\lambda}
 (\alpha^\lambda\Gamma_-^{1-\lambda})^m
 (\Gamma_+^\lambda\beta^{1-\lambda})^n.
\]
At $\lambda_*$ the logarithms of the two factors coincide and equal
$\log\rho_*$.  Summation proves the result.
\end{proof}

\begin{corollary}[Invariant regularity-loss route]\label{cor:loss}
If $Q_a$ contracts on the centered parts of $X_{r+k}$ and $X_r$ and
$K_a^{(k)}:X_{r+k}\to X_r$ is bounded and centered, then
\[
 \abs{\pair{Q_a^nK_a^{(k)}Q_a^m g}{f}}
 \le C_{r,k}\rho_r^n\rho_{r+k}^m
 \norm g_{X_{r+k}}\norm f_{X_r^*}.
\]
All later insertions are bounded compositions on the same invariant ladder.
\end{corollary}

\section{Finite-DQ and higher-source totalization}\label{sec:higher}
For $h\ne0$ set $K_{a,h}=(L_{a+h}-L_a)/h$.

\begin{theorem}[Finite-DQ in the complete response topology]\label{thm:fdq}
Suppose the finite-DQ arrays converge uniformly on every finite rectangle and
the finite-DQ and limiting arrays share one summable product envelope.  Then
\[
 \sum_{m,n\ge0}
 \abs{\pair{Q_{a+h}^nK_{a,h}Q_a^mg-Q_a^nK_aQ_a^mg}{f}}
 \longrightarrow0.
\]
\end{theorem}

\begin{proof}
Choose a rectangle whose common tail is smaller than $\varepsilon/3$ and use
uniform convergence on the finite head.  The two tails are controlled by two
copies of the envelope.
\end{proof}

Let $d=(K,\varepsilon,\mathcal P)$ be a depth, intersection, and refinement
cutoff and write
\[
 G_{3,d}=
 \Tot_{\mathfrak S_d}\{\eta_{a,\omega}:\omega\in I_d\}
 +S_d^{\rm seam}.
\]

\begin{theorem}[Joint-Cauchy totalization]\label{thm:totalization}
Assume the finite sources are continuous in a complete graded source space,
\[
 \sup_a\norm{\eta_{a,\omega}}_{\rm CM2}\le b_\omega,
 \qquad\sum_\omega b_\omega<\infty,
\]
and the vertex and refinement defects have moduli tending to zero.  Suppose
common refinements preserve UIDs and orientations and reduced resolvents
preserve every declared level.  Then, for $d_1,d_2\succeq d$,
\[
 \norm{G_{3,d_1}-G_{3,d_2}}
 \le2\sum_{\omega\notin I_d}b_\omega
 +2\omega_{\rm vert}(\varepsilon)
 +2\omega_{\rm seam}(\mathcal P).
\]
Thus the third source and all graded third-order Kato words converge uniformly
in $a$.
\end{theorem}

\begin{proof}
On a common refinement, common atoms cancel by UID and orientation.  Only the
occurrence tail and the two defect sets remain.  Completeness and level
preservation finish the proof.
\end{proof}

\section{The actual nonconjugate moving-cut family}\label{sec:model}
Let $I=[-1/40,1/40]$ and
\[
 (w_1,w_2,w_3,w_4)
 =(1/10+a,1/5+2a,3/10-a,2/5-2a).
\]
Let $s_0=0$, $s_i=\sum_{j\le i}w_j$, and
\begin{equation}\label{eq:T}
 T_a(x)={x-s_{i-1}(a)\over w_i(a)},
 \qquad x\in[s_{i-1}(a),s_i(a)).
\end{equation}
Its Perron operator is
\begin{equation}\label{eq:L}
 (L_ah)(y)=\sum_{i=1}^4w_i(a)
 h(s_{i-1}(a)+w_i(a)y).
\end{equation}
Lebesgue probability is invariant.

\begin{proposition}[Collision realization]\label{prop:collision}
There is a deterministic unit-speed pinball flow on $\T\times[0,1]$ whose
bottom return map is \eqref{eq:T}.  Its three top-wall collision seams move
with velocities $1,3,2$.
\end{proposition}

\begin{proof}
Launch vertically from the bottom.  On the $i$th top interval choose the
outgoing direction whose next bottom footpoint is $T_a(x)$.  The resulting
return map is \eqref{eq:T}; differentiation of the partial sums gives the seam
velocities.
\end{proof}

\begin{theorem}[Nonconjugacy]\label{thm:nonconjugacy}
If $T_a$ and $T_b$ are $C^1$ conjugate as piecewise expanding maps, then
$a=b$.
\end{theorem}

\begin{proof}
For $i=1,2,3$, the $i$th branch contains the fixed point
$x_i=s_{i-1}/(1-w_i)$ with multiplier $w_i^{-1}$.  The three multiplier ranges
are pairwise disjoint on $I$, so a conjugacy cannot permute them.  Equality of
periodic multipliers gives $w_1(a)=w_1(b)$ and hence $a=b$.
\end{proof}

\section{Uniform contraction and exact parameter letters}\label{sec:letters}
For $r\ge1$ let $X_r=W^{r,1}(\T)$ and let $X_r^0$ be the zero-mean subspace.
Put $\rho=9/20$.

\begin{lemma}[Uniform centered contraction]\label{lem:contraction}
For every finite $r$ there is $C_r$ such that
\[
 \norm{L_a^nh}_{W^{r,1}}
 \le C_r\rho^n\norm h_{W^{r,1}},
 \qquad h\in X_r^0,\ a\in I.
\]
\end{lemma}

\begin{proof}
For $j\ge1$,
\[
 (L_ah)^{(j)}=
 \sum_iw_i^{j+1}h^{(j)}(s_{i-1}+w_i\cdot),
\]
so change of variables gives
$\norm{(L_ah)^{(j)}}_1\le\rho^j\norm{h^{(j)}}_1$.
Iterate and use the periodic zero-mean Poincar\'e inequality to control the
zeroth level by the first derivative.
\end{proof}

Let $v=(1,2,-1,-2)$, $A_i=\sum_{j\le i}v_j$,
\[
 \psi_i^a(y)=s_{i-1}(a)+w_i(a)y,
 \qquad r_i(y)=A_{i-1}+v_i y.
\]

\begin{lemma}[Exact letters]\label{lem:letters}
For every $k\ge1$,
\begin{equation}\label{eq:letters}
 \partial_a^kL_ah=
 \sum_{i=1}^4\left[
 kv_ir_i^{k-1}h^{(k-1)}(\psi_i^a)
 +w_i(a)r_i^kh^{(k)}(\psi_i^a)
 \right].
\end{equation}
Moreover,
\[
 \norm{\partial_a^kL_ah}_{W^{r,1}}
 \le C_{r,k}\norm h_{W^{r+k,1}},
 \qquad \Piop\partial_a^kL_a=0.
\]
\end{lemma}

\begin{proof}
Both $w_i$ and $\psi_i^a$ are affine in $a$.  In the Leibniz rule at most one
derivative hits $w_i$; the rest hit the composition, giving
\eqref{eq:letters}.  Product and affine-composition estimates give the norm
bound.  Differentiate preservation of mass for centering.
\end{proof}

\begin{theorem}[All prescribed finite response orders]\label{thm:allorder}
Fix $N\ge1$.  On the ladder
\[
 W^{N+2,1}\hookrightarrow W^{N+1,1}\hookrightarrow\cdots
 \hookrightarrow W^{1,1},
\]
the family \eqref{eq:T} has:
\begin{enumerate}[label=(\roman*)]
\item a nonzero moving-cut source;
\item product response for every parameter letter of order at most $N+1$;
\item finite-DQ convergence through order $N$;
\item a continuous spectral projection and reduced-resolvent jet through order
$N$;
\item every Kato word of total order at most $N$ as a bounded graded
composition.
\end{enumerate}
In particular, $N=3$ gives the complete $U3$ package, with the fifth input
level available for the fourth-order Taylor remainder.
\end{theorem}

\begin{proof}
The unassembled branch differentiation contains nonzero oriented endpoint
currents, while \eqref{eq:letters} is their complete physical assembly and is
not the zero operator.  Combining \cref{lem:contraction,lem:letters} gives
\[
 \abs{\pair{Q_a^n(\partial_a^kL_a)Q_a^mg}{f}}
 \le C_{r,k}\rho^{m+n}
 \norm g_{W^{r+k,1}}\norm f_{(W^{r,1})^*}.
\]
This proves product response.  Taylor's formula for the order-$N$ difference
quotient uses the $(N+1)$st derivative and therefore the top level
$W^{N+2,1}$; the common product envelope and \cref{thm:fdq} give finite-DQ.
Riesz-contour differentiation yields the spectral jet.  Each letter loses its
order and each reduced resolvent preserves its level, proving composability.
\end{proof}

\section{Exact predictable innovations}\label{sec:innovations}
Let $X_0$ be Lebesgue distributed.  Recursively let $A_k\in I$ be
$\F_k=\sigma(I_0,\ldots,I_{k-1})$-measurable, let $I_k=i$ when
$X_k\in I_i(A_k)$, and set $X_{k+1}=T_{A_k}(X_k)$.

\begin{theorem}[Exact innovation theorem]\label{thm:innovations}
For every $k$:
\begin{enumerate}[label=(\roman*)]
\item conditionally on $\F_k$, $X_k$ is Lebesgue distributed;
\item $\Pp(I_k=i\mid\F_k)=w_i(A_k)$;
\item conditionally on $\F_k\vee\sigma(I_k)$, $X_{k+1}$ is Lebesgue
and independent of that sigma-field.
\end{enumerate}
\end{theorem}

\begin{proof}
Induct on $k$.  Conditional uniformity gives the branch probability as its
length.  Given the branch, the affine bijection to $[0,1)$ sends normalized
Lebesgue measure to Lebesgue measure, independently of the branch and past.
\end{proof}

\begin{corollary}[Endogenous martingale array]\label{cor:martingale}
For bounded predictable vectors $c_i(\theta_k)$, put
\[
 \bar c_k=\sum_iw_i(A_k)c_i(\theta_k),
 \qquad M_k=c_{I_k}(\theta_k)-\bar c_k.
\]
Then
\[
 \mathbb E[M_k\mid\F_k]=0,
 \qquad
 \mathbb E[M_k\otimes M_k\mid\F_k]
 =\sum_iw_i(A_k)(c_i-\bar c_k)^{\otimes2},
\]
and all conditional moments are uniformly bounded.
\end{corollary}

\section{A radial specular channel}\label{sec:specular}
Let a finite-horizon periodic Sinai table have a unique shortest regular normal
two-cycle between two circular scatterers.  Keep centers fixed and set
$R_1(a)=R_1+a$ on a small interval preserving convexity, nonoverlap, and finite
horizon.  The Jacobi trace is
\[
 \operatorname{tr}M(a)=2+4d(a)(\kappa_1(a)+\kappa_2)
 +4d(a)^2\kappa_1(a)\kappa_2,
\]
and has nonzero derivative; the isolated orbit proves nonconjugacy.  In
normalized boundary coordinates the invariant density is
\[
 \rho_a(i,x,\varphi)={\ell_i(a)\cos\varphi\over2\sum_j\ell_j(a)}.
\]

\begin{theorem}[All-order invariant-projector channel]\label{thm:specular}
For every finite $j$,
\[
 S_j=\sum_{k=1}^j{j\choose k}L_0^{[k]}\rho_0^{(j-k)}
 =(I-L_0)\rho_0^{(j)},
 \qquad R_0S_j=\rho_0^{(j)}.
\]
For analytic $F_a,g_a$ and
$\psi_a=c(a)+g_a-g_a\circ T_a$, the susceptibility equals
\[
 \sum_{n\ge0}\int(F_a-\mu_aF_a)
 (\psi_a-c(a))\circ T_a^n\,d\mu_a
 =\int(F_a-\mu_aF_a)g_a\,d\mu_a
\]
and is analytic in $a$.
\end{theorem}

\begin{proof}
Differentiate $L_a\rho_a=\rho_a$ and use that every density jet has zero
integral.  The susceptibility partial sums telescope; fixed-table mixing
removes the terminal term.
\end{proof}

\section{Conclusion}
One nonconjugate deterministic collision family now supplies moving currents,
all finite graded response orders, and exact predictable innovations.  The
spectral and probabilistic companion papers therefore act on the same actual
platform.
