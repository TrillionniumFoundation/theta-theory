\begin{abstract}
We derive a nonlinear terminal semigroup from the same nonconjugate moving-cut
collision system used in the companion response and pressure papers.  The
exact innovation theorem remains valid when the branch parameter is a
predictable function of the slow history.  Centered branch increments are
therefore martingale differences generated by the deterministic nonautonomous
map.  We prove first- and geometric second-level convergence, including the
predictable covariance and antisymmetric area anomaly, and obtain
nonautonomous rough homogenization on the full microscopic sequence.

A sharp Gaussian theorem gives matching
$N^{-(1/2-\eta)}$ upper and lower bounds in the full fractional-Sobolev
step-two rough-path Wasserstein metric.  The rate yields an explicit nonempty
block-freezing window.  Finally, a pointwise finite-branch entropic recursion
converges to a globally posed uniformly elliptic HJB equation.  Cole--Hopf
gives comparison, an exact Feynman--Kac formula, and a time-consistent
theta-semigroup.  No distributional residual is promoted to a pointwise
viscosity statement.
\end{abstract}

\section{Introduction}
Frozen response and nonautonomous homogenization are different assertions.
The bridge here is an exact conditional-independence property of the moving-cut
map itself.  It turns the response coefficients of Papers I--II into the
predictable characteristics of a genuine deterministic martingale array.
The proof uses martingale limit theory \cite{HallHeyde1980,JacodShiryaev2003},
rough paths \cite{Lyons1998,FrizVictoir2010,FrizHairer2014}, deterministic
homogenization \cite{MelbourneStuart2011,KellyMelbourne2016,
KellyMelbourne2017,ChevyrevFrizKorepanovMelbourne2020}, and monotone viscosity
schemes \cite{BarlesSouganidis1991,CrandallIshiiLions1992}.

\section{Exact innovations and Doob branch widths}\label{sec:innovations}
For $a\in I=[-1/40,1/40]$, let $T_a$ be the four-branch full-cover map of
Paper I, with branch widths
\[
 (w_1,w_2,w_3,w_4)
 =(1/10+a,1/5+2a,3/10-a,2/5-2a).
\]
Let $X_0^{\rm f}$ be uniform.  If $A_k$ is measurable with respect to
$\F_k=\sigma(I_0,\ldots,I_{k-1})$, let $I_k$ be the branch of
$X_k^{\rm f}$ and set $X_{k+1}^{\rm f}=T_{A_k}(X_k^{\rm f})$.

\begin{theorem}[Exact predictable innovations]\label{thm:innovations}
Conditionally on $\F_k$, $X_k^{\rm f}$ is uniform and
\[
 \Pp(I_k=i\mid\F_k)=w_i(A_k).
\]
Conditionally on $\F_k\vee\sigma(I_k)$, the image
$X_{k+1}^{\rm f}$ is uniform and independent of that sigma-field.
\end{theorem}

\begin{proof}
Induct on $k$.  Given conditional uniformity, the branch probability is its
length.  The affine branch bijection sends normalized Lebesgue measure on the
branch to Lebesgue measure on $[0,1)$, independently of the branch and past.
\end{proof}

For branch vectors $c_i$ and an external tilt $\zeta$, the Doob probabilities
are
\[
 p_i(a,\zeta)=
 {w_i(a)e^{\zeta\cdot c_i}\over
  \sum_jw_j(a)e^{\zeta\cdot c_j}}.
\]
They are again branch widths of a full-cover map, so
\cref{thm:innovations} applies to predictable $(a_k,\zeta_k)$.  This
selection is made before any value function is solved.

\section{Geometric martingale rough paths}\label{sec:rough}
Let $(M_k^\varepsilon,\F_{k+1}^\varepsilon)$ be $\R^d$-valued martingale
differences and $N_t^\varepsilon=\lfloor t/\varepsilon^2\rfloor$.  At grid
times define
\begin{align}
 W^\varepsilon(t)&=\varepsilon
 \sum_{k<N_t^\varepsilon}M_k^\varepsilon,\label{eq:first}\\
 \mathbb W^\varepsilon(t)&=\varepsilon^2
 \sum_{0\le i<j<N_t^\varepsilon}
 M_i^\varepsilon\otimes M_j^\varepsilon
 +{\varepsilon^2\over2}
 \sum_{i<N_t^\varepsilon}M_i^\varepsilon\otimes M_i^\varepsilon.
 \label{eq:second}
\end{align}
Interpolate each increment linearly.  Then
$(W^\varepsilon,\mathbb W^\varepsilon)$ is the canonical geometric step-two
lift; the diagonal term in \eqref{eq:second} is essential.

\begin{assumption}[Predictable characteristics]\label{ass:chars}
On every finite horizon:
\begin{enumerate}[label=(\roman*)]
\item conditional $q$th moments are uniformly bounded for some $q>4$;
\item conditional Lindeberg holds;
\item
\[
 \varepsilon^2\sum_{k<N_t^\varepsilon}
 \E[M_k^\varepsilon\otimes M_k^\varepsilon\mid\F_k^\varepsilon]
 \longrightarrow\int_0^t\Sigma_s\,ds
\]
uniformly in $t$ in probability;
\item the predictable antisymmetric area compensator converges to
$\int_0^t\Gamma_sds$;
\item the limiting continuous martingale problem is unique.
\end{enumerate}
\end{assumption}

\begin{theorem}[Enhanced martingale limit]\label{thm:rough}
Under \cref{ass:chars}, the canonical lifts converge in every admissible
$p$-variation topology to the geometric lift of the continuous Gaussian
martingale with bracket $\int_0^t\Sigma_sds$, translated in its antisymmetric
second level by $\int_0^t\Gamma_sds$.  Uniformly Lipschitz RDE solution maps
preserve the convergence.
\end{theorem}

\begin{proof}
Conditional Lindeberg and bracket convergence give the martingale functional
CLT.  Burkholder--Davis--Gundy estimates with $q>4$ give first- and
second-level tightness.  The symmetric part of \eqref{eq:second} is fixed by
the geometric identity
\[
 \operatorname{Sym}\mathbb W^\varepsilon_{s,t}
 ={1\over2}W^\varepsilon_{s,t}
  \otimes W^\varepsilon_{s,t}.
\]
The predictable antisymmetric compensator identifies the remaining area.
Uniqueness identifies all subsequences, and continuity of the It\^o--Lyons map
gives the RDE conclusion.
\end{proof}

\section{Actual nonautonomous moving-cut homogenization}\label{sec:actual}
Let $\Theta:\R^m\to I$ and $b:\R^m\to\R^m$ be globally Lipschitz.  Choose
bounded branch vectors $c_i(a)$, put
\[
 \bar c(a)=\sum_iw_i(a)c_i(a),
 \qquad m_i(a)=c_i(a)-\bar c(a),
\]
and assume
\[
 \Sigma(a)=\sum_iw_i(a)m_i(a)\otimes m_i(a)\ge\lambda I_m.
\]
On slow time $t_k=k\varepsilon^2$, set
\begin{align*}
 A_k&=\Theta(X_k^\varepsilon),\\
 X_{k+1}^\varepsilon&=X_k^\varepsilon+
 \varepsilon m_{I_k}(A_k)+\varepsilon^2b(X_k^\varepsilon),\\
 X_{k+1}^{\rm f}&=T_{A_k}(X_k^{\rm f}).
\end{align*}

\begin{theorem}[Endogenous diffusion and rough limit]\label{thm:homogenization}
The slow interpolation converges to
\[
 dX_t=b(X_t)dt+\sigma(\Theta(X_t))dW_t,
 \qquad\sigma(a)\sigma(a)^T=\Sigma(a).
\]
The enhanced driver converges jointly with zero antisymmetric anomaly.  Smooth
vector-field recursions converge to the corresponding geometric rough limit,
with the standard It\^o--Stratonovich bracket conversion when written as an
SDE.
\end{theorem}

\begin{proof}
By \cref{thm:innovations},
$M_k=m_{I_k}(A_k)$ is a martingale difference with conditional covariance
$\Sigma(A_k)$.  It is bounded, so Lindeberg and all required moments hold.
Tightness and continuity make the predictable bracket a Riemann sum for
$\int_0^t\Sigma(\Theta(X_s))ds$.  There is no one-step predictable
antisymmetric area.  Apply \cref{thm:rough}; global Lipschitz structure and
uniform ellipticity give uniqueness.
\end{proof}

Thus the response coefficients and the invariance principle belong to one
actual system, while remaining logically distinct proofs.

\section{Quantitative block freezing}\label{sec:block}
Suppose a frozen enhanced law has a uniform rate
$\eta(N)\le CN^{-\delta}$, $\delta>0$.  A block of $m_\varepsilon$ steps has
duration $h_\varepsilon=\varepsilon^2m_\varepsilon$.

\begin{proposition}[Block window]\label{prop:block}
If
\[
 m_\varepsilon\to\infty,
 \qquad h_\varepsilon\to0,
 \qquad
 \varepsilon^{-2}m_\varepsilon^{-(1+\delta)}\to0,
\]
and accumulated freezing and switching errors vanish, then frozen block
replacement holds on the full sequence.  For
$m_\varepsilon=\varepsilon^{-\kappa}$ the exact nonempty window is
\[
 {2\over1+\delta}<\kappa<2.
\]
\end{proposition}

\begin{proof}
There are $O(h_\varepsilon^{-1})$ blocks, so accumulated law error is bounded
by
$C h_\varepsilon^{-1}\eta(m_\varepsilon)
=C\varepsilon^{-2}m_\varepsilon^{-(1+\delta)}$.
\end{proof}

\section{Sharp Gaussian benchmark}\label{sec:Gaussian}
Let $W_N$ be the polygonal normalized partial sums of a deterministic Gaussian
Bernoulli shift and let $\mathbf W_N$ be its canonical step-two lift.  Choose
\[
 p>6,\qquad {1\over3}<\eta-{1\over p},\qquad\eta<{1\over2}.
\]
Let $\mathcal W_1^{\eta,p}$ be Wasserstein one for the metric combining
$W^{\eta,p}$ at level one and $W^{2\eta,p/2}$ at level two with square-root
scaling.

\begin{theorem}[Sharp rough Wasserstein rate]\label{thm:rate}
For enhanced Brownian motion $\mathbf B$,
\[
 c_{\eta,p}N^{-(1/2-\eta)}
 \le\mathcal W_1^{\eta,p}
 (\mathcal L(\mathbf W_N),\mathcal L(\mathbf B))
 \le C_{\eta,p}N^{-(1/2-\eta)}.
\]
\end{theorem}

\begin{proof}
Couple the Gaussian increments with the Brownian mesh increments; then
$W_N=P_NB$.  The difference is a concatenation of independent Brownian
bridges.  Scaling the fractional-Sobolev double integral gives
\[
 \E\norm{B-P_NB}_{W^{\eta,p}}^p
 \le C N^{-p(1/2-\eta)}.
\]
Chen's identity splits the second-level difference into local bridge L\'evy
areas and bridge--coarse cross integrals.  Local area scales as $N^{-1}$; after
the $W^{2\eta,p/2}$ norm and rough square root it has the same
$N^{-(1/2-\eta)}$ rate.  Cross terms are bounded by the product of the bridge
first-level norm and the coarse Brownian norm.

For the lower bound, let $V_N$ be the closed space of mesh-polygonal paths and
$F_N(\mathbf x)=\dist_{W^{\eta,p}}(x^1,V_N)$.  It is one-Lipschitz and
vanishes on $\mathbf W_N$.  The independent bridge residual on each interval,
a fractional Poincar\'e inequality, and scaling give
$\E F_N(\mathbf B)\ge cN^{-(1/2-\eta)}$.  Kantorovich duality concludes.
\end{proof}

\section{Pointwise entropic recursion and HJB}\label{sec:HJB}
Let $m_i(a)$ be bounded globally Lipschitz, centered under $w_i(a)$, with
uniform covariance $\Sigma(a)\ge\lambda I$.  For globally Lipschitz
$\Theta,b$ and $h>0$, define
\[
 (\mathcal T_h\phi)(x)={1\over\vartheta}\log
 \sum_iw_i(\Theta(x))
 \exp\left\{\vartheta
 \phi(x+h b(x)+\sqrt h\,m_i(\Theta(x)))\right\}.
\]
Let $u_h(T)=\phi$ and $u_h(kh)=\mathcal T_hu_h((k+1)h)$.

\begin{lemma}[Pointwise consistency]\label{lem:consistency}
The scheme is monotone, cash additive, and stable.  For $\phi\in C_b^3$,
\[
 {\mathcal T_h\phi-\phi\over h}\longrightarrow
 b\cdot D\phi+{1\over2}\Sigma(\Theta(x)):D^2\phi
 +{\vartheta\over2}D\phi^T\Sigma(\Theta(x))D\phi
\]
locally uniformly.
\end{lemma}

\begin{proof}
Taylor-expand the four increments.  Centering cancels the $\sqrt h$ term, the
second moment gives the Hessian, and the second logarithmic cumulant gives the
quadratic gradient.  Bounded third moments control the remainder.
\end{proof}

\begin{theorem}[Global HJB and theta-semigroup]\label{thm:HJB}
For every bounded uniformly continuous terminal payoff, $u_h$ converges
locally uniformly to the unique bounded viscosity solution of
\[
 -u_t-b\cdot Du-{1\over2}\Sigma(\Theta(x)):D^2u
 -{\vartheta\over2}Du^T\Sigma(\Theta(x))Du=0,
 \qquad u(T)=\phi.
\]
If $X$ solves
$dX=b(X)dt+\sigma(\Theta(X))dW$ with $\sigma\sigma^T=\Sigma$, then
\[
 u(t,x)={1\over\vartheta}
 \log\E_{t,x}e^{\vartheta\phi(X_T)}.
\]
The operators
\[
 \cE_{s,t}^{\vartheta}[\phi](x)
 ={1\over\vartheta}\log\E_{s,x}e^{\vartheta\phi(X_t)}
\]
are monotone, cash additive, terminally stable, convex for $\vartheta>0$, and
time consistent.
\end{theorem}

\begin{proof}
Monotonicity and \cref{lem:consistency} give viscosity sub- and supersolutions
from the half-relaxed limits.  The transform $v=e^{\vartheta u}$ gives the
linear equation
\[
 -v_t-b\cdot Dv-{1\over2}\Sigma:D^2v=0,
 \qquad v(T)=e^{\vartheta\phi}.
\]
Linear comparison identifies the limits and Feynman--Kac gives the formula.
The tower property proves time consistency.
\end{proof}

\section{Conclusion}
Exact innovations provide the missing positive bridge from the moving-cut
response family to nonautonomous rough homogenization.  The macroscopic value
is produced by a pointwise recursion, and global comparison follows from an
exact transform rather than a compact gradient window.
