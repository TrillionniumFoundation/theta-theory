\section{Conditional uniformity under endogenous parameters}

Let $T_a$ be the full-cover branch map with branch intervals $I_i(a)$ and
widths $w_i(a)$.  Let $A_k$ be measurable with respect to the past branch
sigma-field $\F_k$, and set
\[
 X_{k+1}^{\rm f}=T_{A_k}(X_k^{\rm f}),
 \qquad I_k=i\quad\text{on }X_k^{\rm f}\in I_i(A_k).
\]

\begin{lemma}[Conditional disintegration]\label{lem:disintegration}
Assume $X_k^{\rm f}$ is conditionally Lebesgue given $\F_k$.  Then
\begin{align}
 \Pp(I_k=i\mid\F_k)&=w_i(A_k),\label{eq:branch-law}\\
 \mathcal L(X_k^{\rm f}\mid\F_k,I_k=i)&=
 {\mathbf 1_{I_i(A_k)}(x)dx\over w_i(A_k)},\label{eq:branch-uniform}\\
 \mathcal L(X_{k+1}^{\rm f}\mid\F_k,I_k)&=dx.\label{eq:image-uniform}
\end{align}
The last conditional law is independent of the value of $I_k$.
\end{lemma}

\begin{proof}
Given $\F_k$, the parameter is deterministic and the current density is one.
Integrating the branch indicator gives \eqref{eq:branch-law}, and Bayes'
formula gives \eqref{eq:branch-uniform}.  The affine map from the branch
interval to $[0,1]$ has derivative $1/w_i(A_k)$, so it sends the normalized
conditional density to Lebesgue density one.  The result does not depend on
$i$.
\end{proof}

\begin{theorem}[Endogenous exact innovations]\label{thm:endogenous-innovation}
If $X_0^{\rm f}$ is Lebesgue distributed, then
\cref{lem:disintegration} holds at every time.  In particular, for bounded
branch vectors $c_i(A_k)$ and
\[
 m_i(a)=c_i(a)-\sum_jw_j(a)c_j(a),
 \qquad M_k=m_{I_k}(A_k),
\]
one has
\begin{align}
 \E[M_k\mid\F_k]&=0,\label{eq:MD}\\
 \E[M_k\otimes M_k\mid\F_k]
 &=\Sigma(A_k):=\sum_iw_i(A_k)m_i(A_k)\otimes m_i(A_k).
 \label{eq:bracket}
\end{align}
\end{theorem}

\begin{proof}
Induct using \eqref{eq:image-uniform}.  Equations
\eqref{eq:MD}--\eqref{eq:bracket} follow from the branch law.
\end{proof}

The parameter may depend on an adapted slow state, a control, or a frozen
cotangent signal.  No independence of $A_k$ from the past is used.

\section{Tensorization and arbitrary finite dimension}

A four-branch centered distribution spans at most three dimensions.  To obtain
a uniformly elliptic actual model in arbitrary $d$, take a product of $d$
copies of the symplectic collision map.

\begin{proposition}[Finite-dimensional tensorization]\label{prop:tensorization}
Let
\[
 \mathbf T_{\mathbf a}=T_{a_1}\times\cdots\times T_{a_d}
\]
on $[0,1]^d$, with product Lebesgue measure.  Let each coordinate use a
predictable parameter.  Then:
\begin{enumerate}
\item the product coordinate is conditionally Lebesgue at every time;
\item the $d$ coordinate branch labels are conditionally independent given the
past;
\item choosing in coordinate $j$ a nonzero centered scalar increment
$\zeta^{(j)}$ gives the diagonal covariance
\[
 \Sigma(\mathbf a)=
 \operatorname{diag}igl(\Var_{a_1}\zeta^{(1)},\ldots,
 \Var_{a_d}\zeta^{(d)}\bigr);
\]
\item if the variances have a common positive lower bound, then
$\Sigma\ge\lambda I_d$;
\item the transfer operator, response letters, pressure, and exact-innovation
properties tensorize.
\end{enumerate}
\end{proposition}

\begin{proof}
Conditional Lebesgue measure is a product and each coordinate full-cover map
sends its conditional branch interval to Lebesgue measure.  Fubini gives the
conditional independence.  The covariance and ellipticity statements are
direct.  Tensor products of the one-coordinate centered contractions and
parameter letters give the operator assertions.
\end{proof}

\section{The canonical geometric second level}

For martingale differences $M_k^\varepsilon$, define the piecewise linear
first level
\[
 W_t^\varepsilon=\varepsilon\sum_{k<t/\varepsilon^2}M_k^\varepsilon
\]
and its canonical iterated integral.  At mesh times,
\begin{equation}\label{eq:canonical-second}
 \mathbb W_t^\varepsilon=\varepsilon^2\left(
 \sum_{0\le i<j<t/\varepsilon^2}M_i^\varepsilon\otimes M_j^\varepsilon
 +{1\over2}\sum_{i<t/\varepsilon^2}M_i^\varepsilon\otimes M_i^\varepsilon
 \right).
\end{equation}

\begin{lemma}[Geometric identity]\label{lem:geometric}
For every mesh interval $[s,t]$,
\[
 \operatorname{Sym}\mathbb W_{s,t}^\varepsilon
 ={1\over2}W_{s,t}^\varepsilon\otimes W_{s,t}^\varepsilon.
\]
\end{lemma}

\begin{proof}
Expand the tensor square.  The off-diagonal terms occur twice after
symmetrization, and the diagonal terms are supplied by the one-half sum in
\eqref{eq:canonical-second}.
\end{proof}

\begin{lemma}[Second-level moment bound]\label{lem:BDG2}
Suppose for some $q>4$,
\[
 \sup_{\varepsilon,k}\E[\abs{M_k^\varepsilon}^q\mid\F_k^\varepsilon]
 \le C_q.
\]
Then for every $p\in(2,q)$ there is $C_{p,T}$ such that
\begin{align}
 \E\abs{W_{s,t}^\varepsilon}^{p}
 &\le C_{p,T}\abs{t-s}^{p/2},\label{eq:first-moment}\\
 \E\abs{\mathbb W_{s,t}^\varepsilon}^{p/2}
 &\le C_{p,T}\abs{t-s}^{p/2}.
 \label{eq:second-moment}
\end{align}
The estimates include intervals whose endpoints are not mesh points.
\end{lemma}

\begin{proof}
The first estimate is the discrete Burkholder--Davis--Gundy inequality plus the
bounded within-step interpolation.  For the second level, write the off-
diagonal sum as the martingale transform
\[
 \varepsilon\sum_jW_{0,t_j}^\varepsilon\otimes
 (\varepsilon M_j^\varepsilon)
\]
and apply BDG conditionally, followed by \eqref{eq:first-moment}.  The diagonal
sum is controlled by the predictable quadratic variation and its martingale
fluctuation.  Short partial mesh intervals have the same bounds by the
conditional $q$th moment estimate.
\end{proof}

These estimates give tightness in every rough $p$-variation topology with
$p>2$ after the usual interpolation argument.

\section{Identification of the rough limit}

Let
\[
 \langle W^\varepsilon\rangle_t
 =\varepsilon^2\sum_{k<t/\varepsilon^2}
 \E[M_k^\varepsilon\otimes M_k^\varepsilon\mid\F_k^\varepsilon].
\]
Assume it converges uniformly in probability to
$\int_0^t\Sigma_sds$ and that conditional Lindeberg holds.

\begin{theorem}[Lifted martingale convergence]\label{thm:lifted-martingale}
The first level converges to the continuous Gaussian martingale with the stated
bracket.  The canonical second level converges to its Stratonovich lift plus
the limit of any predictable antisymmetric compensator.  If the increments
are exact one-step innovations depending only on the current branch, the
predictable antisymmetric compensator is zero.
\end{theorem}

\begin{proof}
The martingale FCLT gives the first level.  Tightness follows from
\cref{lem:BDG2}.  The geometric identity fixes the symmetric second level.
For the antisymmetric part, discrete stochastic integration identifies the
limit as the It\^o antisymmetric area plus the predictable compensator.  Exact
one-step innovations have conditionally independent current increments, so the
conditional expectation of an antisymmetric product involving the current
increment vanishes.  Martingale-problem uniqueness identifies the full limit.
\end{proof}

\section{Endogenous bracket convergence}

Let
\[
 X_{k+1}^\varepsilon=X_k^\varepsilon+arepsilon M_k^arepsilon
 +\varepsilon^2b(X_k^\varepsilon),
 \qquad A_k=\Theta(X_k^\varepsilon).
\]
Assume $b,\Theta,\Sigma$ are globally Lipschitz and bounded.

\begin{lemma}[Riemann bracket]\label{lem:Riemann-bracket}
If the slow interpolations are tight in $C([0,T])$, then along every convergent
subsequence,
\[
 \sup_{t\le T}\left\vert
 \varepsilon^2\sum_{k<t/\varepsilon^2}\Sigma(A_k)
 -\int_0^t\Sigma(\Theta(X_s))ds
 \right\vert\longrightarrow0
\]
in probability.
\end{lemma}

\begin{proof}
Split the error into the ordinary Riemann-sum error for a uniformly continuous
limit path and the error from replacing the subsequential slow interpolation
by its limit.  Lipschitz continuity of $\Theta$ and $\Sigma$ controls the
second term uniformly.  Tightness and Skorokhod representation may be used to
perform the argument almost surely on a subsequence.
\end{proof}

Together with \cref{thm:endogenous-innovation,thm:lifted-martingale}, this
identifies the state-dependent diffusion limit.

\section{The microscopic entropic operator}

Let
\[
 (\mathcal T_h\phi)(x)={1\over\vartheta}
 \log\sum_iw_i(\Theta(x))
 e^{\vartheta\phi(x+hb(x)+\sqrt h\,m_i(\Theta(x)))}.
\]

\begin{lemma}[Uniform third-order consistency]\label{lem:uniform-consistency}
Let $K\subset\R^d$ be compact and $\phi\in C_b^3$.  Then, uniformly on $K$,
\begin{align}
 {\mathcal T_h\phi-\phi\over h}
 =&\ b\cdot D\phi+{1\over2}\Sigma:D^2\phi
 +{\vartheta\over2}D\phi^T\Sigma D\phi+R_h,\\
 \sup_{x\in K}\abs{R_h(x)}&\le C_Kh^{1/2}.
 \label{eq:remainder}
\end{align}
\end{lemma}

\begin{proof}
Write
\[
 \Delta_i=hb+\sqrt h\,m_i,
 \qquad
 \phi(x+\Delta_i)-\phi(x)
 =D\phi\cdot\Delta_i+{1\over2}D^2\phi[\Delta_i,\Delta_i]
 +O(\abs{\Delta_i}^3).
\]
The branch mean of $m_i$ is zero and its second moment is $\Sigma$.
The bounded third branch moment makes the averaged Taylor remainder
$O(h^{3/2})$.  Expanding the exponential and then the logarithm, the square of
the centered $\sqrt h$ first-order term contributes
$\vartheta D\phi^T\Sigma D\phi/2$.  All omitted cumulants are
$O(h^{3/2})$, yielding \eqref{eq:remainder} after division by $h$.
\end{proof}

\begin{theorem}[Pointwise viscosity convergence]\label{thm:pointwise-viscosity}
Suppose $b,\Theta,\Sigma$ are globally Lipschitz and
$\Sigma\ge\lambda I$.  The terminal recursion generated by $\mathcal T_h$
converges locally uniformly to the unique bounded viscosity solution of
\[
 -u_t-b\cdot Du-{1\over2}\Sigma:D^2u
 -{\vartheta\over2}Du^T\Sigma Du=0.
\]
\end{theorem}

\begin{proof}
The operator is monotone and cash invariant, so the recursion is stable in the
sup norm.  Let a smooth test touch the upper half-relaxed limit strictly from
above.  Choose ordinary maximizing points of the scalar slow-state functions.
Monotonicity transfers the discrete inequality to the test function, and
\cref{lem:uniform-consistency} gives the viscosity subsolution inequality.
The lower relaxed limit is analogous.  The exponential transform
$v=e^{\vartheta u}$ gives a linear uniformly parabolic equation, hence global
comparison and equality of the relaxed limits.  No microscopic distribution
is evaluated at a contact point.
\end{proof}

\section{Controlled consistency}

For compact control set $U$, let the coefficients depend continuously on
$\alpha\in U$ and define the supremum of the one-step entropic operators minus
$h\ell(x,\alpha)$.

\begin{proposition}[Uniform controlled limit]\label{prop:controlled}
If the coefficient bounds, ellipticity, and third moments are uniform in
$\alpha$, then the controlled one-step operator is monotone and uniformly
consistent with
\[
 \sup_{\alpha\in U}\left\{
 b_\alpha\cdot p+{1\over2}\Sigma_\alpha:X
 +{\vartheta\over2}p^T\Sigma_\alpha p-\ell_\alpha
 \right\}.
\]
Its terminal recursion converges to the unique bounded viscosity solution of
the corresponding risk-sensitive HJB equation.
\end{proposition}

\begin{proof}
The estimate \eqref{eq:remainder} is uniform over the compact control set.
Suprema preserve monotonicity.  Uniformly elliptic HJB comparison and the
half-relaxed-limit argument give convergence.
\end{proof}

\section{Forward theta-independence}

For a past sigma-field $\mathcal G$, a $\mathcal G$-measurable variable $X$,
and a future collision innovation $Y$, define forward
$\vartheta$-independence by
\[
 {1\over\vartheta}\log\E[e^{\vartheta f(X,Y)}\mid\mathcal G]
 =\left.{1\over\vartheta}\log\E e^{\vartheta f(x,Y)}\right|_{x=X}.
\]

\begin{proposition}[Exact collision theta-independence]\label{prop:theta-independent}
The next branch label and normalized image coordinate of the moving full-cover
collision system are forward $\vartheta$-independent of the past, with the
future law evaluated at the current predictable parameter.  Iteration gives
the exact nonlinear tower property of the microscopic recursion.
\end{proposition}

\begin{proof}
Condition on the past and apply \cref{lem:disintegration}.  The conditional
future law is exactly the branch-width law followed by an independent uniform
image coordinate.  Therefore the conditional exponential integral is the
same integral obtained by freezing the past variable at its realized value.
Repeated conditional integration proves the tower property.
\end{proof}

\section{Conclusion}

The same deterministic collision system supplies the martingale array, its
geometric rough lift, the state-dependent bracket, and the pointwise
risk-sensitive DPP.  Tensorization provides arbitrary finite-dimensional
ellipticity.  The proofs do not use a response derivative as a functional
limit theorem and do not convert an anisotropic pairing into point evaluation.
