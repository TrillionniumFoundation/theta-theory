\section{Round-twenty-nine reconstruction: weighted local regularity, typed suspension renewal, and vertical-line memory}
\label{sec:r29-a4}

Weak-Harris transportation and the operator function space serve different
purposes and are no longer identified.  The transportation cost remains
capped, whereas the function seminorm is weighted so that unbounded functions
can have unbounded oscillation.  The power drift is imported from the
fractional moment proved in A3.  Renewal is written on the explicit suspension
state $(h,m,r)$, and the memory expansion uses only a vertical half-plane
resolvent estimate, not an unproved sectoriality assertion.

\subsection{History transportation and a compatible function space}

Let $K(h,dm)$, $U(h,m)$, $W$, and
$PW\le CW^\alpha$, $\alpha<1$, be the A3 objects.  Set
\[
 d_\sigma(h,h')=\sum_{j\ge0}\sigma^j
  (1\wedge d_E(m_{-j},m'_{-j})),\qquad0<\sigma<1.          \tag{A4.1}
\]
The weak-Harris transportation cost is
\[
 d_H(h,h')=\min\left\{1,
 \sqrt{d_\sigma(h,h')[1+W(h)+W(h')] }\right\}.             \tag{A4.2}
\]
For $0<\theta<1$ and $0<\gamma<1$ define instead
\[
 \norm{f}_{\cB_{\gamma,\theta}}
 =\sup_h{\abs{f(h)}\over1+W(h)^\gamma}
 +\sup_{h\ne h'}{\abs{f(h)-f(h')}\over
 d_\sigma(h,h')^\theta[1+W(h)^\gamma+W(h')^\gamma]}.
                                                                    \tag{A4.3}
\]
The second denominator grows with the weights at distant points and tends to
zero with $d_\sigma$ locally.  Thus (A4.3) permits growth of order $W^\gamma$
without forcing globally bounded oscillation.

\begin{lemma}[Local coupling and sublevel contraction]
\label{lem:r29-a4-coupling}
There are $n_0$, $\delta>0$, and $\kappa<1$ such that
\[
 \mathcal W_{d_H}(P^{n_0}(h,\cdot),P^{n_0}(h',\cdot))
 \le\kappa d_H(h,h')                                     \tag{A4.4}
\]
when $d_H(h,h')<\delta$.  For every $L$ there are $n_L$ and
$\epsilon_L>0$ with
\[
 \sup_{W(h)+W(h')\le L}
 \mathcal W_{d_H}(P^{n_L}(h,\cdot),P^{n_L}(h',\cdot))
 \le1-\epsilon_L.                                        \tag{A4.5}
\]
No exact-tail minorizing measure is asserted.
\end{lemma}

\begin{proof}
Histories with a long common recent suffix have next-mark densities whose
weighted total-variation difference is controlled by the A2 stable
holonomy estimate.  On a successful maximal coupling, the common suffix grows
and $d_\sigma$ is multiplied by $\sigma$; the A3 power drift controls the
weight in (A4.2).  On a Lyapunov sublevel, finitely many recent cylinder types
cover the state.  A finite regular connector family couples a new suffix with
positive probability.  The old exact tails remain different, but their
contribution is multiplied by $\sigma^{n_L}$.  This proves (A4.4)--(A4.5).
\end{proof}

\begin{theorem}[Weighted weak-Harris operator gap]
\label{thm:r29-a4-harris}
For suitable $\theta,\gamma$ there are $C<\infty$ and $0<\lambda<1$ such
that
\[
 \norm{P^nf-\pi(f)}_{\cB_{\gamma,\theta}}
 \le C\lambda^n\norm{f}_{\cB_{\gamma,\theta}}.            \tag{A4.6}
\]
The invariant probability is unique among measures with finite $W$ moment.
\end{theorem}

\begin{proof}
The weak-Harris theorem applied to (A4.2), (A4.4), (A4.5), and the affine
Lyapunov drift gives exponential contraction of probability measures in a
weighted Wasserstein distance.  To obtain (A4.6), apply the coupling estimate
to the weighted difference quotient in (A4.3).  On close pairs, successful
coupling contracts $d_\sigma^\theta$; on distant pairs, the denominator is
comparable with $1+W^\gamma+W'^\gamma$ and the power drift controls the
weighted supremum.  These two estimates are stable under completion, proving
the operator gap on the declared Banach space.
\end{proof}

\subsection{Analytic Feynman--Kac sources on one space}

Let $\mathcal U_{\delta,\theta}$ consist of functions $g$ with
\[
 \norm{g}_{\mathcal U}=
 \sup_h{\abs{g(h)}\over1+\log W(h)}+
 \sup_{h\ne h'}{\abs{g(h)-g(h')}\over
 d_\sigma(h,h')^\theta[1+\log W(h)+\log W(h')]}<\infty,   \tag{A4.7}
\]
and $\abs{g(h)}\le g_0+\delta\log W(h)$.  Choose
$\alpha(\gamma+\delta)<\gamma$.

\begin{theorem}[Fixed-space analytic pressure]
\label{thm:r29-a4-fk}
On every sufficiently small ball in $\mathcal U_{\delta,\theta}$,
\[
 P_gf=P(e^gf)                                             \tag{A4.8}
\]
is an analytic bounded-operator family on the single space
$\cB_{\gamma,\theta}$.  Its leading eigenvalue and Riesz projection are simple
and analytic.  For every bounded Holder potential $g_*$, the same conclusion
holds on a neighbourhood of $g_*$ after replacing $W$ by an equivalent
Lyapunov weight.
\end{theorem}

\begin{proof}
The pointwise estimate follows from
\[
 P(e^{g}(1+W^\gamma))
 \le C P(1+W^{\gamma+\delta})
 \le C'(1+W^{\alpha(\gamma+\delta)})
 \le C''(1+W^\gamma).
\]
For the difference quotient, split the multiplier difference and the coupled
function difference.  The source norm (A4.7) gives the same weighted local
factor as (A4.3), while Theorem~\ref{thm:r29-a4-harris} controls the kernel
part.  The operator series in $g$ converges normally because
$(1+\log W)^k/k!$ is absorbed by $W^\delta$.  Analytic perturbation of the
isolated rank-one Riesz projection gives the first claim.  A bounded $g_*$
changes the drift constants but not the Lyapunov exponent, so the argument
restarts on an equivalent weight.  This supplies the finite-amplitude charts
needed by exposing-ray arguments without claiming one global unbounded
analytic ball.
\end{proof}

\subsection{The explicit suspension and renewal identity}

The suspension state is
\[
 \Omega=\{(h,m,r):m\in\operatorname{supp}K(h,\cdot),
                       0\le r<\tau(m)\}.                  \tag{A4.9}
\]
At $r=\tau(m)$ the process returns to the base state $U(h,m)$ and samples a
new mark.  Let $\cX$ be the weighted flight space on $\Omega$ and
$\cY=\cB_{\gamma,\theta}$ the base space.  For $\operatorname{Re}z$ large set
\begin{equation}\tag{A4.10}
\begin{split}
 (R_0(z)f)(h,m,r)&=\int_0^{\tau(m)-r}e^{-zs}f(h,m,r+s)\,ds,\\
 (B(z)f)(h)&=\int K(h,dm)\int_0^{\tau(m)}e^{-zs}f(h,m,s)\,ds,\\
 (T(z)F)(h)&=\int K(h,dm)e^{-z\tau(m)}F(U(h,m)),\\
 (A(z)F)(h,m,r)&=e^{-z(\tau(m)-r)}F(U(h,m)).
\end{split}
\end{equation}
Thus
$R_0:\cX\to\cX$, $B:\cX\to\cY$, $T:\cY\to\cY$, and
$A:\cY\to\cX$.

\begin{proposition}[Typed suspension renewal identity]
\label{prop:r29-a4-renewal}
On the common resolvent set,
\[
 (z-L_{\rm sus})^{-1}
 =R_0(z)+A(z)(I-T(z))^{-1}B(z).                           \tag{A4.11}
\]
All residual-flight coordinates and all base updates are explicit in
(A4.10).
\end{proposition}

\begin{proof}
Integrate the resolvent equation along the residual part of the current
flight.  This gives $R_0(z)f$ and a boundary value at $U(h,m)$ multiplied by
$e^{-z(\tau(m)-r)}$, which is $A(z)$.  A complete subsequent flight contributes
$B(z)f$; each further return contributes $T(z)$.  The exponential roof moment
justifies Fubini and the geometric series for large $\operatorname{Re}z$.
Both sides are analytic operator families, so equality extends to the common
resolvent set.
\end{proof}

\subsection{Stable compression with complete graph hypotheses}

Let $L$ be a closed generator on a Hilbert space $\cH$.  Let $P_0$ be a
finite-rank spectral projection commuting with $L$, put $Q_0=I-P_0$, and
assume
\[
 \norm{e^{tL}Q_0}_{\cH\to\cH}
 +\norm{e^{tL}Q_0}_{D(L)\to D(L)}\le Me^{-\omega t}.       \tag{A4.12}
\]
Let $P$ be another projection with
$\operatorname{rank}P=\operatorname{rank}P_0$, preserving $D(L)$, and suppose
\[
 \norm{P-P_0}_{D(L)\to D(L)}
 +\norm{(P-P_0)LQ_0}_{D(L)\to\cH}
 +\norm{Q_0L(P-P_0)}_{D(L)\to\cH}<\varepsilon.            \tag{A4.13}
\]

\begin{lemma}[Graph-isomorphic compression]
\label{lem:r29-a4-compression}
For $\varepsilon$ sufficiently small, the map
$S=Q_0Q:P\!^\perp D(L)\to Q_0D(L)$ is a graph-norm isomorphism.  In these
coordinates
\[
 S(QLQ)S^{-1}=L|_{Q_0D(L)}+K,\qquad
 \norm K_{D(L)\to\cH}\le C\varepsilon,                   \tag{A4.14}
\]
and $L_Q=QLQ$ generates a semigroup with
\[
 \norm{e^{tL_Q}Q}_{\cH\to\cH}
 +\norm{e^{tL_Q}Q}_{D(L_Q)\to D(L_Q)}
 \le2Me^{-\omega t/2}.                                   \tag{A4.15}
\]
\end{lemma}

\begin{proof}
Equal finite ranks and the first small-angle bound make $Q_0Q$ invertible on
the complementary graph spaces by a Neumann series.  Expanding
$Q_0Q(QLQ)QQ_0$ and using $P_0L=LP_0$ gives (A4.14); every remainder contains
one factor $P-P_0$ and is controlled by (A4.13).  The bounded perturbation
theorem on the graph space and Duhamel's formula yield (A4.15).
\end{proof}

\subsection{Memory on a vertical half-plane}

Let $\mathcal R=P\cH$, $B_Q=QLP$, and $C_Q=PLQ$.  Assume
\[
 B_Q\in\mathcal L(\mathcal R,D(L_Q^3)),\qquad
 C_Q\in\mathcal L(D(L_Q),\mathcal R).                    \tag{A4.16}
\]

\begin{theorem}[Domain-safe memory without sectoriality]
\label{thm:r29-a4-memory}
For $x\in D(L)$ with $Qx\in D(L_Q)$, $u(t)=Pe^{tL}x$ satisfies
\[
 u'(t)=PLPu(t)+\int_0^tK(t-s)u(s)\,ds+F_x(t),             \tag{A4.17}
\]
where
\[
 K(t)=C_Qe^{tL_Q}B_Q,
 \qquad F_x(t)=C_Qe^{tL_Q}Qx.                             \tag{A4.18}
\]
Both terms decay exponentially in operator or graph norm.  For every
$\sigma> -\omega/2$,
\[
 \widehat K(z)=C_Q(z-L_Q)^{-1}B_Q,
 \qquad \operatorname{Re}z\ge\sigma,                     \tag{A4.19}
\]
and, uniformly as $|z|\to\infty$ in that vertical half-plane,
\[
 \widehat K(z)=z^{-1}C_QB_Q+z^{-2}C_QL_QB_Q+O(|z|^{-3}).  \tag{A4.20}
\]
Moreover
\[
 P(z-L)^{-1}P=
 [z-PLP-C_Q(z-L_Q)^{-1}B_Q]^{-1}.                         \tag{A4.21}
\]
\end{theorem}

\begin{proof}
The block equations for $(u,v)=(Pe^{tL}x,Qe^{tL}x)$ and variation of constants
give (A4.17)--(A4.18).  Lemma~\ref{lem:r29-a4-compression} gives graph-norm
decay, which is exactly what the graph-bounded $C_Q$ requires.  Laplace
transformation yields (A4.19).  For $y=B_Qr\in D(L_Q^3)$, repeated use of
$(z-L_Q)^{-1}=z^{-1}I+z^{-1}(z-L_Q)^{-1}L_Q$ gives
\[
 (z-L_Q)^{-1}y=z^{-1}y+z^{-2}L_Qy
 +z^{-3}L_Q^2y+z^{-3}(z-L_Q)^{-1}L_Q^3y.
\]
The semigroup resolvent is uniformly bounded on
$\operatorname{Re}z\ge\sigma$, so the last two terms are $O(|z|^{-3})$ after
applying $C_Q$.  No sectorial estimate is used.  Solving the second block row
of the resolvent equation proves (A4.21).
\end{proof}

The revised platform has a true A3-derived drift, a function norm compatible
with unbounded weights, an explicit suspension renewal calculus, a complete
compression gate, and a high-frequency memory expansion valid without
analytic-semigroup assumptions.
