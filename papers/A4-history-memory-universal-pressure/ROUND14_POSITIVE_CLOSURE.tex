
\section{Round-fourteen positive closure: weighted coupling, rough kernels, and Schur-complement memory}
\label{sec:r14-a4}

The state is the genuine symbolic past and is metrized so that a remote
deterministic tail is exponentially discounted.  We use weighted Wasserstein
contraction, not a common total-variation minorization.  Feynman--Kac
potentials belong to an exponential multiplier algebra, and memory
descriptors are extracted from the final Schur complement after all
pole--zero cancellations.

\subsection{The past kernel}

Let \(\mathsf H\) be the graph-completed past space of A3.  For
\(0<\beta<1\), put
\[
 d_\beta(h,\tilde h)
 =\sum_{j\ge0}\beta^j
  \bigl(1\wedge d_E(h_{-j},\tilde h_{-j})\bigr),
 \qquad
 W(h)=1+\sum_{j\ge0}\beta^jW_E(h_{-j}).
\]
The normalized Gibbs kernel \(P(h,dh')\) appends one excursion and shifts the
past.

\begin{theorem}[Weighted Wasserstein spectral theorem]
\label{thm:r14-a4-wasserstein}
There are \(m\ge1\), \(\rho<1\), and \(C<\infty\) such that
\[
 \mathcal W_{d_\beta}\bigl(P^m(h,\cdot),P^m(\tilde h,\cdot)\bigr)
 \le \rho\,d_\beta(h,\tilde h),
 \qquad
 P^mW\le \rho W+C .
\]
Consequently \(P\) has a unique invariant probability \(\pi\), and on
\[
 \|f\|_{\beta,W}
 =\sup_h{|f(h)|\over W(h)}
 +\sup_{h\ne\tilde h}
 {|f(h)-f(\tilde h)|\over d_\beta(h,\tilde h)(W(h)+W(\tilde h))}
\]
the centered operator has an exponential spectral gap.
\end{theorem}

\begin{proof}
The normalized one-step \(g\)-function has summable past variations:
if two pasts agree in their last \(r\) excursions, bounded distortion gives
\[
 \|P(h,\cdot)-P(\tilde h,\cdot)\|_{\rm TV}
 \le C\vartheta^r .
\]
Use maximal coupling until the first common new excursion.  Once coupled, the
newly appended coordinates agree and the old discrepancy is shifted one
place, hence multiplied by \(\beta\).  If coupling fails, the distance is
bounded by the weighted tail.  Choosing \(m\) so that
\(C\vartheta^m+\beta^m<1\) gives the contraction.

The exponential excursion moment from A3 and the fact that a shift multiplies
the remote contribution by \(\beta\) give
\(P^mW\le\rho W+C\).  The weak Harris argument can be proved directly:
iterate the coupling for the Lipschitz seminorm and use the drift estimate for
the weighted supremum norm.  The resulting Doeblin--Fortet inequality and
compactness of weighted sublevels give the spectral gap.
\end{proof}

\subsection{The admissible Feynman--Kac algebra}

For \(\eta>0\), define
\[
 \mathscr V_\eta=
 \left\{V:\|V\|_{\rm Lip}<\infty,\quad
 |V(h)|\le c_V+\eta\log W(h)\right\}.
\]
Choose \(\eta\) below the exponential moment in the drift theorem.

\begin{lemma}[Exponential multiplier chart]
\label{lem:r14-a4-multiplier}
There is \(\eta_0>0\) such that, for
\(V\in\mathscr V_{\eta_0}\) in a neighborhood of zero,
\[
 P_Vf=P(e^Vf)
\]
is a bounded analytic family on a slightly heavier weighted Lipschitz space.
It has a simple positive eigenvalue \(\lambda_V\), positive eigenfunction
\(h_V\), and dual eigenmeasure \(\pi_V\), all analytic in \(V\).
\end{lemma}

\begin{proof}
Since \(e^{|V|}\le e^{c_V}W^{\eta_0}\), Hölder's inequality and the exponential
excursion moment give \(P(e^{|V|}W)<\infty\) uniformly on the source ball.
The product Lipschitz estimate follows from
\(|e^V-e^{\tilde V}|\le e^{\max(V,\tilde V)}|V-\tilde V|\).
Thus the exponential power series converges in operator norm on a heavier
weight.  The gap in Theorem~\ref{thm:r14-a4-wasserstein} and analytic Riesz
perturbation give the eigen-data.
\end{proof}

Define the Doob kernel
\[
 P_V^Df(h)
 ={1\over\lambda_Vh_V(h)}
 P\bigl(e^Vh_Vf\bigr)(h).
\]

\begin{theorem}[Doob nonlinear tower]
\label{thm:r14-a4-doob}
The kernel \(P_V^D\) is Markov and has the same weighted Wasserstein spectral
properties on a smaller chart.  The nonlinear operators
\[
 \mathcal E_n^V F
 =\log (P_V^D)^n(e^F)
\]
satisfy \(\mathcal E_{n+m}^V=\mathcal E_n^V\mathcal E_m^V\).
\end{theorem}

\begin{proof}
The eigenfunction identity gives \(P_V^D1=1\).  Positivity is immediate.
Analytic closeness to \(P\) preserves the coupling and drift inequalities.
The tower follows from the linear semigroup law before taking the logarithm.
\end{proof}

\subsection{Centered Poisson decomposition and a quenched rough limit}

For centered \(g\in\operatorname{Lip}_{\beta,W}\), set
\[
 h_g=\sum_{j\ge0}P^jg,\qquad
 D_{k+1}=h_g(H_{k+1})-Ph_g(H_k).
\]
Then \(D_{k+1}\) is a martingale difference and
\[
 \sum_{j=1}^ng(H_j)
 =\sum_{j=0}^{n-1}D_{j+1}
  +Ph_g(H_0)-Ph_g(H_n).
\]

\begin{theorem}[Uniform quenched enhanced invariance principle]
\label{thm:r14-a4-rough}
For every finite vector of centered weighted Lipschitz observables, the
rescaled additive process, its roof interpolation, and its second iterated
integrals converge, uniformly for initial pasts in a weighted compact set, to
a Brownian rough path
\[
 (B_t,\mathbb B_{s,t}),\qquad
 \mathbb B_{s,t}
 =\int_s^t(B_r-B_s)\otimes\circ dB_r+(t-s)\Gamma .
\]
The covariance and area anomaly are the absolutely convergent Green--Kubo
and antisymmetric correlation sums.
\end{theorem}

\begin{proof}
The spectral gap makes the Poisson series converge in the weighted Lipschitz
norm.  Conditional centering follows from
\(\mathbb E[D_{k+1}\mid\mathcal F_k]=Ph_g(H_k)-Ph_g(H_k)=0\).
The coupling estimate gives uniform convergence of predictable brackets on
weighted compact sets, while the exponential moment gives the conditional
Lindeberg bound.  The martingale functional CLT follows.

Apply the same decomposition to tensor products and use the
Burkholder inequality to obtain tightness of the iterated integrals.  The
antisymmetric expectation of one-step cross terms converges to \(\Gamma\).
A3 supplies the joint renewal law of the roof and terminal residual; its
exponential moment and deterministic-clock LDP imply uniform negligibility of
the interpolation remainder.  The coboundary is \(o(\sqrt n)\) uniformly on
weighted compact sets by the drift bound.
\end{proof}

\subsection{Continuous-time renewal resolvent}

Let \(\mathcal U_t^V\) be the suspension Doob semigroup and
\(L_V\) its generator on \(L^2(\Pi_V)\).  For a finite-dimensional resolved
subspace \(\mathcal R\subset D(L_V^2)\), let \(P_{\mathcal R}\) be the
orthogonal projection and
\[
 C_V(z)=P_{\mathcal R}(z-L_V)^{-1}P_{\mathcal R}.
\]
Entrance, full-excursion, and terminal-residual operators from A3 give the
renewal identity
\[
 (z-L_V)^{-1}
 =E_V(z)+A_V(z)(I-\mathcal L_{V,z})^{-1}B_V(z)
\]
on the common strip furnished by A2.

\begin{theorem}[Renewal-resolvent vertical theorem]
\label{thm:r14-a4-resolvent}
There is \(\gamma>0\) such that the displayed identity continues
meromorphically to \(\Re z>-\gamma\).  After subtracting its finite Laurent
principal parts in any compact substrip, the first two vertical derivatives
satisfy
\[
 \|\partial_z^jR_{\rm reg}(\sigma+ib)\|
 \le C(1+|b|)^{-1-\delta-j},
 \qquad j=0,1,2,
\]
uniformly for \(-\gamma/2\le\sigma\le1\).
\end{theorem}

\begin{proof}
Laplace transformation of the first-return decomposition gives the renewal
identity, including the entrance and residual operators.  On bounded
frequencies, analytic Fredholm theory applied to the A2 parent bundle gives
the meromorphic continuation.  On large frequencies, the integrable
four-range estimate of A2 applies to \((I-\mathcal L_{V,z})^{-1}\); the
entrance and residual operators gain one roof derivative, and two
integrations by parts give the stated vertical powers.  Compactness leaves
only finitely many poles in a compact substrip, counted with algebraic
multiplicity.
\end{proof}

\subsection{Memory after pole--zero cancellation}

Define the final Schur complement
\[
 \widehat K_V(z)
 =zP_{\mathcal R}-P_{\mathcal R}L_VP_{\mathcal R}-C_V(z)^{-1}.
\]
All descriptor data are extracted from \(\widehat K_V\), not separately from
poles of \(C_V\).

\begin{theorem}[Canonical descriptor--memory decomposition]
\label{thm:r14-a4-memory}
The meromorphic matrix \(\widehat K_V\) has a unique decomposition
\[
 \widehat K_V(z)
 =P_{\rm inst}(z)
  +\sum_{\zeta\in\mathcal Z}
   \sum_{\ell=1}^{m_\zeta}
   {A_{\zeta,\ell}\over(z-\zeta)^\ell}
  +\widehat K_{\rm reg}(z),
\]
where \(P_{\rm inst}\) is the polynomial part at infinity, the finite set
\(\mathcal Z\) and residues are the actual principal parts of the final Schur
complement after every pole--zero cancellation, and
\[
 \|\partial_z^j\widehat K_{\rm reg}(\sigma+ib)\|
 \le C(1+|b|)^{-1-\delta-j}.
\]
The inverse Laplace transform is
\[
 K=P_{\rm inst}(\partial_t)\delta_0
   +K_{\rm desc}+K_{\rm reg},
 \qquad
 \|K_{\rm reg}(t)\|\le Ce^{-\gamma t/2}.
\]
The descriptor realization is minimal after quotienting unreachable and
unobservable modes, and is unique up to similarity.
\end{theorem}

\begin{proof}
Compute the Smith--McMillan form of the finite matrix
\(\widehat K_V\) in the meromorphic strip.  Principal parts are taken only
after inversion of \(C_V\), so the scalar example
\(C(z)=(z+1)^{-1}\) gives \(\widehat K=0\) and contributes no descriptor.
The large-\(z\) expansion on \(\mathcal R\subset D(L_V^2)\) gives the
polynomial instantaneous part.  The vertical theorem and the matrix inverse
identity preserve the residual decay away from the finitely many genuine
zeros.  Bromwich contour shift is justified by the integrable vertical
bound, producing the exponential estimate.  Standard controllable/observable
reduction of the finite principal-part realization gives minimality and
similarity uniqueness.
\end{proof}

\begin{corollary}[Exact projected Volterra equation]
\label{cor:r14-a4-volterra}
For \(A\in D(L_V)\), \(x(t)=P_{\mathcal R}\mathcal U_t^VA\) satisfies the
causal generalized Langevin equation whose memory distribution is
Theorem~\ref{thm:r14-a4-memory}; the rough scaling of this equation is driven
by the enhanced limit in Theorem~\ref{thm:r14-a4-rough}.
\end{corollary}

\begin{proof}
Laplace transform the projected evolution and solve the Schur-complement
identity for \(C_V(z)^{-1}\).  Uniqueness of causal Laplace transforms gives
the equation.  Joint convergence follows from the integrable residual kernel,
the finite descriptor coordinates, and the rough-path continuous mapping
theorem.
\end{proof}

\begin{theorem}[Round-fourteen A4 closure]
\label{thm:r14-a4-main}
The prepared Sinai history has a proved weighted Wasserstein spectral gap, a
correctly centered quenched rough tangent, an admissible Doob nonlinear tower,
and a canonical descriptor--memory decomposition based on the final
Schur complement.
\end{theorem}

\begin{proof}
Combine Theorems~\ref{thm:r14-a4-wasserstein},
\ref{thm:r14-a4-doob}, \ref{thm:r14-a4-rough},
\ref{thm:r14-a4-resolvent}, and \ref{thm:r14-a4-memory}.
\end{proof}
