\section{Round-twenty-one positive closure: derived Harris geometry, fixed-space Feynman--Kac theory, and finite transmission closure}
\label{sec:r21-a4}

This section uses the exact history kernel constructed in A3.  Exponential
variation, drift, and minorization are proved from the billiard geometry; they
are not consequences asserted from summable variation.  The source family is
analytic on one fixed weighted Lipschitz space.  The renewal and memory
statements are obtained from an explicit induced-operator factorization and a
finite spectral transmission closure.

\subsection{Exponential variation of the exact history kernel}

Write $P_R$ for the A3 history kernel on $\mathsf X_R$.  Let
$N(x,y)$ be the common-past length and let $\chi(e)$ be the excursion cost of
A3.  Fix $0<\rho<1$ and define
\[
 V(x)=\sum_{j\ge0}\rho^j\chi(e_{-j}(x)),
 \qquad W(x)=e^{\varepsilon V(x)}.
\]

\begin{lemma}[Derived exponential $g$-variation]
\label{lem:r21-a4-variation}
There are $C<\infty$ and $0<\theta<1$ such that histories agreeing for $n$
steps satisfy
\[
 \|P_R(x,\cdot)-P_R(y,\cdot)\|_{\rm TV}
 \le C\theta^n d_{\rm loc}(x,y),
\]
and the same estimate holds for the logarithm of every branch density on a
common regular chart.
\end{lemma}

\begin{proof}
Two histories with a common suffix of length $n$ lie on stable fibres whose
separation before the current return is at most $C\Lambda^{-n}$.  The
conditional density of the next unstable coordinate is the normalized
inverse Jacobian from A2.  Its logarithm is H\"older along stable holonomies,
with the homogeneity weights already absorbed in the A2 strong norm.  Hence
the unnormalized densities differ by $C\Lambda^{-\alpha n}$.  Their
normalizing integrals satisfy the same estimate and are uniformly bounded
away from zero by the regular return construction.  Dividing and integrating
gives the total-variation bound with
$\theta=\Lambda^{-\alpha}$.  Branch births are paired in the parent-fold
coordinate, so the estimate is uniform in $R$.
\end{proof}

\begin{lemma}[Power drift]
\label{lem:r21-a4-drift}
For $\varepsilon>0$ small enough there are $0<\alpha<1$ and $C<\infty$ such
that
\[
 P_RW\le C W^\alpha.
\]
Consequently, for every $R_0<\infty$, there are $\lambda<1$ and $b<\infty$
with $P_RW\le\lambda W+b$.
\end{lemma}

\begin{proof}
After one history shift,
\[
 V(X_1)=\chi(e_1)+\rho V(x).
\]
Choose $p>1$ with $p\rho<1$.  H\"older's inequality and the exponential
moment of A3 give
\[
 P_RW(x)\le
 \bigl(P_Re^{p\varepsilon\chi(e_1)}\bigr)^{1/p}
 e^{\varepsilon\rho V(x)}
 \le C W(x)^\rho.
\]
Take $\alpha=\rho$.  Since $Cw^\alpha/w\to0$, choose $R_0$ so that
$Cw^\alpha\le\lambda w$ for $w\ge R_0$ and absorb the remaining compact range
into $b$.
\end{proof}

\begin{lemma}[Uniform small sets]
\label{lem:r21-a4-small}
For every $L<\infty$ there are $n_L$, $\epsilon_L>0$, and a probability
$\nu_L$ such that
\[
 P_R^{n_L}(x,\cdot)\ge\epsilon_L\nu_L(\cdot)
 \qquad\text{for all }W(x)\le L
\]
uniformly in $R$.
\end{lemma}

\begin{proof}
The bound $W(x)\le L$ controls the first $N_L$ excursion costs and makes the
uncontrolled past tail contribute at most $C\rho^{N_L}\log L$.  Replace each
history by its $N_L$-cylinder representative.  There are finitely many
regular cylinder types below the cost bound.  Finite primitivity of the common
Markov core supplies, for each type, a connector to one fixed regular
rectangle.  Pad the connectors by a neutral regular loop to obtain a common
length $n_L$.  On every connector, A2 distortion and clearance estimates give
a common positive lower bound for the conditional density on a smaller target
rectangle.  Lemma~\ref{lem:r21-a4-variation} transfers that lower bound from
the cylinder representative to every history in the cylinder once $N_L$ is
large.  Push normalized Lebesgue measure on the target rectangle forward by
the common suffix to obtain $\nu_L$.
\end{proof}

\subsection{A proved operator spectral gap}

Let
\[
 d_\beta(x,y)=\min\{1,\beta^{N(x,y)}+d_{\rm loc}(x,y)\}
\]
and define the weighted distance
\[
 \widetilde d(x,y)=
 \sqrt{d_\beta(x,y)\,[2+W(x)+W(y)]}.
\]
On functions put
\[
 \|f\|_{\mathcal B}=\sup_x{|f(x)|\over1+W(x)}+
 \sup_{x\ne y}{|f(x)-f(y)|\over\widetilde d(x,y)}.
\]

\begin{theorem}[Harris contraction and Banach spectral decomposition]
\label{thm:r21-a4-harris}
The space $\mathcal B$ is complete, $P_R\mathcal B\subset\mathcal B$, and
there are $C<\infty$, $0<\gamma<1$ such that
\[
 \|P_R^nf-\pi_R(f)\|_{\mathcal B}
 \le C\gamma^n\|f\|_{\mathcal B}.
\]
Thus $P_R=\mathbf1\otimes\pi_R+N_R$ with
$r(N_R)<1$ on the stated function space, uniformly in $R$.
\end{theorem}

\begin{proof}
Completeness follows by taking separately uniform weighted limits and
Lipschitz limits.  Couple one step by maximal coupling of the exact kernels.
For nearby histories, Lemma~\ref{lem:r21-a4-variation} makes the mismatch
probability $O(d_\beta)$; on a match, the common-past length increases by one.
This gives a strict contraction of $d_\beta$ below a fixed threshold.  For
large distances, Lemma~\ref{lem:r21-a4-drift} controls the weight.  On a
Lyapunov sublevel, Lemma~\ref{lem:r21-a4-small} gives a fixed chance of coupling
exactly after $n_L$ steps.  Combining the three estimates yields
\[
 \mathcal W_{\widetilde d}(\mu P_R^n,\nu P_R^n)
 \le C\gamma^n\mathcal W_{\widetilde d}(\mu,\nu)
\]
for probability measures with finite $W$ moment.

For $f\in\mathcal B$, duality with the point masses gives the Lipschitz part of
the asserted operator estimate.  The weighted supremum part follows from the
power drift and a decomposition into $\{W\le L\}$ and its complement.  The
same estimates show invariance of $\mathcal B$.  Applying them to
$f-\pi_R(f)$ proves the norm contraction and hence the displayed spectral
decomposition, rather than inferring it only from uniqueness of the invariant
measure.
\end{proof}

\subsection{A fixed analytic Feynman--Kac family}

Choose $0<\delta<1-\rho$ and define the source Banach space
\[
 \mathcal V_\delta=\left\{U:
 \sup_x{|U(x)|\over1+\delta\log W(x)}+
 \operatorname{Lip}_{\widetilde d}U<\infty\right\}.
\]
It is used as a linear parameter space; it is not falsely called an algebra.
Fix $0<\vartheta<1$ and let $\mathcal B_\vartheta$ be the preceding function
space with weight $W^\vartheta$.  Choose the source ball so that
$|U(x)|\le u_0+\delta\log W(x)$ and
$\rho(\vartheta+\delta)<\vartheta$.

\begin{theorem}[Fixed-space analytic pressure]
\label{thm:r21-a4-fk}
For $U$ in that ball,
\[
 P_{R,U}f=P_R(e^Uf)
\]
is a bounded operator on the one fixed space $\mathcal B_\vartheta$, and
$U\mapsto P_{R,U}$ is analytic in operator norm.  It has a simple positive
leading eigenvalue $e^{Q_R(U)}$, positive eigenfunction $h_{R,U}$, and
eigenmeasure $\ell_{R,U}$, all analytic in $U$.  The Doob kernel
\[
 \widehat P_{R,U}f=e^{-Q_R(U)}h_{R,U}^{-1}P_R(e^Uh_{R,U}f)
\]
is a Markov kernel with the same type of spectral gap.
\end{theorem}

\begin{proof}
For $|f|\le CW^\vartheta$,
\[
 |P_R(e^Uf)|\le Ce^{u_0}P_RW^{\vartheta+\delta}
 \le C'W^{\rho(\vartheta+\delta)}
 \le C'(1+W^\vartheta).
\]
The coupling proof of Theorem~\ref{thm:r21-a4-harris}, with one multiplier
difference inserted, gives the Lipschitz bound on the same space.  The
operator series
$P_R(e^Uf)=\sum_{n\ge0}P_R(U^nf)/n!$ converges in operator norm because the
power drift absorbs every factor $(1+\log W)^n/n!$ within the fixed weight.
This proves analyticity.

At $U=0$ the leading eigenvalue is isolated.  On a sufficiently small source
ball the common Riesz contour persists; positivity and the small-set theorem
make its rank one and give strictly positive eigenvectors.  Shrinking the ball
once more bounds $h$ and $h^{-1}$ in the weighted multiplier norm.  Conjugacy
then transfers the gap to the Doob kernel.
\end{proof}

\subsection{Enhanced suspension limits}

Let $A$ be a centered vector observable with
$|A|^{2+\epsilon}\le C(1+W)$ and let
$\widehat A=A+\varphi-\varphi\circ\sigma$ be its martingale-coboundary
decomposition obtained from the Poisson solution on $\mathcal B_\vartheta$.

\begin{theorem}[Rough-path invariance principle]
\label{thm:r21-a4-rough}
The piecewise linear lift of the collision-clock sums of $A$, together with
its antisymmetric iterated integral, converges in every $p$-variation topology
with $p>2$ to a Brownian rough path.  Its area drift is
\[
 \Gamma={1\over2}\sum_{k\ge1}
 \mathbb E_{\pi_R}[A_0\otimes A_k-A_k\otimes A_0].
\]
The result survives the suspension time change, the terminal residual, and
initial conditioning on a compact regular past cylinder, uniformly in $R$ and
in a compact source ball.
\end{theorem}

\begin{proof}
The spectral gap gives a Poisson solution in $L^{2+\epsilon}$ and a stationary
martingale difference sequence.  Conditional brackets converge by the
ergodic theorem, while the $L^{2+\epsilon}$ bound gives the martingale
Lindeberg condition.  Burkholder--Davis--Gundy applied to dyadic blocks gives
uniform $p$-variation moments for the first and second martingale levels.
Expanding the coboundary in the iterated sum leaves only endpoint terms and the
absolutely summable antisymmetric correlation series displayed above.

The roof has an exponential moment.  The renewal law and maximal inequality
therefore make the collision count divided by physical time converge
uniformly on compact intervals, and the terminal residual is
$o_{L^2}(\sqrt n)$.  The continuous mapping theorem for monotone time changes
proves the suspension statement.  Conditioning on a compact past cylinder
changes the initial density by a bounded element of $\mathcal B_\vartheta$;
the spectral estimate makes its contribution exponentially small after a
fixed burn-in, uniformly in the stated parameters.
\end{proof}

\subsection{Operator renewal on a common domain}

Let $\mathfrak H_R$ be the suspension space over the exact history kernel and
let $L_{R,U}$ be the generator of the Doob suspension.  Its graph domain is
\[
 D=\{f:f\text{ is }H^1\text{ along flights and its outgoing trace equals the
 Doob-weighted incoming trace}\}.
\]
The fixed collision charts and multiplication by $h_{R,U}$ identify all these
domains with one reference domain.  Let $R_{R,U}(z)=(z-L_{R,U})^{-1}$.

\begin{theorem}[Induced-operator renewal factorization]
\label{thm:r21-a4-renewal}
On $\operatorname{Re}z$ large,
\[
 R_{R,U}(z)=A_{R,U}(z)
 \bigl(I-\widehat P_{R,U,z}\bigr)^{-1}B_{R,U}(z)+E_{R,U}(z),
\]
where $\widehat P_{R,U,z}$ is the A2 induced operator with roof twist
$e^{-z\tau}$ transported to the exact history realization, and $A,B,E$ are
bounded entry, exit, and residual-flight operators.  The identity extends
meromorphically to a strip $\operatorname{Re}z>-\delta_0$.  Away from its
finitely many poles,
\[
 \|R_{R,U}(\sigma+ib)\|_{D\to\mathfrak H_R}
 \le C(1+|b|)^{-1}
\]
and, after two source derivatives, by $C(1+|b|)^{-1}$ on the corresponding
strong-to-weak scale.
\end{theorem}

\begin{proof}
Integrate the resolvent equation along a flight.  The incoming trace equals
the Laplace transform of the previous outgoing trace; iterating this boundary
equation gives the displayed renewal identity.  The A3 history representation
intertwines its boundary operator with the A2 physical induced operator, so
the A2 weighted bundle and Fourier estimates apply to
$\widehat P_{R,U,z}$.  Split the induced operator into its finite-rank leading
Riesz projection and a complement whose spectral radius is uniformly below
one.  On the complement the inverse is a normally convergent Neumann series;
on the finite-rank block it is a matrix meromorphic function.  This is an
actual Fredholm reduction of the full inverse, not an argument based only on
one eigenvector.

The A2 high-frequency coarea estimate applies to the boundary matrix
coefficients, while integration along the first and last residual flights
gives one additional factor $(1+|b|)^{-1}$.  Differentiating in $U$ inserts
bounded multipliers on the same fixed space by
Theorem~\ref{thm:r21-a4-fk}; the same vertical estimate follows.  Compactness
of the parameter set leaves finitely many poles in every closed substrip.
\end{proof}

\subsection{Finite transmission closure and memory}

Fix a finite-dimensional physical resolved space $\mathcal R\subset D$, with
orthogonal projection $P$.  Put
\[
 C(z)=P(z-L)^{-1}P|_{\mathcal R}.
\]
For $\operatorname{Re}z>0$, dissipativity gives
$\operatorname{Re}\langle C(z)x,x\rangle>0$ for $x\ne0$.
In a strip $\operatorname{Re}z>-\delta$, let $Z$ be the finite multiset of
zeros of $\det C(z)$ away from poles of the full resolvent.  Define once and
for all
\[
 \mathcal R^\sharp=\mathcal R+
 \sum_{z_0\in Z}\sum_{j=0}^{m(z_0)-1}
 (L-z_0)^j\operatorname{Ran}\operatorname*{Res}_{z=z_0}R(z)P.
\]

\begin{lemma}[Finite transmission closure]
\label{lem:r21-a4-transmission}
The space $\mathcal R^\sharp$ is finite dimensional, contained in $D(L^2)$
after replacing each generator vector by its graph-resolvent regularization,
and the compression
$C^\sharp(z)=P^\sharp(z-L)^{-1}P^\sharp$ has no transmission zero in a
slightly smaller strip.  Enlarging the resolved space once does not generate
new zeros there.
\end{lemma}

\begin{proof}
Theorem~\ref{thm:r21-a4-renewal} gives only finitely many zeros in the closed
strip.  At a zero $z_0$, the residue vectors appearing above are exactly the
root vectors through which a full-resolvent mode is invisible to the original
input-output pair.  Adjoining the complete finite Jordan chain makes that
mode an explicit diagonal block of the new compression.  Schur determinant
factorization then replaces its vanishing transmission factor by
$(z-z_0)^{-m}$, which is a pole of the represented mode rather than a zero.
The complementary transfer function is unchanged.  Because every zero in the
strip is treated simultaneously, no sequential redefinition can create a new
complementary zero.  Resolvent regularization preserves the finite span and
places it in $D(L^2)$.
\end{proof}

\begin{theorem}[Domain-safe exponentially integrable memory]
\label{thm:r21-a4-memory}
For $P=P^\sharp$, the exact Mori--Zwanzig identity on $D(L)$ is
\[
 {d\over dt}PX_t=PLPX_t+
 \int_0^tK(t-s)PX_s\,ds+PL e^{tQL}QX_0,
\]
with
\[
 \widehat K(z)=zP-PLP-C(z)^{-1}.
\]
There is $\delta>0$ such that
$\int_0^\infty e^{\delta t}\|K(t)\|dt<\infty$.  All terms are differentiable
in $R$ and in the source on the fixed transported graph/form domain.
\end{theorem}

\begin{proof}
Solve the $Q$-equation by variation of constants and substitute it into the
$P$-equation; this proves the time-domain identity and retains the unresolved
initial forcing.  Taking Laplace transforms and applying the block Schur
formula gives the expression for $\widehat K$.  Lemma
\ref{lem:r21-a4-transmission} removes compression zeros from the strip, while
Theorem~\ref{thm:r21-a4-renewal} supplies meromorphic continuation and the
vertical bounds.  The large-$z$ expansion on $D(L^2)$ cancels the constant and
$z^{-1}$ terms, leaving $\widehat K(\sigma+ib)=O(|b|^{-2})$.  Shift the inverse
Laplace contour to $\operatorname{Re}z=-\delta$ and integrate the
$|b|^{-2}$ bound to obtain exponential $L^1$ decay.  Fixed-domain resolvent
differentiation and the analytic source theorem justify parameter and source
derivatives under the contour integral.
\end{proof}

\begin{theorem}[Round-twenty-one A4 closure]
\label{thm:r21-a4-main}
The exact Sinai history platform has derived exponential variation,
Lyapunov/minorization estimates, a proved weighted-Lipschitz spectral gap, a
fixed-space analytic Feynman--Kac pressure, an enhanced suspension limit, an
operator renewal theorem, and a finite resolved-space memory kernel with
exponential decay.
\end{theorem}

\begin{proof}
Combine Theorems~\ref{thm:r21-a4-harris},
\ref{thm:r21-a4-fk}, \ref{thm:r21-a4-rough},
\ref{thm:r21-a4-renewal}, and \ref{thm:r21-a4-memory}.
\end{proof}
