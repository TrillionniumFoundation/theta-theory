\section{Round-thirteen positive closure: Wasserstein past kernels, Doob pressure, and descriptor memory}
\label{sec:r13-a4}

A complete past cannot satisfy total-variation Doeblin minorization because a
finite number of transitions does not erase its remote tail.  The correct
mechanism is contraction in a metric that discounts that tail.  This section
proves that contraction, constructs the Feynman--Kac eigenchart separately,
and derives memory only after an integrable renewal-resolvent theorem.

\subsection{The complete-past kernel}

Let
\[
 \mathsf H=\bigl\{H=(\ldots,A_{-2},A_{-1},A_0):
               A_j\in\mathcal A\bigr\}
\]
with the weighted metric
\[
 d_\beta(H,\widetilde H)
 =\sum_{j\ge0}e^{-\beta j}
   \bigl(1\wedge d_{\mathcal A}(A_{-j},\widetilde A_{-j})\bigr),
 \qquad \beta>0.
\]
The Markov kernel $P$ appends one excursion according to the normalized Gibbs
kernel and shifts the old past.  Let
$W(H)=1+\sum_{j\ge0}e^{-\beta j}e^{\eta r(A_{-j})}$.

\begin{theorem}[Weighted Wasserstein spectral theorem]
\label{thm:r13-a4-wasserstein}
There are $C<\infty$ and $0<\rho<1$ such that
\[
 W_{d_\beta}\bigl(P^n(H,\cdot),P^n(\widetilde H,\cdot)\bigr)
 \le C\rho^n d_\beta(H,\widetilde H)
      +C\rho^n\bigl(W(H)+W(\widetilde H)\bigr).
\]
Moreover $PW\le\rho_0W+C$, $P$ is Feller, and on the weighted Lipschitz space
$\mathrm{Lip}_W(\mathsf H)$,
\[
 \|P^nf-\pi(f)\|_{\mathrm{Lip}_W}
 \le C\rho^n\|f\|_{\mathrm{Lip}_W}.
\]
No common minorizing measure on the complete-past sigma-field is asserted.
\end{theorem}

\begin{proof}
Couple the newest symbols by the standard common-connector coupling of the
Gibbs kernel.  Once the two newest finite words coalesce, use the same future
innovation.  The remote tails remain different as measures, but after $n$
steps their contribution to $d_\beta$ is multiplied by $e^{-\beta n}$.
Before coalescence, bounded distortion and the finite connector family give a
geometric coupling time; the exponential return tail controls the weight.
Summing the two errors gives the displayed estimate.  The same return-tail
calculation gives the drift inequality.  Kantorovich duality yields the
Doeblin--Fortet estimate on weighted Lipschitz functions, and compactness of
weighted sublevels gives the simple invariant functional and the spectral
gap.
\end{proof}

\subsection{Poisson martingales and the quenched enhanced limit}

For centered $g\in\mathrm{Lip}_W$ define
\[
 h_g=\sum_{j\ge0}P^jg,
 \qquad
 D_{k+1}=h_g(H_{k+1})-Ph_g(H_k).
\]
Then
\[
 g(H_k)=D_{k+1}+h_g(H_k)-h_g(H_{k+1}).
\]

\begin{theorem}[Uniform quenched martingale--rough theorem]
\label{thm:r13-a4-rough}
For every finite vector of centered weighted Lipschitz observables, the
collision-time martingales, their predictable brackets, and their antisymmetric
iterated integrals converge, uniformly for initial pasts in a compact weighted
set, to a Brownian rough path with covariance
\[
 \Sigma_g=\pi\!
 \left[P(h_gh_g^T)-(Ph_g)(Ph_g)^T\right]
\]

after the usual symmetrization.  Suspension by the roof gives the physical-
time rough limit, including the deterministic area anomaly and the terminal
flight residual.
\end{theorem}

\begin{proof}
Theorem~\ref{thm:r13-a4-wasserstein} makes the Poisson series converge in the
weighted Lipschitz norm.  Conditional centering of $D_{k+1}$ is exact.  The
same contraction applied to the bracket observable gives locally uniform
predictable-bracket convergence.  The drift bound yields the conditional
Lindeberg estimate and uniform control of the coboundary.  Martingale rough
invariance then gives the first and second levels; the antisymmetric mean of
the coboundary loop is the area anomaly.  Joint collision/roof exponential
moments and the stopped physical-clock theorem of A3 give time inversion and
control of the terminal residual.
\end{proof}

\subsection{The Feynman--Kac eigenchart and nonlinear tower}

For a bounded weighted Lipschitz potential $V$ put
$P_Vf=P(e^Vf)$.

\begin{theorem}[Feynman--Kac eigenchart and Doob nonlinear semigroup]
\label{thm:r13-a4-doob}
There is a complex neighborhood of $0$ in
$\mathrm{Lip}_W$ on which $P_V$ has a simple eigenvalue $e^{\lambda_V}$,
a positive eigenfunction $h_V$, and a positive eigenfunctional $\ell_V$,
all analytic in $V$.  The kernel
\[
 P_V^Df=e^{-\lambda_V}h_V^{-1}P_V(h_Vf)
\]
is Markov.  Its continuous-time suspension defines the exact nonlinear tower
\[
 \mathcal E_{s,t}^VF
 =\log P_{V,s,t}^D(e^F),
 \qquad
 \mathcal E_{r,t}^V=\mathcal E_{r,s}^V\mathcal E_{s,t}^V.
\]
\end{theorem}

\begin{proof}
The weighted Lipschitz spectral gap of
Theorem~\ref{thm:r13-a4-wasserstein} is stable under sufficiently small
bounded multipliers.  Analytic Riesz perturbation gives the eigen-data and
positivity follows from the cone of positive functions.  The displayed
normalization sends $1$ to $1$.  The linear Markov tower becomes the nonlinear
tower after exponentiation and logarithm.
\end{proof}

\subsection{Continuous-time renewal resolvents}

Let $\mathcal L_{V,z}$ be the roof-twisted returned operator of A2 and let
$E_z$ and $X_z$ be the entrance and terminal-flight operators.  The Laplace
transform of the Doob suspension satisfies the exact renewal identity
\[
 (z-L_V^D)^{-1}=X_z+E_z(I-\mathcal L_{V,z})^{-1}E_z^*.
\]

\begin{theorem}[Integrable renewal-resolvent strip]
\label{thm:r13-a4-renewal}
On a common strip $\Re z>-\gamma_0$, the suspension resolvent is meromorphic.
After subtracting its finite principal parts in a smaller closed strip, the
residual $R_{\rm res}(z)$ satisfies, for $j=0,1,2$,
\[
 \|\partial_y^jR_{\rm res}(c+iy)\|
 \le C_j(1+|y|)^{-1-\delta-j},
 \qquad -\gamma<c\le c_0.
\]
The estimate is uniform in a compact Feynman--Kac source chart.
\end{theorem}

\begin{proof}
The exact renewal decomposition separates the entrance and residual-flight
integrals from the returned resolvent.  The four-range Fourier theorem of A2
gives integrable bounds for $(I-\mathcal L_{V,c+iy})^{-1}$; differentiating
$E_z$ and $X_z$ gains the displayed vertical powers.  Analytic Fredholm theory
gives finitely many poles in a compact part of the strip, while the
very-high-frequency estimate gives invertibility outside that compact part.
Subtract their complete Laurent principal parts.  Product differentiation of
the renewal identity gives the three residual bounds.
\end{proof}

\subsection{Compressed memory with all zero multiplicities}

Let $R$ be a finite-rank orthogonal projection on a resolved Hilbert space for
one Doob phase and set
\[
 C_V(z)=R(z-L_V^D)^{-1}R.
\]
Every pole and every zero of $\det C_V$ in
$\Re z\ge-\gamma$ is listed with algebraic multiplicity.  For a zero
$\zeta$ of order $m$, introduce descriptor coordinates
$t^je^{\zeta t}$, $0\le j<m$.

\begin{theorem}[Exact instantaneous--descriptor--memory decomposition]
\label{thm:r13-a4-memory}
After adjoining all pole and zero Jordan chains in the strip, the compressed
transfer admits the unique normalized decomposition
\[
 \widehat K_V(z)=zR-RL_V^DR-C_V(z)^{-1}
 =P_{\rm inst}(z)+P_{\rm desc}(z)+\widehat K_{\rm reg}(z),
\]
where $P_{\rm inst}$ is the polynomial large-$z$ part,
$P_{\rm desc}$ is the full sum of Laurent principal parts at poles and
transmission zeros, and
\[
 \|\partial_y^j\widehat K_{\rm reg}(c+iy)\|
 \le C_j(1+|y|)^{-1-\delta-j}.
\]
Consequently $K_{\rm reg}$ is an ordinary causal function satisfying
$\|K_{\rm reg}(t)\|\le Ce^{-\gamma t}$.  Descriptor exponents are retained
with their actual real parts; no unproved minimum-phase assertion is made.
\end{theorem}

\begin{proof}
Theorem~\ref{thm:r13-a4-renewal} makes $C_V$ a finite-dimensional meromorphic
matrix with high-frequency invertibility.  Smith--McMillan factorization on
the compact strip records every pole and zero with its full multiplicity.
Subtracting all principal parts and the polynomial large-$z$ terms gives the
residual vertical bounds.  Two integrations by parts on the shifted Bromwich
line make the inverse transform absolutely convergent and yield exponential
decay.  Requiring each Jordan chain to be in monic companion form fixes the
normalization and hence uniqueness.
\end{proof}

\begin{theorem}[Round-thirteen A4 closure]
\label{thm:r13-a4-main}
Prepared Sinai histories possess a weighted-Wasserstein spectral gap, a
correctly centered quenched rough tangent, an eigenfunction-normalized
nonlinear Doob tower, and an exact instantaneous--descriptor--decaying memory
decomposition.  Complete-past total-variation minorization is never used.
\end{theorem}

\begin{proof}
Combine Theorems~\ref{thm:r13-a4-wasserstein},
\ref{thm:r13-a4-rough}, \ref{thm:r13-a4-doob},
\ref{thm:r13-a4-renewal}, and \ref{thm:r13-a4-memory}.
\end{proof}
