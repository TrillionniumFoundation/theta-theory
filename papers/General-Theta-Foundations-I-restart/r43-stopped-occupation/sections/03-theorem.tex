\section{A preparation-injected acquired-geometry theorem}\label{sec:theorem}
\begin{definition}[Geometrically regular controlled instruments]\label{def:class}
In Section~\ref{sec:experiments}, let $K\subset\operatorname{int}D\subset B$ be compact convex sets with $K+B_r\subset D$ for some $r>0$. The preparation $u_0$ lies in $D$. Require the following certificates before a program, $s$, or $M$ is chosen.

\emph{Positive predictive closure.} The normalized image of every input in $D$ lies in $K$ off the common null sets. Likelihoods are strictly positive on $D$. The operator norms of the raw finite-dimensional maps are integrable. Every row $\mu_{a,u}$ is weakly continuous in $u\in D$, and
\begin{equation}\label{eq:report-continuity}
 \TV(\ell_a(\cdot,u)\lambda_a,\ell_a(\cdot,v)\lambda_a)
 \le L_{\rm rep}\norm{u-v}.
\end{equation}
No image condition on an unrepresented infinite-dimensional state is being inferred.

\emph{Actual mass and global cells.} For every $a\in A,u\in D$,
\begin{equation}\label{eq:density}
 \mu_{a,u}\le\Lambda\operatorname{vol}_d|_K.
\end{equation}
Use common finite Borel grid partitions $\mathcal P_m$ of $K$, with at most $m$ cells, compact convex cell closures and mesh at most $C_Km^{-1/d}$. Their boundaries have zero mass under every row. A bounding-cube grid intersected with convex $K$ supplies these partitions; the claimed density is an actual pushforward density, not a distribution assigned to $K$ by hand.

\emph{Feedback-uniform raw overlap.} There are $b\ge1$, $\epsilon>0$ and a probability $\zeta$ on $K$ such that
\begin{equation}\label{eq:overlap}
 \Law_u^\pi(X_b)\ge\epsilon\zeta
 \quad\text{for every }u\in D\text{ and every legal $b$-call feedback strategy }\pi.
\end{equation}
This is a certificate on the true acquired process, not on a quantized observer. The overlap can be verified by a paid positive-mass report word, as in Theorem~\ref{thm:quantum}. No per-step contraction, observer minorization or stationary initial distribution is assumed.
\end{definition}

For any probability $\nu$ define $q_M(\nu)=\inf_{r_1,\ldots,r_M\in B}\int\min_j\norm{x-r_j}^2\nu(dx)$. For a concave function $f$ and a common partition define its conditional gap
\begin{equation}\label{eq:jensen}
 \calJ_{\mathcal P}(f,\mu)
 =\sum_{C\in\mathcal P}\mu(C)f\left(\frac{\int_Cx\,d\mu}{\mu(C)}\right)-\int f\,d\mu,
\end{equation}
omitting zero-mass cells. Introduce the discounted potentials
\begin{equation}\label{eq:bellman-def}
 V_s=\min_{a\in A}K_aH_s,\qquad H_s=s g+\beta V_s,
 \qquad \psi_s=H_s/s.
\end{equation}

\begin{theorem}[Stopped causal acquired-geometry transfer]\label{thm:main}
Assume Definition~\ref{def:class} and the prepared executable task. The value in \eqref{eq:occupation} is the unique continuous solution of \eqref{eq:bellman-def}. There are constants $0<c\le C<\infty$ depending only on the fixed certificates, not on $u_0\in D$, $s\in(0,1]$ or $M$, such that for $M\ge2$
\begin{align}
 \calR_{\rm cp}(s,M;u_0)
 &=\inf_\pi\{B_s^\pi+q_M(\nu_s^\pi)\},\label{eq:cp-exact}\\
 cM^{-2/d}\le\calR_{\rm cp}(s,M;u_0)-V_s(u_0)
 &\le CM^{-2/d},\label{eq:cp-law}\\
 cM^{-2/d}\le\calR_{\rm hom}(s,M;u_0)-V_s(u_0)
 &\le C(M-1)^{-2/d}.\label{eq:hom-law}
\end{align}
The lower bounds hold against all legal randomized Borel finite-label programs, including programs with time-dependent action and update rows. The upper is achieved by one homogeneous program for the given $(u_0,s,M)$, with one initial label and at most $m=M-1$ recurrent labels.

More precisely, that program has a normalized terminal ensemble $Z=(z_j)$, with positive-mass means $q_j$, and a frozen recurrent instrument $P$ satisfying
\begin{equation}\label{eq:resolvent}
 Z=sB_0+\beta PZ
   =s\sum_{t\ge0}\beta^tP^tB_0.
\end{equation}
Here $B_0$ is the true ensemble produced by its first acquisition from $u_0$, not an invariant preparation. Its actual pooled acquired law is
\begin{equation}\label{eq:actual-pooled}
 \mu_s=s\mu_{a_0,u_0}
       +\beta\sum_{i,a}e(z_i)p_{ia}\mu_{a,q_i}.
\end{equation}
Its full excess risk has the exact representation
\begin{equation}\label{eq:exact-gap}
 R(\gamma_{s,m};u_0)-V_s(u_0)
       =\calJ_{\mathcal P_m}(\psi_s,\mu_s).
\end{equation}
The measure in this equality is generated by the executed table and its preparation-injected occupation, not a chosen stationary law. The unique resolvent solution does not assert uniqueness of the nonlinear calibration fixed point or convergence of an algorithm for finding it.
\end{theorem}

The exact construction and \eqref{eq:exact-gap} use closure, common null boundaries and finite actions; overlap is used only for the uniform Lipschitz bound on $\psi_s$. Without \eqref{eq:overlap}, the same identity and an upper with the actual finite constant $\Lip_K\psi_s$ remain valid for each fixed $s>0$. Thus the theorem separates causal realizability from a uniform resource exponent. The uniform density quantifier covers both full-history predictive states and means created by pooling: the latter need not lie on a particular prepared trajectory.

\begin{corollary}[Internally implemented geometric stopping]\label{cor:internal}
For $M\ge3$, under the same hypotheses,
\[
 cM^{-2/d}\le\calR_{\rm int}(s,M;u_0)-V_s(u_0)
 \le C\left\lfloor\frac{M-1}{2}\right\rfloor^{-2/d}.
\]
The construction uses one initial nonhalt label, $m$ recurrent nonhalt labels and $m$ halt labels. Its independent fresh stop coin and halt status are part of the implementation; it does not store an elapsed-time counter. Its mean and pathwise call budgets are those specified after \eqref{eq:stop}.
\end{corollary}
