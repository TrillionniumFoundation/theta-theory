\section{Stationary acquired-geometry transfer}\label{sec:theorem}
\begin{definition}[Stationary acquired-overlap class]\label{def:class}
In the experiment of Section~\ref{sec:experiments}, choose compact convex sets $K\subset\operatorname{int}D\subset B$, with nonempty interior, and $r>0$ such that $K+B_r\subset D$. Assume the following fixed certificates.

\emph{Acquired regularity.} Every normalized instrument output from $D$ lies in $K$ off the common null sets. For $u\in D$ the likelihood is positive almost everywhere, the acquired row is weakly continuous in $u$, and
\[
 \mu_{a,u}\le\Lambda\,\operatorname{vol}_d|_K,
 \qquad \TV(\ell_a(\cdot,u)\lambda_a,\ell_a(\cdot,v)\lambda_a)
       \le L_{\rm rep}\norm{u-v}.
\]
These assertions hold for all actions and all incoming states in $D$, including retained barycenters not produced by an exact filter. $K$ has global Borel grid partitions of at most $m$ cells, with diameter at most $C_Km^{-1/d}$. Their boundaries have zero mass under every acquired row. Convex compact $K$ and the density bound supply such grids.

\emph{Feedback-uniform acquired overlap.} There are $b\ge1$, $\epsilon_*>0$ and a probability $\zeta$ on $K$ such that
\begin{equation}\label{eq:overlap}
 \Law_u^\pi(X_b)\ge\epsilon_*\zeta
 \quad(u\in D,\ \pi\text{ any legal $b$-call feedback policy}).
\end{equation}
Every report and action in the block is included in this actual law.

\emph{Raw averaged minorization.} For a normalized $z_*\in C$, with predictive coordinate in $K$, and $\eta_*>0$,
\begin{equation}\label{eq:minorization}
 \bar L_a z\ \ge_C\ \eta_*e(z)z_*
 \quad(z\in C,\ a\in A).
\end{equation}
This is a condition on unnormalized physical instruments, not a contraction of posterior trajectories. It is separate from \eqref{eq:overlap}. The instruments are integrable in a fixed finite-dimensional operator norm. All constants are independent of the program, $n$ and $M$.
\end{definition}

Let $p=2/d$. Denote by $\bar R(\gamma,u_0)$ the limsup of the protected average risks $R_n$ of a homogeneous program started at $u_0\in D$. The next theorem does not presuppose stationary calibration or an optimal finite observer.

\begin{theorem}[Globally calibrated stationary transfer]\label{thm:main}
Suppose Definition~\ref{def:class} and the executable realization and task hypotheses hold. There are a constant $\rho$ and a continuous concave function $h$ on $D$ satisfying
\begin{equation}\label{eq:acoe}
 \rho+h(u)=g(u)+\min_{a\in A}K_ah(u).
\end{equation}
The function $h+\norm{\cdot}^2$ is concave, and, writing $H=\osc_Dh$,
\begin{equation}\label{eq:bias-bounds}
 H\le b\osc_Dg/\epsilon_*,\qquad \Lip_Kh\le H/r,
 \qquad |J_*(n)-\rho|\le C/n.
\end{equation}
For each $M\ge2$ and each $0<\alpha\le1/2$ there is one homogeneous observer $\gamma_{M,\alpha}$ with at most $M$ labels. One label is reserved for independent erasure at probability $\alpha$ after each report; the physical state is not reset. Its actual invariant label/preparation ensemble is calibrated, and its invariant risk obeys
\begin{equation}\label{eq:stationary-bound}
 \rho+cM^{-p}\ \le\ R_\infty(\gamma_{M,\alpha})
 \ \le\rho+C(M-1)^{-p}+\alpha H.
\end{equation}
The joint physical/label ensemble of this fixed observer contracts in sum base norm by $1-\alpha\eta_*$. In particular, from every initial preparation in $D$ and every initial label,
\begin{equation}\label{eq:entrance}
 R_n(\gamma_{M,\alpha},u_0)
 \le \rho+C(M-1)^{-p}+\alpha H+\frac{2L_0}{n\alpha\eta_*}.
\end{equation}
Conversely, every legal Borel finite-state observer with at most $M$ labels, even one with time-dependent tables, satisfies
\begin{equation}\label{eq:converse-main}
 R_n\ge\rho+cM^{-p}-C/n.
\end{equation}
The converse does not grant time-dependent tables to the constructed upper.

Choosing a dyadic $\alpha$ between $\frac14(M-1)^{-p}$ and $\frac12(M-1)^{-p}$ gives
\begin{equation}\label{eq:average-law}
 cM^{-p}\le\inf_\gamma\bar R(\gamma,u_0)-\rho\le CM^{-p},
\end{equation}
uniformly in the admissible initial preparation. Moreover
\begin{equation}\label{eq:finite-law}
 cM^{-p}-C/n\le\calR_{\rm hom}(n,M)-J_*(n)
 \le C\bigl(M^{-p}+M^p/n\bigr).
\end{equation}
Thus the finite-acquisition orders match whenever $n\ge C_*M^{2p}$, for a fixed sufficiently large $C_*$. All constants depend only on the fixed experiment and task certificates, not on $n,M$.
\end{theorem}

The finite-observer upper constructs the action and compression rules through the same actual invariant ensemble. It is not a fixed-policy comparison. Equation~\eqref{eq:finite-law} gives no matching assertion for its entrance term in the complementary small-$n$ region. It concerns $N=n+2$ as a pathwise call budget for the specified random audit, not a fixed-deadline protocol using exactly $n$ reports. In particular there is no $M-n$ clock subtraction in \eqref{eq:average-law}.

\begin{corollary}[Checkpoint comparison]\label{cor:checkpoint}
For the same protected audit and action class, put
$\nu_n^\pi=n^{-1}\sum_{t=1}^n\Law_\pi(X_t)$ and
$B_n^\pi=n^{-1}\sum_{t=1}^n\E_\pi g(X_t)$. Then
\begin{equation}\label{eq:checkpoint}
 \calR_{\rm cp}(n,M)=\inf_\pi\{B_n^\pi+q_M(\nu_n^\pi)\},\qquad
 cM^{-p}\le\calR_{\rm cp}(n,M)-J_*(n)\le CM^{-p},
\end{equation}
where $q_M(\nu)=\inf_{z_1,\ldots,z_M}\int\min_j\norm{x-z_j}^2\nu(dx)$.
The decoder has no free time argument. Its global cell code suffices for the upper.
\end{corollary}

The two inequalities \eqref{eq:overlap} and \eqref{eq:minorization} have different jobs. Acquired overlap controls the full-history Bellman problem. Raw averaged minorization makes the single compiled finite observer approach its calibrated ensemble. Neither is silently substituted for the other. The exact conditional identities and the all-observer lower below do not need a stationary starting law.
