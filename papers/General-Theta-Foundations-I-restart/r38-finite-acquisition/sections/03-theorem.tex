\section{Acquired accessibility and finite resources}\label{sec:theorem}
The next certificates precede the choice of an observer. In particular, they do not assume that a quantized observer is mixing or that its centroids are calibrated.
\begin{definition}[Accessible positive instruments]\label{def:class}
In Section~\ref{sec:experiments}, let $K\subset\operatorname{int}D\subset B$ be compact convex sets, full-dimensional in the executable test chart, with $K+B_r\subset D$ for some $r>0$. Impose the following conditions, with constants independent of any program or budget.

\emph{Raw interior image.} Off a common reference-null set for each action,
\begin{equation}\label{eq:raw-interior}
 L_a(y)C\subset C_K:=\{w(1,u):w\ge0,\ u\in K\}.
\end{equation}
This is required on the entire physical cone $C$, not just on normalized states in $D$. Likelihoods are positive on $D$. The integrable finite-dimensional positive maps and affine closure are those of Section~\ref{sec:experiments}.

\emph{Acquired regularity.} On $D$, every actual row is weakly continuous and
\begin{equation}\label{eq:acquired-regularity}
 \mu_{a,u}\le\Lambda\operatorname{vol}_d|_K,\qquad
 \TV(\ell_a(\cdot,u)\lambda_a,\ell_a(\cdot,v)\lambda_a)
       \le L_{\rm rep}\norm{u-v}.
\end{equation}
Use common Borel grid cells $C_j$ whose compact convex closures in $K$ have diameter at most $C_K' M^{-1/d}$, with at most $M$ nonempty cells. Boundaries have zero mass under every row. Convex $K$ and \eqref{eq:acquired-regularity} supply these grids; no stationary law is supplied.

\emph{Local actual overlap.} For a closed $G\subset K$, a probability $\zeta$ on $K$ and $\epsilon>0$,
\begin{equation}\label{eq:local-overlap}
 \mu_{a,q}\ge\epsilon\zeta\quad(q\in G,\ a\in A).
\end{equation}
Either $G=K$, in which case put $k=0$ and $\theta=1$, or the following robust accessibility certificate holds. There are Borel report sets $E_a$, a nonnegative $L_V$-Lipschitz function $V$ on $K$, constants $R,v>0$, $0\le\lambda<1$, $c\ge0$, $r_G>0$, an integer $k\ge1$ and a mesh allowance $s_0>0$, such that
\begin{align}
 &V\le R,\qquad \{u\in K:V(u)\le r_G\}\subset G,\qquad
 \inf_{a\in A,x\in K}\int_{E_a}\ell_a(y,x)\,d\lambda_a(y)\ge v,\label{eq:quiet-mass}\\
 &V(F_a(y,q))\le\lambda V(q)+c
       \quad(q\in K,\ y\in E_a),\label{eq:quiet-drift}\\
 &\lambda^k R+(c+L_Vs_0)\sum_{i=0}^{k-1}\lambda^i\le r_G.
       \label{eq:quiet-margin}
\end{align}
The drift assertion is off the common null sets. Set $\theta=v^k$. When $k=0$ the mesh allowance is unrestricted. Only the specified reports must decrease $V$; other reports and optimal action choices need not be contractive.
\end{definition}

Compact containment of $K$ in the interior of the physical base supplies $\kappa>0$ such that
\begin{equation}\label{eq:cone-comparison}
 (1,x)-\kappa(1,q)\in C\quad(x,q\in K).
\end{equation}
We take $\kappa<1$, and set $b=k+1$ and $\gamma=\kappa\epsilon\theta\in(0,1)$. These quantities do not depend on $M$. The acquired overlap for the full-history process, derived rather than assumed separately, is
\begin{equation}\label{eq:overlap}
 \Law_u^\pi(X_{k+2})\ge\epsilon\theta\zeta
 \quad(u\in D,\ \pi\text{ any legal $(k+2)$-call feedback policy}).
\end{equation}

\begin{theorem}[Erasure-free acquired-geometry transfer]\label{thm:main}
Under Definition~\ref{def:class} and the prepared executable-task hypotheses, there are a continuous concave average-cost bias $h$ on $D$ and a scalar $\rho$ such that
\begin{equation}\label{eq:acoe}
 \rho+h(u)=g(u)+\min_{a\in A}K_ah(u).
\end{equation}
Moreover $h+\norm{\cdot}^2$ is concave, and, with $H=\osc_Dh$,
\begin{equation}\label{eq:bias-bounds}
 H\le\frac{(k+2)\osc_Dg}{\epsilon\theta},\qquad
 \Lip_Kh\le H/r,\qquad |J_*(n)-\rho|\le C_0/n.
\end{equation}
There is a fixed $M_0$ such that for every $M\ge M_0$ one homogeneous observer $\gamma_M$ uses at most $M$ labels and, simultaneously for all $n\ge1$ and preparations $u_0\in D$, has the following properties.

Its action rows and readouts are calibrated by its unique actual invariant ensemble. It uses a common report grid and has no erasure label, physical reset, retained phase counter or exact posterior register. Its positive ensemble map $P_M$ satisfies, on the direct-sum cone $C_K^M$,
\begin{equation}\label{eq:ensemble-minor}
 P_M^b Z\ge_{C_K^M}\gamma e_{\rm tot}(Z)Z_M^\circ,
 \qquad e_{\rm tot}(Z_M^\circ)=1,
\end{equation}
for an explicitly constructed ensemble $Z_M^\circ$. Consequently
\begin{equation}\label{eq:stationary-bound}
 \rho+cM^{-2/d}\le R_\infty(\gamma_M)\le\rho+CM^{-2/d},
\end{equation}
\begin{equation}\label{eq:entrance}
 R_n(\gamma_M,u_0)\le\rho+CM^{-2/d}+\frac{2L_0b}{n\gamma}.
\end{equation}
The same observer and table work for every $n$ and every initial label. Its ensemble distance from stationarity after the first acquisition is at most
$2(1-\gamma)^{\lfloor(t-1)/b\rfloor}$ in the sum base norm associated with $C_K$.

Against the unrestricted full-history acquisition comparator in the same protected task,
\begin{equation}\label{eq:finite-law}
 cM^{-2/d}\le\calR_{\rm hom}(n,M)-J_*(n)
 \le C\bigl(M^{-2/d}+n^{-1}\bigr).
\end{equation}
The lower covers every legal randomized Borel observer with at most $M$ retained labels, including time-dependent tables. The upper does not grant that larger architecture to the constructed observer. Thus
\begin{equation}\label{eq:average-law}
 \inf_{\gamma'}\limsup_{n\to\infty}R_n(\gamma',u_0)-\rho
       \asymp M^{-2/d},
\end{equation}
and the finite-acquisition orders match already when $n\ge c_*M^{2/d}$ for any fixed $c_*>0$. Constants may depend on $c_*$ and the fixed certificates, but not on $n,M$.
\end{theorem}

The finite-$n$ conclusion does not claim an unavoidable $n^{-1}$ lower term. In the complementary short-acquisition regime the displayed bounds need not match. The physical-call budget is $N=n+2$ for the hidden uniform audit, not $n$ independent samples or an internally clocked exact deadline. The cone minorization is a proof decomposition of the executed kernel, not a new resetting action. Both $Z_M^\circ$ and the invariant ensemble retain actual subprobability masses.

\begin{corollary}[Checkpoint and all-observer lower]\label{cor:checkpoint}
For $\nu_n^\pi=n^{-1}\sum_{t=1}^n\Law_\pi(X_t)$ and
$B_n^\pi=n^{-1}\sum_{t=1}^n\E_\pi g(X_t)$,
\begin{equation}\label{eq:checkpoint}
 \calR_{\rm cp}(n,M)=\inf_\pi\{B_n^\pi+q_M(\nu_n^\pi)\},\qquad
 cM^{-2/d}\le\calR_{\rm cp}(n,M)-J_*(n)\le CM^{-2/d}.
\end{equation}
Here $q_M(\nu)=\inf_{r_1,\ldots,r_M}\int\min_j\norm{x-r_j}^2\nu(dx)$, with readouts in $B$. The decoder has no free time argument. The lower in \eqref{eq:finite-law} follows for every $n$ without subtracting a Bellman boundary term.
\end{corollary}
