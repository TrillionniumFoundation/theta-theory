\begin{abstract}
We identify one nonquadratic Hamiltonian behind the law of large numbers,
Gaussian fluctuations and local dynamic large deviations of the deterministic
hard-sphere gas.  Its additional collision-flow source records actual binary
collisions.  Exact Legendre duality yields the relative-entropy action under
the kinetic balance constraint.  Collision invariants form gauge directions,
and constrained information projections select endogenous kinetic cotangents.
The Gaussian tangent recovers the fluctuating Boltzmann covariance without
replacing the full exponential collision Hamiltonian.
\end{abstract}

\section{Hamilton--Boltzmann functional}
The deterministic hard-sphere fluctuation/LDP theory of \cite{BGSS2023}
identifies the short-time cumulant Hamiltonian.  Put
\[
 d\mathfrak c=dx\,dv\,dv_*\,d\omega,
 \qquad B(v-v_*,\omega)=((v-v_*)\cdot\omega)_+.
\]
For density $f$ and fields $(p,\psi)$ set
\[
 \mathbb H(f,p,\psi)
 =\langle f,v\cdot\nabla_xp\rangle
 +\frac12\int f f_*B
 \left(e^{\Delta_\omega p+\psi}-1\right)d\mathfrak c.
\]
The elastic collision involution preserves momentum, energy and
$B\,dv\,dv_*\,d\omega$.

\begin{proposition}[Derivatives and convexity]\label{prop:derivatives}
For bounded fields in the source domain,
\[
 D_p\mathbb H(f,0,0)[q]
 =\langle f,v\cdot\nabla q\rangle
 +\frac12\int f f_*B\,\Delta_\omega q\,d\mathfrak c,
\]
\[
 D^2_{pp}\mathbb H(f,0,0)[q_1,q_2]
 =\frac12\int f f_*B\,
 \Delta_\omega q_1\Delta_\omega q_2\,d\mathfrak c.
\]
The full Hessian in $(p,\psi)$ is positive semidefinite.
\end{proposition}

\begin{proof}
Differentiate the exponential under the Gaussian velocity bound.  Every
second directional derivative is an integral of
$e^{\Delta p+\psi}$ times the product of the two increments; the associated
quadratic form is an integral of a square.
\end{proof}

The kinetic Hamilton--Jacobi equation is
\[
 \partial_tU(t,f)+
 \mathbb H\left(f,\frac{\delta U}{\delta f},0\right)=0.
\]

\section{Collision-flow Legendre dual and kinetic action}
For a nonnegative collision measure $\gamma$, impose
\[
 \begin{aligned}
 \langle f_T,\phi_T\rangle-\langle f_0,\phi_0\rangle
 -\int_0^T\langle f_t,\partial_t\phi+v\cdot\nabla_x\phi\rangle dt
 =\int\Delta_\omega\phi\,d\gamma.
 \end{aligned}
\]

\begin{theorem}[Exact density--collision duality]\label{thm:duality}
For regular paths, the Lagrangian dual of $\mathbb H$ is
\[
 \mathbb L(f,\dot f,\gamma)
 =\begin{cases}
 \displaystyle\int\ell(d\gamma/dA_f)dA_f,
 &\text{if the balance constraint holds},\\
 +\infty,&\text{otherwise}.
 \end{cases}
\]
Consequently
\[
 \cI_T(f,\gamma)=\cI_0(f_0)
 +\int_0^T\mathbb L(f_t,\dot f_t,\gamma_t)dt.
\]
\end{theorem}

\begin{proof}
Dualization in $p$ enforces the balance constraint.  Pointwise dualization in
the collision source uses
\[
 \sup_\lambda\{q\lambda-a(e^\lambda-1)\}
 =q\log(q/a)-q+a.
\]
The formula is lower-semicontinuously extended to singular measures.  It is
the deterministic hard-sphere realization of the kinetic collision-flow
action in \cite{Rezakhanlou1998}.
\end{proof}

\section{Conservation gauge and endogenous cotangents}
Let
\[
 \mathscr C=\operatorname{span}\{1,v_1,v_2,v_3,|v|^2\}.
\]
For $c\in\mathscr C$, $\Delta_\omega c=0$.  Under fixed mass, momentum and
energy, adding a time-independent $c$ to the density cotangent changes only a
fixed boundary constant.  The collision cotangent belongs to the quotient by
$\mathscr C$.

Let $C(f,\gamma)=a$ be finitely many continuous macro constraints and set
\[
 (f^a,\gamma^a)
 \in\argmin\{\cI_T(f,\gamma):C(f,\gamma)=a\}.
\]

\begin{theorem}[Kinetic I-projection and cotangent selection]\label{thm:cotangent}
If the minimizer is unique and the constraint derivative has full rank, there
is a unique cotangent class $[(p_a,\psi_a)]$ such that
\[
 \frac{d\gamma^a}{dA_{f^a}}
 =e^{\Delta p_a+\psi_a},
\]
and $(f^a,\gamma^a)$ solves the biased Boltzmann balance.  The multiplier is
the gradient of the constrained rate in strictly convex directions.  With
multiple minimizers, the output is the compact subdifferential set of phases.
\end{theorem}

\begin{proof}
Apply Fenchel--Rockafellar duality on the regular path class.  Strict
convexity of $\ell$ gives the collision-flow formula; the balance and macro
constraints provide the adjoint multipliers.  Quotienting collision invariants
removes the null directions.  Standard convex subdifferential calculus gives
the phase statement.
\end{proof}

\section{Gaussian tangent and fluctuating Boltzmann}
At zero cotangent, the first derivative is the Boltzmann drift and the second
derivative is
\[
 \boxed{
 \mathcal Q_f(q_1,q_2)
 =\frac12\int f f_*B\,
 \Delta q_1\Delta q_2\,d\mathfrak c.}
\]
The rigorous deterministic hard-sphere fluctuation theorem is proved in
\cite{BGSS2023}.

\begin{theorem}[Gaussian tangent hierarchy]\label{thm:gaussian}
Along a regular Boltzmann solution $f_t$, the Hessian of path pressure at zero
is the covariance of the Gaussian field
\[
 d\zeta_t=\mathcal L_{f_t}\zeta_t\,dt+d\eta_t,
\]
where
\[
 \E[d\eta_t(q_1)d\eta_t(q_2)]
 =\mathcal Q_{f_t}(q_1,q_2)dt.
\]
Thus the Boltzmann law, fluctuating Boltzmann equation and dynamic LDP are the
first, second and full nonlinear levels of one Hamiltonian.
\end{theorem}

\begin{proof}
Differentiate the Hamilton--Jacobi equation once and twice at zero.  The first
derivative is the Boltzmann equation.  The second is a Lyapunov equation with
drift $\mathcal L_{f_t}$ and source $\mathcal Q_{f_t}$, identifying the
Gaussian process in the weighted test-function space.
\end{proof}

\begin{theorem}[Hamilton--Boltzmann closure]\label{thm:main}
On the short-time regular source domain, collision-flow rate, endogenous
cotangent, biased kinetic equation and Gaussian tangent are Legendre transforms
or derivatives of $\mathbb H$.
\end{theorem}
