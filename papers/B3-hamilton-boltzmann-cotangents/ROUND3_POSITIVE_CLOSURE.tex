\section{Round-three positive closure: the balance-adjoint gauge complex}
\label{sec:r3-b3}

This section supplies the primal and dual spaces, proves the global integral
conjugacy, identifies the full representation gauge, and derives the joint
Gaussian tangent from the microscopic source expansion of B2 with the
prepared initial covariance of B1.

\subsection{Weighted primal spaces}

Fix \(m>6\) and write \(w(v)=1+|v|^m\).  Let
\[
 \mathcal X_T=C([0,T];\mathcal M_w(\mathbb T^3\times\mathbb R^3))
\]
with the metric generated by weighted bounded-Lipschitz tests.  Let
\(\mathfrak C_T\) be the quotient collision space
\[
 [0,T]\times\mathbb T^3\times\mathbb R^3\times\mathbb R^3
 \times\mathbb S^2
 \big/\big((v,v_*,\omega)\sim(v_*,v,-\omega)\big),
\]
and let \(\mathcal M_{w,c}^+(\mathfrak C_T)\) be positive measures with finite
weight \(1+|v|^m+|v_*|^m\).

For a nonnegative density path \(f\), define
\[
 dA_f=\frac12ff_*((v-v_*)\cdot\omega)_+
 \,dt\,dx\,dv\,dv_*\,d\omega.
\]
A pair \((f,\Gamma)\) is balanced if, for every weighted
\(C^1\) test \(p\),
\[
 \langle p_T,f_T\rangle-\langle p_0,f_0\rangle
 -\int_0^T\langle(\partial_t+v\cdot\nabla_x)p,f\rangle dt
 =\int\Delta_\omega p\,d\Gamma.
\]
Exchange symmetry and the pre/post collision involution are imposed in the
quotient definition of \(\Gamma\).

\begin{lemma}[Closedness of the balance relation]
\label{lem:r3-b3-balance-closed}
On sets with uniformly bounded \(m\)-moment and collision entropy, the balance
relation is closed under weighted weak convergence.
\end{lemma}

\begin{proof}
Truncate velocities at \(R\).  Every term in the balance is then continuous
for weighted weak convergence.  The transport tails are bounded by the
\(m\)-moment.  For the collision term, entropy coercivity and
\(ab\le e^a+\ell(b)\) give uniform integrability of
\((1+|v|^{m-1}+|v_*|^{m-1})\Gamma\).  Let \(R\to\infty\) after taking the
limit.
\end{proof}

\subsection{Integral Fenchel duality}

Let \(\ell(q)=q\log q-q+1\) and \(\ell^*(z)=e^z-1\).  For a fixed \(f\), put
\[
 \mathfrak J_f(\Gamma)
 =\begin{cases}
 \displaystyle\int\ell(d\Gamma/dA_f)dA_f,&\Gamma\ll A_f,\\
 +\infty,&\text{otherwise}.
 \end{cases}
\]
Let \(\mathcal Y_p\) be the completion of smooth weighted tests for the norm
controlling the two endpoint traces, \((\partial_t+v\cdot\nabla)p/w\), and
\(\Delta p/(1+|v|+|v_*|)\).  Let \(\mathcal Y_\psi\) be the exponential
Orlicz heart on the reference collision weight.  Smooth compactly supported
collision tests are dense in this heart.

For \((p,\psi)\in\mathcal Y_p\times\mathcal Y_\psi\), define
\[
 \begin{aligned}
 \mathfrak L_{f,\Gamma}(p,\psi)
 ={}&\langle p_T,f_T\rangle-\langle p_0,f_0\rangle
 -\int_0^T\langle(\partial_t+v\cdot\nabla)p,f\rangle dt\\
 &+\int\psi\,d\Gamma
 -\int\left(e^{\Delta p+\psi}-1\right)dA_f.
 \end{aligned}
\]

\begin{theorem}[Closed path-space duality]
\label{thm:r3-b3-duality}
For every finite-moment pair \((f,\Gamma)\),
\[
 \sup_{(p,\psi)\in\mathcal Y_p\times\mathcal Y_\psi}
 \mathfrak L_{f,\Gamma}(p,\psi)
 =\begin{cases}
 \mathfrak J_f(\Gamma),&(f,\Gamma)\text{ is balanced},\\
 +\infty,&\text{otherwise}.
 \end{cases}
\]
The supremum is unchanged if one starts with smooth compactly supported tests
and closes in the declared norms.
\end{theorem}

\begin{proof}
If balance fails, Lemma~\ref{lem:r3-b3-balance-closed} and Hahn--Banach give a
smooth weighted \(p\) with nonzero balance defect.  Set \(\psi=-\Delta p\).
The exponential term vanishes and scaling \(p\) makes the supremum infinite.

On the balanced set, the first three terms equal
\(\int\Delta p\,d\Gamma\), so
\[
 \mathfrak L_{f,\Gamma}(p,\psi)
 =\int z\,d\Gamma-\int(e^z-1)dA_f,
 \qquad z=\Delta p+\psi.
\]
Since \(p=0\) is allowed, every \(z\) in the exponential Orlicz heart is
obtained by taking \(\psi=z\).  The scalar identity
\(\ell^*(z)=e^z-1\), monotone truncation of
\(\log(d\Gamma/dA_f)\), and the singular-part test
\(z=n1_B\) prove the integral conjugacy.  Orlicz density and the tail estimate
from finite entropy justify smooth approximation and interchange of
supremum and integration.
\end{proof}

\subsection{The complete gauge quotient}

Define
\[
 \mathfrak D:\mathcal Y_p\times\mathcal Y_\psi
 \longrightarrow\mathcal Y_z,
 \qquad
 \mathfrak D(p,\psi)=\Delta p+\psi,
\]
where \(\mathcal Y_z\) is the exponential collision Orlicz space, and define
\[
 \mathfrak G r=(r,-\Delta r).
\]

\begin{theorem}[Exact gauge complex]
\label{thm:r3-b3-gauge}
The sequence
\[
 0\longrightarrow\mathcal Y_p
 \xrightarrow{\mathfrak G}
 \mathcal Y_p\times\mathcal Y_\psi
 \xrightarrow{\mathfrak D}
 \mathcal Y_z\longrightarrow0
\]
is exact.  The range of \(\mathfrak G\) is closed, and
\[
 (\mathcal Y_p\times\mathcal Y_\psi)/\operatorname{Ran}\mathfrak G
 \cong\mathcal Y_z
\]
as Hausdorff locally convex spaces.  On every balanced pair, the dual action
depends only on this quotient class.  If
\(q=d\Gamma/dA_f>0\), its unique optimal cotangent class is
\[
 [p,\psi]\quad\longleftrightarrow\quad
 z=\log q.
\]
Collision invariants are included automatically as gauge directions; they are
not the whole gauge.
\end{theorem}

\begin{proof}
Surjectivity of \(\mathfrak D\) follows from
\(z=\mathfrak D(0,z)\).  If \(\mathfrak D(p,\psi)=0\), then
\((p,\psi)=\mathfrak Gp\), proving exactness.  The range is the graph of the
continuous map \(r\mapsto-\Delta r\), hence closed.  The quotient is therefore
Hausdorff and the induced map to \(\mathcal Y_z\) is a continuous linear
bijection with continuous inverse \(z\mapsto[(0,z)]\).  On a balanced pair,
Theorem~\ref{thm:r3-b3-duality} reduces the action to the function of
\(z\) displayed in its proof.  Strict convexity of \(e^z-1\) gives the unique
optimal \(z=\log q\).
\end{proof}

\subsection{Macro constraints and multiplier existence}

Let \(C:\mathcal X_T\times\mathcal M_{w,c}^+\to\mathbb R^k\) be a continuous
linear macro map whose coordinate classes are independent in the gauge
quotient.  Let
\[
 \mathcal Q(\Theta)
 =\lim_{\varepsilon\to0}\frac1{\mu_\varepsilon}
 \log\mathbb E
 e^{\mu_\varepsilon\langle\Theta,
 (\pi^\varepsilon,\Gamma^\varepsilon)\rangle}
\]
be the prepared pressure on the B2 source ball.

\begin{lemma}[Positive quotient covariance]
\label{lem:r3-b3-positive}
At every regular biased source, the Hessian of \(\mathcal Q\) is positive
definite on every finite-dimensional family of independent quotient classes.
After microcanonical preparation it is the Schur complement of the initial
constraint covariance and remains positive on directions independent of those
constraints.
\end{lemma}

\begin{proof}
B2 identifies the Hessian with the limiting covariance of the joint density
and actual-collision observables.  Zero variance would make the observable
constant under every microscopic configuration in an open hard-sphere phase
set.  Varying one free-flight initial point and one binary contact then shows
that its density and collision parts differ by a balance gauge, contrary to
independence of the quotient classes.  The B1 conditioning formula subtracts
exactly the orthogonal projection onto the initial constraint span, giving the
Schur complement and the final assertion.
\end{proof}

\begin{theorem}[Regular information-projection multiplier]
\label{thm:r3-b3-multiplier}
Let \(a\) lie in the mean image of a regular source ball.  The constrained
information projection is unique, and there is a unique multiplier class
\([p,\psi]\) together with a unique finite-dimensional macro multiplier
\(\lambda\) such that
\[
 \mathfrak D(p,\psi)=\log(d\Gamma/dA_f),
 \qquad C(f,\Gamma)=a.
\]
Both the minimizer and the multipliers depend analytically on \(a\).
\end{theorem}

\begin{proof}
The mean map is the gradient of \(\mathcal Q\).  By
Lemma~\ref{lem:r3-b3-positive}, its derivative on the chosen quotient and
macro directions is invertible.  The analytic inverse-function theorem gives
a unique source and biased phase for every \(a\) in the mean image.  Legendre
duality with the good B2 rate identifies this phase as the unique information
projection.  Theorem~\ref{thm:r3-b3-gauge} gives the unique cotangent class.
\end{proof}

\subsection{Microscopic pressure derivatives}

For the finite hard-sphere system put
\[
 Q_\varepsilon(\Theta)
 =\frac1{\mu_\varepsilon}
 \log\mathbb E e^{\mu_\varepsilon\Theta}.
\]

\begin{proposition}[Exact normalization]
\label{prop:r3-b3-normalization}
For bounded directions \(F,G\),
\[
 DQ_\varepsilon(\Theta)[F]
 =\mathbb E_{\Theta,\varepsilon}F,
 \qquad
 D^2Q_\varepsilon(\Theta)[F,G]
 =\mu_\varepsilon
 \operatorname{Cov}_{\Theta,\varepsilon}(F,G).
\]
All first and second derivatives converge locally uniformly on a regular
source ball.  Under the microcanonical shell, the time-zero part of the limit
Hessian is the B1 Schur-complement covariance.
\end{proposition}

\begin{proof}
The finite-volume identities follow by differentiating the normalized
logarithm.  B2 gives normal convergence on a larger complex source ball, so
Cauchy estimates pass the first two derivatives through the
Boltzmann--Grad limit.  B1 gives the derivative of the source-dependent saddle,
which is exactly the stated Schur complement.
\end{proof}

\subsection{The joint density--collision Gaussian tangent}

Let \((f,\Gamma)\) be a regular biased law with
\(q=d\Gamma/dA_f\), and define centered fields
\[
 \zeta^\varepsilon
 =\sqrt{\mu_\varepsilon}(\pi^\varepsilon-f),
 \qquad
 \Xi^\varepsilon
 =\sqrt{\mu_\varepsilon}(\Gamma^\varepsilon-\Gamma).
\]
For a density test \(\phi\), let
\(\Delta\phi\) denote its collision increment.

\begin{theorem}[Prepared joint fluctuating Boltzmann limit]
\label{thm:r3-b3-gaussian}
The pair \((\zeta^\varepsilon,\Xi^\varepsilon)\) converges in the weighted
distribution path space to the unique centered Gaussian solution of the
linearized biased balance.  In weak form,
\[
 d\langle\zeta_t,\phi\rangle
 =\langle\zeta_t,
 v\cdot\nabla_x\phi+\mathcal L_{f,q}^*\phi\rangle dt
 +dM_t(\phi),
\]
with bracket
\[
 \langle M(\phi),M(\chi)\rangle_t
 =\int_0^t\int
 q(\Delta\phi)(\Delta\chi)\,dA_{f_s}\,ds.
\]
The collision fluctuation is characterized jointly by
\[
 \Xi(\psi)
 =\int\psi\,dM^{\rm coll}
 +\int\psi\,D A_f[\zeta]\,q
 +\int\psi\,A_f\,Dq[\zeta],
\]
with the first term having covariance
\(\int q\psi\chi\,dA_f\).  The initial covariance is the constrained B1
covariance.  For every finite family of density and collision tests, its
covariance is exactly the Hessian of the limiting pressure.
\end{theorem}

\begin{proof}
Normal convergence of the B2 marked cluster series on a complex neighborhood
implies convergence of all joint cumulants.  After centering and multiplying
by \(\sqrt{\mu_\varepsilon}\), a cumulant of order \(k\) is bounded by
\(C_k\mu_\varepsilon^{1-k/2}\); hence all orders \(k\ge3\) vanish, while the
second order converges by Proposition~\ref{prop:r3-b3-normalization}.

Differentiate the marked Hamilton--Jacobi equation of B2 once around the
biased solution.  The density component gives the linearized biased
Boltzmann operator.  Differentiating the collision exponential gives the
white collision bracket and the linear response of \(A_fq\), which is the
displayed joint characterization.  B2 exponential compact containment and
its short-time modulus estimates give tightness in the weighted distribution
space.  B1 supplies the initial Gaussian field and its Schur-complement
covariance.  Uniqueness of the linear Gaussian martingale problem identifies
every subsequential limit.
\end{proof}

\begin{theorem}[Round-three B3 closure]
\label{thm:r3-b3-main}
The B2 action has a closed weighted Orlicz dual, the complete cotangent is the
Hausdorff quotient by \((r,-\Delta r)\), regular macro preparations have
unique multiplier classes, and the microscopic hard-sphere fields converge to
the prepared joint Gaussian tangent of
Theorem~\ref{thm:r3-b3-gaussian}.
\end{theorem}

\begin{proof}
Combine Theorems~\ref{thm:r3-b3-duality},
\ref{thm:r3-b3-gauge}, \ref{thm:r3-b3-multiplier}, and
\ref{thm:r3-b3-gaussian}.
\end{proof}
