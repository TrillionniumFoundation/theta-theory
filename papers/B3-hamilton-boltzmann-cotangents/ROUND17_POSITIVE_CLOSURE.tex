\section{Round-seventeen positive closure: joint cumulants, observability, and quadratic Mosco limits}
\label{sec:r17-b3}

The Gaussian field is constructed first from exact finite-volume cumulants.
Only afterwards is its Cameron--Martin form identified with the second
variation of the density--contact action.  The collision noise convention is
fixed once and for all: the isonormal measure has control $qA_f$, and the
integrand is $\Delta p+\psi$; equivalently one may use control $A_f$ and
integrand $\sqrt q(\Delta p+\psi)$, but never both factors simultaneously.

\subsection{A fixed nuclear triple and the linearized balance}

Let $\mathscr S$ be the projective intersection of weighted Sobolev spaces
$H^m_xH^m_v$ with Gaussian velocity weights and compatible endpoint traces,
and let
\[
 \mathscr S\subset\mathscr H\subset\mathscr S'
\]
be the resulting nuclear Gelfand triple.  For collision tests use the
corresponding symmetric nuclear space $\mathscr S_c$ on
$(t,x,v,v_*,\omega)$ modulo $(v,v_*,\omega)\sim(v_*,v,-\omega)$.
Put $\mathscr E=\mathscr S\oplus\mathscr S_c$.

At a regular biased path $(f,qA_f)$, let $\mathcal L_t$ be the linearized
transport--collision operator.  A tangent pair $(h,K)$ satisfies
\[
 \partial_th+v\cdot\nabla_xh=\Delta^*K,
 \qquad
 K=q\,DA_f[h]+A_f\,k,
\]
where $k$ is the relative collision-intensity tangent.

\begin{theorem}[Finite-time kinetic observability]
\label{thm:r17-b3-observability}
On the orthogonal complement of mass, momentum, energy, and the exact balance
gauge, there are $c,C>0$ such that every smooth backward solution
\[
 -\partial_tp-v\cdot\nabla_xp-\mathcal L_t^*p=F,
 \qquad p(T)=p_T,
\]
satisfies
\[
 \|p\|_{L^2(0,T;H^1_{x,v,w})}^2+
 \|p(0)\|_{L^2_w}^2+\|p(T)\|_{L^2_w}^2
 \le C\left(
 \|F\|_{L^2(0,T;H^{-1}_w)}^2+
 \int|\Delta p|^2q\,dA_f
 \right).
\]
Consequently the linearized balance operator has closed range on the declared
weighted spaces and admits a bounded right inverse on its range.
\end{theorem}

\begin{proof}
Freeze coefficients on a time partition and Fourier transform in $x$.  For
the zero spatial mode, the biased collision operator is a bounded perturbation
of the Maxwellian linearized operator; the spectral gap on the complement of
collision invariants gives velocity coercivity.  For nonzero spatial modes,
the commutator of $iv\cdot k$ with a bounded velocity multiplier transfers the
collision coercivity to the hydrodynamic component.  The standard modified
energy
\[
 E_k(p)=\|p\|_{L^2_w}^2+
 \eta\Re\langle A_kp,p\rangle
\]
then obeys
\[
 -{d\over dt}E_k(p)+c\bigl(\|p_\perp\|_{H^1_v}^2+
 {|k|^2\over1+|k|^2}\|p_\parallel\|^2\bigr)
 \le C\|F\|_{H^{-1}}^2.
\]
The constants are uniform on the small regular source chart because $f$ and
$q$ are bounded above and below relative to the reference Maxwellian and have
uniform weighted derivatives.  Integrating, summing in $k$, and treating the
finite-dimensional invariant block by the endpoint conditions gives the
displayed estimate.  Approximation of the nonautonomous coefficients by the
time partition and Gronwall's lemma pass to the true equation.  The closed
range and bounded inverse follow from the Hilbert closed-range theorem.
\end{proof}

\subsection{Exact finite cumulants and process tightness}

For a test pair $\Theta=(p,\psi)\in\mathscr E$, let
$X_\varepsilon(\Theta)$ be the centered density/contact observable under the
regular biased microscopic law.  B2's normally convergent marked pressure,
combined with B1's exact shell saddle when preparation is imposed, gives
finite-volume cumulants $\kappa_{r,\varepsilon}$.

\begin{lemma}[Localized cumulant estimate]
\label{lem:r17-b3-cumulants}
For every Hilbertian seminorm $|\cdot|_m$ generating the nuclear topology,
there are $m'>m$ and constants $C_r$ such that for tests supported in a time
interval $I\subset[0,T]$,
\[
 |\kappa_{r,\varepsilon}(\Theta_1,\ldots,\Theta_r)|
 \le C_r\mu_\varepsilon^{1-r/2}
 (|I|+\mu_\varepsilon^{-1})
 \prod_{j=1}^r|\Theta_j|_{m'}
\]
for $r\ge2$.  The estimate is uniform over all deterministic intervals,
including intervals shorter than $\mu_\varepsilon^{-1}$, and after exact
microcanonical centering.
\end{lemma}

\begin{proof}
Differentiate the exact marked trajectory-cluster expansion with each source
localized to $I$.  Every connected diagram contributing to an $r$-th
cumulant has one chronologically first marked vertex; its time integral gives
$|I|$.  If two or more marks lie in the same microscopic contact, the diagonal
term contributes one factor $\mu_\varepsilon^{-1}$ instead.  The remaining
tree integrations are bounded by the fixed-horizon B2 factorial majorant.
After multiplying by the central-limit normalization
$\mu_\varepsilon^{-r/2}$, connectedness leaves the factor
$\mu_\varepsilon^{1-r/2}$.  The B1 source-dependent saddle is analytic with
uniform derivatives, so conditional cumulants are obtained by the finite
Schur-complement differentiation and obey the same estimate.
\end{proof}

\begin{theorem}[Joint density--contact functional central limit theorem]
\label{thm:r17-b3-clt}
The centered field
\[
 Z_\varepsilon=\sqrt{\mu_\varepsilon}
 \bigl((\pi^\varepsilon,\Gamma^\varepsilon)-(f,qA_f)\bigr)
\]
converges in $C([0,T];\mathscr S')\times
D([0,T];\mathscr S_c')$ to the unique Gaussian solution of the linearized
balance equation.  Its collision martingale is the isonormal integral
\[
 M(p,\psi)=\int(\Delta p+\psi)\,d\mathbb W_{qA_f},
\]
and hence
\[
 \mathbf E[M(p,\psi)M(\tilde p,\tilde\psi)]
 =\int(\Delta p+\psi)(\Delta\tilde p+\tilde\psi)q\,dA_f.
\]
The initial covariance is the B1 constrained Schur complement and is
propagated by the nonautonomous linearized evolution together with all
density--contact cross terms.
\end{theorem}

\begin{proof}
For every finite set of tests, Lemma~\ref{lem:r17-b3-cumulants} makes all
cumulants of order at least three vanish and identifies the limiting second
cumulant by differentiating the B2 Hamiltonian.  This gives finite-dimensional
Gaussian convergence with the displayed, singly weighted collision bracket.

For tightness, choose a Hilbertian seminorm $|\cdot|_{-m}$ for which the
embedding of the next stronger dual space is Hilbert--Schmidt.  Apply the
second- and fourth-cumulant estimates to every dyadic interval and every vector
of an orthonormal determining family.  Summation using the Hilbert--Schmidt
embedding yields
\[
 \mathbf E\|Z_\varepsilon(t)-Z_\varepsilon(s)\|_{-m}^4
 \le C(|t-s|+\mu_\varepsilon^{-1})^2.
\]
The probability of a jump larger than $\eta$ is bounded by the marked contact
exponential moment and tends to zero.  Kolmogorov--Aldous tightness followed by
Mitoma's theorem gives tightness in the stated nuclear dual path space.  The
exact microscopic balance passes to the limit and identifies the unique
Gaussian martingale problem by Theorem~\ref{thm:r17-b3-observability}.
\end{proof}

\subsection{Second-order action and Mosco identification}

Write a finite-action collision flow as $(q+
ho)A_{f+h}$ near the exposed
path.  The scalar entropy has the uniform expansion
\[
 \ell(q+
ho)=\ell(q)+\rho\log q+{\rho^2\over2q}
 +R_q(\rho),
 \qquad |R_q(\rho)|\le C|\rho|^3
\]
when $q$ stays in a compact subset of $(0,\infty)$.

\begin{theorem}[Quadratic recovery and Mosco convergence]
\label{thm:r17-b3-mosco}
Let $I$ be the B2 joint density--contact action and $x_*$ a regular exposed
path.  The rescaled convex functionals
\[
 I_\eta(z)={I(x_*+\eta z)-I(x_*)-
 \eta\langle DI(x_*),z\rangle\over\eta^2/2}
\]
Mosco-converge on the balanced tangent Hilbert space to
\[
 \mathcal Q(z)=\|\Sigma^{-1/2}z\|^2,
 \qquad D(\mathcal Q)=\operatorname{Ran}\Sigma^{1/2}.
\]
The limsup is supplied by an explicit positive, exactly balanced quadratic
recovery; it does not use the first-order LDP recovery theorem.
\end{theorem}

\begin{proof}
For the liminf, take $z_\eta\rightharpoonup z$ with bounded $I_\eta$.  Entropy
coercivity gives boundedness of the relative collision perturbations in
$L^2(qA_f)$, and the exact balance gives weak compactness of the density
perturbations.  Taylor's formula with truncation at
$|\rho|\le\eta^{-1/3}$ and entropy control of the complement yields the
quadratic lower bound.  Closedness of the balance range identifies its domain
with the quotient tangent space.

For the limsup, first take a smooth tangent $z=(h,K)$ in the range of the
linearized balance.  Use the bounded right inverse from
Theorem~\ref{thm:r17-b3-observability} to construct a second-order correction
$(h^{(2)},K^{(2)})$ cancelling the nonlinear balance defect of
$x_*+\eta z$.  Set
\[
 f_\eta=f+\eta h+\eta^2h^{(2)},
 \qquad
 \Gamma_\eta=qA_f+\eta K+\eta^2K^{(2)}.
\]
After truncating the smooth tangent, $f_\eta$ and $\Gamma_\eta$ remain
strictly positive.  Direct substitution into the entropy expansion gives
$I_\eta(z_\eta)\to\mathcal Q(z)$.  Density of smooth range vectors in
$\operatorname{Ran}\Sigma^{1/2}$ and a diagonal argument complete the
limsup.  Convex duality of the limiting Gaussian pressure identifies the
quadratic form with $\Sigma^{-1}$.
\end{proof}

\begin{theorem}[Round-seventeen B3 closure]
\label{thm:r17-b3-main}
On the regular short-time phase chart, the balance operator has closed range,
the finite microscopic cumulants generate a tight joint density--contact
Gaussian process with correctly normalized collision noise, and the joint
action is twice epi-differentiable with Cameron--Martin domain
$\operatorname{Ran}\Sigma^{1/2}$.
\end{theorem}

\begin{proof}
Combine Theorems~\ref{thm:r17-b3-observability},
\ref{thm:r17-b3-clt}, and~\ref{thm:r17-b3-mosco}, using
Lemma~\ref{lem:r17-b3-cumulants}.
\end{proof}
