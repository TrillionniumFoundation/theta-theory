% ROUND6-REFEREE-POSITIVE-CLOSURE
\section{Round-six positive closure: two-domain cotangents and exact-centred Gaussian fields}
\label{sec:r6-b3}

The local analytic source chart used for pressure derivatives is separated
from the global convex source domain used for Fenchel duality.  The latter is
the union of all bounded compactly supported density/contact cylinders, so it
can be scaled and truncated without leaving the proved B2 pressure domain.
The process Gaussian limit is obtained from time-localized connected
cumulants and conditional block estimates, not from an assumed point-process
compensator.  Every microscopic field is centred at its exact finite-volume
expectation.

\subsection{Weighted primal and cotangent spaces}

Put
\[
w(v)=1+|v|^2,
\qquad
w_c(v,v_*)=1+|v|^2+|v_*|^2.
\]
Let \(\mathcal X_T\) be the density-path space with the topology generated by
weighted bounded-Lipschitz tests and one Maxwellian Lyapunov moment inherited
from B2.  Let \(\mathfrak C_T\) be the exchange-quotiented actual-contact
space, and let \(\mathcal M_{w_c}^+(\mathfrak C_T)\) be the positive Radon
measures with finite \(w_c\)-moment.

Let \(C_{w_c,0}(\mathfrak C_T)\) be the continuous functions \(z\) such that
\(z/w_c\) vanishes at weighted infinity, equipped with the weighted strict
topology \(\beta_{w_c}\).  Let \(\mathcal Y_p\) be the completion of smooth
time-dependent density tests for
\[
\begin{aligned}
\|p\|_{\rm graph}={}&
\|p_0/w\|_\infty+\|p_T/w\|_\infty
+\|((\partial_t+v\cdot\nabla_x)p)/w\|_\infty\\
&+\|\Delta p/w_c\|_\infty
+\text{the corresponding weighted strict moduli}.
\end{aligned}
\]
Then \(p\mapsto\Delta p\) is continuous from \(\mathcal Y_p\) to
\(C_{w_c,0}\).

A pair \((f,\Gamma)\) is balanced if
\[
\langle p_T,f_T\rangle-
\langle p_0,f_0\rangle-
\int_0^T\langle(\partial_t+v\cdot\nabla_x)p,f_t\rangle dt
=\int\Delta p\,d\Gamma
\]
for every \(p\in\mathcal Y_p\).

\subsection{Local analytic and global convex source domains}

For a regular Maxwellian-biased phase with tail exponent \(\beta_*>0\), set
\[
\mathcal U_\eta=
\{(p,\psi):\|\Delta p+\psi\|_{w_c}<\eta\},
\qquad 0<\eta<\beta_*/8.
\]
On \(\mathcal U_\eta\), the exponential Hamiltonian is finite and the B2
pressure is analytic.

For global duality, let \(\mathcal D_{\rm glob}\) be the union, over all
\(R,M<\infty\), of smooth density tests and continuous contact tests supported
in \(w,w_c\le M\) and bounded in absolute value by \(R\).  B2 proves the
finite-horizon pressure on every such bounded real source ball.  Define its
lower-semicontinuous extension to a weighted source by monotone truncation:
\[
\overline Q(p,\psi)=
\sup_{M}\lim_{R\to\infty}
Q\bigl(p^{M,R},\psi^{M,R}\bigr),
\]
whenever the right-hand side is proper.  The local analytic pressure and the
global extension agree on \(\mathcal U_\eta\cap\mathcal D_{\rm glob}\).

\begin{lemma}[Strict Radon dual]
\label{lem:r6-b3-radon}
The continuous dual of
\((C_{w_c,0},\beta_{w_c})\) is the space of countably additive signed Radon
measures \(\mathcal M_{w_c}\), under integration.  No finitely additive
functional occurs.
\end{lemma}

\begin{proof}
Restrict a continuous functional to every compact sublevel
\(\{w_c\le M\}\).  There the strict topology agrees with uniform convergence
on bounded sets, so Riesz representation gives a compatible Radon measure.
Continuity on weighted bounded sets makes the weighted total variations
uniformly summable; the measures paste to one member of
\(\mathcal M_{w_c}\).  Conversely, weighted tightness and dominated
convergence make integration by any such Radon measure strictly continuous.
\end{proof}

\subsection{The exact gauge complex}

Define
\[
\mathfrak G r=(r,-\Delta r),
\qquad
\mathfrak D(p,\psi)=\Delta p+\psi.
\]

\begin{theorem}[Closed balance-adjoint gauge]
\label{thm:r6-b3-gauge}
The sequence
\[
0\longrightarrow\mathcal Y_p
\xrightarrow{\mathfrak G}
\mathcal Y_p\times C_{w_c,0}
\xrightarrow{\mathfrak D}
C_{w_c,0}\longrightarrow0
\]
is exact and has closed first range.  Hence
\[
(\mathcal Y_p\times C_{w_c,0})/
\operatorname{Ran}\mathfrak G
\cong C_{w_c,0}
\]
as Hausdorff locally convex spaces.  On every balanced pair the dual action
depends only on \(z=\Delta p+\psi\).  Collision invariants are included as
normalized gauge directions but do not exhaust the gauge.
\end{theorem}

\begin{proof}
Surjectivity follows from \(z=\mathfrak D(0,z)\).  If
\(\mathfrak D(p,\psi)=0\), then \((p,\psi)=\mathfrak Gp\), proving exactness.
The range of \(\mathfrak G\) is the graph of the continuous map
\(r\mapsto-\Delta r\), hence is closed.  On a balanced pair, the endpoint and
transport terms equal \(\int\Delta p\,d\Gamma\), so the action contains
\(p,\psi\) only through their sum.  Fixing the finite-dimensional collision
invariants selects a representative without changing the quotient class.
\end{proof}

\subsection{Global integral Fenchel duality}

For a density path \(f\), let
\[
dA_f=\frac12ff_*((v-v_*)\cdot\omega)_+
\,dt\,dx\,dv\,dv_*\,d\omega,
\]
and put
\[
\mathfrak J_f(\Gamma)=
\begin{cases}
\displaystyle\int\ell(d\Gamma/dA_f)dA_f,&\Gamma\ll A_f,\\
+\infty,&\text{otherwise},
\end{cases}
\qquad \ell(q)=q\log q-q+1.
\]
For a bounded global test define
\[
\begin{aligned}
\mathfrak L_{f,\Gamma}(p,\psi)={}&
\langle p_T,f_T\rangle-\langle p_0,f_0\rangle
-\int\langle(\partial_t+v\cdot\nabla_x)p,f\rangle dt\\
&+\int\psi\,d\Gamma
-\int(e^{\Delta p+\psi}-1)dA_f.
\end{aligned}
\]

\begin{theorem}[Two-domain path-space duality]
\label{thm:r6-b3-duality}
For every finite-moment pair \((f,\Gamma)\),
\[
\sup_{(p,\psi)\in\mathcal D_{\rm glob}}
\mathfrak L_{f,\Gamma}(p,\psi)
=
\begin{cases}
\mathfrak J_f(\Gamma),&(f,\Gamma)\text{ balanced},\\
+\infty,&\text{otherwise}.
\end{cases}
\]
For a regular pair with
\(0<c\le d\Gamma/dA_f\le C<\infty\), the optimizer belongs to a local
analytic chart and its unique quotient representative is
\[
z=\log(d\Gamma/dA_f).
\]
The global theorem does not assert that arbitrarily large truncations remain
inside one analytic chart.
\end{theorem}

\begin{proof}
If balance fails, a compactly supported smooth density test \(p\) has nonzero
balance defect.  Set \(\psi=-\Delta p\); both are bounded on their compact
velocity support and belong to \(\mathcal D_{\rm glob}\).  Scaling the pair by
any finite constant leaves the exponential argument zero and sends the linear
defect to infinity.

On the balanced set, the action reduces to
\[
\int z\,d\Gamma-\int(e^z-1)dA_f.
\]
For the absolutely continuous part, take bounded truncations of
\(\log(d\Gamma/dA_f)\), followed by compact velocity cutoffs.  Monotone
convergence gives \(\int\ell(d\Gamma/dA_f)dA_f\).  If \(\Gamma\) has a
singular part, choose a compact set \(K\) charged by it and open
neighborhoods \(U_n\) with \(A_f(U_n)\le e^{-2n}\).  A continuous Urysohn
test equal to \(n\) on \(K\) and supported in \(U_n\) gives a linear term of
order \(n\Gamma(K)\) and an exponential cost at most \(e^{-n}\).  Hence the
supremum is infinite.  Every test used is bounded for fixed \(n\), although
the sequence ranges over increasing source balls.  Strict convexity of
\(e^z-1\) gives the regular optimizer.  Its bounded logarithm lies in a local
chart after choosing the chart around that regular phase.
\end{proof}

\subsection{The covariance cohomology}

Let \(Q_\Theta\) be the local analytic prepared pressure at a regular biased
source \(\Theta\), and let
\(\Sigma_\Theta=D^2Q_\Theta\).  Gauge directions are identically zero along
every microscopic balanced path and therefore lie in its kernel.

\begin{theorem}[Complete local covariance kernel]
\label{thm:r6-b3-kernel}
On a regular analytic source chart, the kernel of the speed-normalized
pressure covariance consists exactly of the closed balance gauge, the fixed
initial constraints removed by B1, and endpoint/time coboundaries represented
in \(\mathcal Y_p\).  Equivalently, the Hessian is strictly positive on every
finite-dimensional family of independent quotient classes.
\end{theorem}

\begin{proof}
The B2 lower-bound tilt identifies the local Legendre transform with the
relative-entropy action.  Its second variation at a regular pair is
\[
\delta^2I(\dot f,\dot\Gamma)=
\delta^2I_0(\dot f_0)+
\int\frac{\dot q^2}{q}\,dA_f+
\text{the quadratic variation of }A_f\text{ in }\dot f,
\]
restricted to the linearized balance.  B2's conservative regularization gives
a dense set of smooth balanced tangent directions.  If a source quotient
class is nonzero, Hahn--Banach supplies one such tangent with nonzero pairing;
the positive quadratic action makes the local support function strictly
curved in that direction.  Legendre Hessian inversion therefore gives
positive pressure variance.

Conversely, a zero pressure variance pairs to zero with every finite-action
balanced tangent.  Density of the regular tangents and the closed graph gauge
of Theorem~\ref{thm:r6-b3-gauge} imply that the source lies in the balance
adjoint range, plus an endpoint coboundary.  B1 removes exactly the finite
initial constraint span by its Schur complement.  This proves the stated
kernel without asserting that a zero-variance observable is pointwise
constant.
\end{proof}

\subsection{Exact finite-volume centring and localized cumulants}

Under the finite-volume biased law, define
\[
\bar\pi_\varepsilon(t)=
\mathbb E_{\Theta,\varepsilon}\pi_t^\varepsilon,
\qquad
\bar\Gamma_\varepsilon=
\mathbb E_{\Theta,\varepsilon}\Gamma_T^\varepsilon,
\]
and the exactly centred fields
\[
\zeta^\varepsilon=\sqrt{\mu_\varepsilon}
(\pi^\varepsilon-\bar\pi_\varepsilon),
\qquad
\Xi^\varepsilon=\sqrt{\mu_\varepsilon}
(\Gamma^\varepsilon-\bar\Gamma_\varepsilon).
\]
For an interval \(I\subset[0,T]\), write
\(Y_I^\varepsilon(\phi,\psi)\) for the corresponding density increment plus
actual-contact test, centred at its exact finite expectation.

\begin{lemma}[Time-localized cumulant bounds]
\label{lem:r6-b3-cumulants}
For every bounded finite family of density/contact tests and every
\(k\ge2\),
\[
\left|\kappa_k\bigl(Y_I^\varepsilon\bigr)\right|
\le C_k|I|\mu_\varepsilon^{1-k/2}
+C_k\mu_\varepsilon^{-k/2}.
\]
Conditionally on a stopping time \(\tau\) and the complete trace hierarchy at
\(\tau\), the same estimate holds for
\(I=[\tau,\tau+h]\) with \(|I|=h\), uniformly on exponentially compact trace
balls.  In particular,
\[
\mathbb E|Y_{[s,t]}^\varepsilon|^4
\le C|t-s|^2+C|t-s|/\mu_\varepsilon+C\mu_\varepsilon^{-2}.
\]
\end{lemma}

\begin{proof}
Insert a source supported in \(I\) into the B2 trace-class connected pressure.
A connected contribution meeting that source has one distinguished time in
\(I\), giving the factor \(|I|\); a possible endpoint atom gives the smaller
\(\mu_\varepsilon^{-k/2}\) term.  Normal convergence on a complex
neighborhood bounds every derivative.  Scaling by
\(\sqrt{\mu_\varepsilon}\) gives the stated cumulants.  After a stopping time,
the exact Markov property holds on the complete finite-volume law/trace state,
not on density alone.  B2's conditional trace estimate places every reached
state in the same compact trace ball, so the same local source calculation
applies.  The fourth-moment identity
\(\mathbb EY^4=3\kappa_2(Y)^2+\kappa_4(Y)\) gives the final estimate.
\end{proof}

\subsection{The full joint Gaussian process}

Let \((f,\Gamma=qA_f)\) be the limiting regular biased pair.  The linearized
balance propagator is denoted by \(U(t,s)\).  For density/contact tests
\((\phi,\psi)\) and \((\chi,\eta)\), define the instantaneous collision
covariance
\[
\mathcal C_t((\phi,\psi),(\chi,\eta))
=
\int q_t
(\Delta\phi+\psi)(\Delta\chi+\eta)\,dA_{f_t}.
\]
This one formula includes all density--density, density--contact, and
contact--contact cross terms.

\begin{theorem}[Exact-centred density--actual-contact Gaussian limit]
\label{thm:r6-b3-gaussian}
The fields \((\zeta^\varepsilon,\Xi^\varepsilon)\), centred at their exact
finite-volume expectations, converge in the weighted nuclear distribution
path space to the unique centred Gaussian solution of the linearized biased
balance with initial covariance equal to the B1 constrained Schur complement
and dynamic covariance
\[
\int_0^T
\mathcal C_t\bigl(U(T,t)^*(\phi,\psi),
U(T,t)^*(\chi,\eta)\bigr)dt.
\]
Every finite-dimensional covariance is the Hessian of the limiting prepared
pressure.  No predictable Poisson compensator for the deterministic contact
measure is assumed.
\end{theorem}

\begin{proof}
For finitely many time-localized tests, analytic convergence of the B2
pressure gives convergence of second cumulants.  Lemma~\ref{lem:r6-b3-cumulants}
makes all cumulants of order at least three vanish, hence gives the Gaussian
finite-dimensional limit.  Differentiating the limiting marked
Hamilton--Jacobi equation twice yields the linearized propagator and the
bilinear contact covariance displayed above; because the same actual contact
carries both \(\Delta\phi\) and \(\psi\), the cross terms are included
automatically.

For tightness, apply the conditional version of
Lemma~\ref{lem:r6-b3-cumulants} to stopping-time increments.  The
\(h^2+h/\mu_\varepsilon\) estimate and the fact that individual empirical
jumps are \(O(\mu_\varepsilon^{-1/2})\) give Aldous tightness for every test
coordinate.  A countably Hilbert nuclear scale with Maxwellian weights and
Mitoma's theorem upgrade coordinate tightness to the distribution space.
Every subsequential limit has the Gaussian finite-dimensional laws and the
linearized balance, which determine it uniquely.  B1 supplies the exact
finite-volume conditioned initial covariance and its limiting Schur
complement.
\end{proof}

If one wishes to centre at the limiting means \((f,\Gamma)\), the deterministic
bias
\(\sqrt{\mu_\varepsilon}((\bar\pi_\varepsilon,
\bar\Gamma_\varepsilon)-(f,\Gamma))\) must be added explicitly; it is not
silently discarded without a rate theorem.

\begin{theorem}[Round-six B3 closure]
\label{thm:r6-b3-main}
The density--contact action has a globally valid weighted-strict Radon dual and
a separate local analytic pressure chart.  Its cotangent is the Hausdorff
balance-adjoint quotient, the local pressure covariance is nondegenerate
exactly modulo gauge, endpoint coboundaries, and prepared constraints, and the
exactly centred deterministic hard-sphere density/contact fields converge to
the joint Gaussian process of Theorem~\ref{thm:r6-b3-gaussian}.
\end{theorem}

\begin{proof}
Combine Theorems~\ref{thm:r6-b3-gauge},
\ref{thm:r6-b3-duality}, \ref{thm:r6-b3-kernel}, and
\ref{thm:r6-b3-gaussian}, together with
Lemma~\ref{lem:r6-b3-radon}.
\end{proof}
