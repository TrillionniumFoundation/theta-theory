\section{Round-twelve positive closure: perturbative kinetic control, exact covariance kernel, and process Gaussianity}
\label{sec:r12-b3}

The regular source chart is made quantitative.  On it the biased collision
operator is a small graph-bounded perturbation of the Maxwellian linearized
operator.  Covariance is constructed first from microscopic cumulants; its
dual kernel and Cameron--Martin inverse are then identified from the local
noise representation, without a hard-sphere Liv\v{s}ic assertion.

\subsection{A nonautonomous biased energy theorem}

Let $M_{\rho,u,T}$ be the local Maxwellian determined by the prepared conserved
fields.  The regular chart consists of paths and collision ratios satisfying
\[
 cM\le f\le CM,
 \quad
 \|f-M\|_{W^{2,\infty}_{x,v}(M^{-1})}
 +\|q-1\|_{W^{1,\infty}_w}\le\delta,
 \quad 0<q_-\le q\le q_+.
\]
The B2 small source/time map places every regular exposed phase in this chart.

\begin{lemma}[Perturbative biased collision gap]
\label{lem:r12-b3-gap}
For $\delta$ small, the symmetric part of the linearized biased collision
operator satisfies
\[
 -\langle h,L^{\rm sym}_{f,q}h\rangle_{L^2(M^{-1})}
 \ge c\|(I-\Pi_5)h\|_\nu^2-C\delta\|\Pi_5h\|^2,
\]
where $\Pi_5$ is the five-dimensional velocity-invariant projection.  The
constants are uniform on the chart.
\end{lemma}

\begin{proof}
At $(M,1)$ the Dirichlet form is the classical Maxwellian form with its
five-dimensional kernel.  Expand $f=M+(f-M)$ and $q=1+(q-1)$ in the binary
collision integral.  The difference is bounded, in the collision graph norm,
by $C\delta\|h\|_\nu^2$.  Absorb the microscopic part into the equilibrium
gap; the remaining finite-rank invariant contribution gives the displayed
term.
\end{proof}

Let $u$ solve the linearized biased balance with prepared initial data.  Write
$u=\Pi_5u+(I-\Pi_5)u$; the coefficients of $\Pi_5u$ are fields on
$[0,T]\times\mathbb T^3$, not five scalars.

\begin{theorem}[Variable-coefficient hypocoercive control]
\label{thm:r12-b3-energy}
For $T\le T_*$ and the regular chart,
\[
 \sup_{t\le T}\|u_t\|_{L^2(M^{-1})}^2
 +\int_0^T\|(I-\Pi_5)u_t\|_\nu^2dt
 +\|\Pi_5u\|_{L^2_tH^{-1}_x}^2
 \le C\bigl(\|u_0\|^2+\|F\|_{X^*}^2\bigr).
\]
The transport/contact trace map is continuous in this energy space.  The same
estimate holds for the backward adjoint equation with endpoint data.
\end{theorem}

\begin{proof}
Multiply the microscopic equation by $u/M$ and use
Lemma~\ref{lem:r12-b3-gap}.  Commutators with the variable Maxwellian and $q$
are $C\delta$ graph-bounded errors.  Taking the five collision moments gives
the linear acoustic system for the spatial coefficient fields, forced by the
microscopic flux.  Testing that system in $H^{-1}_x$ and using the microscopic
dissipation controls the macroscopic part.  Endpoint and contact terms cancel
through the B2 Green trace identity.  Gronwall on a short interval absorbs the
remaining $C(\delta+T)$ terms.  Time reversal gives the adjoint estimate.
\end{proof}

\subsection{Covariance first and the exact dual kernel}

For a density/contact source $F$, let
\[
 \Sigma(F,G)=\lim_{\varepsilon\to0}
 \mu_\varepsilon\operatorname{Cov}(F(\pi^\varepsilon,\Gamma^\varepsilon),
                                    G(\pi^\varepsilon,\Gamma^\varepsilon)).
\]
B1 supplies the prepared initial block and B2 supplies the marked contact
cumulants.

\begin{theorem}[Joint Gaussian noise representation]
\label{thm:r12-b3-noise}
On the regular chart, $\Sigma$ is the covariance of the unique linearized
biased Boltzmann martingale problem
\[
 d\langle u_t,\phi\rangle
 =\langle u_t,\mathcal A_{f,q}^*\phi\rangle dt+dM_t(\phi),
\]
with prepared initial covariance and bracket
\[
 d\langle M(\phi),M(\psi)\rangle_t
 =\int q\,\Delta\phi\,\Delta\psi\,dA_fdt.
\]
For a dual source pair, variance is zero exactly when its backward adjoint
solution has zero projected initial source and zero local collision-noise
coefficient.  This kernel is the closed balance/endpoint gauge; no long-time
Liv\v{s}ic theorem is invoked.
\end{theorem}

\begin{proof}
Differentiate the B2 marked Hamiltonian twice and use the B1 Schur complement
at time zero.  The result is the displayed linear Gaussian martingale
problem.  Solve its backward adjoint equation by
Theorem~\ref{thm:r12-b3-energy}.  It represents every source as the sum of an
initial Gaussian pairing and a stochastic integral against the collision
noise.  Orthogonality of those two terms makes the variance zero precisely
when both coefficients vanish.  Reversing the weak balance identifies this
condition with the endpoint-normalized representation gauge.
\end{proof}

\begin{proposition}[Typed cotangent and Cameron--Martin spaces]
\label{prop:r12-b3-typed}
The continuous dual source space is quotiented by $\ker\Sigma$, a closed
linear subspace.  The primal Cameron--Martin form is
\[
 \mathfrak q(z)=
 \begin{cases}
  \|\Sigma^{-1/2}z\|^2,&z\in\operatorname{Ran}\Sigma^{1/2},\\
  +\infty,&\text{otherwise}.
 \end{cases}
\]
It has no nonzero null vector.  The dual quotient and the primal form domain
are not identified with one another.
\end{proposition}

\begin{proof}
The covariance kernel is closed by continuity.  The spectral theorem for the
positive covariance operator gives the displayed closed extended quadratic
form and its exact domain.  Positivity is strict on that domain after the dual
kernel has been quotiented.
\end{proof}

\subsection{Second-order action and Mosco convergence}

\begin{theorem}[Second epi-derivative]
\label{thm:r12-b3-mosco}
Let $I$ be the B2 prepared path action.  At a regular exposed path, its second
epi-derivative on the balanced Gelfand triple equals $\mathfrak q$.  Finite
conditional-expectation projections of the Gaussian noise give an increasing
sequence of closed forms converging to $\mathfrak q$ in the Mosco sense.
\end{theorem}

\begin{proof}
On a finite family of tests, analytic pressure duality gives the inverse
covariance Hessian.  Realize all projections on the same initial-plus-collision
Gaussian noise space; conditional expectation onto increasing noise
sigma-fields makes the energies monotone.  The martingale convergence theorem
gives the liminf and recovery of the closed forms.  The energy estimate of
Theorem~\ref{thm:r12-b3-energy} identifies the closure of smooth balanced
variations with this Gaussian control space.  Hence the second epi-derivative
is exactly $\mathfrak q$.
\end{proof}

\subsection{Process tightness with the microscopic diagonal term}

Let $Z_t^\varepsilon$ be the centered $\sqrt{\mu_\varepsilon}$ density/contact
field.

\begin{lemma}[Aldous increment estimate]
\label{lem:r12-b3-tight}
For every bounded stopping time $\tau$ and $0\le h\le1$,
\[
 \mathbb E|Z_{\tau+h}^\varepsilon-Z_\tau^\varepsilon|^4
 \le C\left(h^2+{h\over\mu_\varepsilon}\right).
\]
The maximal jump in the declared distribution norm converges to zero in
probability.
\end{lemma}

\begin{proof}
The fourth cumulant of a contact increment contains one connected diagonal
term of size $h/\mu_\varepsilon$ and pairings of size $h^2$.  The localized B2
cluster density bounds all remaining connected diagrams by the same right
side.  Each microscopic density/contact jump has size
$O(\mu_\varepsilon^{-1/2})$ in the negative test norm; the exponential contact
count bound controls their maximum.
\end{proof}

\begin{theorem}[Prepared joint process CLT]
\label{thm:r12-b3-clt}
The fields $Z^\varepsilon$ converge in the weighted distribution Skorokhod
space to the Gaussian martingale problem of
Theorem~\ref{thm:r12-b3-noise}.  All density/contact cross-brackets and the
prepared initial covariance are included.
\end{theorem}

\begin{proof}
Finite-dimensional cumulants converge by normal pressure convergence and all
orders at least three vanish after central scaling.  Lemma~\ref{lem:r12-b3-tight}
gives Aldous tightness; Mitoma's criterion lifts it to the distribution space.
Closedness of the B2 balance graph identifies every limit, and uniqueness of
the linear Gaussian martingale problem gives full convergence.
\end{proof}

\begin{theorem}[Round-twelve B3 closure]
\label{thm:r12-b3-main}
On the explicit perturbative regular chart, the prepared hard-sphere tangent
has a proved nonautonomous energy estimate, exact covariance kernel, correctly
typed Cameron--Martin form, second-order action, and joint process Gaussian
limit.
\end{theorem}

\begin{proof}
Combine Theorems~\ref{thm:r12-b3-energy},
\ref{thm:r12-b3-noise}, \ref{thm:r12-b3-mosco}, and
\ref{thm:r12-b3-clt}, with Proposition~\ref{prop:r12-b3-typed}.
\end{proof}
