\section{Round-seven reconstruction: exact tangent action and closed balance range}
\label{sec:r7-b3}

This section replaces the round-six controlling module.  The nonlinear
reference collision measure $A_f$ is differentiated exactly.  Covariance is
constructed by Lax--Milgram on the linearized balanced tangent space rather
than by an unqualified infinite-dimensional Legendre inversion.

\subsection{Local analytic and global convex source domains}

Let $\mathscr S_{\rm an}$ be the union of bounded weighted source balls on
which the B2 pressure has a zero-free complex neighborhood.  It is used only
for analytic response.  Let
\[
 \mathscr S_{\rm cvx}=
 \bigcup_{M,R<\infty}
 \{(p,\psi):\operatorname{supp}(p,\psi)\subset K_R,
   \ \|p\|_\infty+\|\psi\|_\infty\le M\}.
\]
It is used for Fenchel duality.  For a general signed source there is no
asserted monotone truncation limit; the global convex pressure is the directed
supremum over the finite test family appearing in the dual pairing.

\begin{theorem}[Two-domain Fenchel theorem]
\label{thm:r7-b3-duality}
For every balanced pair $(f,\Gamma)$,
\[
 I(f,\Gamma)=
 \sup_{(p,\psi)\in\mathscr S_{\rm cvx}}
 \left\{\langle p,f\rangle+\langle\psi,\Gamma\rangle
       -Q(p,\psi)\right\}.
\]
If the pair is unbalanced or if $\Gamma$ has a component singular to $A_f$,
the supremum is $+\infty$.  On $\mathscr S_{\rm an}$, $Q$ is analytic and its
finite derivatives agree with the corresponding finite-volume cumulants.
\end{theorem}

\begin{proof}
Compactly supported bounded sources separate Radon measures and can be scaled
without leaving $\mathscr S_{\rm cvx}$.  Balance defects are therefore assigned
infinite cost.  On the absolutely continuous contact part, pointwise conjugacy
of $e^z-1$ and $\ell(q)=q\log q-q+1$, followed by monotone spatial and velocity
cutoffs, gives the entropy.  A singular contact part is detected by bounded
sources supported where $A_f$ is arbitrarily small.  Analyticity on the local
domain follows from the B2 pressure theorem and Cauchy differentiation.
\end{proof}

\subsection{The exact second variation}

For positive measures $g,a$ write the perspective entropy density
\[
 \Phi(g,a)=g\log(g/a)-g+a.
\]
At $g=qa$, its Hessian in affine directions $(h,b)$ is
\[
 D^2\Phi_{(qa,a)}[(h,b)]^2
 =\frac{(h-qb)^2}{qa}.
\]
If $a_s=A_{f_s}$ is nonlinear, then
\[
 \frac{d^2}{ds^2}\Phi(g_s,A_{f_s})\bigg|_{s=0}
 =\frac{(\dot\Gamma-qDA_f[\dot f])^2}{qA_f}
 +(1-q)D^2A_f[\dot f,\dot f].
\]
This includes the term omitted in the preceding revision.

Let $I_0(f_0)$ be the prepared initial entropy.  For a regular exposed phase
$(f,\Gamma=qA_f)$ define
\[
 \mathfrak q_{f,q}(\dot f,\dot\Gamma)=
 D^2I_0(f_0)[\dot f_0]^2
 +\int_0^T\int
  \frac{|\dot\Gamma-qDA_f[\dot f]|^2}{q}\,dA_fdt
 +\int_0^T\int(1-q)D^2A_f[\dot f,\dot f]dt.
\]
The tangent is required to satisfy the exact linearized balance equation.

\begin{lemma}[Coercivity of the on-shell tangent action]
\label{lem:r7-b3-coercive}
On every regular B2 analytic phase chart there are $c,C>0$ such that
\[
 \mathfrak q_{f,q}(\dot f,\dot\Gamma)
 \ge c\left(
 \|\dot f_0\|_{L^2(f_0^{-1})}^2
 +\int\frac{|\dot\Gamma-qDA_f[\dot f]|^2}{q}\,dA_fdt
 +\|\dot f\|_{\mathsf X_T}^2\right).
\]
Here $\mathsf X_T$ is the weighted transport energy norm.  The constants are
uniform after a fixed finite horizon is divided into analytic continuation
blocks.
\end{lemma}

\begin{proof}
For hard spheres,
$D^2A_f[\dot f,\dot f]$ is twice the collision measure with density
$\dot f\dot f_*$.  Cauchy--Schwarz in $(v,v_*,\omega)$ and the Maxwellian
bounds give
\[
 \left|\int(1-q)D^2A_f[\dot f,\dot f]\right|
 \le \eta\|\dot f\|_{\mathsf X_T}^2
 +C_\eta\int
  \frac{|\dot\Gamma-qDA_f[\dot f]|^2}{q}\,dA_f.
\]
The linearized balance energy estimate controls $\mathsf X_T$ by the initial
term and the square contact residual.  On a sufficiently short analytic
continuation block the coefficient $\eta$ is absorbed.  Concatenating finitely
many blocks propagates a positive constant over the fixed horizon.
\end{proof}

\subsection{The balance operator and its closed range}

Let $\mathsf H_{\rm tan}$ be the completion of smooth tangent pairs for the
norm in Lemma~\ref{lem:r7-b3-coercive}.  Let $\mathsf Y$ be the weighted
negative Sobolev space of balance defects including endpoint traces.  Define
\[
 \mathcal B(\dot f,\dot\Gamma)
 =\partial_t\dot f+v\cdot\nabla_x\dot f
  -\mathcal C_f\dot\Gamma,
\]
with the prepared endpoint map included.  Its adjoint is the backward
transport--collision operator on the source pair $(p,\psi)$.

\begin{theorem}[Closed balance range and the gauge annihilator]
\label{thm:r7-b3-range}
The operator $\mathcal B:\mathsf H_{\rm tan}\to\mathsf Y$ has closed range.
Its kernel is the balanced tangent Hilbert space
$\mathsf H_{\rm bal}$, and
\[
 (\mathsf H_{\rm bal})^\perp
 =\overline{\operatorname{Ran}\mathcal B^*}.
\]
Under the source identification,
$\operatorname{Ran}\mathcal B^*$ is exactly the balance gauge
$(r,-\Delta r)$ together with endpoint coboundaries and the prepared
constraint covectors.  There are no additional zero-variance source classes.
\end{theorem}

\begin{proof}
Solve the backward adjoint equation with terminal value zero.  Multiplying by
the solution and using the collision Dirichlet form gives the observability
estimate
\[
 \|(p,\psi)\|_{\mathsf Y^*/\ker\mathcal B^*}
 \le C\|\mathcal B^*(p,\psi)\|_{\mathsf H_{\rm tan}^*}.
\]
The transport trace terms are controlled by the measure-flux Green identity of
the B2 trace theorem.  This estimate is equivalent to closed range.  The
Hilbert closed-range theorem gives the annihilator identity.  Expanding the
adjoint balance pairing yields $(r,-\Delta r)$ in the bulk/contact variables;
the only boundary terms are the declared endpoint and prepared constraints.
\end{proof}

\subsection{Covariance by Lax--Milgram}

For a source class $\xi$ modulo the gauge, define $u_\xi\in\mathsf H_{\rm bal}$
by
\[
 \mathfrak q_{f,q}(u_\xi,v)=\langle\xi,v\rangle
 \quad\hbox{for every }v\in\mathsf H_{\rm bal}.
\]

\begin{theorem}[Tangent covariance isomorphism]
\label{thm:r7-b3-covariance}
The equation above has a unique solution and
\[
 \operatorname{Cov}(\xi,\eta)
 =\mathfrak q_{f,q}(u_\xi,u_\eta)
\]
is a positive continuous bilinear form on the quotient source Hilbert space.
Its kernel is exactly the closed gauge of
Theorem~\ref{thm:r7-b3-range}.  Every finite-dimensional projected pressure
Hessian equals the corresponding covariance matrix.
\end{theorem}

\begin{proof}
Lemma~\ref{lem:r7-b3-coercive} and Lax--Milgram give existence, uniqueness,
and a bounded inverse on $\mathsf H_{\rm bal}$.  A source has zero covariance
if and only if it annihilates every balanced tangent, which by
Theorem~\ref{thm:r7-b3-range} is precisely the gauge class.  Differentiating
the finite-dimensional B2 variational formula twice and using the unique
linearized minimizer gives the same Lax--Milgram equation, hence the projected
Hessian identity.  No global Frechet inversion of an arbitrary lower
semicontinuous Legendre transform is invoked.
\end{proof}

\subsection{Localized deterministic-interval cumulants}

For a bounded test $\xi$ supported in an interval $I$, let
$Y_I^\varepsilon(\xi)$ be the exactly finite-volume centered density/contact
increment.  Differentiating the B2 connected expansion with time-localized
sources gives joint cumulant densities $c_k^\varepsilon(t_1,\ldots,t_k)$.
The measure-trace renewal theorem bounds these densities in $L^1$ uniformly on
bounded tilted charts.

\begin{lemma}[Interval and two-interval cumulant bounds]
\label{lem:r7-b3-cumulants}
For $k\ge2$,
\[
 |\kappa_k(Y_I^\varepsilon)|
 \le C_k\left(|I|\mu_\varepsilon^{1-k/2}
             +\mu_\varepsilon^{-k/2}\right).
\]
For adjacent disjoint intervals $I,J$,
\[
 \mathbb E|Y_I^\varepsilon|^2|Y_J^\varepsilon|^2
 \le C|I||J|+C\mu_\varepsilon^{-1}(|I|+|J|)
 +C\mu_\varepsilon^{-2}.
\]
These are integrated deterministic-interval estimates and do not condition on
a point collision.
\end{lemma}

\begin{proof}
A connected cumulant meeting a source supported in $I$ has one distinguished
marked time integrated over $I$.  The remaining time variables are integrated
against the $L^1$ connected density supplied by the B2 measure-trace renewal
bound.  Coincident endpoint atoms have total mass
$O(\mu_\varepsilon^{-k/2})$.  For two intervals, expand the fourth joint moment
into connected cumulants and pairings.  Every connected block meeting both
intervals carries the product $|I||J|$; blocks meeting one endpoint give the
smaller terms.
\end{proof}

\subsection{The Gaussian process}

Choose a countably Hilbert nuclear test space
\[
 \mathsf S=\bigcap_{m\ge0}H^{m}_{x,v}(e^{\beta_m|v|^2})
 \times\bigcap_{m\ge0}H^{m}_{\rm contact}
 (e^{\beta_m(|v|^2+|v_*|^2)}),
\]
with Hilbert--Schmidt embeddings obtained by increasing both derivative order
and Maxwellian weight.  Let $\mathsf S'$ be its dual.  The linearized biased
balance generates a continuous propagator $U(t,s)$ on this scale.

\begin{theorem}[Exact-centred density--actual-contact Gaussian limit]
\label{thm:r7-b3-gaussian}
The fields centered at their exact finite-volume tilted expectations converge
in $D([0,T],\mathsf S')$ to the unique centered Gaussian solution of the
linearized balance equation.  Its initial covariance is the B1 constrained
Schur complement and its dynamic covariance is
\[
 \int_0^T\int q_t
  (\Delta\phi_t+\psi_t)(\Delta\chi_t+\eta_t)
  \,dA_{f_t}dt,
\]
with tests propagated backward by $U(T,t)^*$.  The covariance kernel is the one
in Theorem~\ref{thm:r7-b3-covariance}.
\end{theorem}

\begin{proof}
For finitely many localized tests, second cumulants converge by the B2 pressure
and higher cumulants vanish by
Lemma~\ref{lem:r7-b3-cumulants}.  This gives Gaussian finite-dimensional
limits.  The two-interval estimate yields a uniform fractional Sobolev bound
$W^{\alpha,4}$ for every $\alpha<1/4$ after the microscopic jump correction;
compact embedding gives scalar path tightness without stopping times.  Mitoma's
theorem applies to the explicitly nuclear space $\mathsf S$.  Differentiating
the limiting balance at the regular phase identifies the propagator and the
contact covariance.  Coercivity and the linearized energy estimate give
uniqueness of the Gaussian martingale problem.
\end{proof}

\begin{theorem}[Round-seven B3 closure]
\label{thm:r7-b3-main}
The global entropy dual and local analytic response use distinct source
domains; the nonlinear contact action has its exact second variation; the
linearized balance has closed range; covariance is the Lax--Milgram inverse on
the balanced quotient; and process tightness follows from integrated interval
cumulants rather than an unavailable stopping trace density.
\end{theorem}

\begin{proof}
Combine Theorems~\ref{thm:r7-b3-duality},
\ref{thm:r7-b3-range},
\ref{thm:r7-b3-covariance}, and
\ref{thm:r7-b3-gaussian}, together with
Lemmas~\ref{lem:r7-b3-coercive} and
\ref{lem:r7-b3-cumulants}.
\end{proof}
