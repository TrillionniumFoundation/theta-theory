satisfies a uniform vector-valued almost-sure invariance principle.  In
particular, after enlarging the probability space, its partial sums differ
from a two-dimensional Brownian motion with the same Green--Kubo covariance by
$o(n^{1/2-\eta})$ almost surely for some $\eta>0$ uniform on $\mathcal K$.
The polygonal lift satisfies the corresponding rough invariance principle.
\end{theorem}
\begin{proof}
The driven operator has the same exponential Young-tower and twisted spectral
packet as the base Lorentz map, uniformly on compact $q$-sets.  The observables
are bounded and piecewise H\"older.  Apply the vector ASIP and spectral
methods of \cite{MelbourneNicol2005,MelbourneNicol2009,Gouezel2010}; rough
convergence follows from the enhanced invariance principle
\cite{FrizHairer2014}.
\end{proof}

\section{Actual physical time and cell position}\label{sec:physical}
Let
\[
 S_n\tau_R=\sum_{k=0}^{n-1}\tau_R\circ T_R^k,
 \qquad
 N_R(t)=\max\{n:S_n\tau_R\le t\},
\]
and let
\[
 C_R(n)=\sum_{k=0}^{n-1}J_R\circ T_R^k
\]
be the first lattice coordinate of the lifted cell position after $n$
collisions.

\begin{theorem}[Ballistic--diffusive Lorentz limit]\label{thm:physical-limit}
Under $\mu_{R,q}$,
\[
 \frac{C_R(N_R(t))}{t}\longrightarrow v_{R,q}
 \qquad\text{almost surely}.
\]
Moreover, for every $T<\infty$,
\[
 \left
 \{
 \frac{C_R(N_R(nt))-v_{R,q}nt}{\sqrt n}
 \right\}_{0\le t\le T}
 \Longrightarrow
 \{\sqrt{D_{R,q}}\,W_t\}_{0\le t\le T}.
\]
The canonical rough lift converges jointly.
\end{theorem}
\begin{proof}
The ergodic theorem gives $N_R(t)/t\to1/\bar\tau$ and the ballistic law.
Write
\[
 C_R(N_R(t))-vt
 =S_{N_R(t)}G_{R,q}
 +v\bigl(S_{N_R(t)}\tau_R-t\bigr).
\]
The second term is uniformly bounded by $v\sup\tau_R$.  Apply the joint ASIP
and monotone time change to the first term.  Its collision variance is
$\Sigma^{\rm phys}$ and division by the mean collision time gives
$D_{R,q}$.
\end{proof}

\section{A deterministic comoving slow system}\label{sec:slow}
Let $b:\R\to\R$ be bounded and globally Lipschitz.  Define at collisions
\[
 X_{k+1}^\varepsilon
 =X_k^\varepsilon
 +\varepsilon G_{R,q}(T_R^kz)
 +\varepsilon^2\tau_R(T_R^kz)b(X_k^\varepsilon),
\]
\[
 t_{k+1}^\varepsilon=t_k^\varepsilon
 +\varepsilon^2\tau_R(T_R^kz).
\]
For each complete initial billiard state $z$, the evolution is deterministic.

\begin{corollary}[Physical homogenization]\label{cor:sde}
The physical-time interpolation converges weakly to
\[
 dX_t=b(X_t)\,dt+\sqrt{D_{R,q}}\,dW_t.
\]
\end{corollary}
\begin{proof}
Use \cref{thm:physical-limit} and the deterministic fast--slow homogenization
argument of \cite{KellyMelbourne2017}.  The drift sum is weighted by the actual
roof, so it converges to physical-time drift $b$ without an additional factor.
\end{proof}

\section{Spectral Bernstein and random physical horizons}\label{sec:window}
Let
\[
 n_\varepsilon(T)=\left\lfloor
 \frac{T}{\varepsilon^2\bar\tau_{R,q}}\right\rfloor,
 \qquad
 N_\varepsilon(T)=N_R(T/\varepsilon^2).
\]

\begin{theorem}[Exponential random-window control]\label{thm:window}
For every compact preparation set $\mathcal K$, $T<\infty$, and fixed
$\lambda>0$, there is $C$ such that uniformly on $\mathcal K$,
\[
 \log\E_{\mu_{R,q}}
 \exp\left\{
 \lambda\varepsilon
 \left|\sum_{k=n_\varepsilon(T)}^{N_\varepsilon(T)-1}
 G_{R,q}\circ T_R^k\right|
 \right\}
 \le C\sqrt\varepsilon,
\]
and
\[
 \log\E_{\mu_{R,q}}
 \exp\left\{
 \lambda\varepsilon^2
 \abs{N_\varepsilon(T)-n_\varepsilon(T)}
 \right\}
 \le C\varepsilon.
\]
\end{theorem}
\begin{proof}
The complex twisted spectral packet gives, for small $u$ and all $m$,
\[
 \E e^{uS_mG}\le Ce^{Cu^2m},
 \qquad
 \E e^{uS_m(\tau-\bar\tau)}\le Ce^{Cu^2m}.
\]
Chernoff bounds and monotone inversion give a sub-Gaussian tail for the inverse
clock.  Split the random index difference into dyadic windows.  On each
window, apply the first spectral moment estimate and sum the sub-Gaussian
probabilities.  Typical window length is $O(\varepsilon^{-1})$, so the
$\varepsilon$-scaled current has exponential Orlicz size
$O(\sqrt\varepsilon)$.
\end{proof}

\section{Parameter continuity and the moving family}\label{sec:parameter}
