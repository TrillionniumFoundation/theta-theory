
\begin{abstract}
We derive the physical-time stochastic limit of the actual lattice displacement
in the current-prepared periodic Sinai billiards of Paper I.  The microscopic
orbit is deterministic and the only probability is the canonical equilibrium
preparation selected by the Liouville current large deviation.  A nonzero
prepared current creates ballistic transport.  The diffusive variable is
therefore the comoving displacement, and its collision observable is not
$J-j$ but the clock-corrected mechanical increment
\[
 G=J-v\tau,\qquad v=\frac{j}{\bar\tau}.
\]
This correction closes the current--roof covariance interface that is lost if
one divides the collision variance by the mean roof mechanically.

Using the twisted Lorentz spectral packet, we prove a joint almost-sure
invariance principle for cell current and actual free-flight time, identify the
physical drift and variance through a pressure-root formula, and obtain a
ballistic law of large numbers plus Brownian fluctuations in the comoving
frame.  Spectral Bernstein estimates yield exponential control of the random
collision window at a fixed physical horizon.  These estimates are precisely
strong enough for the nonlinear free-energy DPP in Paper III.
\end{abstract}

\section{Prepared equilibrium and mechanical characteristics}\label{sec:data}
Fix a radius $R\in\mathcal I$ and a canonical field $q$ in the spectral
interval of Paper I.  Let $\mu_{R,q}$ be the driven invariant preparation and
put
\[
 j_{R,q}=\int J_R\,d\mu_{R,q}=\Lambda_R'(q),
 \qquad
 \bar\tau_{R,q}=\int\tau_R\,d\mu_{R,q}.
\]
The physical ballistic velocity in the first lattice coordinate is
\[
 v_{R,q}=\frac{j_{R,q}}{\bar\tau_{R,q}}.
\]
Define the collision-centred and physical-time-centred observables
\[
 g_{R,q}=J_R-j_{R,q},
 \qquad
 G_{R,q}=J_R-v_{R,q}\tau_R.
\]
Both have zero mean under $\mu_{R,q}$ only in their respective clocks:
\[
 \int g_{R,q}\,d\mu_{R,q}=0,
 \qquad
 \int G_{R,q}\,d\mu_{R,q}=0.
\]

\section{Joint pressure and the physical cumulant root}\label{sec:pressure}
Let $\Psi_{R,q}(p,s)$ be the logarithm of the simple leading eigenvalue of the
Doob-prepared transfer operator additionally twisted by
$e^{pJ_R-s\tau_R}$.  It is analytic near $(0,0)$ and
\[
 \partial_p\Psi_{R,q}(0,0)=j_{R,q},
 \qquad
 \partial_s\Psi_{R,q}(0,0)=-\bar\tau_{R,q}.
\]
Let $\kappa_{R,q}(p)$ be the unique local root
\[
 \Psi_{R,q}\bigl(p,\kappa_{R,q}(p)\bigr)=0,
 \qquad \kappa_{R,q}(0)=0.
\]

\begin{theorem}[Physical drift and diffusion from one pressure root]\label{thm:root}
One has
\[
 \kappa_{R,q}'(0)=v_{R,q}
\]
and
\[
 \kappa_{R,q}''(0)=D_{R,q},
\]
where
\[
 D_{R,q}
 =\frac{\Sigma_{R,q}^{\rm phys}}{\bar\tau_{R,q}},
\]
\[
 \Sigma_{R,q}^{\rm phys}
 =\lim_{n\to\infty}\frac1n
 \Var_{\mu_{R,q}}
 \left(\sum_{k=0}^{n-1}G_{R,q}\circ T_R^k\right)>0.
\]
Equivalently,
\[
 \Sigma_{R,q}^{\rm phys}
 =\sum_{n\in\mathbb Z}
 \Cov_{\mu_{R,q}}
 \bigl(G_{R,q},G_{R,q}\circ T_R^n\bigr).
\]
\end{theorem}
\begin{proof}
The implicit-function theorem gives
$\kappa'=-\Psi_p/\Psi_s=j/\bar\tau$.  Differentiating a second time gives
\[
 -\Psi_s\kappa''
 =\Psi_{pp}+2v\Psi_{ps}+v^2\Psi_{ss}.
\]
The right side is the Green--Kubo asymptotic variance of
$J-v\tau=G$.  Since $-\Psi_s=\bar\tau$, the formula follows.  Positivity
would fail only if $G$ were a coboundary.  The marked travelling and normal
periodic orbits from Paper I have different ratios of lattice displacement to
physical period, excluding that possibility.
\end{proof}

\begin{remark}[Why $\Lambda''/\bar\tau$ is not the general answer]
If $q=0$, time reversal gives $j=0$, so $G=J$ and
$D_{R,0}=\Lambda_R''(0)/\bar\tau_{R,0}$.  At nonzero prepared current, roof
fluctuations and current--roof correlations contribute.  The pressure-root
formula is the invariant physical-time expression.
\end{remark}
