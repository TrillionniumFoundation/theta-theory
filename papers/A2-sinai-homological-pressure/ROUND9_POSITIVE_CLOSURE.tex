\section{Round-nine reconstruction: blow-up transport, explicit arithmetic, and a density-level vector--roof theorem}
\label{sec:r9-a2}

We work on the compact finite-horizon radius interval $K$ fixed in the paper.
Moving tangencies are resolved by a parabolic blow-up and a material
connection; compact-frequency arithmetic and high-frequency UNI are proved by
different orbit packets.  The local limit result is a theorem for the joint
density, so arbitrarily small sharp roof windows are obtained by integration
rather than by smoothing an indicator.

\subsection{Parabolic blow-up at a simple birth}

Near a simple tangency choose collision coordinates $(u,v)$ and a parameter
$r=R-R_0$ in which the two newborn branches satisfy
\[
 u=u_0+a_1r+a_2v^2+O(r^2+|r||v|+|v|^3),
 \qquad a_2\ne0.
\]
Blow up $(r,v)=(0,0)$ by $v=|r|^{1/2}\zeta$.  For a test distribution define
the paired traces
\[
 T^{\rm ev}_r=T_{r,+}+T_{r,-},
 \qquad
 T^{\rm od}_r=|r|^{-1/2}(T_{r,+}-T_{r,-}).
\]
Both have finite limits on the front face.  Let $\mathbb B^j$ be the usual
strong/weak anisotropic bulk space enlarged by even/odd trace jets through
order $j$.  Define the material connection on the blown-up chart by
\[
 \nabla_r=\partial_r-\frac{\zeta}{2r}\partial_\zeta+\mathcal C_r,
\]
where $\mathcal C_r$ is the explicit Jacobian correction which makes
$\partial_r(E_rU)=E_r(\nabla_rU)$.

\begin{lemma}[Uniform birth graph and connection]
\label{lem:r9-a2-birth}
The graph of the physical realization
\[
 E_r:\mathbb B^{j+1}\longrightarrow\mathscr D'
\]
is closed uniformly through $r=0$.  The paired even/odd trace map has a bounded
inverse onto its physical graph norm, and
\[
 \partial_r^k(E_rU)=E_r\nabla_r^kU,
 \qquad 0\le k\le j,
\]
with constants independent of the side of the birth.
\end{lemma}

\begin{proof}
Taylor expansion in the blown-up variables gives
\[
 T_{r,+}+T_{r,-}=2T_0+rT_2+O(r^2),
 \qquad
 |r|^{-1/2}(T_{r,+}-T_{r,-})=2T_1+O(r).
\]
Thus the even/odd basis has a nondegenerate limit, unlike the unscaled birth
coordinate.  The anisotropic trace theorem bounds $T_0,T_1,T_2$ by one bulk
strong order and places their remainders in the weak order.  Differentiating
$v=|r|^{1/2}\zeta$ gives the singular Euler field; the displayed connection
cancels it exactly.  The remaining Jacobian coefficients are smooth on the
blown-up compact chart.  A finite cover and the closed graph theorem give the
uniform statement.
\end{proof}

\begin{theorem}[Uniform moving-billiard spectral bundle]
\label{thm:r9-a2-bundle}
The untwisted and vector--roof twisted transfer operators act on one finite
strong/weak scale of the blow-up bundle and satisfy, uniformly in $R\in K$,
\[
 \|\mathcal L_{R,\xi,b}^nf\|_{\mathbb B^j}
 \le C\vartheta^n\|f\|_{\mathbb B^j}
   +C(1+|\xi|+|b|)^j\|f\|_{\mathbb B^{j-1}}.
\]
Covariant parameter derivatives through the declared finite order obey the
same estimate with the expected loss of one strong level per derivative.
At zero frequency the peripheral eigenvalue is simple.
\end{theorem}

\begin{proof}
On every homogeneous branch, finite horizon and uniform curvature give the
standard cone, distortion, and one-step expansion constants.  Differentiating
a branch creates the paired trace jets of Lemma~\ref{lem:r9-a2-birth}; their
codimension is triangular and their all-depth coefficients are summable by
the homogeneity-strip expansion.  The material connection commutes with
physical realization, so no inverse power of $r$ appears.  Compactness follows
from stable-curve truncation, homogeneity truncation, and one-level loss.  The
mixing collision map has no nonconstant peripheral eigenfunction; the trace
extension has none because its diagonal blocks are restrictions of the same
physical operator and every off-diagonal block lowers the jet order.
\end{proof}

\subsection{A concrete compact-frequency certificate}

Choose lattice basis vectors $e_1,e_2$.  On each radius chart select the
following regular periodic orbits, continued by the implicit-function theorem:
$w_1,w_2$ are the normal-incidence winding orbits with homology $e_1,e_2$;
$w_3$ is a three-scatterer triangular loop of zero homology; $w_4$ is a
four-scatterer rhombic loop of zero homology; and $w_0$ is the two-scatterer
backtracking loop.  Put
\[
 c_R(w)=\bigl(\kappa_R^{(1)}(w),\kappa_R^{(2)}(w),
              \tau_R(w),n(w)\bigr).
\]

\begin{lemma}[Explicit finite arithmetic packets]
\label{lem:r9-a2-arithmetic}
There is a finite cover $K=\bigcup_{\alpha=1}^M K_\alpha$ and, on each chart,
a five-orbit packet of the preceding type for which
\[
 \left|\det\bigl(c_R(w_j)-c_R(w_0)\bigr)_{j=1}^4\right|
 \ge d_\alpha>0,
 \qquad R\in K_\alpha.
\]
Consequently no nonzero character of
$(\kappa^{(1)},\kappa^{(2)},\tau,n)$ is a coboundary.
\end{lemma}

\begin{proof}
The first two rows have homology matrix equal, after orientation, to the
identity.  The remaining $2\times2$ minor is
\[
 (\tau_3-\tau_0)(n_4-n_0)-(\tau_4-\tau_0)(n_3-n_0).
\]
For the normal-incidence geometric representatives, the free lengths are
explicit analytic functions of $R$.  The triangular and rhombic loops have
different mean free lengths, so the minor is nonzero except at isolated radii.
At an exceptional radius replace the rhombic loop by the next regular polygonal
loop; its mean-length analytic function is not identical to either previous
one.  Compactness gives finitely many replacements and positive determinant
lower bounds.  A coboundary character vanishes on every periodic orbit, which
contradicts the corresponding determinant.
\end{proof}

\subsection{A returned-branch UNI theorem}

Fix a Markov rectangle $Y$ and a small unstable interval $J\subset Y$.  Choose
two inverse branches which share the same last $m$ collisions, but before that
common suffix pass around opposite sides of a triangular gap.  Let $h_R,k_R$
be the resulting inverse branches of one common return iterate and let
\[
 \Delta_R(x)=\tau_N(h_Rx)-\tau_N(k_Rx).
\]

\begin{theorem}[Uniform returned-branch temporal nonintegrability]
\label{thm:r9-a2-uni}
The branches and $m$ can be chosen on finitely many radius charts so that
\[
 \inf_{R\in K_\alpha}\inf_{x\in J_\alpha}
 |\partial_x\Delta_R(x)|\ge c_{\rm UNI}>0.
\]
The suffix is not declared identical in physical space; its derivative
contribution is bounded by $C\Lambda^{-m}$ and is absorbed by choosing $m$
large.
\end{theorem}

\begin{proof}
For a flight from boundary points $r(x)$ and $r'(x)$, the first variation of
its length is
\[
 \partial_x\tau=\bigl\langle\widehat{r'-r},
 \partial_xr'-\partial_xr\bigr\rangle.
\]
At the symmetric reference orbit, the two gap itineraries have outgoing angles
$\pm\theta_R$ with $\inf_K|\sin\theta_R|>0$.  Their first unequal legs
therefore contribute $2\sin\theta_R$ to the temporal-distance derivative.
Inverse-branch contraction makes the derivative of the preceding and common
suffix tails at most $C\Lambda^{-m}$.  Choose $m$ so this is less than half the
uniform main contribution.  Distortion and the implicit-function theorem then
extend the lower bound to one common interval on each compact chart.
\end{proof}

\begin{theorem}[All-iterate Dolgopyat bound]
\label{thm:r9-a2-dolgopyat}
There are $A,C,c,\gamma>0$ such that, for $|b|\ge2$ and every $n\ge1$,
\[
 \|\mathcal L_{R,\xi,b}^n\|_{\mathbb B^j\to\mathbb B^{j-1}}
 \le C|b|^A
 \exp\left(-\frac{cn}{\log(2+|b|)}\right)
 +C\vartheta^n,
\]
uniformly in $R\in K$ and homology frequency in its fundamental domain.
\end{theorem}

\begin{proof}
Theorem~\ref{thm:r9-a2-uni} gives a phase derivative at least
$c_{\rm UNI}|b|$ on a returned interval.  Split it into pieces of length
$|b|^{-1}$ and pair the two inverse branches on pieces where their phases are
within a fixed angle of opposition.  Bounded distortion gives comparable
amplitudes.  The growth lemma returns a fixed fraction of every standard
family to $J$ in at most $A\log(2+|b|)$ iterates.  Hence each such block loses a
fixed fraction.  Iterating the blocks gives the exponential factor.  The
strong-to-weak loss and the blow-up trace jets are controlled by
Theorem~\ref{thm:r9-a2-bundle}; all off-diagonal jet terms are time-ordered and
summable.
\end{proof}

\subsection{A density-level vector--roof local limit theorem}

Let $K_n\in\mathbb Z^2$ be the homology sum and $T_n\in\mathbb R$ the roof
sum under a real tilted phase.  Let $p_{n,R}(k,t)$ denote the density in $t$ of
$\mathbb P_R(K_n=k,T_n\in dt)$.  The covariance matrix $\Sigma_R$ is
nonsingular by Lemma~\ref{lem:r9-a2-arithmetic} and
Theorem~\ref{thm:r9-a2-uni}.

\begin{theorem}[Uniform joint density LLT]
\label{thm:r9-a2-density-llt}
For some $\eta>0$, uniformly in $R$, in central targets
$|k-n\bar\kappa_R|+|t-n\bar\tau_R|\le C\sqrt n$, and in the declared real
source chart,
\[
 \left|
 n^{3/2}p_{n,R}(k,t)
 -\varphi_{\Sigma_R}\left(
 \frac{k-n\bar\kappa_R}{\sqrt n},
 \frac{t-n\bar\tau_R}{\sqrt n}
 \right)
 \right|
 \le Cn^{-\eta}.
\]
The same estimate holds after insertion of a fixed central cylinder, with the
Gaussian coefficient multiplied by its tilted conditional expectation.
\end{theorem}

\begin{proof}
Fourier inversion is taken over the two lattice frequencies and the full roof
frequency.  On a ball of radius $n^{-2/5}$, analytic perturbation of the simple
eigenvalue gives the Gaussian expansion with a uniform cubic remainder.
Compact nonzero frequencies are uniformly contracted by
Lemma~\ref{lem:r9-a2-arithmetic}.  Intermediate and large roof frequencies are
integrable by Theorem~\ref{thm:r9-a2-dolgopyat}; one further roof derivative is
available on the strong/weak scale and supplies an integrable Fourier tail.
The resulting inversion is a density estimate, not an estimate for a smoothed
indicator.  Central insertions are placed between the left and right spectral
projectors.
\end{proof}

\begin{corollary}[Sharp windows in all three regimes]
\label{cor:r9-a2-windows}
For every bounded Borel set $B$ with negligible boundary and every $G_n>0$,
provided the central window remains in the source chart,
\[
 \begin{aligned}
 &\mathbb P_R\left(
 K_n=k_n,\ \frac{T_n}{n}\in t_n+G_nB
 \right)\\
 &\quad=\frac1n
 \int_{\sqrt n(t_n-\bar\tau_R)+\sqrt nG_nB}
 \varphi_{\Sigma_R}\left(
 \frac{k_n-n\bar\kappa_R}{\sqrt n},u
 \right)du
 \left(1+O(n^{-\eta})\right).
 \end{aligned}
\]
Thus the local, central, and saturated orders are respectively
$G_nn^{-1/2}$, $n^{-1}$ times a finite Gaussian interval mass, and $n^{-1}$
times the saturated conditional mass.  The relative error remains valid even
when $G_n\ll n^{-1/2}$.
\end{corollary}

\begin{proof}
Integrate the density estimate of
Theorem~\ref{thm:r9-a2-density-llt} over the roof-sum interval
$nt_n+nG_nB$ and use $u=(t-n\bar\tau_R)/\sqrt n$.  The pointwise relative
scale is preserved under integration; no boundary smoothing or absolute
$o(n^{-1})$ remainder is used.
\end{proof}
