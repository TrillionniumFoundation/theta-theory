\section{Referee-round positive closure: uniform vector--roof spectral and local-limit theory}
\label{sec:rr2-a2}

This section supplies the parameter-uniform billiard analysis previously
compressed into the homological packet.  The geometric parameter interval is
the compact radius interval used in the body of the paper.

\subsection{A finite atlas of anisotropic spaces}

Write \(M_R=\partial K_R\times[-\pi/2,\pi/2]\) in angular collision
coordinates \((\alpha,\varphi)\).  These coordinates identify every \(M_R\)
with the same compact cylinder.  Let \(\mathcal S_R\) be the singularity set
of the collision map and let \(\mathcal H_k\) be the usual homogeneity strips.

\begin{lemma}[Uniform geometric constants]
\label{lem:rr2-a2-geometry}
There are constants
\[
 \tau_->0,\quad \tau_+<\infty,\quad
 \Lambda>1,\quad C_{\rm dist}<\infty,\quad \vartheta<1
\]
independent of \(R\) such that: free flights lie in
\([\tau_-,\tau_+]\); stable and unstable cones are invariant with expansion
at least \(\Lambda\); distortion on homogeneous branches is bounded by
\(C_{\rm dist}\); and the one-step expansion sum is at most \(\vartheta\).
\end{lemma}

\begin{proof}
The compact radius interval stays a positive distance from both overlap and
corridor thresholds.  Hence the free-flight lower and upper bounds are
uniform.  Curvatures equal \(R^{-1}\) and are uniformly separated from zero
and infinity.  The standard mirror equation then gives uniform cone
invariance and expansion.  The derivative and second derivative formulas in
\((\alpha,\varphi)\) have denominators consisting only of the flight length,
curvature, and \(\cos\varphi\).  On each homogeneity strip these quantities
have the standard strip bounds with constants uniform in \(R\).  Summing the
inverse unstable expansions over children gives a continuous function of
\(R\); compactness and strict one-step expansion at each parameter give a
common \(\vartheta<1\).
\end{proof}

Choose finitely many parameter neighborhoods \(U_\alpha\) so that the
singularity curves in each \(U_\alpha\) admit common graph charts.  Let
\(\mathcal B_\alpha\) and \(\mathcal B_{\alpha,w}\) be the strong and weak
anisotropic completions defined by stable-curve testing in those charts.
Coordinate changes induce bounded maps \(T_{\alpha\beta}\) and
\(R_{\alpha\beta}\) on overlaps.

\begin{theorem}[Uniform spectral atlas]
\label{thm:rr2-a2-atlas}
The atlas can be chosen so that
\[
 R_{\alpha\beta}T_{\alpha\beta}=I,\qquad
 T_{\alpha\beta}\mathcal L_{R,\xi,s}
 =
 \mathcal L_{R,\xi,s}T_{\alpha\beta}
\]
on overlaps.  For \(|\xi|+|s|<\delta\), with complex \((\xi,s)\), the twisted
operator
\[
 \mathcal L_{R,\xi,s}h
 =\mathcal L_R\!\left(
 e^{\langle\xi,\kappa_R\rangle-s\tau_R}h\right)
\]
satisfies the uniform Lasota--Yorke inequalities
\[
 \|\mathcal L_{R,\xi,s}^nh\|_{\mathcal B_\alpha}
 \le C\rho^n\|h\|_{\mathcal B_\alpha}
     +C\|h\|_{\mathcal B_{\alpha,w}},
 \qquad
 \|\mathcal L_{R,\xi,s}^nh\|_{\mathcal B_{\alpha,w}}
 \le C\|h\|_{\mathcal B_{\alpha,w}}.
\]
It has a simple leading eigenvalue
\(e^{P_R(\xi,s)}\), analytic jointly in \((\xi,s)\) and \(C^1\) in \(R\),
and the complementary spectrum is contained in a disk of radius
\(e^{\Re P_R(\xi,s)}\rho\), with one common \(\rho<1\).
\end{theorem}

\begin{proof}
Lemma~\ref{lem:rr2-a2-geometry} supplies the uniform growth and distortion
constants used in the strong/weak norm estimates.  The displacement is
constant on each continuity component and finite horizon bounds its range.
The roof and its first two branch derivatives are uniformly bounded on
homogeneity strips.  Multiplication by
\(e^{\langle\xi,\kappa_R\rangle-s\tau_R}\) is therefore a bounded analytic
multiplier on the graded atlas, with a norm bounded uniformly for
\(|\xi|+|s|<\delta\).  The compact embedding from strong to weak space and
the uniform Lasota--Yorke estimate give uniform quasicompactness.
At \((0,0)\), mixing gives simplicity of the eigenvalue one.  A finite atlas
and compactness of the radius interval produce a common perturbation radius
and spectral gap.  The overlap identities follow because the charts
represent the same distributions on physical stable curves; thus local
spectral projectors patch to an atlas-independent projector.
\end{proof}

\subsection{Joint arithmetic exclusion}

For \(u\in\mathbb R^2\) and \(t\in\mathbb R\), a peripheral mode would imply
\[
 e^{i\langle u,S_n\kappa_R\rangle-itS_n\tau_R}=1
\]
on every periodic orbit.  The nearest-neighbor displacement branches generate
\(\mathbb Z^2\).  Three explicitly continued periodic collision words have
zero net displacement and periods \(T_1(R),T_2(R),T_3(R)\) whose two
differences have irrational ratio, uniformly in \(R\).

\begin{lemma}[Uniform aperiodicity]
\label{lem:rr2-a2-aperiodic}
If the twisted operator has peripheral spectral radius one at \((iu,it)\),
then \(u\in2\pi\mathbb Z^2\) and \(t=0\).  Consequently there exist
\(\epsilon,c>0\) such that, outside an \(\epsilon\)-neighborhood of the
origin modulo \(2\pi\mathbb Z^2\),
\[
 \|\mathcal L_{R,iu,it}^n\|_{\mathcal B_\alpha\to\mathcal B_{\alpha,w}}
 \le Ce^{-cn}.
\]
\end{lemma}

\begin{proof}
Liv\v{s}ic's periodic-orbit criterion applied to the unit-modulus
eigenfunction gives the periodic identities.  The displacement-generating
orbits force \(u\in2\pi\mathbb Z^2\).  The three zero-displacement periods
force \(t(T_2-T_1)\) and \(t(T_3-T_1)\) to be integer multiples of \(2\pi\).
The explicitly computed ratio of the two differences is irrational, hence
\(t=0\).  Upper semicontinuity of the spectral radius and compactness of the
parameter/frequency annulus give the uniform exponential bound.
\end{proof}

\subsection{Joint local limit and ratio conditioning}

Let \(K_n=S_n\kappa_R\), \(T_n=S_n\tau_R\), and let
\(\nu_{R,\xi}\) be the normalized tilted equilibrium law on the pressure-root
surface \(s=\mathcal H_R(\xi)\).

\begin{theorem}[Uniform lattice--nonlattice local limit]
\label{thm:rr2-a2-jllt}
Uniformly for \(R\) and \(\xi\) in compact subsets of the simple branch, for
bounded central cylinder \(F\), \(k-na_\xi=O(\sqrt n)\), and intervals
\(J_n=[t_n-b_n,t_n+b_n]\) with \(b_n\to\infty\), \(b_n=o(n)\),
\[
 \mathbb E_{\nu_{R,0}}\!\left[
 F;\ K_n=k,\ T_n\in J_n\right]
 =
 e^{-nI_R(k/n,t_n/n)}
 \frac{2b_n}{(2\pi n)^{3/2}\sqrt{\det\Sigma_{R,\xi}}}
 \left(\nu_{R,\xi}(F)+\varepsilon_n(F)\right),
\]
where \(\sup_{\|F\|_{\rm cyl}\le1}|\varepsilon_n(F)|\to0\).
The formula remains valid when \(b_n/\sqrt n\to0\); in that regime the
displayed \(2b_n\) factor is the temporal local-limit scale.
\end{theorem}

\begin{proof}
Use Fourier inversion in the two lattice coordinates and a smooth
upper/lower approximation of the interval indicator in the temporal
frequency.  On the major arc, Theorem~\ref{thm:rr2-a2-atlas} gives the
third-order eigenvalue expansion
\[
 P_R(\xi+iu,s+it)
 =P_R(\xi,s)+i\langle(u,t),m\rangle
 -\tfrac12\langle(u,t),\Sigma_{R,\xi}(u,t)\rangle
 +O(|(u,t)|^3)
\]
uniformly, including the left/right eigenvectors with the cylinder
insertion.  Gaussian integration gives the main term.  On the intermediate
arc, strict positivity of the covariance gives \(e^{-cn|(u,t)|^2}\).
On the minor arc, Lemma~\ref{lem:rr2-a2-aperiodic} gives exponential decay.
The smoothing error is optimized with cutoff \(b_n^{-1}\); the bounds are
uniform even when \(b_n=o(\sqrt n)\).
\end{proof}

\begin{corollary}[Two-sided current--clock Gibbs conditioning]
\label{cor:rr2-a2-conditioning}
For a fixed central window of length \(2m+1\), exact displacement \(k_n\), and
the clock window above,
\[
 \left|
 \mathbb E_{\nu_{R,0}}\!\left[
 F\mid K_{2n}=k_n,\ T_{2n}\in J_n\right]
 -\nu_{R,\xi}(F)
 \right|
 \le C_F\left((n-m)^{-1/2}+b_n/n+b_n^{-1}+\delta_n\right),
\]
where \(\delta_n\) is the mismatch between the constrained means and the
tilted means.
\end{corollary}

\begin{proof}
Apply Theorem~\ref{thm:rr2-a2-jllt} to the numerator with a central insertion
and to the denominator with \(F=1\).  The central placement leaves
\(n-m\) iterates on both sides, so the left and right eigenvectors pair into
the stationary tilted density.  Divide the two expansions and use the
uniform denominator lower bound supplied by the Gaussian main term.
\end{proof}
