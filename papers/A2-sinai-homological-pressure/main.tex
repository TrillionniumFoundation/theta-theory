\begin{abstract}
We upgrade scalar cell-current conditioning to a coordinate-free homological
path theory for a compact periodic finite-horizon Sinai billiard.  The full
lattice displacement cocycle takes values in the first homology group and is
paired with a covector field.  A joint displacement--roof Lorentz transfer
operator yields an analytic physical pressure root.  Its gradient is ballistic
velocity, its Hessian is the clock-corrected transport tensor, and its Legendre
dual selects the endogenous homological field of a prepared velocity.

We also construct a projective empirical-path large-deviation principle on a
convergence-determining cylinder algebra.  A two-sided vector-current/clock
Gibbs-conditioning theorem removes endpoint eigenfunction bias and identifies
the canonical same-billiard preparation.  The scalar v8 theorem becomes every
one-dimensional contraction of this coordinate-free result.
\end{abstract}

\section{The compact periodic Sinai platform}\label{sec:platform}
Let
\[
 e_1=(1,0),\qquad e_2=(1/2,\sqrt3/2),\qquad
 \Lambda=\Z e_1\oplus\Z e_2,
\]
and put one circular scatterer of radius
\[
 R\in\mathcal I=[9/20,47/100]
\]
in every lattice cell.  The quotient table is compact, the lifted table is
periodic, and the particle moves at unit speed with equal-angle specular
reflection.  The finite-horizon and disjointness estimates of the v8 platform
remain in force.  In common angular boundary coordinates
\(M=\T\times(-\pi/2,\pi/2)\), the normalized collision law is
\[
 d\mu=(4\pi)^{-1}\cos\varphi\,d\alpha\,d\varphi.
\]
This is the sole natural preparation used in the present series.

For a collision state \(z\), lift the present collision to the disk at the
origin.  If the next collision is on the disk at
\(m_1e_1+m_2e_2\), define the full homology displacement
\[
 \kappa_R(z)=(m_1,m_2)\in H_1(\T_\Lambda^2;\Z)\cong\Z^2.
\]
The actual free-flight time is \(\tau_R(z)>0\).  Finite horizon makes both
\(\kappa_R\) and \(\tau_R\) bounded and piecewise dynamically H\"older.

\begin{proposition}[Coordinate-free mechanical input]\label{prop:mechanical}
The pair
\[
 (\kappa_R,\tau_R)\in H_1(\T_\Lambda^2;\Z)\times\R_+
\]
is determined by the lifted specular orbit.  Under velocity reversal,
\[
 \kappa_R\circ\mathscr R\circ T_R=-\kappa_R,
 \qquad
 \tau_R\circ\mathscr R\circ T_R=\tau_R.
\]
In particular \(\int\kappa_R\,d\mu=0\).
\end{proposition}
\begin{proof}
Reversal traverses the same free segment in the opposite lattice direction
and with the same length.  Liouville invariance then gives zero mean
homology current.
\end{proof}

\begin{remark}
The scalar current used in v8 is the contraction
\(J_\ell=\pair{\ell}{\kappa_R}\) for one covector
\(\ell\in H^1(\T_\Lambda^2;\R)\).  It is no longer a primitive input.
\end{remark}

\section{The joint homology--clock spectral packet}\label{sec:spectral}
For \(\xi\in H^1(\T_\Lambda^2;\R)\cong\R^2\) and \(s\in\R\), define
\[
 \cL_{R,\xi,s}f
 =\cL_R\left(e^{\pair{\xi}{\kappa_R}-s\tau_R}f\right),
\]
where \(\cL_R\) is the Liouville transfer operator.

\begin{theorem}[Uniform homological Lorentz packet]\label{thm:spectral}
There are anisotropic spaces \(\mathcal B\hookrightarrow\mathcal B_w\), a
neighbourhood \(\mathcal U\subset\R^2\times\R\) of the origin, and constants
\(C>0\), \(0<\rho_*<1\), uniform in \(R\in\mathcal I\), such that
\[
 \cL_{R,\xi,s}^n
 =e^{nP_R(\xi,s)}\Pi_{R,\xi,s}+N_{R,\xi,s}^n,
 \qquad
 \norm{N_{R,\xi,s}^n}_{\mathcal B\to\mathcal B_w}
 \le C\rho_*^n.
\]
The leading eigenvalue is simple; \(P_R\), the projector, and the normalized
Doob data are real analytic in \((\xi,s)\in\mathcal U\).  The Hessian of
\(P_R\) is the asymptotic covariance form of the joint observable
\((\kappa_R,-\tau_R)\).
\end{theorem}
\begin{proof}
The untwisted finite-horizon Lorentz operator has a uniform spectral gap on
the perturbation spaces for the compact radius family.  Multiplication by the
bounded piecewise H\"older weight
\(e^{\pair{\xi}{\kappa_R}-s\tau_R}\) is analytic and bounded on the graded
strong/weak pair.  Analytic perturbation of an isolated simple eigenvalue gives
the decomposition.  Differentiating the eigenvalue equation and summing the
reduced-resolvent series gives the Green--Kubo derivatives.  The functional
analytic perturbation input is the Lorentz-gas framework of
\cite{DemersZhang2016}; the vector lattice local-limit input is compatible
with \cite{SzaszVarju2004}.
\end{proof}

\begin{remark}[Review boundary]
The theorem is the principal imported analytic packet.  External reviewers
must verify that the common singularity complexity, multiplication bounds, and
lattice aperiodicity are uniform on \(\mathcal I\).
\end{remark}

\section{Projective empirical-path large deviations}\label{sec:level3}
Let \(\Omega_R\subset M^{\Z}\) be the natural two-sided path space of the
collision map and let \(\Theta\) be the left shift.  Denote by \(\Pp_R\) the
stationary path law induced by \(\mu\).  Choose a countable algebra
\(\mathscr A_R\) of bounded dynamically H\"older cylinder functions that
separates probability measures on \(\Omega_R\) and is stable under time
translation.

For \(F\in\mathscr A_R\), set
\[
 Q_R(F)=\lim_{n\to\infty}\frac1n
 \log\E_{\Pp_R}\exp\left\{\sum_{k=0}^{n-1}F\circ\Theta^k\right\}.
\]
For a finite tuple \(\bm F=(F_1,\ldots,F_m)\), the same notation is used for
the joint log-moment function.

\begin{theorem}[Canonical projective empirical-process LDP]\label{thm:level3}
Let
\[
 L_n=\frac1n\sum_{k=0}^{n-1}\delta_{\Theta^k\omega}.
\]
For every finite tuple \(\bm F=(F_1,\ldots,F_m)\subset\mathscr A_R\), let
\(U_{\bm F}\) be a multiplier neighbourhood on which the corresponding
higher-block twisted operator has a simple leading eigenvalue, and put
\(V_{\bm F}=\nabla Q_{R,\bm F}(U_{\bm F})\).  The vector
\((L_nF_1,\ldots,L_nF_m)\) satisfies the full LDP upper and lower bounds for
Borel sets whose closures lie compactly in \(V_{\bm F}\), with the local
Legendre rate.  These consistent finite projections define a canonical
projective LDP on the cylinder topology, with exposed-point rate
\[
 \boxed{
 \cI_R^{\rm proj}(\nu)=
 \sup_{\substack{F\in\mathscr A_R\\ F\text{ in the declared pressure domain}}}
 \left\{\int F\,d\nu-Q_R(F)\right\}.}
\]
For every finite constraint whose multiplier remains in the common simple
spectral component, the information projection is unique and is the stationary
equilibrium path law generated by the normalized twisted Lorentz operator.
We write \(\cI_R=\cI_R^{\rm proj}\) on this canonical domain.
\end{theorem}
\begin{proof}
For each finite tuple, the higher-block transfer representation and the
perturbation packet give an essentially smooth log-moment function on a
smaller multiplier neighbourhood.  The finite-dimensional
G\"artner--Ellis theorem gives the stated LDP on compact subsets of its mean
image.  Consistency under coordinate deletion gives the projective cylinder
bounds over the countable algebra; compactness of the product of the bounded
coordinate ranges supplies exponential tightness in that topology
\cite{DemboZeitouni1998,DawsonGartner1987}.  Strict convexity modulo
coboundaries gives uniqueness of each exposed local information projection.
No global empirical-process LDP is inferred from a merely local spectral
branch.
\end{proof}

\begin{remark}[Variational continuation and phase coexistence]
The lower-semicontinuous Legendre extension of \(\cI_R^{\rm proj}\) provides a
well-defined convex variational continuation.  Outside the proved canonical
projective domain, its subgradients classify candidate phase coexistence, but
actual microscopic realization requires a separate global empirical-process
LDP and is not claimed here.
\end{remark}

\section{The physical pressure root and endogenous homological field}
Because \(\partial_sP_R(0,0)=-\int\tau_Rd\mu<0\), there is a unique analytic
root
\[
 \boxed{P_R(\xi,\cH_R(\xi))=0}
\]
for \(\xi\) in a neighbourhood \(\mathcal V_R\subset\R^2\).  Let
\(\nu_{R,\xi}\) be the stationary equilibrium law for the potential
\(\pair{\xi}{\kappa_R}-\cH_R(\xi)\tau_R\).

\begin{theorem}[Pressure-root geometry]\label{thm:root}
For \(\xi\in\mathcal V_R\),
\[
 v_R(\xi)=\nabla\cH_R(\xi)
 =\frac{\int\kappa_R\,d\nu_{R,\xi}}
        {\int\tau_R\,d\nu_{R,\xi}},
\]
and
\[
 \boxed{
 D_R(\xi)=\nabla^2\cH_R(\xi)
 =\frac{\AVar_{\nu_{R,\xi}}
 (\kappa_R-v_R(\xi)\tau_R)}
 {\int\tau_R\,d\nu_{R,\xi}}.}
\]
The matrix is positive definite on a common local field window.  Hence
\(\xi\mapsto v_R(\xi)\) is a local analytic diffeomorphism and
\[
 \boxed{\xi_R(v)=\nabla I_R^{\rm phys}(v)}
\]
is the unique homological conjugate field of the prepared physical velocity.
\end{theorem}
\begin{proof}
Differentiate the root equation once and twice.  The first derivative divides
the mean homology increment by the mean roof.  The second derivative is the
Schur complement of the joint current--clock covariance, which is exactly the
asymptotic covariance of \(\kappa-v\tau\) divided by the mean roof.  Positive
definiteness follows from the nondegenerate two-dimensional Lorentz local-limit
packet and persists under local analytic tilting.
\end{proof}

The local spectral branch has a canonical convex continuation.  Let
\(\cI_R^{\rm proj}\) denote the lower-semicontinuous projective Legendre
functional generated by the pressure domain in \cref{thm:level3}, and define
\[
 \boxed{
 \cH_R^{\rm var}(\xi)
 =\sup_{\nu\in\mathcal M_\Theta,\,\nu(\tau_R)>0}
 \frac{\pair{\xi}{\nu(\kappa_R)}-\cI_R^{\rm proj}(\nu)}
 {\nu(\tau_R)}.}
\]

\begin{proposition}[Convex variational continuation]\label{prop:global-root}
The function \(\cH_R^{\rm var}\) is convex and lower semicontinuous.  On the
common simple spectral component it agrees with the analytic physical root
\(\cH_R\).  Outside that component it is a variational continuation only:
maximizers are generalized canonical candidates, and their realization as
actual invariant path preparations requires a corresponding global
empirical-process LDP.  Multiple maximizers encode phase coexistence.
\end{proposition}
\begin{proof}
On the declared pressure domain, Legendre duality gives
\[
 Q_R(\pair{\xi}{\kappa_R}-s\tau_R)
 =\sup_\nu\{\pair{\xi}{\nu(\kappa_R)}
 -s\nu(\tau_R)-\cI_R^{\rm proj}(\nu)\}.
\]
Solving its zero-level variational equation gives the displayed ratio.
Inside the simple spectral component, uniqueness of the equilibrium state and
the implicit-function theorem identify it with the spectral root.  The final
sentence is a scope statement, not an additional microscopic claim.
\end{proof}

On the smooth branch define the centered full pressure Hamiltonian
\[
 \boxed{
 \widehat\cH_{R,\xi}(p)
 =\cH_R(\xi+p)-\cH_R(\xi)-v_R(\xi)\cdot p.}
\]
It retains every physical current cumulant available on the translated simple
spectral component and has Hessian \(D_R(\xi)\) at the origin.  The analogous
formula with \(\cH_R^{\rm var}\) defines a comparison-stable convex PDE
continuation but is not silently identified with microscopic dynamics outside
the proved pressure domain.

\section{Central space-time Gibbs conditioning}\label{sec:conditioning}
Let
\[
 K_{-n,n}=\sum_{k=-n}^{n-1}\kappa_R\circ\Theta^k,
 \qquad
 T_{-n,n}=\sum_{k=-n}^{n-1}\tau_R\circ\Theta^k.
\]
Fix \(\xi\in\mathcal V_R\), and put
\[
 a_\xi=\int\kappa_R\,d\nu_{R,\xi},
 \qquad
 t_\xi=\int\tau_R\,d\nu_{R,\xi}.
\]
Take admissible lattice vectors \(k_n\) and roof half-widths \(b_n\) such that
\[
 \frac{k_n}{2n}\longrightarrow a_\xi,
 \qquad b_n\longrightarrow\infty,
 \qquad b_n/n\longrightarrow0.
\]

\begin{theorem}[Two-sided vector-current/clock conditioning]\label{thm:conditioning}
Let \(F_m\) be a central dynamically H\"older cylinder observable supported
on coordinates \([-m,m]\), with block norm
\(\|F_m\|_{\mathrm{cyl}}\).  There are constants \(C,c>0\), uniform on compact
local field and geometry windows, such that
\[
 \begin{aligned}
 &\left|\E_{\Pp_R}\left[
 F_m\mid K_{-n,n}=k_n,
 \ |T_{-n,n}-2nt_\xi|\le b_n
 \right]
 -\int F_m\,d\nu_{R,\xi}\right|\\
 &\quad\le C e^{cm}\|F_m\|_{\mathrm{cyl}}
 \left((n-m)^{-1/2}+\frac{b_n}{n}+\frac1{b_n}
 +\left|\frac{k_n}{2n}-a_\xi\right|\right).
 \end{aligned}
\]
The estimate permits growing central blocks whenever \(m=m_n=o(n)\) and the
right-hand side tends to zero.  No assertion that the finite conditioned path
law is Markov is used.
\end{theorem}
\begin{proof}
Use Fourier inversion only in the \(\Z^2\)-valued displacement coordinate.
The full lattice span removes peripheral arithmetic modes and the local
Fourier integral gives the \((n-m)^{-1/2}\) term.  For the continuous roof,
apply the joint spectral Laplace principle on the shrinking macroscopic
interval
\([t_\xi-b_n/(2n),t_\xi+b_n/(2n)]\).  Its unique constrained minimizer has a
joint multiplier converging to \((\xi,-\cH_R(\xi))\); displacement mismatch
and interval width give the two macroscopic error terms, while smoothing the
roof indicator costs \(b_n^{-1}\).  Uniform spectral perturbation makes the
central local-observable ratio converge to the equilibrium law at this
minimizer.  The central placement leaves \(n-m\) iterates on each side, so the
left and right leading eigenvectors pair to the stationary density; block
norms cost at most \(e^{cm}\).  This proves a vector space-time Gibbs
conditioning theorem without a temporal lattice local-limit assumption.
\end{proof}

\section{Coordinate covariance and main theorem}
If a new lattice basis is obtained by \(A\in GL(2,\Z)\), then
\[
 \kappa'=A^{-1}\kappa,\qquad
 \xi'=A^T\xi,
\]
and
\[
 \cH_R'(\xi')=\cH_R(A^{-T}\xi'),\quad
 v'=A^{-1}v,\quad
 D'=A^{-1}DA^{-T}.
\]
Thus the theory lives on \(H_1\) and \(H^1\), not on a preferred coordinate.

\begin{theorem}[Homological first-principles closure]\label{thm:main}
The compact finite-horizon Sinai family with natural Liouville law determines:
\begin{enumerate}[label=(\roman*)]
\item the full homology/clock path observable;
\item a projective empirical-path rate functional;
\item a joint physical pressure root \(\cH_R\);
\item the endogenous covector \(\xi_R(v)=\nabla I_R^{\rm phys}(v)\);
\item a central microcanonical-to-canonical preparation theorem;
\item the complete centered Hamiltonian \(\widehat\cH_{R,\xi}\).
\end{enumerate}
The scalar v8 construction is every one-dimensional contraction
\(\xi=\theta\ell\).
\end{theorem}
\begin{proof}
Combine \cref{prop:mechanical,thm:spectral,thm:level3,thm:root,thm:conditioning}.
\end{proof}
