\section{Round-three positive closure: the complete vector--roof packet}
\label{sec:r3-a2}

This section replaces the imported spectral packet and the unproved arithmetic
paragraph in the body.  The proof is organized as a finite Banach bundle over
the radius interval, an explicit temporal non-integrability rectangle, and a
three-frequency Fourier argument.  The constants are uniform for
\[
 \mathcal I=[9/20,47/100].
\]

\subsection{Uniform geometry and the moving-cut bundle}

Write
\[
 g_*=1-2(47/100)=3/50,
 \qquad
 h_*=2(9/20)-\sqrt3/2>0.
\]
The first number is the minimum separation of neighboring disks and the
second is the strict blocking margin for the three primitive corridor
directions.  Curvature lies in
\([100/47,20/9]\).

\begin{lemma}[Uniform finite-horizon geometry]
\label{lem:r3-a2-geometry}
There are constants
\[
 0<\tau_-<\tau_+<\infty,
 \quad \Lambda_*>1,
 \quad C_{\rm d}<\infty,
 \quad 0<\vartheta_*<1,
\]
which work for every \(R\in\mathcal I\).  More precisely, every regular free
flight has length in \([\tau_-,\tau_+]\); the standard stable and unstable
cones are invariant; homogeneous unstable curves expand by at least
\(\Lambda_*\); homogeneous distortion is bounded by \(C_{\rm d}\); and the
one-step expansion sum is at most \(\vartheta_*\).
\end{lemma}

\begin{proof}
Disjointness and the lower flight bound follow from \(g_*>0\).  A free line
of unbounded length would have a limiting direction.  Reducing this direction
modulo the triangular lattice gives one of the three primitive corridor
strips.  Their transverse width is \(\sqrt3/2\), whereas the two disks which
bound a strip occupy transverse width at least \(2R\ge 9/10\).  The positive
margin \(h_*\) therefore bounds the number of crossed cells, and compactness
of direction and base point gives \(\tau_+\).

For a dispersing collision, the mirror equation gives, on an unstable vector,
\[
 \mathcal B^+
 =\frac{1}{\tau+(\mathcal B^-+\kappa/\cos\varphi)^{-1}}
   +\frac{\kappa'}{\cos\varphi'} .
\]
The bounds on flight length and curvature give a common invariant cone and a
common expansion away from grazing.  On the homogeneity strip
\(k^{-2}\le\cos\varphi<(k-1)^{-2}\), the first two derivatives have the
usual polynomial bounds in \(k\), with coefficients depending only on
\(\tau_\pm\) and the curvature interval.  Summing inverse expansions over
children gives a continuous function of \(R\).  It is strictly below one for
each dispersing table, and the compact parameter interval therefore gives a
common \(\vartheta_*<1\).  The same formulas give \(C_{\rm d}\).
\end{proof}

Use common angular coordinates \((r,\varphi)\in\mathbb T\times
(-\pi/2,\pi/2)\).  Let \(\mathcal W_R^s\) be stable curves of length at most
\(\delta_0\), cut by singularities and homogeneity boundaries.  For a
distribution \(h\), define
\[
 \|h\|_w
 =\sup_{W\in\mathcal W_R^s}
   \sup_{|\psi|_{C^\alpha(W)}\le1}
   \left|\int_W h\psi\,dm_W\right|,
\]
let \(\|h\|_s\) be the same expression with the standard length weight, and
let \(\|h\|_u\) compare the pairings on two stable curves joined by unstable
holonomy.  Put
\(\|h\|_{\mathcal B_R}=\|h\|_s+b\|h\|_u\) and denote its weak completion by
\(\mathcal B_{R,w}\).

\begin{lemma}[Finite moving-cut atlas]
\label{lem:r3-a2-atlas}
There is a finite cover \((U_j)_{j=1}^J\) of \(\mathcal I\), reference spaces
\(\mathcal B_j\hookrightarrow\mathcal B_{j,w}\), and bounded pullback and
restriction maps
\[
 T_{R,j}:\mathcal B_R\longrightarrow\mathcal B_j,
 \qquad
 S_{R,j}:\mathcal B_j\longrightarrow\mathcal B_R,
 \qquad R\in U_j,
\]
such that \(S_{R,j}T_{R,j}=I\), the norms are uniformly equivalent, and the
physical pairing with every stable curve is preserved.  Singular curves which
enter or leave a chart are represented as cut boundaries and do not create an
uncontrolled norm term.
\end{lemma}

\begin{proof}
For a fixed \(R_0\), every singular curve of depth one is a \(C^2\) graph
between finitely many tangencies.  The implicit-function theorem continues
these graphs for \(R\) near \(R_0\).  Near a tangency use the homogeneous
coordinate \(u=\operatorname{sgn}(\varphi)(\pi/2-|\varphi|)^{1/2}\); in this
coordinate the continued curves and their holonomies have uniform \(C^2\)
bounds.  Cutting at every continued endpoint makes the incidence pattern
constant in the chart.  Pullback along the graph continuation preserves the
stable-curve integral and changes the strong and weak norms by
\(1+O(|R-R_0|)\).  A finite subcover of the compact interval gives the atlas.
When a curve crosses a chart boundary, the two pieces add with their physical
orientations, so no trace or seam mass is introduced.
\end{proof}

\subsection{Uniform twisted spectrum}

The displacement \(\kappa_R\) is constant on every continuity component and
has finite range.  On a homogeneous branch, the roof \(\tau_R\) and its first
two stable/unstable derivatives obey the same strip weights as the Jacobian.

\begin{lemma}[Vector--roof multiplier bound]
\label{lem:r3-a2-multiplier}
There are \(\delta>0\) and \(C<\infty\) such that, for complex
\((\xi,s)\in\mathbb C^2\times\mathbb C\) with
\(|\xi|+|s|<\delta\), multiplication by
\[
 e^{\langle\xi,\kappa_R\rangle-s\tau_R}
\]
is analytic from the strong space to itself and satisfies
\[
 \|M_{R,\xi,s}-I\|_{\mathcal B_R\to\mathcal B_R}
 +\|M_{R,\xi,s}-I\|_{\mathcal B_{R,w}\to\mathcal B_{R,w}}
 \le C(|\xi|+|s|)
\]
uniformly in \(R\).
\end{lemma}

\begin{proof}
The displacement factor is branchwise constant.  For the roof factor, expand
the exponential in its absolutely convergent power series.  The product
estimate in the stable norm uses bounded oscillation on a homogeneous stable
curve; the unstable norm uses the holonomy estimate for the roof.  The strip
weights from Lemma~\ref{lem:r3-a2-geometry} make the sums uniform.  Termwise
differentiation proves analyticity and the displayed Lipschitz estimate.
\end{proof}

\begin{theorem}[Uniform complex spectral bundle]
\label{thm:r3-a2-spectrum}
For \(|\xi|+|s|<\delta\),
\[
 \mathcal L_{R,\xi,s}h
 =\mathcal L_R\left(
 e^{\langle\xi,\kappa_R\rangle-s\tau_R}h\right)
\]
has a simple leading eigenvalue \(e^{P_R(\xi,s)}\), and
\[
 \mathcal L_{R,\xi,s}^n
 =e^{nP_R(\xi,s)}\Pi_{R,\xi,s}+N_{R,\xi,s}^n,
 \qquad
 \|N_{R,\xi,s}^n\|_{\mathcal B_R\to\mathcal B_{R,w}}
 \le C\rho^n e^{n\Re P_R(\xi,s)}
\]
with one \(\rho<1\).  The pressure and projector are analytic in
\((\xi,s)\), continuously differentiable in \(R\), and agree on chart
overlaps.
\end{theorem}

\begin{proof}
The growth lemma and distortion bounds give, for suitable \(b\),
\[
 \|\mathcal L_R^nh\|_{\mathcal B_R}
 \le C\rho_0^n\|h\|_{\mathcal B_R}+C\|h\|_{\mathcal B_{R,w}},
 \qquad
 \|\mathcal L_R^nh\|_{\mathcal B_{R,w}}
 \le C\|h\|_{\mathcal B_{R,w}},
\]
with uniform constants.  Compactness of the strong unit ball in the weak
space gives uniform quasicompactness.  Mixing of the finite-horizon collision
map makes the eigenvalue one simple.  Lemma~\ref{lem:r3-a2-multiplier} and a
resolvent contour lying in the common spectral gap give analytic perturbation
with a common radius.  The moving-cut atlas identifies the resolvent pairings
on overlaps, hence also the Riesz projectors.  Differentiating the charted
operators, including the boundary-motion term already represented by the cut
norm, gives continuous \(R\)-dependence.
\end{proof}

The same proof applies to a fixed finite collision cylinder inserted in the
middle of the orbit.  In particular left and right eigenvectors can be
paired with a central-window observable without replacing it by an evolved
smooth observable.

\subsection{A concrete temporal non-integrability rectangle}

Consider the triangular gap formed by the disks at centers
\(0,e_1,e_2\).  Choose a small unstable interval \(J_R\) on the inward-facing
arc of the disk at \(0\).  There are two regular three-flight inverse branches
\(h_R^+\) and \(h_R^-\): the first visits the inward arcs at
\(e_1,e_2,0\), and the second visits \(e_2,e_1,0\).  The strict margins
\(g_*\) and \(h_*\) allow the interval to be chosen with one size for all
\(R\).

Let
\[
 D_R(x)=
 \sum_{n=0}^{\infty}
 \left[
 \tau_R\big((h_R^+h_R^0)^n x\big)
 -\tau_R\big((h_R^-h_R^0)^n x\big)
 \right],
\]
where \(h_R^0\) is the common return branch from the terminal inward arc to
\(J_R\).  The series and its derivative converge by hyperbolicity.

\begin{lemma}[Uniform temporal UNI]
\label{lem:r3-a2-uni}
After shrinking \(J_R\) by a radius-independent factor, there is
\(c_{\rm UNI}>0\) such that
\[
 \inf_{R\in\mathcal I}
 \inf_{x\in J_R}|D_R'(x)|\ge c_{\rm UNI}.
\]
\end{lemma}

\begin{proof}
For a free segment joining boundary points with outgoing and incoming angles
\(\varphi,\varphi'\), differentiation with respect to boundary arclength gives
\[
 \partial_r\tau=\sin\varphi-\sin\varphi'\,\partial_r r'.
\]
At the symmetric center point of the triangular gap, the two itineraries are
mirror images.  Their first unequal legs have outgoing angles separated by
\(2\beta_R\), where elementary equilateral-triangle geometry gives
\[
 \sin\beta_R
 \ge \frac{h_*}{2(1+\tau_+)}.
\]
Thus the derivative difference of the first unequal pair is at least
\(2\sin\beta_R\).  The derivative contribution of all later paired legs is
bounded by
\[
 C_{\rm d}\sum_{n\ge1}\Lambda_*^{-n}.
\]
Choose the common return branch with sufficiently many iterates before the
next unequal pair; this makes the tail at most half the first-pair bound.
Continuity then permits one common shrink of the interval.  The resulting
lower bound depends only on \(g_*,h_*,\tau_+,C_{\rm d},\Lambda_*\), and is
therefore uniform in \(R\).
\end{proof}

\begin{theorem}[Joint aperiodicity and high-frequency contraction]
\label{thm:r3-a2-high}
If \((u,t)\in[-\pi,\pi]^2\times\mathbb R\) and
\(\mathcal L_{R,iu,it}\) has peripheral spectral radius one, then
\(u=0\) and \(t=0\).  Moreover there are \(A,c,C>0\) such that
\[
 \|\mathcal L_{R,iu,it}^n\|_{\mathcal B_R\to\mathcal B_{R,w}}
 \le C|t|^A e^{-cn/\log(2+|t|)}
\]
for \(|t|\ge1\), uniformly in \(R\) and \(u\).
\end{theorem}

\begin{proof}
A peripheral eigenfunction has constant modulus and therefore makes
\(u\cdot S_n\kappa_R-tS_n\tau_R\) constant modulo \(2\pi\) on every inverse
branch.  Differentiating the difference of the two branches in
Lemma~\ref{lem:r3-a2-uni} forces \(t=0\).  With \(t=0\), open nearest-neighbor
branches with displacements \(\pm e_1,\pm e_2\) force
\(u\in2\pi\mathbb Z^2\), hence \(u=0\) in the fundamental domain.

For the quantitative estimate, pair the two inverse branches on each return
to the UNI rectangle.  On unstable intervals of length comparable with
\(|t|^{-1}\), the phase difference has derivative at least
\(c_{\rm UNI}|t|\); one of the paired complex weights therefore loses a fixed
fraction after a bounded number of iterates.  The growth lemma returns a fixed
fraction of every standard family to the rectangle in
\(O(\log(2+|t|))\) iterates.  Iterating the cancellation and using the strong
to weak Lasota--Yorke estimate gives the displayed bound.  Frequencies in a
compact annulus are handled by upper semicontinuity of the spectral radius and
the absence of peripheral modes just proved.
\end{proof}

\subsection{Joint lattice--nonlattice local limit theorem}

Let \(K_n=S_n\kappa_R\), \(T_n=S_n\tau_R\), and let
\((a,b)=\nabla P_R(\xi,s)\) on a real simple branch.  Write
\(\Sigma_{R,\xi,s}\) for the covariance matrix of
\((\kappa_R,\tau_R)\) under the tilted equilibrium law.

\begin{lemma}[Uniform nondegeneracy]
\label{lem:r3-a2-cov}
On every compact subset of the real simple branch,
\[
 cI_3\le\Sigma_{R,\xi,s}\le CI_3
\]
with constants independent of \(R\).
\end{lemma}

\begin{proof}
A zero variance direction would make
\(u\cdot\kappa_R-t\tau_R\) a measurable coboundary.  Periodic-orbit
identities and Lemma~\ref{thm:r3-a2-high} then give \((u,t)=0\).  Continuity
of the Hessian and compactness give the uniform lower and upper bounds.
\end{proof}

\begin{theorem}[Uniform vector--clock local limit]
\label{thm:r3-a2-llt}
Let \(F\) be a bounded central cylinder.  Uniformly for \(R\), for tilts in a
compact simple branch, for
\(k_n-na=O(\sqrt n)\), and for intervals
\(J_n=[t_n-b_n,t_n+b_n]\) with
\(b_n\to\infty\) and \(b_n=o(n)\),
\[
 \begin{aligned}
 &\mathbb E_{R,0}
 \left[F;\ K_n=k_n,\ T_n\in J_n\right]\\
 &\quad=
 e^{-nI_R(k_n/n,t_n/n)}
 \frac{2b_n}{(2\pi n)^{3/2}
 \sqrt{\det\Sigma_{R,\xi,s}}}
 \left(\nu_{R,\xi,s}(F)+o(1)\right).
 \end{aligned}
\]
The \(o(1)\) is uniform for \(F\) in the unit ball of a fixed cylinder
space.  The conclusion includes the regime \(b_n=o(\sqrt n)\).
\end{theorem}

\begin{proof}
Exponentially tilt to the unique real parameters whose means equal
\((k_n/n,t_n/n)\).  Fourier inversion is taken on
\([ -\pi,\pi]^2\times\mathbb R\).  On
\(|(u,t)|\le n^{-2/5}\), Theorem~\ref{thm:r3-a2-spectrum} gives, uniformly
with the central insertion,
\[
 P_R(\xi+iu,s+it)-P_R(\xi,s)
 =i\langle(u,t),(a,b)\rangle
 -\tfrac12\langle(u,t),\Sigma_{R,\xi,s}(u,t)\rangle
 +O(|(u,t)|^3).
\]
After the change of variables \((u,t)=n^{-1/2}z\), dominated convergence and
Lemma~\ref{lem:r3-a2-cov} give the three-dimensional Gaussian factor.  A
smooth upper and lower approximation to \(1_{J_n}\) has Fourier mass
\(2b_n(1+o(1))\) on this scale and error \(o(b_nn^{-3/2})\) because
\(b_n\to\infty\).

On the intermediate region, the covariance lower bound and the spectral
perturbation estimate give \(e^{-cn|(u,t)|^2}\).  On compact minor arcs, the
spectral radius is uniformly below one; on unbounded temporal frequencies use
Theorem~\ref{thm:r3-a2-high} and integrate by parts in the smoothed interval
transform.  These contributions are \(o(b_nn^{-3/2})\).  The left and right
projectors surrounding the central insertion pair to the stationary tilted
law, which gives \(\nu_{R,\xi,s}(F)\).
\end{proof}

\begin{corollary}[Two-sided current--clock conditioning]
\label{cor:r3-a2-conditioning}
For every fixed central window and every admissible sequence in
Theorem~\ref{thm:r3-a2-llt}, the conditional central-window law given
\(K_n=k_n\) and \(T_n\in J_n\) converges in total variation on that finite
window to the stationary tilted path law.  Quantitatively the error is
\[
 O\left(n^{-1/2}+b_n/n+b_n^{-1}
\right)
\]
plus the error in matching the prescribed means to the tilt.
\end{corollary}

\begin{proof}
Apply Theorem~\ref{thm:r3-a2-llt} to the numerator with a central cylinder and
to the denominator with \(F=1\).  The Gaussian main term supplies a uniform
positive denominator.  The left and right spectral projectors cancel the
endpoint eigenfunction bias, and division gives the estimate.
\end{proof}

\subsection{Physical pressure root}

Because \(\partial_sP_R(0,0)=-\mu_R(\tau_R)<0\), the equation
\[
 P_R(\xi,\mathcal H_R(\xi))=0
\]
has a uniform analytic solution near zero.

\begin{theorem}[Closed homological pressure theorem]
\label{thm:r3-a2-main}
The pressure root is analytic jointly in the homological field and
continuously differentiable in \(R\), and
\[
 \nabla\mathcal H_R(\xi)
 =\frac{\nu_{R,\xi}(\kappa_R)}{\nu_{R,\xi}(\tau_R)},
\]
\[
 \nabla^2\mathcal H_R(\xi)
 =\frac{\operatorname{AVar}_{\nu_{R,\xi}}
 (\kappa_R-\nabla\mathcal H_R(\xi)\tau_R)}
 {\nu_{R,\xi}(\tau_R)}.
\]
The Hessian is uniformly positive definite on a common field ball.  Exact
vector-current and submacroscopic clock conditioning on a central window are
given by Corollary~\ref{cor:r3-a2-conditioning}.
\end{theorem}

\begin{proof}
The implicit-function theorem applies to
Theorem~\ref{thm:r3-a2-spectrum}.  Differentiating the root once and twice
gives the two formulas; the second is the Schur complement of the joint
vector--roof covariance.  Lemma~\ref{lem:r3-a2-cov} and the positive mean roof
give uniform positivity.  The conditioning statement was proved above.
\end{proof}
