\section{Round-thirteen positive closure: the parent transfer bundle and an integrable Fourier theorem}
\label{sec:r13-a2}

The proof is carried out on one physical parent-branch bundle.  Symbolic
births are changes of coordinates inside that bundle, not new operator
summands multiplied by a Heaviside factor.  The Fourier line is divided into
ranges whose estimates are genuinely integrable; in particular no uniform
gap is claimed at frequencies tending to zero and no frequency-independent
remainder is left on an infinite interval.

\subsection{The moving physical bundle}

Let $R\in\mathcal I=[9/20,47/100]$ and let $T_R$ be the collision map of the
finite-horizon triangular Lorentz gas.  Stable curves are represented in a
fixed tubular atlas of the scatterer boundary.  A physical inverse branch is
continued as one parent graph until the first tangency.  When a symbolic cut
is born, the two cut pieces are represented by the restrictions of the same
parent distribution.

For an integer $m\ge2$ let $\mathbb B_R^m\hookrightarrow\mathbb B_{R,w}^m$
be the completion of bulk pairings on stable curves together with oriented
trace pairings on every homogeneity boundary of depth at most $m$.  The norm
contains the usual stable Hölder, unstable holonomy, distortion, and trace
terms with one depth weight $\vartheta^j$, $0<\vartheta<1$.

\begin{theorem}[Uniform parent-bundle Lasota--Yorke theorem]
\label{thm:r13-a2-bundle}
There are common constants $C<\infty$, $0<\rho<1$, and a finite parameter
atlas such that, for every $R\in\mathcal I$,
\[
 \|\mathcal L_R^nh\|_{\mathbb B_R^m}
 \le C\rho^n\|h\|_{\mathbb B_R^m}
      +C\|h\|_{\mathbb B_{R,w}^m},
 \qquad
 \|\mathcal L_R^nh\|_{\mathbb B_{R,w}^m}
 \le C\|h\|_{\mathbb B_{R,w}^m}.
\]
The strong unit ball is compact in the weak space.  In a chart, the physical
operator and every bounded cylinder insertion are $C^q$ in $R$ as maps
$\mathbb B^{m+q}	o\mathbb B^m$.  At a symbolic branch birth the derivative
is the derivative of the parent push-forward; no one-sided square-root term or
Heaviside derivative occurs in the physical operator.
\end{theorem}

\begin{proof}
Finite horizon, uniform curvature, and the compact radius interval give common
cone, expansion, distortion, and one-step-growth constants.  The trace part
records each moving cut before taking a material derivative, so boundary
motion is a bounded strong-to-weak operator.  In a fold chart the two symbolic
pieces satisfy
\[
 \int_{J_+(R)}g_R+\int_{J_-(R)}g_R
 =\int_{J_{\rm parent}}g_R,
\]
with the parent coordinate and density $C^q$ in $R$.  The endpoint terms of
the two restrictions cancel with opposite orientations.  Iterating the graph
growth estimate and summing the depth weights gives the displayed
Lasota--Yorke bounds.  Arzelà--Ascoli on stable tests and the compact trace
embedding give strong-to-weak compactness.  Differentiating the parent branch
loses one strong level per derivative and proves the material statement.
\end{proof}

\subsection{Joint twists, covariance damping, and arithmetic}

Let $\kappa_R\in\mathbb Z^2$ be displacement, $r_R\in\mathbb Z_+$ the return
count, and $\tau_R>0$ the roof.  For
$(u,v,b)\in\mathbb T^2\times\mathbb T\times\mathbb R$ define
\[
 \mathcal L_{R,u,v,b}h
 =\mathcal L_R\left(
 e^{i(u\cdot\kappa_R+v r_R+b\tau_R)}h\right).
\]
For real exponential tilts in a compact regular chart, denote the centered
joint covariance of $(\kappa,r,\tau)$ by $\Sigma_R$.

\begin{lemma}[Uniform covariance damping near zero]
\label{lem:r13-a2-small}
There are $b_0,c,C>0$ such that, uniformly in $R$ and the regular real tilt,
\[
 \|\mathcal L_{R,u,v,b}^n\|_{\rm s\to w}
 \le C\exp\{-cn(|u|^2+|v|^2+b^2)\}
\]

after removing the leading oscillatory mean, whenever
$|(u,v,b)|\le b_0$.  The covariance matrices satisfy
$cI\le\Sigma_R\le CI$.
\end{lemma}

\begin{proof}
Theorem~\ref{thm:r13-a2-bundle} and analytic multiplier estimates give a
simple eigenvalue
\[
 \lambda_R(i\zeta)
 =\exp\{i m_R\cdot\zeta-	frac12
          \zeta^T\Sigma_R\zeta+O(|\zeta|^3)\}.
\]
If a covariance direction vanished, the corresponding joint observable would
be a measurable coboundary.  The returned-branch temporal-distance argument
below eliminates the roof coefficient, and open regular branches with
independent homology and return count eliminate the remaining lattice
coefficients.  Compactness then gives the uniform lower bound.  The remainder
of the spectrum is uniformly separated and the Taylor expansion gives the
stated $e^{-cn|\zeta|^2}$ damping, including the range where $|b|$ tends to
zero with $n$.
\end{proof}

\begin{theorem}[Uniform joint aperiodicity from returned UNI]
\label{thm:r13-a2-arithmetic}
There are two returned inverse branches $h_R^\pm$ on one common unstable
interval and a temporal-distance function $D_R$ such that
\[
 \inf_{R\in\mathcal I}\inf_x|D_R'(x)|>0.
\]
If $u\cdot\kappa_R+vr_R+b\tau_R$ is a circle-valued coboundary, then
$b=0$ and $(u,v)=(0,0)$ in the fundamental torus.  Consequently every compact
annulus separated from the origin has spectral radius uniformly below one.
\end{theorem}

\begin{proof}
Choose the two mirror return words through the triangular gap and append one
fixed regular connector.  The first unequal free flights have opposite
momentum derivatives bounded away from zero by the finite-horizon clearance.
The derivatives of the common tails form a convergent geometric series, so a
fixed number of appended common returns makes their total smaller than half
the first difference.  This proves the displayed UNI bound uniformly in $R$.

A peripheral eigenfunction makes the phase difference of the two inverse
branches constant modulo $2\pi$.  Differentiation gives $bD_R'=0$, hence
$b=0$.  Regular open branches realizing two independent lattice displacements
and two different return counts are closed by the uniform connector family;
the coboundary periodic sums force $u\in2\pi\mathbb Z^2$ and
$v\in2\pi\mathbb Z$.  Upper semicontinuity of the spectral radius on a compact
annulus gives the last assertion.  This argument uses actual open branch
families and closing, not nonexistent one-collision winding orbits.
\end{proof}

\subsection{An integrable four-range theorem}

Fix $N\ge6$.  Choose $\kappa>0$ sufficiently small for the moderate-frequency
estimate below, and set $L_n=e^{\kappa\sqrt n}$.

\begin{theorem}[Complete Fourier-line estimate]
\label{thm:r13-a2-fourier}
Uniformly in $R$, in a compact real saddle chart, and with one bounded central
cylinder insertion, the following bounds hold.
\begin{enumerate}[label=(\roman*)]
\item For $|(u,v,b)|\le n^{-2/5}$, the simple eigenvalue has a third-order
      Gaussian expansion with a uniform remainder.
\item For $n^{-2/5}<|(u,v,b)|\le b_0$,
\[
 \|\mathcal L_{R,u,v,b}^n\|_{\rm s\to w}
 \le Ce^{-cn|(u,v,b)|^2}.
\]
\item For $b_0<|b|\le L_n$,
\[
 \|\mathcal L_{R,u,v,b}^n\|_{\rm s\to w}
 \le C|b|^A\exp\{-cn/\log(2+|b|)\}.
\]
The constant $\kappa$ is chosen so that the integral of this bound over the
range is $O(e^{-c_1\sqrt n})$.
\item For $|b|>L_n$,
\[
 \|\mathcal L_{R,u,v,b}^n\|_{\rm s\to w}
 \le C_N n^{C_N}(1+|b|)^{-N}
      +Ce^{-cn}(1+|b|)^{-2}.
\]
Hence the integral over the very-high range is $O(e^{-c_2\sqrt n})$.
\end{enumerate}
All four estimates are stable under a fixed number of material derivatives.
\end{theorem}

\begin{proof}
Parts (i)--(ii) are Lemma~\ref{lem:r13-a2-small}.  On the compact annulus
$[b_0,B]$ use Theorem~\ref{thm:r13-a2-arithmetic}.  For larger moderate
frequencies, pair the returned UNI branches in standard families.  One fixed
fraction of mass returns to the UNI rectangle in $O(\log|b|)$ iterates, and
the phase derivative gives a fixed cancellation.  Iteration yields the
Dolgopyat bound in (iii).  Since $\log|b|\le\kappa\sqrt n$, its integral is
bounded by
\[
 \exp\{(A+1)\kappa\sqrt n-c\sqrt n/\kappa\};
\]
choose $\kappa^2<c/(2(A+1))$.

For (iv), select $N$ well-separated returns to the UNI rectangle, with smooth
cutoffs supported strictly inside their branch domains.  Integrate once in the
unstable coordinate of each selected block.  Derivatives of holonomies and
amplitudes grow only polynomially in $n$ because $N$ is fixed.  Boundary
terms vanish on the interior cutoffs.  The complement is partitioned by a
terminal roof coordinate; two integrations there give
$e^{-cn}(1+|b|)^{-2}$ rather than a frequency-independent error.  Summing the
standard-family partition gives the displayed bound.  Material derivatives
consume fresh strong levels but do not change the fixed number of integrations.
\end{proof}

\subsection{The joint density local limit theorem}

Let $(k_n,\ell_n,t_n)$ be a lattice--lattice--continuous target and let the
real saddle be chosen so that its mean is
$(k_n/n,\ell_n/n,t_n/n)$.  Define the normalized deviation
\[
 z_n={1\over\sqrt n}
 \begin{pmatrix}
 k_n-na_R\\[1mm]
 \ell_n-n\bar r_R\\[1mm]
 t_n-n\bar\tau_R
 \end{pmatrix}.
\]

\begin{theorem}[Uniform vector--count--roof density LLT]
\label{thm:r13-a2-llt}
For a bounded central cylinder $F$, uniformly on compact regular saddle
charts,
\[
 \begin{aligned}
 &\mathbb E_R\!
 \left[F;K_n=k_n,\ R_n=\ell_n,\ T_n\in dt\right]\\
 &\quad=
 e^{-nI_R(k_n/n,\ell_n/n,t/n)}
 {\nu_{R,*}(F)+O(n^{-\eta})
  \over(2\pi n)^{3/2}\sqrt{\det\Sigma_R}}
 \exp\{-\tfrac12 z_n^T\Sigma_R^{-1}z_n\}\,dt,
 \end{aligned}
\]
locally uniformly for $|z_n|\le n^{1/10}$.  Integrating the conditional roof
coordinate over any interval gives its exact Gaussian interval mass.  In
particular the subcentral, central, and saturated window regimes follow from
one formula, with no missing factor of $n$.
\end{theorem}

\begin{proof}
Use lattice Fourier inversion in $(u,v)$ and ordinary Fourier inversion in
$b$.  The small cube gives the Gaussian density after the change of variables
$\zeta=\sqrt n(u,v,b)$.  The covariance lower bound supplies domination.  The
remaining small annulus is integrable by part (ii) of
Theorem~\ref{thm:r13-a2-fourier}; the compact/moderate and very-high ranges are
integrable by parts (iii)--(iv).  The Riesz projector surrounding the central
insertion converges to the tilted stationary expectation.  Integrating the
last Gaussian coordinate over the requested unnormalized roof interval gives
the exact conditional Gaussian interval probability.
\end{proof}

\begin{theorem}[Round-thirteen A2 closure]
\label{thm:r13-a2-main}
The compact Sinai radius family has a uniform parent-branch anisotropic
bundle, joint arithmetic/UNI control, an integrable estimate on the entire
Fourier line, and the vector--count--roof density LLT of
Theorem~\ref{thm:r13-a2-llt}.
\end{theorem}

\begin{proof}
Combine Theorems~\ref{thm:r13-a2-bundle},
\ref{thm:r13-a2-arithmetic}, \ref{thm:r13-a2-fourier}, and
\ref{thm:r13-a2-llt}.
\end{proof}
