\section{Two pressures on an admissible language}\label{sec:v6-pressure}
All pressures in this section are taken on $\Sigma_A$. A H\"older
potential $F$ has uniformly bounded oscillation of $S_nF$ on cylinders,
and its partition sums are comparable to $e^{nP_A(F)}$, uniformly in
$n$. These are classical finite-state thermodynamic facts
\cite{Bowen2008}. The new issue is the renewal-weighted sum of suffix
scales, which is not an additive potential.

\begin{lemma}[Mixed partition pressure]\label{lem:v6-partition}
For $0<s<1$, under \eqref{eq:v6-tail-rate},
\begin{equation}
 \lim_{n\to\infty}\frac1n\log\sum_{v\in\mathcal L_n}a_w(v)^s
 =\max\{P_A(s\varphi+\tau s\psi),\ P_A(s\varphi)+s\log q\}.
                                                        \label{eq:v6-mixed-pressure}
\end{equation}
\end{lemma}
\begin{proof}
Put $P_g(s)=P_A(s\varphi+\tau s\psi)$ and $P_0(s)=P_A(s\varphi)$.
The first and last terms in \eqref{eq:v6-energy}, separately, imply
\[
 \sum_{|v|=n}a_w(v)^s\ge
 c_s\max\{w_0^s e^{nP_g(s)},\ W(n)^s e^{nP_0(s)}\}.
\]
For the upper bound use $(\sum x_k)^s\le\sum x_k^s$ and split an
admissible word after $k$ letters. Its mass and suffix scale are,
within fixed factors, the exponential of the corresponding Birkhoff
sums. Dropping the constraint joining the two pieces can only increase
the sum. Thus
\begin{equation}
 \sum_{|v|=n}a_w(v)^s\le C_s\left(
 \sum_{k<n}w_k^s e^{kP_0(s)}e^{(n-k)P_g(s)}
       +W(n)^s e^{nP_0(s)}\right).                       \label{eq:v6-convolution}
\end{equation}
No factorization of arbitrary words across a forbidden transition is
asserted. Given $\delta>0$, the weight assumption gives
$w_k\le C_\delta e^{k(\log q+\delta)}$, and $W(n)\asymp w_n$.
The upper sum has at most $n+1$ terms with exponential rate at most
$\max(P_g(s),P_0(s)+s\log q+s\delta)$. The lower bound has rate at
least the same maximum with $\delta=0$. Let $\delta\downarrow0$.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:v6-pressure}]
We first verify uniqueness of the roots. Every vertex has at least two
continuations of a common length $\ell$. By the positive conditional
mass bound, no particular continuation of that length has conditional
probability larger than $1-\zeta^\ell$. Hence
$\mu[v]\le C\eta^{|v|}$ for an $\eta<1$. The Gibbs estimate gives
$S_n\varphi\le n\log\eta+O(1)$ on every cylinder. Similarly
$S_n\psi\le n\log r_++O(1)$. The pressure functions in
\eqref{eq:v6-roots} are continuous and strictly decreasing, with
positive value $\log\lambda_A$ at zero. At one their values are
strictly negative, since $P_A(\varphi)=0$. Their roots lie in $(0,1)$.

For $s>s_*$, both exponents in \eqref{eq:v6-convolution} are negative.
Choose $q''>q$ close enough that they remain negative with $q''$ in
place of $q$. Bounding $w_k$ and $W(k)$ by a constant times $(q'')^k$
and summing \eqref{eq:v6-convolution} over all $n$ yields
\begin{equation}
                  \sum_{v\in\mathcal L}a_w(v)^s<\infty. \label{eq:v6-summability}
\end{equation}
For a balanced partition with $L\asymp N$ leaves, this implies
$L(a_0\theta_N)^s\le C_s$, and therefore
$G_N\le L\theta_N\le C_s'N^{-(1-s)/s}$.

For the lower bound at $s_g$, use the Gibbs probability $\nu_g$ for
the zero-pressure potential $s_g\varphi+\tau s_g\psi$. On each cylinder,
\[
 \nu_g[v]\le C\mu[v]^{s_g}r_v^{\tau s_g}\le C' a_w(v)^{s_g}.
\]
Summing over $\mathcal P_N$ gives $1\le C'L\theta_N^{s_g}$.
Balance then gives $G_N\ge cN^{-(1-s_g)/s_g}$.

For the survival root, a polynomial correction to $w_n$ prevents
using $q^n$ as a uniform lower bound. Instead fix $0<q'<q$. There
is $c_{q'}>0$ with $W(n)\ge c_{q'}(q')^n$ for all $n$.
Let $s_t(q')$ be the zero of $P_A(s\varphi)+s\log q'$, and use the
Gibbs measure of this zero-pressure potential. Its cylinder masses
are bounded by $C\mu[v]^{s_t(q')}(q')^{|v|s_t(q')}$, hence by
$C'a_w(v)^{s_t(q')}$. The same balanced-partition argument proves
the lower bound with exponent $(1-s_t(q'))/s_t(q')$.
The roots vary continuously with $q'$, by strict monotonicity.
Letting $q'\uparrow q$ and $s\downarrow s_*$ proves the common exponent
for $G_N$. Theorem~\ref{thm:v6-profile} transfers it to both errors.
Finally $s_t=s_g$ precisely when
$P_A(s_g\varphi)+s_g\log q=0$, proving the threshold formula.
\end{proof}

\begin{corollary}[Markov matrices with zero entries]\label{cor:v6-markov-matrix}
Suppose the $h_i$ are similarities and $\mu$ is a stationary Markov
measure with transition probabilities $p_{ij}>0$ exactly when $A_{ij}=1$.
The roots in \eqref{eq:v6-roots} are characterized by
\begin{equation}
 \operatorname{spr}\bigl((A_{ij}p_{ij}^{s_g}r_i^{\tau s_g})_{ij}\bigr)=1,
 \qquad
 q^{s_t}\operatorname{spr}\bigl((A_{ij}p_{ij}^{s_t})_{ij}\bigr)=1.
                                                        \label{eq:v6-perron}
\end{equation}
Only admissible entries are powered; other entries are zero. At $s=0$
the limiting matrix is $A$, not a matrix of ones.
\end{corollary}
\begin{proof}
Take $\varphi(\omega)=\log p_{\omega_0\omega_1}$ on admissible edges
and $\psi(\omega)=\log r_{\omega_0}$. The nonuniform stationary initial
factor and the terminal factor are bounded above and below, so do not
change pressure. The finite transfer matrices of these edge potentials
are exactly those in \eqref{eq:v6-perron}. Perron--Frobenius theory and
Theorem~\ref{thm:v6-pressure} give the assertion.
\end{proof}

\begin{example}[A noninjective planar readout with a forbidden transition]
\label{ex:v6-delay}
On $[0,1]^2$ take
\[
 h_0(x,y)=(ry,rx),\qquad h_1(x,y)=(ry+2/3,rx),\qquad r=1/3,
 \qquad A=\begin{pmatrix}1&1\\1&0\end{pmatrix}.
\]
Rescaling the ambient domain makes its diameter one without changing
the argument. Both maps are similarities and their images have a
positive horizontal gap. Give the golden-mean shift any stationary
H\"older Gibbs law. For $f(x,y)=x$, the relation
$f(T(x,y))=y/r$ on $K$ gives
\[
 |f(x,y)-f(x',y')|^2+|f(T(x,y))-f(T(x',y'))|^2
 =(x-x')^2+r^{-2}(y-y')^2.
\]
Thus \eqref{eq:v6-delay} holds with $J=1$ and $\beta=1$.
The sequences $000\cdots$ and $01000\cdots$ are admissible, have the
same first coordinate, and have different second coordinates. Hence
$f$ is not injective. This is a higher-dimensional geometric system
with a genuinely forbidden transition and a delayed, rather than
instantaneously identifying, observable.
\end{example}
