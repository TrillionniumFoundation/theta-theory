\section{Round-three positive closure: a shielded two-clock level-three theorem}
\label{sec:r3-a3}

This section proves the coding, exponential approximation, random-clock upper
and lower bounds, finite-rate support, and cotangent quotient which were only
listed in the first circulation.  The construction uses the uniform billiard
packet of Paper A2, but no level-three assertion is imported from it.

\subsection{A finite-connector induced code}

Let \(Y\) be a Young magnet with finitely many crossing components
\(Y_1,\ldots,Y_m\).  Cut each first-return branch by its complete finite
collision itinerary and record the tuple
\[
 a=(i,j,r,\mathfrak t,\zeta),
\]
where \(i,j\) are the entrance and exit components, \(r\) is the collision
length, \(\mathfrak t\) the physical duration, and \(\zeta\) the regular
finite orbit branch.  Denote the alphabet by \(\mathcal A\).

\begin{lemma}[Finite connectors and exponential marks]
\label{lem:r3-a3-connect}
The magnet and its refinement may be chosen so that:
\begin{enumerate}[label=(\roman*)]
\item there is a finite set \(\mathcal C\subset\mathcal A^*\) such that for
      every pair of symbols \(a,b\) one of the words in \(\mathcal C\)
      connects \(a\) to \(b\);
\item the induced Jacobian potential \(\widehat\phi\) has summable
      variations;
\item for some \(\eta>0\),
\[
 \int e^{\eta r+\eta\mathfrak t}\,d\widehat{\mathbb P}<\infty;
\]
\item the natural extension and suspension of the code push forward to the
      Liouville collision and physical path laws.
\end{enumerate}
\end{lemma}

\begin{proof}
Mixing of the collision map gives, for each ordered pair
\((Y_i,Y_j)\), a regular orbit segment from a full unstable crossing of
\(Y_i\) to a full stable crossing of \(Y_j\).  Choose one segment for each of
the finitely many pairs and refine the return partition along these segments.
Their words form \(\mathcal C\), proving finite primitivity.  Homogeneous
bounded distortion gives
\[
 \operatorname{var}_n(\widehat\phi)
 \le C\rho^n,
\]
so the variations are summable.  The growth lemma gives an exponential tail
for the first return.  Finite horizon yields
\(\tau_-r\le\mathfrak t\le\tau_+r\), hence the joint exponential moment.
The tower projection follows every recorded collision and therefore agrees
with the billiard orbit outside the null singular set; suspension gives the
physical path law.
\end{proof}

For a return branch \(a\), let \(h(a)\) be the largest
homogeneity-strip index met by its recorded excursion.  For \(L,K\ge1\), put
\[
 \mathcal A_{L,K}=\{a:r(a)\le L,\ h(a)\le K\}.
\]
The finite-time singularity arrangement has only finitely many components
after both cutoffs, so \(\mathcal A_{L,K}\) is finite.  The high-homogeneity
set has Gibbs mass \(O(K^{-\zeta})\) for some \(\zeta>0\), uniformly on every
fixed return cutoff.

\begin{theorem}[Block empirical-measure LDP without an unverified coding
packet]
\label{thm:r3-a3-block}
The empirical block measures
\[
 \widehat L_N=\frac1N\sum_{j=0}^{N-1}
 \delta_{\widehat\Theta^j\widehat\omega}
\]
satisfy a full weak LDP with a good rate \(\widehat{\mathcal I}\).  On
invariant probabilities of finite mark moment,
\[
 \widehat{\mathcal I}(\widehat\nu)
 =\sup_{F\in C_b^{\rm cyl}}
 \left\{\widehat\nu(F)-
 \big(P(\widehat\phi+F)-P(\widehat\phi)\big)\right\},
\]
and it is infinite off invariant probabilities or when the required mark
moment is infinite.
\end{theorem}

\begin{proof}
Give the countable code the weighted variation norm
\[
 \|g\|_{\eta}=\sup e^{-\eta(r+\mathfrak t)}|g|
 +\sum_{n\ge1}\rho^{-n}\operatorname{var}_n(g),
\]
with \(0<\eta\) below the exponential-mark exponent.  Finite primitivity,
summable distortion, and the connector words give the Doeblin--Fortet
estimate
\[
 \|\widehat{\mathcal L}^{\,n}g\|_\eta
 \le C\rho_0^n\|g\|_\eta+C\|g\|_{w,\eta},
 \qquad \rho_0<1.
\]
The exponential mark tail makes the strong unit ball compact in the weighted
weak norm.  Thus every bounded cylinder twist has a simple pressure
eigenvalue on a common neighborhood, and every finite tuple of cylinder
coordinates has the finite-dimensional upper bound and exposed-point lower
bound obtained by exponential tilting.

The two missing global ingredients are exponential tightness and density of
exposed finite-state phases.  For long marks, normal convergence of the
operator twisted by \(s r1_{\{r>L\}}\) gives, for every fixed \(s<\eta\),
\[
 \lim_{L\to\infty}\big(P(\widehat\phi+s r1_{\{r>L\}})
 -P(\widehat\phi)\big)=0.
\]
Chernoff's inequality makes the empirical long-mark mass exponentially
negligible with a cost tending to infinity.  For grazing branches, the weak
operator norm of multiplication by \(1_{\{h>K\}}\) tends to zero by the
homogeneity-strip mass estimate.  Hence, for every fixed \(s>0\),
\[
 P(\widehat\phi+s1_{\{h>K\}})-P(\widehat\phi)\longrightarrow0.
\]
First let \(K\to\infty\) and then \(s\to\infty\); this gives an arbitrarily
large cost for a fixed positive empirical grazing frequency.  These two
bounds give exponential tightness in the cylinder topology augmented by the
mark moment.

Finally let \(\widehat\nu\) be invariant with finite rate.  Restrict it to
\(\mathcal A_{L,K}\), and replace each discarded excursion by the fixed
connector word having the same entrance and exit components.  The discarded
frequency and mark mass tend to zero.  Summable variations make the potential
and entropy/free-energy costs of the connector insertions tend to zero as
well.  The resulting invariant finite-state phases
\(\widehat\nu_{L,K}\) converge to \(\widehat\nu\) on every cylinder and
satisfy
\[
 \widehat{\mathcal I}_{L,K}(\widehat\nu_{L,K})
 \longrightarrow\widehat{\mathcal I}(\widehat\nu).
\]
The finite-state LDP lower bound followed by this diagonal approximation gives
the lower bound at every finite-rate law.  The projective upper bound and
exponential tightness give the full upper bound.  Compactness of rate
sublevels and the displayed pressure duality follow from the same two tail
estimates.
\end{proof}

\subsection{A quantitative singularity shield}

Fix a collision-window radius \(q\), a return cutoff \(L\), and a
homogeneity cutoff \(K\).  Let \(\mathcal S_{q,L,K}\) be the union, over
retained branches, of singularity curves met by the expanded window.  The
finite-cut complexity and strip-mass bounds give, for fixed \(c_0\),
\[
 \#\mathcal S_{q,L,K}\le CLK^{c_0}e^{\kappa q},
 \qquad
 \widehat{\mathbb P}\big(N_\delta(\mathcal S_{q,L,K})\big)
 \le CLK^{c_0}e^{\kappa q}\delta^\alpha .
\]

\begin{lemma}[Exponential frequency ledger]
\label{lem:r3-a3-frequency}
For every \(\epsilon>0\),
\[
 \lim_{q,L,K\to\infty}
 \lim_{\delta\downarrow0\atop LK^{c_0}e^{\kappa q}\delta^\alpha\downarrow0}
 \limsup_{N\to\infty}\frac1N
 \log\widehat{\mathbb P}\left(
 \widehat L_N(N_\delta(\mathcal S_{q,L,K}))>\epsilon
 \right)=-\infty.
\]
In addition,
\[
 \lim_{L\to\infty}\limsup_{N\to\infty}
 \frac1N\log\widehat{\mathbb P}\left(
 \frac1N\sum_{j<N}r_j1_{\{r_j>L\}}>\epsilon
 \right)=-\infty,
\]
and the same assertion holds with \(r\) replaced by \(\mathfrak t\).
If \(h_j\) is the maximal homogeneity index of block \(j\), then
\[
 \lim_{K\to\infty}\limsup_{N\to\infty}\frac1N
 \log\widehat{\mathbb P}\left(
 \frac1N\sum_{j<N}1_{\{h_j>K\}}>\epsilon
 \right)=-\infty.
\]
\end{lemma}

\begin{proof}
For a bounded indicator approximation \(b_{q,L,K,\delta}\), the pressure
increment of the twisted operator satisfies
\[
 P(\widehat\phi+s b_{q,L,K,\delta})-P(\widehat\phi)
 \le C(e^s-1)LK^{c_0}e^{\kappa q}\delta^\alpha.
\]
This follows by expanding the transfer operator once at every visit to the
bad set and using bounded distortion to replace conditional branch mass by
its Gibbs mass.  Chernoff's inequality gives
\[
 -s\epsilon+C(e^s-1)LK^{c_0}e^{\kappa q}\delta^\alpha.
\]
Choose \(s=\frac12\log((LK^{c_0}e^{\kappa q}\delta^\alpha)^{-1})\) and then take the
ordered limit.  The mark statements are the exponential-tilt argument used in
Theorem~\ref{thm:r3-a3-block}.  For high homogeneity, use the
fixed-\(s\) pressure convergence for \(s1_{\{h>K\}}\) proved in that theorem;
Chernoff first sends \(K\to\infty\) and then \(s\to\infty\).
\end{proof}

On the complement of the bad set, expanding a block, choosing a collision or
physical root, and evaluating any fixed path window are uniformly continuous.
Let \(\mathscr S_{c,q,L,K,\delta}\) and
\(\mathscr S_{p,q,L,K,\delta}\) be the corresponding continuous Palm maps.

\begin{theorem}[Exponentially good collision and physical factors]
\label{thm:r3-a3-factor}
For every metric generating weak convergence on collision or physical path
probabilities and every \(\epsilon>0\),
\[
 \lim_{q,L,K\to\infty}
 \lim_{\delta\downarrow0\atop LK^{c_0}e^{\kappa q}\delta^\alpha\downarrow0}
 \limsup_{N\to\infty}\frac1N\log
 \widehat{\mathbb P}\left(
 d(\mathscr S_\bullet\widehat L_N,
   \mathscr S_{\bullet,q,L,K,\delta}\widehat L_N)>\epsilon
 \right)=-\infty
\]
for \(\bullet=c,p\).
\end{theorem}

\begin{proof}
A discrepancy in one of the first \(q\) path coordinates is caused by a long
block, a visit to the singularity neighborhood, or an initial/final residual
block.  The first two are controlled by Lemma~\ref{lem:r3-a3-frequency}.  The
residual carries at most \(2q\) collision units or \(2q\tau_+\) physical
units; after normalization its contribution vanishes exponentially well
because the total clock has a positive linear lower tail.  The tail of the
weak metric is made smaller than \(\epsilon\) by choosing finitely many path
tests.  This proves the assertion.
\end{proof}

\subsection{Two random clocks and the lower-bound surgery}

For an invariant block law \(\widehat\nu\), define
\[
 \int F\,d\mathscr S_c\widehat\nu
 =\frac{1}{\widehat\nu(r)}
 \int\sum_{j=0}^{r-1}F(\Theta^j\Gamma^c\omega)
 \,d\widehat\nu,
\]
and
\[
 \int F\,d\mathscr S_p\widehat\nu
 =\frac{1}{\widehat\nu(\mathfrak t)}
 \int_0^{\mathfrak t(\omega)}F(\Theta_t\Gamma^p\omega)
 \,dt\,d\widehat\nu.
\]

\begin{lemma}[Terminal clock surgery]
\label{lem:r3-a3-surgery}
Let \(\widehat\nu\) be ergodic with finite exponential mark moment.  For every
weak neighborhood \(U\) of \(\widehat\nu\) and every \(o(n)\) tolerance,
there are block words whose empirical block measure lies in \(U\), whose
collision total straddles \(n\) with an overshoot \(o(n)\), and whose physical
total straddles a prescribed \(T\) with an overshoot \(o(T)\).  The connector
and terminal modifications use \(o(n)\) symbols and have logarithmic Gibbs
cost \(o(n)\).
\end{lemma}

\begin{proof}
Choose a large mark cutoff \(L\) for which the \(\widehat\nu\)-mass and mark
mass of discarded symbols are smaller than the tolerance.  A typical word of
length \(N\) then has block empirical law in \(U\) and clock totals
\(N\widehat\nu(r)+o(N)\) and
\(N\widehat\nu(\mathfrak t)+o(N)\).  Select
\(N=\lfloor n/\widehat\nu(r)\rfloor\), or the analogous physical value.
The finite connector set joins the word to a fixed positive-mass return loop.
Repeat or delete that bounded loop until the target is crossed, and stop
inside the last block.  At most \(O(L)+o(n)\) collision units and physical time
are changed.  The Gibbs distortion of connectors and terminal loops is
bounded per added symbol, so their logarithmic cost is \(o(n)\).
\end{proof}

\begin{theorem}[Full collision- and physical-time empirical-path LDP]
\label{thm:r3-a3-two-clock}
The collision empirical path measures satisfy a full weak LDP with speed
\(n\) and good rate
\[
 \mathcal I^c(\nu)
 =\inf_{\mathscr S_c\widehat\nu=\nu}
 \frac{\widehat{\mathcal I}(\widehat\nu)}
      {\widehat\nu(r)}.
\]
The physical empirical path measures satisfy a full weak LDP with speed
\(T\) and good rate
\[
 \mathcal I^p(\nu)
 =\inf_{\mathscr S_p\widehat\nu=\nu}
 \frac{\widehat{\mathcal I}(\widehat\nu)}
      {\widehat\nu(\mathfrak t)}.
\]
Both upper and lower bounds hold on the local Skorokhod weak topology.
\end{theorem}

\begin{proof}
First use bounded marks and the continuous shielded maps.  Partition the
possible empirical mean clock into intervals of width \(\epsilon\).  If a
block empirical law is close to \(\widehat\nu\), then the number of complete
blocks before collision time \(n\) equals
\(n/\widehat\nu(r)+O(\epsilon n)\).  The block LDP therefore assigns exponent
\(-n\widehat{\mathcal I}(\widehat\nu)/\widehat\nu(r)+O(\epsilon n)\).
A finite union over clock bins gives the upper bound.  Exponential mark
truncation and Theorem~\ref{thm:r3-a3-factor} remove the bounded-mark and
continuity restrictions.

For the lower bound, start with an ergodic \(\widehat\nu\) whose Palm image
lies in the desired open set.  The block LDP lower bound supplies typical
words at cost \(N\widehat{\mathcal I}(\widehat\nu)+o(N)\).  Apply
Lemma~\ref{lem:r3-a3-surgery}; the terminal modification has sublinear cost and
sublinear effect on the Palm empirical path.  Since
\(N=n/\widehat\nu(r)+o(n)\), this gives the collision lower exponent.  The
physical proof is identical with \(\mathfrak t\).  Ergodic decomposition and
finite convex approximation extend the lower bound to all invariant laws.
Goodness follows from goodness of \(\widehat{\mathcal I}\), uniform
integrability of both marks on its sublevels, and tightness of the shielded
Palm images.
\end{proof}

\begin{proposition}[Presentation independence]
\label{prop:r3-a3-independent}
The two rates in Theorem~\ref{thm:r3-a3-two-clock} do not depend on the chosen
finite-connector exponential Young presentation.
\end{proposition}

\begin{proof}
Given two presentations, refine their inducing partitions by the first common
return to the product magnet.  The common refinement has exponential return
marks and factors onto both codes.  Applying the two-clock contraction through
either factor gives the same microscopic collision or physical empirical path
law.  Uniqueness of an LDP rate on a Hausdorff space makes the two variational
formulas equal.
\end{proof}

\subsection{Finite-rate support and preparation}

Let \(\overline\Omega_{\rm det}\) be the graph completion which records the
incoming state and both one-sided continuations at a singular collision.

\begin{proposition}[Finite-rate deterministic support]
\label{prop:r3-a3-support}
If \(\mathcal I^c(\nu)<\infty\) or \(\mathcal I^p(\nu)<\infty\), then
\(\nu(\overline\Omega_{\rm det})=1\).  Every finite-rate law gives zero mass
to genuinely multivalued singular continuations.
\end{proposition}

\begin{proof}
Every shielded factor is supported on the closed graph completion.  The LDP
upper bound and exponential approximation make the rate infinite outside it.
If a law charged genuinely singular continuations, every sufficiently small
singularity neighborhood would have a fixed positive empirical frequency.
Lemma~\ref{lem:r3-a3-frequency} makes the cost of that event tend to infinity.
\end{proof}

Let \(C\) be a continuous affine finite-dimensional macro map on physical
path probabilities and let \(a\) be finite-rate feasible.

\begin{theorem}[Information projections and Gibbs conditioning]
\label{thm:r3-a3-projection}
The minimizer set
\[
 \mathcal M(a)=\operatorname*{argmin}_{C(\nu)=a}\mathcal I^p(\nu)
\]
is nonempty and compact.  If the minimizer is a unique exposed point, then
conditioning on shrinking continuity neighborhoods of \(a\) concentrates
exponentially at that law, including at a uniform interior time root.  Without
uniqueness, every subsequential conditioned limit is a probability mixture
supported on \(\mathcal M(a)\), and the phase label is retained.
\end{theorem}

\begin{proof}
A minimizing sequence lies in a compact rate sublevel; continuity of \(C\)
and lower semicontinuity give attainment.  Exposed uniqueness gives a positive
rate gap outside every neighborhood of the minimizer.  The LDP upper bound
controls the numerator and the lower bound at the minimizer controls the
conditioning denominator.  A uniform interior root changes only a vanishing
boundary fraction.  In the nonunique case the same rate-gap argument puts all
limit mass on the compact minimizer set; disintegration over that set gives
the phase mixture.
\end{proof}

\subsection{The closed path cotangent quotient}

Let \(\mathscr A_\beta\) be the weighted dynamically H\"older algebra on
physical paths with norm
\[
 \|F\|_\beta=\|F/(1+W)\|_\infty+[F]_{\rm dyn,\beta},
\]
where \(W\) is the exponential clock/velocity weight.  Define the closed
linear subspace
\[
 \mathscr N_\beta
 =\overline{\operatorname{span}}^{\mathscr A_\beta}
 \left(
 \mathbb R\mathbf1+
 \{U-U\circ\Theta_t:
 U\in\mathscr A_\beta,\ t\in\mathbb Q\}
 \right).
\]

\begin{theorem}[Hausdorff cotangents and annihilator]
\label{thm:r3-a3-cotangent}
The quotient \(\mathscr A_\beta/\mathscr N_\beta\) is a Hausdorff Banach
space.  Every invariant finite-weight path law annihilates the coboundary
part.  Conversely, if a continuous linear functional on
\(\mathscr A_\beta\) annihilates all invariant finite-weight probabilities,
then it lies in the annihilator generated by \(\mathscr N_\beta\).  Hence two
potentials define the same pressure tangent on all invariant phases exactly
when they determine the same quotient class.
\end{theorem}

\begin{proof}
Closedness makes the quotient Hausdorff and complete.  Invariance gives
\(\nu(U-U\circ\Theta_t)=0\).  For the converse, if a function class were not
in \(\mathscr N_\beta\), Hahn--Banach would separate it from that closed
subspace.  Averaging the separating functional over rational time shifts and
using the exponential weight produces an invariant continuous functional;
its positive and negative Jordan parts normalize to invariant finite-weight
measures, contradicting the assumed annihilation.  Thus the closed
coboundary space is exactly the kernel seen by invariant phases.
\end{proof}

\begin{theorem}[Round-three A3 closure]
\label{thm:r3-a3-main}
The periodic finite-horizon Sinai billiard satisfies full weak empirical-path
LDPs in collision and physical time with the good rates of
Theorem~\ref{thm:r3-a3-two-clock}.  The rates are presentation independent,
finite only on deterministic graph-completed paths, support compact
information projections, and have cotangents in the Hausdorff quotient
\(\mathscr A_\beta/\mathscr N_\beta\).
\end{theorem}

\begin{proof}
Combine Theorems~\ref{thm:r3-a3-block},
\ref{thm:r3-a3-factor}, \ref{thm:r3-a3-two-clock},
\ref{thm:r3-a3-projection}, and \ref{thm:r3-a3-cotangent}, together with
Propositions~\ref{prop:r3-a3-independent} and
\ref{prop:r3-a3-support}.
\end{proof}
