\section{Round-thirty-one reconstruction: transition-clock entropy and two-scale chronological currents}
\label{sec:r31-a3}

The controlled entropy is charged once per selected excursion, not once per
unit of holding time.  The state therefore contains both a transition-count
measure and a holding occupation, together with their exact renewal relation.
A mesoscopic current retains the slow chronological derivative that disappears
from the leading fast balance.  Random stopping is handled by a terminal-state
Markov-renewal kernel; no unproved scalar independence of the overshoot is
used.

\subsection{Legal marked histories and two empirical measures}

Let $E$ be the Polish space of legal completed excursion marks
\[
 m=(R,\tau,\kappa,e^-,e^+,\mathcal G),                    \tag{A3.1}
\]
with $R\in\N$, $\tau>0$, $\kappa\in\Z^2$, and completed graph
$\mathcal G$.  Let $\mathsf H=E^{\Z_-}$ be the complete-past space,
$K(h,\dd m)$ the exact stable disintegration, and $U(h,m)$ the append--shift
map.  The A2 return estimates give, for suitable $0<\rho<1$,
\[
 W(h)=\exp\left\{\epsilon\left(1+
 \sum_{j\ge0}\rho^j\chi(m_{-j})\right)\right\},           \tag{A3.2}
\]
\[
 \int e^{\epsilon\chi(m)}K(h,\dd m)\le C W(h)^\zeta,
 \qquad \rho+\zeta<1.                                    \tag{A3.3}
\]
Hence $PW\le C W^{\rho+\zeta}$.

For a legal word $(m_1,\ldots,m_{\nu_N})$, put
$h_k=U(h_{k-1},m_k)$ and
\[
 s_{0,N}=0,\qquad s_{k,N}=N^{-1}\sum_{j\le k}R(m_j).       \tag{A3.4}
\]
The transition-count occupation and marked transition flow are
\[
 \nu_N={1\over N}\sum_{k\le\nu_N}
       \delta_{(s_{k-1,N},h_{k-1})},                      \tag{A3.5}
\]
\[
 Q_N={1\over N}\sum_{k\le\nu_N}
       \delta_{(s_{k-1,N},h_{k-1},m_k)}.                  \tag{A3.6}
\]
The holding occupation is the different measure
\[
 L_N(\dd s,\dd h)=
 \sum_{k\le\nu_N}
  1_{[s_{k-1,N},s_{k,N})}(s)\,\dd s\,\delta_{h_{k-1}}(\dd h).           \tag{A3.7}
\]
Thus $Q_N$ and $\nu_N$ charge one unit $1/N$ per transition, while $L_N$
charges $R(m_k)/N$ units of the scaled chronological clock.

\begin{proposition}[Exact renewal relation]
\label{prop:r31-a3-renewal-relation}
For every bounded continuous $g(s,h)$,
\begin{equation}\tag{A3.8}
 \int g\,\dd L_N
 =\int\left[\int_0^{R(m)/N}g(s+r,h)\,\dd r\right]
    NQ_N(\dd s,\dd h,\dd m).
\end{equation}
In particular, if $Q_N\Rightarrow Q$ with uniformly integrable return length,
then
\[
 L(\dd s,\dd h)=\int R(m)Q(\dd s,\dd h,\dd m).            \tag{A3.9}
\]
The transition marginal of $Q$ is $\nu$, not $L$.
\end{proposition}

\begin{proof}
The $k$th term on the right of (A3.8) is precisely
$\int_{s_{k-1,N}}^{s_{k,N}}g(s,h_{k-1})\dd s$.  Summation gives (A3.7).
Uniform integrability from (A3.3) permits passage to (A3.9).
\end{proof}

\subsection{Exact entropy representation}

A predictable controlled mark kernel has the form
$q_{s,h}(m)K(h,\dd m)$ with
$\int q_{s,h}\dd K=1$.  Its one-transition relative entropy is
\[
 \operatorname{Ent}(q_{s,h}K\mid K)
 =\int q_{s,h}\log q_{s,h}\,K(h,\dd m).                  \tag{A3.10}
\]

\begin{theorem}[Transition-count entropy action]
\label{thm:r31-a3-action}
For controlled words stopped at $\nu_N$, the normalized path entropy is
\[
 {1\over N}E^q\sum_{k\le\nu_N}
 \operatorname{Ent}(q_{s_{k-1,N},h_{k-1}}K\mid K).        \tag{A3.11}
\]
Every weak limit has action
\[
 \cA(Q)=
 \int q_{s,h}(m)\log q_{s,h}(m)
       K(h,\dd m)\nu(\dd s,\dd h),                       \tag{A3.12}
\]
where
$Q(\dd s,\dd h,\dd m)=q_{s,h}(m)K(h,\dd m)
u(\dd s,\dd h)$.
The holding occupation enters only through (A3.9).  If $R\equiv r$, then
$\nu([0,1]\times\mathsf H)=1/r$ and (A3.12) equals the exact entropy
$H/r$, not $H$.
\end{theorem}

\begin{proof}
Apply the chain rule for relative entropy to the successive conditional mark
laws.  Each selected mark contributes once, giving (A3.11).  The empirical
measure for this sum is (A3.5), so lower semicontinuity of relative entropy
and the conditional variational formula give (A3.12).  Recovery uses
conditional expectations of $q$ on finite partitions and preserves the same
transition count.  Proposition~\ref{prop:r31-a3-renewal-relation} determines
$L$ afterward; replacing $\nu$ by $L$ would incorrectly multiply each cost by
its return length.
\end{proof}

\subsection{Fast balance and slow chronological divergence}

For a cylindrical $C^1$ test $\varphi(s,h)$ define
$\Delta_m\varphi(s,h)=\varphi(s,U(h,m))-\varphi(s,h)$.
The exact word identity is
\begin{equation}\tag{A3.13}
\begin{split}
 {\varphi(s_{\nu_N,N},h_{\nu_N})-\varphi(0,h_0)\over N}
 ={}&\int\Delta_m\varphi(s,h)\,Q_N(\dd s,\dd h,\dd m)\\
 &+{1\over N}\int\partial_s\varphi(s,h)\,L_N(\dd s,\dd h)
 +\epsilon_N(\varphi),
\end{split}
\end{equation}
with $|\epsilon_N(\varphi)|\le C_\varphi/N$.  At leading order this yields
the fast stationarity relation $\int\Delta_m\varphi\dd Q=0$.

To retain the slow derivative, choose $\delta_N\downarrow0$ with
$N\delta_N\to\infty$ and let $\rho$ be a smooth unit-mass kernel.  Define the
mesoscopic incoming and outgoing history measures
\[
 \nu_N^\pm(t,\dd h)=
 {1\over N\delta_N}\sum_k
 \rho\left({t-s_{k-1,N}\over\delta_N}\right)
 \delta_{h_{k-1\pm1}}(\dd h),                             \tag{A3.14}
\]
with the natural endpoint convention, and the divergence current
\[
 \cJ_N(t)=N\delta_N\,[\nu_N^+(t)-\nu_N^-(t)].             \tag{A3.15}
\]

\begin{theorem}[Two-scale chronological current]
\label{thm:r31-a3-two-scale}
On every action sublevel, $(Q_N,L_N,\cJ_N)$ is tight in the product of the
weak marked-flow topology, the holding topology, and
$H^{-1}_{\mathrm{loc}}((0,T);\mathsf H')$.  Every limit satisfies
\[
 Q(\dd s,\dd h,\dd m)=q_{s,h}(m)K(h,\dd m)\nu(\dd s,\dd h),             \tag{A3.16}
\]
\[
 \int\Delta_m\phi(h)\,Q_s(\dd h,\dd m)=0
 \quad\text{for a.e. }s,                                 \tag{A3.17}
\]
\[
 \partial_sL+\operatorname{div}_{U}\cJ
 =\delta_{(0,h_0)}-\delta_{(T,h_T)}                       \tag{A3.18}
\]
in cylindrical distributions, together with the renewal relation (A3.9).
Conversely, every finite-action state satisfying (A3.16)--(A3.18) is obtained
by legal finite-memory recovery words.
\end{theorem}

\begin{proof}
Equation (A3.13) gives (A3.17).  Apply the same telescoping identity after
multiplying by the translated kernel
$\rho((t-s)/\delta_N)$ and summing $t$ against a smooth test.  Discrete
summation by parts produces (A3.18); the error is
$O((N\delta_N)^{-1}+\delta_N)$ and vanishes.  The fractional exponential
moment controls return lengths and completed-graph variation, yielding
compactness.  For recovery, approximate $q$, $L$, and $\cJ$ on a finite
space-time/cylinder partition, use conditional-reference mark laws, and join
blocks with the regular connector library.  The number of connectors is
$o(N)$, while their exact endpoint increments realize the discrete divergence
of the partitioned current.  Refinement gives all three relations.
\end{proof}

\subsection{Two graph clocks and recession}

Let $C_N=\sum_{k\le\nu_N}R(m_k)$ and
$T_N=\sum_{k\le\nu_N}\tau(m_k)$.  The collision and physical completed paths
are separately normalized by $C_N$ and $T_N$; neither is evaluated on the
other clock.  For rational $r>0$, marks with $\chi(m)>rN$ enter a radial
recession measure weighted by $\chi(m)/N$, and are removed from the recurrent
part.  The entropy inequality based on (A3.3) gives
\[
 \sup_N E^q{1\over N}\sum_{k\le\nu_N}\chi(m_k)<\infty,   \tag{A3.19}
\]
so the projective recession and terminal-mark compactifications have bounded
mass.

\begin{theorem}[Chronological two-clock Laplace principle]
\label{thm:r31-a3-ldp}
The state consisting of
\[
 (Q_N,\nu_N,L_N,\cJ_N,\Gamma_N^{\rm col},
  \Gamma_N^{\rm phys},C_N/N,T_N/N,\cR_N)                 \tag{A3.20}
\]
satisfies a good Laplace principle at speed $N$.  Its rate is (A3.12) on the
closed constraint set (A3.9), (A3.16)--(A3.18), graph compatibility, and
projective recession consistency, and is $+\infty$ otherwise.
\end{theorem}

\begin{proof}
The conditional entropy variational formula gives the upper bound on finite
partitions.  Equations (A3.3) and (A3.19) give controlled compactness.  The
recovery part of Theorem~\ref{thm:r31-a3-two-scale} gives the lower bound with
exact legal marks and exact chronological divergence.  Refinement and the
Dawson--Gartner theorem assemble the projective coordinates.  The rate uses
the transition marginal throughout.
\end{proof}

\subsection{Terminal-state Markov-renewal local theorem}

Let
\[
 \nu_N^*=\inf\{k:S_kR\ge N\},\qquad O_N=S_{\nu_N^*}R-N,   \tag{A3.21}
\]
and let
$X_N=(\nu_N^*,S_{\nu_N^*}\kappa,S_{\nu_N^*}\tau)$.
Introduce the residual-clock transfer operator on
$\mathsf H\times\N_0$,
\[
 (\mathscr T_{u,b,z}F)(h,r)
 =\int e^{iu\cdot\kappa(m)+ib\cdot\tau(m)}
 F(U(h,m),r-R(m))1_{\{R(m)\le r\}}K(h,\dd m),             \tag{A3.22}
\]
and the terminal kernel for $R(m)>r$.  The exact A2 arithmetic acts on the
enlarged terminal-state operator and excludes every peripheral pole except the
simple renewal root.

\begin{theorem}[Matrix-amplitude stopped local theorem]
\label{thm:r31-a3-stopped}
For central $x_N$ and fixed overshoot $o$, terminal cylinder $D$, and bounded
finite-memory source $H$,
\begin{equation}\tag{A3.23}
 P_H\{X_N=x_N,O_N=o,h_{\nu_N^*}\in D\}
 =N^{-d/2}\phi_{\Sigma_H}
   \left({x_N-Nm_H\over\sqrt N}\right)
 \left[a_H(o,D;x_N/\sqrt N)+O(N^{-1/2})\right].
\end{equation}
The nonnegative terminal amplitude satisfies
\[
 \sum_{o\ge0}a_H(o,\mathsf H;z)=1,\qquad
 a_H(o,\mathsf H;z)\le Ce^{-co},                          \tag{A3.24}
\]
and is $C^2$ in the central argument and source.  A scalar product
$\rho_H(o)\phi_{\Sigma_H}$ is asserted only when the terminal operator is
proved rank one independently of the central Fourier direction.
\end{theorem}

\begin{proof}
Apply Fourier--Laplace inversion to the block operator made of
$\mathscr T_{u,b,z}$ and the terminal kernel.  The simple renewal pole gives
the Gaussian factor; its left/right eigenvectors and the terminal operator
give the amplitude $a_H(o,D;\cdot)$.  The A2 noncentral bound is integrable
with two source derivatives and controls the complement.  Exponential return
tails give (A3.24).  This keeps the possible dependence of overshoot and
terminal state on the central coordinate instead of assuming asymptotic
independence.
\end{proof}

\begin{corollary}[Exact conditioned path principle]
\label{cor:r31-a3-conditioned}
Regular conditioning on $X_N=x_N$, a specified terminal cylinder, and either
a fixed or summed overshoot gives the contraction of
Theorem~\ref{thm:r31-a3-ldp} to that coarea/lattice fibre, with its minimum
subtracted.
\end{corollary}

\begin{proof}
Use (A3.23) separately for numerator and denominator.  The same nonzero
terminal matrix amplitude occurs at the fixed conditioning datum and cancels.
Projective compactness extends finite-memory sources to bounded continuous
functionals.
\end{proof}

The corrected theory now uses the exact entropy clock, distinguishes the fast
balance from the slow chronological divergence, and treats the overshoot by
the terminal-state renewal operator that actually produces it.
