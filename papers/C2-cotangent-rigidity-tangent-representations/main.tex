    \begin{abstract}
    We organize the preceding papers as one path-space cotangent theory.  The full
Liouville empirical-path rate contracts to homological transport, scalar
currents, physical clocks, impact statistics, and any bounded continuous
mechanical path constraint.  Heat-bath and entropy-production theories are
obtained after augmenting the mechanical platform, not by changing the
variational principle.

On exposed branches, pressure derivatives are tangent path laws and integrated
covariances; at phase coexistence they become compact subdifferential sets.
Cash additivity and strong time consistency select the scalar entropic family,
while the mechanical information projection fixes its parameter on calibrated
rays.  A universal finite-horizon likelihood-ratio martingale contracts to the
usual diffusive Girsanov and quadratic BSDE representations.  The exact
Mori--Zwanzig memory equation and the history nonlinear semigroup are shown to
be two representations of the same prepared deterministic path law.
    \end{abstract}

\section{Universal contractions of the full rate}
Let \(\cI_R^p\) be the good physical empirical-path rate.  For a continuous
affine map \(C:\mathcal P(\Omega_R^p)\to E\) into a finite-dimensional space,
\[
 I_C(a)=\inf\{\cI_R^p(\nu):C(\nu)=a\}.
\]

\begin{theorem}[All finite mechanical contractions]\label{thm:contractions}
The same rate functional produces by contraction:
\begin{enumerate}[label=(\roman*)]
\item the v9 full homology current and joint current--clock pressure;
\item every scalar lattice-direction current, including v8;
\item collision activity, occupation, roof, and bounded path work;
\item the v7 impact-frequency preparation after changing to the open-billiard
      platform;
\item the v6 exact branch-current model after changing to its Lebesgue platform.
\end{enumerate}
Energy transfer and entropy production are obtained from the identical
construction on an augmented closed mechanical platform.  They are not
silently asserted for the fixed-energy single-particle table.
\end{theorem}
\begin{proof}
Apply the contraction principle.  The parent rate changes with the physical
platform, while the variational mechanism does not.
\end{proof}

\begin{proposition}[Coboundary quotient]
If \(F-G=c+U-U\circ\Theta_t\), then their long-time pressures differ by
\(c\), their information projections coincide, and they define the same
cotangent direction in \(\mathscr V_R\).
\end{proposition}

\section{Tangent path laws and phase subdifferentials}
At a potential \(\Psi\) with a unique maximizing law \(\Pp_R^\Psi\), define
\(Q_R(F)\) as in Paper III.

\begin{theorem}[First and second pressure variations]\label{thm:tangent}
For bounded regular path directions \(G,H\),
\[
 DQ_R(\Psi)G=\int Gd\Pp_R^\Psi,
\]
and whenever the covariance series converges absolutely,
\[
 D^2Q_R(\Psi)(G,H)
 =\int_{\R}\Cov_{\Pp_R^\Psi}
 (G,H\circ\Theta_t)\,dt
\]
in continuous time, with the bilateral sum in collision time.  Conditional
differentiation of Paper III's tower gives a tangent-kernel cocycle.
\end{theorem}
\begin{proof}
Differentiate the variational pressure on its exposed smooth branch; the
linear response of the equilibrium law gives the integrated covariance.  The
conditional cocycle follows from the chain rule for exact time consistency.
\end{proof}

At a nonsmooth potential,
\[
 \partial Q_R(\Psi)=\cM(\Psi)
\]
is a compact set of path phases.  No single tangent probability or covariance
is selected without additional phase preparation.

\section{Scalar entropic rigidity and mechanical calibration}
Let \(\mathfrak C\) be a monotone, cash-additive, strictly relevant, strongly
time-consistent scalar certainty equivalent generated by one smooth utility
function.

\begin{theorem}[Entropic rigidity with path calibration]\label{thm:rigidity}
The certainty equivalent is either linear expectation or
\[
 \mathfrak C(X)=\frac1\vartheta\log\E e^{\vartheta X}
\]
for a constant \(\vartheta\ne0\).  The axioms do not determine
\(\vartheta\).  If the macro preparation lies on a calibrated cotangent ray
\(\Psi=\theta G\), and terminal work is measured in the dual units of \(G\),
then the first-principles branch has \(\vartheta=\theta\).  Other values are
external valuation parameters.
\end{theorem}
\begin{proof}
Cash additivity yields the translation functional equation for the utility;
smoothness forces constant absolute risk aversion, hence affine or
exponential form.  Strong time consistency fixes the parameter through time
\cite{KupperSchachermayer2009}.  Paper I supplies the mechanical cotangent.
\end{proof}

\section{Universal likelihood ratios and diffusive representations}
For a prepared phase \(\Pp_R^\Psi\) and bounded terminal path work \(F\), put
\[
 M_t=
 \frac{\E_{\Pp_R^\Psi}[e^F\mid\mathcal F_t]}
 {\E_{\Pp_R^\Psi}e^F}.
\]

\begin{theorem}[Path-space tangent density]\label{thm:density}
The process \(M_t\) is a uniformly integrable martingale and defines the
finite-horizon tilted path law.  It is the universal tangent likelihood ratio
of the excess-pressure semigroup.  After a scalar diffusive contraction with
\(F=\theta\Phi(X_T)\), it becomes the stochastic exponential
\[
 M=\mathcal E\left(
 \int\theta\sigma^\top Du\,dW\right),
\]
producing the standard Girsanov drift shift and the quadratic BSDE
\[
 Y_t=\Phi(X_T)+\int_t^T\frac\theta2\abs{Z_s}^2ds-
 \int_t^TZ_sdW_s.
\]
\end{theorem}
\begin{proof}
The first assertion is the conditional-expectation martingale identity.  The
diffusive representation follows from Ito's formula applied to the scalar
entropic contraction, followed by Girsanov and the usual BSDE identification
\cite{PardouxPeng1990,Kobylanski2000}.
\end{proof}

\section{Memory and nonlinear pressure are compatible representations}
Let \(A\) be a resolved observable.  Paper II gives its exact
domain-safe Mori--Zwanzig family under the prepared law; Paper III gives the exact
conditional pressure semigroup on the full resolved history.

\begin{theorem}[Representation compatibility]\label{thm:compatibility}
The regularized Mori--Zwanzig memory family, any verified orthogonal force,
history transition kernel,
and excess-pressure tangent density are all constructed from the same
stationary prepared path law.  Changing the preparation changes all four
objects coherently.  A Markov diffusion/BSDE representation is valid only
after a proved short-memory/homogenization contraction.
\end{theorem}
\begin{proof}
Each object is a conditional or operator-theoretic construction on
\(L^2(\Pp_R^\Psi)\) or its path filtration.  Paper II's preparation-first
projection rule and Paper III's exact history tower identify the common law.
\end{proof}

\begin{theorem}[Cotangent theory closure]\label{thm:main}
One full Liouville empirical-path rate simultaneously yields information
projections, path cotangents, universal excess pressure, exact memory-aware
dynamics, scalar entropic rigidity, tangent path laws, phase
subdifferentials, and the usual Girsanov/BSDE representations after justified
contraction.
\end{theorem}
\begin{proof}
Combine \cref{thm:contractions,thm:tangent,thm:rigidity,thm:density,thm:compatibility}.
\end{proof}
