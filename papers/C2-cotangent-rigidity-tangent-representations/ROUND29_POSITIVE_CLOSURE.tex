\section{Round-twenty-nine reconstruction: weighted strict duality without local compactness and channel-specific likelihood calculus}
\label{sec:r29-c2}

The functional-analytic statement is now valid on an arbitrary Polish space:
no compact exhaustion and no continuous proper weight are assumed.  The
kinetic annihilator retains the full B3 contact space and therefore forces the
contact cotangent to annihilate physical cycle directions.  Prediction lives
in one fixed stopped-path space, and discrete, point-process, and Brownian
observation likelihoods are treated by their own stochastic calculi.

\subsection{Weighted mixed strict topology on a Polish space}

Let $E$ be Polish and let $W:E\to[1,\infty)$ be positive and continuous; its
sublevels need not be compact.  Define
\[
 C_W(E)=\{f\in C(E):f/W\in C_b(E)\}.                      \tag{C2.1}
\]
For an arbitrary sequence of compact sets $(K_n)$ and numbers
$a_n\downarrow0$, put
\[
 p_{K,a}(g)=\sup_{n\ge1}a_n\sup_{x\in K_n}|g(x)|.          \tag{C2.2}
\]
The mixed strict topology $\beta_0$ on $C_b(E)$ is generated by all seminorms
(C2.2), together with uniform convergence on each fixed compact set.  No
single sequence $(K_n)$ is required to exhaust $E$.  Let
$Tf=f/W$ and define $\beta_W=T^{-1}\beta_0$.

\begin{theorem}[Weighted Polish strict dual]
\label{thm:r29-c2-dual}
The continuous dual of $(C_W(E),\beta_W)$ is
\[
 \mathcal M_W(E)=\left\{\mu\text{ finite signed Radon}:
                         \int W\,d|\mu|<\infty\right\}.    \tag{C2.3}
\]
A family is equicontinuous if and only if its weighted total variations are
uniformly bounded and it is uniformly $W$-tight:
\[
 \sup_{\mu}\int_{K^c}W\,d|\mu|\longrightarrow0
 \quad\text{along a directed family of compact }K.        \tag{C2.4}
\]
The theorem applies to non-locally-compact path and measure spaces.
\end{theorem}

\begin{proof}
The map $T$ is a linear homeomorphism onto $(C_b(E),\beta_0)$.  For Polish
$E$, every $\beta_0$-continuous functional is integration against a finite
Radon measure: restriction to each compact gives a Riesz measure, the family
is compatible, and the seminorms (C2.2) give tightness needed to glue it.  If
$\nu$ represents the functional on $C_b(E)$, then
$d\mu=W^{-1}d\nu$ represents it on $C_W(E)$ and
$\int Wd|\mu|=|\nu|(E)$.  The converse follows immediately from compact
truncation and (C2.4).  This argument uses arbitrary compact sets, not local
compactness or a compact exhaustion.
\end{proof}

\subsection{The full kinetic annihilator}

Use the B3 tangent pair $(u,h)$ with
$h=h_{\rm n}+h_{\rm cyc}$ in
$\mathcal K^\perp\oplus\mathcal K$.  The balance operator is
\[
 \mathfrak B(u,h)=
 (R(u,h_{\rm n}),u(0)),\qquad
 R=\partial_tu+v\cdot\nabla_xu-DQ_f[u]-\mathcal Ch_{\rm n}.              \tag{C2.5}
\]
A source triple $(\zeta_f,\zeta_c,\zeta_0)$ pairs with both normal and cycle
contact tangents.

\begin{theorem}[Complete full-contact annihilator]
\label{thm:r29-c2-annihilator}
A source triple annihilates the homogeneous tangent kernel of (C2.5) if and
only if, modulo the five collision-invariant gauges,
\[
 \zeta_c\perp\ker\mathcal C,
 \qquad
 \zeta_f=-\partial_tr-v\cdot\nabla_xr-\mathcal L_f^*r,
 \qquad
 \zeta_c=-\Delta r,
 \qquad
 \zeta_0=r_0-r(0),\quad r(T)=0.                           \tag{C2.6}
\]
The gauge-fixed pair $(r,r_0)$ is unique.
\end{theorem}

\begin{proof}
Cycle tangents $(0,h_{\rm cyc})$ belong to the homogeneous kernel, so
annihilation first forces $\zeta_c\perp\ker\mathcal C$.  On the remaining
normal graph, B3 gives closed range.  The Hilbert identity
$(\ker\mathfrak B)^\perp=\operatorname{Ran}\mathfrak B^*$ and integration by
parts yield
\[
 \int_0^T\ip{r}{R(u,h_{\rm n})}\,dt+\ip{r_0}{u(0)}
 =\int\ip{-\partial_tr-v\cdot\nabla_xr-\mathcal L_f^*r}{u}
   -\int\ip{\Delta r}{h_{\rm n}}
   +\ip{r(T)}{u(T)}+\ip{r_0-r(0)}{u(0)}.
\]
This is exactly (C2.6).  Terminal and invariant gauges give uniqueness.
\end{proof}

\subsection{Pressure rigidity through finite-amplitude charts}

Let $\Sigma$ be the A4 history shift.  For a regular periodic orbit
$\mathcal O(p)$ let
$d_p(x)=1\wedge d_\Sigma(x,\mathcal O(p))^\alpha$.  Let $\chi_m$ be a bounded
Holder tail cutoff and define, for finite $t,R$,
\[
 V_{t,m,R}=-t d_p-R\chi_m.                                \tag{C2.7}
\]
Every such potential is bounded.  A4 supplies an analytic pressure chart
around each finite-amplitude potential; overlapping charts give continuation
along the finite path from zero to $V_{t,m,R}$.

\begin{lemma}[Existence, tightness, and orbit localization]
\label{lem:r29-c2-localize}
For each finite $(t,m,R)$, the A4 Riesz projection gives a unique equilibrium
state $\mu_{t,m,R}$.  First choose $R=R_m\to\infty$ so that the Lyapunov drift
makes
$\mu_{t,m,R_m}\{W>2m\}\to0$, and then let $t\to\infty$.  Every limit is the
uniform invariant measure on $\mathcal O(p)$.
\end{lemma}

\begin{proof}
The bounded Feynman--Kac operator is positive and has a simple leading
projection by A4, giving existence and uniqueness.  The tilted drift estimate
bounds the tail probability by the exponentially small cost of the cutoff.
On a fixed Lyapunov sublevel, compactness follows from the history metric.
Every invariant measure not supported on the orbit has positive integral of
$d_p$; the variational principle then makes its weight exponentially smaller
as $t\to\infty$.  The only invariant probability on one periodic orbit is the
uniform measure.
\end{proof}

\begin{theorem}[Pressure-to-Livsic rigidity]
\label{thm:r29-c2-rigidity}
Let $\phi$ be a Holder cocycle.  If two pressure functionals agree along every
finite path of potentials (C2.7) and on a Holder neighbourhood in the
$\phi$ direction at each point, then every periodic sum of $\phi$ vanishes and
\[
 \phi=u-u\circ\sigma                                    \tag{C2.8}
\]
for a Holder transfer $u$.
\end{theorem}

\begin{proof}
Differentiate the pressure identity in the $\phi$ direction at each finite
$V_{t,m,R}$.  The derivative is
$\int\phi d\mu_{t,m,R}$.  Lemma~\ref{lem:r29-c2-localize} gives zero average
on every regular periodic orbit.  The standard connector construction defines
$u$ by path sums from a fixed cylinder; periodic cancellation makes it
well-defined, and exponential variation makes it Holder.
\end{proof}

\subsection{One fixed stopped-path prediction space}

Fix the horizon $T$ and let $\mathbb D_T=D([0,T];E_X)$.  Define the stopping
map
\[
 (S_tx)(s)=x(s\wedge t),\qquad 0\le s\le T.               \tag{C2.9}
\]
Thus every stopped path belongs to the same Polish space $\mathbb D_T$.  The
prediction state is
\[
 \bar\Pi_t=\mathcal L(S_tX,X_t\mid\mathcal F_t^Y)          \tag{C2.10}
\]
on $\mathbb D_T\times E_X$.  It is augmented by the continuation kernel
$K_{t,T}((S_tx,x_t),d\widetilde x)$ supplied by the hidden Markov model.

\begin{theorem}[Optional projections of path functionals]
\label{thm:r29-c2-optional}
Suppose the hidden/observation kernels converge on a countable cylinder core,
the A3/B3 path laws are tight, and the one-step or compensator characteristics
converge with a uniform $2+\delta$ moment.  Then for every bounded continuous
$F:\mathbb D_T\to\R$,
\[
 E[F(X^n)\mid\mathcal F_t^{Y^n}]
 \Longrightarrow E[F(X)\mid\mathcal F_t^Y]               \tag{C2.11}
\]
as a process.  Predictable brackets are derived from the convergent primitive
kernels/compensators; they are not assumed and then restated.
\end{theorem}

\begin{proof}
For cylinder $F$, the optional projection is the integral of $F$ against
$\bar\Pi_t^nK_{t,T}^n$, a continuous functional away from the null set of
limiting jump times.  Core convergence identifies every subsequential filter
limit.  Conditional second moments computed from the primitive transition or
compensator kernels converge, which yields bracket convergence.  Uniform
$2+\delta$ moments give uniform tightness of the resulting martingales.
Finally, path tightness makes cylinder functions uniformly dense on compact
subsets of $\mathbb D_T$, proving (C2.11).
\end{proof}

\subsection{Likelihoods for three observation types}

\begin{proposition}[Discrete-time product likelihood]
\label{prop:r29-c2-discrete}
For the C1 discrete observation experiment and two parameters
$\vartheta,\vartheta_0$,
\[
 L_n=\prod_{k=1}^{n}
 {r_\vartheta(\Pi_{k-1},A_k,Y_k)
  \over r_{\vartheta_0}(\Pi_{k-1},A_k,Y_k)}               \tag{C2.12}
\]
is the likelihood martingale.  Its logarithm is a sum of increments and its
predictable quadratic characteristic is the sum of conditional score
covariances.  No Brownian representation is invoked.
\end{proposition}

\begin{proposition}[Marked-point likelihood]
\label{prop:r29-c2-point}
If observations form a marked point process with predictable intensities
$\lambda_t(dy)$ and $\lambda_t^0(dy)$, then
\[
 L_t=\mathcal E\left(\int_0^t\int
   \left({d\lambda_s\over d\lambda_s^0}(y)-1\right)
   (N(ds,dy)-\lambda_s^0(dy)ds)\right),                   \tag{C2.13}
\]
provided the usual exponential compensator integral is finite.  The jump
terms are retained in the Dol\'eans exponential.
\end{proposition}

\begin{proposition}[Brownian diagnostic likelihood]
\label{prop:r29-c2-brownian}
Only for a continuous diagnostic channel
$dY_t=h_\vartheta(X_t,A_t)dt+dB_t$ does
\[
 L_t=\mathcal E\left(\int_0^t
       [\widehat h_\vartheta(s)-\widehat h_{\vartheta_0}(s)]\,dI_s\right)
                                                                    \tag{C2.14}
\]
hold.  The bounded diagnostic score from C1 gives Novikov on every finite
horizon.  This statement is not applied to discrete or surface-counting
channels.
\end{proposition}

\begin{proof}[Proof of the three propositions]
Equation (C2.12) is repeated Bayes change of measure.  Equation (C2.13) is the
Radon--Nikodym formula for marked point compensators.  Equation (C2.14) follows
from the Brownian innovation theorem after the observation model has actually
identified a Brownian filtration.  The three formulas have different
quadratic characteristics and are therefore stated separately.
\end{proof}

\begin{corollary}[Channel-correct stochastic and BSDE stability]
\label{cor:r29-c2-bsde}
Under Theorem~\ref{thm:r29-c2-optional}, the appropriate likelihood
characteristics above, convergence of terminal conditions in $L^2$, and
locally uniform Lipschitz convergence of BSDE drivers, the likelihoods,
stochastic integrals, and BSDE solutions converge jointly in their natural
continuous or jump path spaces.
\end{corollary}

\begin{proof}
Uniform integrability of the channel-specific likelihoods follows from their
respective exponential characteristic bounds.  Martingale convergence plus
terminal and driver convergence gives the standard stability estimate; the
jump case includes the compensated-jump norm.  Gronwall closes the estimate.
\end{proof}

The strict dual now covers genuinely non-locally-compact Polish platforms, the
annihilator sees contact cycles, the prediction state has a fixed ambient
space and continuation kernel, and likelihood/BSDE statements match the
actual observation channel.
