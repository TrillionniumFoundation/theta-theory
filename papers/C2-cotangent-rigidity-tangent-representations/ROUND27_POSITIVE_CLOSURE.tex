
\section{Round-twenty-seven reconstruction: Polish strict duality, complete boundary adjoints, and path-valued prediction}
\label{sec:r27-c2}

The platform states supplied by A3, B4, and C1 are Polish and generally not
locally compact.  We therefore use the mixed strict topology on $C_b(E)$ for
Polish $E$.  The kinetic annihilator includes the initial-output component of
the B3 graph operator.  Rigidity is obtained from pressure identities on
complete exposing rays using continuous Holder localizers, and optional
projection is stated for a path-augmented prediction process.

\subsection{Weighted mixed strict topology on a Polish space}

Let $(E,d)$ be Polish and let $W:E\to[1,\infty)$ be continuous with compact
sublevels.  Put
\[
 C_W(E)=\{f\in C(E):f/W\in C_b(E)\}.
\]
For compact sets $K_n\uparrow E$ and numbers $a_n\downarrow0$, define
\[
 p_{K,a}(g)=\sup_n a_n\sup_{x\in K_n}|g(x)|.
\]
The mixed strict topology $\beta_0$ on $C_b(E)$ is generated by these
seminorms together with uniform convergence on compact sets.  Define
\[
 \beta_W=T^{-1}\beta_0,\qquad T f=f/W.                    \tag{C2.1}
\]

\begin{theorem}[Weighted Polish strict dual]
\label{thm:r27-c2-dual}
The continuous dual of $(C_W(E),\beta_W)$ is
\[
 \mathcal M_W(E)=\{\mu\text{ finite signed Radon}:
                    \int W\,d|\mu|<\infty\}.              \tag{C2.2}
\]
A set of measures is equicontinuous precisely when its weighted total
variation is bounded and it is uniformly $W$-tight.
\end{theorem}

\begin{proof}
Multiplication by $W^{-1}$ is a linear homeomorphism from
$(C_W,\beta_W)$ to $(C_b,\beta_0)$.  On a Polish space, the mixed strict dual
of $C_b$ is the finite signed Radon measures; this follows by restricting a
continuous functional to each compact $K_n$, applying Riesz representation,
and using the $a_n\downarrow0$ bound to glue the compatible measures with
tight total mass.  Pulling a finite measure $\nu$ back through $T$ gives
$d\mu=W^{-1}d\nu$ and
$\int W\,d|\mu|=|\nu|(E)$.  The compact gluing argument also gives the stated
equicontinuity criterion.  No local compactness of $E$ is used.
\end{proof}

\subsection{The complete kinetic adjoint}

B3 uses
\[
 \mathfrak B^\circ(u,[h])
 =\bigl(R(u,[h]),u(0)\bigr),
 \quad
 R=\partial_tu+v\cdot\nabla_xu-DQ_f[u]-\mathcal Ch^\circ.               \tag{C2.3}
\]
Pair its outputs with $(r,r_0)$.  Integration by parts gives
\[
 \int_0^T\langle r,R\rangle\,dt+\langle r_0,u(0)\rangle
 =
 \int_0^T\langle-\partial_tr-v\cdot\nabla_xr-\mathcal L_f^*r,u\rangle dt
 -\int_0^T\langle\Delta r,h^\circ\rangle dt
 +\langle r(T),u(T)\rangle+\langle r_0-r(0),u(0)\rangle.                 \tag{C2.4}
\]

\begin{theorem}[Complete hard-sphere annihilator]
\label{thm:r27-c2-annihilator}
On the B3 quotient graph spaces, a source triple
$(\zeta_f,\zeta_c,\zeta_0)$ annihilates the homogeneous tangent kernel if and
only if, modulo the five invariant sources,
\[
 \zeta_f=-\partial_tr-v\cdot\nabla_xr-\mathcal L_f^*r,\qquad
 \zeta_c=[-\Delta r],\qquad
 \zeta_0=r_0-r(0),\qquad r(T)=0.                          \tag{C2.5}
\]
The gauge-fixed pair $(r,r_0)$ is unique.
\end{theorem}

\begin{proof}
Formula (C2.4) proves inclusion of the displayed range in the annihilator.
Conversely, B3's closed quotient range gives
$(\ker\mathfrak B^\circ)^\perp=\operatorname{Ran}
(\mathfrak B^\circ)^*$.  The explicit integration-by-parts identity identifies
all three components, including the initial output.  Quotienting the five
collision invariants and fixing $r(T)=0$ removes the gauge.
\end{proof}

\subsection{Pressure rigidity on complete exposing rays}

Let $\Sigma$ be the A4 history shift and let $\phi$ be the difference of two
Holder cocycles.  For every regular periodic orbit $\mathcal O(p)$ choose a
bounded Holder distance
\[
 d_p(x)=\min\{1,d_\Sigma(x,\mathcal O(p))^\alpha\}.
\]
Let $W$ be the A4 Lyapunov weight and choose continuous Holder cutoffs
$\chi_m$ satisfying $0\le\chi_m\le1$, $\chi_m=0$ on $\{W\le m\}$, and
$\chi_m=1$ on $\{W\ge2m\}$.  Define the exposing ray
\[
 V_{t,m}=-t\,d_p-R_m\chi_m,\qquad t\ge0.                  \tag{C2.6}
\]
Assume the two pressure functionals agree on every complete ray (C2.6) and
on a common Holder neighbourhood in the cocycle direction at every point of
the ray.  A4 supplies the local analytic charts and their overlap
continuation.

\begin{lemma}[Equilibrium localization]
\label{lem:r27-c2-localize}
For a diagonal choice $R_m\to\infty$ followed by $t\to\infty$, the unique
equilibrium states of $V_{t,m}$ converge to the uniform measure on
$\mathcal O(p)$.
\end{lemma}

\begin{proof}
The continuous tail cutoff and the A4 exponential return estimate make
equilibrium mass outside $\{W\le2m\}$ at most $e^{-cR_m}$.  On the compact
sublevel, every invariant measure not supported on $\mathcal O(p)$ has
positive integral of $d_p$.  The variational principle then forces its mass
away from the orbit to vanish as $t\to\infty$.  The only invariant probability
supported on one periodic orbit is its uniform measure.  Tightness permits
the diagonal passage.  No discontinuous indicator is used.
\end{proof}

\begin{theorem}[Pressure-to-Livsic rigidity]
\label{thm:r27-c2-rigidity}
Equality of the pressure functionals on the complete exposing rays implies
\[
 S_{n(p)}\phi(p)=0
\]
for every regular periodic orbit, and hence
\[
 \phi=u-u\circ\sigma
\]
for a Holder transfer function $u$ on the prepared history space.
\end{theorem}

\begin{proof}
Differentiate the pressure equality in the cocycle direction at every
$V_{t,m}$.  The derivative is the equilibrium expectation of $\phi$, hence it
vanishes.  Apply Lemma \ref{lem:r27-c2-localize} to obtain every periodic
sum.  Construct $u$ by summing $\phi$ along connectors from a fixed reference
cylinder.  Periodic cancellation makes the definition independent of the
connector; exponential variation makes $u$ Holder.
\end{proof}

\subsection{Path-augmented prediction}

For a signal $X$ define the stopped-path state
\[
 \bar X_t=(X_{[0,t]},X_t)\in D([0,t];E_X)\times E_X.
\]
Its prediction process is
\[
 \bar\Pi_t^n=\mathcal L(\bar X_t^n\mid\mathcal F_t^{Y^n}).               \tag{C2.7}
\]
This process determines conditional expectations of whole-path functionals,
unlike the current-state filter alone.

Assume:
\begin{enumerate}[label=(P\arabic*)]
\item $(Y^n,\bar\Pi^n)\Rightarrow(Y,\bar\Pi)$ in the path-prediction space;
\item a countable cylinder algebra solves a convergent filter martingale
      problem;
\item observation and likelihood martingales have uniformly tight
      characteristics;
\item predictable covariations converge uniformly on compact time intervals.
\end{enumerate}

\begin{theorem}[Optional projections of path functionals]
\label{thm:r27-c2-optional}
For every bounded continuous cylinder functional $F$ of the stopped path,
\[
 E[F(X^n_{[0,T]})\mid\mathcal F_t^{Y^n}]
 \Rightarrow
 E[F(X_{[0,T]})\mid\mathcal F_t^{Y}]                      \tag{C2.8}
\]
as a process after adjoining the future continuation kernel to
$\bar\Pi_t^n$.  Innovation martingales and predictable brackets converge
jointly.  The result extends to bounded continuous path functionals by the
tight cylinder approximation.
\end{theorem}

\begin{proof}
For a cylinder functional, the conditional expectation is a continuous
functional of the stopped-path prediction and the Markov continuation kernel,
except at a null set of limiting jump times.  Assumptions (P1)--(P2) identify
every subsequential limit.  Uniform tightness of characteristics permits
passage of stochastic integrals, while (P4) identifies brackets.  Tightness of
the path state gives a uniform cylinder approximation and completes the
extension.
\end{proof}

\subsection{Likelihood exponentials}

On the positive diagnostic experiment of C1, let
$L_t=dP/dQ|_{\mathcal F_t^Y}$.  Strict positivity follows from the independent
noise density.  The condition used for Girsanov is the exponential bracket
bound
\[
 E_Q\exp\left\{\tfrac12(1+\delta)
 \int_0^T|H_s|^2\,ds\right\}<\infty,                      \tag{C2.9}
\]
where $dL_t=L_tH_t\,dI_t$.  It is verified by the C1 score envelope and
finite-horizon action-frequency bound; reciprocal polynomial moments are not
used as a substitute.

\begin{proposition}[Stable innovation exponential]
\label{prop:r27-c2-girsanov}
Under (C2.9),
\[
 {L_t\over L_0}
 =\mathcal E\!\left(\int_0^tH_s\,dI_s\right),\qquad
 \log{L_t\over L_0}
 =\int_0^tH_s\,dI_s-\tfrac12\int_0^t|H_s|^2ds.             \tag{C2.10}
\]
Under (P3)--(P4), these likelihoods, their stochastic integrals, and bounded
Lipschitz BSDE solutions converge jointly.
\end{proposition}

\begin{proof}
Martingale representation gives $H$ after localization.  The exponential
bracket condition is Novikov's criterion and removes the localization.
Uniform tightness and bracket convergence give stability of stochastic
integrals; a Gronwall estimate gives the BSDE conclusion.
\end{proof}

The strict dual now matches the actual Polish platform, the kinetic
annihilator contains every boundary component, periodic localization is
Holder, path functionals use a path-valued prediction, and the Girsanov gate
is an actual exponential condition consistent with C1.
