\section{A sharp occupation law}\label{sec:width}
The argument is useful in an inhomogeneous form. At epoch $t$, let the
fresh command have a prescribed law $p_t$, independently across epochs,
and let its angular operator satisfy
\[
 \norm{P_t|_{L^2_0(\sigma)}}\le\lambda_t\le1,
 \qquad g_t=1-\lambda_t^2.
\]
The alphabets and orthogonal matrices can also depend on $t$ in this
probabilistic statement. No strict gap is required at every epoch.
Define the dimension-only constants
\begin{equation}\label{eq:constants}
 a_d=3+\frac2{d-1},\qquad
 B_d=2048d(16A_d)^{2/(d-1)}.
\end{equation}

\begin{theorem}[Gap-weighted occupation without a mixing-time loss]
\label{thm:occupation}
For the hidden process of Lemma~\ref{lem:terminal}, suppose
$q_N\ge\kappa>0$. For every integer $k\ge1$,
\begin{equation}\label{eq:occupation}
 \sum_{\substack{0\le t<N\\K_t\le k}}g_t
       \le B_d\kappa^{-a_d}k^{2/(d-1)}.
\end{equation}
The other cuts may have arbitrarily many labels. The inequality also
holds when $K_t$ is replaced by the number of positive-probability
states under this independent-command test law.
\end{theorem}
\begin{proof}
Fix $k$ and choose one smoothing scale $\tau$ by
\eqref{eq:smoothing-scale}; keep it fixed throughout the run. Set
$f_t=\Law(Z_t)*\gamma_\tau$ and
$\widetilde Z_t=U_{a_{t+1}}Z_t$. Orthogonal invariance of Gaussian noise
and independence of the fresh command imply
$f_{\widetilde Z_t}=P_tf_t$.
By \eqref{eq:martingale-update},
$Z_{t+1}=\E[\widetilde Z_t\mid S_{t+1}]$.
Lemma~\ref{lem:centroid-entropy} therefore yields
\begin{equation}\label{eq:entropy-step}
 \Ent(f_{t+1})-\Ent(f_t)
 \ge \Ent(P_tf_t)-\Ent(f_t)-\frac{\Delta_t}{2\tau^2}.
\end{equation}
Whenever $K_t\le k$, Lemmas~\ref{lem:entropy-gap} and
\ref{lem:sparse-root}, together with $q_t\ge q_N\ge\kappa$, bound the
entropy increment from randomizing the command below by $g_t\kappa/16$.
At all other cuts that increment remains nonnegative.

Sum \eqref{eq:entropy-step}. The total quadratic loss is at most one.
Furthermore, entropy concavity gives
$\Ent(f_0)\ge\frac d2\log(2\pi e\tau^2)$, while
$\E\norm{Z_N+\tau G}^2\le1+d\tau^2$ and Gaussian maximal entropy give
\[
 \Ent(f_N)\le\frac d2\log\bigl(2\pi e(\tau^2+1/d)\bigr).
\]
The latter bound follows directly by comparing $f_N$ with the centered
Gaussian of that covariance and using nonnegative relative entropy.
It does not require $Z_N$ to be centered. Consequently
\begin{equation}\label{eq:finite-entropy-budget}
 \frac\kappa{16}\sum_{K_t\le k}g_t
 \le\frac d2\log\left(1+\frac1{d\tau^2}\right)
                  +\frac1{2\tau^2}
 \le\frac1{\tau^2}.
\end{equation}
Since
$\tau^{-2}=128d\kappa^{-2}(16A_dk/\kappa)^{2/(d-1)}$,
this is \eqref{eq:occupation}. Nothing in this summation restricts the
intermediate hidden affine dimension, projected vertices, or the widths
of the other registers. Under the test law, only positive-mass labels
occur in the mixture $f_t$, proving the final assertion.
\end{proof}

For a fixed gap $g=1-\lambda^2>0$, \eqref{eq:occupation} gives
$\#\{0\le t<N:K_t\le k\}\le B_dg^{-1}\kappa^{-a_d}k^{2/(d-1)}$.
Counting endpoints $1,\ldots,N$ instead changes this bound by at most
one. Thus the earlier logarithmic loss in occupation, as well as in
peak width, is removed. The proof does not assume a spectral gap for
the stochastic state matrices; its gap belongs to the \emph{external}
orthogonal action on the sphere.

\begin{lemma}[Exact spherical synthesis, inherited upper bound]
\label{lem:net-upper}
For every $N\ge1$ and finite orthogonal command alphabet, the experiment
\eqref{eq:law} admits a common-row realization with at most
$C_dN^{(d-1)/2}$ labels. Rational commands, seeds and signal admit
rational rows with at most $d+1$ successors.
\end{lemma}
\begin{proof}
This is the construction from \cite{GTF36}; we include it to make the
matched theorem self-contained. Write $m=d-1$ and
$\eta=1-1/(8N)$. The two rational stereographic charts
\[
 t\longmapsto\left(\frac{2t}{1+\norm t^2},
          \pm\frac{1-\norm t^2}{1+\norm t^2}\right),
 \qquad t\in[-1,1]^m,
\]
are $2$-Lipschitz. Every sphere point has a preimage in the unit ball
in one chart. On a grid of mesh $1/M$, with
$M=\lceil2\sqrt{mN}\rceil$, their images form a rational set $V$ of
at most $2(2M+1)^m$ points. Every unit direction is within chordal
distance $\sqrt m/M$ of $V$. Hence
$\eta\mathbb B^d\subset P:=\conv V$ by the support-function criterion.

Associate label $u\in V$ at time $t$ with
$a_tu$, where $a_t=\eta^{-t}/4$. On command $a$ choose a convex
representation of $\eta U_au$ in $V$. These are common command rows.
Initialize the mean to $\rho x$ in $a_0P$, which is possible because
$\eta/4\ge7/32>\rho$. Use terminal mean $a_Ne_j^Tu$.
Bernoulli's inequality gives $\eta^N\ge7/8$, so $a_N\le2/7<1$.
The identity
$a_{t+1}\E[u'\mid u,a]=a_tU_au$ proves exactness on every word.
For rational data, each convex feasibility problem has a rational basic
solution with at most $d+1$ nonzero coefficients. The state holds only
the label. A charged final bit uses two labels and is absorbed in $C_d$.
No efficient table-generation algorithm is inferred from finite feasibility.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:main}]
Use the independent command law $p$ from \eqref{eq:gap-intro} and any
fixed seed. Lemma~\ref{lem:terminal} gives
$\kappa=\rho/\sqrt d-2\varepsilon>0$.
With $K=\max_tK_t$, Theorem~\ref{thm:occupation} gives the explicit bound
\begin{equation}\label{eq:peak-explicit}
 K\ge\left(\frac{(1-\lambda^2)\kappa^{a_d}}{B_d}N\right)^{(d-1)/2}.
\end{equation}
The preceding lemma proves the exact upper bound.
Proposition~\ref{prop:separate} proves the isolated-cut assertion.
\end{proof}

\begin{corollary}[An obstruction for actual terminal error]
\label{cor:actual-error}
For an available-label profile $\mathbf K$, let
$\mathcal E_N(\mathbf K)$ be the least actual worst-row terminal error
among all machines with that profile. For every integer $k$ with
$I_k:=\#\{0\le t<N:K_t\le k\}>0$,
\begin{equation}\label{eq:actual-error}
 \mathcal E_N(\mathbf K)\ge\frac12
 \left[\frac\rho{\sqrt d}-
 \left(\frac{B_dk^{2/(d-1)}}{(1-\lambda^2)I_k}\right)^{1/a_d}\right]_+.
\end{equation}
In particular, peak $o(N^{(d-1)/2})$ entails
$\liminf\mathcal E_N\ge\rho/(2\sqrt d)$.
\end{corollary}
\begin{proof}
The finite parameter domain of a fixed-profile machine is compact and
the wordwise objective continuous, so the minimum exists.
Apply \eqref{eq:occupation} with $\kappa=q_N$ when $q_N>0$, and use
$q_N\ge\rho/\sqrt d-2\mathcal E_N$. If $q_N=0$, the same conclusion
follows directly from calibration. The last assertion uses $I_K=N$.
\end{proof}

This is an actual-error bound: it does not add local defects or rule out
their cancellation. If a separately supplied diagram has an accumulated
local deficiency certificate $D$, the classical composition argument
produces a machine with final error at most $D$; only then does
\eqref{eq:actual-error} lower-bound that certificate. It need not equal
the minimum terminal error.
