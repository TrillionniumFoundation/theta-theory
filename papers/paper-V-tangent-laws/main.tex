\begin{abstract}
We develop the representation theory of the physical theta-semigroups
constructed in Paper III.  The representation layer is strictly downstream:
it is not used to derive the billiard response, rough limit, microscopic DPP,
or HJB equation.

First, a parabolic implicit-function theorem proves Fr\'echet differentiability
of the terminal-data solution map under explicit uniform parabolic and
H\"older hypotheses.  For the entropic physical theta-semigroup, the terminal
map is real analytic on bounded payoffs.  Its first derivative is expectation
under a payoff-dependent tangent probability and its second derivative is the
corresponding covariance.  Tangent kernels form a cocycle obtained by
differentiating nonlinear time consistency.

We prove a converse characterization: a monotone cash-additive twice
differentiable functional whose tangent probabilities have one fixed
covariance curvature is necessarily logarithmic moment generating.  The
finite collision recursions of Paper III carry exact exponential tangent
measures; these laws and their first two derivative operators converge to the
continuum tangent diffusion.

For smooth data, the density process is a genuine stochastic exponential with
integrand $\vartheta\sigma^TDu$.  Bounded terminal data give a uniformly
integrable martingale, and a BMO argument yields a valid Girsanov change of
measure without inverting $\sigma^TDu$.  Quadratic and tangent BSDEs follow.
Path-dependent, pure-game, and volatility-uncertain representations are
recorded as typed optional branches rather than as inputs to the core
characterization.
\end{abstract}

\section{Introduction}

A nonlinear semigroup cannot be represented by one payoff-independent Markov
kernel for every terminal payoff.  It may nevertheless possess a canonical
bundle of payoff-dependent tangent laws.  For the entropic semigroup this
bundle is rigid: first derivatives are probabilities, second derivatives are
covariances multiplied by one constant, and the tangent kernels compose.

The underlying diffusion coefficients in this paper are those obtained from
the same moving open billiard in Papers I--III:
\[
 a_{\rm phys}(x)
 =
 \frac1{7\bar\tau_{\alpha(x)}}I_2
\]
in covariance convention, with generator coefficient $a_{\rm phys}/2$.
The finite microscopic laws are the collision recursions of Paper III.

We use classical BSDE theory \cite{PardouxPeng1990,Kobylanski2000},
functional It\^o/PPDE theory \cite{Dupire2009,ContFournie2013,
EkrenTouziZhang2014}, and second-order BSDE theory
\cite{SonerTouziZhang2012}.  The new core is the tangent-law
characterization and its microscopic collision convergence.

\section{A fixed-law obstruction}\label{sec:obstruction}

Let $E$ be a Polish space.

\begin{theorem}[A payoff-independent law forces linearity]\label{thm:fixed-law}
Suppose that for fixed $s<t$ and every $x$ there is a probability kernel
$P_{s,t}(x,dy)$ such that
\[
 \cE_{s,t}[\phi](x)
 =
 \int_E\phi(y)P_{s,t}(x,dy)
\]
for every bounded continuous $\phi$.  Then $\cE_{s,t}$ is linear, positive,
and constant preserving.
\end{theorem}

\begin{proof}
All three properties are inherited from integration against a fixed
probability kernel.
\end{proof}

Thus a genuinely nonlinear semigroup requires payoff-dependent tangent laws,
a controlled family, a game, or a nondominated representation.

\section{A parabolic implicit-function theorem}\label{sec:ift}

Work on $\T^d$.  Fix $0<\alpha<1$ and put
\[
 \mathbb X=C^{1+\alpha/2,2+\alpha}([0,T]\times\T^d),
\]
\[
 \mathbb Y=
 C^{\alpha/2,\alpha}([0,T]\times\T^d)
 \times C^{2+\alpha}(\T^d).
\]
Let $F(t,x,r,p,X)$ be $C^k$, $k\ge2$, near the relevant jet set and consider
\[
 -u_t-F(t,x,u,Du,D^2u)=0,\qquad u(T)=\phi.
\]

\begin{assumption}[Uniform classical window]\label{ass:window}
There are $0<\lambda\le\Lambda$ such that
\[
 \lambda I\le F_X\le\Lambda I
\]
on the jet neighbourhood.  All derivatives through order $k$ are bounded and
H\"older there, and one classical solution $(u_0,\phi_0)$ is given.
\end{assumption}

\begin{theorem}[Terminal-map differentiability]\label{thm:ift}
Under \cref{ass:window}, the terminal solution map is $C^{k-1}$ near
$\phi_0$.  Its first derivative in direction $\eta$ is the unique solution of
\[
 -v_t-F_rv-F_p\cdot Dv-F_X:D^2v=0,\qquad v(T)=\eta,
\]
with coefficients evaluated along the nonlinear solution.
\end{theorem}

\begin{proof}
Define
\[
 \mathfrak F(u,\phi)
 =
 \bigl(-u_t-F(t,x,u,Du,D^2u),u(T)-\phi\bigr).
\]
H\"older composition makes $\mathfrak F$ a $C^{k-1}$ map from
$\mathbb X\times C^{2+\alpha}$ to $\mathbb Y$.  Its $u$-derivative is a
uniformly parabolic terminal operator and is a bounded isomorphism by Schauder
theory.  Apply the Banach implicit-function theorem and differentiate
$\mathfrak F(u[\phi],\phi)=0$.
\end{proof}

\section{The physical entropic semigroup}\label{sec:entropic}

Let
\[
 dX_s=b(s,X_s)\,ds+\sigma(s,X_s)\,dW_s,\qquad X_t=x,
\]
where $b,\sigma$ are bounded globally Lipschitz and
$a=\sigma\sigma^T$ is uniformly elliptic.  In the same-platform application,
$a$ is the physical covariance from Paper III.

For $\vartheta\ne0$, define
\begin{equation}\label{eq:theta}
 \cE_{t,T}^{\vartheta}[\phi](x)
 =
 \frac1\vartheta
 \log \E_{t,x}e^{\vartheta\phi(X_T)}.
\end{equation}

\begin{theorem}[Global analytic terminal map]\label{thm:analytic}
For fixed $(t,x)$, the map
\[
 C_b(\R^d)\ni\phi\longmapsto
 \cE_{t,T}^{\vartheta}[\phi](x)
\]
is real analytic.  Its first two Fr\'echet derivatives are
\[
 D\cE^\vartheta[\phi]\eta
 =
 \frac{\E(e^{\vartheta\phi(X_T)}\eta(X_T))}
 {\E e^{\vartheta\phi(X_T)}},
\]
and
\[
 D^2\cE^\vartheta[\phi](\eta_1,\eta_2)
 =
 \vartheta\Cov_{\Qp^\phi}
 (\eta_1(X_T),\eta_2(X_T)).
\]
Higher derivatives are tilted cumulants multiplied by
$\vartheta^{n-1}$.
\end{theorem}

\begin{proof}
The exponential Nemytskii map has an absolutely convergent power series on
bounded balls of $C_b$.  Expectation is bounded and positive, and its value on
$e^{\vartheta\phi}$ is bounded away from zero on each ball.  The logarithm is
analytic on that range.  Differentiate the composition.
\end{proof}

Define
\[
 \frac{d\Qp_{t,x}^{\phi}}{d\Pp_{t,x}}
 =
 \frac{e^{\vartheta\phi(X_T)}}
 {\E_{t,x}e^{\vartheta\phi(X_T)}}.
\]

\section{Tangent kernels and nonlinear time consistency}\label{sec:cocycle}

For $s<t$ and a bounded function $\psi$, define
\[
 K_{s,t}^{\psi}(x,dy)
 =
 \frac{e^{\vartheta\psi(y)}P_{s,t}(x,dy)}
 {P_{s,t}e^{\vartheta\psi}(x)}.
\]

\begin{theorem}[Tangent-kernel cocycle]\label{thm:cocycle}
The nonlinear semigroup is time consistent and
\[
 K_{r,t}^{\phi}
 =
 K_{r,s}^{\cE_{s,t}^{\vartheta}[\phi]}
 K_{s,t}^{\phi}.
\]
Equivalently,
\[
 D\cE_{r,t}^{\vartheta}[\phi]
 =
 D\cE_{r,s}^{\vartheta}
 [\cE_{s,t}^{\vartheta}[\phi]]
 \circ
 D\cE_{s,t}^{\vartheta}[\phi].
\]
\end{theorem}

\begin{proof}
The Markov property gives
\[
 P_{r,t}e^{\vartheta\phi}
 =
 P_{r,s}
 \exp\{\vartheta\cE_{s,t}^{\vartheta}[\phi]\}.
\]
The Radon--Nikodym factors multiply and the intermediate normalization
cancels.  The derivative identity is the Banach chain rule.
\end{proof}

\section{Characterization by covariance curvature}\label{sec:characterization}

Let $\mathfrak E:L^\infty(E)\to\R$ be twice Fr\'echet differentiable.

\begin{assumption}[Tangent probability bundle]\label{ass:curvature}
The functional is monotone, cash additive, and $\mathfrak E(0)=0$.  For every
$\phi$ there is a probability $Q_\phi$ such that
\[
 D\mathfrak E(\phi)f=Q_\phi(f)
\]
and
\[
 D^2\mathfrak E(\phi)(f,g)
 =
 \vartheta\bigl(Q_\phi(fg)-Q_\phi(f)Q_\phi(g)\bigr)
\]
for one fixed $\vartheta\ne0$.
\end{assumption}

\begin{theorem}[Static tangent-law characterization]\label{thm:static-char}
Under \cref{ass:curvature},
\[
 \mathfrak E(\phi)
 =
 \frac1\vartheta
 \log Q_0(e^{\vartheta\phi}).
\]
Moreover,
\[
 \frac{dQ_\phi}{dQ_0}
 =
 \frac{e^{\vartheta\phi}}
 {Q_0(e^{\vartheta\phi})}.
\]
\end{theorem}

\begin{proof}
Fix $\phi,f$ and set $Q_t=Q_{t\phi}$.  Differentiating
$Q_t(f)=D\mathfrak E(t\phi)f$ gives
\[
 \frac{d}{dt}Q_t(f)
 =
 \vartheta\{Q_t(f\phi)-Q_t(f)Q_t(\phi)\}.
\]
The unique solution of this measure-valued ODE is the exponential tilt of
$Q_0$.  Since
\[
 \frac{d}{dt}\mathfrak E(t\phi)=Q_t(\phi),
\]
integration from zero to one gives the logarithmic moment formula.
\end{proof}

\begin{theorem}[Dynamic characterization]\label{thm:dynamic-char}
Let a monotone cash-additive time-consistent semigroup be twice differentiable
in terminal data.  If each tangent kernel has the covariance curvature of
\cref{ass:curvature} with the same $\vartheta$, and the kernels satisfy the
chain-rule cocycle, then the semigroup is the entropic semigroup of its
zero-payoff tangent Markov family.
\end{theorem}

\begin{proof}
Apply \cref{thm:static-char} at every $(s,t,x)$.  At zero payoff the tangent
kernels compose by the cocycle and hence form a Markov family.  The static
formula identifies every nonlinear transition with the logarithmic moment
transition of that family.
\end{proof}

\section{Microscopic collision tangent laws}\label{sec:microscopic}

Let $\Pp_h$ be the path law of the finite collision recursion in Paper III and
let $X_T^h$ be its terminal state.  For terminal payoff $\phi$, define
\[
 \frac{d\Qp_h^\phi}{d\Pp_h}
 =
 \frac{e^{\vartheta\phi(X_T^h)}}
 {\E_{\Pp_h}e^{\vartheta\phi(X_T^h)}}.
\]
The finite recursion differentiates exactly:
\[
 D\cE_h^\vartheta[\phi]\eta
 =
 \E_{\Qp_h^\phi}\eta(X_T^h),
\]
\[
 D^2\cE_h^\vartheta[\phi](\eta_1,\eta_2)
 =
 \vartheta\Cov_{\Qp_h^\phi}
 (\eta_1(X_T^h),\eta_2(X_T^h)).
\]

\begin{theorem}[Microscopic tangent convergence]\label{thm:micro-conv}
Assume the collision path laws of Paper III converge weakly to the physical
diffusion law, and let $\phi,\eta_1,\eta_2$ be bounded continuous.  Then
\[
 \Qp_h^\phi\Longrightarrow\Qp^\phi,
\]
and the first two derivative operators converge locally uniformly in the
initial state.
\end{theorem}

\begin{proof}
The exponential tilts are uniformly bounded above and below because $\phi$ is
bounded.  Weak convergence therefore passes to numerator and denominator in
every tilted expectation.  The same argument applies to the products
$\eta_1\eta_2$.
\end{proof}

\begin{remark}
The theorem imports the path convergence of Paper III, not the filtering or
game extension of Paper IV.
\end{remark}

\section{A genuine Girsanov theorem}\label{sec:girsanov}

Let
\[
 u(t,x)
 =
 \frac1\vartheta
 \log\E_{t,x}e^{\vartheta\phi(X_T)}.
\]
For smooth bounded data, It\^o's formula gives
\[
 M_s
 =
 \exp\{\vartheta u(s,X_s)-\vartheta u(t,x)\}.
\]

\begin{theorem}[Tangent-law density process]\label{thm:girsanov}
The process $M$ is the uniformly integrable martingale
\[
 M_s
 =
 \frac{\E[e^{\vartheta\phi(X_T)}\mid\F_s]}
 {\E e^{\vartheta\phi(X_T)}}
\]
and
\[
 M_s
 =
 \mathcal E\left(
 \int_t^s\vartheta\sigma^TDu(r,X_r)\,dW_r
 \right).
\]
Under $\Qp^\phi$ defined by $M_T$, the process
\[
 W_s^{\Qp}
 =
 W_s-\int_t^s\vartheta\sigma^TDu(r,X_r)\,dr
\]
is Brownian motion, and
\[
 dX_s
 =
 \left(b+\vartheta aDu\right)(s,X_s)\,ds
 +\sigma(s,X_s)\,dW_s^{\Qp}.
\]
\end{theorem}

\begin{proof}
The conditional-expectation formula makes $M$ a positive uniformly integrable
martingale.  Applying It\^o to $e^{\vartheta u(s,X_s)}$ and using the entropic
PDE cancels the drift and gives the stochastic-exponential representation.
The stochastic logarithm is BMO for bounded terminal data in the uniformly
elliptic bounded-coefficient window; localization and uniform integrability
justify Girsanov.  No inversion of $\sigma^TDu$ is used.
\end{proof}

\section{BSDE bundle}\label{sec:bsde}

Let
\[
 Y_s=u(s,X_s),\qquad Z_s=\sigma^TDu(s,X_s).
\]

\begin{theorem}[Quadratic and tangent BSDEs]\label{thm:bsde}
The entropic solution satisfies
\[
 Y_s
 =
 \phi(X_T)
 +\int_s^T\frac{\vartheta}{2}\abs{Z_r}^2\,dr
 -\int_s^TZ_r\,dW_r.
\]
For a terminal direction $\eta$, the tangent process
\[
 V_s=D\cE_{s,T}^{\vartheta}[\phi]\eta(X_s)
\]
satisfies
\[
 V_s
 =
 \eta(X_T)
 -\int_s^T\widetilde Z_r\,dW_r^{\Qp}.
\]
\end{theorem}

\begin{proof}
Apply It\^o's formula to $u(s,X_s)$ and use the entropic PDE.  Differentiate
the PDE in terminal data.  Under the tangent law of
\cref{thm:girsanov}, the linear first-order drift cancels, leaving the
martingale representation.
\end{proof}

\section{Typed optional representation branches}\label{sec:branches}

The following branches are not part of the proof of
\cref{thm:static-char,thm:dynamic-char,thm:micro-conv}.

\begin{proposition}[Path-dependent branch]\label{prop:ppde}
On a canonical path space, if the path-dependent terminal payoff and
coefficients satisfy the regularity and comparison hypotheses of functional
It\^o/PPDE theory, the same exponential transform yields a quadratic BSDE and
a viscosity PPDE representation.
\end{proposition}

\begin{proposition}[Pure-game branch]\label{prop:game}
If the local Hamiltonian has the unique pure saddle supplied by Paper IV, the
resulting feedback game represents the corresponding Isaacs semigroup.  This
branch imports Paper IV and is optional.
\end{proposition}

\begin{proposition}[Volatility-uncertain branch]\label{prop:2bsde}
For a stable nondominated family of uniformly elliptic volatility laws,
dynamic programming and comparison yield the associated second-order BSDE
representation.  This branch has platform identifier
\texttt{SV-CONTROL-v1}; it is not a same-platform statement about
\texttt{OB3-MG-v1}.
\end{proposition}

\section{Main theorem and dependency statement}

\begin{theorem}[Tangent-law representation theorem]\label{thm:main}
The physical theta-semigroup of Paper III has:
\begin{enumerate}[label=(\roman*)]
\item an analytic terminal map;
\item a probability-valued tangent-kernel cocycle;
\item covariance second variation;
\item convergent microscopic collision tangent laws;
\item a genuine Girsanov density;
\item quadratic and linear tangent BSDE representations.
\end{enumerate}
Conversely, the tangent probability and covariance-curvature axioms
characterize the entropic semigroup.
\end{theorem}

\begin{proof}
Combine \cref{thm:analytic,thm:cocycle,thm:static-char,thm:dynamic-char,thm:micro-conv,thm:girsanov,thm:bsde}.
\end{proof}
