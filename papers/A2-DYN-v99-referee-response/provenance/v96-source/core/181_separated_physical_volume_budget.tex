\section{The angular volume estimate on the complete source}
\label{sec:v79-volume}
Fix one original exact event, and keep its source measure
$\bar\nu=(4\pi c_*)^{-1}d\alpha\,dp$. On its regular part set
\[
 \omega(x)=|B_m(x)|^{-1},\qquad
 d\bar\nu=\frac{\omega(q)}{4\pi c_*}\,dq,\qquad
 d\mu_t=\frac{1}{4\pi c_*|\nabla L|}\,d\mathcal H^1|_{\{L=t\}}.
\]
Thus $p(t)=\mu_t(\Pi)$ and $b^\varepsilon(t)=\int W^\varepsilon\,d\mu_t$.
All fibers below are those of this same event. The gradient in the
coarea formula is in the original collision coordinates.

Let $S_h(t)$ be the starting points on this fiber whose canonical
forward flow survives until $h$. Its physical area Jacobian and its
optical ratio are denoted by $J_h$ and $\eta_h$, respectively. Define
three nonnegative quantities before any collision limit:
\begin{align}
 E_h^\varepsilon(t)&=\int_{\{L=t\}\setminus S_h(t)}W^\varepsilon\,d\mu_t,
                                                   \label{eq:v79-exit}\\
 V_h^\varepsilon(t)&=\int_{S_h(t)}W^\varepsilon[1-J_h]_+\,d\mu_t,
                                                   \label{eq:v79-volume}\\
 O_h^\varepsilon(t)&=\int_{S_h(t)}W^\varepsilon[1-\eta_h]_+\,d\mu_t.
                                                   \label{eq:v79-optical-loss}
\end{align}
The first is the unchanged initial mass of killed trajectories. The
second is the full physical volume loss. The third isolates the
optical inverse-twist loss. Neither $E_h$ nor $O_h$ omits a physical
word. We write $\bar E_r=r^{-1}\int_0^rE_h\,dh$, and similarly for
$V$ and $O$. All assertions use joint coarea versions in $(t,h)$.

\begin{theorem}[Exact Jacobian factorization and a complete angular bound]
\label{thm:v79-volume-factorization}
On every survivor,
\begin{equation}\label{eq:v79-physical-jacobian}
 \eta_h(x)=\frac{\omega(\Phi_hx)}{\omega(x)},\qquad
 J_h(x)=\eta_h(x)\frac{\chi(x)}{\chi(\Phi_hx)}.
\end{equation}
There is a uniform $C_{\mathrm{ang}}$ such that, for each fixed $r>0$,
\begin{equation}\label{eq:v79-angular-volume-error}
 |\bar V_r^\varepsilon(t)-\bar O_r^\varepsilon(t)|
                            \le C_{\mathrm{ang}}\sqrt r\,A_rp(t)
\end{equation}
for almost every $t$. Consequently, from the stated fixed-width
analytic input,
\begin{equation}\label{eq:v79-complete-angular-bound}
 \limsup_m\sup_{R,n,k}m^2
       \|\bar V_r^\varepsilon-\bar O_r^\varepsilon\|_\infty
                                             \le C\sqrt r.
\end{equation}
The constants are independent of $\varepsilon$. This is a bound for
the complete source, not a per-word estimate multiplied by a word count.
\end{theorem}
\begin{proof}
Multiplying the endpoint-coordinate Jacobian
\eqref{eq:v79-q-jacobian} by the ratio of the physical area densities
gives \eqref{eq:v79-physical-jacobian}. Put
$k_h=\chi(x)/\chi(\Phi_hx)$. The positive-part function is
one-Lipschitz, so
\[
 |[1-\eta_hk_h]_+-[1-\eta_h]_+|
                         \le\eta_h|1-k_h|.
\]
Corollary~\ref{cor:v79-angular-jacobian} bounds $|1-k_h|$ by
$C\sqrt h$. Moreover $\eta_h\le(\Lambda_\circ/\lambda_\circ)J_h$.
Coarea covariance in the target direction therefore yields
\[
 |V_h^\varepsilon(t)-O_h^\varepsilon(t)|
 \le C\sqrt h\int_{S_h(t)}J_h\,d\mu_t
 \le C\sqrt h\,p(t+h).
\]
The target sum contains all its actual words, and $W^\varepsilon\le1$
was used only for an upper bound. Average over $0<h<r$ and bound the
forward average of the nonnegative density by $2A_rp$.
This proves \eqref{eq:v79-angular-volume-error}.

For clarity, the unweighted input used in the last step is
\[
 \limsup_m\sup_{R,n,k}m^2\|A_rp\|_\infty\le C
 \quad\text{for every previously fixed }r>0.
\]
It follows from Input statement~\ref{input:v76-analytic}: for the auxiliary
parameter $1$ the initial factor is $\Theta(c_0)=0$, whence $W^1=1$
and $b^1=p$. The precise inherited source propositions are pinned in
Section~\ref{sec:v79-input-map}. No raw pointwise bound for $p$ or
collision-dependent averaging width is used.
\end{proof}

Define the separate ordered heights
\begin{equation}\label{eq:v79-ordered-budgets}
 \begin{split}
 \mathfrak X_{\varepsilon,M}(r)
 &=\limsup_m\sup_{R,n,k}m^2\esssup_{\mathcal C(M)}\bar E_r^\varepsilon,\\
 \mathfrak V_{\varepsilon,M}(r)
 &=\limsup_m\sup_{R,n,k}m^2\esssup_{\mathcal C(M)}\bar V_r^\varepsilon,\\
 \mathfrak O_{\varepsilon,M}(r)
 &=\limsup_m\sup_{R,n,k}m^2\esssup_{\mathcal C(M)}\bar O_r^\varepsilon.
 \end{split}
\end{equation}
Equation~\eqref{eq:v79-complete-angular-bound} gives both
$\mathfrak V\le\mathfrak O+C\sqrt r$ and
$\mathfrak O\le\mathfrak V+C\sqrt r$ as extended nonnegative inequalities.
Thus the unestimated interior volume contribution has been reduced to
inverse-twist loss, with no unestimated third derivative of the reduced
roof left in this comparison.

\begin{theorem}[The unchanged pointwise error with separated defects]
\label{thm:v79-separated-endpoint}
Under Input statement~\ref{input:v76-analytic}, for sufficiently large
fixed $B$ and every fixed $r>0$,
\begin{equation}\label{eq:v79-separated-endpoint}
 \mathcal E_M\le C_MB^{-1/192}
       +\mathfrak X_{\varepsilon(B),M}(r)
       +\mathfrak O_{\varepsilon(B),M}(r)
       +C_\Theta(1+\varepsilon(B)^{-1})\sqrt r.
\end{equation}
Here $\varepsilon(B)=A_0B^{-1/12}$. The order is
\[
 B,\ \varepsilon(B),\ r,\ \text{auxiliary band fixed};\quad
 m\to\infty;\quad \text{auxiliary band}\to\infty;\quad B\to\infty.
\]
In particular, separate bounds
\begin{equation}\label{eq:v79-two-hypotheses}
 \mathfrak X_{\varepsilon,M}(r)\le C_M\varepsilon^{-q_X}r^{\beta_X},
 \qquad
 \mathfrak O_{\varepsilon,M}(r)\le C_M\varepsilon^{-q_O}r^{\beta_O}
\end{equation}
with finite nonnegative $q_X,q_O$ and positive $\beta_X,\beta_O$
imply the original pointwise assertion. One may choose
$r=c\varepsilon(B)^\gamma$ with
$\gamma>\max\{2,q_X/\beta_X,q_O/\beta_O\}$.
The two estimates in \eqref{eq:v79-two-hypotheses} are proof obligations,
not conclusions of the present theorem.
\end{theorem}
\begin{proof}
The stopped subcoupling has density
$\min\{W(x),W(\Phi_hx)J_h(x)\}$ on survivors. Its two marginals
are dominated by the original weighted fibers. The elementary bound
$[w-w'J]_+\le w[1-J]_++J|w-w'|$, the complete guard modulus of
Corollary~\ref{cor:v78-guard}, and target coarea covariance give
\[
 b^\varepsilon(t)\le b^\varepsilon(t+h)+E_h^\varepsilon(t)
       +V_h^\varepsilon(t)
       +K_\Theta\varepsilon^{-1}\sqrt h\,p(t+h).
\]
Average at the previously fixed width and apply
\eqref{eq:v79-angular-volume-error}. The averaged bad source is
bounded at the ordered height by $2C\varepsilon^{1/16}$; the
unweighted target is bounded as in the preceding proof. Inserting the
positive scalar remainder gives \eqref{eq:v79-separated-endpoint}.
All inequalities precede the suprema and the collision limit.
The displayed choice of $\gamma$ makes each of the three explicit
powers positive, proving the implication.
\end{proof}

\begin{corollary}[An exact separated criterion]
\label{cor:v79-separated-criterion}
Under the same analytic input, the original pointwise assertion is
equivalent to, for every $M$ and every fixed $r_0>0$,
\begin{equation}\label{eq:v79-separated-criterion}
 \lim_{B\to\infty}\inf_{0<r<r_0}
 \left\{\mathfrak X_{\varepsilon(B),M}(r)
       +\mathfrak O_{\varepsilon(B),M}(r)
       +(1+\varepsilon(B)^{-1})\sqrt r\right\}=0.
\end{equation}
\end{corollary}
\begin{proof}
Sufficiency follows by approximating the infimum before the collision
limit. For necessity, the positive remainder and the pointwise assertion
give ordered height at most $C_MB^{-1/192}$ for $b^{\varepsilon(B)}$.
Both $E_h^\varepsilon$ and $O_h^\varepsilon$ are bounded by this same
positive density at the starting roof. Their separate ordered heights
are therefore at most this bound, uniformly in $r$. Take the infimum
and then $B\to\infty$. This argument does not claim that the infimum
is attained or select a width depending on $m$.
\end{proof}
