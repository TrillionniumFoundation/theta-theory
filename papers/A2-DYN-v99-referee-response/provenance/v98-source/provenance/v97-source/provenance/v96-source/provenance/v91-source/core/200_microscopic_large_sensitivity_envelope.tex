\section{A finite-count envelope on collapsing optical scales}
\label{sec:v86-envelope}
Put $\kappa=(1-\eta)/2$ and fix admissible $p,\eta$ as in
Theorem~\ref{thm:v86-moments}. Choose
\begin{equation}\label{eq:v86-exponents}
 \gamma>\frac{\kappa}{p-1},\qquad d=\gamma(p-1)-\kappa>0.
\end{equation}
The following source is defined directly through the common physical
fiber kernels, not through separately chosen versions of a density:
\begin{equation}\label{eq:v86-large-source}
 \mathcal D_{\eta,\gamma,h}(t)
 =\int_0^h s^{-\kappa}
       \int I_m(y)z(y)^{-\eta}
                 \one_{\{I_m(y)>s^{-\gamma}\}}\,d\mu_{t+s}(y)\,ds.
\end{equation}
It contains every actual word of the fixed exact event. The selector
may shrink with the microscopic offset because the input to which it
will be compared is a finite-count estimate uniform in all intervals.
The Fourier band remains fixed.

\begin{theorem}[A complete microscopic large-sensitivity envelope]
\label{thm:v86-envelope}
For $0<h<h_0<1$, every $m\ge2$ and every fixed band $B_*$,
\begin{align}
 m^2\|\mathcal D_{\eta,\gamma,h}\|_\infty
 \le{}&C_{p,\eta,\gamma}
              (h^{d+1}+B_*^{-1}h^d)\notag\\
 &+C_{B_*,p,\eta,\gamma}(1+B_*h)m^2
                    \rho_{B_*}^{m\sigma_p/2}h^d.
                                         \label{eq:v86-envelope-finite}
\end{align}
The essential supremum is global in the roof variable and uniform
in radius and exact labels. In particular, for each fixed band there
is a finite $K_{B_*,p,\eta,\gamma}$ such that
\begin{equation}\label{eq:v86-uniform-dini}
 \sup_{m\ge2,R,n,k}m^2\|\mathcal D_{\eta,\gamma,h}\|_\infty
                       \le K_{B_*,p,\eta,\gamma}h^d.
\end{equation}
After the collision limit and then enlargement of the auxiliary band,
\begin{equation}\label{eq:v86-large-height}
 \limsup_m\sup_{R,n,k}m^2\|\mathcal D_{\eta,\gamma,h}\|_\infty
                             \le C_{p,\eta,\gamma}h^{d+1}.
\end{equation}
Thus the entire infinite microscopic tail of this actual selected
source has zero ordered concentration charge.
\end{theorem}
\begin{proof}
Let $\ell_j=2^{-j}h$. On $\ell_j/2<s\le\ell_j$ the indicator
in \eqref{eq:v86-large-source} implies $I_m>\ell_j^{-\gamma}$,
and consequently
\[
 I_m\one_{\{I_m>s^{-\gamma}\}}
                       \le\ell_j^{\gamma(p-1)}I_m^p.
\]
Also $s^{-\kappa}\le2^\kappa\ell_j^{-\kappa}$. Tonelli therefore
gives, at finite count,
\begin{equation}\label{eq:v86-dyadic-majorant}
 \mathcal D_{\eta,\gamma,h}(t)
 \le2^\kappa\sum_{j=0}^{\infty}\ell_j^d
                \int_{\ell_j/2}^{\ell_j}V_{p,\eta}(t+s)\,ds.
\end{equation}
For a finite partial sum apply \eqref{eq:v86-full-moment} to each
actual interval $(t+\ell_j/2,t+\ell_j]$. The leading terms are
bounded by $C\ell_j^d(\ell_j+B_*^{-1})$, and the remainders by
$C_{B_*}m^2\rho_{B_*}^{m\sigma_p/2}
\ell_j^d(1+B_*\ell_j)$. Both series are geometric since $d>0$.
Their sums give \eqref{eq:v86-envelope-finite}, first for all
partial sums and then for the entire nonnegative source by monotone
convergence. The countable dyadic comparisons use one common
almost-everywhere fiber version. No infinite sum of separate
collision limsups occurs.

At a fixed $B_*$, the supremum of
$m^2\rho_{B_*}^{m\sigma_p/2}$ over $m\ge2$ is finite.
This proves the finite-count uniform envelope
\eqref{eq:v86-uniform-dini}. For \eqref{eq:v86-large-height}, first
take the collision limsup with $h$ and $B_*$ fixed, and only then
let $B_*\to\infty$. The leading constant is band independent.
The same bounds apply to the complete tail below $2^{-N}h$ by
replacing $h$ with $2^{-N}h$. In particular its finite-count bound
decays as $2^{-Nd}$ before any collision limit.
\end{proof}

\begin{corollary}[Explicit exponents]\label{cor:v86-explicit}
One admissible choice is
\[
 p=33/32,\qquad\eta=1/4,\qquad\sigma_p=31/64,\qquad
 \gamma=16,\qquad\kappa=3/8,\qquad d=1/8.
\]
The finite-count microscopic envelope is $K_{B_*}h^{1/8}$, and
its ordered essential-height bound is $Ch^{9/8}$.
\end{corollary}
\begin{proof}
Here $p<17/16$, $\eta<31/64$ and
$16(33/32-1)-3/8=1/8$. Substitute these identities in the theorem.
\end{proof}

\subsection{What is removed from the fractional charge}
Define $\mathsf P_{\eta,h}^{\rm mod}$ by replacing the indicator
in \eqref{eq:v86-large-source} with
$\one_{\{I_m\le s^{-\gamma}\}}$. At finite count,
\[
 \mathsf P_{\eta,h}v^\eta
       =\mathsf P_{\eta,h}^{\rm mod}+\mathcal D_{\eta,\gamma,h}.
\]
Each side uses the original source; this is an exact positive
partition. With $\mathcal H_M$ as in the preceding section,
\begin{equation}\label{eq:v86-canonical-localization}
 \mathfrak A_{\eta,M}
   =\lim_{h\downarrow0}\mathcal H_M(\mathsf P_{\eta,h}^{\rm mod}).
\end{equation}
Indeed positivity gives one inequality, and subadditivity of
$\mathcal H_M$ with \eqref{eq:v86-large-height} gives the other.
The equality holds also for infinite charges. The theorem removes a
complete high-sensitivity microscopic sector, not the whole optical
charge. It gives no pointwise estimate for its complementary sector.
