\begin{abstract}
\noindent\textbf{Platform: }\texttt{OB3-PE-v1}.\par
We identify and represent the physical theta-semigroups derived from specular
billiard path conditioning in Papers I--III.  The conjugate field is already
fixed mechanically by $\theta=I_\rho'(r)$; the present paper does not choose it
again.

A cash-additive certainty-equivalent rigidity theorem shows that, among smooth
law-based certainty equivalents, only the affine expectation and the
exponential family are compatible with translation by deterministic work.
Relevance and the nonzero mechanically selected conjugate field leave the
entropic branch.  For the fixed-law physical semigroup the terminal map is
analytic, its first derivative is a payoff-dependent tangent probability, and
its second derivative is the corresponding covariance times $\theta$.
Tangent kernels form a nonlinear chain-rule cocycle.

The finite Doob-normalized Ruelle evaluations have exact exponential tangent
measures.  Deterministic homogenization passes those measures and their first
two derivative operators to the diffusion limit.  For smooth data the tangent
density is a genuine stochastic exponential, giving a valid Girsanov theorem,
a quadratic BSDE, and a linear tangent BSDE.  Optimized values are kept
separate: Paper IV's saddle-envelope correction is retained in their second
variation.
\end{abstract}

\section{Cash-additive certainty-equivalent rigidity}\label{sec:rigidity}
Let $u:\R\to\R$ be strictly increasing and $C^2$, and define for bounded
random variables
\[
 \mathfrak C_u(X)=u^{-1}(\E u(X)).
\]

\begin{theorem}[Affine-or-exponential rigidity]\label{thm:utility-rigidity}
Assume
\[
 \mathfrak C_u(X+c)=\mathfrak C_u(X)+c
\]
for every bounded $X$ and every constant $c$.  Then, after an affine change of
scale in $u$, either
\[
 u(x)=x
\]
or
\[
 u(x)=e^{\vartheta x}
\]
for one nonzero constant $\vartheta$.
\end{theorem}
\begin{proof}
Apply cash additivity to two-point random variables.  Equality of certainty
equivalents for every mixing probability implies that for each $c$ there are
numbers $a(c)>0,b(c)$ such that
\[
 u(x+c)=a(c)u(x)+b(c)
\]
for all $x$.  The composition rule for two translations gives
$a(c+d)=a(c)a(d)$.  Continuity yields either $a(c)=1$ for all $c$ or
$a(c)=e^{\vartheta c}$.  In the first case the additive equation makes $u$
affine.  In the second, subtracting a fixed point of the affine action and
rescaling gives the exponential form.
\end{proof}

\begin{corollary}[Mechanical identification]\label{cor:mechanical-theta}
For the path preparation of Paper I, the nonzero branch in
\cref{thm:utility-rigidity} has
\[
 \vartheta=\theta_\rho(r)=I_\rho'(r).
\]
Thus axiomatic rigidity identifies the functional form, while the billiard
large-deviation principle fixes its parameter.
\end{corollary}

This division is compatible with law-invariant, relevant, strongly
time-consistent rigidity results for dynamic monetary evaluations
\cite{KupperSchachermayer2009}.

\section{The fixed-law physical semigroup}\label{sec:fixed-law}
Let
\[
 dX_s=b(X_s)ds+\sqrt{A_{\rho,r}}\,dW_s,
 \qquad X_t=x,
\]
and put $\theta=\theta_\rho(r)$.  At $\theta=0$ all logarithmic
formulas use the continuous linear limit.  For bounded $\phi$ define
\[
 \cE_{t,T}^{\rho,r}[\phi](x)
 =\frac1\theta\log\E_{t,x}e^{\theta\phi(X_T)}.
\]

\begin{theorem}[Analytic terminal map]\label{thm:analytic}
For fixed $(t,x)$ the map
$C_b(\R)\ni\phi\mapsto\cE_{t,T}^{\rho,r}[\phi](x)$ is real analytic.  If
\[
 \frac{d\Qp_{t,x}^{\phi}}{d\Pp_{t,x}}
 =\frac{e^{\theta\phi(X_T)}}
 {\E_{t,x}e^{\theta\phi(X_T)}},
\]
then
\[
 D\cE[\phi]\eta
 =\E^{\Qp^\phi}\eta(X_T),
\]
\[
 D^2\cE[\phi](\eta_1,\eta_2)
 =\theta\Cov_{\Qp^\phi}
 (\eta_1(X_T),\eta_2(X_T)).
\]
Higher derivatives are tilted cumulants multiplied by powers of $\theta$.
\end{theorem}
\begin{proof}
The exponential Nemytskii map is analytic on bounded balls of $C_b$.
Expectation is positive and bounded, and the logarithm is analytic away from
zero.  Differentiate the composition.
\end{proof}

\section{Tangent kernels and characterization}\label{sec:tangent}
For $s<t$ and bounded $\psi$, let
\[
 K_{s,t}^{\psi}(x,dy)
 =\frac{e^{\theta\psi(y)}P_{s,t}(x,dy)}
 {P_{s,t}e^{\theta\psi}(x)}.
\]

\begin{theorem}[Tangent-kernel cocycle]\label{thm:cocycle}
The kernels satisfy
\[
 K_{r,t}^{\phi}
 =K_{r,s}^{\cE_{s,t}[\phi]}K_{s,t}^{\phi}.
\]
Equivalently, terminal derivatives obey the nonlinear chain-rule cocycle.
\end{theorem}
\begin{proof}
Use the Markov property and cancellation of the intermediate exponential
normalization.
\end{proof}

\begin{theorem}[Covariance-curvature characterization]\label{thm:characterization}
Let $\mathfrak E:L^\infty(E)\to\R$ be monotone, cash additive, normalized at
zero, and twice Frechet differentiable.  Suppose for every $\phi$ there is a
probability $Q_\phi$ such that
\[
 D\mathfrak E(\phi)f=Q_\phi(f),
\]
\[
 D^2\mathfrak E(\phi)(f,g)
 =\theta\{Q_\phi(fg)-Q_\phi(f)Q_\phi(g)\}.
\]
Then
\[
 \mathfrak E(\phi)
 =\frac1\theta\log Q_0(e^{\theta\phi}),
 \qquad
 \frac{dQ_\phi}{dQ_0}
 =\frac{e^{\theta\phi}}{Q_0(e^{\theta\phi})}.
\]
The dynamic version, together with the cocycle, characterizes the entropic
semigroup of the zero-payoff tangent Markov family.
\end{theorem}
\begin{proof}
Set $Q_t=Q_{t\phi}$.  Differentiation gives the measure-valued ODE
\[
 \frac d{dt}Q_t(f)
 =\theta\{Q_t(f\phi)-Q_t(f)Q_t(\phi)\}.
\]
Its unique solution is the exponential tilt of $Q_0$.  Integrate
$d\mathfrak E(t\phi)/dt=Q_t(\phi)$ from zero to one.  Apply the result at every
time interval for the dynamic statement.
\end{proof}

\section{Microscopic Ruelle tangent laws}\label{sec:micro}
Let $\Pp_\varepsilon$ be the finite physical-path law generated by the
Doob-normalized Ruelle preparation and the deterministic slow recursion of
Paper III.  For a bounded terminal payoff $\phi$, define
\[
 \frac{d\Qp_\varepsilon^\phi}{d\Pp_\varepsilon}
 =\frac{e^{\theta\phi(X_T^\varepsilon)}}
 {\E_{\Pp_\varepsilon}e^{\theta\phi(X_T^\varepsilon)}}.
\]

\begin{theorem}[Microscopic tangent convergence]\label{thm:micro-conv}
For bounded continuous $\phi,\eta_1,\eta_2$,
\[
 \Qp_\varepsilon^\phi\Longrightarrow\Qp^\phi,
\]
and the first two terminal derivative operators converge locally uniformly in
the initial slow state.  The conclusion uses the fixed-law or fixed-policy
path law; it does not apply to a changing optimizer without Paper IV's envelope
correction.
\end{theorem}
\begin{proof}
Paper III gives weak convergence of the complete slow paths.  Boundedness of
$\phi$ makes the exponential densities uniformly bounded above and below.
Weak convergence therefore passes to every tilted numerator, denominator, and
the products $\eta_1\eta_2$.
\end{proof}

\section{Genuine Girsanov change of measure}\label{sec:girsanov}
Let
\[
 u(t,x)=\frac1\theta\log\E_{t,x}e^{\theta\phi(X_T)}
\]
for smooth bounded data.  Put
\[
 M_s=\exp\{\theta u(s,X_s)-\theta u(t,x)\}.
\]

\begin{theorem}[Tangent density process]\label{thm:girsanov}
The process $M$ is the uniformly integrable martingale
\[
 M_s=\frac{\E[e^{\theta\phi(X_T)}\mid\mathcal F_s]}
 {\E e^{\theta\phi(X_T)}}
\]
and
\[
 M_s=\mathcal E\left(
 \int_t^s\theta\sqrt{A_{\rho,r}}\,u_x(q,X_q)dW_q
 \right).
\]
Under $\Qp^\phi$ defined by $M_T$,
\[
 W_s^{\Qp}=W_s-
 \int_t^s\theta\sqrt{A_{\rho,r}}\,u_x(q,X_q)dq
\]
is Brownian motion and
\[
 dX_s=
 \left(b+\theta A_{\rho,r}u_x\right)(X_s)ds
 +\sqrt{A_{\rho,r}}dW_s^{\Qp}.
\]
\end{theorem}
\begin{proof}
The conditional-expectation identity gives uniform integrability.  Apply
Ito's formula to $e^{\theta u(s,X_s)}$ and use Paper III's theta-PDE to cancel
the drift.  The stochastic logarithm is BMO for bounded terminal data and
bounded uniformly elliptic coefficients; localization and uniform
integrability justify Girsanov.  No inversion of the gradient process is used.
\end{proof}

\section{Quadratic and tangent BSDEs}\label{sec:bsde}
Let
\[
 Y_s=u(s,X_s),\qquad
 Z_s=\sqrt{A_{\rho,r}}\,u_x(s,X_s).
\]

\begin{theorem}[BSDE bundle]\label{thm:bsde}
The fixed-law entropic solution satisfies
\[
 Y_s=\phi(X_T)+
 \int_s^T\frac\theta2\abs{Z_q}^2dq
 -\int_s^TZ_qdW_q.
\]
For a terminal direction $\eta$, the tangent process is a martingale under the
law $\Qp^\phi$:
\[
 V_s=\eta(X_T)-\int_s^T\widetilde Z_qdW_q^{\Qp}.
\]
\end{theorem}
\begin{proof}
Apply Ito's formula to $u(s,X_s)$ and use the theta-PDE.  Differentiate the PDE
in terminal data and use \cref{thm:girsanov} to absorb the first-order tangent
drift \cite{PardouxPeng1990,Kobylanski2000}.
\end{proof}

\section{Controlled second variations}\label{sec:controlled}
For an optimized risk-sensitive value, Paper IV proves
\[
 D^2V
 =\theta\Cov_{Q^*}
 -G_{\phi z}G_{zz}^{-1}G_{z\phi}.
\]
The second term is part of the theorem, not an error.  Consequently the pure
covariance-curvature characterization in \cref{thm:characterization} is a
fixed-law/fixed-policy result.  A controlled value belongs to a different
representation class unless the optimizer-response term vanishes.

\section{Main theorem and scope}\label{sec:main}
\begin{theorem}[Billiard-derived theta representation theorem]\label{thm:main}
The physical theta-semigroup derived from the specular path ensemble has:
\begin{enumerate}[label=(\roman*)]
\item a mechanically fixed parameter $\theta=I_\rho'(r)$;
\item an analytic terminal map and probability-valued tangent cocycle;
\item covariance second variation in the fixed-law and fixed-policy types;
\item convergent microscopic Ruelle tangent laws;
\item a genuine Girsanov density and quadratic/tangent BSDEs;
\item a separately typed optimizer-response correction for controlled values.
\end{enumerate}
Axiomatic rigidity confirms the entropic form but does not replace the
mechanical selection of $\theta$.
\end{theorem}
\begin{proof}
Combine \cref{thm:utility-rigidity,cor:mechanical-theta,thm:analytic,thm:cocycle,thm:characterization,thm:micro-conv,thm:girsanov,thm:bsde} and
Paper IV's envelope theorem.
\end{proof}

\section*{Scope and review status}
The core representation theorem is fixed-law/fixed-policy.  Optimized games
carry the explicit envelope correction.  Independent external review is
