\begin{abstract}
\noindent\textbf{Platform: }\texttt{OB3-PE-v1}.\par
We develop controlled and partially observed extensions of the fixed-law
specular theta-semigroup of Paper III.  The base path ensemble, physical
covariance, clock, and endogenous parameter remain those of the same
open-billiard platform.  Controls act only on the slow collision port and
running work; they do not alter the fast obstacle geometry or silently replace
the prepared Gibbs state.

A deterministic refresh--autoregressive product shift supplies a noncompact
hidden coordinate.  Bounded nondegenerate observations, a common refresh
component, and a Lyapunov drift give a uniform posterior moment ball and
strategy-tree filter contraction.  An exact-observation no-go theorem keeps
this branch separate from full fast-state information.

For optimized risk-sensitive values we prove a saddle-envelope theorem.  The
first terminal derivative is expectation under the optimal tilted law, while
the second derivative equals the covariance term minus an explicit
optimizer-response correction.  Quadratic actuator energies give a strongly
monotone saddle inclusion and a unique Lipschitz pure selector.  This closes
the type gap between fixed-law tangent theory and controlled theta-games.
\end{abstract}

\section{Typed controlled values}\label{sec:types}
Let $A=A_{\rho,r}>0$ and $\theta=\theta_\rho(r)$ be the coefficients of
Paper III.  Expressions divided by $\theta$ use their continuous expectation
limit when $\theta=0$.  Let $U\subset\R^m$ and $V\subset\R^n$ be closed convex
action sets.  Controls change only bounded Lipschitz functions $b^{uv}(x)$ and
$c^{uv}(x)$.  The fast specular billiard, its preparation, and its physical
clock are action independent.

For a fixed feedback pair $(u,v)$, the value is the fixed-policy logarithmic
expectation.  The lower optimized value is
\[
 \mathcal V[\Phi]
 =\sup_u\inf_v\frac1\theta\log
 \E^{u,v}\exp\left\{\theta\left[
 \Phi(X_T)+\int_0^Tc^{uv}(X_s)ds\right]\right\}.
\]
The optimization is outside the path logarithm and must not be identified with
a single fixed-law entropic functional.

\section{Curvature-compensated pure saddles}\label{sec:curvature}
Let $F_0(z,u,v)$ be $C^2$ in $(u,v)$ and suppose
\[
 D_{uu}^2F_0\le L_uI,
 \qquad
 D_{vv}^2F_0\ge-L_vI.
\]
Define
\[
 F(z,u,v)=F_0(z,u,v)-\frac\mu2\abs u^2+\frac\nu2\abs v^2,
 \qquad \mu>L_u,\quad\nu>L_v.
\]
Let
\[
 \mathscr G_z(u,v)=(-D_uF(z,u,v),D_vF(z,u,v)).
\]

\begin{theorem}[Pure-saddle selector]\label{thm:saddle}
If $\mathscr G_z+N_{U\times V}$ is coercive, it is strongly maximal monotone
with constant
\[
 \lambda=\min\{\mu-L_u,\nu-L_v\}>0.
\]
There is a unique global pure saddle $(u_*(z),v_*(z))$, and it is Lipschitz in
$z$ whenever $\mathscr G_z$ is uniformly Lipschitz in $z$.
\end{theorem}
\begin{proof}
The symmetric quadratic form of $D\mathscr G_z$ is
\[
 -\dot u^TD_{uu}^2F\dot u+\dot v^TD_{vv}^2F\dot v.
\]
The mixed Hessian terms cancel.  The energy curvatures bound the form below by
$\lambda(\abs{\dot u}^2+\abs{\dot v}^2)$.  Add the maximal monotone normal cone
and use coercivity for existence; strong monotonicity gives uniqueness and the
Lipschitz comparison estimate.
\end{proof}

\section{The risk-sensitive saddle envelope}\label{sec:envelope}
Let $\phi$ be a terminal payoff parameter in a Banach space and let
$G(\phi,z)$ be a twice Frechet differentiable local risk-sensitive saddle
functional, with $z=(u,v)$.  Suppose $z_*(\phi)$ is an interior saddle and
\[
 G_z(\phi,z_*(\phi))=0.
\]
Let
\[
 H_*(\phi)=G_{zz}(\phi,z_*(\phi)).
\]

\begin{theorem}[Saddle-envelope derivatives]\label{thm:envelope}
Assume $H_*(\phi)$ is invertible with uniformly bounded inverse in the signed
saddle norm.  Then $z_*(\phi)$ is differentiable and the optimized value
$V(\phi)=G(\phi,z_*(\phi))$ satisfies
\[
 DV(\phi)\eta=G_\phi(\phi,z_*)\eta,
\]
\begin{equation}\label{eq:envelope-hessian}
 D^2V(\phi)(\eta,\zeta)
 =G_{\phi\phi}(\eta,\zeta)
 -G_{\phi z}[\eta]H_*^{-1}G_{z\phi}[\zeta].
\end{equation}
For a logarithmic path functional, the first term in
\eqref{eq:envelope-hessian} is
$\theta\Cov_{Q_{\phi,z_*}}(\eta,\zeta)$; the second term is the optimizer
response.
\end{theorem}
\begin{proof}
Differentiate $G_z(\phi,z_*(\phi))=0$ to obtain
\[
 Dz_*[\eta]=-H_*^{-1}G_{z\phi}[\eta].
\]
The first derivative of $V$ loses the $G_zDz_*$ term by stationarity.
Differentiating again gives \eqref{eq:envelope-hessian}.  Differentiating a
logarithmic moment at fixed $z$ gives the tilted covariance.
\end{proof}

\begin{remark}
The correction in \eqref{eq:envelope-hessian} can have either sign because a
saddle Hessian is indefinite.  Omitting it would incorrectly identify an
optimized game with a fixed-law entropic semigroup.
\end{remark}

\section{Controlled theta-HJB}\label{sec:controlled-hjb}
Assume the local saddle of
\[
 c^{uv}(x)+b^{uv}(x)p
\]
is unique and Lipschitz, uniformly on compact $(x,p)$ sets, by
\cref{thm:saddle}.

\begin{theorem}[Physical controlled theta limit]\label{thm:controlled}
The lower and upper microscopic risk-sensitive collision schemes converge to
the same bounded viscosity solution of
\[
 -u_t
 =\sup_{u\in U}\inf_{v\in V}
 \{c^{uv}(x)+b^{uv}(x)u_x\}
 +\frac A2u_{xx}+\frac{\theta A}2(u_x)^2.
\]
The local supremum and infimum are attained by the unique pure Lipschitz saddle
selector.  The fast covariance and theta parameter remain the coefficients
selected by the uncontrolled specular path preparation.
\end{theorem}
\begin{proof}
For fixed actions, Paper III gives the local free-energy expansion and
physical-time limit.  Uniformity over the action sets permits the max--min to
pass through the consistency limit.  The pure-saddle theorem identifies the
lower and upper Hamiltonians.  Monotonicity, stability, uniform ellipticity,
and viscosity comparison give convergence
\cite{BarlesSouganidis1991,FlemingSoner2006}.
\end{proof}

\section{A deterministic noncompact hidden extension}\label{sec:hidden}
Independently of the billiard coordinate, let a deterministic product shift
realize the kernel
\[
 P_Y(y,\cdot)=\delta\gamma+(1-\delta)N(qy,1-q^2),
 \qquad 0<\delta<1,\quad\abs q<1,
\]
where $\gamma=N(0,1)$.  The probability is the product preparation of the
initial hidden sequence; the update map itself is deterministic.  Let
$W(y)=1+y^2$ and use binary observations with
\[
 g_o(y)=\frac{1+o\epsilon_o\tanh y}{2},
 \qquad o\in\{-1,1\},\quad0<\epsilon_o<1.
\]

\begin{lemma}[Prediction and Bayes bounds]\label{lem:filter-bounds}
For all probabilities $\mu,\nu$,
\[
 \norm{\mu P_Y-\nu P_Y}_{\TV}
 \le(1-\delta)\norm{\mu-\nu}_{\TV}.
\]
With $\alpha=(1-\delta)q^2$,
\[
 P_YW=\alpha W+2(1-\alpha).
\]
If $0<g_-\le g_o\le g_+$, then the Bayes map is
$2g_+/g_-$-Lipschitz in total variation.
\end{lemma}
\begin{proof}
The common refresh part cancels in signed differences.  The moment identity is
direct.  For Bayes, add and subtract one numerator over the other denominator
and use the lower likelihood bound.
\end{proof}

\begin{theorem}[Soft-filter stability]\label{thm:filter}
Choose $m_*$ so that
\[
 \frac{2g_+}{g_-}(1-\delta)^{m_*}<1,
 \qquad
 \frac{g_+}{g_-}\alpha^{m_*}<1.
\]
Then one block of $m_*$ predictions and one observation contracts total
variation, preserves an explicit quadratic-moment ball, and contracts
uniformly along all common strategy and observation trees.  A vanishing slow
initial layer removes the initial hidden belief from the macroscopic value.
\end{theorem}
\begin{proof}
Iterate \cref{lem:filter-bounds}; the moment update is an affine contraction.
Couple two filters on the same observation tree.  Repeating the strict
contraction gives prior removal before the first order-one slow reward.
\end{proof}

\section{Exact observation is not soft filtering}\label{sec:no-go}
\begin{theorem}[Pre-action exact-observation obstruction]\label{thm:no-go}
Let a centered nonzero hidden payoff $h$ be revealed exactly before a control
$u\in\{-1,1\}$ is selected.  Then
\[
 \int\sup_u uh(y)d\mu(y)=\int\abs{h(y)}d\mu(y)>0,
\]
whereas
\[
 \sup_u\int uh(y)d\mu(y)=0.
\]
Thus exact high-rate pre-action observation cannot be replaced by the blind or
soft-filter Hamiltonian without retaining a sufficient information state.
\end{theorem}
\begin{proof}
Choose $u(y)=\operatorname{sign}h(y)$ after observation.  Before observation,
every fixed $u$ has zero mean reward.
\end{proof}

\section{Main theorem and dependency boundary}\label{sec:main}
\begin{theorem}[Typed specular control extension]\label{thm:main}
Paper III's fixed-law specular theta-semigroup extends to:
\begin{enumerate}[label=(\roman*)]
\item fixed-policy logarithmic evaluations;
\item controlled pure-saddle theta-HJB equations;
\item a differentiable saddle envelope with an explicit optimizer-response
term;
\item a separately typed noncompact soft-filter product extension;
\item an exact-observation no-go branch.
\end{enumerate}
None of these extensions changes the geometry-selected fast coefficients
without a new platform theorem.
\end{theorem}
\begin{proof}
Combine \cref{thm:saddle,thm:envelope,thm:controlled,thm:filter,thm:no-go}.
\end{proof}

\section*{Scope and review status}
The filtering product is an extension, not part of the first-principles
specular core.  The controlled second variation is not asserted to be a pure
covariance.  Independent review remains pending.
