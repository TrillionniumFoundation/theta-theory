\section{Zero-width trace concentration by positive absorption}
\label{sec:v67-stratified-trace}
Keep $\chi,\varepsilon$ fixed, and use the same disjoint physical
critical charts as in Section~\ref{sec:v66-collar-trace}, restricted
to clearance seams. For a fixed $K>0$ define the exact-label stratum
\[
 \mathcal Z_{m,R,\lambda}(K)
   =\{z:\theta_{z,\lambda}>0,\ \mathfrak a_{z,\lambda}\le K\}.
\]
Selection depends only on the center data and the exact label, not
on the collar width. Define
\begin{align}
 \mathcal B^K_{m;\lambda,R}
    &=\sum_{z\in\mathcal Z_{m,R,\lambda}(K)}
                   \beta_{z,\lambda}\delta_{t_z},\notag\\
 \mathcal Q^K_{m,s;\lambda,R}
    &=\sum_{z\in\mathcal Z_{m,R,\lambda}(K)}
       \left(s^{-1}\int_0^s b_{z,\lambda}^{\varepsilon,\mathrm{clr}}
                                      (t_z+u)\,du\right)\delta_{t_z}.
                                                        \label{eq:v67-stratified-traces}
\end{align}
Let $\mathcal D^K_{m,s}$ denote the same restriction of
\eqref{eq:v66-angular-loss}. Subscripts suppressed below are unchanged.
Theorem~\ref{thm:v66-ordered-collar-trace} applies to these subfamilies:
selection only removes positive source pieces. No extra regularity of
the selected family is needed for that comparison.

\begin{theorem}[An ordered zero-width bound on finite-jet strata]
\label{thm:v67-stratified-trace}
For every fixed $K,\chi,\varepsilon$ and $h>0$,
\begin{equation}\label{eq:v67-stratified-concentration}
 \limsup_{m\to\infty}\sup_{R,\lambda,t}
       m^2\mathcal B^K_{m;\lambda,R}([t-h,t+h])\le2Ch,
\end{equation}
where $C$ is the collar constant, independent of $K$.
Moreover, for $s<h_\chi$ with $K\sqrt{2s}<1$,
\begin{align}
 \mathcal B^K_m(I)
 &\le\frac{e^{K_\chi\sqrt{2s}}}{1-K\sqrt{2s}}
                            \mathcal Q^K_{m,s}(I),       \label{eq:v67-absorption}\\
 \mathcal D^K_{m,s}(I)
 &\le\tfrac23K\sqrt{2s}\,\mathcal B^K_m(I).             \label{eq:v67-stratified-loss}
\end{align}
In particular
\begin{equation}\label{eq:v67-ordered-stratified-loss}
 \limsup_{m\to\infty}\sup_{R,\lambda,t}
      m^2\mathcal D^K_{m,s}([t-1,t+1])
                  \le\tfrac43 C K\sqrt{2s}.
\end{equation}
The statements require no concentration estimate for
$\sum_zJ_z\delta_{t_z}$.
\end{theorem}
\begin{proof}
For $0<u<s$, Theorem~\ref{thm:v67-relative-angular} and the lower
multiplicative density estimate give
\[
 b_{z,\lambda}^{\varepsilon,\mathrm{clr}}(t_z+u)
  \ge e^{-K_\chi\sqrt{2s}}(1-K\sqrt{2s})\beta_{z,\lambda}.
\]
Average over $u$, then sum at the unchanged critical values in $I$.
This proves \eqref{eq:v67-absorption}. All coefficients are positive;
coincident critical values simply add. Alternatively the same inequality
absorbs a relative angular-loss term into the left hand side of
\eqref{eq:v66-atomic-collar-comparison}.

Fix $s,h,K,\chi,\varepsilon$ before taking the collision limsup.
The positive collar comparison gives
\[
 \limsup_m\sup_{R,\lambda,t}m^2\mathcal B^K_m([t-h,t+h])
 \le C\frac{e^{K_\chi\sqrt{2s}}}{1-K\sqrt{2s}}(2h+s).
\]
The left side does not depend on $s$. Let $s\downarrow0$ now, with
$K,h,\chi,\varepsilon$ still fixed, to obtain
\eqref{eq:v67-stratified-concentration}. This does not interchange
$m$ with the fixed-count limit of $\mathcal Q_{m,s}$.

The one-sided angular estimate is at most
$K\sqrt{2u}\theta_{z,\lambda}$ on the stratum. Since
$s^{-1}\int_0^s\sqrt{2u}\,du=\tfrac23\sqrt{2s}$,
it proves \eqref{eq:v67-stratified-loss}. Apply
\eqref{eq:v67-stratified-concentration} with $h=1$ to obtain
\eqref{eq:v67-ordered-stratified-loss}.
\end{proof}

\subsection{A pointwise height estimate for the original caustic collars}
For $H>0$ define a positive density on the original source by
\begin{equation}\label{eq:v67-stratified-collar-density}
 b^{K,H}_{m;\lambda,R}(t)
 =\sum_{z\in\mathcal Z_{m,R,\lambda}(K)}
    \one_{\{0<t-t_z<H\}}\,
                 b_{z,\lambda}^{\varepsilon,\mathrm{clr}}(t).
\end{equation}
The width $H$ is a roof width, whereas the chart radius is $\sqrt{2H}$.

\begin{corollary}[Ordered essential height on each stratum]
\label{cor:v67-collar-height}
If $H<h_\chi$ and $K\sqrt{2H}<1$, then
\begin{equation}\label{eq:v67-collar-height}
 \limsup_{m\to\infty}\sup_{R,\lambda}
    m^2\|b^{K,H}_{m;\lambda,R}\|_\infty
 \le C H e^{K_\chi\sqrt{2H}}(1+K\sqrt{2H}).
\end{equation}
Thus this original-source essential height tends to zero as
$H\downarrow0$ after the collision limit, with $K,\chi,\varepsilon$
fixed. A bounded insertion is dominated by its supremum norm times
the same right hand side.
\end{corollary}
\begin{proof}
The upper multiplicative profile and
\eqref{eq:v67-relative-angular} bound each summand, on the common
coarea representative, by
$e^{K_\chi\sqrt{2H}}(1+K\sqrt{2H})\beta_{z,\lambda}$.
Contributing centers lie in $[t-H,t]$. Apply
\eqref{eq:v67-stratified-concentration} with center $t-H/2$ and
radius $H/2$. At each finite count the union of the exceptional roof
sets is null, so this gives an essential-supremum estimate for the
summed density. Source domination proves the insertion statement.
\end{proof}

\paragraph{What is uniform.}
The constant in \eqref{eq:v67-stratified-concentration} is independent
of the cutoff $K$, but the collision-count convergence is not asserted
to be uniform in $K$. Every fixed word with positive angular fraction
belongs to some finite stratum; this fixed-count exhaustion does not
permit $K\to\infty$ before $m\to\infty$. The next section keeps the
missing tail as a positive measure. Likewise, no height for an annulus
or for a zero-angular-fraction source follows from
\eqref{eq:v67-collar-height}.
