\begin{abstract}
\noindent\textbf{Platform: }\texttt{OB3-PE-v1}.\par
We derive a nonlinear physical-time semigroup from the same genuine specular
billiard used in Papers I and II.  A macroscopic impact-frequency preparation
$r$ first selects the conjugate parameter
$\theta=I_\rho'(r)$.  The normalized Ruelle operator of the tilted geometric
potential is then a Markov disintegration of the deterministic invariant path
ensemble.  Its logarithmic transform is an exact Doob-normalized excess
pressure, not an independently postulated exponential utility.

We write the microscopic slow recursion and its exact conditional free-energy
DPP on the symbolic fast state.  Paper II's deterministic physical-time
homogenization implies convergence of the full path laws, and bounded
exponential functionals therefore converge to the entropic functional of the
limiting diffusion.  The resulting theta-generator is
\[
 \mathfrak G_{\rho,r}f
 =c+b f'+\frac{A_{\rho,r}}2f''
 +\frac{\theta_\rho(r)A_{\rho,r}}2(f')^2.
\]
An exponential random-horizon estimate closes the passage from collision count
to the actual free-flight clock.  Fixed-law, fixed-policy, and optimized
risk-sensitive values are kept as distinct types.
\end{abstract}

\section{The normalized conditional path operator}\label{sec:path-op}
Fix $(\rho,r)$ and abbreviate
\[
 \theta=\theta_\rho(r),\qquad
 \cP=\cP_{\rho,\theta},\qquad
 g=J-r,\qquad
 \tau=\widehat\tau_\rho.
\]
The operator $\cP$ acts on a one-sided symbolic state $z\in\Sigma^+$ by
averaging over admissible preimages $y$ with $\sigma y=z$.  This is the
reverse-time conditional law of the two-sided deterministic billiard ensemble.
Let $\mathscr R$ reverse the obstacle code and reverse the collision velocity.
The specular law gives
$B_\rho^{-1}=\mathscr R B_\rho\mathscr R$, while the impact observable and
free-flight roof are invariant under the corresponding edge reversal.  Hence
the reverse Ruelle chain is a forward conditional description after applying
$\mathscr R$.  It is a disintegration of an invariant deterministic path
measure, not microscopic random forcing.

For bounded $F$ define
\[
 \cE_n^{\theta}[F](z)
 =\frac1\theta\log\cP^n(e^{\theta F})(z),
\]
with continuous extension at $\theta=0$.

\begin{proposition}[Exact nonlinear tower]\label{prop:tower}
The operators $\cE_n^\theta$ are monotone, cash additive, constant preserving,
and
\[
 \cE_{n+m}^\theta
 =\cE_n^\theta\circ\cE_m^\theta.
\]
They coincide with the Doob-normalized excess pressure of Paper I.
\end{proposition}
\begin{proof}
Use $\cP^{n+m}=\cP^n\cP^m$, exponentiate, and take the logarithm.  The
partition-ratio identity is Paper I's exact Ruelle formula.
\end{proof}

\section{The microscopic slow path and DPP}\label{sec:dpp}
Let $b,c:\R\to\R$ be bounded globally Lipschitz and let
$\Phi:\R\to\R$ be bounded Lipschitz.  A transition $z\leftarrow y$ sampled by
$\cP$ produces
\[
 x^+=x+\varepsilon g(y)
 +\varepsilon^2\tau(y)b(x),
 \qquad
 t^+=t+\varepsilon^2\tau(y).
\]
For a deterministic collision horizon $N$, put
$V_N^\varepsilon(x,z)=\Phi(x)$ and define backwards
\begin{equation}\label{eq:DPP}
\begin{aligned}
 V_k^\varepsilon(x,z)
 =\frac1\theta\log\cP\Bigl[
 \exp\{\theta(&\varepsilon^2\tau(y)c(x)\\
 &+V_{k+1}^\varepsilon(
 x+\varepsilon g(y)+\varepsilon^2\tau(y)b(x),y))\}
 \Bigr](z).
\end{aligned}
\end{equation}

\begin{theorem}[Exact canonical Ruelle free-energy DPP]\label{thm:DPP}
Equation \eqref{eq:DPP} is the exact finite-horizon conditional excess pressure
of the canonical tilted preparation selected by Paper I.  Paper I's central-
window Gibbs-conditioning theorem identifies this canonical preparation as the
local limit of finite microcanonical impact conditioning.  The theorem does
not claim that a finite-horizon microcanonical conditioning is itself Markov.
\end{theorem}
\begin{proof}
Condition the Doob-normalized canonical path partition function on the first
predecessor $y$.  Add the actual physical-time work
$\varepsilon^2\tau(y)c(x)$ and update the deterministic slow coordinate.
Apply \cref{prop:tower} and proceed by backward induction.  The
microcanonical interpretation is asymptotic and follows separately from Paper
I's conditioning theorem.
\end{proof}

\section{Stationary preparation and optional boundary forgetting}\label{sec:fast-collapse}
The load-bearing semigroup starts the fast billiard in the stationary
canonical equilibrium state $\nu_{\rho,\theta}$.  Thus no unproved uniform
collapse from an arbitrary symbolic boundary condition is needed in the main
theorem.

For later boundary-value comparisons, the normalized Ruelle operator has the
standard spectral gap: for Holder $f$,
\[
 \norm{\cP^mf-\nu(f)}_\alpha
 \le C\kappa^m\norm f_\alpha,
 \qquad 0<\kappa<1.
\]

\begin{proposition}[Optional terminal-state forgetting]\label{thm:fast-collapse}
For every bounded Holder terminal observable $F:\Sigma^+\to\R$ with a fixed
Holder norm bound,
\[
 \osc_z\left[\frac1\theta\log\cP^m(e^{\theta F})(z)\right]
 \le C_F\kappa^m,
\]
with the continuous linear interpretation at $\theta=0$.  Consequently a
nonstationary terminal symbolic boundary can be separated from a macroscopic
observation by $m_\varepsilon\asymp|\log\varepsilon|$ collisions at physical
cost $O(\varepsilon^2|\log\varepsilon|)$, provided the terminal observables
have a uniform Holder bound.
\end{proposition}
\begin{proof}
Apply the spectral decomposition to the positive Holder function
$e^{\theta F}$.  Uniform upper and lower bounds permit taking logarithms.  The
roof and running work are bounded, giving the stated physical cost.  This
proposition is auxiliary; the stationary main chain does not require it.
\end{proof}

\section{Nonlinear homogenization of path functionals}\label{sec:nonlinear-hom}
Let $X^{\varepsilon,x}$ be the physical-time interpolation of the slow
recursion.  Define
\[
 \mathcal W_T^\varepsilon
 =\Phi(X_T^{\varepsilon,x})
 +\int_0^Tc(X_s^{\varepsilon,x})\,ds,
\]
using the piecewise-constant interpolation of the running work.

\begin{theorem}[Exponential functional homogenization]\label{thm:exp-hom}
Under the stationary canonical preparation $\nu_{\rho,\theta}$, uniformly
for $x$ in compact sets,
\[
 \frac1\theta\log
 \E\exp\{\theta\mathcal W_T^\varepsilon\}
 \longrightarrow
 \frac1\theta\log
 \E_x\exp\left\{\theta\left[
 \Phi(X_T)+\int_0^Tc(X_s)\,ds\right]\right\},
\]
where
\[
 dX_t=b(X_t)dt+\sqrt{A_{\rho,r}}\,dW_t.
\]
At $\theta=0$ the assertion is the corresponding expectation convergence.
\end{theorem}
\begin{proof}
Paper II gives uniform weak convergence of the physical slow paths.  The map
\[
 \gamma\mapsto\Phi(\gamma_T)+\int_0^Tc(\gamma_s)ds
\]
is bounded and continuous after the usual harmless polygonal interpolation.
The exponential is therefore bounded and continuous.  Weak convergence passes
to numerator and denominator, and the logarithm is continuous on the positive
range.  Uniformity in the slow initial state follows from the globally Lipschitz slow
coefficients and Paper II's stationary, compact-parameter homogenization
statement.  The optional boundary-forgetting proposition is not used.
\end{proof}

\section{The physical theta-semigroup}\label{sec:semigroup}
For $0\le s\le t$ define
\begin{equation}\label{eq:limit-semigroup}
 \mathfrak E_{s,t}^{\rho,r}[\Phi](x)
 =\frac1\theta\log\E_{s,x}\exp\left\{
 \theta\left[
 \Phi(X_t)+\int_s^tc(X_q)dq\right]\right\}.
\end{equation}

\begin{theorem}[Physical theta-generator]\label{thm:generator}
The operators in \eqref{eq:limit-semigroup} form a monotone, cash-additive,
strongly time-consistent semigroup.  Its bounded viscosity solution satisfies
\begin{equation}\label{eq:theta-pde}
 -u_t
 =c(x)+b(x)u_x
 +\frac{A_{\rho,r}}2u_{xx}
 +\frac{\theta_\rho(r)A_{\rho,r}}2(u_x)^2,
 \qquad u(T,x)=\Phi(x).
\end{equation}
Equivalently, for smooth $f$,
\[
 \mathfrak G_{\rho,r}f
 =c+\frac1\theta e^{-\theta f}
 \left(b\partial_x+\frac{A_{\rho,r}}2\partial_{xx}\right)
 e^{\theta f}.
\]
\end{theorem}
\begin{proof}
The Markov property of the limiting diffusion gives time consistency.
Set $v=e^{\theta u}$.  Linear Feynman--Kac theory gives
\[
 -v_t=bv_x+\frac{A_{\rho,r}}2v_{xx}+\theta c v.
\]
Dividing the derivatives of $v=e^{\theta u}$ by $\theta e^{\theta u}$ gives
\eqref{eq:theta-pde}.  For $\theta=0$ take the continuous linear limit.
The Cole--Hopf transform also gives bounded viscosity comparison.
\end{proof}

\begin{corollary}[Endogenous coefficient identity]\label{cor:endogenous}
The two nonlinear parameters in \eqref{eq:theta-pde} are derived from the same
specular preparation:
\[
 \theta_\rho(r)=I_\rho'(r),
 \qquad
 A_{\rho,r}
 =\frac{\Lambda_\rho''(\theta_\rho(r))}
 {\int\tau_\rho d\nu_{\rho,\theta_\rho(r)}}.
\]
\end{corollary}

\section{Restoring the random physical horizon}\label{sec:clock}
Let $V_{\rm phys}^{\varepsilon}$ stop at the last billiard collision before
physical time $T$, and let $V_{\rm coll}^{\varepsilon}$ use
\[
 n_\varepsilon(T)=\left\lfloor
 \frac{T}{\varepsilon^2\bar\tau_{\rho,r}}\right\rfloor
\]
collisions.

\begin{lemma}[Entropic Orlicz comparison]\label{lem:orlicz}
For bounded random variables $X,Y$ and $\theta\ne0$, define
$dQ_X/dP=e^{\theta X}/Ee^{\theta X}$ and similarly $Q_Y$.  Then
\[
 \abs{\frac1\theta\log Ee^{\theta X}
 -\frac1\theta\log Ee^{\theta Y}}
 \le\frac1{\abs\theta}
 \max_{Q\in\{Q_X,Q_Y\}}
 \log E_Qe^{\abs\theta\abs{X-Y}}.
\]
\end{lemma}
\begin{proof}
Use
$Ee^{\theta X}/Ee^{\theta Y}=E_{Q_Y}e^{\theta(X-Y)}$ and the reverse ratio.
\end{proof}

\begin{theorem}[Physical-horizon entropic replacement]\label{thm:replacement}
On compact subsets of $I_\rho\times(0,1/2)$,
\[
 \sup_x\abs{
 V_{\rm phys}^{\varepsilon}(T,x)
 -V_{\rm coll}^{\varepsilon}(n_\varepsilon(T),x)}
 \le C_T\sqrt\varepsilon.
\]
\end{theorem}
\begin{proof}
Under the stationary canonical preparation, couple the two evaluations on the
same deterministic billiard path.  The
terminal difference is controlled by the random-window current; running work
and drift differ by $O(\varepsilon^2|N_\varepsilon-n_\varepsilon|)$.
Paper II gives exponential moment bounds for both quantities.  Apply
\cref{lem:orlicz}; boundedness of the total payoff controls the tilted
Radon--Nikodym factors.
\end{proof}

\section{Three noninterchangeable value types}\label{sec:types}
\begin{definition}[Typed theta values]
The \emph{fixed-law value} is \eqref{eq:limit-semigroup}.  A
\emph{fixed-policy value} freezes a prescribed feedback in the slow drift and
work and retains the same logarithmic expectation.  A
\emph{controlled risk-sensitive value} takes a supremum, infimum, or saddle
over policies outside the logarithmic path functional.  Only the first two
have a pure covariance second variation without optimizer-response terms.
\end{definition}

This typing is normative for Papers IV and V.  No tangent theorem for a
fixed-law entropic functional is silently imported into an optimized game.

\section{Main theorem and scope}\label{sec:main}
\begin{theorem}[Specular billiard to theta-generator]\label{thm:main}
Within the declared real specular platform,
\[
\begin{aligned}
 \text{geometry-selected path ensemble}
 &\longrightarrow \theta=I'(r)\\
 &\longrightarrow \text{deterministic physical homogenization}\\
 &\longrightarrow \text{canonical Ruelle free-energy DPP}\\
 &\longrightarrow \mathfrak G_{\rho,r}.
\end{aligned}
\]
is a closed theorem chain.  The exponential form and its parameter are both
selected before the macroscopic PDE is written.
\end{theorem}
\begin{proof}
Combine Papers I--II with
\cref{thm:DPP,thm:exp-hom,thm:generator,thm:replacement}.
\end{proof}
