\section{Reconstruction of boundary profiles from physical counts}
\label{sec:v10-acquisition}
The stability theorem of Section~\ref{sec:v9-compatibility} takes a
function and three of its derivatives as data. Here the observations
are instead finitely many binary reports from independent physical
preparations. We recover both full profiles on a closed energy collar,
including its origin. No derivatives of empirical proportions are
supplied as observations.

Fix $0<D\le d_*$ and an even flight number $j$. Write $e_0=e$,
$e_1=\bar e$ and
\begin{equation}\label{eq:v10-observation}
 p_{j,b}(d)=\Pr(E_{j,e_b}(d)),\qquad
 a_j(d)=\frac{2A\sinh(j\gamma)}{d^2},\qquad
 F_{j,b}(d)=a_j(d)p_{j,b}(d).
\end{equation}
The gap $g$, free area $A$, multiplier parameter $\gamma$, and the
selected facing patches are supplied. Thus the window $jg+d$ and its
normalization are known before sampling. Each preparation reports only
whether the selected event occurred; a failure is recorded and charged.
This is a labelled binary experiment, not the unlabelled experiment
of Section~\ref{sec:v8-fixed}.

\subsection{Uniform classes and the statement}
Let $m\ge4$, $0<\beta\le1$ and $B\ge1$. Set
\[
 \|V\|_{m-1,1}=\sum_{r=0}^{m-1}\|V^{(r)}\|_\infty
                         +\operatorname{Lip}(V^{(m-1)}),
\]
and let $\mathcal K=\mathcal K_m(B,\beta;D)$ be the class of real
$C^{m-1,1}([0,D])$ functions with $V(0)=1$, $V\ge\beta$ and
$\|V\|_{m-1,1}\le B$. The true profiles belong to $\mathcal K$.
Assume supplied uniform constants $C_0,M$ and $0<\tau<1$ such that,
for both orientations and every even $j\ge2$,
\begin{equation}\label{eq:v10-physical-bounds}
 \|F_{j,b}-\mathcal F_{bb}\|_{C^3([0,D])}\le C_0\tau^j,
 \qquad 0\le F_{j,b}\le M.
\end{equation}
These are conclusions of Theorem~\ref{thm:v8-main-relative} on each
fixed compact smooth family, with $B,\beta$ chosen for that family.
The theorem below applies to any collection of physical experiments
with these common bounds. It does not postulate that every member
of $\mathcal K$ is realizable by a billiard. The larger class is used
only for regularization. Constants in this section may depend on
$m,D,B,\beta,C_0,M,A,\gamma,\tau$, not on the accuracy or confidence.

For $V,W\in\mathcal K$ define the explicit bilinear map
\begin{equation}\label{eq:v10-forward-map}
 T(V,W)(d)=\frac2\pi\int_{s,t>0,\ s+t<1}
     \frac{1-s-t}{\sqrt{st}}V(ds)W(dt)\,\dd s\dd t,
 \qquad T(V,W)(0)=1.
\end{equation}
Thus $\mathcal F_{bc}=T(\mathcal V_b,\mathcal V_c)$.

\begin{theorem}[Finite-preparation recovery of full profiles]
\label{thm:v10-acquisition}
Under \eqref{eq:v10-physical-bounds}, choose an integer $L\ge m$,
$h=D/L\le1$, an even $j\ge2$, and $0<t,\delta\le1$. At each positive
node $d_k=kh$, $1\le k\le L$, and for each $b=0,1$, make
\begin{equation}\label{eq:v10-allocation}
 N_{b,k}=\left\lceil
 2a_j(d_k)(M+t/3)t^{-2}\log\frac{4L}{\eta}
                       \right\rceil
\end{equation}
independent preparations, where $0<\eta<1/2$.
There is a measurable finite-dictionary estimator
$(\widehat V_0,\widehat V_1)\in\mathcal K^2$, defined below, for which
with probability at least $1-\eta$,
\begin{equation}\label{eq:v10-error}
 \max_{b=0,1}\|\widehat V_b-\mathcal V_b\|_\infty
 \le C\bigl(h^{m-3}+\tau^j+t h^{-3}+\delta\bigr).
\end{equation}
Its prediction $T(\widehat V_0,\widehat V_1)$ of the odd limiting
law has at most $2B$ times the error on the left. Its deterministic
total preparation count satisfies
\begin{equation}\label{eq:v10-budget-general}
 N_{\mathrm{tot}}\le 2L+
 C A\sinh(j\gamma)h^{-2}t^{-2}\log\frac{4L}{\eta}.
\end{equation}
Only positive-offset, even-flight observations are used. Dictionary
cardinality and computational running time are not bounded by this
preparation estimate.
\end{theorem}

\begin{corollary}[An explicit accuracy budget]
\label{cor:v10-budget}
For sufficiently small $\varepsilon>0$, the two profiles and the odd
limiting law can be recovered in uniform norm to accuracy
$\varepsilon$ with confidence $1-\eta$, using at most
\begin{equation}\label{eq:v10-budget}
 C\varepsilon^{-\left(2+\frac8{m-3}
                   +\frac{\gamma}{|\log\tau|}\right)}
                        \log\frac{C}{\eta\varepsilon}
\end{equation}
independent preparations. The windows remain in the fixed collar,
and the even flight number is $O(\log(1/\varepsilon))$.
This is a sufficient cost in the stated labelled experiment; no
optimality over designs or billiard observations is asserted.
\end{corollary}

\subsection{Positive-node differentiation and regularization}
For each cell $J_\ell=[\ell h,(\ell+1)h]$, $0\le\ell<L$, put
\[
 s_\ell=\max\{1,\min\{\ell,L-m+1\}\}.
\]
Let $I_hz$ on $J_\ell$ be the polynomial of degree at most $m-1$
interpolating $z_k$ at the $m$ nodes
$d_{s_\ell},\ldots,d_{s_\ell+m-1}$. We also write $I_hf$ for
interpolation of the sampled values of a function $f$. Polynomials
on adjacent cells need not have matching derivatives. Their norm is
\[
 \|P\|_{3,h}=\max_{0\le\ell<L}\max_{0\le r\le3}
                         \|P^{(r)}\|_{C^0(J_\ell)}.
\]
The stencil on the first cell uses $h,\ldots,mh$. There is neither
a preparation at zero nor an assumed exact value of $F_{j,b}(0)$.
The limited extrapolation to zero uses the same fixed stencil bounds.

\begin{lemma}[Three interpolation estimates]\label{lem:v10-stencils}
With constants depending only on $m,D$, one has
\begin{align}
 \|I_h f-f\|_{3,h}&\le C h^{m-3}\|f\|_{m-1,1},
                   &&f\in C^{m-1,1}([0,D]),\label{eq:v10-interpolation}\\
 \|I_h f\|_{3,h}&\le C\|f\|_{C^3([0,D])},
                   &&f\in C^3([0,D]),\label{eq:v10-stability}\\
 \|I_h z\|_{3,h}&\le C h^{-3}\max_k|z_k|.
                                      \label{eq:v10-noise}
\end{align}
In particular a smooth $C^3$ model error is not amplified by
$h^{-3}$, whereas an arbitrary scalar error can be.
\end{lemma}
\begin{proof}
Rescale a cell by $d=\ell h+hu$. Its interpolation nodes are distinct
integers lying within distance $m$ of $[0,1]$. There are only finitely
many possible relative stencils. Their Lagrange basis polynomials
and derivatives through order three are therefore bounded uniformly
on $[0,1]$. Scaling back gives
$\|(I_hz)^{(r)}\|_{C^0(J_\ell)}\le C h^{-r}\max|z_k|$;
since $h\le1$, this proves \eqref{eq:v10-noise}.

For the first estimate, subtract the degree-$(m-1)$ Taylor polynomial
of $f$ at $\ell h$. The Lipschitz bound on its last derivative gives
remainder $C\|f\|_{m-1,1}h^{m-r}$ after $r\le3$ derivatives
on the cell, and remainder $C\|f\|_{m-1,1}h^m$ at every stencil
node. The interpolant reproduces the Taylor polynomial. Applying the
scaled basis bound to the nodal remainder and adding the differentiated
function remainder proves \eqref{eq:v10-interpolation}.

For \eqref{eq:v10-stability}, instead subtract the quadratic Taylor
polynomial. Its interpolant is itself, since $m\ge4$; its derivatives
through order three are bounded by $C\|f\|_{C^3}$. Its nodal
remainder is bounded by $C h^3\|f\|_{C^3}$, including in the first
and last cells. Scaling the remainder gives
$C h^{3-r}\|f\|_{C^3}$ for $r\le3$, proving the assertion.
\end{proof}

\begin{lemma}[Compact regularization class]\label{lem:v10-dictionary}
The class $\mathcal K$ is compact in $C^{m-1}([0,D])$. The functions
$T(V,V)$, $V\in\mathcal K$, have uniformly bounded
$C^{m-1,1}$ norms. For each $h$ and $\delta>0$, there is a finite
ordered set $\mathcal D_{h,\delta}\subset\mathcal K$ such that
\begin{equation}\label{eq:v10-net}
 \inf_{W\in\mathcal D_{h,\delta}}
    \|I_hT(V,V)-I_hT(W,W)\|_{3,h}\le\delta
                    \quad(V\in\mathcal K).
\end{equation}
Moreover for $V,W\in\mathcal K$,
\begin{equation}\label{eq:v10-inverse-class}
 \|V-W\|_\infty\le C\|T(V,V)-T(W,W)\|_{C^3([0,D])}.
\end{equation}
\end{lemma}
\begin{proof}
The derivatives through order $m-1$ are uniformly bounded and
equicontinuous. Successive use of the fundamental theorem of calculus
identifies the limits of these derivatives under an Arzel\`a--Ascoli
subsequence. The norm bound, positivity, normalization and Lipschitz
bound pass to the limit. This proves compactness.

The weight in \eqref{eq:v10-forward-map} is integrable and has total
mass one. Differentiating under the integral through order $m-1$
uses bounded products of derivatives of $V$, with factors $s,t\le1$.
The same product estimate bounds the Lipschitz seminorm of the last
derivative. It gives a uniform $C^{m-1,1}$ bound. At a fixed grid,
$V\mapsto I_hT(V,V)$ is continuous even for the uniform topology:
the forward nodal values are continuous, and interpolation is a
linear map between finite-dimensional spaces. Its image is compact,
so it admits a finite $\delta$-net with representatives in
$\mathcal K$. Fix any order of those representatives in advance.

Finally the proof of Theorem~\ref{thm:v9-profile-stability} uses only
normalization, the first-derivative bound and the three derivatives
of the forward law. It applies to the present $C^{m-1,1}$ functions
without a smoothness extension. More explicitly, put
$f=x^{-1/2}(V-W)$ and $a=x^{-1/2}(V+W)$. Their Abel convolution
has the form $x^{-1/2}*a=2\pi+R$, with $R(0)=0$ and
$\|R'\|_\infty\le\pi B$. If
$Q=(\pi/2)(d^2[T(V,V)-T(W,W)])''$, then
$2\pi f+R'*f=x^{-1/2}*Q'$. The normalization makes $f$ continuous
at zero. The Abel bound and Gronwall give
$\|V-W\|_\infty\le(D/\pi)e^{BD/2}\|Q'\|_\infty$.
The identity
\[
 Q'=\frac\pi2\left(6\Delta F'+6d\Delta F''+d^2\Delta F'''\right),
 \qquad \Delta F=T(V,V)-T(W,W),
\]
proves \eqref{eq:v10-inverse-class} in the stated regularity class.
\end{proof}

Given nodal observations $z=(z_k)_{k=1}^L$, define
\begin{equation}\label{eq:v10-estimator}
 \widehat V(z)=\text{the first minimizer over }W\in\mathcal D_{h,\delta}
       \text{ of }\|I_hT(W,W)-I_hz\|_{3,h}.
\end{equation}
Each criterion is continuous in the finite data vector. The prescribed
tie rule makes the estimator Borel measurable. Neither the dictionary
nor its order depends on the unknown profile or the observed counts.
This construction specifies a statistical estimator; it does not
claim a tractable enumeration of the dictionary.

\subsection{Proof of the acquisition theorem}
\begin{proof}[Proof of Theorem~\ref{thm:v10-acquisition}]
For orientation $b$ and node $k$, let $\overline X_{b,k}$ be the
empirical proportion of successes and put
$Z_{b,k}=a_j(d_k)\overline X_{b,k}$. Its expectation is exactly
$F_{j,b}(d_k)$, and its variance is
\begin{equation}\label{eq:v10-variance}
 \operatorname{Var}(Z_{b,k})=
 \frac{a_j(d_k)F_{j,b}(d_k)(1-p_{j,b}(d_k))}{N_{b,k}}.
\end{equation}
For a Bernoulli variable $X$ of mean $p$,
$\log\E e^{s(X-p)}\le p(e^s-1-s)$.
The exponential series gives
$e^s-1-s\le s^2/[2(1-s/3)]$ for $0<s<3$; for the negative
exponent use $e^{-s}-1+s\le s^2/2$. Independence, exponential
Markov bounds, and the parameter $s=t/(M+t/3)$ for $X-p$ give
\begin{equation}\label{eq:v10-bernstein}
 \Pr\{|Z_{b,k}-F_{j,b}(d_k)|>t\}
 \le2\exp\left\{-\frac{N_{b,k}t^2}
            {2a_j(d_k)(M+t/3)}\right\}.
\end{equation}
The allocation \eqref{eq:v10-allocation} and a union bound over
$2L$ observations yield an event of probability at least $1-\eta$
on which all scalar errors are at most $t$.

Work on this event and fix $b$. Abbreviate $\mathcal F=\mathcal
F_{bb}$ and $\widehat F=T(\widehat V_b,\widehat V_b)$, where
$\widehat V_b=\widehat V((Z_{b,k})_k)$. The last two bounds of
Lemma~\ref{lem:v10-stencils} give
\begin{equation}\label{eq:v10-discrepancy}
 \|I_hZ_b-I_h\mathcal F\|_{3,h}
                    \le C(t h^{-3}+C_0\tau^j).
\end{equation}
In particular the deterministic relative error in
\eqref{eq:v10-physical-bounds} has not been multiplied by $h^{-3}$.
Choose a dictionary representative for the true profile using
\eqref{eq:v10-net}. The minimizing property and the triangle
inequality imply
\[
 \|I_h\widehat F-I_h\mathcal F\|_{3,h}
       \le \delta+2\|I_hZ_b-I_h\mathcal F\|_{3,h}.
\]
The first interpolation bound, applied separately to $\widehat F$
and $\mathcal F$, now gives
\[
 \|\widehat F-\mathcal F\|_{C^3([0,D])}
       \le C(h^{m-3}+t h^{-3}+\tau^j+\delta).
\]
Both functions are globally $C^3$; the broken norm of their difference
therefore bounds each of its derivatives on the whole closed collar.
Apply \eqref{eq:v10-inverse-class} to prove \eqref{eq:v10-error}.
The mass-one positive weight in \eqref{eq:v10-forward-map} gives
\[
 \|T(V_0,V_1)-T(W_0,W_1)\|_\infty
       \le B(\|V_0-W_0\|_\infty+\|V_1-W_1\|_\infty),
\]
which proves the odd-law prediction bound.

To count preparations, the ceilings contribute at most $2L$.
Since $\sum_{k=1}^L d_k^{-2}\le h^{-2}\sum_{k\ge1}k^{-2}
\le2h^{-2}$, summing \eqref{eq:v10-allocation} gives
\eqref{eq:v10-budget-general}. All preparations, including failures,
have been counted. This completes the proof of
Theorem~\ref{thm:v10-acquisition}.
\end{proof}

\begin{proof}[Proof of Corollary~\ref{cor:v10-budget}]
Choose $h=D/L$ comparable to a sufficiently small constant times
$\varepsilon^{1/(m-3)}$, with $L\ge m$, and put
$t=c\varepsilon h^3$, $\delta=c\varepsilon$.
Choose the least even $j\ge2$ with $\tau^j\le c\varepsilon$.
Writing $\lambda=|\log\tau|$, rounding changes $j$ by less than two
once the lower bound $j\ge2$ is inactive. Consequently
\[
 e^{j\gamma}\le C\varepsilon^{-\gamma/\lambda},\qquad
 h^{-2}t^{-2}=c^{-2}\varepsilon^{-2}h^{-8}.
\]
A small enough fixed $c$, and the corresponding constant in the mesh,
make both profile errors and the odd-law prediction error at most
$\varepsilon$. The logarithm of $4L/\eta$ is bounded by
$C\log(C/(\eta\varepsilon))$. Substitute these estimates into
\eqref{eq:v10-budget-general}; its $2L$ term is of smaller order.
This proves \eqref{eq:v10-budget} with a deterministic design.
\end{proof}

\subsection{What the budget measures}
The three accuracy-dependent contributions to the power in
\eqref{eq:v10-budget} have distinct origins. Scalar accuracy costs
$\varepsilon^{-2}$. Resolving three derivatives while sampling
near a quadratic onset costs $h^{-8}$. Suppressing finite-bridge bias
costs $e^{j\gamma}$, with $j$ chosen from the relative decay
$\tau^j$. No equality between $\tau$ and $e^{-\gamma}$ is assumed;
a slower certified relative rate gives a larger sufficient budget.

The same experiment was implicit in the direct-variance calculation
of the referee memorandum~\cite{A2ReviewV9}. Formula
\eqref{eq:v10-variance} retains that cost rather than treating the
limiting law as a freely observed function. Regularization is needed
because the inverse differentiates the data. Its finite smoothness
bound is an additional quantitative assumption for acquisition, not a
new restriction on the general smooth forward theorem. The recovered
objects remain the weighted, symmetrized energy profiles, not arbitrary
asymmetric boundary graphs. The two oriented even sequences suffice
for their reconstruction and for prediction of the odd law. Separate
odd records provide an independent compatibility test.
