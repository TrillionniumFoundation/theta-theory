\begin{abstract}
We derive a nonlinear kinetic semigroup from finite deterministic
hard-sphere path laws.  A canonical path tilt has an exact conditional
log-Laplace tower on complete histories.  Microcanonical ensemble transfer and
the collision-marked cumulant limit produce the Hamilton--Boltzmann
Hamilton--Jacobi semigroup after Boltzmann--Grad scaling.  Preparation precedes
homogenization, and finite conditioned laws are not declared Markov.  Scalar
theta-expectations arise only on mechanically calibrated cotangent rays; the
familiar quadratic theta-generator is a Gaussian contraction of the full
nonquadratic kinetic theory.
\end{abstract}

\section{Finite-volume prepared path laws}
Let $\mathbb P_{\varepsilon}^{\rm mc,a}$ be the macro-prepared microcanonical
law of Paper I and let $\mathcal F_t^\varepsilon$ contain the empirical
particle histories and collision marks up to $t$.

For a bounded additive kinetic path potential $\Theta$ define
\[
 \frac{d\mathbb P_{\varepsilon,T}^{\Theta}}
 {d\mathbb P_{\varepsilon}^{\rm mc,a}}
 =\frac{\exp\{\mu_\varepsilon A_T^\varepsilon(\Theta)\}}
 {\E\exp\{\mu_\varepsilon A_T^\varepsilon(\Theta)\}}.
\]
For additional bounded work $F$ put
\[
 \cE_{s,t}^{\varepsilon,\Theta}[F]
 =\frac1{\mu_\varepsilon}
 \log\E_{\mathbb P_{\varepsilon,T}^{\Theta}}
 \left[e^{\mu_\varepsilon F}\mid\mathcal F_s^\varepsilon\right].
\]

\begin{proposition}[Exact microscopic tower]\label{prop:tower}
For $r\le s\le t$,
\[
 \cE_{r,t}^{\varepsilon,\Theta}
 =\cE_{r,s}^{\varepsilon,\Theta}
 \circ\cE_{s,t}^{\varepsilon,\Theta}.
\]
The exact sufficient state is the complete conditional history law; no finite
empirical-density Markov property is asserted.
\end{proposition}

\begin{proof}
Exponentiate and apply the linear conditional-expectation tower.  The
finite-volume canonical normalization cancels in the conditional
Radon--Nikodym ratio.
\end{proof}

Once the complete microstate is observed, the future is deterministic; the
nontrivial semigroup comes from coarse preparation and partial information.

\section{Boltzmann--Grad nonlinear semigroup}
For a terminal functional $\Phi$ on kinetic densities define
\[
 U_\varepsilon(t,\pi)
 =\frac1{\mu_\varepsilon}
 \log\E_{\pi}\exp\{\mu_\varepsilon\Phi(\pi_T^\varepsilon)\}.
\]
Here $\pi$ denotes a prepared conditional law, not one finite-$N$ empirical
measure.

\begin{theorem}[Kinetic excess-pressure limit]\label{thm:pressure-limit}
On the source ball and time interval of Paper I, $U_\varepsilon$ converges
locally uniformly on the regular prepared class to the unique mild solution
\[
 \partial_tU(t,f)
 +\mathbb H\left(f,\frac{\delta U}{\delta f},0\right)=0,
 \qquad U(T,f)=\Phi(f).
\]
A bounded collision work is incorporated as the collision source in
$\mathbb H$.
\end{theorem}

\begin{proof}
The logarithmic transform is the cumulant functional controlled by the
source-decorated cluster expansion.  The density-only hard-sphere convergence
and Hamilton--Jacobi identification are in \cite{BGSS2023}; stability follows
the nonlinear-generator framework of \cite{FengKurtz2006}.  Paper II
identifies the extended Hamiltonian.  Local analytic well-posedness gives
uniqueness, while \cref{prop:tower} and compact-uniform convergence pass the
semigroup law to the limit.
\end{proof}

The collision nonlinearity remains exponential; it is not replaced by a
quadratic diffusion unless the Gaussian tangent is taken.

\section{Microcanonical preparation before kinetic scaling}
Let $a$ be the macro constraint and $f_a^0$ its information projection.  The
order of limits is
\[
 \text{microcanonical conditioning}
 \longrightarrow f_a^0
 \longrightarrow \text{Boltzmann--Grad kinetic limit}.
\]
Finite microcanonical path laws are neither product nor Markov.

\begin{theorem}[Preparation--semigroup diagonal limit]\label{thm:diagonal}
Choose shell and macro windows with
$\mu_\varepsilon^{-1/2}\ll\eta_\varepsilon,\delta_\varepsilon\downarrow0$.
For every compact source set there exists a diagonal choice for which the
microcanonical nonlinear values converge to the same kinetic semigroup as the
technical hard-core grand-canonical law with density $f_a^0$.
\end{theorem}

\begin{proof}
Paper I gives $o(\mu_\varepsilon)$ equivalence of the required path
log-Laplace functionals.  Division by $\mu_\varepsilon$ removes the error.
Uniform source analyticity and local HJ comparison allow the preparation and
Boltzmann--Grad limits to be diagonalized.
\end{proof}

\section{Calibrated theta rays and diffusion tangents}
Let $G$ be a mechanical path functional and suppose a one-dimensional macro
preparation selects the cotangent ray
\[
 \Theta_a=\theta(a)G.
\]
Define
\[
 \cE_G^{\theta}[F]
 =\frac1\theta
 \left(Q(\theta G+\theta F)-Q(\theta G)\right).
\]

\begin{theorem}[Endogenous theta expectation]\label{thm:theta}
If the constrained rate is differentiable along the $G$-ray, then
$\theta(a)=I_G'(a)$ and $\cE_G^\theta$ is the excess kinetic pressure of the
prepared gas.  Cash additivity and the dynamic tower hold exactly for the
associated canonical phase.  No free risk parameter is added after the
preparation is fixed.
\end{theorem}

At a Gaussian contraction $X_t$ with covariance $A_t$, the tangent generator
is
\[
 \mathfrak G^\theta f
 =b\cdot Df+\frac12A:D^2f
 +\frac\theta2Df^TADf.
\]
It is a tangent of the full kinetic exponential generator, not its definition.

\begin{theorem}[Microcanonical-to-theta closure]\label{thm:main}
The exact finite-volume history tower, microcanonical preparation transfer,
kinetic HJ limit and calibrated scalar contractions form a closed chain in the
declared short-time domain.
\end{theorem}
