\section{Round-twenty-nine reconstruction: preparation--dynamics factorization and energy-supported semigroups}
\label{sec:r29-b4}

Initial preparation entropy and dynamic running action are different
functionals.  The former is charged once, outside the nonlinear semigroup; the
latter concatenates and gives the identity at time zero.  Velocity containment
is proved at the quadratic scale supported by the Gaussian reference and by
elastic energy conservation.  No exponential moment of
$|v|^2\log|v|$ is asserted.

\subsection{Joint and contracted actions}

Let $\mathcal X_E$ be the space of probability densities or measures with
fixed mass and finite second velocity moment, equipped with $W_2$ and the
weak collision topology.  For an exactly balanced path
$\Gamma=qA_f$, define the running action conditional on $f_0=x$ by
\[
 J_T^x(f,\Gamma)=
 \begin{cases}
 \displaystyle\int_0^T\!\int\ell(q)\,dA_f,&f_0=x,\\
 +\infty,&\text{otherwise},
 \end{cases}
 \qquad \ell(q)=q\log q-q+1.                              \tag{B4.1}
\]
The joint microscopic rate and its density contraction are
\[
 I_T^{\rm joint}(f,\Gamma)
 =H(f_0\mid f_0^{\rm ref})+J_T^{f_0}(f,\Gamma),            \tag{B4.2}
\]
\[
 I_T^{\rm dens}(f)=\inf_{\Gamma}I_T^{\rm joint}(f,\Gamma). \tag{B4.3}
\]

\begin{proposition}[Correct effective-domain representation]
\label{prop:r29-b4-action}
On the B2 attainable closure,
\[
 I_T^{\rm joint}(f,\Gamma)
 =H(f_0\mid f_0^{\rm ref})+
   \int\ell\left({d\Gamma\over dA_f}\right)dA_f           \tag{B4.4}
\]
for every retained joint pair.  For density alone the representation is the
infimum (B4.3); it is not equal to the cost of every contact control producing
the same density path.
\end{proposition}

\begin{proof}
B2 positive recovery gives the upper bound for each joint pair.  The
variational formula
\[
 \int\ell(q)dA=
 \sup_{\phi\in C_c}\left\{\int\phi\,d\Gamma-
                         \int(e^\phi-1)dA\right\}
\]
gives lower semicontinuity under the collision topology and hence the lower
bound.  Taking the infimum over all $\Gamma$ with the same density path proves
(B4.3).  B3 contact cycles show why equality with an arbitrary chosen
$\Gamma$ would be false.
\end{proof}

\subsection{Quadratic energy and entropy containment}

Let $M(v)=Z^{-1}e^{-a|v|^2}$ be the Gaussian velocity reference.  Microscopic
elastic dynamics conserves
\[
 E^\varepsilon_t=\left\langle f_t^\varepsilon,|v|^2\right\rangle
 =E^\varepsilon_0                                             \tag{B4.5}
\]
pathwise.  Therefore, for every $0<\lambda<a$,
\[
 {1\over\mu_\varepsilon}\log
 E\exp\{\lambda\mu_\varepsilon E_t^\varepsilon\}
 ={1\over\mu_\varepsilon}\log
 E\exp\{\lambda\mu_\varepsilon E_0^\varepsilon\}\le C_\lambda.       \tag{B4.6}
\]
This is an initial-law exponential moment, not a fictitious collision-counting
martingale.

\begin{lemma}[Dynamic relative-entropy estimate]
\label{lem:r29-b4-entropy}
For every regular controlled Boltzmann path,
\[
 \sup_{t\le T}H(f_t\mid M)
 \le H(f_0\mid M)+J_T^{f_0}(f,\Gamma)+C_T.                 \tag{B4.7}
\]
Consequently every joint-action sublevel is uniformly integrable for
$|v|^2$ and relatively compact at each time in $W_2$.
\end{lemma}

\begin{proof}
Test the controlled balance equation with
$\log(f/M)$ after a positive truncation.  The free-transport term vanishes.
For one collision, Young's inequality in the form
$ab\le\ell(a)+e^b-1$ bounds the controlled entropy production by
$\ell(q)$ plus the ordinary Maxwellian collision entropy term, which is
integrable on the regular chart.  Integrating gives (B4.7), and approximation
extends it to the attainable closure.

For $R>0$, apply the entropy inequality to
$\lambda|v|^2 1_{|v|>R}$ relative to $M$.  Since
$\int e^{\lambda|v|^2}M(dv)<\infty$ for $\lambda<a$, the right side of
(B4.7) makes the second-moment tail tend to zero uniformly.  This is precisely
$W_2$ compactness at a fixed time.
\end{proof}

\begin{theorem}[Energy-supported path compactness]
\label{thm:r29-b4-compact}
For every $L,T$, the set
\[
 \{(f,\Gamma):I_T^{\rm joint}(f,\Gamma)\le L\}            \tag{B4.8}
\]
is relatively compact in
$C([0,T];\mathcal X_E)$ with the $W_2$ topology and in the weak contact
current topology.
\end{theorem}

\begin{proof}
For a test with one bounded spatial and velocity derivative, a free flight on
$[s,t]$ changes its expectation by
$C|t-s|(1+E)^{1/2}$.  A binary collision changes a Lipschitz velocity test by
at most $C|v-v_*|$.  Since $A_f$ already contains one factor
$|v-v_*|$, the part $q\le R$ is bounded by
$CR|t-s|\int ff_*|v-v_*|^2$, which uses only the second moment.  On $q>R$,
first truncate velocities, use
$\int_{q>R}q\,dA_f\le C J/\log R$, and then remove the velocity truncation
with Lemma~\ref{lem:r29-b4-entropy}.  This gives a modulus tending to zero
uniformly on (B4.8).  Pointwise $W_2$ compactness and the modulus give path
compactness; the same entropy estimate controls contact mass on compact
velocity sets.
\end{proof}

\subsection{The running-action Nisio semigroup}

For $\phi\in\operatorname{BUC}(\mathcal X_E)$ define
\[
 (V_t\phi)(x)=
 \sup_{(f,\Gamma):f_0=x}
 \{\phi(f_t)-J_t^x(f,\Gamma)\}.                           \tag{B4.9}
\]
The one-shot prepared value is separately
\[
 (\mathcal P_t\phi)=
 \sup_{x\in\mathcal X_E}\{V_t\phi(x)-H(x\mid f_0^{\rm ref})\}.          \tag{B4.10}
\]

\begin{theorem}[Preparation--dynamics factorization]
\label{thm:r29-b4-semigroup}
The operators $(V_t)_{t\ge0}$ form a strongly continuous max-plus contraction
semigroup:
\[
 V_0=I,\qquad V_{t+s}=V_tV_s.                             \tag{B4.11}
\]
The preparation functional is charged exactly once through (B4.10) and does
not enter the concatenated running action.
\end{theorem}

\begin{proof}
At $t=0$, the only path has zero running action, so $V_0\phi=\phi$.
Concatenating two exactly balanced controls adds their integrals in (B4.1) and
preserves the intermediate density/contact trace.  Conversely every control
on $[0,t+s]$ disintegrates at time $t$, so the dynamic programming inequalities
hold in both directions.  Compactness from Theorem~\ref{thm:r29-b4-compact}
permits passage from regular controls to the attainable closure.  The same
modulus proves strong continuity.
\end{proof}

On smooth cylindrical functions,
\[
 \mathcal H\phi(f)=
 \ip{f}{v\cdot\nabla_xD\phi(f)}+
 \int A_f\left(e^{\Delta D\phi(f)}-1\right).              \tag{B4.12}
\]
The initial entropy does not appear in this generator.

\subsection{Collision entropy topology and comparison}

A sequence $(f_n,\Gamma_n)$ converges in collision entropy topology if
$f_n\to f$ in $C(W_2)$, $A_{f_n}\Rightarrow A_f$,
$\Gamma_n\Rightarrow\Gamma$, and
\[
 \int\phi\,d\Gamma_n-
 \int(e^\phi-1)dA_{f_n}\to
 \int\phi\,d\Gamma-
 \int(e^\phi-1)dA_f                                     \tag{B4.13}
\]
for every compactly supported continuous collision test $\phi$.

\begin{proposition}[Varying-reference closure]
\label{prop:r29-b4-closure}
On an action sublevel, $A_{f_n}\Rightarrow A_f$ and (B4.13) imply
$\Gamma=qA_f$ and
\[
 \int\ell(q)dA_f\le\liminf_n\int\ell(q_n)dA_{f_n}.        \tag{B4.14}
\]
\end{proposition}

\begin{proof}
Truncate velocities.  On the truncated set, $W_2$ convergence gives product
convergence and the hard-sphere kernel is bounded except on a set controlled
by angular coarea.  Lemma~\ref{lem:r29-b4-entropy} removes the velocity tail.
The variational representation of entropy and (B4.13) give (B4.14).
\end{proof}

\begin{theorem}[Comparison on energy--entropy shells]
\label{thm:r29-b4-comparison}
On every compact set
\[
 K_{E,H}=\{f:E(f)\le E_0,\ H(f\mid M)\le H_0\},            \tag{B4.15}
\]
the Hamilton--Jacobi equation $\partial_tu=\mathcal Hu$ has at most one
bounded uniformly continuous viscosity solution with cylindrical test core,
and that solution is $V_t\phi$.
\end{theorem}

\begin{proof}
Use the doubled-variable penalty
$\alpha W_2(f,g)^2+\epsilon[H(f\mid M)+H(g\mid M)]$.
The shell is compact by Lemma~\ref{lem:r29-b4-entropy}.  A common optimal
coupling transports the free part.  For the collision part, truncate
velocities and use the local Lipschitz bound for
$e^{\Delta D\phi}-1$; the entropy/energy tail is uniform by (B4.15).
The standard viscosity inequalities then let
$\alpha\to\infty$ and $\epsilon\downarrow0$.  Dynamic programming from
Theorem~\ref{thm:r29-b4-semigroup} gives existence.
\end{proof}

\subsection{Canonical microscopic realization}

For a finite separating family $\Phi_n$ of bounded continuous observables,
let $\mathfrak R^{(n)}_{\varepsilon,N}(f)$ be the unique minimizer of relative
entropy with respect to the symmetric hard-core Maxwellian law subject to
\[
 E_{\mathfrak R^{(n)}}\langle f^{\varepsilon},\phi\rangle
 =\langle f,\phi\rangle,
 \qquad \phi\in\Phi_n,                                    \tag{B4.16}
\]
and the exact mass, momentum, and energy constraints.  Strict convexity gives
uniqueness.  Let $n=n(\varepsilon)\uparrow\infty$ slowly enough that the B1
saddle remains in its uniform chart, and write
$\mathfrak R_{\varepsilon,N}(f)$ for this canonical diagonal realization.

\begin{lemma}[Recovery-independent realization]
\label{lem:r29-b4-realization}
On every shell (B4.15),
$\mathfrak R_{\varepsilon,N}(f)$ is well defined, depends continuously on the
finite constraint vector, and every subsequential empirical limit equals
$f$.  Its definition uses no arbitrary regular recovery sequence.
\end{lemma}

\begin{proof}
Existence follows from compactness of entropy sublevels and feasibility of the
B2 positive approximants.  Strict convexity gives uniqueness.  The Lagrange
multipliers are the B1 finite-dimensional saddle and vary continuously.
Density of $\cup_n\Phi_n$ in the bounded-Lipschitz class identifies every
empirical limit with $f$.
\end{proof}

\begin{theorem}[Nonlinear microscopic semigroup limit]
\label{thm:r29-b4-limit}
Let $V_t^\varepsilon$ be the exact finite-volume exponential semigroup.  For
$T<T_*$, $\phi\in\operatorname{BUC}(\mathcal X_E)$, and every shell
$K_{E,H}$,
\[
 \sup_{t\le T}\sup_{f\in K_{E,H}}
 \left|V_t^\varepsilon\phi(\mathfrak R_{\varepsilon,N}(f))
       -V_t\phi(f)\right|\longrightarrow0.                \tag{B4.17}
\]
\end{theorem}

\begin{proof}
For a cylindrical core function, truncate the B2 kernel expansion at a fixed
vertex number.  The B2 loop theorem removes recollisions and the remaining
tree terms converge to (B4.12); the factorial tail is uniform.  B1 controls
the canonical realization multipliers.  Theorem~\ref{thm:r29-b4-compact}
gives half-relaxed-limit compactness and
Theorem~\ref{thm:r29-b4-comparison} identifies both limits.  Contraction and
core density extend the convergence to all bounded uniformly continuous
$\phi$.
\end{proof}

The revised theory keeps the original entropy Hamiltonian, but it now uses the
correct joint contraction, charges preparation only once, relies on the
quadratic moment actually supported by the model, and employs a canonical
rather than recovery-dependent microscopic realization.
