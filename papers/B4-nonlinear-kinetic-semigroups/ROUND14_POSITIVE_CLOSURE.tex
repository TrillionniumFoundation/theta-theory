
\section{Round-fourteen positive closure: ensemble boundary generators and the kinetic Nisio semigroup}
\label{sec:r14-b4}

Five objects are kept distinct: the deterministic hard-sphere flow, its
Koopman action on microscopic observables, the pushforward of probability
laws, the ensemble log-Laplace functional, and the limiting kinetic
Lax--Oleinik semigroup.  A deterministic Koopman derivation is never claimed
to create an exponential Hamiltonian.  The exponential collision term arises
from the boundary Green formula after the empirical jump is multiplied by
the large-deviation speed.

\subsection{The finite-volume boundary identity}

Let \(\Phi_t^\varepsilon\) be the hard-sphere flow,
\(U_t^\varepsilon G=G\circ\Phi_t^\varepsilon\), and
\((\Phi_t^\varepsilon)_\#\rho\) the law pushforward.  Let
\(X^\varepsilon=(\pi^\varepsilon,\Gamma^\varepsilon)\) be the empirical
density/contact state.  For a smooth density cylinder
\(F(\pi)\), write \(p=\delta F/\delta\pi\).

\begin{theorem}[Ensemble exponential Green identity]
\label{thm:r14-b4-green}
If \(\rho_t\) is an ensemble law in the B2 trace graph, then
\[
 \begin{aligned}
 {d\over dt}\int e^{\mu_\varepsilon F(\pi^\varepsilon)}\,d\rho_t
 &=
 \mu_\varepsilon\int
 e^{\mu_\varepsilon F}
 \langle\pi^\varepsilon,v\cdot\nabla_xp\rangle\,d\rho_t\\
 &\quad+
 \sum_{i<j}\int_{\Sigma_{ij}^-}
 \left[
 e^{\mu_\varepsilon F(\pi^{\varepsilon,+})}
 -e^{\mu_\varepsilon F(\pi^{\varepsilon,-})}
 \right]d\gamma_{\rho_t,ij}^- .
 \end{aligned}
\]
At one contact,
\[
 \mu_\varepsilon
 \bigl(F(\pi^{\varepsilon,+})-F(\pi^{\varepsilon,-})\bigr)
 =\Delta_\omega p+O(\mu_\varepsilon^{-1}).
\]
Thus the ensemble logarithmic derivative contains
\(e^{\Delta_\omega p}-1\).  Pointwise in the regular interior,
\(A_\varepsilon e^{\mu F}=\mu e^{\mu F}A_\varepsilon F\), as required for a
Koopman derivation.
\end{theorem}

\begin{proof}
Apply the B2 Green trace formula to the microscopic observable
\(e^{\mu_\varepsilon F(\pi^\varepsilon)}\).  Free transport obeys the ordinary
chain rule.  Across a contact, the empirical measure changes by
\[
 \pi^{\varepsilon,+}-\pi^{\varepsilon,-}
 ={1\over\mu_\varepsilon}
 (\delta_{v'}+\delta_{v_*'}-\delta_v-\delta_{v_*}),
\]
so Taylor expansion gives the second formula.  The boundary jump, not the
interior derivation, is responsible for the exponential.
\end{proof}

For a law \(\rho\), define the finite ensemble Hamiltonian by dividing the
right side by
\(\mu_\varepsilon\int e^{\mu_\varepsilon F}d\rho\).  B2 trace convergence
then gives the Hamilton--Boltzmann Hamiltonian.

\subsection{Factorial hierarchy intertwining}

Let \(g_j^\varepsilon\) be the factorial correlation hierarchy.  A polynomial
law cylinder has expansion
\[
 \mathcal I_\varepsilon F(g)
 =
 \sum_{j\ge0}{1\over j!}
 \langle f_j[F],g_j^\varepsilon\rangle .
\]
The hierarchy norm is
\[
 \|g\|_{\alpha,\beta}
 =
 \sum_{j\ge0}{e^{\alpha j}\over j!}
 \|g_j\|_\beta .
\]

\begin{lemma}[Exact law--hierarchy intertwiner]
\label{lem:r14-b4-intertwine}
The map \(\rho\mapsto(g_j^\varepsilon)_j\) intertwines the Liouville
pushforward of laws with the BBGKY hierarchy, including the B2 normal-flux
boundary operator.  For every polynomial cylinder \(F\), the ensemble Green
identity equals the hierarchy duality
\[
 {d\over dt}\mathcal I_\varepsilon F(g_t)
 =
 \mathcal I_\varepsilon(\mathbb A_\varepsilon F)(g_t).
\]
\end{lemma}

\begin{proof}
Expand the symmetric expectation in factorial moments.  Free transport acts
on each marginal.  Integrating the contact boundary of a \(j\)-particle
marginal gives the loss term, while summing over one unobserved label gives
the \(j+1\to j\) creation term with its exact combinatorial coefficient.
This is the BBGKY hierarchy.  Absolute convergence follows from the
\(1/j!\) weight.
\end{proof}

\subsection{The full triangular corrector}

Let \(F\) be a smooth kinetic cylinder.  Expanding the finite ensemble
Hamiltonian gives a triangular sequence of connected defects
\(D_j^\varepsilon[F]\).  Let \(\mathbb A_{\varepsilon}^{(J)}\) be the
finite \(J\)-level observable hierarchy generator on its resolvent graph
domain.

\begin{theorem}[Normally summable hierarchy corrector]
\label{thm:r14-b4-corrector}
For every regular moment ball and \(T\le T_0\), there are graph-domain
correctors \(c_j^\varepsilon[F]\) satisfying the backward equations
\[
 \partial_t c_j^\varepsilon
 +\mathbb A_{\varepsilon,j}c_j^\varepsilon
 =-D_j^\varepsilon[F],\qquad c_j^\varepsilon(T)=0,
\]
and
\[
 F_\varepsilon
 =F+\sum_{j\ge2}\mu_\varepsilon^{1-j}c_j^\varepsilon[F].
\]
Moreover,
\[
 \sum_{j\ge2}{e^{\alpha'j}\over j!}
 \left(
 \|c_j^\varepsilon\|+
 \|\mathbb A_{\varepsilon,j}c_j^\varepsilon\|
 \right)<\infty
\]
uniformly in \(\varepsilon\), and
\[
 \mathscr H_\varepsilon F_\varepsilon
 \longrightarrow
 \mathbb H\left(f,{\delta F\over\delta f},\psi\right)
\]
locally uniformly.
\end{theorem}

\begin{proof}
Solve the finite triangular terminal system by the resolvent/Yosida
propagator.  B2 connected genealogies give
\[
 \|D_j^\varepsilon[F]\|
 \le j!C^jT^{j-1}.
\]
The \(j!\) is exactly cancelled by the hierarchy norm, while the
Boltzmann--Grad coefficient \(\mu_\varepsilon^{1-j}\) and a loss
\(\alpha-\alpha'>CT\) make the series geometric.  The same estimate holds for
the generator images and time derivatives, so closedness of the graph domain
passes the finite-\(J\) solutions to the infinite sum.  Lemma
\ref{lem:r14-b4-intertwine} and the ensemble boundary identity identify the
remaining binary term with the exponential collision Hamiltonian.  B2
dominated genealogy convergence removes cycles.
\end{proof}

\subsection{The dynamic action and Nisio semigroup}

For a balanced density/contact path on \([s,t]\), let
\[
 \mathcal A_{s,t}(f,\Gamma)
 =\int_s^t\ell(d\Gamma/dA_f)\,dA_f .
\]
This transition action contains no initial preparation cost.  Define
\[
 (S_{s,t}\Phi)(f_s)
 =
 \sup_{(f,\Gamma):f(s)=f_s}
 \{\Phi(f_t)-\mathcal A_{s,t}(f,\Gamma)\}.
\]

\begin{theorem}[Kinetic Nisio semigroup]
\label{thm:r14-b4-nisio}
On every B2 compact action sublevel,
\(S_{s,t}\) is monotone, cash additive, a contraction in the sup norm, and
\[
 S_{r,t}=S_{r,s}S_{s,t}.
\]
For \(\lambda>0\), the discounted nonlinear resolvent
\[
 (J_\lambda h)(f)
 =
 \sup_{(f_\cdot,\Gamma)}
 \int_0^\infty e^{-\lambda t}
 \bigl[h(f_t)-\ell(d\Gamma/dA_f)\bigr]\,dt
\]
is well defined, satisfies
\(\|J_\lambda h-J_\lambda k\|_\infty\le\lambda^{-1}\|h-k\|_\infty\), and is
the resolvent of the Nisio generator associated with \(\mathbb H\).
\end{theorem}

\begin{proof}
Concatenation of balanced paths preserves balance and adds the dynamic
entropy, which proves both inequalities in the semigroup identity.
Exponential discount and entropy coercivity make the infinite-horizon
supremum finite.  Splitting at a deterministic time gives the resolvent
dynamic-programming equation.  Subtracting two such equations proves the
contraction estimate.  On smooth cylinders, Fenchel duality for
\(\ell(q)\) gives
\[
 \lambda u-h
 =
 \mathbb H\left(f,{\delta u\over\delta f},0\right)
\]
in the graph-viscosity sense, identifying the generator.
\end{proof}

\subsection{Comparison without uncontrolled doubled jets}

A bounded upper-semicontinuous \(u\) is a subsolution when
\(u\le J_\lambda(h+\lambda u)\), and a bounded lower-semicontinuous \(v\) is
a supersolution when \(v\ge J_\lambda(h+\lambda v)\).

\begin{theorem}[Primal resolvent comparison]
\label{thm:r14-b4-comparison}
If \(u\) and \(v\) are respectively a sub- and supersolution of the same
discounted equation, then \(u\le v\).  The backward terminal equation
\[
 \partial_sU+\mathbb H(f,\delta U/\delta f,0)=0,\qquad U(T)=\Phi
\]
has the unique bounded graph-viscosity solution \(U(s)=S_{s,T}\Phi\).
No doubled exponential jet is used.
\end{theorem}

\begin{proof}
Monotonicity and the resolvent contraction give
\[
 \|(u-v)^+\|_\infty
 \le
 \|[J_\lambda(h+\lambda u)-J_\lambda(h+\lambda v)]^+\|_\infty
 \le\|(u-v)^+\|_\infty .
\]
Insert the strict perturbation \(u_\eta=u-\eta\) in the subsolution
inequality; the first inequality then improves by \(\eta/\lambda\), forcing
\((u_\eta-v)^+=0\).  Let \(\eta\downarrow0\).
Time discretization by the resolvent and the Crandall--Liggett product
formula gives the Nisio semigroup.  Its dynamic-programming equation proves
the terminal statement.
\end{proof}

\subsection{Microscopic convergence and preparation}

\begin{theorem}[Ensemble-to-kinetic semigroup convergence]
\label{thm:r14-b4-convergence}
For every bounded terminal cylinder and compact regular initial-law family,
the finite ensemble log-Laplace values converge locally uniformly to
\(S_{s,t}\Phi\).  Primitive microcanonical preparation is applied once at
the initial boundary through the B1 source-dependent saddle.  It is not
reselected on subintervals.
\end{theorem}

\begin{proof}
The corrected tests of
Theorem~\ref{thm:r14-b4-corrector} give upper and lower nonlinear-generator
limits on a separating graph core.  B2 exponential compact containment gives
half-relaxed limits.  The primal comparison theorem identifies both limits
with the Nisio value.  B1 conditions the initial law before the deterministic
evolution; the fixed initial multiplier may be carried as a static coordinate
but does not enter the dynamic action.
\end{proof}

\begin{theorem}[Gaussian risk-sensitive tangent]
\label{thm:r14-b4-gaussian}
Under the B3 central scaling, the kinetic semigroups converge on smooth
finite-dimensional cylinders to the risk-sensitive diffusion semigroup with
generator
\[
 b\cdot DF+\tfrac12A:D^2F+\tfrac\theta2DF^TADF,
\]
where \(A\) is the full density/contact covariance, including pure-contact
noise and the B1 initial Schur complement.
\end{theorem}

\begin{proof}
B3 gives process convergence and a uniform complex moment neighborhood.
Taylor expansion of the finite ensemble pressure shows that cumulants of
order at least three vanish under the central scaling, while the first two
give \(b\) and \(A\).  Uniform integrability passes the exponential
transform to the limit.  Applying the logarithmic transform to the diffusion
generator yields the quadratic term.
\end{proof}

\begin{theorem}[Round-fourteen B4 closure]
\label{thm:r14-b4-main}
The deterministic hard-sphere ensemble has a correctly typed boundary
nonlinear generator, a normally summable hierarchy corrector, a constructed
kinetic Nisio semigroup with primal comparison, a one-time microcanonical
interface, and the full Gaussian risk-sensitive tangent.
\end{theorem}

\begin{proof}
Combine Theorems~\ref{thm:r14-b4-green},
\ref{thm:r14-b4-corrector}, \ref{thm:r14-b4-nisio},
\ref{thm:r14-b4-comparison}, \ref{thm:r14-b4-convergence}, and
\ref{thm:r14-b4-gaussian}.
\end{proof}
