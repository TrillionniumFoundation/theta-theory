\section{Round-twelve positive closure: typed endpoint values, resolvent cores, and primal comparison}
\label{sec:r12-b4}

Five levels remain distinct: microscopic states, Koopman observables,
push-forward laws, endpoint values depending on an initial law, and the
limiting kinetic action semigroup.  Uniqueness of the limiting Hamilton--Jacobi
equation is proved from the primal entropy action, not from a doubled penalty
whose exponential jets escape every controlled source ball.

\subsection{The exact finite-volume objects}

Let $\Phi_t^\varepsilon$ be the hard-sphere flow, $U_t^\varepsilon F=F\circ
\Phi_t^\varepsilon$ its Koopman group, and
$(U_t^\varepsilon)_\#\rho$ the push-forward of a law.  For a terminal
microscopic observable $G$ define
\[
 V_{s,t}^\varepsilon(G;\rho)
 ={1\over\mu_\varepsilon}\log
 \int e^{\mu_\varepsilon U_{t-s}^\varepsilon G(x)}\,d\rho(x).
\]

\begin{proposition}[Typed endpoint tower]
\label{prop:r12-b4-tower}
For $r<s<t$,
\[
 V_{r,t}^\varepsilon(G;\rho)
 ={1\over\mu_\varepsilon}\log
 \int\exp\{\mu_\varepsilon
 V_{s,t}^\varepsilon(G;\delta_{\Phi_{s-r}^\varepsilon x})\}\,d\rho(x).
\]
The inner function is a microscopic observable; no finite-volume Markov
closure on the one-particle density is asserted.
\end{proposition}

\begin{proof}
Substitute the definition and use
$U_{t-r}=U_{s-r}U_{t-s}$.  The types of the inner Dirac endpoint value and the
outer initial-law integral are exactly those displayed.
\end{proof}

\subsection{A genuine observable graph core}

Let $A_\varepsilon$ be the closed Koopman generator on the weighted uniformly
continuous observable space with specular boundary matching.  Let
$\mathscr D_0$ be smooth finite-coordinate functions satisfying that matching,
and define
\[
 \mathscr C_\varepsilon=R_{\lambda,\varepsilon}\mathscr D_0,
 \qquad
 R_{\lambda,\varepsilon}=(\lambda-A_\varepsilon)^{-1}.
\]

\begin{theorem}[Resolvent graph core]
\label{thm:r12-b4-core}
$\mathscr D_0$ is dense in the underlying observable Banach space and
$\mathscr C_\varepsilon$ is a core for $A_\varepsilon$.  More precisely, for
$F\in D(A_\varepsilon)$ and
$G=(\lambda-A_\varepsilon)F$, any
$G_n\in\mathscr D_0$ with $G_n\to G$ gives
$R_{\lambda,\varepsilon}G_n\to F$ in graph norm.
\end{theorem}

\begin{proof}
Unfold specular trajectories across each collision face.  On every finite
chronological chart the flow is translation in unfolded coordinates, so
mollification and velocity truncation approximate bounded uniformly
continuous observables while preserving the reflected matching after folding
back.  Grazing and multiple-collision sets are bypassed in the base norm,
not claimed to have zero graph capacity.  This proves density of
$\mathscr D_0$.  The resolvent identity then gives
$A R_\lambda G_n=\lambda R_\lambda G_n-G_n$, which is graph convergence.
\end{proof}

\subsection{The full hierarchy corrector in one sequence space}

Use the exponential generating hierarchy norm
\[
 \|G\|_{\alpha}
 =\sum_{j\ge0}{e^{\alpha j}\over j!}\|g_j\|_{j,\beta}.
\]
Let $\mathbb A_\varepsilon$ be the full triangular BBGKY/ledger generator on
its closed graph domain.

\begin{theorem}[Normally summable graph corrector]
\label{thm:r12-b4-corrector}
For every smooth density cylinder $F$ and $T\le T_*$ there is a unique
sequence-space perturbed test $F_\varepsilon^{\rm corr}$ satisfying the
backward graph equation
\[
 (\partial_t+\mathbb A_\varepsilon)
 F_\varepsilon^{\rm corr}
 =\mathbb H(f,DF,\psi)+r_\varepsilon,
 \qquad F_\varepsilon^{\rm corr}(T)=F,
\]
with
\[
 \sup_{t\le T}
 \bigl(\|F_\varepsilon^{\rm corr}-F\|_\alpha
      +\|r_\varepsilon\|_{\alpha'}\bigr)\longrightarrow0.
\]
The factorial tree growth is cancelled explicitly by the $1/j!$ hierarchy
weight and the remaining series is geometric for short time.
\end{theorem}

\begin{proof}
Write the Duhamel equation on the complete weighted sequence space.  A
connected order-$j$ term is at most $j!C^jT^{j-1}$; multiplication by the
sequence weight leaves $(Ce^{\alpha-\alpha'}T)^j$.  Choose $T_*$ so this is
summable, including one generator image and time derivative.  B2 removes the
summed cyclic sector and identifies the tree limit.  Closedness of
$\mathbb A_\varepsilon$ passes the normally convergent series through the
graph equation.
\end{proof}

\subsection{The dynamic action semigroup}

Let $I_{s,t}^{\rm dyn}$ be only the B2 collision entropy on $[s,t]$, with no
initial preparation term.  Define
\[
 S_{s,t}\Phi(f)=
 \sup_{(\rho,\Gamma):\rho_s=f}
 \{\Phi(\rho_t)-I_{s,t}^{\rm dyn}(\rho,\Gamma)\}.
\]

\begin{theorem}[Action compactness and dynamic programming]
\label{thm:r12-b4-action}
On every moment sublevel, $I_{s,t}^{\rm dyn}$ has compact sublevels, is
lower semicontinuous, and is additive under concatenation.  Hence
$S_{r,t}=S_{r,s}S_{s,t}$ and $S$ is a sup-norm contraction.  Primitive
microcanonical preparation is charged once through the B1 initial rate.
\end{theorem}

\begin{proof}
B2 exponential moments and the Green balance give weighted tightness and
endpoint compactness.  Entropy lower semicontinuity and closed balance give
closed sublevels.  Concatenating balanced pairs adds the entropy integral;
splitting an admissible pair gives the reverse dynamic-programming
inequality.  The initial rate never appears in a transition interval.
\end{proof}

\subsection{Comparison from the primal control problem}

A bounded upper semicontinuous $u$ is a subsolution when every smooth bounded
cylinder test gives the Hamiltonian inequality.  Supersolutions are analogous.

\begin{theorem}[Primal verification comparison]
\label{thm:r12-b4-comparison}
Let $u$ be a bounded subsolution and $v$ a bounded supersolution with the same
terminal data and the common moment-containment condition.  Then $u\le v$.
The proof never evaluates the Hamiltonian on an uncontrolled doubled jet.
\end{theorem}

\begin{proof}
Fix a smooth positive control pair $(\rho,\Gamma)$ with bounded collision ratio
$q$.  Fenchel's inequality applied to the subsolution test gives, after
partitioning time and using the chain rule along the weak balance,
\[
 u(s,\rho_s)\le u(t,\rho_t)+I_{s,t}^{\rm dyn}(\rho,\Gamma)+o(1).
\]
For a supersolution, use the regular tilted recovery supplied by B2 and the
reverse local inequality to obtain
\[
 v(s,f)\ge
 \sup_{(\rho,\Gamma):\rho_s=f}
 \{v(t,\rho_t)-I_{s,t}^{\rm dyn}(\rho,\Gamma)\}.
\]
Approximate finite-action controls by the B2 positive regularization; entropy
uniform integrability removes the bounded-$q$ restriction.  Iterating the two
inequalities to the terminal time gives
$u\le S\Phi\le v$.  No exponential continuity estimate at a diverging
cotangent is used.
\end{proof}

\begin{theorem}[Microscopic nonlinear-semigroup convergence]
\label{thm:r12-b4-limit}
The finite-volume endpoint values converge locally uniformly on regular moment
sets to $S_{s,t}\Phi$.  The limit is the unique viscosity/semigroup solution of
\[
 \partial_sU+\mathbb H(f,DU,\psi)=0
\]
in the preceding class.  Under B3 central scaling it converges to the
risk-sensitive Gaussian semigroup with quadratic term
$\frac\theta2DU^TADU$.
\end{theorem}

\begin{proof}
Theorem~\ref{thm:r12-b4-corrector} gives upper and lower nonlinear-generator
limits on the resolvent core.  B2 compact containment gives half-relaxed
limits, and Theorem~\ref{thm:r12-b4-comparison} identifies them with the action
value.  The exact endpoint tower and local uniform convergence give the
semigroup law.  B3 Gaussian convergence and a uniform complex moment
neighborhood give the final exponential-transform limit.
\end{proof}

\begin{theorem}[Round-twelve B4 closure]
\label{thm:r12-b4-main}
The exact microscopic objects are correctly typed, the Koopman generator has
a resolvent graph core, the full hierarchy corrector is normally summable, and
the kinetic action semigroup is the unique limit by primal comparison.
\end{theorem}

\begin{proof}
Combine Proposition~\ref{prop:r12-b4-tower} and
Theorems~\ref{thm:r12-b4-core}, \ref{thm:r12-b4-corrector},
\ref{thm:r12-b4-action}, \ref{thm:r12-b4-comparison}, and
\ref{thm:r12-b4-limit}.
\end{proof}
