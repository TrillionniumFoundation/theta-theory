
\section{Round-twenty-seven reconstruction: attainable entropy action and microscopic realization}
\label{sec:r27-b4}

A formal finite value of $\int\ell(q)\,dA_f$ does not by itself control a
superquadratic velocity moment.  The microscopic rate is therefore defined as
the lower-semicontinuous closure of the regular, microscopically realizable
entropy action in the topology in which the hard-sphere laws are exponentially
tight.  On its effective domain the formula is still exactly the original
relative-entropy action; no artificial moment penalty is added to the
Hamiltonian.

\subsection{The attainable action}

Choose a convex superquadratic function
\[
 \Psi(v)=(1+|v|^2)\log(2+|v|^2)
\]
and let $\tau_\Psi$ be narrow convergence together with convergence of
$\int\Psi\,df$.  For a smooth positive controlled Boltzmann path,
\[
 \partial_tf+v\cdot\nabla_xf=\mathcal C(qA_f),\qquad q\ge0,
\]
put
\[
 \mathcal A_T(f,q)
 =H(f_0\mid f_0^{\rm ref})+\int_0^T\!\int\ell(q)\,dA_f.   \tag{B4.1}
\]
Let $\mathscr R$ be the regular positive class produced by the B2 recovery
compiler and define
\[
 I_T(f)=\inf\left\{\liminf_n\mathcal A_T(f_n,q_n):
 (f_n,q_n)\in\mathscr R,\ f_n\to f
 \text{ in }C([0,T];\tau_\Psi)\right\}.                   \tag{B4.2}
\]
This is an attainable closure, not a new additive term.

\begin{theorem}[Representation on the effective domain]
\label{thm:r27-b4-representation}
If $(f,q)$ is a controlled path with
\[
 \sup_{t\le T}\int\Psi\,df_t<\infty
\]
and is approximable by the B2 regular controls, then
\[
 I_T(f)=\mathcal A_T(f,q).                                \tag{B4.3}
\]
If a formal finite-entropy pair has no such $\tau_\Psi$ approximation, then
$I_T(f)=+\infty$.  In particular the high-velocity multiplier counterexample
from the referee report is outside the microscopic effective domain, rather
than being assigned a false compactness estimate.
\end{theorem}

\begin{proof}
The upper bound is the B2 positive strong recovery with velocity truncation,
Maxwellian replacement, and collision-simplex balance correction.  For the
lower bound, use the variational representation
\[
 \int\ell(q)\,dA_f
 =\sup_\phi\left\{\int\phi\,q\,dA_f-\int(e^\phi-1)\,dA_f\right\}.
\]
On $\tau_\Psi$-compact sets, truncate $\phi$ in velocity, pass to the limit,
and remove the truncation using uniform $\Psi$-integrability.  This proves
lower semicontinuity and (B4.3).  The final statement is the definition of the
lower-semicontinuous microscopic closure and prevents the formal action from
being extended beyond the topology supported by the microscopic laws.
\end{proof}

\subsection{Exponential-source containment}

B2 pressure analyticity is available for truncated collision sources
$\lambda\Delta\Psi_R$, where
$\Psi_R=\Psi\wedge R$.  The angular Povzner identity gives constants
$c,C>0$ independent of $R$ such that
\[
 \int_{\mathbb S^2}B\,
 [\Psi_R(v')+\Psi_R(v_*')-\Psi_R(v)-\Psi_R(v_*)]\,d\omega
 \le C(1+|v|^2+|v_*|^2)-cD_R(v,v_*),                     \tag{B4.4}
\]
where $D_R\uparrow D_\Psi$ is nonnegative and superquadratic.

\begin{lemma}[Microscopic exponential collision-source bound]
\label{lem:r27-b4-source}
There are $\lambda_0,C_T>0$ such that, uniformly in $R$ and $\varepsilon$,
\[
 {1\over\mu_\varepsilon}\log
 E\exp\left\{\lambda_0\mu_\varepsilon
 \left[\sup_{t\le T}\langle f_t^\varepsilon,\Psi_R\rangle
       +\int_0^TD_R(f_t^\varepsilon)\,dt\right]\right\}
 \le C_T.                                                 \tag{B4.5}
\]
\end{lemma}

\begin{proof}
Insert the exponential martingale associated with the microscopic increment
of $\Psi_R$ at each actual collision.  Elastic energy conservation controls
the quadratic remainder, while (B4.4) contributes the nonnegative
dissipation.  The B2 connected-history pressure bounds the compensator
uniformly for $|\lambda|\le\lambda_0$; surplus histories vanish by the causal
loop theorem.  Doob's inequality gives the time supremum.  All constants are
uniform in $R$ because the Povzner estimate is.
\end{proof}

\begin{theorem}[Action-coercive containment]
\label{thm:r27-b4-containment}
Every sublevel $\{I_T\le L\}$ satisfies
\[
 \sup_{f:I_T(f)\le L}\left[
 \sup_{t\le T}\int\Psi\,df_t+\int_0^TD_\Psi(f_t)\,dt
 \right]\le C_{L,T}.                                     \tag{B4.6}
\]
\end{theorem}

\begin{proof}
Apply the entropy inequality between a regular controlled microscopic tilt and
the reference law to the random variable in (B4.5).  Divide by
$\mu_\varepsilon$, use the relative entropy bound, and pass to the B2 limit.
Then let $R\uparrow\infty$ by monotone convergence.  Finally use the
lower-semicontinuous definition (B4.2).  This derives containment from the
original microscopic action and its pressure domain; it does not use the
invalid pointwise Young inequality with $e^{c|v|^p}$.
\end{proof}

\subsection{A time modulus in Wasserstein topology}

For a collision from $(v,v_*)$ to $(v',v_*')$, couple the pre- and
post-collisional empirical measures by moving two atoms along the collision
chord.  The squared transport cost is bounded by
$C(|v|^2+|v_*|^2)$.  Free transport over a time interval of length $h$ costs
$h^2\int|v|^2df$.

\begin{lemma}[Dynamic $W_2$ modulus]
\label{lem:r27-b4-modulus}
On $\{I_T\le L\}$,
\[
 W_2(f_t,f_s)^2
 \le C_{L,T}\bigl(|t-s|+|t-s|^{1+\alpha}\bigr)            \tag{B4.7}
\]
for some $\alpha>0$.  Hence action sublevels are relatively compact in
$C([0,T];W_2)$.
\end{lemma}

\begin{proof}
For regular controls, integrate the collision coupling cost against
$qA_f$.  Split at $q\le M$ and use the entropy bound on $q>M$ together with
the $\Psi$ containment (B4.6); optimize $M$ as a negative power of
$|t-s|$.  Add the free-transport coupling.  The estimate passes to the
attainable closure by lower semicontinuity.  Pointwise $W_2$ compactness follows
from $\Psi$-uniform integrability, and (B4.7) gives uniform-in-time
compactness.
\end{proof}

\subsection{Varying-reference entropy topology}

A sequence $(f_n,\Gamma_n)$ converges in collision entropy topology when
$f_n\to f$ in $C(W_2)$, $A_{f_n}\Rightarrow A_f$, $\Gamma_n\Rightarrow\Gamma$,
and for every bounded continuous $\phi$ with compact velocity support,
\[
 \int\phi\,d\Gamma_n-\int(e^\phi-1)\,dA_{f_n}
 \longrightarrow
 \int\phi\,d\Gamma-\int(e^\phi-1)\,dA_f.                  \tag{B4.8}
\]
This definition is intrinsic for varying reference measures.

\begin{proposition}[Collision-product closure]
\label{prop:r27-b4-product}
On an action sublevel, $f_n\to f$ in $C(W_2)$ implies
$A_{f_n}\Rightarrow A_f$ with uniform integrability of the collision rate.
Convergence in (B4.8) then implies
$\Gamma_n=q_nA_{f_n}\Rightarrow qA_f$ and
\[
 \int\ell(q)\,dA_f\le\liminf_n\int\ell(q_n)\,dA_{f_n}.     \tag{B4.9}
\]
\end{proposition}

\begin{proof}
Product convergence follows from $W_2$ convergence.  The superquadratic
containment controls the linear hard-sphere factor $|v-v_*|$.  Truncate
velocities, pass to the bounded continuous kernel, and remove the truncation.
The variational formula for relative entropy and (B4.8) give (B4.9).
\end{proof}

\subsection{Nisio semigroup and Hamiltonian}

For $\phi\in\operatorname{BUC}(\mathcal X_\Psi)$ define
\[
 (V_t\phi)(f)
 =\sup_{f_\cdot:f_0=f}\{\phi(f_t)-I_t(f_\cdot)\}.          \tag{B4.10}
\]
Compactness and concatenation of attainable controls give a strongly
continuous max-plus semigroup.  On smooth cylindrical functions,
\[
 \mathcal H\phi(f)
 =\langle f,v\cdot\nabla_xD\phi(f)\rangle
 +\int A_f(e^{\Delta D\phi(f)}-1).                        \tag{B4.11}
\]
No containment penalty appears in (B4.11).

\begin{theorem}[Comparison on the attainable shell]
\label{thm:r27-b4-comparison}
The Hamilton--Jacobi equation
$\partial_tu=\mathcal Hu$ has at most one bounded uniformly continuous
viscosity solution on every $I_T$ sublevel, and that solution is $V_t\phi$.
\end{theorem}

\begin{proof}
Penalize a doubled maximum by
$\alpha W_2(f,g)^2+\epsilon[\langle f,\Psi\rangle+
\langle g,\Psi\rangle]$.  The containment theorem makes maximizing pairs
compact.  Couple free transport and collisions synchronously; the collision
exponential is locally Lipschitz on the cylindrical core and the tail is
uniformly small by (B4.6).  The transport and collision errors vanish as
$\alpha\to\infty$, then $\epsilon\downarrow0$.  Standard time doubling gives
comparison.  Dynamic programming proves that $V_t\phi$ is a solution.
\end{proof}

\subsection{A typed microscopic realization map}

For a smooth density $f$ with conserved moments define
$\mathfrak R_{\varepsilon,N}(f)$ to be the symmetric canonical law obtained
from the conditioned product $\prod_i f(x_i,v_i)$ on the hard-core domain,
normalized and corrected by the B1 exact-number/momentum/energy saddle.  For a
general $f$, choose a regular recovery and take the graph closure.  A
microscopic observable acts on $\mathfrak R_{\varepsilon,N}(f)$, never directly
on an abstract continuum density.

Let $\phi$ be a cylindrical core function.  Truncate the B2 history expansion
at $K$ vertices and solve the exact backward specular problem to obtain
$\phi_{\varepsilon,K}$ in the BBGKY generator domain.

\begin{lemma}[Core generator convergence]
\label{lem:r27-b4-core}
For every action sublevel,
\[
 \lim_{K\to\infty}\limsup_{\varepsilon\to0}
 \sup_f\left|
 H_\varepsilon\phi_{\varepsilon,K}
     (\mathfrak R_{\varepsilon,N}(f))
 -\mathcal H\phi(f)\right|=0.                             \tag{B4.12}
\]
\end{lemma}

\begin{proof}
At fixed $K$, the B2 causal loop theorem removes every recollision graph and
the tree terms sum to (B4.11).  B1 controls the realization saddle uniformly.
The B2 factorial radius estimate makes the discarded history tail
$O((a'/a)^K)$.  The containment theorem makes velocity truncation uniform.
\end{proof}

\begin{theorem}[Nonlinear microscopic semigroup limit]
\label{thm:r27-b4-limit}
Let $V_t^\varepsilon$ be the exact finite-volume exponential semigroup.  For
every $\phi\in\operatorname{BUC}(\mathcal X_\Psi)$ and $T<T_*$,
\[
 \sup_{t\le T}\sup_{f:I_T(f)\le L}
 \left|V_t^\varepsilon\phi
       (\mathfrak R_{\varepsilon,N}(f))-V_t\phi(f)\right|\to0.            \tag{B4.13}
\]
\end{theorem}

\begin{proof}
Use Lemma \ref{lem:r27-b4-core} in the upper and lower half-relaxed-limit
argument.  Exponential containment gives compactness, and Theorem
\ref{thm:r27-b4-comparison} identifies both limits.  Density of the
resolvent-regularized cylindrical core and contraction of both semigroups
extend the result to all bounded uniformly continuous functions.
\end{proof}

This construction keeps the original microscopic entropy and Hamiltonian,
but uses their correct attainable lower-semicontinuous domain.  It supplies
the missing superquadratic containment, temporal $W_2$ modulus,
varying-reference topology, comparison calculation, and continuum-to-
microscopic realization map.
