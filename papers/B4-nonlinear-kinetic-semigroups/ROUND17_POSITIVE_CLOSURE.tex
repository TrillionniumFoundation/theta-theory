\section{Round-twenty-one positive closure: nonlinear resolvents, realizable hierarchy cores, and semigroup convergence}
\label{sec:r21-b4}

This section replaces the former Round-Seventeen source.  The state space, the
controlled action, and the function space are fixed before the semigroup is
constructed.  The discounted variational operator satisfies the nonlinear
resolvent identity for an accretive graph; the linear pseudo-resolvent formula
is nowhere used.

\subsection{The kinetic state and compact action topology}

Let $\mathcal E_M$ be the set of nonnegative densities $f$ on
$\mathbb T^3\times\mathbb R^3$ with prescribed mass and with
\[
 \int (1+|v|^2)f\,dxdv\le M.
\]
Choose a dense sequence $(\phi_j)_{j\ge1}$ in the unit ball of smooth tests
satisfying
$|\phi_j|+|\nabla_x\phi_j|+|\Delta\phi_j|\le C_j(1+|v|^2)$, and put
\[
 d_*(f,g)=\sum_{j\ge1}2^{-j}
 {\left|\int\phi_j(f-g)\right|\over
  1+\left|\int\phi_j(f-g)\right|}.
\]
The full state space is the increasing union
$\mathcal E=\bigcup_M\mathcal E_M$ with weight
$\mathfrak w(f)=1+\int(1+|v|^2)f$.  Let
$\mathrm{BUC}_{\mathfrak w}(\mathcal E)$ be the functions $h$ for which
$h/\mathfrak w$ is bounded and uniformly continuous on every energy shell,
with norm
\[
 \|h\|_{\mathfrak w}=\sup_f{|h(f)|\over\mathfrak w(f)}.
\]

For an admissible path $(f,\Gamma)$ on $[0,T]$, write
$\Gamma=qA_f$ and use the B2 action
\[
 \mathcal A_T(f,\Gamma)=\int_0^T\int\ell(q)\,dA_f,
 \qquad \ell(q)=q\log q-q+1.
\]
The path satisfies the exact weak balance equation and begins from the stated
initial condition.

\begin{theorem}[Compact dynamic action sublevels]
\label{thm:r21-b4-compact}
Fix an energy shell $\mathcal E_M$, a time $T$, and $L<\infty$.  The set of
admissible paths beginning in $\mathcal E_M$ and satisfying
$\mathcal A_T\le L$ is compact after passing to the topology of uniform
convergence of all density coordinates
$t\mapsto\int\phi_j f_t$ and weak convergence of the contact current on
compact time--velocity sets.  The terminal-state map is continuous on this
set, and the action is lower semicontinuous.
\end{theorem}

\begin{proof}
The entropy inequality gives, for every nonnegative contact test $G$,
\[
 \int Gq\,dA_f\le \mathcal A_T+
 \int(e^G-1)\,dA_f.
\]
Choose successively truncated quadratic velocity tests.  Conservation of
mass and energy controls $A_f$ on compact velocity sets, while the B2
exponential preparation estimate controls its velocity tails.  Thus the
contact currents are uniformly tight.  The weak balance identity yields
\[
 \left|\int\phi_j(f_t-f_s)\right|
 \le |t-s|\|v\cdot\nabla_x\phi_j\|_{L^\infty_{2}}
       \sup_r\int(1+|v|^2)f_r
 +\int_{(s,t]}|\Delta\phi_j|\,d\Gamma.
\]
Uniform integrability of the contact measure makes the right side a common
modulus in probability and, on an action sublevel, after discarding an
arbitrarily small contact-time set.  Reparametrize those finitely many jumps
by their cumulative contact mass; the density path is then equicontinuous in
$d_*$.  Helly--Arzel\`a compactness and a diagonal argument give a convergent
subsequence, and the B2 closed Green graph identifies its limit as admissible.

On compact velocity sets the density coordinates converge strongly after
velocity averaging of the balance equation.  Hence
$f_nf_{n,*}B\,dtdxdvdv_*d\omega$ converges weakly to $A_f$ there; the tail
estimate removes the truncation.  The variational representation
\[
 \int\ell(d\Gamma/dA_f)dA_f
 =\sup_G\left\{\int G\,d\Gamma-
                 \int(e^G-1)dA_f\right\}
\]
then gives lower semicontinuity.  Continuity of the terminal coordinates
follows from the chosen uniform density topology.
\end{proof}

\subsection{State-dependent transfer without signed corrections}

Let $(f,qA_f)$ be a regular controlled path on $[0,T]$ and let $g_0$ be near
$f_0$ in $d_*$ on the same energy shell.  Define $g$ as the unique mild
solution of
\[
 \partial_tg+v\cdot\nabla_xg=\mathcal Q_{qA_g},
 \qquad g(0)=g_0.
\]

\begin{lemma}[Positive multiplier transfer]
\label{lem:r21-b4-transfer}
On every compact regular control chart there are $C,\delta>0$ such that, if
$d_*(f_0,g_0)<\delta$, the path $(g,qA_g)$ exists on $[0,T]$, is positive and
exactly balanced, and satisfies
\[
 \sup_{t\le T}d_*(f_t,g_t)\le C d_*(f_0,g_0),
 \qquad
 |\mathcal A_T(g,qA_g)-\mathcal A_T(f,qA_f)|
 \le C d_*(f_0,g_0).
\]
No signed collision-current correction is added.
\end{lemma}

\begin{proof}
For bounded $q$ on the regular chart, the controlled Boltzmann collision map
is locally Lipschitz in the weighted $L^1$ norm and preserves positivity and
all collision invariants.  Picard iteration of the mild equation therefore
gives the unique positive path.  Subtract the two mild equations, use the
bilinear estimate
\[
 \|\mathcal Q_{qA_g}-\mathcal Q_{qA_f}\|_{L^1_2}
 \le C(\|f\|_{L^1_2}+\|g\|_{L^1_2})\|f-g\|_{L^1_2},
\]
and apply Gronwall.  Approximation of the metric tests by the weighted
$L^1$ class gives the $d_*$ bound.  Since the relative multiplier is the same
$q$, the action difference is
$\int\ell(q)(dA_g-dA_f)$ and is controlled by the same bilinear estimate.
The balance equation holds by construction, so there is no residual and no
appeal to a finite-dimensional correction of an infinite-dimensional PDE.
\end{proof}

\subsection{The Nisio semigroup}

For $h\in\mathrm{BUC}_{\mathfrak w}(\mathcal E)$ define
\[
 (S_Th)(f_0)=\sup_{(f,\Gamma):f(0)=f_0}
 \left\{h(f_T)-\mathcal A_T(f,\Gamma)\right\}.
\]
The zero-cost choice is the ordinary Boltzmann path $q=1$.

\begin{theorem}[Weighted strongly continuous Nisio semigroup]
\label{thm:r21-b4-nisio}
The operators $(S_T)_{T\ge0}$ form an order-preserving, constant-additive
contraction semigroup on $\mathrm{BUC}_{\mathfrak w}(\mathcal E)$.  On each
energy shell,
\[
 \sup_{f\in\mathcal E_M}|S_Th(f)-h(f)|\longrightarrow0
 \qquad(T\downarrow0).
\]
The supremum is attained for bounded terminal rewards on an energy shell.
\end{theorem}

\begin{proof}
Concatenating admissible paths and using additivity of the action gives
$S_{t+s}=S_tS_s$.  Order preservation and constant additivity are immediate;
taking the same path for two rewards gives the contraction estimate.  For
short times, the zero-cost path moves each test coordinate by $O_M(T)$.
Conversely, an optimizer moving a coordinate by more than $\eta$ must spend a
positive amount determined by the entropy inequality and the balance formula.
Uniform continuity of $h$ on $\mathcal E_M$ therefore gives the stated
uniform short-time limit.  The weight is propagated by mass and energy
conservation.

For an attaining sequence, truncate to a common action sublevel using the
bounded reward.  Theorem~\ref{thm:r21-b4-compact} supplies a convergent
subsequence; continuity of the terminal reward and lower semicontinuity of the
action make its limit optimal.
\end{proof}

\subsection{The correct nonlinear discounted resolvent}

For $\lambda>0$ define
\[
 (J_\lambda h)(f_0)=
 \sup_{(f,\Gamma):f(0)=f_0}
 \int_0^\infty e^{-t/\lambda}
 \left[{1\over\lambda}h(f_t)-
       \ell(q_t)A_{f_t}(\mathbf1)\right]dt,
\]
where the action density notation means integration over the collision
variables.  The exponential moment and energy shell bounds make the integral
finite.

\begin{theorem}[Nonlinear resolvent identity and accretive graph]
\label{thm:r21-b4-resolvent}
The maps $J_\lambda$ are contractions and satisfy, for
$0<\mu<\lambda$,
\[
 J_\lambda h
 =J_\mu\left({\mu\over\lambda}h+
          \left(1-{\mu\over\lambda}\right)J_\lambda h\right).
\]
Define the graph $\mathcal H$ by
\[
 (u,g)\in\mathcal H
 \quad\Longleftrightarrow\quad
 u=J_\lambda(u+\lambda g)
\]
for one, and hence every, $\lambda>0$.  Then $\mathcal H$ is dissipative,
$I-\lambda\mathcal H$ has full range for every $\lambda>0$, and the semigroup
generated by $\mathcal H$ is exactly $(S_t)$.
\end{theorem}

\begin{proof}
Split the discounted integral at a stopping time $s$ and use the dynamic
programming principle.  Equivalently,
\[
 J_\lambda h=\sup_{s>0}
 \left\{\int_0^se^{-t/\lambda}
  \left({h(f_t)\over\lambda}-L_t\right)dt
  +e^{-s/\lambda}J_\lambda h(f_s)\right\}.
\]
Multiply this identity by the exponential density of a mean-$\mu$ waiting
time and integrate in $s$.  Fubini's theorem and the memoryless property split
the mean-$\lambda$ horizon into a mean-$\mu$ first stage followed by the
continuation value.  The resulting reward is exactly
$(\mu/\lambda)h+(1-\mu/\lambda)J_\lambda h$, proving the displayed nonlinear
resolvent identity.  No subtraction or linear composition identity is used.

The contraction follows by comparing two rewards under the same controls.
The nonlinear identity proves that the graph definition is independent of
$\lambda$.  If $u_i=J_\lambda(u_i+\lambda g_i)$, contraction gives
\[
 \|u_1-u_2\|\le
 \|u_1-u_2+\lambda(g_1-g_2)\|,
\]
which is dissipativity.  For every $h$, $u=J_\lambda h$ gives
$(u,(h-u)/\lambda)\in\mathcal H$, so the range is full.  The nonlinear
Crandall--Liggett exponential formula defines the unique contraction semigroup.
Applying the resolvent DPP repeatedly and letting the mesh tend to zero
identifies that formula with the finite-horizon Nisio DPP, hence with $S_t$.
\end{proof}

\subsection{Exact law--hierarchy realization}

In a bounded finite-volume hard-sphere domain, the exclusion condition gives
a finite maximal particle number in every compact position cell.  For a point
process law $P$, let $\rho^{(k)}$ be its factorial moment densities.

\begin{theorem}[Finite-volume Janossy reconstruction and BBGKY domain]
\label{thm:r21-b4-intertwiner}
A sequence $(\rho^{(k)})_{k\ge0}$ is the factorial hierarchy of a unique
hard-sphere law with exponential number moment if and only if it is symmetric,
consistent, supported on the exclusion configurations, Lenard positive, and
satisfies
$\sum_k z^k\|\rho^{(k)}\|_1/k!<\infty$ for some $z>1$.  The inverse is
\[
 j_n={1\over n!}\sum_{k\ge0}{(-1)^k\over k!}
 \int\rho^{(n+k)}\,dX_{n+1}\cdots dX_{n+k},
\]
and $P=\sum_nj_n\,dX_n$.  On the Green--specular graph domain of B2, the
Liouville evolution is intertwined exactly with the BBGKY hierarchy and
preserves the realizable cone.
\end{theorem}

\begin{proof}
The exponential bound makes the inclusion--exclusion series for $j_n$
absolutely convergent.  Lenard positivity makes every Janossy functional
nonnegative; consistency gives normalization.  Conversely, direct factorial
expansion of a point-process law yields the displayed conditions and
inclusion--exclusion inversion.  Absolute convergence proves uniqueness.

Apply the finite-particle Green identity to a smooth factorial observable.
Bulk transport gives the free hierarchy term and the incoming/outgoing trace
relation gives the binary BBGKY boundary term.  The B2 closed trace theorem
permits graph closure.  Because the finite evolution is a push-forward of a
probability law, its hierarchy remains Lenard positive and consistent; the
exponential number bound is unchanged.
\end{proof}

\subsection{A quantitative microscopic graph core}

Let $F$ be a smooth cylinder in the limiting kinetic state.  Truncate the B2
connected expansion at genealogy size $K$.  For each retained genealogy solve
backward the finite terminal-value transport equation with the exact specular
boundary relation; call the sum $F_{\varepsilon,K}$.

\begin{lemma}[Diagonal choice with nonuniform genealogy exponents]
\label{lem:r21-b4-diagonal}
There is a deterministic $K(\varepsilon)\uparrow\infty$ such that
\[
 \sum_{|G|\le K(\varepsilon)}C_G\varepsilon^{\alpha_G}\to0,
 \qquad
 \sum_{|G|>K(\varepsilon)}(CT)^{|G|}\to0.
\]
\end{lemma}

\begin{proof}
For every fixed $K$, the first sum tends to zero because it contains finitely
many positive exponents.  Choose a decreasing sequence
$\varepsilon_K$ so that the first sum is below $2^{-K}$ whenever
$\varepsilon<\varepsilon_K$, and define $K(\varepsilon)$ as the largest $K$
with that property.  Reduce the short slicing interval so that $CT<1$; then
the geometric tail tends to zero.  This diagonal argument requires no
uniform lower bound on the genealogy-dependent exponents.
\end{proof}

\begin{theorem}[Realizable Liouville graph-core corrector]
\label{thm:r21-b4-corrector}
The observables $F_\varepsilon=F_{\varepsilon,K(\varepsilon)}$ belong to the
exact finite-particle Liouville graph domain, are factorial observables of
realizable laws, and satisfy on every kinetic action sublevel
\[
 F_\varepsilon\to F,
 \qquad
 H_\varepsilon F_\varepsilon\to HF
\]
locally uniformly, where $H_\varepsilon$ is the microscopic nonlinear
Hamiltonian and $H$ the B2 kinetic Hamiltonian.
\end{theorem}

\begin{proof}
The backward terminal-value equation is solved separately on every
finite-particle collision chamber.  Its incoming boundary value is defined as
the outgoing value after the exact specular reflection, so
$F_{\varepsilon,K}$ lies in the graph domain without later regularization.
Symmetrizing over labels makes it a factorial observable; Theorem
\ref{thm:r21-b4-intertwiner} identifies it on the realizable cone.

For genealogies of size at most $K$, the B2 precontact theorem bounds every
surplus contribution by $C_G\varepsilon^{\alpha_G}$.  The normalized
connected-tree estimate gives the geometric tail beyond $K$.  Lemma
\ref{lem:r21-b4-diagonal} makes both errors vanish.  Compact action sublevels
provide uniform velocity/contact truncation, so the convergence is locally
uniform.  Differentiating the finite terminal equation gives the generator
convergence with the same estimates.
\end{proof}

\subsection{Nonlinear semigroup convergence}

Let $S_t^\varepsilon$ be the finite-volume logarithmic semigroup on law
observables transported to factorial hierarchies, and let $S_t$ be
Theorem~\ref{thm:r21-b4-nisio}.

\begin{theorem}[Nonlinear Trotter--Kato limit]
\label{thm:r21-b4-limit}
For every weighted uniformly continuous kinetic observable $h$ and every
sequence of microscopic laws whose empirical states converge to $f$ with
uniform energy/exponential moments,
\[
 S_t^\varepsilon h_\varepsilon\longrightarrow S_th(f)
\]
locally uniformly for $t$ in compact intervals and $f$ in kinetic action
sublevels.  The limiting semigroup is the unique one generated by the graph
$\mathcal H$ of Theorem~\ref{thm:r21-b4-resolvent}.
\end{theorem}

\begin{proof}
Exponential tightness and compact containment follow from B2 and
Theorem~\ref{thm:r21-b4-compact}.  Apply
Theorem~\ref{thm:r21-b4-corrector} to a graph core of smooth cylinders.  Its
locally uniform generator convergence supplies both upper and lower extended
generator inclusions by adding and subtracting a vanishing constant.  Hence
upper and lower half-relaxed limits are respectively a viscosity subsolution
and supersolution of the resolvent equation for $\mathcal H$.

The contraction resolvent of Theorem~\ref{thm:r21-b4-resolvent} gives
comparison: apply dissipativity to the doubled variables and then let the
penalty vanish.  Therefore the two relaxed limits coincide.  Density of the
core, contraction of both semigroups, and the exponential formula extend the
convergence to every $h\in\mathrm{BUC}_{\mathfrak w}$.  Initial convergence and
compact containment identify the value at $t=0$, completing the nonlinear
Trotter--Kato argument.
\end{proof}

\begin{theorem}[Round-twenty-one B4 closure]
\label{thm:r21-b4-main}
The kinetic value theory has compact action sublevels, positive
state-dependent control transfer, a weighted strongly continuous Nisio
semigroup, the correct nonlinear resolvent identity and maximal dissipative
graph, an exact realizable hierarchy intertwiner, a quantitative
specular-domain graph core, and nonlinear semigroup convergence.
\end{theorem}

\begin{proof}
Combine Theorems~\ref{thm:r21-b4-compact},
\ref{thm:r21-b4-nisio}, \ref{thm:r21-b4-resolvent},
\ref{thm:r21-b4-intertwiner}, \ref{thm:r21-b4-corrector}, and
\ref{thm:r21-b4-limit} with Lemma~\ref{lem:r21-b4-transfer}.
\end{proof}
