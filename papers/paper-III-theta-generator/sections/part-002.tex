Its stationary bracket and correlation correction reproduce the collision

\begin{proposition}[Quadratic cell average]\label{prop:cell-average}
For every $p,X\in\R$, the normalized second-order cell equation has mean
\[
 \bar\tau(c+bp)
 +\frac{\Sigma_{R,q}^{\rm phys}}2X
 +\frac{\theta\Sigma_{R,q}^{\rm phys}}2p^2.
\]
\end{proposition}
\begin{proof}
Use the perturbed test
\[
 \phi^\varepsilon(x,\xi)
 =\phi(x)+\varepsilon p h(\xi)+\varepsilon^2\chi(\xi).
\]
The Poisson equation cancels the order-$\varepsilon$ fast term.  At order
$\varepsilon^2$, the additive Taylor contribution is one half the
Green--Kubo variance times $X$.  Expanding the logarithm contributes one half
$\theta$ times the same variance of the first-order martingale increment,
namely $\theta\Sigma p^2/2$.  The roof-weighted drift and work average to
$\bar\tau(bp+c)$.  The remaining centred state dependence is removed by the
second Poisson equation for $\chi$.
\end{proof}

\section{Local consistency and fast-state collapse}\label{sec:consistency}
\begin{theorem}[Collision-clock consistency]\label{thm:consistency}
For $\phi\in C_b^3(\R)$ and the corrected test above,
\[
 \frac{\cT_\varepsilon^{R,q}\phi^\varepsilon-
       \phi^\varepsilon}{\varepsilon^2}
 \longrightarrow
 \bar\tau(c+b\phi')
 +\frac{\Sigma_{R,q}^{\rm phys}}2\phi''
 +\frac{\theta\Sigma_{R,q}^{\rm phys}}2(\phi')^2
\]
locally uniformly in $x$ and in the weighted quotient norm.
\end{theorem}
\begin{proof}
Taylor expand the spatial shift and the exponential to third order.  Bounded
cell displacement, finite horizon, and bounded correctors make the normalized
remainder $O(\varepsilon)$.  Apply \cref{prop:cell-average}.
\end{proof}

\begin{lemma}[Canonical future-kernel memory loss]\label{lem:memory}
On every compact preparation set there are $C<\infty$, $\rho\in(0,1)$ and a
weighted H\"older norm $\norm{\cdot}_{\alpha,w}$ such that
\[
 \norm{K_{R,q}^m f-\pi_{R,q}(f)}_\infty
 \le C\rho^m\norm f_{\alpha,w}.
\]
The estimate also holds for the small positive Feynman--Kac perturbations
appearing in \eqref{eq:operator}, after normalization, with constants uniform
for sufficiently small $\varepsilon$.
\end{lemma}
\begin{proof}
Doob conjugacy transfers the simple eigenvalue and spectral gap of the twisted
Lorentz operator to the normalized future kernel.  Multiplication by the
bounded factor
$\exp\{\theta\varepsilon^2\tau c\}$ and the
$O(\varepsilon)$ Lipschitz spatial shift are a small operator perturbation on
the weighted H\"older tower space.  The common Riesz contour and the resolvent
identity preserve the gap and give the displayed normalized contraction.
\end{proof}

\begin{proposition}[Loss of the fast preparation state]\label{prop:collapse}
For a macroscopic collision horizon $O(\varepsilon^{-2})$, values started from
two quotient states in a common weighted compact set differ by $o(1)$ after a
vanishing initial layer.  Every half-relaxed limit is therefore independent of
$\xi$.
\end{proposition}
\begin{proof}
Exponentiate the value recursion so that each fast block is a positive
Feynman--Kac operator.  Apply \cref{lem:memory} over
$m_\varepsilon=\lceil C_0\abs{\log\varepsilon}\rceil$ collisions.  The
normalized dependence on the initial quotient state is
$O(\rho^{m_\varepsilon})$.  The Duhamel terms caused by the spatial shifts are
a geometric convolution of $O(\varepsilon)$ Lipschitz errors and are
$O(\varepsilon)$; the running-work perturbation is $O(\varepsilon^2)$.  Over
the initial block the slow state moves by
$O(\varepsilon m_\varepsilon)=o(1)$ and physical time by
$O(\varepsilon^2m_\varepsilon)=o(1)$.  Taking $C_0$ large makes the spectral
term $o(1)$, and monotonicity plus cash additivity propagate the estimate
through the remaining recursion.
\end{proof}
