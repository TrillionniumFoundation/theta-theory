
\begin{abstract}
We derive a nonlinear physical-time generator from the compact periodic Sinai
billiard and Liouville current preparation of Papers I--II.  The canonical
field $\theta$ is fixed by the prepared cell current through
$\theta=I_R'(j)$.  It is not introduced as a free risk parameter.  The exact
microscopic nonlinear recursion is the conditional excess pressure of the
Doob-normalized Lorentz transfer operator on the one-sided tower quotient.
The associated kernel is a disintegration of a deterministic two-sided
billiard path measure, not a stochastic collision law.

A spectral Poisson corrector and a quadratic multiplicative cell problem yield
collision-clock consistency.  The correct physical observable is the
clock-centred cell increment $G=J-v\tau$.  Its pressure-root variance produces
both the linear diffusion term and the quadratic entropic term.  Exponential
random-horizon control converts the collision recursion to actual physical
time.  The limiting generator is
\[
 \mathfrak G_{R,j}f
 =c+bf'+\frac{D_{R,j}}2f''
 +\frac{\theta_R(j)D_{R,j}}2(f')^2.
\]
We distinguish exact canonical dynamic programming from finite
microcanonical conditioning: the latter justifies the canonical preparation
through an iterated or diagonal preparation limit and is never declared
Markov at finite size.
\end{abstract}

\section{The canonical past-state kernel}\label{sec:kernel}
Fix $(R,q)$ in the spectral range of Paper I and write
\[
 j=\Lambda_R'(q),\qquad \theta=q,
 \qquad \bar\tau=\bar\tau_{R,q},
 \qquad v=j/\bar\tau,
 \qquad G=J_R-v\tau_R.
\]
Let $\Xi_R$ be the one-sided quotient state of a standard Young tower for
$T_R$.  The Doob-normalized transfer operator from Paper I defines a positive
kernel
\[
 K_{R,q}(\xi,d\xi')
\]
with stationary law $\pi_{R,q}$ and natural extension $\mu_{R,q}$.
Edge observables $G(\xi,\xi')$ and $\tau(\xi,\xi')$ project to the actual
cell displacement and free-flight time of the billiard.

\begin{proposition}[Deterministic interpretation]\label{prop:deterministic}
A path drawn from the stationary kernel and lifted to the natural extension is
an ordinary deterministic orbit of $T_R$.  Conditional randomness appears
only because $\xi$ omits the complementary stable/future coordinate.
\end{proposition}
\begin{proof}
This is the natural-extension statement of Paper I.  The factor from the
two-sided tower to the billiard intertwines the shift with $T_R$ almost
everywhere.
\end{proof}

\section{Exact microscopic excess-pressure DPP}\label{sec:dpp}
Let $b,c:\R\to\R$ be bounded and globally Lipschitz and let the terminal work
$\Phi$ be bounded Lipschitz.  At one collision set
\[
 x'=x+\varepsilon G(\xi,\xi')
 +\varepsilon^2\tau(\xi,\xi')b(x).
\]
For $\theta\ne0$ define
\begin{equation}\label{eq:operator}
\begin{aligned}
 (\cT_\varepsilon^{R,q}F)(x,\xi)
 =\frac1\theta\log\int K_{R,q}(\xi,d\xi')
 \exp\Bigl\{\theta[&\varepsilon^2\tau(\xi,\xi')c(x)\\
 &+F(x',\xi')]\Bigr\}.
\end{aligned}
\end{equation}
At $\theta=0$ take the continuous conditional-expectation limit.

\begin{theorem}[Exact canonical microscopic DPP]\label{thm:exact-dpp}
For a deterministic collision horizon $N$, the recursion
\[
 V_N^\varepsilon(x,\xi)=\Phi(x),
 \qquad
 V_k^\varepsilon=\cT_\varepsilon^{R,q}V_{k+1}^\varepsilon
\]
is exactly the conditional excess path pressure of the current-prepared Sinai
billiard.  It is monotone, cash additive, constant preserving, and strongly
time consistent.
\end{theorem}
\begin{proof}
Condition the canonical path partition function on the current quotient state
$\xi$.  The conditional future law is $K_{R,q}$.  Multiplicativity of the
exponential and the linear conditional tower give \eqref{eq:operator} and
backward induction.  Positivity gives monotonicity, while multiplication by
$e^{\theta a}$ gives cash additivity.
\end{proof}
