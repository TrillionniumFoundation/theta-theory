 =\bar\tau(c+bu_x)
 +\frac{\Sigma_{R,q}^{\rm phys}}2u_{xx}
 +\frac{\theta\Sigma_{R,q}^{\rm phys}}2(u_x)^2,
 \qquad u(T)=\Phi.
\]
\end{theorem}
\begin{proof}
The scheme is monotone and stable, consistency is
\cref{thm:consistency}, and scalar compactness is
\cref{prop:collapse}.  Uniform ellipticity holds on compact preparation sets.
For fixed law, the Cole--Hopf transform $w=e^{\theta u}$ converts the equation
to a linear uniformly parabolic equation (with the $\theta=0$ limit already
linear), giving comparison.  Apply the half-relaxed-limit theorem
\cite{BarlesSouganidis1991,CrandallIshiiLions1992}.
\end{proof}

Let
\[
 D_{R,q}=\frac{\Sigma_{R,q}^{\rm phys}}{\bar\tau_{R,q}}.
\]

\begin{theorem}[Physical theta-generator]\label{thm:physical-generator}
At a fixed physical horizon, the values converge to
\[
 -u_t=c+bu_x+\frac{D_{R,q}}2u_{xx}
 +\frac{qD_{R,q}}2(u_x)^2,
 \qquad u(T)=\Phi.
\]
Equivalently, the nonlinear generator is
\[
 \mathfrak G_{R,q}f
 =c+\frac1q e^{-qf}
 \left(b\partial_x+\frac{D_{R,q}}2\partial_{xx}\right)e^{qf},
\]
with continuous extension at $q=0$.
\end{theorem}
\begin{proof}
Paper II gives exponential control of the random collision window and the
roof-weighted work.  The entropic Orlicz comparison lemma below transfers this
control inside the logarithmic partition function.  Divide the collision
Hamiltonian by $\bar\tau$.
\end{proof}

\begin{lemma}[Entropic Orlicz comparison]\label{lem:orlicz}
For bounded random variables $X,Y$ and $q\ne0$,
\[
 \left|\frac1q\log\E e^{qX}-\frac1q\log\E e^{qY}\right|
 \le\frac1{\abs q}
 \max_{Q\in\{Q_X,Q_Y\}}
 \log\E_Qe^{\abs q\abs{X-Y}},
\]
where $dQ_X/dP=e^{qX}/\E e^{qX}$ and similarly for $Y$.
\end{lemma}
\begin{proof}
Use
$\E e^{qX}/\E e^{qY}=\E_{Q_Y}e^{q(X-Y)}$ and the reverse ratio.
\end{proof}

\section{Microcanonical preparation and order of limits}\label{sec:microcanonical}
Finite current conditioning is not Markov.  Let a preparation window of length
$2n$ surround an observation block containing all collisions used by a fixed
$\varepsilon$-problem.  Paper I gives convergence of its central conditional
law to the canonical driven law as $n\to\infty$.

\begin{theorem}[Preparation-before-homogenization principle]\label{thm:order}
For every fixed $\varepsilon$ and finite physical horizon, the
microcanonically conditioned value converges to the canonical value as the
central preparation window tends to infinity.  Taking $\varepsilon\to0$
afterwards yields \cref{thm:physical-generator}.  Equivalently, there exists a
diagonal sequence $n(\varepsilon)\to\infty$ for which the same conclusion
holds.
\end{theorem}
\begin{proof}
For fixed $\varepsilon$, the game uses finitely many collisions almost surely
up to the bounded roof horizon.  Truncate the exponentially small collision
count tail, apply the central Gibbs-conditioning theorem to the remaining
finite path sigma-field, and then remove the truncation.  A diagonal choice of
$n(\varepsilon)$ gives the joint sequence.
\end{proof}

\section{Type separation}\label{sec:types}
\begin{definition}[Three exports]
The \emph{fixed-law theta-semigroup} is \cref{thm:physical-generator}.  A
\emph{fixed-policy theta-semigroup} freezes a feedback before taking excess
pressure.  A \emph{controlled risk-sensitive value} optimizes over policies
outside the logarithmic expectation.  Only the first two have a pure
covariance second terminal variation; the controlled object has an additional
optimizer-response term treated in Paper IV.
\end{definition}

\section{Main theorem and scope}\label{sec:main}
\begin{theorem}[Liouville-to-theta time-production theorem]\label{thm:main}
For each current preparation $(R,j)$ in the local LDP range, the compact
periodic Sinai billiard yields, in one typed chain,
\[
\begin{aligned}
 \text{Liouville current conditioning}
 &\longrightarrow q=I_R'(j),\\
 &\longrightarrow \text{canonical excess-pressure DPP},\\
 &\longrightarrow \mathfrak G_{R,q}.
\end{aligned}
\]
The diffusion coefficient is the physical Green--Kubo coefficient of the
clock-corrected actual cell displacement.  Neither microscopic noise nor an
independent exponential-utility parameter is inserted.
\end{theorem}
\begin{proof}
