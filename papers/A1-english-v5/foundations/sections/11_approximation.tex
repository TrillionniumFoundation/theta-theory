\section{Experimental distances, normalization, and error propagation}\label{sec:approximation}
\subsection{Basic normalization estimates}
\begin{lemma}[Normalization of finite positive measures]\label{lem:normalization}
For positive finite measures $\alpha,\beta$ with masses $a,b>0$,
\[
 \norm{\alpha/a-\beta/b}_{\rm var}\le2\norm{\alpha-\beta}_{\rm var}/a.
\]
Consequently, for a common bounded $V$,
\[
 d_{\TV}(P^V,Q^V)\le2e^{\osc V}d_{\TV}(P,Q).
\]
If $\norm{V-W}_\infty\le\delta$, then
$d_{\TV}(P^V,P^W)\le e^{2\delta}-1$.
\end{lemma}
\begin{proof}
Write the normalized difference as $(\alpha-\beta)/a+\beta(1/a-1/b)$ and use $|a-b|\le\norm{\alpha-\beta}_{\rm var}$. For the tilted estimate subtract $\inf V$ so the weight lies in $[1,e^{\osc V}]$; both masses are at least one and the unnormalized variation difference is at most $e^{\osc V}\norm{P-Q}_{\rm var}$. Divide by two. For source error, $dP^V/dP^W$ lies between $e^{-2\delta}$ and $e^{2\delta}$, giving the stated, slightly conservative bound.
\end{proof}

\begin{proposition}[Rare-event and chronological budgets]\label{prop:tv-budget}
If $d_{\TV}(P,Q)\le\epsilon$ and $P(A),Q(A)\ge p_*>0$, then
\[
 d_{\TV}(P(\cdot\mid A),Q(\cdot\mid A))\le2\epsilon/p_*.
\]
If successive readout kernels differ by at most $\epsilon_n$ in TV at every common history and action, then under a common policy
$d_{\TV}(P_{0:N},Q_{0:N})\le\epsilon_{-1}+\sum_{n<N}\epsilon_n$, where $\epsilon_{-1}$ is the initial-record error.
\end{proposition}
\begin{proof}
Apply Lemma~\ref{lem:normalization} to the restricted positive measures $\one_AP$ and $\one_AQ$. For the path estimate use
\[
 \norm{\mu K-\nu L}_{\rm var}
 \le\norm{\mu-\nu}_{\rm var}+\int\norm{K_h-L_h}_{\rm var}\nu(dh).
\]
A common policy kernel is a variation contraction. Induction and division by two give the result. No independence between time steps is required.
\end{proof}

These bounds can be too strong for mechanical paths. The probabilities $\delta_x$ and $\delta_{x+h}$ have TV distance one for $h\ne0$, although any Lipschitz observation differs by at most its Lipschitz constant times $|h|$. We therefore give estimates for the tests actually used.

\subsection{Tilted experimental tests instead of microscopic TV}
Let $R$ be a report, $\mathcal G$ a class of report tests with $|f|\le B$, and define
\[
 Z=P e^V,\quad \widetilde Z=\widetilde P e^{\widetilde V},\qquad
 A(f)=P[e^Vf(R)],\quad\widetilde A(f)=\widetilde P[e^{\widetilde V}f(\widetilde R)].
\]
The microscopic sample spaces may differ. Only these report integrals are compared.

\begin{theorem}[Observable-specific tilted error]\label{thm:test-tilt}
If $Z,\widetilde Z\ge z_*>0$, $|Z-\widetilde Z|\le\epsilon_0$, and
$\sup_{f\in\mathcal G}|A(f)-\widetilde A(f)|\le\epsilon_\mathcal G$, then
\[
 \sup_{f\in\mathcal G}\left|\frac{A(f)}Z-\frac{\widetilde A(f)}{\widetilde Z}\right|
 \le\frac{\epsilon_\mathcal G+B\epsilon_0}{z_*},
 \qquad |\log Z-\log\widetilde Z|\le\epsilon_0/z_*.
\]
The same statement applies after event conditioning when $A,Z$ are replaced by their event-restricted integrals and the positive evidence bound is replaced accordingly.
\end{theorem}
\begin{proof}
Subtract the two normalized integrals, using denominator $Z$ for the numerator difference. The remaining term is bounded by
$|\widetilde A|\,|Z-\widetilde Z|/(Z\widetilde Z)\le B\epsilon_0/Z$.
The logarithm is $1/z_*$-Lipschitz on $[z_*,\infty)$. Event restriction changes none of this algebra; it changes the required lower bound and the test class.
\end{proof}

An experimental distance can be defined by taking these suprema also over a declared common continuation-policy family. It is not automatically stable under tilt, derivatives, or conditioning: the associated tests $e^V$, $e^Vf$, $\one_Ae^V$, and their derivatives must belong to the controlled class or be approximated with an explicit error.

\begin{proposition}[A mechanical coupling supplies weak-test errors]\label{prop:coupling}
Suppose reports $X,\widetilde X$ are coupled, $|V(X)|,|\widetilde V(\widetilde X)|\le M$, $|f|\le B$, and $f$ is $L_f$-Lipschitz. Then
\[
 |A(f)-\widetilde A(f)|
 \le e^M\{L_fE d(X,\widetilde X)+B E|V(X)-\widetilde V(\widetilde X)|\},
\]
while the denominator error is at most $e^M E|V(X)-\widetilde V(\widetilde X)|$. Both denominators are at least $e^{-M}$.
\end{proposition}
\begin{proof}
Split the difference into $e^V[f(X)-f(\widetilde X)]$ and
$f(\widetilde X)[e^V-e^{\widetilde V}]$. The exponential is $e^M$-Lipschitz on $[-M,M]$, and the report test is Lipschitz. The denominator calculation omits the report test. This remains meaningful for mutually singular deterministic microscopic laws.
\end{proof}

\subsection{Derivative errors cannot be inferred from value errors}
Let $N_a,\widetilde N_a$ be Banach-valued numerator curves, $Z_a,\widetilde Z_a$ positive scalar curves, all $C^r$ on a parameter interval, with $Z_a,\widetilde Z_a\ge z_*$. Set $U=N/Z$ and $\widetilde U=\widetilde N/\widetilde Z$. Suppose
\[
 \norm{N^{(j)}-\widetilde N^{(j)}}\le\epsilon_{N,j},\quad
 |Z^{(j)}-\widetilde Z^{(j)}|\le\epsilon_{Z,j},\quad
 |Z^{(j)}|\le M_j,\quad\norm{\widetilde U^{(j)}}\le B_j.
\]

\begin{theorem}[Normalized response error recursion]\label{thm:jet-error}
An upper bound $\Delta_j$ for $\norm{U^{(j)}-\widetilde U^{(j)}}$ is obtained recursively from
\begin{equation}
 \Delta_j=\frac1{z_*}\left[
 \epsilon_{N,j}+\epsilon_{Z,0}B_j+
 \sum_{k=1}^j\binom jk(M_k\Delta_{j-k}+\epsilon_{Z,k}B_{j-k})\right].
 \label{eq:response-error}
\end{equation}
For $j=0$ the sum is empty. Each $B_j$ can itself be computed from the quotient-jet recursion and bounds on the tilded derivatives.
\end{theorem}
\begin{proof}
Differentiate $ZU=N$ and $\widetilde Z\widetilde U=\widetilde N$ $j$ times. Subtract, keeping the coefficient $Z$ on the highest unknown difference. The term $(Z-\widetilde Z)\widetilde U^{(j)}$ gives the second error term. For every $k\ge1$, split
$Z^{(k)}U^{(j-k)}-\widetilde Z^{(k)}\widetilde U^{(j-k)}$
into $Z^{(k)}(U^{(j-k)}-\widetilde U^{(j-k)})$ and the coefficient error times the tilded derivative. Divide by $Z\ge z_*$ and use induction. The same Leibniz identity without subtraction computes $B_j$.
\end{proof}

For example $\epsilon^2\sin(a/\epsilon^3)$ tends uniformly to zero but its first derivative can diverge. An exceptional set with small probability is therefore not a response-tail estimate. Formula~\eqref{eq:response-error} makes every derivative and evidence cost explicit.

\subsection{Dynamic values and the correct continuation tests}
Suppose a true history $h$ is compressed to $s=\sigma_n(h)$, and the law of the next readout and compressed state differs in TV from a model kernel by at most $\epsilon_n$, uniformly over common feasible actions. Both models have the same feasible-action correspondence, reward and resource accounting on retained variables. Let $|r_n|\le R_0$, $|f|\le B_0$, and $\eta>0$.

\begin{theorem}[Finite-horizon value approximation]\label{thm:value-error}
The full and compressed resource-entropic values satisfy
\[
 \sup_h|v_n(h)-\bar v_n(\sigma_n(h))|
 \le\sum_{j=n}^{N-1}\frac{e^{2\eta[(N-j)R_0+B_0]}-1}{\eta}\epsilon_j.
\]
More generally, if the error on each actual normalized continuation test
$\exp\{\eta(g-\inf g)\}$ is at most $\zeta_j$, the corresponding bound is $\eta^{-1}\sum_{j=n}^{N-1}\zeta_j$. A double-sided kernel density-ratio bound $e^{-\alpha_j}\bar K\le K\le e^{\alpha_j}\bar K$ gives $\eta^{-1}\sum\alpha_j$ instead.
\end{theorem}
\begin{proof}
For bounded $g$,
\[
 |\eta^{-1}\log Pe^{\eta g}-\eta^{-1}\log Qe^{\eta g}|
 \le\eta^{-1}(e^{\eta\osc g}-1)d_{\TV}(P,Q).
\]
Subtract $\inf g$; the exponentials lie in $[1,e^{\eta\osc g}]$, their integrals are at least one, and their difference is bounded by oscillation times TV. If one controls the normalized exponential test directly, replace that bound by $\zeta_j$. A density ratio bounds the logarithm directly by $\alpha_j$. At stage $j$, the continuation payoff oscillation is at most $2[(N-j)R_0+B_0]$. The entropic operator is $1$-Lipschitz in its continuation value, and maxima over the same feasible set preserve uniform errors. Backward induction gives the three assertions. If feasible sets differ, an additional action-approximation or feasibility error is required.
\end{proof}

These estimates do not claim uniform stability at infinite time. Such stability requires an actual contraction, recurrence, or stronger likelihood-ratio mechanism appropriate to the model.
