\section{Finite spectral-gap alphabets}\label{sec:spectral}
Let $d\ge2$, $m=d-1$, and $\sigma$ be normalized measure on
$\mathbb S^{d-1}$. For a distribution $p$ on the finite alphabet, put
$Pf(v)=\sum_ap(a)f(U_av)$. Our hypothesis is the absolute $L^2$ gap
\begin{equation}\label{eq:gap}
 \norm{P|_{L^2_0(\sigma)}}\le\lambda<1,\qquad 0<\lambda<1.
\end{equation}
Laziness makes an upper spectral gap absolute for symmetric walks.
We assume \eqref{eq:gap} on the full mean-zero function space, not only
on the defining $d$-dimensional representation. Density alone is not
substituted for this analytic hypothesis.

\begin{lemma}[Uniform mixing for Lipschitz tests]\label{lem:mixing}
There is $C_d$ such that every real $1$-Lipschitz $f$ with values in
$[-1,1]$ satisfies
\begin{equation}\label{eq:mixing}
 \sup_v|P^Bf(v)-\sigma f|\le C_d\lambda^{2B/(m+2)}.
\end{equation}
For $d=3$ one can take $C_3=3$.
\end{lemma}
\begin{proof}
Orthogonal actions preserve distances, so $h=P^Bf-\sigma f$ is
$1$-Lipschitz. Its $L^2$ norm is at most $\lambda^B$, since
$\norm{f-\sigma f}_2^2$ is a variance bounded by one.
A chordal radius-$r$ ball has mass at least $c_dr^m$ for $0<r\le1$.
Averaging $|h(v)|\le |h(u)|+r$ over this ball and using
Cauchy--Schwarz gives
\[
 |h(v)|\le r+c_d^{-1/2}r^{-m/2}\lambda^B.
\]
Take $r=\lambda^{2B/(m+2)}$. The ball estimate follows from polar
integration and $\sin t\ge2t/\pi$ on $[0,\pi/2]$.
On $\mathbb S^2$ the chordal cap mass is exactly $r^2/4$;
taking $r=\lambda^{B/2}$ gives $3\lambda^{B/2}$.
\end{proof}

\begin{lemma}[Finite-word orbit distortion]\label{lem:finite-distortion}
There are constants $c_d>0$ and integers $B_k=O_{d,\lambda}(\log(k+1))$
such that the product law $\mu_{B_k}$ of $B_k$ independent command letters
satisfies
\begin{equation}\label{eq:finite-distortion}
 \Gamma_k(\mu_{B_k})\ge c_d k^{-2/m}.
\end{equation}
For $d=3$, the explicit choices
\begin{equation}\label{eq:B3}
 B_k=\left\lceil\frac{2\log(12k)}{-\log\lambda}\right\rceil,
 \qquad \Gamma_k(\mu_{B_k})\ge\frac1{4k}
\end{equation}
are valid.
\end{lemma}
\begin{proof}
A chordal radius-$r$ cap has mass at most $A_dr^m$, with $A_d\ge1$.
For any $k$ unit centers, choose $r=(2A_dk)^{-1/m}$. Outside their
union, of mass at least one half, the loss
$1-\max_i\langle v,u_i\rangle$ is at least $r^2/2$.
Consequently the spherical average loss is at least
$a_dk^{-2/m}$, where $a_d=(2A_d)^{-2/m}/4$.
The function $f(v)=\max_i\langle v,u_i\rangle$ is $1$-Lipschitz
and lies in $[-1,1]$. Lemma~\ref{lem:mixing} gives, uniformly in the
centers and starting direction,
\[
 P^Bf(v)\le1-a_d k^{-2/m}+C_d\lambda^{2B/(m+2)}.
\]
Choose $B_k$ so the last term is at most $a_dk^{-2/m}/2$.
This proves the first assertion. Powers of $P$ and chronologically
ordered products give the same law here, since the independent letters
have a common distribution and their joint law is invariant under
reversal.

On $\mathbb S^2$ take $r=\sqrt{2/k}\le\sqrt2$. The exact cap mass
$r^2/4$ gives spherical loss at least $1/(2k)$. Equation~\eqref{eq:B3}
makes $3\lambda^{B_k/2}\le1/(4k)$, proving the sharper constants.
Notice that $B_k$ counts ordinary inputs, not one Haar input.
\end{proof}

\begin{theorem}[Finite-alphabet width and occupation]\label{thm:spectral}
Assume \eqref{eq:gap}, $0<\rho\le1/10$, and
$0\le\varepsilon<\rho/(2\sqrt d)$. Then
\begin{equation}\label{eq:spectral-width}
 c_{d,\lambda,\rho,\varepsilon}
 \left(\frac N{\log(N+1)}\right)^{m/2}
 \le W_{N,\varepsilon}\le W_{N,0}\le C_d N^{m/2}.
\end{equation}
In particular $\lim_{N\to\infty}\log W_{N,\varepsilon}/\log N=m/2$.
Every available-label profile with this error satisfies
\begin{equation}\label{eq:spectral-occupation}
 \#\{1\le t\le N:K_t\le k\}
 \le B_k-1+c_d^{-1}B_k k^{2/m}\log(1/\kappa_d).
\end{equation}
The other cuts may have arbitrarily large registers. For $d=3$ one may
replace the second term by $4B_k k\log(1/\kappa_3)$ using \eqref{eq:B3}.
The exact upper realization has common command rows and an
$N$-dependent decoder.
\end{theorem}
\begin{proof}[Proof of the lower and occupation bounds]
Under the independent-letter law, every consecutive length-$B_k$ block
is a fresh packet. Lemma~\ref{lem:terminal} and
Theorem~\ref{thm:packet-budget} imply for a peak-$k$ machine
\begin{equation}\label{eq:gap-finite}
 \lfloor N/B_k\rfloor c_d k^{-2/m}\le\log(1/\kappa_d).
\end{equation}
Thus $N<B_k(1+c_d^{-1}k^{2/m}\log(1/\kappa_d))$.
As $B_k\le C\log(k+1)$, this implies the first inequality in
\eqref{eq:spectral-width}. Explicitly, if $k\ge N^{m/2}$ it is
immediate; otherwise $\log(k+1)\le C_m\log(N+1)$ and rearrangement
suffices. Constants absorb the finitely many small $N$.

For occupation, discard at most $B_k-1$ early endpoints and greedily
select low-width endpoints at spacing at least $B_k$. Each selected
interval removes at most $B_k$ candidate endpoints. If $l$ intervals
are selected, their total loss is at least $lc_dk^{-2/m}$, irrespective
of the internal widths. The packet budget bounds this by
$\log(1/\kappa_d)$ and proves \eqref{eq:spectral-occupation}.
The initialization cut is not in this count. The argument remains
valid for positive-mass states under the selected independent-letter law.
\end{proof}

\begin{lemma}[Rational spherical synthesis]\label{lem:net-upper}
For every $N\ge1$ there is a rational finite set
$V_N\subset\mathbb S^{d-1}$ of cardinality at most $C_d N^{m/2}$ such that
\[
 \eta_N\mathbb B^d\subset\conv V_N,\qquad \eta_N=1-\frac1{8N}.
\]
For finitely many orthogonal commands, this gives the exact upper bound
in \eqref{eq:spectral-width}. If the commands, seeds and $\rho$ are
rational, all transition, initialization and decoder entries may be
chosen rational, with at most $d+1$ successors in each command row.
\end{lemma}
\begin{proof}
Use the two stereographic charts
\[
 t\longmapsto\left(\frac{2t}{1+\norm t^2},
             \pm\frac{1-\norm t^2}{1+\norm t^2}\right),
 \qquad t\in[-1,1]^m.
\]
Every sphere point has a preimage in the unit ball in one chart.
Each chart is $2$-Lipschitz. A grid of mesh $1/M$, with
$M=\lceil2\sqrt{mN}\rceil$, is therefore within chordal distance
$\sqrt m/M$ of every sphere point. Its at most $2(2M+1)^m$ images
are rational. In every unit direction the support function of their
convex hull is at least $1-m/(2M^2)\ge\eta_N$.

Write $P=\conv V_N$, $a_t=\eta_N^{-t}/4$. From label $u\in V_N$
on command $a$, choose coefficients expressing $\eta_NU_au$ as a
convex combination of $V_N$. This is possible by ball containment.
The coefficients depend on $a,u,N$, not on the epoch. Initialize to
mean $\rho x$ in $a_0P$, since $\rho\le1/10<7/32\le\eta_N/4$.
At the final cut use mean decoder $a_N e_j^Tu$.
Bernoulli's inequality gives $\eta_N^N\ge7/8$, so $a_N\le2/7<1$.
The conditional mean update is
$a_{t+1}\E[u'\mid u,a]=a_tU_au$, proving exactness by induction.
Only the vertex label is retained.

For rational data, convex feasibility is a rational linear system.
A basic feasible solution has at most $d+1$ nonzero coefficients and
is rational; apply this separately to the finitely many rows and seeds.
An extra binary-output cut has two labels and does not change the bound.
This proves the upper assertion of Theorem~\ref{thm:spectral}.
\end{proof}

The mixing and quantization inequalities are classical analytic and
geometric mechanisms. Their use above is a finite-word, all-hidden-state
compression bound. A logarithmic mixing time is explicitly paid. Neither
an atomic Haar command nor a uniform-in-$d$ constant is used.
