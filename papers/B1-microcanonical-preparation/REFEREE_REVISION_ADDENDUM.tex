\section{Referee-round positive closure: source-dependent constrained pressure}
\label{sec:rr2-b1}

The fixed zero-source saddle is replaced by a source-dependent coefficient
extraction.  This section is controlling for every downstream use of
microcanonical preparation.

Let \(C_N\in\mathbb R^d\) collect particle number, total momentum, total
energy, and the finitely many prepared moments, divided by the
Boltzmann--Grad speed \(\mu_\epsilon\).  Let
\[
 Z_\epsilon(H,\lambda)
 =
 \mathbb E_{\rm hc}
 \exp\left\{
 \mu_\epsilon H+\mu_\epsilon\langle\lambda,C_N\rangle
 \right\}
\]
be the hard-core grand-canonical dynamical partition function, with bounded
path/collision source \(H\).

\begin{theorem}[Source-dependent microcanonical pressure]
\label{thm:rr2-b1-constrained}
Assume \(a\) lies in the interior of the admissible moment set and that the
source-decorated grand-canonical pressure
\[
 Q(H,\lambda)=
 \lim_{\epsilon\to0}\mu_\epsilon^{-1}\log Z_\epsilon(H,\lambda)
\]
is analytic for \(\|H\|<\delta\), \(|\lambda-\lambda_0|<\delta\), with
positive-definite constraint Hessian.  Then the thin-shell microcanonical
pressure exists and equals
\[
 P_a(H)
 =
 \inf_{\lambda\in\mathbb R^d}
 \{Q(H,\lambda)-\langle\lambda,a\rangle\}
 -
 \inf_{\lambda\in\mathbb R^d}
 \{Q(0,\lambda)-\langle\lambda,a\rangle\}.
\]
The minimizer \(\lambda(H)\) is analytic and satisfies
\[
 \nabla_\lambda Q(H,\lambda(H))=a .
\]
In particular the saddle is reoptimized when \(H\) changes.
\end{theorem}

\begin{proof}
Represent the shell indicator by Fourier inversion in the \(d\) constrained
coordinates, with a smooth window whose width tends to zero slower than
\(\mu_\epsilon^{-1/2}\).  Shift the contour through the unique real saddle
\(\lambda(H)\).  Uniform analyticity and positive Hessian give a Gaussian
main contribution
\[
 e^{\mu_\epsilon[Q(H,\lambda(H))-\langle\lambda(H),a\rangle]}
 \mu_\epsilon^{-d/2}
 \frac{(2\pi)^{d/2}}{\sqrt{\det\nabla^2_{\lambda\lambda}Q}}
 (1+o(1)).
\]
Outside a fixed saddle neighborhood, strict convexity and the complex
Fourier gap give \(e^{-c\mu_\epsilon}\) relative decay.  The identical
expansion at \(H=0\) cancels the polynomial factors and gives the displayed
difference.  The implicit-function theorem gives analyticity of
\(\lambda(H)\).
\end{proof}

\begin{corollary}[Resolution of the time-zero-source test]
\label{cor:rr2-b1-test}
If a constrained component is
\(C_N^\chi=N^{-1}\sum_i\chi(z_i)\) and
\(H=tC_N^\chi\), then
\[
 P_a(tC^\chi)=ta.
\]
\end{corollary}

\begin{proof}
The source is absorbed by translating the corresponding multiplier:
\(Q(tC^\chi,\lambda)=Q(0,\lambda+te_\chi)\).  Changing variables in the
infimum gives the value \(ta\).  Thus the strict-convexity discrepancy found
by the referee is exactly cancelled by the source-dependent saddle shift.
\end{proof}

\subsection{Hard-core coefficient extraction}

\begin{lemma}[Source-uniform dilute correction]
\label{lem:rr2-b1-hardcore}
For bounded source norm below \(\delta\), the logarithm of the ratio between
the hard-core and ideal-gas source-decorated partition functions is
\(O(N_\epsilon^2\epsilon^3)\), uniformly in the complex saddle
neighborhood.  Hence
\[
 \mu_\epsilon^{-1}
 O(N_\epsilon^2\epsilon^3)=O(\epsilon)\to0.
\]
\end{lemma}

\begin{proof}
Expand the exclusion indicator in Mayer bonds.  A connected graph with
\(k\) vertices has integral bounded by
\(C^k\epsilon^{3(k-1)}\), while the bounded dynamical source changes each
vertex activity by at most \(e^{C\delta}\).  The tree-graph inequality sums
connected graphs and gives an exponential whose logarithm begins at
\(N_\epsilon^2\epsilon^3\).  The same estimate holds after the complex
constraint shift because the velocity exponential moments are uniform in
the saddle neighborhood.
\end{proof}

\begin{theorem}[Uniform coefficient-extraction error]
\label{thm:rr2-b1-coeff}
For the declared shell widths and source ball,
\[
 \log\mathbb E_{\rm mc,a}
 e^{\mu_\epsilon H}
 =
 \mu_\epsilon P_a(H)+o(\mu_\epsilon)
\]
uniformly.  The \(o(\mu_\epsilon)\) includes the hard-core correction,
Gaussian coefficient factor, and shell-smoothing error.
\end{theorem}

\begin{proof}
Combine Theorem~\ref{thm:rr2-b1-constrained} with
Lemma~\ref{lem:rr2-b1-hardcore}.  The smoothing window contributes
\(o(\mu_\epsilon)\) because its logarithmic width is \(o(\mu_\epsilon)\);
the Gaussian determinant contributes \(O(\log\mu_\epsilon)\).  All estimates
are uniform on the compact source ball.
\end{proof}

\subsection{First and second derivatives}

Let \(G(H,\lambda)=Q(H,\lambda)-\langle\lambda,a\rangle\).  Differentiating
the saddle equation gives
\[
 D\lambda(H)
 =
 -G_{\lambda\lambda}^{-1}G_{\lambda H}.
\]

\begin{proposition}[Constrained tangent and Schur complement]
\label{prop:rr2-b1-schur}
At a smooth source,
\[
 DP_a=G_H,\qquad
 D^2P_a
 =
 G_{HH}
 -
 G_{H\lambda}G_{\lambda\lambda}^{-1}G_{\lambda H}.
\]
Thus the preparation response is the canonical covariance corrected by the
constraint-saddle Schur complement.
\end{proposition}

\begin{proof}
Apply the envelope theorem to the analytic minimizing saddle.  One
differentiation removes the \(\lambda'\)-term by \(G_\lambda=0\).
A second differentiation and substitution of the displayed formula for
\(D\lambda\) gives the Schur complement.
\end{proof}

\subsection{Conditional initial empirical-measure rate}

\begin{theorem}[Prepared initial LDP]
\label{thm:rr2-b1-initial-ldp}
The initial empirical measure under the microcanonical shell satisfies an LDP
on the weighted weak topology with rate
\[
 I_{0,a}(f)=
 \begin{cases}
 \operatorname{Ent}(f\mid f_{\rm ref})
 -\displaystyle\inf_{C(g)=a}
  \operatorname{Ent}(g\mid f_{\rm ref}),
 & C(f)=a,\\[1ex]
 +\infty,& C(f)\ne a.
 \end{cases}
\]
On the interior admissible set it has a unique zero \(f_a^0\), and its
quadratic tangent is the reference covariance projected orthogonally to the
constraint directions.
\end{theorem}

\begin{proof}
Sanov's theorem with exponential velocity moments gives the ideal-gas rate.
Lemma~\ref{lem:rr2-b1-hardcore} is subexponential at the LDP speed, so
exponential equivalence transfers the rate to the dilute hard-core law.
Conditioning on shrinking continuity neighborhoods of \(C(f)=a\) gives the
conditional rate.  Strict entropy convexity yields uniqueness; the Hessian
restricted to the tangent space of the constraints gives the projected
covariance.
\end{proof}
