\begin{abstract}
We isolate the preparation layer of the deterministic hard-sphere programme.
Starting from a thin microcanonical Liouville shell with fixed particle number,
momentum, energy, and finitely many smooth macroscopic constraints, we prove an
exponential-scale transfer to the corresponding source-dependent information
projection.  The hard-core static correction is shown to be negligible at the
Boltzmann--Grad large-deviation speed.  This theorem is the sole preparation
input used by the collision-cluster and kinetic-semigroup papers.
\end{abstract}

\section{Deterministic hard-sphere platform and Liouville preparation}
Fix $d=3$ and let $\varepsilon>0$.  For $N$ unit-mass hard spheres of diameter
$\varepsilon$ in $\T^3$, set
\[
 \mathcal D_N^\varepsilon
 =\{(X_N,V_N): |x_i-x_j|>\varepsilon\ \text{for }i\ne j\}.
\]
Between collisions $\dot x_i=v_i$, $\dot v_i=0$; at a binary contact with
normal $\omega=(x_i-x_j)/\varepsilon$,
\[
 v_i'=v_i-((v_i-v_j)\cdot\omega)\omega,\qquad
 v_j'=v_j+((v_i-v_j)\cdot\omega)\omega.
\]
The flow $\Phi_{N,\varepsilon}^t$ is defined on a full Liouville-measure set and
preserves particle number, total momentum and kinetic energy.  This is the
classical deterministic setting of the Boltzmann--Grad program
\cite{Lanford1975,GallagherSaintRaymondTexier2014}.

Put $\mu_\varepsilon=\varepsilon^{-2}$ and
$N_\varepsilon=\lfloor\mu_\varepsilon\rfloor$.  Given mean momentum $u$ and
specific kinetic energy $e>|u|^2/2$, let
$\lambda_{\varepsilon}^{\rm mc}$ be normalized Liouville volume on
\[
 \begin{aligned}
 \mathcal S_\varepsilon(u,e;\eta_\varepsilon)
 =\bigg\{Z_{N_\varepsilon}\in\mathcal D_{N_\varepsilon}^\varepsilon:\;&
 \left|N_\varepsilon^{-1}\sum_i v_i-u\right|\le\eta_\varepsilon,\\
 &\left|N_\varepsilon^{-1}\sum_i\frac{|v_i|^2}{2}-e\right|
 \le\eta_\varepsilon\bigg\}.
 \end{aligned}
\]
Here $\mu_\varepsilon^{-1/2}\ll\eta_\varepsilon\downarrow0$.  This is the
primitive microcanonical preparation.  Canonical or grand-canonical ensembles
below are technical representations justified by an explicit transfer theorem,
not replacements of the physical preparation; compare \cite{Touchette2015}.

Let $\chi=(\chi_1,\ldots,\chi_m)$ be bounded $C^1$ observables with Gaussian
velocity weights, linearly independent modulo $1,v,|v|^2$.  Condition on
\[
 \left|\frac1{N_\varepsilon}\sum_i\chi(z_i)-a\right|
 \le\delta_\varepsilon,
 \qquad \mu_\varepsilon^{-1/2}\ll\delta_\varepsilon\downarrow0.
\]
The information projection is
\[
 f_a^0(x,v)=Z_a^{-1}M_{u,\beta}(v)
 \exp\{\lambda_a\cdot\chi(x,v)\},
\]
where $\beta$ and $\lambda_a$ enforce all constraints.

\begin{theorem}[Microcanonical preparation transfer]\label{thm:mc-transfer}
Assume the constraint covariance is nonsingular at $a$.  For every fixed
source radius $r$ there is $T_*(r)>0$ such that, uniformly over bounded
single-particle path sources and bounded collision sources of norm at most
$r$, the conditioned microcanonical and hard-core grand-canonical
log-Laplace functionals differ by $o(\mu_\varepsilon)$ on $[0,T_*(r)]$.
The technical one-particle density is $f_a^0$.
\end{theorem}

\begin{proof}
Use a joint Fourier--Laplace inversion for particle number, momentum, energy
and the $\chi$ constraints.  The nonsingular covariance gives a unique saddle
whose one-particle activity is $f_a^0$.  The complement of a fixed saddle
neighbourhood is exponentially smaller by strict convexity.  Inside that
neighbourhood, hard-core exclusion changes the logarithm of the partition
function by
\[
 O(N_\varepsilon^2\varepsilon^3)
 =O(\varepsilon^{-1})=o(\mu_\varepsilon).
\]
The short-time dynamical cluster pressure is analytic uniformly in the
root activity and in the two source families.  Substitution of the saddle
activity therefore changes the pressure by $o(1)$ per particle.  Multiplying
by $\mu_\varepsilon$ and combining the central and tail regions proves the
claim.
\end{proof}

\begin{remark}
The theorem is an exponential-scale equivalence, not total-variation
equivalence of the full $N$-particle laws.
\end{remark}
