
\section{Strong filter jets and posterior uncertainty for memory}
\label{sec:filter-memory}

The initial-state likelihood factor in \eqref{eq:nuisance-factorization}
defines the normalized probability
\[
 w_{n,\vartheta}^{\nu_n}(dz)
 =e^{V_n^\vartheta(z)-R_n^{\nu_n}(\vartheta)}\nu_n(dz).
\]
The physical current-state filter at fixed parameter is
\begin{equation}
 \Pi_{n,\vartheta}^{\nu_n}
 =\bigl(z\mapsto x_n^\vartheta+T_\vartheta(t_n)z\bigr)_\#
   w_{n,\vartheta}^{\nu_n},
 \qquad T_\vartheta(t)=e^{tA_\vartheta}.
 \label{eq:physical-filter}
\end{equation}

\subsection{The Banach spaces and strong derivatives}

For an integer \(r\ge0\), let \(C_b^r(\Hh)\) be the Banach space of bounded
functions with bounded continuous Fr\'echet derivatives through order \(r\),
with norm
\[
 \norm\phi_{C_b^r}
 =\max_{0\le j\le r}
 \sup_{x\in\Hh}\norm{D^j\phi(x)}_{\operatorname{Sym}^j(\Hh)^*}.
\]
Parameter derivative tensors use the operator norm on
\(\operatorname{Sym}^j(\R^5; (C_b^r(\Hh))')\).  For a \(C^r\) curve
\(y_\vartheta\in\Hh\), the notation
\(\partial_\vartheta^j\delta_{y_\vartheta}\) means the strong derivative in
\((C_b^{j+1}(\Hh))'\); on a test function it is the corresponding chain-rule
combination of \(D^\ell\phi(y_\vartheta)\) and the trajectory derivatives.

The following deterministic lemma is the functional-analytic step hidden by a
pointwise-on-tests calculation.

\begin{lemma}[Concentrated push-forward jets]
\label{lem:strong-pushforward}
Let \(U\subset\R^d\) be compact.  Suppose \(x_\theta\in\Hh\) and
\(S_\theta\in\mathcal L(\Hh)\) are \(C^{r+1}\), with uniformly bounded
derivatives, and
\[
 \sup_{\theta\in U,\abs\alpha\le r+1}
 \norm{\partial_\theta^\alpha S_\theta}\le\varepsilon.
\]
Let \(\mu_\theta\) be probabilities supported in the radius-\(R\) ball and
\(C^r\) in total variation, with
\(\sup_{\abs\alpha\le r}\norm{\partial_\theta^\alpha\mu_\theta}_{TV}\le M\).
Then
\[
 \theta\mapsto (x_\theta+S_\theta z)_\#\mu_\theta
\]
is strongly \(C^r\) into \((C_b^{r+1}(\Hh))'\), and for \(0\le j\le r\),
\begin{equation}
 \sup_{\theta\in U}
 \norm{\partial_\theta^j[(x_\theta+S_\theta z)_\#\mu_\theta]
       -\partial_\theta^j\delta_{x_\theta}}_{(C_b^{j+1})'}
 \le C_{r,R,M}\varepsilon.
 \label{eq:strong-pushforward-bound}
\end{equation}
The derivative-valued maps are strongly Borel measurable.  The supremum in
\eqref{eq:strong-pushforward-bound} is measurable.
\end{lemma}

\begin{proof}
For \(\norm\phi_{C_b^{r+1}}\le1\), Taylor's formula with integral remainder
gives, uniformly in \(\theta\) and \(\norm z\le R\),
\[
 \abs{\phi(x_\theta+S_\theta z)-\phi(x_\theta)}\le R\varepsilon.
\]
Differentiate this difference at most \(r\) times.  Every resulting term
contains either \(S_\theta z\) or a parameter derivative of \(S_\theta z\);
all remaining factors are bounded trajectory derivatives and derivatives of
\(\phi\) through order \(r+1\).  Hence its absolute value is at most
\(C\varepsilon\), uniformly over the test unit ball.  Leibniz's rule after
integration against \(\mu_\theta\) proves \eqref{eq:strong-pushforward-bound}.

For strong differentiability, apply the Banach-space Taylor formula to
\(\phi(x_{\theta+h}+S_{\theta+h}z)\), uniformly in the same test unit ball and
in \(z\), and use total-variation Taylor expansion of \(\mu_{\theta+h}\).
The remainder divided by \(\abs h^r\) tends to zero uniformly.  This is norm,
not merely weak-on-tests, convergence.  Continuity of the derivative maps on
the compact parameter set implies strong Borel measurability in the separable
closed subspace generated by their continuous range.  The dual norm is then
the supremum of a countable dense subset of that range's test restrictions,
which proves measurability of the displayed supremum.
\end{proof}

\begin{theorem}[Time-uniform strong filter jets]
\label{thm:filter-jets}
For each fixed integer \(r\ge0\) and \(q<\infty\), there are
\(C_{r,q},\gamma_{r,q}>0\) such that
\begin{equation}
 \sup_{E\in\frE_n}
 \E_E\left[
 \sup_{\vartheta\in\Theta}
 \norm{\partial_\vartheta^j\Pi_{n,\vartheta}^{\nu_n}
       -\partial_\vartheta^j\delta_{x_n^\vartheta}}_{(C_b^{j+1})'}^q
 \right]
 \le C_{r,q}e^{-\gamma_{r,q}n},
 \qquad 0\le j\le r.
 \label{eq:filter-jet-decay}
\end{equation}
The assertion holds for arbitrary Borel \(\nu_n\) supported in the Hilbert
ball.  For two such working priors, the difference of their filter jets is
bounded by the sum of the two right sides.  In particular,
\[
 W_1(\Pi_{n,\vartheta}^{\nu_n^1},
     \Pi_{n,\vartheta}^{\nu_n^2})
 \le2MRe^{-\lambda n\tau}
\]
holds deterministically for the undifferentiated filters.
\end{theorem}

\begin{proof}
Lemma~\ref{lem:nuisance-budget} and explicit differentiation of the normalized
density show that \(w_{n,\vartheta}^{\nu_n}\) is strongly \(C^r\) in total
variation.  For example,
\[
 \partial_jw=(\partial_jV-\E_w\partial_jV)w,
\]
and higher derivatives are finite polynomials in centered derivatives of
\(V\).  Their total-variation norms are bounded by
\(C_r(1+W_{n,r,E})^{C_r}\), whose moments are uniform.
Lemma~\ref{lem:five-energy} gives
\[
 \sup_{\vartheta,\abs\alpha\le r+1}
 \norm{\partial_\vartheta^\alpha T_\vartheta(t_n)}
 \le C_re^{-\lambda_rt_n}\le C_re^{-\lambda_rn\tau}.
\]
Apply Lemma~\ref{lem:strong-pushforward} pathwise with
\(S_\vartheta=T_\vartheta(t_n)\), then take the \(q\)-th moment.  H\"older's
inequality and the polynomial moment bound for the weights prove
\eqref{eq:filter-jet-decay}.  Coupling any two posterior initial states in the
radius-\(R\) ball gives the deterministic Wasserstein estimate.
\end{proof}

\begin{corollary}[Finite-rank current-state Gaussian image]
\label{cor:state-image}
Let \(G_n=D_\vartheta x_n^{\vartheta_0}:\R^5\to\Hh\).  Under the joint
parameter--initial-state quasi-posterior, the law of
\(\sqrt n\{z(t_n)-x_n^{\vartheta_0}\}\) is at bounded-Lipschitz distance
\(o_{\frE}(1)\) from the image of
\(N(I_n^{-1}\Delta_n,I_n^{-1})\) under \(G_n\).  Its covariance is
\(G_nI_n^{-1}G_n^*\), of rank at most five.
\end{corollary}

\begin{proof}
On every fixed \(h\)-ball,
\[
 \sqrt n\{x_n^{\vartheta_0+h/\sqrt n}-x_n^{\vartheta_0}\}
 =G_nh+O(\abs h^2/\sqrt n)
\]
uniformly, while the posterior initial-state displacement is at most
\(\sqrt nMRe^{-\lambda n\tau}\).  Use Theorem~\ref{thm:lattice-bvm}, its
root-\(n\) tightness, and Theorem~\ref{thm:filter-jets}.  A bounded-Lipschitz
push-forward cannot increase total variation, and the two deterministic
remainders vanish.  The covariance statement is immediate.
\end{proof}

\section{The inferred bath produces an inferred memory kernel}
\label{sec:memory}

Resolve \(r=(q_0,v_0)\in\R^2\) and put every site \(j\ge1\) in the unresolved
state.  The unresolved generator is
\[
 A_Q(\epsilon,c_b,k_b)
 =\begin{pmatrix}0&I\\-(k_bI+\epsilon L_D)&-c_bI\end{pmatrix},
\]
where \(L_D\) is the Dirichlet half-line Jacobi matrix with diagonal two and
off-diagonal minus one.  The resolved and coupling blocks are
\[
 A_P(c,k,\epsilon)=
 \begin{pmatrix}0&1\\-(k+\epsilon)&-c\end{pmatrix},
\]
while \(B_Qr\) inserts \(\epsilon q_0\) into the acceleration at site one and
\(C_Qz_Q=(0,\epsilon q_1)\).

Variation of constants gives the exact generalized Langevin equation
\begin{equation}
 \dot r(t)=A_Pr(t)+\binom0{u(t)}
 +\int_0^t\mathcal K_\vartheta(t-s)r(s)ds
 +C_Qe^{tA_Q}z_{Q,0},
 \qquad
 \mathcal K_\vartheta(t)=C_Qe^{tA_Q}B_Q.
 \label{eq:gle}
\end{equation}
For \(0<\omega<\inf_\Theta\lambda_Q(\vartheta)\), let
\[
 \mathcal X_\omega=\{K\in C([0,\infty);\mathcal L(\R^2)):
 \norm K_\omega=\sup_{t\ge0}e^{\omega t}\norm{K(t)}<\infty\}.
\]
Duhamel differentiation and uniform stability show that
\begin{equation}
 \mathscr M:\Theta\to\mathcal X_\omega,\qquad
 \mathscr M(\vartheta)=\mathcal K_\vartheta,
 \label{eq:memory-map}
\end{equation}
is \(C^\infty\), with bounded derivatives of every fixed order.

For \(\operatorname{Re}z\) large, put
\[
 p_\vartheta(z)=z^2+c_bz+k_b+2\epsilon.
\]
A scalar Schur complement of the constant Jacobi tail gives
\begin{equation}
 \widehat{\mathcal K}_\vartheta(z)=
 \begin{pmatrix}0&0\\s_\vartheta(z)&0\end{pmatrix},
 \qquad
 s_\vartheta(z)=
 \frac{p_\vartheta(z)-
 \sqrt{p_\vartheta(z)^2-4\epsilon^2}}2,
 \label{eq:memory-transform}
\end{equation}
where the branch is determined by
\(s_\vartheta(z)\sim\epsilon^2z^{-2}\) on the positive real axis.  The
boundary transfer function is consequently
\begin{equation}
 \widehat h_\vartheta(z)=
 \frac1{z\{z^2+cz+k+\epsilon-s_\vartheta(z)\}}.
 \label{eq:boundary-transfer}
\end{equation}
Thus \((\epsilon,c_b,k_b)\), inferred in
Theorem~\ref{thm:lattice-bvm}, determine both the memory strength and its full
shape.

\begin{corollary}[Functional posterior for the memory kernel]
\label{cor:memory-posterior}
Let \(D\mathscr M_{\vartheta_0}:\R^5\to\mathcal X_\omega\) be the derivative
of \eqref{eq:memory-map}.  Under the parameter quasi-posterior,
\begin{equation}
 \BL\left(
 \mathcal L\{\sqrt n(\mathcal K_\vartheta-
 \mathcal K_{\vartheta_0})\mid\cF_n\},
 (D\mathscr M_{\vartheta_0})_\#
 N(I_n^{-1}\Delta_n,I_n^{-1})
 \right)=o_{\frE}(1)
 \label{eq:memory-posterior}
\end{equation}
in the Banach space \(\mathcal X_\omega\).  The same conclusion holds for
the transform on every fixed right half-plane on which the resolvents are
uniformly bounded.
\end{corollary}

\begin{proof}
The bounded second derivative of \(\mathscr M\) gives, on a fixed \(h\)-ball,
\[
 \sqrt n\{\mathscr M(\vartheta_0+h/\sqrt n)-
 \mathscr M(\vartheta_0)\}
 =D\mathscr M_{\vartheta_0}h+O(\abs h^2/\sqrt n)
\]
in \(\mathcal X_\omega\).  Combine this with the total-variation
approximation and root-\(n\) tightness in
Theorem~\ref{thm:lattice-bvm}.  The resolvent version follows from the
resolvent identity and the same uniform derivative bounds.
\end{proof}
