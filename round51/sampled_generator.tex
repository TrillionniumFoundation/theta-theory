\section{A single-atom sampled inverse and weighted certificate}
\label{sec:sampled}
Fix the atom in Theorem~\ref{thm:main} and put
\[
 \Delta=2t_0,\quad U_\beta=e^{\Delta A_\beta},\quad X_\beta=U_\beta-I,
 \quad H_\beta=\int_0^{t_0}e^{uA_\beta}\,du.
\]
Write the pulse responses as $b_{k,\beta}=\ell U_\beta^kH_\beta B
=h_\beta(k\Delta+t_0)-h_\beta(k\Delta)$.
The extra parameter subscript distinguishes these responses from the physical
diagonal coefficients $b_k$.

\begin{lemma}[Channel-corrected transform with geometric remainder]
\label{lem:sampled}
For $N\ge R=4J+8$, there are parameter-independent coefficients $T_{r l}$,
$1\le r\le R$, $0\le l\le N$, such that, with
\begin{equation}
 \mathcal A=8\Delta^{-R}4^N,\qquad
 \mathcal E=\Delta^{-R}16^{-N},\qquad
 \mathcal W=\max_{1\le r\le R}\sum_{l=0}^N\frac{T_{rl}^2}{N+1-l},
 \label{eq:transform-budgets}
\end{equation}
we have $0<\mathcal W\le\mathcal A^2$. For an admissible pair put
$\mathcal D_R=\max_{0\le r\le R}|h_\beta^{(r)}(0)-h_\gamma^{(r)}(0)|$.
The following two bounds hold:
\begin{align}
 \mathcal D_R&\le\mathcal A\max_{l\le N}|b_{l,\beta}-b_{l,\gamma}|+\mathcal E,
 \label{eq:sampled-infty}\\
 \mathcal D_R&\le\sqrt{\mathcal W}
 \left(\sum_{l=0}^N(N+1-l)|b_{l,\beta}-b_{l,\gamma}|^2\right)^{1/2}
 +\mathcal E.\label{eq:sampled-weighted}
\end{align}
For rational $\Delta$, every $T_{rl}$ and $\mathcal W$ is rational.
\end{lemma}
\begin{proof}
The fast-clock assumption gives $\|X_\beta\|\le e^{1/64}-1\le1/32$
and $\Delta\le1$. Absolutely convergent scalar power-series identities remain
valid in the Banach algebra generated by $X_\beta$, and give
$\log(I+X_\beta)=\Delta A_\beta$ and
$(I+X_\beta)^{1/2}=e^{\Delta A_\beta/2}$. For $r\ge1$ define
\[
 F_r(x)=\Delta^{-r}[\log(1+x)]^{r-1}
       \frac{\log(1+x)}{x}(\sqrt{1+x}+1)=\sum_{k\ge0}c_{rk}x^k.
\]
Removable quotients are defined by their analytic continuations.
Since $(\sqrt{1+x}-1)(\sqrt{1+x}+1)=x$, we obtain
$F_r(X_\beta)H_\beta=A_\beta^{r-1}$; this is the correction for the
\emph{integrated} short-pulse channel, not a substitution of it for an impulse.
On $|x|=1/2$, $|\log(1+x)|\le\log2<1$,
$|\log(1+x)/x|\le2$ and $|\sqrt{1+x}+1|<3$. Thus
$|c_{rk}|\le6\Delta^{-R}2^k$, while
$\|H_\beta\|\le t_0e^{t_0\Lambda}\le\Delta$.
The difference of the two tails after degree $N$ is at most
\begin{equation}
 12\Delta^{1-R}\frac{16^{-(N+1)}}{1-1/16}
 =\frac45\Delta^{1-R}16^{-N}\le\mathcal E.
 \label{eq:sharp-tail}
\end{equation}
No replacement of $16^{-N}$ by a slower geometric rate is necessary.
Expanding $(U-I)^k$, set
\[
 T_{rl}=\sum_{k=l}^N c_{rk}(-1)^{k-l}\binom{k}{l}.
\]
Its row one-norm is at most
$6\Delta^{-R}\sum_{k=0}^N4^k\le\mathcal A$.
This proves \eqref{eq:sampled-infty}; weighted Cauchy--Schwarz proves
\eqref{eq:sampled-weighted}. Also
$\sum_lT_{rl}^2/(N+1-l)\le(\sum_l|T_{rl}|)^2$.
The first transform row is nonzero, since its polynomial evaluated at $U=1$
is $c_{10}=2/\Delta$. Rational power-series arithmetic proves rationality.
\end{proof}

For $j=J+1$, $d_0=\log\Delta^{-1}$, $v=\log(1/\delta)$, $0<\delta\le1$, set
\begin{equation}
 N=\max\left\{R,\left\lceil\frac{\log2+25j\log Q+Rd_0+v}{\log16}
                   \right\rceil\right\},\quad m=N+1,\quad
 p_*=1-\rho+\rho w.
 \label{eq:choose-N}
\end{equation}
Then $L\mathcal E\le\delta/2$. Define the two valid certificates
\begin{equation}
 \kappa=p_*^m\frac{\delta^2}{4L^2\mathcal A^2},\qquad
 \kappa_W=p_*^m\frac{\delta^2}{4L^2\mathcal W}\ge\kappa.
 \label{eq:kappa}
\end{equation}
The weighted version is exactly computable at rational inputs without taking
a square root. The elementary version gives a convenient closed-form region:
\begin{equation}
 \begin{split}
 K_b&=12+\frac{\log2+25\log Q+8d_0}{\log16},\qquad
 A_b=\log(1/p_*)+\log16,\\
 C_b&=\log256+50\log Q+16d_0+K_bA_b,\qquad
 p_b=2+A_b/\log16=3+\log(1/p_*)/\log16.
 \end{split}\label{eq:constants}
\end{equation}

\begin{proposition}[Explicit resource envelope]
\label{prop:envelope}
For all $J\ge0$, $0<\delta\le1$,
$m\le K_bj+v/\log16$ and $\kappa_W\ge\kappa\ge e^{-C_bj}\delta^{p_b}$.
\end{proposition}
\begin{proof}
Use $R\le8j$, the ceiling error at most one, and $j\ge1$ in
\eqref{eq:choose-N}. Moreover,
\[
 -\log\kappa=m\log(1/p_*)+2v+\log256+50j\log Q+2Rd_0+N\log16
 \le C_bj+p_bv.
\]
The whole-block probability, not a one-stage surrogate, appears here.
\end{proof}

\begin{corollary}[The original bounded multiscale law]
\label{cor:original}
For the labeled law with half its mass on a dense rational enumeration of
$(0,T]$ and masses $2^{-K-s-2}/R_K$ at
$k\zeta_{R_K}2^{-s}$, where $R_K=4K+8$,
$\zeta_R=\min(1,T/(2R))$, choose
\[
 s_0=\min\{s\ge1:2\zeta_8 2^{-s}\Lambda\le1/64\},\quad
 t_0=\zeta_8 2^{-s_0},\quad w_0=2^{-s_0-5}.
\]
Theorem~\ref{thm:main} applies with $w=w_0$, without reallocating any mass.
Counting any additional coincident durations only improves the bound.
\end{corollary}
