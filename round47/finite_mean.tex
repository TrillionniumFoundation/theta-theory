\section{A finite-mean probe law with exponential depth cost}
\label{sec:long}
Set $u_0=1/(256\Lambda)$. For $K\ge0$ and $1\le k\le N_K=4K+8$, define
\begin{equation}
 t_{K,k}=ku_0,\qquad \nu_{K,k}=2^{-K-1}/N_K.
 \label{eq:long-law}
\end{equation}
This law has total mass one. Its duration is unbounded, but its first moment
and an exponential moment are explicit. No dense auxiliary family is needed:
the following theorem itself proves injectivity of the discrete response map.

For $j,R,L,v$ as above put
\begin{equation}
\begin{split}
 H_\infty&=\max\left\{8,R,
 \frac{\log4+25j\log Q+R\log(u_0^{-1})+v}{\log4}\right\},\\
 N_\infty&=4\lceil H_\infty/4\rceil,\quad
 A_\infty=8^{N_\infty}u_0^{-R},\quad
 p_\infty=2^{-N_\infty/4+1}/N_\infty,\\
 \kappa_\infty(J,\delta)&=\frac{p_\infty\delta^2}{4L^2A_\infty^2},\\
 K_\infty&=24+25\log Q+8\log(u_0^{-1}),\\
 C_\infty&=1+50\log Q+16\log(u_0^{-1})+6K_\infty.
\end{split}\label{eq:long-certificate}
\end{equation}

\begin{theorem}[Exponential separation on a finite-mean duration law]
\label{thm:long}
For every $d_J(\beta,\gamma)\ge\delta$, $0<\delta\le1$,
\begin{equation}
 D_\nu(\beta,\gamma)\ge\kappa_\infty(J,\delta)
 \ge e^{-C_\infty(J+1)}\delta^8.
 \label{eq:long-separation}
\end{equation}
The law \eqref{eq:long-law} satisfies $\E_\nu t=13u_0/2$. With
$\theta_0=(\log2)/(8u_0)$ and $M_0=(1-1/\sqrt2)^{-1}$,
$\E_\nu e^{\theta_0t}\le M_0$.
\end{theorem}
\begin{proof}
Let $N=N_\infty$ and use Lemma~\ref{lem:grid} with spacing $u_0$. The bound
$8e^{\Lambda u_0}e\Lambda u_0\le48/256<1/4$ and $e\Lambda u_0<1$ give
\[
 A_\infty E_{N,u_0}\le 2u_0^{-R}4^{-N}.
\]
Our choice of $N$ makes $LA_\infty E_{N,u_0}\le\delta/2$. A positive grid
value is at least $\delta/(2LA_\infty)$, with label mass $p_\infty$, proving
the first bound. Rounding and the defining maximum give
$N\le K_\infty j+v$. As $\log N\le N$ and $a_0+1<6$,
\begin{align*}
 -\log\kappa_\infty
 &=\log2+\log N+2v+50j\log Q+a_0N+2R\log(u_0^{-1})\\
 &\le C_\infty j+8v.
\end{align*}
For the clock, $K$ has probabilities $2^{-K-1}$ and mean one. Conditional on
$K$, the mean duration is $u_0(N_K+1)/2$. Hence the overall mean is
$13u_0/2$. For $0<\theta<(\log2)/(4u_0)$,
\[
 \E e^{\theta t}\le\sum_{K\ge0}2^{-K-1}e^{\theta u_0(4K+8)}
 =\frac{e^{8\theta u_0}}{2(1-e^{4\theta u_0}/2)}.
\]
Substitution of $\theta_0$ gives $M_0$.
\end{proof}

\begin{proposition}[Physical clock of the long-probe experiment]
\label{prop:clock}
In the chronological experiment using $\nu$ on exploration and durations at
most $T$ otherwise, uniformly over the baseline policy and for every $x>0$,
\begin{equation}
 \Pp\left\{\mathsf T_n>
 n\left(\tau_g+T_0+T+\frac{\log M_0}{\theta_0}\right)
       +\frac{x}{\theta_0}\right\}\le e^{-x}.
 \label{eq:clock-tail}
\end{equation}
Also $\mathsf T_n\ge\tau_g n$. For any fixed infinite admissible policy,
$\limsup_n\mathsf T_n/n\le\tau_g+T_0+T+13u_0/2$ almost surely.
For the compact-duration law, $\mathsf T_n\le(\tau_g+T_0+T)n$ deterministically.
\end{proposition}
\begin{proof}
Enlarge the probability space by drawing an independent $\nu$-duration at
\emph{every} stage, including unused draws on nonexploration stages. Pathwise,
$\mathsf T_n\le n(\tau_g+T_0+T)+\sum_{i\le n}t_i$. The exponential-moment
bound and Markov's inequality prove \eqref{eq:clock-tail}, and the strong law
proves the fixed-policy assertion. This is not a deterministic upper bound on
individual long probes. The lower bound follows from the baseline gap.
\end{proof}
