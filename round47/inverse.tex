\section{A linear-depth response-jet inverse}
\label{sec:jet}
Put $\mu_m=\langle e_0,\Jop_\beta^m e_0\rangle$ and
$v_r=h_\beta^{(r+2)}(0)=\ell A_\beta^{r+1}B$. Then $\mu_0=v_0=1$ and
$c=-v_1$. Throughout this section $j=J+1$, $R=4J+8$, and
$L=Q^{25j}$.

\begin{lemma}[Closed response--moment inversion]
\label{lem:moments}
For $m\ge0$,
\begin{equation}
 \mu_m=(-1)^m\sum_{k=0}^m\binom mk c^{m-k}v_{m+k}.
 \label{eq:closed-moment}
\end{equation}
For two parameters and $M\ge1$, the maximum difference of damping and moments
through order $M$ is at most $Q^{4M}$ times the maximum difference of $v_r$
through order $2M$.
\end{lemma}
\begin{proof}
The identity $A^2+cA=-\operatorname{diag}(\Jop,\Jop)$ commutes with $A$.
Expand $(A^2+cA)^m A$, apply $\ell$ and $B$, and use
$\ell\operatorname{diag}(\Jop^m,\Jop^m)AB=\mu_m$. This proves
\eqref{eq:closed-moment}. If $e=\max_{r\le2M}|v_r-\widetilde v_r|$, then
$|c-\widetilde c|\le e$ and $|v_r|\le Q^{r+1}$.
Subtracting the finite sum, for $1\le m\le M$, gives
\[
 |\mu_m-\widetilde\mu_m|
 \le\{(1+c_+)^m+m2^{m-1}Q^{2m}\}e\le Q^{4m}e.
\]
Indeed $m2^{m-1}\le4^m\le Q^m$ and $Q^m+Q^{3m}\le Q^{4m}$.
No recursive amplification of previously estimated moments is necessary.
\end{proof}

Let $p_k$ be the monic orthogonal polynomial for the boundary spectral measure,
$\rho_k=\int p_k^2\,d\mu$, and $H_k=(\mu_{r+s})_{0\le r,s\le k}$.
The three-term recurrence, or Gram orthogonalization of
$1,x,x^2,\ldots$, gives
\begin{equation}
 \rho_k=\prod_{r<k}a_r^2,\quad
 p_k(x)=\det(xI-\Jop_{[0,k-1]}),\quad
 \|p_k\|_1\le(1+B_0)^k\le Q^k.
 \label{eq:poly-norm}
\end{equation}
The signs of off-diagonal entries do not change the positive recurrence
coefficients $a_k^2$. In particular $H_k$ is strictly positive definite.
Writing $p_k=x^k+\sum_{r<k}\alpha_r^{(k)}x^r$, one has
\[
 H_{k-1}\alpha^{(k)}=-(\mu_k,\ldots,\mu_{2k-1})^T,
 \quad t_k=\sum_{r,s\le k}p_{k,r}p_{k,s}\mu_{r+s+1},
 \quad b_k=t_k/\rho_k,\quad a_k=\sqrt{\rho_{k+1}/\rho_k}.
\]

\begin{lemma}[Orthogonal Gram factorization and secant bounds]
\label{lem:gram}
For padded coefficient vectors $p_0,\ldots,p_{k-1}$,
\begin{equation}
 H_{k-1}^{-1}=\sum_{r=0}^{k-1}\rho_r^{-1}p_rp_r^T,
 \qquad \|H_{k-1}^{-1}\|_\infty\le Q^{5k}\quad(k\ge1).
 \label{eq:gram-inverse}
\end{equation}
Moments through $2J+2$ determine the coefficients through $J$, with Lipschitz
constant at most $Q^{17(J+1)}$ between admissible moment vectors.
\end{lemma}
\begin{proof}
Stack the padded monic vectors as the rows of a unitriangular matrix $P$.
Then $PH_{k-1}P^T=\operatorname{diag}(\rho_0,\ldots,\rho_{k-1})$.
Inverting gives \eqref{eq:gram-inverse}; its row sums are bounded by
$\sum_{r<k}\|p_r\|_1^2/\rho_r\le\sum_{r<k}Q^{4r}\le kQ^{4(k-1)}\le Q^{5k}$.

For clarity every subsequent operation is bounded, not just the inverse
matrix. Write $S_k=Q^k$, $A_k=Q^{5k}$ and
\[
 V_k=kA_k(1+kS_k),\quad
 R_k'=S_k^2+2B_0^{2k}S_kV_k,\quad
 T_k'=S_k^2+2B_0^{2k+1}S_kV_k,\quad l_k=a_-^{2k}.
\]
For a maximum moment error $e$, subtraction of the two Gram systems and then
of their quadratic forms gives
\[
 \|p_k-\widetilde p_k\|_1\le V_ke,\quad
 |\rho_k-\widetilde\rho_k|\le R_k'e,\quad
 |t_k-\widetilde t_k|\le T_k'e.
\]
Here the matrix error contributes $S_k^2e$, and the two coefficient changes
contribute $2B_0^{2k}S_kV_ke$ or $2B_0^{2k+1}S_kV_ke$. Subtracting ratios,
using $\rho_k\ge l_k$, $t_k=b_k\rho_k$, and
$a_k+\widetilde a_k\ge2a_-$, gives respectively the constants
\[
 (T_k'+b_+R_k')/l_k,
 \qquad (R_{k+1}'+a_+^2R_k')/(2a_-l_k).
\]
For all $k\ge1$ these quantities obey
\[
 \begin{array}{c|ccccc}
 &V_k&R_k'&T_k'&(T_k'+b_+R_k')/l_k&
 (R_{k+1}'+a_+^2R_k')/(2a_-l_k)\\\hline
 \text{bound}&Q^{9k}&Q^{13k}&Q^{14k}&Q^{17k}&Q^{15(k+1)}.
 \end{array}
\]
For example the intermediate exponents for the first three entries are
$8k+1$, $12k+1$, and $12k+2$; the last two are at most $16k+1$ and
$15k+14$. These follow from $k\le Q^k$, $2\le Q$, and
$B_0,a_+,a_-^{-1}\le Q$. For the last entry retain cancellation of the
factor two in numerator and denominator. At $k=0$, $b_0=\mu_1$ has constant
one and the $a_0$ ratio has constant at most $Q^{14}/2$ because $\rho_0=1$
is fixed. Maximizing through $J$ proves the assertion, including $J=0$.
\end{proof}

\begin{theorem}[All-depth jet inverse]
\label{thm:jet}
For every $J\ge0$ and admissible $\beta,\gamma$,
\begin{equation}
 d_J(\beta,\gamma)\le Q^{25(J+1)}
 \max_{0\le r\le4J+8}|h_\beta^{(r)}(0)-h_\gamma^{(r)}(0)|.
 \label{eq:jet}
\end{equation}
\end{theorem}
\begin{proof}
Take $M=2J+2$ in Lemma~\ref{lem:moments} and compose with
Lemma~\ref{lem:gram}. The exponents add to $8(J+1)+17(J+1)$.
Derivatives only through $4J+6$ are necessary; the displayed two spare orders
keep the original diagnostic labeling. All identities involve finitely many
moments and are uniform in the unspecified infinite tail.
\end{proof}

Equality of the response on any time interval implies equality of all jets,
and hence all coefficients, by bounded-generator analyticity. Equivalently,
$\widehat h(z)=z^{-1}m_0(z^2+cz)$, where
$m_k(w)^{-1}=w+b_k-a_k^2m_{k+1}(w)$; this is the classical Weyl recursion
\cite{Teschl}. Positivity of every $a_k$ makes $e_0$ cyclic, so a fixed finite
Jacobi chain cannot reproduce the infinite-chain Weyl function.
