
\section{No-washout linear-time adaptive identification}
\label{sec:linear-time-protocol}

The logarithmic-washout experiment of
Section~\ref{sec:jacobi-statistics} gives a full two-sided posterior LDP.  We
now give a complementary protocol that removes the washout altogether.  Its
key device is a fresh Rademacher pulse drawn after the entire predictable
baseline action has been fixed.  The baseline may contain the accumulated
response to all earlier adaptive actions; conditional sign orthogonality makes
that predictable intercept contribute a nonnegative square to information.
The result retains the same guaranteed response exponent, up to the chosen
exploration fraction, in total physical time of order \(n\).

Put
\[
 g_\beta(t)=h_\beta'(t)=\ell e^{tA_\beta}B,
 \qquad t\ge0.
\]

\begin{lemma}[Compact \(L^1\) impulse-response geometry]
\label{lem:l1-impulse-geometry}
The map
\begin{equation}
 \mathcal G:\mathfrak B\longrightarrow L^1(0,\infty),
 \qquad \mathcal G(\beta)=g_\beta,
 \label{eq:l1-impulse-map}
\end{equation}
is continuous and has compact image.  Consequently, for every measurable
input \(u\) with \(\|u\|_\infty\le U_*\),
\begin{equation}
 \sup_{t\ge0}\left|
 \int_0^t\{g_\beta(t-s)-g_\gamma(t-s)\}u(s)\,ds
 \right|
 \le U_*\|g_\beta-g_\gamma\|_{L^1}.
 \label{eq:policy-uniform-convolution}
\end{equation}
The bound is independent of the number, order, and history dependence of the
input switches.
\end{lemma}

\begin{proof}
On every finite interval, product convergence of \(\beta_n\) to \(\beta\)
and the proof of Lemma~\ref{lem:product-continuity} give
\(g_{\beta_n}\to g_\beta\) uniformly.  Uniform exponential stability gives
\[
 |g_\beta(t)|\le C_*\|\ell\|\|B\|e^{-\lambda_*t}
 \quad\text{for every }\beta\in\mathfrak B.
\]
The common integrable envelope and dominated convergence prove continuity in
\(L^1(0,\infty)\).  Compactness follows from compactness of \(\mathfrak B\),
and \eqref{eq:policy-uniform-convolution} is Young's inequality.
\end{proof}

\subsection{Chronology and predictable-intercept information}

Fix constants \(\tau_g>0\), \(T_0<\infty\), \(U_0<\infty\), an exploration
fraction \(\rho\in(0,1]\), and the multiscale duration law \(\omega\) from
\eqref{eq:multiscale-design}.  Stage \(i\) has the following chronology.

\begin{enumerate}
\item From the observed past, a parameter-independent controller chooses a
baseline segment of duration \(r_i\in[\tau_g,\tau_g+T_0]\) and a baseline
force bounded by \(U_0\).  It may be reward-seeking, stabilizing, randomized,
and horizon dependent.
\item Draw an independent Bernoulli exploration flag \(E_i\) with parameter
\(\rho\).  If \(E_i=1\), draw \(M_i\) independently with law \(\omega\); if
\(E_i=0\), the controller may choose any duration index predictably.  In all
cases \(t_{M_i}\le T\).
\item After the baseline, flag, and duration are fixed, draw an independent
Rademacher sign \(S_i\).  During the final interval of length \(t_{M_i}\),
add the constant pulse \(aS_i\) to the bounded baseline force.
\item Only after the pulse is complete, draw the independent
\(N(0,\sigma^2)\) readout noise and observe the boundary coordinate.
\end{enumerate}
Thus, with \(\mathcal F_{i-1}\) denoting the information before stage \(i\),
\begin{equation}
 \mathcal F_{i-1}\longrightarrow(r_i,u_i^0,E_i,M_i)
 \longrightarrow S_i\longrightarrow\xi_i\longrightarrow Y_i.
 \label{eq:linear-time-chronology}
\end{equation}
There is no zero-force washout and the physical state is never reset.

Let \(m_i^\beta\) be the zero-initial-state endpoint mean under the complete
realized input record.  Linearity and the placement of the current signed
pulse give, for every candidate \(\beta\),
\begin{equation}
 m_i^\beta-m_i^{\beta_0}
 =P_i(\beta)+aS_i
 \{h_\beta(t_{M_i})-h_{\beta_0}(t_{M_i})\},
 \label{eq:predictable-intercept-decomposition}
\end{equation}
where \(P_i(\beta)\) is measurable before \(S_i\) is drawn.  It includes the
entire difference generated by past signed pulses and by the current
baseline.  Lemma~\ref{lem:l1-impulse-geometry} gives, with
\(U_*=U_0+a\),
\begin{equation}
 |P_i(\beta)-P_i(\gamma)|
 \le U_*\|g_\beta-g_\gamma\|_{L^1},
 \qquad
 |P_i(\beta)|\le U_*\|g_\beta-g_{\beta_0}\|_{L^1}.
 \label{eq:predictable-intercept-continuity}
\end{equation}
These bounds hold simultaneously for all stages and all admissible adaptive
controllers.

\begin{lemma}[Predictable-intercept Rademacher information]
\label{lem:predictable-intercept-information}
For every \(\beta\in\mathfrak B\),
\begin{align}
 &\mathbb E\left[
 (m_i^\beta-m_i^{\beta_0})^2
 \mid\mathcal F_{i-1},r_i,u_i^0,E_i,M_i\right]\notag\\
 &\hspace{18mm}=P_i(\beta)^2
 +a^2\{h_\beta(t_{M_i})-h_{\beta_0}(t_{M_i})\}^2,
 \label{eq:predictable-intercept-square}
\end{align}
and therefore
\begin{equation}
 \mathbb E\left[(m_i^\beta-m_i^{\beta_0})^2
 \mid\mathcal F_{i-1}\right]
 \ge \rho a^2D(\beta,\beta_0).
 \label{eq:linear-time-information-floor}
\end{equation}
The same assertions hold for directional derivatives at \(\beta_0\) along
any fixed finite coefficient cylinder.
\end{lemma}

\begin{proof}
Condition on every variable preceding \(S_i\) in
\eqref{eq:linear-time-chronology}.  The cross term in the square of
\eqref{eq:predictable-intercept-decomposition} has conditional expectation
zero.  This gives \eqref{eq:predictable-intercept-square}.  On exploration
stages, averaging the second term over \(M_i\) gives \(a^2D\); every
nonexploration and predictable-intercept contribution is nonnegative.  The
derivative statement follows by differentiating the same linear
decomposition.
\end{proof}

\subsection{Posterior exponent in linear physical time}

Let \(T_i^{\rm phys}\) be the global endpoint time of stage \(i\).  The stage
lengths obey
\begin{equation}
 \tau_g n\le T_n^{\rm phys}
 \le \tau_+n,
 \qquad \tau_+=\tau_g+T_0+T.
 \label{eq:linear-physical-time}
\end{equation}
For a working initial-state prior \(\nu_n\) supported in the radius-\(R\)
Hilbert ball, let \(L_{n,{\rm lin}}^{\nu_n}\) and
\(\Pi_{n,{\rm lin}}^{\nu_n}\) denote the exact integrated likelihood and
posterior under the protocol above.

\begin{theorem}[No-washout adaptive Jacobi posterior bound]
\label{thm:linear-time-adaptive-jacobi}
For every admissible bounded adaptive baseline, every sequence of working
initial-state priors supported in the fixed Hilbert ball, and every closed
\(F\subset\mathfrak B\), on one event of
\(P_{\beta_0,z_0}\)-probability one,
\begin{equation}
 \limsup_{n\to\infty}\frac1n
 \log\Pi_{n,{\rm lin}}^{\nu_n}(F)
 \le-\frac{\rho a^2}{2\sigma^2}
       \inf_{\beta\in F}D(\beta,\beta_0).
 \label{eq:linear-posterior-upper}
\end{equation}
Consequently, whenever the separated set is nonempty,
\begin{equation}
 \limsup_{n\to\infty}\frac1n
 \log\Pi_{n,{\rm lin}}^{\nu_n}
 \{d_J(\beta,\beta_0)\ge\delta\}
 \le-\frac{\rho a^2}{2\sigma^2}
       \underline\kappa_J(\delta)<0.
 \label{eq:linear-coefficient-upper}
\end{equation}
In elapsed physical time the uniform guaranteed exponent is
\begin{equation}
 \limsup_{n\to\infty}\frac1{T_n^{\rm phys}}
 \log\Pi_{n,{\rm lin}}^{\nu_n}(F)
 \le-\frac{\rho a^2}{2\sigma^2\tau_+}
       \inf_{\beta\in F}D(\beta,\beta_0).
 \label{eq:linear-elapsed-rate}
\end{equation}
Thus arbitrary bounded adaptive baselines coexist with an explicit
information rate and \(\Theta(n)\) total physical time.
\end{theorem}

\begin{proof}
Put
\(\Delta_i(\beta)=m_i^\beta-m_i^{\beta_0}\).  By
\eqref{eq:predictable-intercept-continuity}, compactness of the two response
images in \(L^1(0,\infty)\) and \(C([0,T])\) supplies finite deterministic
nets for the class \(\{\Delta_i(\beta):\beta\in\mathfrak B\}\), uniformly in
\(i\) and in the policy.  At each net point, the centered quadratic terms
are bounded martingale differences.  The Gaussian scores
\(\xi_i\Delta_i(\beta)\) are conditionally sub-Gaussian martingale
differences.  Azuma--Hoeffding and Gaussian exponential bounds, followed by
finite-net interpolation, give on one event
\begin{align}
 \sup_{\beta\in\mathfrak B}\left|
 \frac1n\sum_{i=1}^n\left\{
 \Delta_i(\beta)^2-
 \mathbb E[\Delta_i(\beta)^2\mid\mathcal F_{i-1}]
 \right\}\right|&\longrightarrow0,
 \label{eq:linear-uniform-quadratic-slln}\\
 \sup_{\beta\in\mathfrak B}\left|
 \frac1n\sum_{i=1}^n\xi_i\Delta_i(\beta)
 \right|&\longrightarrow0.
 \label{eq:linear-uniform-score-slln}
\end{align}
The interpolation error is controlled by the empirical first moments of the
bounded intercepts and of \(|\xi_i|\), exactly as in
Lemma~\ref{lem:supnorm-response-slln}.

For the zero-initial-state Gaussian likelihood, the usual square expansion,
Lemma~\ref{lem:predictable-intercept-information}, and
\eqref{eq:linear-uniform-quadratic-slln}--
\eqref{eq:linear-uniform-score-slln} imply
\begin{equation}
 \limsup_n\sup_{\beta\in\mathfrak B}
 \left\{\frac1n\log
 \frac{L_{n,{\rm lin}}^\star(\beta)}
      {L_{n,{\rm lin}}^\star(\beta_0)}
 +\frac{\rho a^2}{2\sigma^2}D(\beta,\beta_0)\right\}\le0.
 \label{eq:linear-uniform-likelihood-upper}
\end{equation}
The global initial-state term at stage \(i\) is
\(\ell e^{T_i^{\rm phys}A_\beta}z\).  Repeated Duhamel differentiation and
\eqref{eq:linear-physical-time} give, for every fixed cylinder derivative,
\[
 \sup_{\beta,\|z\|\le R}
 |\partial^\alpha\ell e^{T_i^{\rm phys}A_\beta}z|
 \le C_\alpha(1+i)^{|\alpha|}e^{-\lambda\tau_gi}.
\]
The right side is summable.  The Gaussian completion used in
Lemma~\ref{lem:washout-correction} therefore changes the exact integrated
log likelihood by only \(O(1)\), uniformly in \(\beta\), the policy, and
\(\nu_n\).

It remains to normalize.  For every \(\varepsilon>0\), continuity in
\eqref{eq:predictable-intercept-continuity} provides an open neighbourhood
\(V_\varepsilon\) of \(\beta_0\) such that
\(\sup_i\sup_{\beta\in V_\varepsilon}|\Delta_i(\beta)|\le\varepsilon\)
for every admissible policy.  The square expansion and the strong law for
\(n^{-1}\sum|\xi_i|\) give
\[
 \liminf_n\frac1n\log
 \int_{V_\varepsilon}
 \frac{L_{n,{\rm lin}}^{\nu_n}(\beta)}
      {L_{n,{\rm lin}}^{\nu_n}(\beta_0)}\Pi(d\beta)
 \ge-C(\varepsilon+\varepsilon^2).
\]
Full support gives \(\Pi(V_\varepsilon)>0\); letting
\(\varepsilon\downarrow0\) shows that the posterior denominator has
exponential rate zero.  Integrating
\eqref{eq:linear-uniform-likelihood-upper} over a closed set proves
\eqref{eq:linear-posterior-upper}.  The coefficient statement follows from
Theorem~\ref{thm:effective-jacobi-stability}, and
\eqref{eq:linear-elapsed-rate} follows from
\(T_n^{\rm phys}/n\le\tau_+\).
\end{proof}

\begin{remark}[Why the result is complementary to the full LDP]
Theorem~\ref{thm:jacobi-rate} remains the exact matching upper-and-lower LDP
for the logarithmic-washout protocol.  Under an arbitrary adaptive baseline,
its predictable coefficient-dependent response can add information, so the
linear-time posterior may contract faster than the guaranteed response rate.
Theorem~\ref{thm:linear-time-adaptive-jacobi} keeps the sharp universal lower
floor on information without discarding that additional signal.  In
particular, removing washout does not weaken any theorem already proved in
this article.
\end{remark}

\subsection{Washout-free honest response and coefficient cylinders}

Let \(\mathcal Q\) be a finite collection of multiscale cells.  Count only
exploration visits and put
\[
 N_q(n)=\#\{i\le n:E_i=1,\ M_i=q\},
 \qquad
 \widehat h_q^{\rm lin}
 =\frac1{aN_q(n)}
   \sum_{i\le n:E_i=1,\,M_i=q}S_iY_i
\]
when \(N_q(n)>0\).  Uniform stability and the global input bound give the
deterministic predictable-intercept envelope
\begin{equation}
 B_* = C_*\|\ell\|\left\{R+
       \frac{(U_0+a)\|B\|}{\lambda_*}\right\}.
 \label{eq:linear-intercept-envelope}
\end{equation}
Define
\begin{equation}
 R_{q,n}^{\rm lin}(\alpha)=
 \frac{\sqrt{2(\sigma^2+B_*^2)
 \log(2n|\mathcal Q|/\alpha)}}{a\sqrt{N_q(n)}},
 \label{eq:linear-confidence-radius}
\end{equation}
with value \(+\infty\) when \(N_q(n)=0\).

\begin{theorem}[Honest no-washout coefficient cylinders]
\label{thm:linear-time-honest-cylinders}
For every \(0<\alpha<1\),
\begin{equation}
 \mathcal C_n^{\rm lin}(\alpha)=
 \left\{\beta\in\mathfrak B:
 |h_\beta(t_q)-\widehat h_q^{\rm lin}|
 \le R_{q,n}^{\rm lin}(\alpha)
 \text{ for all }q\in\mathcal Q\right\}
 \label{eq:linear-confidence-set}
\end{equation}
has frequentist coverage at least \(1-\alpha\), uniformly over the full
coefficient box, all admissible bounded adaptive baselines, all true initial
states in the radius-\(R\) ball, and all sample sizes.  If
\(\mathcal Q\) contains the \(R_J+1\) grid cells at scale
\(\varepsilon_{J,\delta}\) and
\begin{equation}
 2\max_{q\in\mathcal Q}R_{q,n}^{\rm lin}(\alpha)
 <\frac{\delta\varepsilon_{J,\delta}^{R_J}}
        {2C_{R_J}L_J},
 \label{eq:linear-cylinder-threshold}
\end{equation}
then
\(\operatorname{diam}_{d_J}\mathcal C_n^{\rm lin}(\alpha)\le\delta\).
The assertion remains valid for triangular finite cell sets
\(\mathcal Q_n\) after replacing \(|\mathcal Q|\) by
\(|\mathcal Q_n|\) in \eqref{eq:linear-confidence-radius}.
\end{theorem}

\begin{proof}
At an exploration visit to cell \(q\), write
\[
 Y_i=aS_ih_{\beta_0}(t_q)+B_i^0+\xi_i,
 \]
where \(B_i^0\) contains the true initial-state response and every input
contribution except the current signed pulse.  It is measurable before
\(S_i\) and satisfies \(|B_i^0|\le B_*\).  Therefore, conditionally on the
past and the selected cell,
\[
 \mathbb E\exp\{\lambda S_i(B_i^0+\xi_i)\}
 =e^{\sigma^2\lambda^2/2}\cosh(\lambda B_i^0)
 \le\exp\{(\sigma^2+B_*^2)\lambda^2/2\}.
\]
For each cell and each possible visit count \(1\le m\le n\), the selected
sum is thus sub-Gaussian with variance proxy
\(m(\sigma^2+B_*^2)\).  A two-sided tail bound and a union bound over cells
and visit counts prove simultaneous coverage.  For any two elements of the
confidence set, their response difference on the grid is at most twice the
radius.  Lemma~\ref{lem:finite-grid-certificate}, followed by
Theorem~\ref{thm:effective-response-jet-audit}, gives the diameter statement.
\end{proof}

\begin{corollary}[Linear-time shrinking honest blocks]
\label{cor:linear-time-shrinking-cylinders}
For every fixed \(J\) and \(\delta>0\), choose the finite grid required in
Theorem~\ref{thm:linear-time-honest-cylinders}.  Since
\(N_q(n)/n\to\rho\omega_q\) almost surely, the diameter is eventually at
most \(\delta\) with probability at least \(1-\alpha\), while total physical
time remains bounded by \(\tau_+n\).  More generally, any triangular
\((J_n,\delta_n,\mathcal Q_n)\) for which
\eqref{eq:linear-cylinder-threshold} holds eventually gives honest
simultaneous uncertainty for an increasing coefficient block without a
washout interval.
\end{corollary}
