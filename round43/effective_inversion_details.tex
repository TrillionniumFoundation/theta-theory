
\section{Algorithmic moment--Gram inversion and explicit shrinking radii}
\label{sec:effective-inversion-details}

The proof of Lemma~\ref{lem:effective-response-jet} can be made completely
algorithmic.  This section records the exact triangular identities and the
finite linear algebra used by that lemma.  In particular, no coefficient of
the inverse map is selected by compactness, and no unnamed continuity modulus
enters the growing-depth conclusion.

For \(\beta\in\mathfrak B\), put
\[
 \mu_k(\beta)=\langle e_0,J_\beta^ke_0\rangle,
 \qquad d_r(\beta)=h_\beta^{(r+2)}(0).
\]
Thus \(\mu_0=d_0=1\) and \(c=-d_1\).

\begin{lemma}[Exact response--moment triangularity]
\label{lem:response-moment-triangularity}
In a neighbourhood of the origin one has the convergent generating identity
\begin{equation}
 \sum_{r=0}^{\infty}d_r(\beta)x^r
 =\sum_{k=0}^{\infty}(-1)^k\mu_k(\beta)x^{2k}
       (1+cx)^{-k-1}.
 \label{eq:response-moment-generating-identity}
\end{equation}
Consequently, for every \(m\ge1\),
\begin{equation}
 \mu_m=(-1)^m\left[
 d_{2m}-\sum_{k=0}^{m-1}(-1)^k
 \binom{2m-k}{2m-2k}c^{2m-2k}\mu_k
 \right].
 \label{eq:explicit-moment-recursion}
\end{equation}
Hence \(c,\mu_0,\ldots,\mu_M\) are recovered by a finite triangular
calculation from derivatives through order \(2M+2\).
\end{lemma}

\begin{proof}
The large-\(z\) expansion of \eqref{eq:jacobi-transfer}, with \(x=z^{-1}\),
gives
\[
 \sum_{n=0}^{\infty}h_\beta^{(n)}(0)x^n
 =x^2\langle e_0,(I+cx+x^2J_\beta)^{-1}e_0\rangle.
\]
Factor the inverse as
\[
 (I+cx+x^2J_\beta)^{-1}
 =(1+cx)^{-1}\sum_{k=0}^{\infty}
 \left\{-\frac{x^2J_\beta}{1+cx}\right\}^k.
\]
The coefficient box makes this series norm convergent for all sufficiently
small \(x\), uniformly in \(\beta\).  Applying the boundary vector proves
\eqref{eq:response-moment-generating-identity}.  Since
\[
 (1+cx)^{-k-1}=\sum_{r=0}^{\infty}
 (-1)^r\binom{k+r}{r}c^rx^r,
\]
the coefficient of \(x^{2m}\) is
\[
 d_{2m}=\sum_{k=0}^{m}(-1)^k
 \binom{2m-k}{2m-2k}c^{2m-2k}\mu_k.
\]
The last summand is \((-1)^m\mu_m\), which gives
\eqref{eq:explicit-moment-recursion}.
\end{proof}

\begin{proposition}[Quantified triangular moment map]
\label{prop:quantified-moment-map}
Let \(M\ge0\), and let \(\beta,\gamma\in\mathfrak B\).  With the constant
\(Q\) of Section~\ref{sec:quantitative-jacobi},
\begin{equation}
 \max\left\{|c-\widetilde c|,
       \max_{0\le k\le M}|\mu_k(\beta)-\mu_k(\gamma)|\right\}
 \le Q^{4(M+1)^2}
       \max_{0\le r\le2M}|d_r(\beta)-d_r(\gamma)|.
 \label{eq:quantified-moment-map}
\end{equation}
The map on the left is therefore an explicitly given polynomial-triangular
map on the declared derivative box.
\end{proposition}

\begin{proof}
Use \eqref{eq:explicit-moment-recursion} inductively.  At level \(m\), there
are at most \(m\) previous moments, every binomial coefficient is at most
\(2^{2m}\), every moment is at most \(B_J^m\), and every power of \(c\) is
bounded by \(\Lambda_A^{2m}\).  The difference of a power satisfies
\(|c^r-\widetilde c^r|\le r\max(c_+,1)^{r-1}|c-\widetilde c|\).
Since \(Q\ge8\max\{2,\Lambda_A\}\), induction gives the displayed exponent;
the increase from level \(m-1\) to \(m\) is bounded by \(Q^{8m+4}\), and
\(\sum_{m=0}^{M}(8m+4)\le4(M+1)^2\).  Every operation in this induction is
the addition or multiplication displayed in
\eqref{eq:explicit-moment-recursion}.
\end{proof}

We next spell out the moment-to-coefficient step.  For \(j\ge0\), put
\[
 H_j=(\mu_{r+s})_{r,s=0}^{j},\qquad
 \Delta_j=\det H_j,\qquad \Delta_{-1}=1.
\]
For \(j\ge1\), let \(\alpha^{(j)}\in\mathbb R^j\) be the unique solution of
\begin{equation}
 H_{j-1}\alpha^{(j)}
 =-(\mu_j,\mu_{j+1},\ldots,\mu_{2j-1})^T,
 \label{eq:gram-linear-system}
\end{equation}
and define
\[
 p_0(x)=1,\qquad
 p_j(x)=x^j+\sum_{r=0}^{j-1}\alpha_r^{(j)}x^r.
\]
Write
\begin{equation}
 \rho_j=\sum_{r,s=0}^{j}p_{j,r}p_{j,s}\mu_{r+s}
       =\frac{\Delta_j}{\Delta_{j-1}},
 \label{eq:monic-norm-determinant}
\end{equation}
where \(p_{j,j}=1\).

\begin{proposition}[Finite Gram reconstruction with an explicit condition
number]
\label{prop:effective-gram-reconstruction}
The coefficients through depth \(J\) are obtained from
\(\mu_0,\ldots,\mu_{2J+2}\) by
\begin{align}
 b_j&=\rho_j^{-1}
 \sum_{r,s=0}^{j}p_{j,r}p_{j,s}\mu_{r+s+1},
 &&0\le j\le J,
 \label{eq:gram-recover-b}\\
 a_j^2&=\rho_{j+1}/\rho_j,
 &&0\le j\le J.
 \label{eq:gram-recover-a}
\end{align}
Moreover, for two admissible moment vectors,
\begin{equation}
 \max_{0\le j\le J}
 \left\{|a_j-\widetilde a_j|,|b_j-\widetilde b_j|\right\}
 \le Q^{8(J+1)^3}
       \max_{0\le k\le2J+2}|\mu_k-\widetilde\mu_k|.
 \label{eq:effective-gram-lipschitz}
\end{equation}
Thus the inverse is a finite list of linear solves, quadratic forms, and
positive square roots with a displayed uniform condition bound.
\end{proposition}

\begin{proof}
Equation~\eqref{eq:gram-linear-system} is the orthogonality condition for the
monic polynomial \(p_j\).  The monic three-term recurrence is
\[
 xp_j=p_{j+1}+b_jp_j+a_{j-1}^2p_{j-1},
\]
so taking inner products with \(p_j\), and then comparing monic norms, gives
\eqref{eq:gram-recover-b}--\eqref{eq:gram-recover-a}.

For completeness, every denominator is bounded directly.  The Jacobi
factorization gives
\begin{equation}
 \Delta_j=\prod_{r=0}^{j-1}a_r^{2(j-r)}
 \ge a_-^{j(j+1)},
 \qquad
 a_-^{2j}\le\rho_j\le a_+^{2j}.
 \label{eq:gram-denominator-bounds}
\end{equation}
All entries of \(H_j\) are bounded by \(B_J^{2j}\).  Cramer's rule,
Hadamard's inequality, and \eqref{eq:gram-denominator-bounds} therefore give
\[
 \|H_{j-1}^{-1}\|_\infty
 \le j\,j!\,B_J^{2j(j-1)}a_-^{-j(j-1)}
 \le Q^{3(j+1)^2}.
\]
The resolvent identity for two inverses gives the same bound, squared, for
their difference.  Substitution in
\eqref{eq:gram-linear-system}, followed by the quadratic forms in
\eqref{eq:monic-norm-determinant} and
\eqref{eq:gram-recover-b}, costs at most \(Q^{7(j+1)^2}\) at level \(j\).
The positive square root in \eqref{eq:gram-recover-a} is
\(a_-^{-1}\)-Lipschitz on the relevant squared interval.  Multiplying the
level bounds, or equivalently summing their logarithms, gives
\[
 Q^{\sum_{j=0}^{J}7(j+1)^2+J+1}
 \le Q^{8(J+1)^3},
\]
which proves \eqref{eq:effective-gram-lipschitz}.  This calculation also
specifies an executable reconstruction algorithm at every finite depth.
\end{proof}

\begin{theorem}[Line-addressable effective response-jet inversion]
\label{thm:effective-response-jet-audit}
For every \(J\ge0\) and \(\beta,\gamma\in\mathfrak B\),
\begin{equation}
 d_J(\beta,\gamma)
 \le Q^{12(J+1)^3}
 \max_{0\le r\le4J+8}
 |h_\beta^{(r)}(0)-h_\gamma^{(r)}(0)|.
 \label{eq:audited-effective-jet-inverse}
\end{equation}
In particular, the constant \(L_J\) in
\eqref{eq:effective-jet-inverse} is produced by the explicit recursions
\eqref{eq:explicit-moment-recursion},
\eqref{eq:gram-linear-system},
\eqref{eq:gram-recover-b}, and \eqref{eq:gram-recover-a}.
\end{theorem}

\begin{proof}
Take \(M=2J+2\) in Proposition~\ref{prop:quantified-moment-map} and compose
with Proposition~\ref{prop:effective-gram-reconstruction}.  Derivatives
through order \(4J+6\) already suffice, and the stated maximum through
\(4J+8\) only enlarges the right side.  Since
\(4(2J+3)^2+8(J+1)^3\le12(J+1)^3\) after enlarging the harmless cases
\(J=0,1\) by the standing factor \(Q\ge16\), the claimed bound follows.
\end{proof}

The preceding estimates also turn the qualitative shrinking-separation
statement into a concrete sequence.

\begin{theorem}[Explicit shrinking radius at increasing depth]
\label{thm:explicit-shrinking-block-rate}
Assume the product-prior hypotheses of
Theorem~\ref{thm:growing-depth-recovery}.  Fix \(0<\eta<1\), and let
\(J_n\uparrow\infty\) satisfy
\begin{equation}
 (J_n+1)^5=o(\log n).
 \label{eq:explicit-depth-regime}
\end{equation}
Set
\begin{equation}
 \delta_n=\exp\left\{-\frac{\eta\log n}{80(J_n+1)}\right\},
 \qquad
 r_{n,\tau}=3\delta_n+C_{\rm box}e^{-2\tau(J_n+1)}.
 \label{eq:explicit-shrinking-radius}
\end{equation}
Then \(\delta_n\downarrow0\), and
\begin{align}
 \Pi_n^{\nu_n}\{d_{J_n}(\beta,\beta_0)\ge\delta_n\}
 &\longrightarrow0,
 \label{eq:explicit-block-contraction}\\
 \Pi_n^{\nu_n}\{\|W_\tau(J_\beta-J_{\beta_0})W_\tau\|_{\rm op}
                 >r_{n,\tau}\}
 &\longrightarrow0
 \label{eq:explicit-weighted-contraction}
\end{align}
in \(P_{\beta_0,z_0}\)-probability.  The same conclusion holds under the
adaptive exploration policy of
Theorem~\ref{thm:adaptive-exploration-floor} whenever
\(\inf_n\rho_n>0\).
\end{theorem}

\begin{proof}
Let \(j_n=J_n+1\) and
\(k_n=\underline\kappa_{J_n}(\delta_n)\).  By
\eqref{eq:kappa-asymptotic-lower},
\[
 k_n\ge \exp\{-A_0j_n^4\}\delta_n^{20j_n}
      =\exp\{-A_0j_n^4\}n^{-\eta/4}
      =n^{-\eta/4-o(1)}.
\]
Condition~\eqref{eq:explicit-depth-regime} also implies
\(\log n/j_n\to\infty\), so \(\delta_n\to0\).  The entropy and prior
bounds in \eqref{eq:response-entropy-bound} give
\[
 \mathcal H(k_n/16)+\mathcal P(\sqrt{k_n}/8)
 =O\!\left(\frac{(\log n)^2}{\log\log n}\right).
\]
On the other hand,
\[
 nk_n^2\ge n^{1-\eta/2-o(1)},
 \qquad nk_n\ge n^{1-\eta/4-o(1)}.
\]
Both conditions in \eqref{eq:growing-depth-conditions} follow, and
Theorem~\ref{thm:growing-depth-recovery} proves
\eqref{eq:explicit-block-contraction}.  The deterministic bound
\eqref{eq:weighted-operator-bound} gives
\eqref{eq:explicit-weighted-contraction}.  A uniform positive exploration
floor only multiplies \(k_n\) by a fixed positive constant, so the same
estimates verify \eqref{eq:adaptive-growing-depth-conditions}.
\end{proof}
