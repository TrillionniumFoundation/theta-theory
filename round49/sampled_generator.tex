\section{A sampled-generator inverse for delayed short pulses}
\label{sec:sampled}
Choose an atom $q_0=(0,s_0,1)$ of \eqref{eq:omega}, where
\begin{equation}
 s_0=\min\{s\ge1:2\zeta_8 2^{-s}\le(64\Lambda)^{-1}\},\quad
 t_0=\zeta_8 2^{-s_0},\quad\Delta=2t_0,\quad w_0=\omega_{q_0}=2^{-s_0-5}.
 \label{eq:fast-atom}
\end{equation}
All these are determined by the original box and law. Set
$U_\beta=e^{\Delta A_\beta}$, $X_\beta=U_\beta-I$, and
$H_\beta=\int_0^{t_0}e^{uA_\beta}\,du$. Define delayed short-pulse responses
\begin{equation}
 b_{k,\beta}=\ell e^{k\Delta A_\beta}H_\beta B
      =h_\beta(k\Delta+t_0)-h_\beta(k\Delta),\qquad k=0,1,\ldots.
 \label{eq:pulse-responses}
\end{equation}

\begin{lemma}[Finite sampled-logarithm certificate]
\label{lem:sampled-log}
For any $N\ge R=4J+8$, put
\begin{equation}
 \mathcal A_{N,R}=8\Delta^{-R}4^N,\qquad
 \mathcal E_{N,R}=16\Delta^{-R}8^{-N}.
 \label{eq:sampled-constants}
\end{equation}
Then all admissible pairs satisfy
\begin{equation}
 \max_{0\le r\le R}|h_\beta^{(r)}(0)-h_\gamma^{(r)}(0)|
 \le\mathcal A_{N,R}\max_{0\le k\le N}|b_{k,\beta}-b_{k,\gamma}|
                +\mathcal E_{N,R}.
 \label{eq:sampled-jet}
\end{equation}
The polynomial transform coefficients depend only on $(\Delta,r,N)$, not on
the unknown parameter, and are rational when $\Delta$ is rational.
\end{lemma}
\begin{proof}
Since $\Delta\Lambda\le1/64$,
$\|X_\beta\|\le e^{1/64}-1\le1/32$. The norm-convergent logarithm satisfies
$\log(I+X_\beta)=\Delta A_\beta$: substitution into the absolutely convergent
scalar power-series identity proves the identity in this Banach algebra.
The principal square-root series similarly gives
$(I+X_\beta)^{1/2}=e^{\Delta A_\beta/2}$. No choice among distant logarithm
branches is being made.

For $r\ge1$, define the analytic function on $|x|<1$
\begin{equation}
 F_r(x)=\Delta^{-r}[\log(1+x)]^{r-1}
              \frac{\log(1+x)}{x}(\sqrt{1+x}+1).
 \label{eq:sampled-function}
\end{equation}
The quotient at zero is one. The elementary identity
$(\sqrt{1+x}-1)(\sqrt{1+x}+1)=x$ shows
$F_r(X_\beta)H_\beta=A_\beta^{r-1}$, by analytic continuation also at zero.
Thus $h_\beta^{(r)}(0)=\ell F_r(X_\beta)H_\beta B$.
On $|x|=1/2$, $|\log(1+x)|\le\log2<1$,
$|\log(1+x)/x|\le2$, and $|\sqrt{1+x}+1|<3$.
Writing $F_r(x)=\sum_{k\ge0}c_{rk}x^k$, Cauchy's coefficient bound gives
$|c_{rk}|\le6\Delta^{-R}2^k$ for $1\le r\le R$, since $\Delta\le1$.
Also $\|H_\beta\|\le t_0e^{t_0\Lambda}\le\Delta$.
The difference of the two series tails after $N$ is at most
\[
 12\Delta^{1-R}\frac{16^{-(N+1)}}{1-1/16}
       \le16\Delta^{-R}8^{-N}.
\]
For the finite part, $(U-I)^k=\sum_{l=0}^k\binom kl(-1)^{k-l}U^l$.
Its coefficient row sum in the responses $b_0,\ldots,b_N$ is at most
$6\Delta^{-R}\sum_{k=0}^N4^k\le8\Delta^{-R}4^N$.
This proves \eqref{eq:sampled-jet}; $h(0)=0$ is fixed.
Logarithm and square-root series have rational coefficients, proving the last
assertion without numerical differentiation of data.
\end{proof}

For $j=J+1$, $L=Q^{25j}$, $d_0=\log\Delta^{-1}$ and $v=\log(1/\delta)$ choose
\begin{equation}
 N=\max\left\{R,\left\lceil
   \frac{\log32+25j\log Q+Rd_0+v}{\log8}\right\rceil\right\},\quad
 m=N+1,
 \quad \kappa_b=(\rho w_0)^m\frac{\delta^2}{4L^2\mathcal A_{N,R}^2}.
 \label{eq:block-certificate}
\end{equation}
The remainder obeys $L\mathcal E_{N,R}\le\delta/2$. Hence any separated pair
has a lag $0\le k\le N$ with
$|b_{k,\beta}-b_{k,\gamma}|\ge\delta/(2L\mathcal A_{N,R})$.
Define fixed constants
\begin{equation}
 \begin{split}
 K_b&=12+\frac{\log32+25\log Q+8d_0}{\log8},\qquad
 a_b=\log(1/(\rho w_0))+\log16,\\
 C_b&=\log256+50\log Q+16d_0+K_b a_b,\qquad
 p_b=2+a_b/\log8.
 \end{split}
 \label{eq:block-constants}
\end{equation}
Here $p_b$ is an accuracy exponent, not a sampling or confidence weight.

\begin{proposition}[Exponential resource certificate for the unchanged law]
\label{prop:block-certificate}
For all $J\ge0$ and $0<\delta\le1$,
\begin{equation}
 m\le K_bj+v/\log8,\qquad \kappa_b\ge e^{-C_bj}\delta^{p_b}.
 \label{eq:block-separation}
\end{equation}
\end{proposition}
\begin{proof}
Use $R\le8j$, the ceiling error at most one, and $j\ge1$ to obtain the first
inequality from \eqref{eq:block-certificate}. Moreover
\[
 -\log\kappa_b=m\log(1/(\rho w_0))+2v+\log256
                     +50j\log Q+2Rd_0+2N\log4
 \le C_bj+p_bv.
\]
The estimate includes the probability of the entire successful run, not just
one atom. It does not assign extra mass to the selected labels.
\end{proof}
