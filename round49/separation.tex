\section{The two earlier response laws and their quantitative separation}
\label{sec:separation}
The original compact-duration law $\omega$ puts weights $2^{-m-1}$ on a dense
rational enumeration of $(0,T]$, $m\ge1$, and assigns its other half of mass to
\begin{equation}
 R_K=4K+8,\quad q=(K,s,k),\quad
 t_q=k\zeta_{R_K}2^{-s},\quad \omega_q=2^{-K-s-2}/R_K,
 \quad\zeta_R=\min(1,T/(2R)),
 \label{eq:omega}
\end{equation}
where $K\ge0$, $s\ge1$, $1\le k\le R_K$. Coincident times retain separate
labels. There is no noisy zero-time cell. For a law $\chi$ write
$D_\chi(\beta,\gamma)=\sum_q\chi_q[h_\beta(t_q)-h_\gamma(t_q)]^2$.

For reference, the preceding finite-mean law is
\begin{equation}
 u_0=(256\Lambda)^{-1},\quad N_K=4K+8,\quad
 t_{K,k}=ku_0,\quad \nu_{K,k}=2^{-K-1}/N_K.
 \label{eq:nu}
\end{equation}
Its support, unlike that of $\omega$, is unbounded. Its mean is $13u_0/2$.
With $\theta_0=\log2/(8u_0)$,
$\E_\nu e^{\theta_0t}\le M_0=(1-1/\sqrt2)^{-1}$: dominate each duration by
$u_0(4K+8)$ and sum the geometric series. This distinction is unchanged.

\begin{lemma}[Oversampled response jets]
\label{lem:grid}
Let $N\ge R$, $0<u\le1$ and $f=h_\beta-h_\gamma$. Put
\[
 A_{N,R,u}=8^Nu^{-R},\qquad
 E_{N,u}=2e^{\Lambda Nu}(\Lambda Nu)^{N+1}/(N+1)!.
\]
Then
\begin{equation}
 \max_{r\le R}|f^{(r)}(0)|\le
 A_{N,R,u}\left(\max_{1\le k\le N}|f(ku)|+E_{N,u}\right).
 \label{eq:grid}
\end{equation}
\end{lemma}
\begin{proof}
For $V_N=(k^r/r!)_{k,r=0}^N$, Lagrange interpolation gives
$(V_N^{-1})_{rk}=r![x^r]\prod_{l\ne k}(x-l)/\prod_{l\ne k}(k-l)$.
The absolute numerator coefficient, multiplied by $r!\le N^r$, is at most
the absolute-coefficient polynomial evaluated at $N$. Its value is bounded
by $(2N)!/N!$; division by $k!(N-k)!$ and summation give
$\|V_N^{-1}\|_\infty\le2^N\binom{2N}{N}\le8^N$.
The bound $|f^{(N+1)}(t)|\le2e^{\Lambda t}\Lambda^{N+1}$ controls the
Taylor remainder at all nodes by $E_{N,u}$. Invert for $u^rf^{(r)}(0)$
and use $u^{-r}\le u^{-R}$ for $r\le R$. The origin has exact value zero.
\end{proof}

For $0<\delta\le1$, write $v=\log(1/\delta)$ and $j,R,L$ as in
Theorem~\ref{thm:jet}. With $\tau=T/4$ choose
\begin{align*}
 H_c&=\max\{8,2R,T/2,256e^2\Lambda^2\tau,
  (\log4+\Lambda\tau+25j\log Q+v)/\log2\},\quad N_c=4\lceil H_c/4\rceil,\\
 A_c&=8^{N_c}(N_c/\tau)^R,\quad p_c=2^{-N_c/4-1}/N_c,\quad
 \kappa_c=p_c\delta^2/(4L^2A_c^2),\\
 K_c&=32+T+256e^2\Lambda^2\tau+2\Lambda\tau+50\log Q,\\
 C_c&=32+50\log Q+5K_c+17\log K_c+16|\log\tau|.
\end{align*}
For $\nu$ choose instead
\begin{align*}
 H_\infty&=\max\{8,R,(\log4+25j\log Q+R\log u_0^{-1}+v)/\log4\},\\
 N_\infty&=4\lceil H_\infty/4\rceil,\quad
 A_\infty=8^{N_\infty}u_0^{-R},\quad p_\infty=2^{-N_\infty/4+1}/N_\infty,\\
 \kappa_\infty&=p_\infty\delta^2/(4L^2A_\infty^2),\quad
 K_\infty=24+25\log Q+8\log u_0^{-1},\\
 C_\infty&=1+50\log Q+16\log u_0^{-1}+6K_\infty.
\end{align*}
The dependence of these certificates on $(J,\delta)$ is suppressed only in
these displayed definitions.

\begin{proposition}[Explicit separation, retained and used below]
\label{prop:separation}
For $d_J(\beta,\gamma)\ge\delta$, $0<\delta\le1$,
\begin{equation}
 D_\omega\ge\kappa_c\ge e^{-C_cj\log(ej)}\delta^{12},\qquad
 D_\nu\ge\kappa_\infty\ge e^{-C_\infty j}\delta^8.
 \label{eq:separation}
\end{equation}
\end{proposition}
\begin{proof}
For the compact law use spacing $u=\tau/N_c$ on existing cells
$(K,1,k)$ with $K=(N_c-8)/4$, each of mass $p_c$. Since $N_c\ge2R$,
$N_c/\tau\ge1$, and $(N+1)!\ge((N+1)/e)^{N+1}$,
\[
 A_cE_{N_c,u}\le2e^{\Lambda\tau}
 (8e\Lambda\sqrt{\tau/N_c})^{N_c}\,e\Lambda\tau/N_c
 \le2e^{\Lambda\tau}2^{-N_c}.
\]
Our choice makes $LA_cE\le\delta/2$. Equations~\eqref{eq:jet}--\eqref{eq:grid}
force a positive node of size at least $\delta/(2LA_c)$, giving $\kappa_c$.
Rounding yields $N_c\le K_cj+2v$. Put $a_0=2\log8+(\log2)/4<5$. Then
\[
 -\log\kappa_c=2v+\log8+50j\log Q+a_0N_c
                      +2R\log(N_c/\tau)+\log N_c.
\]
Use $R\le8j$ and $\log N_c\le\log(K_cj)+2v/(K_cj)$.
The coefficient of $v$ is at most $2+2a_0+34/K_c<12$; the remaining
terms are at most $C_cj\log(ej)$, including $j=1$.
For the long law, $8e^{\Lambda u_0}e\Lambda u_0<1/4$ and
$e\Lambda u_0<1$ give $A_\infty E\le2u_0^{-R}4^{-N_\infty}$.
Absorb this as before and use $N_\infty\le K_\infty j+v$.
Here
$-\log\kappa_\infty=\log2+\log N_\infty+2v+50j\log Q
+a_0N_\infty+2R\log u_0^{-1}\le C_\infty j+8v$.
\end{proof}

\subsection{Chronology and the complete working likelihood}
Fix $R_0<\infty$, $a,\sigma,\tau_g>0$, $U_0,T_0\ge0$, and $\rho\in(0,1]$.
At stage $i$ a bounded baseline segment of duration in
$[\tau_g,\tau_g+T_0]$ is chosen before the fresh exploration flag and probe.
On exploration a fresh cell is drawn from $\chi$; otherwise any duration at
most $T$ may be chosen. A baseline on the probe interval is fixed before
an independent Rademacher sign $S_i$, and $aS_i$ is added to it.
The complete force is bounded by $U_*=U_0+a$. The noise $\xi_i$ is a fresh
$N(0,\sigma^2)$ variable at the readout. All actions precede that noise;
there are no unrecorded observations. Policy kernels are parameter independent
functions of the observed history and may depend on the horizon.

If $m_i^\beta$ is the zero-state mean under the entire realized action record,
$b_i^\beta(z)=\ell e^{\mathsf T_iA_\beta}z$, and $\zeta_n$ is any probability
on the radius-$R_0$ ball, define
\begin{equation}
 L_n^{\zeta_n}(\beta)=\int\prod_{i\le n}
 \varphi_\sigma(Y_i-m_i^\beta-b_i^\beta(z))\,\zeta_n(dz),\quad
 \Pi_n^{\zeta_n}(d\beta)\propto L_n^{\zeta_n}(\beta)\Pi(d\beta).
 \label{eq:posterior}
\end{equation}
The true initial state may lie outside the support of $\zeta_n$; this is then
an exact working likelihood, not an assertion of a correctly specified
initial-state prior. Action kernels cancel in the chronological likelihood
ratio at the same history. They are never differentiated or conditioned on
as a final fixed design.
